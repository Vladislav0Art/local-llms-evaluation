package org.jsoup.helper;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;
import org.springframework.test.context.junit4.SpringRunner;

import java.net.MalformedURLException;
import java.net.URI;
import java.net.URL;
import java.util.HashMap;
import java.util.Map;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNotNull;
import static org.junit.Assert.assertNull;
import static org.mockito.Mockito.when;

@RunWith(SpringRunner.class)
public class UrlBuilderTest {

    @Mock
    private Connection.KeyVal kv;

    @Test
    public void [UrlBuilder_init][withInputUrl]Test() {
        URL u = new URL("http://example.com/path");
        UrlBuilder builder = new UrlBuilder(u);
        assertNotNull(builder.u);
        assertNull(builder.q);
    }

    @Test
    public void [UrlBuilder_build][withoutQueryAndRef][TestBasicNormalization] {
        URL u = new URL("http://example.com/path");
        UrlBuilder builder = new UrlBuilder(u);
        URL result = builder.build();
        assertNotNull(result);
        assertEquals("http://example.com/path", result.toString());
    }

    @Test
    public void [UrlBuilder_build][withQueryAndRef][TestBasicNormalization] {
        URL u = new URL("http://example.com/path?query=value#ref");
        UrlBuilder builder = new UrlBuilder(u);
        URL result = builder.build();
        assertNotNull(result);
        assertEquals("http://example.com/path?query=value#ref", result.toString());
    }

    @Test
    public void [UrlBuilder_build][withNonAsciiPathComponent][TestNormalization] {
        URL u = new URL("http://example.com/äthpath");
        UrlBuilder builder = new UrlBuilder(u);
        URL result = builder.build();
        assertNotNull(result);
        assertEquals("http://example.com/%C3%A4thpath", result.toString());
    }

    @Test
    public void [UrlBuilder_appendKeyVal][withConnectionKeyVal][TestQueryAppend] {
        Connection.KeyVal kv = new Connection.KeyVal();
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com/path"));
        builder.appendKeyVal(kv);
        assertNotNull(builder.q);
        assertEquals("key=value", builder.q.toString());
    }

    @Test
    public void [UrlBuilder_appendKeyVal][withMultipleConnectionKeyVals][TestQueryAppend] {
        Connection.KeyVal kv1 = new Connection.KeyVal();
        Connection.KeyVal kv2 = new Connection.KeyVal();
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com/path"));
        builder.appendKeyVal(kv1);
        builder.appendKeyVal(kv2);
        assertNotNull(builder.q);
        assertEquals("key=value&k2=v2", builder.q.toString());
    }

    @Test
    public void [UrlBuilder_decodePart][withEncodedString][TestDecode] {
        String encoded = "http%3A%2F%2Fexample.com";
        String result = UrlBuilder.decodePart(encoded);
        assertEquals("http://example.com", result);
    }

    @Test
    public void [UrlBuilder_build][ MalformedURLException][TestExceptionHandling] {
        URL u = new URL("invalid:uri");
        UrlBuilder builder = new UrlBuilder(u);
        try {
            builder.build();
            assert false;
        } catch (MalformedURLException e) {
            assertNotNull(e);
        }
    }

    @Test
    public void [UrlBuilder_build][URISyntaxException][TestExceptionHandling] {
        URL u = new URL("http://example.com/path?query=value#ref");
        UrlBuilder builder = new UrlBuilder(u);
        try {
            builder.build();
            assert false;
        } catch (URISyntaxException e) {
            assertNotNull(e);
        }
    }

}
