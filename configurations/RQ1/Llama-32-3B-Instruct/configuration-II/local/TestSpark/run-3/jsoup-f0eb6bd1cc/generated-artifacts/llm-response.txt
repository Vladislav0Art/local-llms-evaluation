package org.jsoup.helper;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

@RunWith(MockitoJUnitRunner.class)
public class UrlBuilderTest {

    @Mock
    private Connection.KeyVal kv;

    @Test
    public void build_url_with_absolute_path() {
        // arrange
        URL u = new URL("http://example.com/path?query=value#anchor");
        UrlBuilder builder = new UrlBuilder(u);

        // act
        builder.build();

        // assert
        Mockito.verify(builder.u).getHost();
        Mockito.verify(builder.q);
    }

    @Test
    public void build_url_with_relative_path() {
        // arrange
        URL u = new URL("http://example.com/path?query=value#anchor");
        UrlBuilder builder = new UrlBuilder(u);

        builder.u.setPath("");
        builder.build();

        // assert
        Mockito.verify(builder.u).getHost();
        Mockito.verify(builder.q);
    }

    @Test
    public void build_url_with_query_string() {
        // arrange
        URL u = new URL("http://example.com/path?query=value#anchor");
        UrlBuilder builder = new UrlBuilder(u);

        builder.build();

        // assert
        Mockito.verify(builder.u).getHost();
        Mockito.verify(builder.q);
    }

    @Test
    public void appendKeyVal_key_and_value() {
        // arrange
        Connection.KeyVal kv = new Connection.KeyVal("key", "value");
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com/path"));

        // act
        builder.appendKeyVal(kv);

        // assert
        Mockito.verify(builder.q).append(Mockito.anyString()).times(2);
    }

    @Test
    public void appendKeyVal_key_only() {
        // arrange
        Connection.KeyVal kv = new Connection.KeyVal("key", null);
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com/path"));

        // act
        builder.appendKeyVal(kv);

        // assert
        Mockito.verify(builder.q).append(Mockito.anyString());
    }

    @Test
    public void appendToAscii_no_non_ascii() {
        // arrange
        String s = "hello";
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com/path"));

        // act
        StringBuilder sb = new StringBuilder();
        builder.appendToAscii(s, false, sb);

        // assert
        Mockito.verify(builder.u).getPath();
    }

    @Test
    public void appendToAscii_non_ascii() {
        // arrange
        String s = "\u00e9";
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com/path"));

        // act
        StringBuilder sb = new StringBuilder();
        builder.appendToAscii(s, false, sb);

        // assert
        Mockito.verify(builder.u).getPath();
    }
}
