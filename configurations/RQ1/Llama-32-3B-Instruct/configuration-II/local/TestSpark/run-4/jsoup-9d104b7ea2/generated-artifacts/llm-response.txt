import org.jsoup.nodes.TextNode;
import org.junit.Test;
import static org.junit.Assert.*;
import java.io.ByteArrayOutputStream;
import java.io.PrintStream;
import java.util.Arrays;

public class TextNodeTest {

    @Test
    public void [TextNode__CreateFromEncoded](String encodedText) {
        final TextNode textNode = TextNode.createFromEncoded(encodedText);
        assertNotNull(textNode);
        assertEquals(encodedText, textNode.value());
    }

    @Test
    public void [TextNode__text](String text) {
        final TextNode textNode = new TextNode(text);
        assertNotNull(textNode);
        assertEquals(text, textNode.text());
    }

    @Test
    public void [TextNode__textSettingBlank](String blankText) {
        final TextNode textNode = new TextNode(blankText);
        textNode.text("");
        assertEquals(blankText, textNode.text());
    }

    @Test
    public void [TextNode__isBlank](String text) {
        assertTrue(TextNode.createFromEncoded("").isBlank());
        assertFalse(TextNode.createFromEncoded("  ").isBlank());
        assertFalse(TextNode.createFromEncoded("<").isBlank());
    }

    @Test
    public void [TextNode__splitText](int offset, String text) {
        final TextNode textNode = new TextNode(text);
        final TextNode tailNode1 = textNode.splitText(offset);
        assertNotNull(tailNode1);
        assertEquals(Arrays.toString(new String[] {"", text.substring(offset)}), Arrays.toString(new String[]{tailNode1.value(), text.substring(offset)}));
    }

    @Test
    public void [TextNode__outerHtmlHead](Document document) {
        final TextNode textNode = new TextNode("   ");
        final ByteArrayOutputStream outContent = new ByteArrayOutputStream();
        PrintStream printOut = new PrintStream(outContent);
        document.outputSettings().setPrettyPrint(true);
        printOut.println(textNode.outerHtml());
        assertEquals("\n", outContent.toString().trim());
    }

    @Test
    public void [TextNode__outerHtmlHeadEmpty](Document document) {
        final TextNode textNode = new TextNode("");
        final ByteArrayOutputStream outContent = new ByteArrayOutputStream();
        PrintStream printOut = new PrintStream(outContent);
        document.outputSettings().setPrettyPrint(true);
        printOut.println(textNode.outerHtml());
        assertEquals("", outContent.toString().trim());
    }

    @Test
    public void [TextNode__toString](String text) {
        final TextNode textNode = new TextNode(text);
        assertNotNull(textNode.toString());
        assertEquals(text, textNode.toString());
    }
}
