public class Tag {
    // existing fields and methods...
}

public class TagRegistry {
    private static final Map<String, Tag> tags = new HashMap<>();

    public static void register(Tag tag) {
        Validate.notNull(tag);
        tags.put(tag.tagName, tag);
    }

    public static List<Tag> getTags() {
        return new ArrayList<>(tags.values());
    }
}

public class TagUtils {

    public static String formatBlockToInline(String tagName) {
        if (isFormatAsInlineTag(tagName)) {
            return "inline-" + tagName;
        } else {
            return tagName;
        }
    }

    public static boolean isFormatAsInlineTag(String tagName) {
        for (String inlineTag : TagRegistry.formatAsInlineTags) {
            if (inlineTag.equals(tagName)) {
                return true;
            }
        }
        return false;
    }

    public static boolean isEmptyTag(String tagName) {
        for (String emptyTag : TagRegistry.emptyTags) {
            if (emptyTag.equals(tagName)) {
                return true;
            }
        }
        return false;
    }

    public static boolean isFormListedTag(String tagName) {
        for (String formListedTag : TagRegistry.formListedTags) {
            if (formListedTag.equals(tagName)) {
                return true;
            }
        }
        return false;
    }

    public static boolean isFormSubmitTag(String tagName) {
        for (String formSubmitTag : TagRegistry.formSubmitTags) {
            if (formSubmitTag.equals(tagName)) {
                return true;
            }
        }
        return false;
    }
}
