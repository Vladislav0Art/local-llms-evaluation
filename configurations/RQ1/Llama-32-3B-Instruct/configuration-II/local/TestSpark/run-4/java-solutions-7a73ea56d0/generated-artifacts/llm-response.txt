import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import java.util.ArrayList;
import java.util.List;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNotEquals;
import static org.junit.Assert.assertTrue;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class OnlineStockSpanTest {

    private OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
    @Mock
    List<Integer> mockList;

    public void testInitialize() {
        when(mockList.size()).thenReturn(0);
        when(mockList.isEmpty()).thenReturn(true);

        assertEquals(0, onlineStockSpan.list.size());
    }

    public void testNextSinglePriceIncrease() {
        int[] mockPrices = {10};
        when(mockList.get(0)).thenReturn(5);
        when(mockList.get(mockList.size())).thenReturn(15);

        int expectedOutput = 1;
        assertEquals(expectedOutput, onlineStockSpan.next(mockPrices[0]));
    }

    public void testNextMultiplePriceIncreases() {
        int[] mockPrices = {10, 20, 30};
        when(mockList.get(0)).thenReturn(5);
        when(mockList.get(mockList.size())).thenReturn(15);

        int expectedOutput = 1;
        assertEquals(expectedOutput, onlineStockSpan.next(mockPrices[0]));
        assertEquals(2, onlineStockSpan.next(mockPrices[1]));
        assertEquals(3, onlineStockSpan.next(mockPrices[2]));
    }

    public void testCalculateSpansSinglePrice() {
        int[] mockPrices = {10};
        when(mockList.get(0)).thenReturn(10);

        int[] expectedOutput = {1};
        assertArrayEquals(expectedOutput, onlineStockSpan.calculateSpans(mockPrices));
    }

    public void testCalculateSpansMultiplePrices() {
        int[] mockPrices = {10, 20, 30};

        when(mockList.get(0)).thenReturn(5);
        when(mockList.get(mockList.size())).thenReturn(15);

        int[] expectedOutput = {1, 2, 3};
        assertArrayEquals(expectedOutput, onlineStockSpan.calculateSpans(mockPrices));
    }

    public void testCalculateSpansEmptyList() {
        int[] mockPrices = {};
        when(mockList.size()).thenReturn(0);
        when(mockList.isEmpty()).thenReturn(true);

        int[] expectedOutput = new int[0];
        assertArrayEquals(expectedOutput, onlineStockSpan.calculateSpans(mockPrices));
    }

    public void testCalculateSpansDifferentPricing() {
        int[] mockPrices = {10, 20, 30, 40};

        when(mockList.get(0)).thenReturn(5);
        when(mockList.get(mockList.size())).thenReturn(15);

        int[] expectedOutput = {1, 2, 3, 4};
        assertArrayEquals(expectedOutput, onlineStockSpan.calculateSpans(mockPrices));
    }

}
