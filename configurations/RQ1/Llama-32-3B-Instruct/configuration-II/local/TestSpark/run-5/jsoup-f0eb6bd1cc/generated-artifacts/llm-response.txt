        Mockito.when(kv.key()).thenReturn("a");
        Mockito.when(kv.value()).thenReturn("value");

        UrlBuilder builder = new UrlBuilder(u);
        when(builder.q).thenReturn(null);

        builder.build();

        //Assert

    }
