    private static final String TEST_VALUE_1 = "test_value_1";

    private static final String TEST_KEY_2 = "test_key_2";
    private static final String TEST_VALUE_2 = "test_value_2";

    public static String decodePart(String encodedString) {
        return new String(encodedString.toCharArray(), 0, encodedString.length() - 1);
    }

    public static void appendKeyValWithEmptyQuery(String query) {
        Map<String, List<String>> paramsMap = new HashMap<>();
        if (query != null && !query.isEmpty()) {
            try {
                URI uri = new URI(query);
                String queryString = uri.getQuery();
                String[] params = queryString.split("&");
                for (String param : params) {
                    int idx = param.indexOf("=");
                    if (idx > 0) {
                        String key = decodePart(param.substring(0, idx));
                        String value = decodePart(param.substring(idx + 1));
                        paramsMap.computeIfAbsent(key, k -> new ArrayList<>()).add(value);
                    }
                }
            } catch (URISyntaxException e) {
                System.out.println("Invalid query string: " + e.getMessage());
            }
        }

        // Append key-value pairs to query
        if (!paramsMap.isEmpty()) {
            StringBuilder sb = new StringBuilder(query);
            if (sb.length() > 0 && sb.charAt(sb.length() - 1) == '&') {
                sb.append("&");
            }
            for (List<String> values : paramsMap.values()) {
                for (int i = 0; i < values.size(); i++) {
                    String value = values.get(i);
                    if (i == 0) {
                        sb.append(TEST_KEY_1).append("=").append(value);
                    } else if (i > 0 && i <= 1) {
                        sb.append("&").append(TEST_KEY_2).append("=").append(value);
                    }
                }

        }

    }
