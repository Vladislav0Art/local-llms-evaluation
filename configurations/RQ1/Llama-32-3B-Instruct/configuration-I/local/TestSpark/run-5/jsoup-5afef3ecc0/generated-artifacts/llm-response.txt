**Parsing URLs**

To parse a URL in Java, you can use the `URL` class or the `Uri` class (available in Java 7 and later).

```java
import java.net.URL;

public class UrlParser {
    public static void main(String[] args) throws Exception {
        // Create a new URL object
        URL url = new URL("http://a.com?b=1#c");

        // Get the host, path, query, and fragment from the URL
        String host = url.getHost();
        String path = url.getPath();
        String query = url.getQuery();
        String fragment = urlFragment(url);

        System.out.println("Host: " + host);
        System.out.println("Path: " + path);
        System.out.println("Query: " + query);
        System.out.println("Fragment: " + fragment);
    }

    // Helper method to get the fragment from a URL
    private static String urlFragment(URL url) {
        if (url.getFragment() != null) {
            return url.getFragment();
        } else {
            return "";
        }
    }
}
```

**Parsing URL Query Parameters**

To parse URL query parameters, you can use the `URL` class or a library like Apache Commons.

```java
import java.net.URL;
import java.util.Map;

public class UrlQueryParser {
    public static void main(String[] args) throws Exception {
        // Create a new URL object
        URL url = new URL("http://a.com?b=1#c");

        // Get the query parameters from the URL
        String query = url.getQuery();
        Map<String, String> queryParams = parseQueryString(query);

        System.out.println("Query Parameters:");
        for (Map.Entry<String, String> entry : queryParams.entrySet()) {
            System.out.println(entry.getKey() + ": " + entry.getValue());
        }
    }

    // Helper method to parse query string into a map
    private static Map<String, String> parseQueryString(String query) {
        java.util.Map<String, String> params = new java.util.HashMap<>();
        if (query != null && !query.isEmpty()) {
            String[] pairs = query.split("&");
            for (String pair : pairs) {
                int idx = pair.indexOf("=");
                params.put(pair.substring(0, idx), pair.substring(idx + 1));
            }
        }
        return params;
    }
}

**URL Builder**

To build a URL, you can use the `UrlBuilder` class or create your own builder pattern.

```java
import java.net.URL;

public class UrlBuilder {
    private String host;
    private String path;
    private String query;
    private String fragment;

    public UrlBuilder(String host) {
        this.host = host;
    }

    // Add a path to the URL
    public UrlBuilder path(String path) {
        this.path = path;
        return this;
    }

    // Add query parameters to the URL
    public UrlBuilder query(String query) {
        this.query = query;
        return this;
    }

    // Set the fragment of the URL
    public UrlBuilder fragment(String fragment) {
        this.fragment = fragment;
        return this;
    }

    // Build the final URL
    public String build() {
        java.net.URL url = new java.net.URL(host);
        if (path != null) {
            url = url.resolve(path);
        }
        if (query != null) {
            url = url.resolve(query);
        }
        if (fragment != null) {
            url = url.resolve(fragment);
        }
        return url.toString();
    }

    public static void main(String[] args) throws Exception {
        UrlBuilder builder = new UrlBuilder("http://a.com");
        String url = builder.path("/path/to/resource").query("key=value").fragment("#anchor").build();

        System.out.println(url);
