public class OnlineStockSpan {

    private Stack<Integer> indexStack = new Stack<>();

    public int next(int price) {
        if (indexStack.isEmpty()) {
            return 0;
        }

        while (!indexStack.isEmpty() && indexStack.peek() <= price) {
            indexStack.pop();
        }

        if (indexStack.isEmpty()) {
            indexStack.push(price);
            return price - indexStack.peek();
        } else {
            indexStack.push(price);
            return price - indexStack.peek() + 1;
        }
    }

    public int[] calculateSpans(int[] prices) {
        int n = prices.length;
        int[] result = new int[n];

        for (int i = 0; i < n; i++) {
            result[i] = next(prices[i]);
        }

        return result;
    }
}

public class OnlineStockSpan {

    private Stack<Integer> indexStack = new Stack<>();

    public int next(int price) {
        if (indexStack.isEmpty()) {
            return 0;
        }

        while (!indexStack.isEmpty() && indexStack.peek() <= price) {
            indexStack.pop();
        }

        if (indexStack.isEmpty()) {
            indexStack.push(price);
            return price - indexStack.peek();
        } else {
            indexStack.push(price);
            return price - indexStack.peek() + 1;
        }
    }

    public int[] calculateSpans(int[] prices) {
        int n = prices.length;
        int[] result = new int[n];

        for (int i = 0; i < n; i++) {
            result[i] = next(prices[i]);
        }

        return result;
    }
}

public class OnlineStockSpanTest {

    private OnlineStockSpan onlineStockSpan;

    public void setup() {
        onlineStockSpan = new OnlineStockSpan();
    }

    public void testNext_DivisiblePrice_ReturnsSpanOfAllPreviousPrices() {
        int[] prices = {1, 2, 3, 4, 5};
        for (int price : prices) {
            onlineStockSpan.next(price);
        }
        assertEquals(0, onlineStockSpan.next(prices[prices.length - 1]));
    }

    public void testNext_NonDivisiblePrice_ReturnsSpanFromLastDivisiblePrice() {
        int[] prices = {2, 4, 6, 8, 10};
        for (int price : prices) {
            onlineStockSpan.next(price);
        }
        assertEquals(1, onlineStockSpan.next(prices[prices.length - 1]));
    }

    public void testCalculateSpans_DivisiblePrices_ReturnsCorrectSpans() {
        int[] prices = {10, 20, 30};
        int[] result = onlineStockSpan.calculateSpans(prices);
        assertEquals(new int[]{1, 2, 3}, result);
    }

    public void testCalculateSpans_NonDivisiblePrices_ReturnsCorrectSpans() {
        int[] prices = {20, 10, 30};
        int[] result = onlineStockSpan.calculateSpans(prices);
        assertEquals(new int[]{2, 1, 3}, result);
    }

    public void testCalculateSpans_EmptyPricesArray_ReturnsCorrectSpans() {
        int[] prices = {};
        int[] result = onlineStockSpan.calculateSpans(prices);
        assertEquals(new int[]{0}, result);
    }

    public void testNext_EmptyListThrowsNullPointerException() {
        assertThrows(NullPointerException.class, () -> onlineStockSpan.next());
    }
}
