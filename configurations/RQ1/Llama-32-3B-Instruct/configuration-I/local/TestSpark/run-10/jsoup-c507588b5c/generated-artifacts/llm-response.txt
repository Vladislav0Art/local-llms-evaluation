public class TextNode {
    private String text;
    private Node parent;

    public TextNode(String text) {
        this.text = text;
    }

    // Method to check if the text node is blank
    public boolean isBlank() {
        return text.isEmpty();
    }

    // Method to split the text at a given offset
    public void splitText(int offset) {
        String textBeforeOffset = text.substring(0, offset);
        String textAfterOffset = text.substring(offset);

        this.text = textBeforeOffset;
        parent.getChildNodes().add(new TextNode(textAfterOffset));
    }

    // Method to get the outer HTML of the text node
    public void outerHtmlHead(Appendable accum, int offset, Appendable out) {
        if (text.isEmpty()) {
            return;
        }
        accum.append("<span>");
        accum.append(this.text);
        accum.append("</span>");
        out.append(accum.getContents());
    }

    // Method to get the unescaped text of the text node
    public String text() {
        return text;
    }
}
