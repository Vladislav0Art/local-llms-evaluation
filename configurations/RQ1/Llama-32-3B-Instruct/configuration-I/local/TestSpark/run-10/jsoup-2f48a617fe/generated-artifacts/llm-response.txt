        this.value = value;

    public String getValue() {

    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj)
            return true;
        if (obj == null || getClass() != obj.getClass())
            return false;

        LeafNode leafNode = (LeafNode) obj;

        return Objects.equals(name, leafNode.name);
    }
}

                '}';
