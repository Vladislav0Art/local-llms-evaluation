import org.jsoup.Connection;
import org.jsoup.helper.UrlBuilder;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNotNull;
import static org.junit.Assert.assertThrows;
import static org.junit.Assert.fail;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class UrlBuilderTest {

    @Mock
    private Connection.KeyVal kv;

    @Test
    public void buildUrl_EmptyUrl_ReturnsSameUrl() {
        when(kv.key()).thenReturn("");
        when(kv.value()).thenReturn("");

        UrlBuilder urlBuilder = new UrlBuilder(null);
        URL result = urlBuilder.build();

        assertEquals(urlBuilder.u, result);
    }

    @Test
    public void buildUrl_PunctuatedSchemeReturnsCorrectProtocol() {
        when(kv.key()).thenReturn("https:");
        when(kv.value()).thenReturn("/path");

        UrlBuilder urlBuilder = new UrlBuilder(null);
        URL result = urlBuilder.build();

        assertEquals("https", result.getProtocol());
    }

    @Test
    public void buildUrl_PunctuatedUserAndPortReturnsCorrectUserInfo() {
        when(kv.key()).thenReturn("user:pass");
        when(kv.value()).thenReturn("/path");

        UrlBuilder urlBuilder = new UrlBuilder(null);
        URL result = urlBuilder.build();

        assertEquals("user:pass", urlBuilder.u.getUserInfo());
    }

    @Test
    public void buildUrl_PunctuatedHostReturnsCorrectHostAfterPunyCode() {
        when(kv.key()).thenReturn("");
        when(kv.value()).thenReturn("example.com");

        UrlBuilder urlBuilder = new UrlBuilder(new URL("http://example.com"));
        URL result = urlBuilder.build();

        assertEquals("http", result.getProtocol());
    }

    @Test
    public void buildUrl_PunctuatedPathReturnsCorrectPath() {
        when(kv.key()).thenReturn("");
        when(kv.value()).thenReturn("/path");

        UrlBuilder urlBuilder = new UrlBuilder(new URL("http://example.com/path"));
        URL result = urlBuilder.build();

        assertEquals("/path", result.getPath());
    }

    @Test
    public void buildUrl_QuestionMarkPreserved() {
        when(kv.key()).thenReturn("");
        when(kv.value()).thenReturn("?query");

        UrlBuilder urlBuilder = new UrlBuilder(new URL("http://example.com"));
        URL result = urlBuilder.build();

        assertEquals("http", result.getProtocol());
        assert result.getQuery().startsWith("?query");
    }

    @Test
    public void buildUrl_FragmentPreserved() {
        when(kv.key()).thenReturn("");
        when(kv.value()).thenReturn("#fragment");

        UrlBuilder urlBuilder = new UrlBuilder(new URL("http://example.com"));
        URL result = urlBuilder.build();

        assertEquals("http", result.getProtocol());
        assert result.getRef().equals("#fragment");
    }

    @Test
    public void buildUrl_InvalidUrlThrowsMalformedURLException() {
        assertThrows(MalformedURLException.class, () -> new UrlBuilder(null));
    }

    @Test
    public void appendKeyVal_AppendsKeyValuePairsToQuery() {
        String query = "?key=value&foo=bar";

        UrlBuilder urlBuilder = new UrlBuilder(null);
        urlBuilder.appendKeyVal(new Connection.KeyVal("key", "value"));
        urlBuilder.appendKeyVal(new Connection.KeyVal("foo", "bar"));

        assertNotNull(urlBuilder.q);
        assertEquals(query, urlBuilder.q.toString());
    }

    @Test
    public void decodePart_DecodesAsciiEncodedString() {
        when(kv.key()).thenReturn("");
        when(kv.value()).thenReturn("%20");

        String result = UrlBuilder.decodePart("%20");

        assertEquals(" ", result);
    }

    @Test
    public void appendToAscii_AppendsKeyAndValueAsUrlEncoded() {
        StringBuilder sb = StringUtil.borrowBuilder();
        String key = "key";
        String value = "value";

        UrlBuilder.appendToAscii(key, true, sb);
        UrlBuilder.appendToAscii(value, false, sb);

        assertNotNull(sb);
        assertEquals(String.format("%s=%s", URLEncoder.encode(key, UTF_8.name()), URLEncoder.encode(value, UTF_8.name())), sb.toString());
    }
}
