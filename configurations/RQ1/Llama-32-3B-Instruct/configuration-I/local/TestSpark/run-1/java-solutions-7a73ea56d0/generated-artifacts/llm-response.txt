public class Solution {

    public static int[] findDiagonalOrder(int[][] nums) {
        if (nums.length == 0) return new int[0];
        int m = nums.length, n = nums[0].length;
        Map<Integer, List<Integer>> map = new HashMap<>();
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                map.computeIfAbsent(i + j, k -> new ArrayList<>()).add(nums[i][j]);
            }
        }

        List<Integer> result = new ArrayList<>();
        for (int i : map.keySet()) {
            if ((i & 1) == 0) {
                result.addAll(map.get(i));
            } else {
                Collections.reverse(map.get(i));
                result.addAll(map.get(i));
            }
        }
        return result.stream().mapToInt(x -> x).toArray();
    }

    public static void main(String[] args) {
        int[][] nums = {{1, 2}, {3, 4}};
        System.out.println(java.util.Arrays.toString(findDiagonalOrder(nums)));
    }
}

public class Solution {
    public int[] findDiagonalOrder(int[][] nums) {
        List<List<Integer>> ans = new ArrayList<>();
        
        if (nums.length == 0) return new int[0];
        
        for (int i = 0; i < nums.length; i++) {
            for (int j = 0; j < nums[i].length; j++) {
                int key = i + j;
                if (!ans.get(key).isEmpty()) {
                    if ((key & 1) == 0) {
                        ans.get(key).add(nums[i][j]);
                    } else {
                        ans.get(key).add(0, nums[i][j]); // add at the beginning
                    }
                } else {
                    List<Integer> list = new ArrayList<>();
                    list.add(nums[i][j]);
                    ans.add(list);
                }
            }
        }
        
        int[] res = new int[ans.size()];
        for (int i = 0; i < ans.size(); i++) {
            for (int j = 0; j < ans.get(i).size(); j++) {
                if ((i + j) % 2 == 0) { // even index
                    res[i] += ans.get(i).get(j);
                } else { // odd index
                    res[i] += ans.get(i).get(ans.get(i).size() - j - 1); 
                }
            }
        }
        
        return res;
    }

    public static void main(String[] args) {
        int[][] nums = {{1, 2}, {3, 4}};
        System.out.println(Arrays.toString(findDiagonalOrder(nums)));
    }
}
