import static org.junit.Assert.*;

public class UrlBuilderTest {

    private Connection.KeyVal kv;

    public void [UrlBuilder_build][Scenario1Test]() throws Exception {
        UrlBuilder builder = new UrlBuilder(new URL("https://example.com/path?query#fragment"));
        URI uri = Mockito.mock(URI.class);
        Mockito.when(uri.toASCIIString()).thenReturn("https%3A//example.com/path");
        URL normalizedUrl = builder.build();
        assertNotNull(normalizedUrl);
    }

    public void [UrlBuilder_build][Scenario2Test]() throws Exception {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com:8080/path?query#fragment"));
        URI uri = Mockito.mock(URI.class);
        Mockito.when(uri.toASCIIString()).thenReturn("http%3A//example.com%3A8080/path");
        URL normalizedUrl = builder.build();
        assertNotNull(normalizedUrl);
    }

    public void [UrlBuilder_build][Scenario3Test]() throws Exception {
        UrlBuilder builder = new UrlBuilder(new URL("https://example.com:8080/path?query=äöü#fragment"));
        URI uri = Mockito.mock(URI.class);
        Mockito.when(uri.toASCIIString()).thenReturn("https%3A//example.com%3A8080/path");
        URL normalizedUrl = builder.build();
        assertNotNull(normalizedUrl);
    }

    public void [UrlBuilder_appendKeyVal][Scenario1Test]() throws Exception {

        assertEquals("key=value", builder.q.toString());
    }

    public void [UrlBuilder_appendKeyVal][Scenario2Test]() throws Exception {

        builder.appendKeyVal(kv);

        assertEquals("key=value", builder.q.toString());
    }

        assertNotNull(builder.u);
    }

    public void [UrlBuilder_appendToAscii][Scenario2Test]() throws Exception {
        UrlBuilder builder = new UrlBuilder(new URL(""));
        StringBuilder sb = Mockito.mock(StringBuilder.class);
        Mockito.when(sb.append((Object) String)).thenReturn(sb);
        builder.appendToAscii("äöü", true, sb);
        assertNotNull(builder.u);
    }

        UrlBuilder builder = new UrlBuilder(new URL(""));
        String decoded = UrlBuilder.decodePart("");
        assertEquals("", decoded);
    }

    }
