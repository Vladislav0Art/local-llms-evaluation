package org.jsoup.nodes;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import java.io.IOException;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertFalse;
import static org.junit.Assert.assertNotNull;
import static org.junit.Assert.assertNull;
import static org.junit.Assert.assertTrue;
import static org.mockito.ArgumentMatchers.any;
import static org.mockito.Mockito.doThrow;

@RunWith(MockitoJUnitRunner.class)
public class CommentTest {

    @Mock
    private Appendable accum;

    @Test
    public void createComment_withData_setsValue() {
        String data = "Hello";
        Comment comment = new Comment(data);
        assertEquals(data, comment.getData());
    }

    @Test
    public void createComment_withoutData_isNull() {
        Comment comment = new Comment("");
        assertNull(comment.getData());
    }

    @Test
    public void outerHtmlHead_withPrettyPrint_setsDepthAndAccum() throws IOException {
        doThrow(new IOException()).when(accum).append(any());
        Comment comment = new Comment("Hello");
        comment.outerHtmlHead(accum, 0, null);
        assertEquals(1, accum.depth());
        assertEquals(comment, accum);
    }

    @Test
    public void outerHtmlTail_doesNothing() {
        doThrow(new IOException()).when(accum).append(any());
        Comment comment = new Comment("Hello");
        comment.outerHtmlHead(accum, 0, null);
        assertNull(accum.appendLine(""));
    }

    @Test
    public void toString_returnsOuterHtml() {
        String data = "Hello";
        Comment comment = new Comment(data);
        assertEquals("#comment" + "\n<!--" + data + "-->", comment.toString());
    }

    @Test
    public void clone_returnsSelf() {
        Comment comment = new Comment("Hello");
        Comment clonedComment = comment.clone();
        assertTrue(comment == clonedComment);
    }

    @Test
    public void isXmlDeclaration_isTrue_forValidData() {
        String data = "Hello World!";
        Comment comment = new Comment(data);
        assertTrue(comment.isXmlDeclaration());
    }

    @Test
    public void isXmlDeclaration_isFalse_forInvalidData() {
        String data = "Hello";
        Comment comment = new Comment(data);
        assertFalse(comment.isXmlDeclaration());
    }

    @Test
    public void asXmlDeclaration_returnsNull_forInvalidData() {
        String data = "";
        Comment comment = new Comment(data);
        assertNull(comment.asXmlDeclaration());
    }

    @Test
    public void asXmlDeclaration_returnsXmlDeclarationForValidData() throws IOException {
        String fragment = "<Hello World!";
        Document doc = Parser.htmlParser().settings(ParseSettings.preserveCase).parseInput(fragment, null);
        Comment comment = new Comment(doc.body().child(0).tagName());
        XmlDeclaration decl = comment.asXmlDeclaration();
        assertNotNull(decl);
        assertTrue(decl.tag().equals("Hello"));
    }

    @Test
    public void asXmlDeclaration_throwsIOException_onParseError() throws IOException {
        doThrow(new IOException()).when(Parser.htmlParser())
                .settings(ParseSettings.preserveCase)
                .parseInput(any(), any());
        Comment comment = new Comment("");
        assertNull(comment.asXmlDeclaration());
    }
}
