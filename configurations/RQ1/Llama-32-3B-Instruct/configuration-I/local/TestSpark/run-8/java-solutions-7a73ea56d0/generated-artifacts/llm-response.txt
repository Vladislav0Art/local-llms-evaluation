package leetcode.medium;

import java.util.Stack;
import java.util.List;

public class OnlineStockSpan {
    private Stack<Integer> indexStack = new Stack<>();
    private List<Integer> list = new ArrayList<>();

    public int next(int price) {
        while (!indexStack.isEmpty() && list.get(indexStack.peek()) <= price) {
            indexStack.pop();
        }
        if (indexStack.isEmpty()) {
            list.add(price);
            indexStack.push(list.size() - 1);
        } else {
            list.add(price - list.get(indexStack.peek()));
            indexStack.push(list.size() - 1);
        }
        return indexStack.size();
    }

    public int[] calculateSpans(int[] prices) {
        for (int price : prices) {
            next(price);
        }
        return new int[list.size()];
    }
}

        onlineStockSpan.next(20);

        assertEquals(4, onlineStockSpan.list.get(3));
    }

    @Test
    public void testNext_LessThanPrice_ReturnsCorrectCount() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        onlineStockSpan.next(50);
        onlineStockSpan.next(100);
        onlineStockSpan.next(90);

        assertEquals(2, onlineStockSpan.list.get(2));
    }

    @Test
    public void testNext_ZeroPrice_ReturnsCorrectCount() {

public class OnlineStockSpan {

    private Stack<Integer> indexStack = new Stack<>();
    private List<Integer> list = new ArrayList<>();

    public int next(int price) {
        while (!indexStack.isEmpty() && list.get(indexStack.peek()) <= price) {
            indexStack.pop();
        }
        if (indexStack.isEmpty()) {
            list.add(price);
            indexStack.push(list.size() - 1);
        } else {
            list.add(price - list.get(indexStack.peek()));
            indexStack.push(list.size() - 1);
        }
        return indexStack.size();
    }

    public int[] calculateSpans(int[] prices) {
        for (int price : prices) {
            this.next(price);
        }
        List<Integer> result = new ArrayList<>();
        for (int i = 0; i  list.size(); i++) {
            result.add(list.get(i));
        }
        return result.stream().mapToInt(Integer::intValue).toArray();
    }

    public void testNext() {
        int[] prices = {1,2,3,4,5};
        assertEquals(1, next(1));
        assertEquals(2, next(2));
        assertEquals(2, next(3));
        assertEquals(3, next(4));
        assertEquals(4, next(5));
    }

    public void testCalculateSpans() {
        int[] prices = {1,2,3,4,5};
        int[] result = calculateSpans(prices);
        assertArrayEquals(new int[]{0, 1, 2, 3, 4}, result);
    }
}
