import org.jsoup.helper.UrlBuilder;
import org.junit.Test;
import static org.junit.Assert.*;
import org.junit.runner.RunWith;
import org.junit.runners.JUnit4;

@RunWith(JUnit4.class)
public class UrlBuilderTest {

    @Test
    public void urlBuilderConstructWithoutQueryAndFragmentTests() {
        URL u = new URL("http://example.com");
        UrlBuilder builder = new UrlBuilder(u);
        assertNull(builder.q);
    }

    @Test
    public void urlBuilderConstructWithQueryAndFragmentTests() {
        URL u = new URL("http://example.com?query=hello&fragment=world");
        UrlBuilder builder = new UrlBuilder(u);
        assertNotNull(builder.q);
    }

    @Test
    public void urlBuilderBuildUrlWithoutQueryAndFragmentTests() {
        URL u = new URL("http://example.com/path/to/resource");
        UrlBuilder builder = new UrlBuilder(u);
        assertEquals("http://example.com/ path%2Fto%2Fresource", builder.u.toString());
    }

    @Test
    public void urlBuilderBuildUrlWithQueryAndFragmentTests() {
        URL u = new URL("http://example.com/path/to/resource?query=hello&fragment=world");
        UrlBuilder builder = new UrlBuilder(u);
        assertEquals("http://example.com/ path%2Fto%2Fresource?query=hello&fragment=world", builder.u.toString());
    }

    @Test
    public void urlBuilderAppendKeyValTests() {
        Connection.KeyVal kv = new Connection.KeyVal("key1", "value1");
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com/path/to/resource"));
        builder.appendKeyVal(kv);
        assertEquals("?key1=value1", builder.q.toString());
    }

    @Test
    public void urlBuilderAppendMultipleKeyValsTests() {
        Connection.KeyVal kv1 = new Connection.KeyVal("key1", "value1");
        Connection.KeyVal kv2 = new Connection.KeyVal("key2", "value2");
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com/path/to/resource"));
        builder.appendKeyVal(kv1);
        builder.appendKeyVal(kv2);
        assertEquals("?key1=value1&key2=value2", builder.q.toString());
    }

    @Test
    public void urlBuilderDecodePartTests() {
        String encoded = "https%3A%2F%2Fexample.com";
        String decoded = UrlBuilder.decodePart(encoded);
        assertEquals("https://example.com", decoded);
    }

    @Test
    public void urlBuilderAppendToAsciiWithSpaceAsPlusTests() {
        StringBuilder sb = new StringBuilder();
        UrlBuilder.appendToAscii("Hello World", true, sb);
        assertEquals("+%20Hello+World", sb.toString());
    }

    @Test
    public void urlBuilderAppendToAsciiWithoutSpaceAsPlusTests() {
        StringBuilder sb = new StringBuilder();
        UrlBuilder.appendToAscii("Hello World", false, sb);
        assertEquals("%3DHallo%20World", sb.toString());
    }
}
