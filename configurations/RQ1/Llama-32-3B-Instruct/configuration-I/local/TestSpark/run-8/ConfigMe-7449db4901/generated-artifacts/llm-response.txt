import org.junit.Test;
import static org.junit.Assert.*;

public class CommentsConfigurationTest {

    @Test
    public void testComments() {
        // Arrange
        String[] path = {"path", "line"};
        String comment = "This is a comment";
        String[] lines = {"line1"};

        CommentsConfiguration comments = new CommentsConfiguration();

        // Act
        comments.addPath(path, Arrays.asList(comment));
        Map<String, List<String>> map = comments.getCommentsMap();
        map.put("path", lines);

        // Assert
        assertEquals(comment, map.get("path").get(0));
    }
}

class CommentsConfiguration {
    public Map<String, List<String>> commentsMap;

    public void addPath(String[] path, List<String> commentList) {
        if (commentsMap == null) {
            commentsMap = new HashMap<>();
        }
        commentsMap.put(path[0], commentList);
    }

    public Map<String, List<String>> getCommentsMap() {
        return commentsMap;
    }
}
