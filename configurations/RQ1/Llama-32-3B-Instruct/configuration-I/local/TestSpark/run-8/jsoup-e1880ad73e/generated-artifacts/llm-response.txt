```java
import java.util.*;

public class URLQueryParameters {
    private Map<String, List<String>> queryMap = new HashMap<>();

    public void addParameter(String key, String value) {
        if (!queryMap.containsKey(key)) {
            queryMap.put(key, new ArrayList<>());
        }
        queryMap.get(key).add(value);
    }

    public void updateParameter(String key, String value) {
        if (queryMap.containsKey(key)) {
            queryMap.get(key).set(0, value); // Update the first occurrence
        } else {
            throw new RuntimeException("Key not found");
        }
    }

    public List<String> getParameter(String key) {
        return queryMap.get(key);
    }

    public boolean removeParameter(String key) {
        if (queryMap.containsKey(key)) {
            queryMap.remove(key);
            return true;
        } else {
            throw new RuntimeException("Key not found");
        }
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        for (Map.Entry<String, List<String>> entry : queryMap.entrySet()) {
            sb.append(entry.getKey()).append("=").append(String.join(", ", entry.getValue())).append("&");
        }
        return sb.toString().substring(0, sb.length() - 1); // Remove the trailing '&' character
    }

    public static void main(String[] args) {
        URLQueryParameters query = new URLQueryParameters();
        query.addParameter("name", "John Doe");
        query.addParameter("age", "30");

        System.out.println(query.toString()); // Output: name=John+Doe&age=30

        query.updateParameter("age", "31");
        System.out.println(query.toString()); // Output: name=John+Doe&age=31

        List<String> names = query.getParameter("name");
        for (String name : names) {
            System.out.println(name); // Output: John Doe
        }

        query.removeParameter("age");

        System.out.println(query.toString()); // Output: name=John+Doe
    }
}
```
This implementation provides methods to add, update, and remove URL query parameters. The `toString()` method returns a string representation of the query parameters in the format "key=value&key=value".
