    public void [TextNode_SplitText]Test() {
        String text = "Hello";
        TextNode textNode = new TextNode(text);

        assertNotNull(tailNode);
        assertEquals(4, tailNode.text().length());
    }

    public void [TextNode_CreateFromEncoded]Test() {
        String encodedText = "&lt;Hello, World!&gt;";

        assertEquals("Hello, World!", textNode.text());
    }

    @Test
    public void [TextNode_isBlank]Test() {
        String text = "   ";
        TextNode textNode = new TextNode(text);
        assertTrue(textNode.isBlank());
    }

    public void [TextNode_textSetters]Test() {

        TextNode textNode = new TextNode(text);
        textNode.text("World");

    public void [TextNode_getWholeText]Test() throws IOException {

        assertNotNull(accum.toString());
    }

    public void [TextNode_Clone]Test() throws IOException {
        String text = "Hello";
        TextNode textNode = new TextNode(text);

import org.jsoup.Jsoup;

import org.junit.Test;
import static org.junit.Assert.*;

public class CloneTest extends TestCase {

    @Test
    public void testClone() {
        Document document = Jsoup.parse("<html><body>Hello World!</body></html>");
        Object clonedDocument = document.clone();
        assertTrue(clonedDocument instanceof Document);
        assertEquals(document, (Document) clonedDocument);
    }

    @Test
    public void testNotCloningComments() {
        Document document = Jsoup.parse("<html><body><!-- This is a comment -->Hello World!</body></html>");
        Object clonedDocument = document.clone();
        assertFalse(clonedDocument.contains("#"));
    }
}
