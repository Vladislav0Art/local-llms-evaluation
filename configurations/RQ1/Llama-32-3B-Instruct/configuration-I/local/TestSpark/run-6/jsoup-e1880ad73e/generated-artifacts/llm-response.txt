import org.jsoup.Connection;
import org.jsoup.internal.StringUtil;
import org.junit.Test;
import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertThrows;

public class UrlBuilderTest {

    @Test
    public void newUrlBuilder() {
        URL u = new URL("http://example.com/path");
        UrlBuilder builder = new UrlBuilder(u);
        assertEquals(u, builder.u);
        assertNull(builder.q);
    }

    @Test
    public void buildUrl() {
        URL u = new URL("http://example.com/path?key=value&space+as+plus#anchor");
        UrlBuilder builder = new UrlBuilder(u);
        builder.build();
        URI uri = u.getURI();
        assertEquals("http://example.com/path", uri.getHost());
        assertEquals("value", uri.getQuery().getValues().iterator().next());
    }

    @Test
    public void buildUrlWithNonAsciiPath() {
        URL u = new URL("http://example.com/äoth?key=value&space+as+plus#anchor");
        UrlBuilder builder = new UrlBuilder(u);
        builder.build();
        URI uri = u.getURI();
        assertEquals("äoth", uri.getPath());
    }

    @Test
    public void appendKeyVal() {
        Connection.KeyVal kv = new Connection.KeyVal("key", "value");
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com/path"));
        builder.appendKeyVal(kv);
        assertEquals("key=value", StringUtil.toString(builder.q));
    }

    @Test
    public void appendKeyValWithMultipleQuery() {
        Connection.KeyVal kv1 = new Connection.KeyVal("key1", "value1");
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com/path"));
        builder.appendKeyVal(kv1);
        Connection.KeyVal kv2 = new Connection.KeyVal("key2", "value2");
        builder.appendKeyVal(kv2);
        assertEquals("key1=value1&key2=value2", StringUtil.toString(builder.q));
    }

    @Test
    public void appendToAscii() {
        StringBuilder sb = new StringBuilder();
        String s = "äoth";
        UrlBuilder.builder.appendToAscii(s, true, sb);
        assertEquals("%e0th", sb.toString());
    }

    @Test
    public void decodePart() {
        String encoded = "%E0th";
        assertEquals("äoth", UrlBuilder.decodePart(encoded));
    }

    @Test(expected = RuntimeException.class)
    public void buildUrlWithInvalidEncoding() {
        URL u = new URL("http://example.com/path?key=value");
        UrlBuilder builder = new UrlBuilder(u);
        builder.build();
    }
}

import org.junit.Test;
import static org.junit.Assert.assertEquals;

public class URL {
    private String scheme;
    private String userInfo;
    private String host;
    private int port;
    private String path;
    private String query;
    private String fragment;

    public URL(String url) {
        String[] parts = url.split(":");
        if (parts.length == 2 && parts[0].equals("http")) {
            this.scheme = "http";
            String[] hostAndPort = parts[1].split("/");
            this.host = hostAndPort[0];
            this.port = Integer.parseInt(hostAndPort[1]);
            parts = url.split("/", 4);
            if (parts.length == 3) {
                this.path = "/" + parts[2] + "?" + parts[3];
            } else {
                this.path = "/";
            }
        } else {
            throw new IllegalArgumentException("Invalid URL");
        }
    }

    public String getScheme() {
        return scheme;
    }

    public String getUserInfo() {
        return userInfo;
    }

    public String getHost() {
        return host;
    }

    public int getPort() {
        return port;
    }

    public String getPath() {
        return path;
    }

    public String getQuery() {
        return query;
    }

    public String getFragment() {
        return fragment;
    }
}

public class URLBuilder {
    private URL url;

    public static URLBuilder builder() {
        return new URLBuilder();
    }

    public void setScheme(String scheme) {
        this.url = new URL("http://" + scheme);
    }

    public void setUserInfo(String userInfo) {
        this.url = new URL("http://" + userInfo);
    }

    public void setHost(int host, int port) {
        url = new URL("http://localhost:" + host + "/" + url.getPath());
    }

    public void appendToAscii(String s, boolean isQuery, StringBuilder sb) {
        if (!isQuery) {
            url = new URL(url.getScheme() + "://" + url.getHost() + "/" + url.getPath());
        }
        sb.append(url.getScheme());
        sb.append("://");
        sb.append(url.getHost());
        sb.append('/');
        sb.append(url.getPath());
    }

    public String getURI() {
        return url.getScheme() + "://" + url.getHost() + "/" + url.getPath() + "?" + url.getQuery() + "#" + url.getFragment();
    }
}

public class ConnectionKeyVal {
    private String key;
    private String value;

    public static ConnectionKeyVal builder() {
        return new ConnectionKeyVal();
    }

    public void appendTo(String s, boolean isQuery, StringBuilder sb) {
        URLBuilder builder = UrlBuilder.builder();
        builder.setScheme(s);
        if (isQuery) {
            sb.append(builder.getURI());
        } else {
            sb.append(builder.getScheme());
            sb.append("://");
            sb.append(builder.getHost());
            sb.append('/');
            sb.append(builder.getPath());
        }
    }

    public String getKey() {
        return key;
    }

    public void setKey(String key) {
        this.key = key;
    }

    public String getValue() {
        return value;
    }

    public void setValue(String value) {
        this.value = value;
    }
}

public class StringUtil {
    public static String toString(StringBuilder builder) {
        return builder.toString();
    }
}
