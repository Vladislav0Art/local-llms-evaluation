package leetcode.medium;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import java.util.ArrayList;
import java.util.List;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertThrows;
import static org.mockito.ArgumentMatchers.any;
import static org.mockito.ArgumentMatchers.anyInt;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class OnlineStockSpanTest {

    @Mock
    private Stack<Integer> indexStack;

    @Mock
    private List<Integer> list;

    private OnlineStockSpan onlineStockSpan;

    @BeforeAll
    public void setUp() {
        onlineStockSpan = new OnlineStockSpan();
        onlineStockSpan.list = list;
    }

    [NextTest][EmptyList]Test
    public void testCalculateSpans_EmptyList_ReturnsZeroArray() {
        int[] expectedOutput = new int[0];
        int[] actualOutput = onlineStockSpan.calculateSpans(new int[0]);
        assertArrayEquals(expectedOutput, actualOutput);
    }

    [NextTest][SmallInput]Test
    public void testCalculateSpans_SmallInput_ReturnsCorrectSpans() {
        onlineStockSpan.list = new ArrayList<>();
        int[] prices = {1, 2, 3};
        int[] expectedOutput = {1, 1, 2};
        int[] actualOutput = onlineStockSpan.calculateSpans(prices);
        assertArrayEquals(expectedOutput, actualOutput);
    }

    [NextTest][LargeInput]Test
    public void testCalculateSpans_LargeInput_ReturnsCorrectSpans() {
        onlineStockSpan.list = new ArrayList<>();
        int[] prices = {1, 2, 3, 4, 5};
        int[] expectedOutput = {1, 1, 2, 2, 3};
        int[] actualOutput = onlineStockSpan.calculateSpans(prices);
        assertArrayEquals(expectedOutput, actualOutput);
    }

    [NextTest][PriceIncrease]Test
    public void testCalculateSpans_PriceIncreases_ReturnsCorrectSpans() {
        onlineStockSpan.list = new ArrayList<>();
        List<Integer> list = new ArrayList<>();
        list.add(1);
        list.add(2);
        when(indexStack.pop()).thenReturn(list.size());
        int[] prices = {3, 4};
        int[] expectedOutput = {2, 1};
        int[] actualOutput = onlineStockSpan.calculateSpans(prices);
        assertArrayEquals(expectedOutput, actualOutput);
    }

    [NextTest][PriceDecrease]Test
    public void testCalculateSpans_PriceDecreases_ReturnsCorrectSpans() {
        onlineStockSpan.list = new ArrayList<>();
        List<Integer> list = new ArrayList<>();
        list.add(1);
        when(indexStack.pop()).thenReturn(list.size());
        int[] prices = {2, 3};
        int[] expectedOutput = {2, 1};
        int[] actualOutput = onlineStockSpan.calculateSpans(prices);
        assertArrayEquals(expectedOutput, actualOutput);
    }

    }
