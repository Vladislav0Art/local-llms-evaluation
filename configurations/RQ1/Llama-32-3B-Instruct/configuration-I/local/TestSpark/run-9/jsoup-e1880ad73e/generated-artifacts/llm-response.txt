```java
package org.jsoup.helper;

public class UrlBuilder {
    private URL url;
    private StringBuilder q = new StringBuilder();

    public UrlBuilder(URL url) {
        this.url = url;
    }

    public void appendKeyVal(Connection.KeyVal kv) throws UnsupportedEncodingException {
        if (q.length() > 0 && !q.charAt(q.length() - 1).equals('&')) {
            q.append("&");
        }
        q.append(kv.getKey());
        String value = URLEncoder.encode(kv.getValue(), IDN.getCharset());
        q.append('=');
        q.append(value);

        // Add a space as + in the query string
        if (kv.getValue().contains(" ")) {
            value = URLEncoder.encode("+" + kv.getValue(), IDN.getCharset());
            q.append(' ');
            q.append('=').append(value);
        }
    }

    public void appendToAscii(String str, boolean addQuery, StringBuilder sb) {
        // Convert non-ASCII to ASCII
        for (int i = 0; i < str.length(); i++) {
            char c = str.charAt(i);
            if (c > 127) {
                str = str.substring(0, i) + '%' + Integer.toString(c, 16).toUpperCase() + str.substring(i + 1);
            }
        }

        // Add to the StringBuilder
        sb.append(str);
    }

    public URL build() throws MalformedURLException {
        String scheme = url.getProtocol();
        if (scheme.equals("http")) {
            scheme += "s";
        }
        return new URL(scheme, url.getHost(), url.getPort(), q.toString(), url.getPath());
    }

    public static String decodePart(String part) throws UnsupportedEncodingException {
        String decodedPart = new String(part.getBytes(IDN.getCharset()), IDN.getCharset());
        // Remove spaces
        int spaceIndex = decodedPart.indexOf(' ');
        if (spaceIndex != -1) {
            return decodedPart.substring(0, spaceIndex);
        } else {
            return decodedPart;
        }
    }

    public static String encodePart(String part) throws UnsupportedEncodingException {
        return URLEncoder.encode(part, IDN.getCharset());
    }

    public void appendToUrl(String str, boolean addQuery) throws MalformedURLException {
        if (addQuery && q.length() > 0) {
            if (!q.toString().contains("?")) {
                q.append("?");
            } else {
                q.append("&");
            }
        }
        String scheme = url.getProtocol();
        if (scheme.equals("http")) {
            scheme += "s";
        }
        URL newUrl = new URL(scheme, url.getHost(), url.getPort(), q.toString() + str, url.getPath());
        url = newUrl;
    }
}

public class Connection {
    public static class KeyVal {
        private String key;
        private String value;

        public KeyVal(String key, String value) {
            this.key = key;
            this.value = value;
        }

        // getters and setters
    }
}
```

```java
package org.jsoup.helper;

import org.junit.Test;
import static org.junit.Assert.assertEquals;
import java.net.URL;
import java.io.UnsupportedEncodingException;

public class UrlBuilderTest {

    public static class MockUrlBuilder extends UrlBuilder {
        @Override
        public void appendKeyVal(Connection.KeyVal kv) throws UnsupportedEncodingException {
            // just return the StringBuilder, mock it later in a method call
            q.append(kv.getKey() + "=" + kv.getValue());
        }
        
        @Override
        public void appendToAscii(String str, boolean encode, StringBuilder sb) {
            if (encode) {
                for (char c : str.toCharArray()) {
                    char asciiVal = (char) c;
                    if (asciiVal >= 32 && asciiVal = 126) {
                        sb.append('%');
                        String hex = Integer.toHexString(asciiVal);
                        while(hex.length() == 1)
                            hex = "0" + hex;
                        sb.append(hex.toLowerCase());
                    } else
                        sb.append(c);
                }
            } else {
                sb.append(str);
            }
        }

        public StringBuilder getQ() {
            return q;
        }

        private StringBuilder q = new StringBuilder();
    }

    @Test
    public void testUrlBuilder() throws UnsupportedEncodingException {
        MockUrlBuilder builder = new MockUrlBuilder();
        URL inputUrl = new URL("http://test.com/path");
        Connection.KeyVal kv1 = new Connection.KeyVal("key1", "value1");
        Connection.KeyVal kv2 = new Connection.KeyVal("key2", "value2");
        
        // Test appending key-value pairs
        builder.appendKeyVal(kv1);
        builder.appendKeyVal(kv2);

        assertEquals("key1=value1&key2=value2", builder.getQ().toString());

        // Test appendToAscii with encoding and without encoding
        String str = "Hello World!";
        StringBuilder sb = new StringBuilder();
        builder.appendToAscii(str, true, sb);
        assertEquals("%48%65%6c%6c%6f %57%6f%72%6c%64%21", sb.toString());

        // Test appendToAscii without encoding
        str = "Hello World!";
        sb.setLength(0);  // Reset StringBuilder
        builder.appendToAscii(str, false, sb);
        assertEquals("Hello World!", sb.toString());
    }
}
```
