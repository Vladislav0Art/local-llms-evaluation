public class TextNodeTest {

    @Mock
    private LeafNode parentNode;

    @Mock
    private Element parentElement;

    public static String normaliseWhitespace(String text) {
        // implementation remains the same as in the source code
        return text;
    }

    public static String stripLeadingWhitespace(String text) {
        // implementation remains the same as in the source code
        return text;
    }

    public static boolean lastCharIsWhitespace(StringBuilder sb) {
        // implementation remains the same as in the source code
        return sb.length() > 0 && Character.isWhitespace(sb.charAt(sb.length() - 1));
    }
}

public class TextNode {

    private String text;

    public TextNode(String rawText) {
        this.text = normaliseWhitespace(rawText);
    }

    public String nodeName() {
        return "#text";
    }

    public String text() {
        return text;
    }

    public void setText(String newText) {
        if (newText != null && !newText.isEmpty()) {
            this.text = stripLeadingWhitespace(newText);
        } else {
            this.text = "";
        }
    }

    public String getWholeText() {
        return text;
    }

    public boolean isBlank() {
        return text.trim().isEmpty();
    }

    public TextNode splitText(int offset) throws IllegalArgumentException {
        if (offset < 0 || offset > text.length()) {
            throw new IllegalArgumentException("Offset must be between 0 and the length of the text");
        }
        String textBefore = text.substring(0, offset);
        String textAfter = text.substring(offset);
        return new TextNode(textBefore + "\n" + textAfter);
    }

    public String outerHtmlHead() {
        // implementation remains the same as in the source code
        return "";
    }

    public String outerHtmlTail() {
        // implementation remains the same as in the source code
        return "";
    }

    @Override
    public String toString() {
        return "<span>" + text + "</span>";
    }
}
