public class UrlBuilder {
    private URI url;

    public static URL build(String url) throws MalformedURLException {
        return new URL(new URI(url));
    }

    public static String normalizeQuery(String query) {
        if (query == null || query.isEmpty()) {
            return "";
        }
        Map<String, String> params = new HashMap<>();
        for (String param : query.split("&")) {
            int idx = param.indexOf("=");
            if (idx > 0) {
                String key = param.substring(0, idx);
                String value = param.substring(idx + 1);
                params.put(key, value);
            }
        }
        StringBuilder sb = new StringBuilder();
        for (Map.Entry<String, String> entry : params.entrySet()) {
            if (sb.length() > 0) {
                sb.append("&");
            }
            sb.append(entry.getKey()).append("=").append(entry.getValue());
        }
        return sb.toString();
    }

    public static String normalizeRef(String ref) {
        while (ref.startsWith(" ")) {
            ref = ref.substring(1);
        }
        while (ref.endsWith(" ")) {
            ref = ref.substring(0, ref.length() - 1);
        }
        return ref;
    }

    private UrlBuilder(String url) throws MalformedURLException {
        this.url = new URI(url);
    }

    public static URL buildUrl(String url) throws MalformedURLException {
        return new URL(url);
    }

    public static String decodePart(String encoded) {
        // Simple implementation, may not cover all cases
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < encoded.length(); ) {
            if (encoded.charAt(i) == '%') {
                try {
                    int c = Integer.parseInt(encoded.substring(i + 1, i + 3), 16);
                    sb.append((char) c);
                    i += 3;
                } catch (NumberFormatException e) {
                    // Handle invalid encoding
                }
            } else {
                sb.append(encoded.charAt(i));
                i++;
            }
        }
        return sb.toString();
    }

}

class URL extends java.net.URL {
    public URL(URI uri) throws MalformedURLException {
        super(uri);
    }

    @Override
    public String toString() {
        return uri.toString();
    }
}
