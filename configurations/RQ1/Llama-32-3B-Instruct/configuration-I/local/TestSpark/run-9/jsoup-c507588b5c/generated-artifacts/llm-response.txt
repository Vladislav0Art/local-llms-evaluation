        String end = "</span>" + tail;
        accum.append(end);
    }

    public TextNode clone() {
        return new TextNode(value);
    }
}

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.LeafNode;

import org.jsoup.nodes.TextNode;

public class GeneratedTest {

    public void testTextNode() {
        TestTextNode testTextNode = new TestTextNode();
        
        // assertEquals("test", testTextNode.text());
        
        Document document = new Document();
        Element element = (Element)document.body().firstChild();
        TextNode textNode = (TextNode)element.firstChild();
        int offset = 3;
        TextNode tail = textNode.splitText(offset);
        assert tail.value().length() == 2; // Should be "te"
        
        // assertTrue(testTextNode.isBlank());
        
        ByteArrayOutputStream out = new ByteArrayOutputStream(StandardCharsets.UTF_8.name().getBytes().length);
        Appendable accum = out;
        TextNode textNode2 = new TextNode("test");
        textNode2.outerHtmlHead(accum, 0, new Document.OutputSettings());
        String outerHtml = out.toString();
        assert outerHtml.equals("<span>test"); // Should be "<span>test"
        
        // assertEquals("", tail.outerHtmlTail(new Document.OutputSettings()));
        String tailOuterHtml = tail.outerHtmlTail(out, 0, new Document.OutputSettings());
