import java.util.*;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

public class Parser {

    private List<Element> elements;

    public Parser() {
        this.elements = new ArrayList<>();
    }

    // Public method to parse HTML
    public void parse(String html) {
        Pattern pattern = Pattern.compile("<(\\w+)[^>]*>");
        Matcher matcher = pattern.matcher(html);

        while (matcher.find()) {
            String tag = matcher.group(1);
            Tag currentTag = Tag.getTag(tag);

            if (currentTag == null) {
                throw new IllegalArgumentException("Invalid tag name: " + tag);
            }

            // Create a new element
            Element element = Element.createElement(currentTag.tagName, currentTag.isBlock ? new Tag(currentTag.tagName).isBlock : false,
                    currentTag.formatAsBlock ? new Tag(currentTag.tagName).formatAsBlock : false,
                    currentTag.preserveWhitespace ? new Tag(currentTag.tagName).preserveWhitespace : false,
                    currentTag.formList ? new Tag(currentTag.tagName).formList : false,
                    currentTag.formSubmit ? new Tag(currentTag.tagName).formSubmit : false);

            // Add the element to the list
            elements.add(element);
        }
    }

    // Public method to print parsed HTML
    public void printElements() {
        for (Element element : elements) {
            System.out.println(element.name);
            for (Tag tag : element.tags) {
                System.out.print(tag.tagName + ":");
                if (tag.formatAsBlock) {
                    System.out.print(" true");
                } else {
                    System.out.print(" false");
                }
                System.out.print(",");
                System.out.print(tag.isBlock ? "true" : "false");
                System.out.print(",");
                System.out.print(tag.preserveWhitespace ? "true" : "false");
                System.out.print(",");
                System.out.print(tag.formList ? "true" : "false");
                System.out.print(",");
                System.out.print(tag.formSubmit ? "true" : "false");
                System.out.println();
            }
        }
    }

}

        Parser parser = new Parser();
        String html = "<html><body><p>Hello World!</p></body></html>";
