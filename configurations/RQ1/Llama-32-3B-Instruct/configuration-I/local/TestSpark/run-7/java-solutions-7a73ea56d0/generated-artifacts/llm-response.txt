package leetcode.medium;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import java.util.ArrayList;
import java.util.List;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertTrue;

@RunWith(MockitoJUnitRunner.class)
public class OnlineStockSpanTest {

    private OnlineStockSpan stockSpan = new OnlineStockSpan();

    @Mock
    private List<Integer> list;

    @Mock
    private Stack<Integer> indexStack;

    @Test
    public void next_IntPrice_ReturnsCount() {
        // Given
        int price = 10;
        Mockito.when(list.get(Mockito.anyInt())).thenReturn(price);

        // When
        int actualCount = stockSpan.next(price);
        Mockito.verifyNoMoreInteractions(list, indexStack);

        // Then
        assertEquals(1, actualCount);
    }

    @Test
    public void next_SamePrice_ReturnsSameCount() {
        // Given
        int price = 10;
        stockSpan.list.add(price);
        Mockito.when(list.get(Mockito.anyInt())).thenReturn(price);

        // When
        int actualCount = stockSpan.next(price);
        Mockito.verifyNoMoreInteractions(list, indexStack);

        // Then
        assertEquals(1, actualCount);
    }

    @Test
    public void next_DifferentPrice_ReturnsDifferentCount() {
        // Given
        int price1 = 10;
        int price2 = 20;
        stockSpan.list.add(price1);
        stockSpan.list.add(price2);

        Mockito.when(list.get(Mockito.anyInt())).thenReturn(price1).thenReturn(price2);

        // When
        int actualCount = stockSpan.next(price2);
        Mockito.verifyNoMoreInteractions(list, indexStack);

        // Then
        assertEquals(2, actualCount);
    }

    @Test
    public void calculateSpans_EmptyPricesArray_ReturnsEmptyArray() {
        // Given
        int[] prices = new int[0];

        // When
        int[] actualSpans = stockSpan.calculateSpans(prices);

        // Then
        assertTrue(actualSpans.length == 0);
    }

    @Test
    public void calculateSpans_SingleElementArray_ReturnsCorrectSpan() {
        // Given
        int[] prices = {10};
        int expectedSpan = 1;

        // When
        int[] actualSpans = stockSpan.calculateSpans(prices);

        // Then
        assertEquals(expectedSpan, actualSpans[0]);
    }

    @Test
    public void calculateSpans_MultipleElementArray_ReturnsCorrectSpans() {
        // Given
        int[] prices = {10, 20, 15};
        int[] expectedSpans = {1, 2, 2};

        stockSpan.list.add(10);
        stockSpan.list.add(20);

        Mockito.when(indexStack.peek()).thenReturn(0);
        Mockito.when(indexStack.pop()).thenReturn(0);

        // When
        int[] actualSpans = stockSpan.calculateSpans(prices);

        // Then
        for (int i = 0; i < expectedSpans.length; i++) {
            assertEquals(expectedSpans[i], actualSpans[i]);
        }
    }

}
