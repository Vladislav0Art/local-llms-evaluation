package org.jsoup.nodes;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;
import org.junit.runners.JUnit4;

import static org.junit.Assert.*;
import static org.mockito.Mockito.when;

@RunWith(JUnit4.class)
public class TextNodeTest {

    @Mock
    private Node parentNode;

    @Mock
    private Appendable accum;

    @Mock
    private Document.OutputSettings out;

    @Before
    public void setup() {
        MockitoAnnotations.initMocks(this);
    }

    public String getWholeText() {
        return "Hello World";
    }

    public String text() {
        return "Hello World";
    }

    public TextNode clone() {
        return this;
    }

    public String outerHtml() {
        return "Hello World";
    }

    @Test
    public void createTextNodeFromRawText_GetsProperlyCreated() {
        when(parentNode).thenReturn(this);
        String text = "Hello World";
        when(coreValue()).thenReturn(text);

        TextNode textNode = new TextNode(text);
        assertNotNull(textNode);
    }

    @Test
    public void nodeName_GetsCorrectValue() {
        assertEquals("#text", nodeName());
    }

    @Test
    public void getWholeText_ReturnsTextContent() {
        when(coreValue()).thenReturn("Hello World");
        String expected = "Hello World";
        String actual = text();
        assertEquals(expected, actual);
    }

    @Test
    public void isBlank_GetsCorrectValue() {
        assertTrue(isBlank());
    }

    @Test
    public void splitText_SplitsTextAtOffset() {
        when(coreValue()).thenReturn("Hello");
        when(parentNode).thenReturn(this);

        int offset = 5;
        TextNode tailNode = splitText(offset);
        assertNotNull(tailNode);
        assertEquals(5, tailNode.getWholeText().length());
    }

    @Test
    public void outerHtmlHead_SplitsTextIntoTwoNodes() {
        when(parentNode).thenReturn(this);
        when(out.prettyPrint()).thenReturn(true);

        StringBuilder accum = new StringBuilder();
        int depth = 0;
        Document.OutputSettings out = this.out;

        outerHtmlHead(accum, depth, out);
        assertEquals("Hello\r\nWorld", accum.toString());
    }

    @Test
    public void text_SetTextAndReturnsThis() {
        TextNode textNode = new TextNode(getWholeText());

        when(parentNode).thenReturn(this);

        String newText = "Hello";
        textNode.text(newText);
        assertEquals(newText, text());
    }
}
