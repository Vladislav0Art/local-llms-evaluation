import org.jsoup.nodes.Document;

public class TextNodeTest {

    @Test
    public void createTextNode_EmptyText() {
        TextNode node = new TextNode("");
        assert node.text().isEmpty();
    }

    @Test
    public void createTextNode_TrailingWhitespace() {
        TextNode node = new TextNode("   ");
        assertEquals("   ", node.text());
    }

    @Test
    public void createTextNode_LedgingWhitespace() {
        TextNode node = new TextNode("   \n");
        assertEquals("\n", node.text());
    }

    @Test
    public void textEmpty() {
        TextNode node = new TextNode("");
        assertTrue(node.isBlank());
    }

    @Test
    public void textNotBlank() {
        TextNode node = new TextNode("Hello");
        assertFalse(node.isBlank());
    }

    @Test
    public void createTextNode splitTextOffsetInMiddle() {
        String text = "Hello World";
        TextNode node = new TextNode(text);
        TextNode tail = node.splitText(6);
        assertEquals("World", tail.text());
        assertEquals("Hello ", node.text());
    }

    @Test
    public void createTextNode splitTextOffsetAtStart() {
        String text = "Hello World";
        TextNode node = new TextNode(text);
        TextNode tail = node.splitText(0);
        assertEquals("", tail.text());
        assertEquals("Hello", node.text());
    }

    @Test
    public void createTextNode splitTextOffsetAtEnd() {
        String text = "Hello World";
        TextNode node = new TextNode(text);
        TextNode head = node.splitText(text.length());
        assertEquals("", head.text());
        assertEquals("World ", node.text());
    }

    @Test
    public void getWholeText_Empty() {
        TextNode node = new TextNode("");
        assertEquals("", node.getWholeText());
    }

    @Test
    public void getWholeText_NonEmpty() {
        String text = "Hello World";
        TextNode node = new TextNode(text);
        assertEquals("Hello World", node.getWholeText());
    }
}

import org.junit.Test;
import static org.junit.Assert.*;

public class NodeTest {

    @Test
    public void test() {
        // Arrange
        String text = "Hello World";

        // Act
        Node node = new Node(text);
        assertEquals("Hello World", Entities.unescape(node.text()), 0, new StringBuilder());

        // Assert
        assertTrue(lastCharIsWhitespace(new StringBuilder()));
    }

    private boolean lastCharIsWhitespace(StringBuilder sb) {
        if (sb.length() > 0 && Character.isWhitespace(sb.charAt(sb.length() - 1))) {
            return true;
        }
        return false;
    }
}

class Node {
    private String text;

    public Node(String text) {
        this.text = text;
    }

    public String text() {
        return text;
    }
}

public class Entities {

    public static String unescape(Node node) {
        // implementation of unescape
        return node.text();
    }
}
