public class Connection {
    public static class KeyVal {
        private String key;
        private String value;

        public KeyVal(String key, String value) {
            this.key = key;
            this.value = value;
        }

        @Override
        public String toString() {
            return key + "=" + value;
        }
    }
}

public class UrlBuilder {

    private URL u;
    private java.util.Map<String, String> q;

    public UrlBuilder(URL u) {
        this.u = u;
        this.q = new java.util.HashMap<>();
    }

    public void appendKeyVal(Connection.KeyVal kv) {
        if (q == null) {
            q = new java.util.HashMap<>();
        }
        q.put(kv.key, kv.value);
    }

    private static String decodePart(String encoded) throws java.io.UnsupportedEncodingException {
        return new java.net.URLDecoder().decode(encoded);
    }

    public URL getURL() {
        StringBuilder sb = new StringBuilder(u.getProtocol());
        sb.append("://");
        sb.append(decodePart(u.getHost()));
        if (u.getPort() != -1) {
            sb.append(":").append(u.getPort());
        }
        sb.append(u.getPath());

        java.util.Map<String, String> query = u.getQuery();
        for (String key : query.keySet()) {
            sb.append("?");
            sb.append(decodePart(key)).append("=").append(decodePart(query.get(key)));
        }

        if (u.getRef() != null) {
            sb.append("#").append(decodePart(u.getRef()));
        }
        return new URL(sb.toString());
    }

    public String getQuery() {
        StringBuilder sb = new StringBuilder();
        for (java.util.Map.Entry<String, String> entry : q.entrySet()) {
            if (sb.length() > 0) {
                sb.append("&");
            }
            sb.append(entry.getKey()).append("=").append(decodePart(entry.getValue()));
        }
        return sb.toString();
    }

    public void setURL(URL u) {
        this.u = u;
    }

    public static UrlBuilder fromInputUrl(URL inputUrl) {
        UrlBuilder builder = new UrlBuilder(inputUrl);
        // remove the following line
        // builder.appendKeyVal(new Connection.KeyVal("key", "value"));
        return builder;
    }
}
