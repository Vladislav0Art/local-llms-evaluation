public class TextNodeTest {

    private Document document;
    private TextNode textNode;

    @Before
    public void setup() {
        document = new Document();
        textNode = new TextNode("Hello World");
        textNode.parentNode = document;
    }

    public static String stripLeadingWhitespace(String text) {
        return StringUtil.stripLeadingWhitespace(text);
    }

    public static String normalizeWhitespace(String text) {
        return StringUtil.normalizeWhitespace(text);
    }

    public static String escapeEntities(String text, Document.OutputSettings outputSettings) {
        return Entities.escape(text, outputSettings, false);
    }

    @Test
    public void createTextNode_CreateTextNode() {
        TextNode createdTextNode = new TextNode("Hello World");
        assertNotNull(createdTextNode);
    }

    @Test
    public void createTextNodecreateFromEncoded/CreateTextNodeFromEncoded() {
        String encodedText = "&lt;Hello World&gt;";
        TextNode createdTextNode = TextNode.createFromEncoded(encodedText);
        assertNotNull(createdTextNode);
    }

    @Test
    public void text_NodeText() {
        assertEquals("Hello World", textNode.text());
    }

    @Test
    public void isBlank_EmptyText_ReturnsTrue() {
        TextNode emptyTextNode = new TextNode("");
        assertTrue(emptyTextNode.isBlank());
    }

    @Test
    public void isBlank_BlankText_ReturnsFalse() {
        TextNode blankTextNode = new TextNode("   ");
        assertFalse(blankTextNode.isBlank());
    }

    @Test
    public void splitText_SplitTextAtOffset() {
        String text = "Hello World";
        int offset = 5;
        TextNode tailNode = textNode.splitText(offset);
        assertEquals(text.substring(0, offset), textNode.getWholeText());
        assertNotNull(tailNode);
        assertEquals("World", tailNode.text());
    }

    @Test
    public void outerHtml_IndentationPreserved() {
        Appendable appendable = new StringBuilder();
        Document.OutputSettings outputSettings = new Document.OutputSettings(true, false);
        textNode.outerHtmlHead(appendable, 0, outputSettings);
        assertEquals("<p>Hello World</p>", escapeEntities(appendable.toString(), outputSettings));
    }

    @Test
    public void isBlank_CoercionWorks() {
        TextNode textNode = new TextNode("   ");
        assertTrue(textNode.isBlank());
    }

    @Test
    public void normalizeWhitespace() {
        assertEquals("Hello World", normalizeWhitespace("Hello World"));
    }

    @Test
    public void stripLeadingWhitespace() {
        assertEquals("World", stripLeadingWhitespace("   Hello World "));
    }

    @Test
    public void clone_CloneTextNode() {
        TextNode clonedTextNode = textNode.clone();
        assertNotNull(clonedTextNode);
    }
}
