package ch.jalu.configme.configurationdata;

import org.junit.Test;
import static org.junit.Assert.*;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Arrays;

public class CommentsConfigurationTest {

    @Test
    public void newCommentsConfigurationInstanceCreated() {
        CommentsConfiguration instance = new CommentsConfiguration();
        assertNotNull(instance);
        assertTrue(instance.comments instanceof HashMap);
    }

    @Test
    public void commentsConfigurationInstanceCreatedFromMap() {
        Map<String, List<String>> map = new HashMap<>();
        map.put("path", Arrays.asList("line1", "line2"));
        CommentsConfiguration instance = new CommentsConfiguration(map);
        assertNotNull(instance);
        assertEquals(map, instance.comments);
    }

    @Test
    public void setCommentSetsCommentsForPath() {
        CommentsConfiguration instance = new CommentsConfiguration();
        instance.setComment("path", "line1");
        assertTrue(instance.comments.containsKey("path"));
        assertEquals(Arrays.asList("line1"), ((List<String>) instance.comments.get("path")));
    }

    @Test
    public void setCommentSetsEmptyLineWithoutMarkerForNewline() {
        CommentsConfiguration instance = new CommentsConfiguration();
        instance.setComment("path", "\n");
        assertTrue(instance.comments.containsKey("path"));
        assertEquals(Arrays.asList(), ((List<String>) instance.comments.get("path")));
    }

    @Test
    public void setCommentOverwritesExistingComments() {
        CommentsConfiguration instance = new CommentsConfiguration();
        instance.setComment("path", "line1");
        instance.setComment("path", "line2");
        assertTrue(instance.comments.containsKey("path"));
        assertEquals(Arrays.asList("line2"), ((List<String>) instance.comments.get("path")));
    }

    @Test
    public void getAllCommentsReturnsReadOnlyView() {
        CommentsConfiguration instance = new CommentsConfiguration();
        Map<String, List<String>> map = instance.getAllComments();
        assertTrue(map instanceof java.util.Map);
        assertFalse(java.util.Collections.unmodifiableMap(map).containsKey("new"));
    }
}
