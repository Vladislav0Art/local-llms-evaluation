import java.net.URL;
import java.util.ArrayList;
import java.util.List;

public class Main {

    public static void main(String[] args) {
        String url = "http://example.com/path?key=value";
        URL parsedUrl = parseUrl(url);
        System.out.println(parsedUrl.getQuery());
    }

    public static URL parseUrl(String url) {
        int queryIndex = url.indexOf('?');
        if (queryIndex == -1) {
            return new URL(url);
        }
        String path = url.substring(0, queryIndex);
        String query = url.substring(queryIndex + 1);
        List<String> keyValuePairs = new ArrayList<>();
        while (query.length() > 0) {
            int equalsIndex = query.indexOf('=');
            if (equalsIndex != -1) {
                keyValuePairs.add(query.substring(0, equalsIndex));
                query = query.substring(equalsIndex + 1);
            } else {
                break;
            }
        }
        return new URL("http://example.com/" + path + "?" + String.join("&", keyValuePairs));
    }
}

import java.net.URL;
import java.util.Arrays;

public class Main {

    public static void main(String[] args) {
        URL url = new URL("http://example.com/path?key=value");
        System.out.println(Arrays.toString(url.getQuery().split("&")));
    }
}

class Connection {
    public String key;
    public String value;

    public Connection(String key, String value) {
        this.key = key;
        this.value = value;
    }

    @Override
    public String toString() {
        return "key=" + key + "&value=" + value;
    }
}

class UrlBuilder {
    private String path;
    private List<String> query;

    public UrlBuilder(String path) {
        this.path = path;
    }

    public void addParameter(String key, String value) {
        if (query == null) {
            query = new ArrayList<>();
        }
        query.add(key + "=" + value);
    }

    @Override
    public String toString() {
        StringBuilder builder = new StringBuilder(path);
        for (String parameter : query) {
            builder.append("?").append(parameter);
        }
        return builder.toString();
    }
}

class KeyVal {
    private String key;
    private String value;

    public KeyVal(String key, String value) {
        this.key = key;
        this.value = value;
    }

    @Override
    public String toString() {
        return "key=" + key + "&value=" + value;
    }
}

class AsciiUrlBuilder extends UrlBuilder {
    private boolean appendAscii = true;

    public void append(String value) {
        if (appendAscii) {
            append(value);
        } else {
            append(value, false);
        }
    }

    @Override
    public String toString() {
        StringBuilder builder = new StringBuilder(getPath());
        for (String parameter : getQuery()) {
            builder.append("?").append(parameter);
        }
        return builder.toString();
    }
}

class UrlUtils {

    public static String urlencode(String str) {
        StringBuilder sb = new StringBuilder(str.length * 2);
        for (char c : str.toCharArray()) {
            if ((c < 0x20) || (c > 0xD7FF)) {
                sb.append("%").append(Integer.toHexString(c));
            } else if (c >= 0xE000 && c <= 0xFDCF) {
                sb.append("%").append("e2".equals(String.valueOf(c)) ? "e2" : "ef");
                sb.append(Integer.toHexString((int)c - 0xE0));
            } else if (c >= 0xFDE0 && c <= 0xFFFD) {
                sb.append("%").append("f8".equals(String.valueOf(c)) ? "f8" : "fa");
                sb.append(Integer.toHexString((int)c - 0xFE));
            } else {
                sb.append(c);
            }
        }
        return sb.toString();
    }

    public static String urlDecode(String str) {
        StringBuilder sb = new StringBuilder(str.length());
        for (int i = 0; i < str.length(); i += 2) {
            int c = Integer.parseInt(str.substring(i, i + 2), 16);
            if ((c >= 0x20) && (c <= 0xD7FF)) {
                sb.append((char)c);
            } else if ((c >= 0xE000) && (c <= 0xFDCF)) {
                sb.append((char)(0xE0 | (c - 0xE0)));
            } else if ((c >= 0xFDE0) && (c <= 0xFFFD)) {
                sb.append((char)(0xFE | (c - 0xFE)));
            }
        }
        return sb.toString();
    }

    public static String unquote(String str) {
        return urlDecode(urlEncode(str));
    }

    public static String quote(String str) {
        return urlEncode(str);
    }

    public static String urlEncode(String str) {
        StringBuilder sb = new StringBuilder(str.length * 2);
        for (int i = 0; i < str.length(); ) {
            char c = str.charAt(i++);
            if ((c >= 'a') && (c <= 'z') || (c >= 'A') && (c <= 'Z') ||
                    (c == '_') || (c == '-') || (c == '.') || (c == '~')) {
                sb.append(c);
            } else {
                sb.append('%').append(Integer.toHexString(c).toUpperCase()).append("2");
                i++;
            }
        }
        return sb.toString();
    }

    public static String urlDecode(String str) {
        StringBuilder sb = new StringBuilder(str.length());
        for (int i = 0; i < str.length(); ) {
            char c = str.charAt(i++);
            if ((c == '%') && (i + 2 < str.length())) {
                int codePoint = Integer.parseInt(str.substring(i, i + 3), 16);
                sb.append((char)codePoint);
                i += 3;
            } else {
                sb.append(c);
            }
        }
        return sb.toString();
    }

    public static boolean isAscii(char c) {
        return (c >= 'a') && (c <= 'z') || (c >= 'A') && (c <= 'Z')
                || (c == '_') || (c == '-') || (c == '.') || (c == '~');
    }
}
