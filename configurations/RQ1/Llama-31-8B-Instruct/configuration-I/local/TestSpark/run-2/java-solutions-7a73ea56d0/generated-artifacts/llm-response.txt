import org.junit.runner.RunWith;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertTrue;
import static org.mockito.Mockito.*;

public class OnlineStockSpanTest {

  public void next_EmptyList_ReturnsZero() {

  public void next_SingleElementList_ReturnsOne() {

  }

    stockSpan.next(30);
    int result = stockSpan.next(40);
    assertEquals(1, result);
  }

  @Test
  public void next_DecreasingList_ReturnsCount() {
    OnlineStockSpan stockSpan = new OnlineStockSpan();
    stockSpan.next(40);

    stockSpan.next(10);
    int result = stockSpan.next(5);
    assertEquals(4, result);
  }

    stockSpan.next(10);
    stockSpan.next(20);
    stockSpan.next(20);
    stockSpan.next(30);
    int result = stockSpan.next(30);
    assertEquals(1, result);
  }

  @Test
  public void calculateSpans_EmptyArray_ReturnsEmptyArray() {
