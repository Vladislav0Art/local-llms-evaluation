package org.jsoup.helper;

import org.jsoup.Connection;
import org.jsoup.internal.StringUtil;
import javax.annotation.Nullable;
import java.io.UnsupportedEncodingException;
import java.net.IDN;
import java.net.MalformedURLException;
import java.net.URI;
import java.net.URISyntaxException;
import java.net.URL;
import java.net.URLDecoder;
import java.net.URLEncoder;
import static org.jsoup.helper.DataUtil.UTF_8;

public class UrlBuilderTest {
    public void testUrlBuilder() throws UnsupportedEncodingException {
        UrlBuilder urlBuilder = new UrlBuilder(new URI("http://example.com/path?a=b"));
        
        urlBuilder.appendKeyVal(new java.util.HashMap<String, String>() {{
            put("key1", "value1");
            put("key2", "value2");
        }});
        
        assertEquals("http://example.com/path?key1=value1&key2=value2", urlBuilder.build().toString());
        
        urlBuilder = new UrlBuilder(new URI("http://example.com/path?a=b#anchor"));
        
        assertEquals("http://example.com/path#anchor", urlBuilder.build().toString());
    }

    public void testUrlBuilderNormalize() throws UnsupportedEncodingException {
        URL u = new URI("http://example.com/path?query=abc");
        UrlBuilder urlBuilder = new UrlBuilder(u);
        
        String normUrl = urlBuilder.build();
        assertEquals("http://example.com/path?query=abc", normUrl.toString());
    }

    public void testUrlBuilderAppendKeyVal() throws UnsupportedEncodingException {
        URL u = new URI("http://example.com/path?a=b&c=d");
        UrlBuilder urlBuilder = new UrlBuilder(u);
        
        urlBuilder.appendKeyVal(new java.util.HashMap<String, String>() {{
            put("key3", "value3");
        }});
        
        assertEquals("http://example.com/path?a=b&key3=value3", urlBuilder.build().toString());
    }

    public void testUrlBuilderAppendKeyValNonAscii() throws UnsupportedEncodingException {
        URL u = new URI("http://example.com/path?query=abc&c=d");
        UrlBuilder urlBuilder = new UrlBuilder(u);
        
        urlBuilder.appendKeyVal(new java.util.HashMap<String, String>() {{
            put("key3", "value3");
            put("key4", "value4");
        }});
        
        assertEquals("http://example.com/path?query=abc&c=d?key3=value3&key4=value4", urlBuilder.build().toString());
    }

    public void testUrlBuilderAppendKeyValEscape() throws UnsupportedEncodingException {
        URL u = new URI("http://example.com/path?query=abc&c=d");
        UrlBuilder urlBuilder = new UrlBuilder(u);
        
        urlBuilder.appendKeyVal(new java.util.HashMap<String, String>() {{
            put("key4", "value4");
        }});
        
        assertEquals("http://example.com/path?query=abc&c=d?key4=value4", urlBuilder.build().toString());
    }
}
