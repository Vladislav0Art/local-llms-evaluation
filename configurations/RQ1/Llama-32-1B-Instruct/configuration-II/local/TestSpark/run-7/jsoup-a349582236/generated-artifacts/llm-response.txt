class Tag {
    private String name;

    public Tag(String name) {
        this.name = name;
    }

    public void isBlock(boolean block) {
        this.block = block;
    }

    public void setEmpty(boolean empty) {
        if (!empty) {
            Method method = new ConstructorTag(name, 0).constructor;
            register(method);
        }
    }

    public boolean getEmpty() {
        return false;
    }

    @Override
    public String toString() {
        StringBuilder builder = new StringBuilder();
        builder.append("Tag {");
        builder.append("\"name\":\"").append(name).append("\",");
        builder.append("\"block\":").append(block).append(",");
        builder.append("\"empty\":").append(empty).append("}");
        return builder.toString();
    }
}

class ConstructorTag {
    private String name;
    private int constructor;

    public ConstructorTag(String name, int constructor) {
        this.name = name;
        this.constructor = constructor;
    }

    @Override
    public String toString() {
        StringBuilder builder = new StringBuilder();
        builder.append("ConstructorTag {");
        builder.append("\"name\":\"").append(name).append("\",");
        builder.append("\"constructor\":").append(constructor);
        builder.append("}");
        return builder.toString();
    }
}
