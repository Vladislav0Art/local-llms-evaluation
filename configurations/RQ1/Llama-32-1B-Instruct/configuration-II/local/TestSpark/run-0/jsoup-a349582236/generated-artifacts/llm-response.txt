public class Tag {
    private final String tagName;
    private boolean isBlock;
    private boolean formatAsBlock;
    private boolean preserveWhitespace;

    public Tag(String tagName) {
        this.tagName = tagName;
    }

    public void setIsBlock(boolean isBlock) {
        this.isBlock = isBlock;
    }

    public void setFormatAsBlock(boolean formatAsBlock) {
        this.formatAsBlock = formatAsBlock;
    }

    public boolean isBlock() {
        return isBlock;
    }

    public boolean formatAsBlock() {
        return formatAsBlock;
    }

    public boolean preserveWhitespace() {
        return preserveWhitespace;
    }

    public void setPreserveWhitespace(boolean preserveWhitespace) {
        this.preserveWhitespace = preserveWhitespace;
    }
}

public class TagRegistry {
    private final Map<String, Tag> tags = new HashMap<>();

    public void register(Tag tag) {
        tags.put(tag.tagName, tag);
    }

    public Tag get(String tagName) {
        return tags.get(tagName);
    }
}

public class HtmlEmitter implements Emitter {
    private static final TagRegistry TAG_REGISTRY = new TagRegistry();
    private static final Map<String, String> FORM_LISTED_TAGS = new HashMap<>();
    private static final Map<String, String> FORM_SUBMIT_TAGS = new HashMap<>();

    public static void register(String tagName) {
        TAG_REGISTRY.register(new Tag(tagName));
    }

    @Override
    public void emitTag(Tag tag) {
        if (FORM_LISTED_TAGS.containsKey(tag.tagName)) {
            // Emit the list item tag
            System.out.println("Listing item: " + tag.tagName);
            System.out.println("HTML Code:");
            // Generate HTML code for listing items
            // For simplicity, we will just print it out here
            System.out.println("<li>" + tag.tagName + "</li>");
        } else if (FORM_SUBMIT_TAGS.containsKey(tag.tagName)) {
            // Emit the submit button tag
            System.out.println("Submit Button: " + tag.tagName);
            System.out.println("HTML Code:");
            // Generate HTML code for submit buttons
            // For simplicity, we will just print it out here
            System.out.println("<button>" + tag.tagName + "</button>");
        } else {
            System.out.println("Unknown Tag: " + tag.tagName);
        }
    }

    public static void main(String[] args) {
        register("title");
        register("a");

        // Test the emitter with some tags
        HtmlEmitter emitter = new HtmlEmitter();
        emitter.emitTag(new Tag("h1"));
        emitter.emitTag(new Tag("button"));

        // List item test
        emitter.emitTag(new Tag("li"));
    }
}
