public class UrlBuilderTest {

    private String url;

    @Before
    public void setup() {
        url = "http://example.com/path?query=value#anchor";
    }

    public void testUrlBuilderConstructors() throws MalformedURLException, URISyntaxException, UnsupportedEncodingException {
        URLBuilder builder1 = new URLBuilder(url);
        URLBuilder builder2 = new URLBuilder(new URI("http://example.com"));
        assert builder1.u.equals(builder2.u);

        try {
            URLBuilder builder3 = new URLBuilder(null); // null input
        } catch (NullPointerException e) {
            assert false;
        }
    }

    public void testUrlBuilderAppendKeyVal() throws UnsupportedEncodingException, MalformedURLException {
        UrlBuilder urlBuilder = new URLBuilder(url);
        String[] kv = {"key1", "value1"};
        for (String kvValue : kv) {
            urlBuilder.appendKeyVal(kvValue.getBytes(UTF_8.NAME));
        }
        assert urlBuilder.q.toString().contains("&");
    }

    public void testUrlBuilderAppendKeyValEmpty() throws UnsupportedEncodingException, MalformedURLException {
        UrlBuilder urlBuilder = new URLBuilder(url);
        String[] kv = {"key1"};
        for (String kvValue : kv) {
            urlBuilder.appendKeyVal(kvValue.getBytes(UTF_8.NAME));
        }
    }

    public void testUrlBuilderAppendEscape() throws UnsupportedEncodingException, MalformedURLException {
        UrlBuilder urlBuilder = new URLBuilder(url);
        String encodedPath = "C:\\path\\to\\file.txt";
        StringBuilder sb = new StringBuilder();
        appendToAscii(encodedPath, true, sb);
        assert sb.toString().contains("%5B");
    }

    public void testUrlBuilderNormalize() throws MalformedURLException {
        UrlBuilder urlBuilder = new URLBuilder(url);

        // Test that puny-coding is done correctly
        String encodedHost = "punycoded-host";
        String normHost = URLDecoder.decode(encodedHost, UTF_8.name());
        assert normHost.equals("host");

        // Test that encoding non-ascii path components works as expected
        String encodedPath = "/path/with/descriptive/non/ascii-character.txt";
        StringBuilder sb = new StringBuilder();
        appendToAscii(encodedPath, false, sb);
        assert sb.toString().contains("%2F%25A1%20%29%3B");
    }

    public void testUrlBuilderEncodeNonAscii() throws MalformedURLException {
        UrlBuilder urlBuilder = new URLBuilder(url);

        String encodedHost = "punycoded-host";
        StringBuilder sb = new StringBuilder();
        appendToAscii(encodedHost, true, sb);
        assert sb.toString().contains("%5B");
    }

}
