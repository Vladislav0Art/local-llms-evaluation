public class CommentsConfigurationTests {

    private CommentsConfiguration commentsConfiguration;

    @Before
    void setUp() {
        commentsConfiguration = new CommentsConfiguration();
        SettingsHolder.registerComments("path1");
    }

    public void testSetComment_SingleLine() throws Exception {
        String path = "path1";
        String commentLines = "\nHello World";
        commentsConfiguration.setComment(path, commentLines);
        Assert.assertTrue(commentsConfiguration.getAllComments().containsKey(path));
        Assert.assertTrue(commentsConfiguration.getAllComments().get(path).containsAll(Arrays.asList(commentLines)));
    }

    public void testSetComment_SingleLineNoNewline() throws Exception {
        String path = "path1";
        String commentLines = "";
        commentsConfiguration.setComment(path, commentLines);
        Assert.assertTrue(commentsConfiguration.getAllComments().containsKey(path));
        Assert.assertTrue(commentsConfiguration.getAllComments().get(path).containsAll(Arrays.asList(commentLines)));
    }

    public void testSetCommentEmptyList() throws Exception {
        String path = "path1";
        String... commentLines = {};
        commentsConfiguration.setComment(path, commentLines);
        Assert.assertFalse(commentsConfiguration.getAllComments().containsKey(path));
        Assert.assertFalse(commentsConfiguration.getAllComments().get(path).containsAll(Arrays.asList(commentLines)));
    }

    public void testSetCommentMultipleLines() throws Exception {
        String path = "path1";
        String... commentLines = {"Hello World", "\nThis is another line"};
        commentsConfiguration.setComment(path, commentLines);
        Assert.assertTrue(commentsConfiguration.getAllComments().containsKey(path));
        Assert.assertTrue(commentsConfiguration.getAllComments().get(path).containsAll(Arrays.asList(commentLines)));
    }

    public void testGetAllComments() throws Exception {
        String path = "path1";
        commentsConfiguration.setComment(path, Collections.singletonList("Hello World"));
        Map<String, List<String>> expectedComments = new HashMap<>();
        expectedComments.put(path, Arrays.asList("Hello World"));
        Assert.assertTrue(commentsConfiguration.getAllComments().containsKey(path));
        Assert.assertTrue(commentsConfiguration.getAllComments().get(path).equals(expectedComments));
    }
}
