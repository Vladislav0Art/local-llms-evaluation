package org.jsoup.nodes;

import org.jsoup.helper.Validate;
import org.jsoup.internal.StringUtil;
import java.io.IOException;
import java.util.*;

/**
 * Create a new TextNode representing the supplied (unencoded) text).
 */
public class TextNodeTest {

    @org.junit.Test
    public void testIsBlank() {
        TextNode node = new TextNode("");
        assertTrue(node.isBlank());
        node.text("Hello World");
        assertFalse(node.isBlank());
    }

    @org.junit.Test
    public void testSplitText_0Offset() {
        TextNode node = new TextNode("   Hello World  ");
        TextNode splitNode = node.splitText(5);
        assertEquals("Hello World", splitNode.text());
    }

    @org.junit.Test
    public void testSplitText NegativeOffset() {
        TextNode node = new TextNode("   Hello World  ");
        try {
            node.splitText(-1);
            fail("Expected IOException");
        } catch (IOException e) {
            // Expected exception
        }
        TextNode splitNode = node.splitText(5);
        assertEquals("Hello World", splitNode.text());
    }

    @org.junit.Test
    public void testSplitText_SiblingOffset() {
        TextNode node = new TextNode("   Hello World  ");
        node.siblingIndex = 1;
        TextNode splitNode = node.splitText(0);
        assertEquals("   Hello World", splitNode.text());
        node.siblingIndex++;
        splitNode = node.splitText(-1);
        assertEquals("Hello World", splitNode.text());
    }

    @org.junit.Test
    public void testSplitText_SplitNode() {
        TextNode node = new TextNode("   Hello World  ");
        String text = node.text();
        TextNode splitNode = node.splitText(5);
        assertEquals(text.substring(0, 6), splitNode.text());
    }

    @org.junit.Test
    public void testGetWholeText_EmptyString() {
        TextNode node = new TextNode("");
        assertEquals("", node.getWholeText());
    }

    @org.junit.Test
    public void testGetWholeText_SingleChar() {
        TextNode node = new TextNode("a");
        assertEquals("a", node.getWholeText());
    }

    @org.junit.Test
    public void testGetWholeText_SplitString() {
        TextNode node = new TextNode("   Hello World  ");
        assertEquals("Hello World", node.getWholeText());
    }
