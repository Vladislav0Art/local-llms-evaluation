public class Comment {

    private final String comment;

    public Comment(String comment) {
        this.comment = comment;
    }

    @Override
    public String nodeName() {
        return "#comment";
    }

    @Override
    public String getData() {
        return comment;
    }

    @Override
    public void outerHtmlHead(Appendable accum, int depth, Document.OutputSettings out) throws IOException {
        if (out.prettyPrint()) {
            indent(accum, depth, out);
        }
        accum.append("<!--").append(comment).append("-->");
    }

    private void indent(Appendable accum, int depth, Document.OutputSettings out) throws IOException {
        for (int i = 0; i < depth; i++) {
            accum.append("\n");
        }
        accum.append("<!--").append(comment).append("-->");
    }

    public static Comment asComment(String data) {
        return new Comment(data);
    }

    public boolean isXmlDeclaration() {
        String comment = getData();
        return isXmlDeclarationData(comment);
    }

    private static boolean isXmlDeclarationData(String comment) {
        if (comment.length() > 1 && (comment.startsWith("!") || comment.startsWith("?"))) {
            return true;
        }
        return false;
    }

    @Override
    public Comment clone() {
        return new Comment(getData());
    }

    /**
     * Check if this comment looks like a JavaDoc comment.
     * 
     * @return true if it looks like, maybe, it's a JavaDoc comment.
     */
    public boolean isJavaDocComment() {
        String comment = getData();
        return comment.startsWith("@author") || comment.startsWith("@param") || comment.startsWith("@return");
    }

    private static Comment asJavaDocComment(String data) {
        return new Comment(data);
    }
}
