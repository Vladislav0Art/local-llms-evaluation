import org.junit.Test;
import static org.junit.Assert.assertTrue;

public class TextNodeTest {

    @Test
    public void testIsBlank() {
        String text = "   ";
        assertTrue(TextNode.isBlank(text));
        text = "";
        assertTrue(TextNode.isBlank(text));
        text = null;
        assertTrue(TextNode.isBlank(text));
    }

    @Test
    public void testSplitText() {
        String text = "Hello World";
        TextNode result = TextNode.splitText(6);
        assertEquals("World", result.text());
        assertTrue(!result.text().isEmpty());
    }

    @Test
    public void testSplitTextBlankOffset() {
        String text = "   Hello ";
        TextNode result = TextNode.splitText(-1);
        assertEquals("Hello ", result.text());
        assertTrue(result.text().isEmpty());
    }

    @Test
    public void testOuterHtmlHead() {
        Document document = new Document();
        Appendable accum = new StringBuilder();
        String text = "This is a test.\nIt has two lines.";
        TextNode node = TextNode.createFromEncoded(text);
        node.outerHtmlHead(accum, 0, document.OutputSettings());
        assertTrue(accum.toString().contains("test"));
    }

    @Test
    public void testOuterHtmlTail() {
        Document document = new Document();
        Appendable accum = new StringBuilder();
        String text = "This is a test.\nIt has two lines.";
        TextNode node = TextNode.createFromEncoded(text);
        TextNode tail = node.outerHtmlTail(accum, 0, document.OutputSettings());
        assertTrue(tail.text().contains("lines"));
    }

    @Test
    public void testClone() {
        Document document = new Document();
        Appendable accum = new StringBuilder();
        String text = "This is a test.\nIt has two lines.";
        TextNode node = TextNode.createFromEncoded(text);
        TextNode clone = (TextNode) node.clone();
        assertTrue(clone.text().equals(node.text()));
    }

    @Test
    public void testToString() {
        Document document = new Document();
        Appendable accum = new StringBuilder();
        String text = "This is a test.\nIt has two lines.";
        TextNode node = TextNode.createFromEncoded(text);
        assertEquals("TextNode{@node=text, nodeName=#text}", node.toString());
    }

}
