public class TextNodeTest {

    private Node parentNode;
    private Appendable accum;

    @Before
    public void setUp() {
        Document document = new Document();
        parentNode = document.createNode(TextNode.class);
        accum = new StringBuilder();
    }

    public void testBlank() throws IOException {
        parentNode.setCoreValue(StringUtil.normaliseWhitespace(""));
        verifyIsBlank(true);
    }

    public void testSplitText() throws IOException {
        parentNode.setCoreValue("hello world");
        TextNode result = parentNode.splitText(5);
        assertEquals(TextNode.class, result.getClass());
        assertEquals(2, result.getSiblingIndex());
        verifySiblings();
    }

    public void testOuterHtmlHead() throws IOException {
        parentNode.setCoreValue("<p>Hello <span>world</span></p>");
        parentNode.outerHtmlHead(accum, 0, Document.OutputSettings.PrettyPrint());
        verifyOutput("Hello <span>world</span>\n");
    }

    public void testOuterHtmlTail() throws IOException {
        parentNode.setCoreValue("<p>Hello <span>world</span></p>");
        parentNode.outerHtmlTail(accum, 0);
        verifyTrimmedText();
    }

    public void testToString() throws IOException {
        TextNode result = parentNode.toString();
        verifyResult(result.getClass());
    }

    public void testClone() throws IOException {
        parentNode.setCoreValue("<p>Hello <span>world</span></p>");
        TextNode clone = parentNode.clone();
        assertEquals(TextNode.class, clone.getClass());
        assertEquals(clone.getSiblingIndex(), parentNode.getSiblingIndex());
    }

    private void verifyIsBlank(boolean condition) {
        if (condition) {
            assertTrue("Is blank is true", parentNode.isBlank());
        } else {
            assertFalse("Is blank is false", parentNode.isBlank());
        }
    }

    private void verifySiblings() {
        assertEquals(2, parentNode.getSiblingIndex());
    }

    private void verifyTrimmedText() throws IOException {
        String text = parentNode.getText();
        assertTrue("Trimming text is true", trimLeading(text));
        assertTrue("Trimming trailing text is true", trimTrailing(text));
    }

    private void verifyOutput(String expected) throws IOException {
        assertEquals(expected, parentNode.outerHtml());
    }

    private void verifyResult(Class<?> clazz) {
        assertEquals(clazz, parentNode.getClass());
    }

    private void verifyTrimmedText() throws IOException {
        String text = parentNode.getText();
        assertTrue("Trimming text is true", trimLeading(text));
        assertTrue("Trimming trailing text is true", trimTrailing(text));
    }
}
