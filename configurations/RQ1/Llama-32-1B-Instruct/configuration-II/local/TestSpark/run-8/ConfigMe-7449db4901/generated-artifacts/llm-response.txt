import ch.jalu.configme.configurationdata.CommentsConfiguration;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.mockito.Mock;
import org.mockito.Mockito;

import java.util.Collections;
import java.util.List;

public class CommentsConfigurationTest {

    private CommentsConfiguration commentsConfiguration;

    @Mock
    private Map<String, List<String>> mockComments;

    @BeforeEach
    public void setUp() {
        commentsConfiguration = new CommentsConfiguration(mockComments);
    }

    @Test
    public void setComment_SingleLine_ReturnsEmptyList() {
        commentsConfiguration.setComment("path", "\n");
        Mockito.verify(mockComments).put("path", Collections.emptyList());
    }

    @Test
    public void setComment_HiddenLines_ReturnsEmptyList() {
        commentsConfiguration.setComment("path", new String[]{"line1", "line2"});
        Mockito.verify(mockComments).put("path", Collections.emptyList());
    }

    @Test
    public void getAllComments_FullMap_ReturnsFullMap() {
        List<String> commentLines = Arrays.asList("comment1", "comment2");
        commentsConfiguration.setComment("path", commentLines);
        Mockito.verifyAll();
    }

    @Test
    public void getAllComments_EmptyMap_ReturnsEmptyMap() {
        commentsConfiguration.setComment("path", new String[0]);
        Mockito.verifyAll();
    }
}
