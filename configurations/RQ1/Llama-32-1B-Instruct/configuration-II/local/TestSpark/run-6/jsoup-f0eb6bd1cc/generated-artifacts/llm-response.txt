public class UrlBuilderTest {

    private static final String TEST_URL = "https://example.com/path/to/resource";

    public void testUrlNormalization() throws UnsupportedEncodingException {
        UrlBuilder urlBuilder = new UrlBuilder(UrlBuilder.build());
        assertEquals("https://example.com/path%2Fto%2Fresource", urlBuilder.u.toString());
    }

    public void testAppendKeyVal() throws UnsupportedEncodingException {
        String key = "key";
        String value = "value";
        UrlBuilder urlBuilder = new UrlBuilder(UrlBuilder.build());
        urlBuilder.appendKeyVal(new javax.net.KeyVal(key, value));
        assertEquals("key=value", urlBuilder.q.toString());
    }

    public void testURIEncoding() throws UnsupportedEncodingException {
        URL url = new URI(TEST_URL);
        StringBuilder normUrl = new StringBuilder();
        UriBuilder uriBuilder = new UriBuilder(url.getScheme(), null, null, null);
        String[] decodedParts = uriBuilder.decodePath();
        assertEquals("https://example.com/path%2Fto%2Fresource", String.join("%3A%2F", decodedParts));
    }

    public void testURLDecoding() throws UnsupportedEncodingException {
        URL url = new URI(TEST_URL);
        String[] decodedParts = url.getQuery().split("&");
        assertEquals("key=value", decodePart(decodedParts[0]));
    }

    public void testNullInputUrl() {
        try {
            UrlBuilder.urlBuilder(null);
            assert false;
        } catch (NullPointerException e) {
            // expected
        }
    }

    public void testNonAsciiPathComponent() throws UnsupportedEncodingException {
        URL url = new URI(TEST_URL);
        String[] decodedParts = url.getQuery().split("&");
        assertEquals("path%2Fto%2Fresource", decodePart(decodedParts[0]));
    }

    public void testAppendToAscii() throws UnsupportedEncodingException {
        String inputString = "This is a test string.";
        UrlBuilder urlBuilder = new UrlBuilder(UrlBuilder.build());
        StringBuilder result = new StringBuilder();
        urlBuilder.appendToAscii(inputString, false, result);
        assertEquals("This%20is%20a%20test%20string.", result.toString());
    }

    public void testValidateAssertions() throws MalformedURLException, URISyntaxException {
        assert Validate.assertFail("Invalid input URL");
    }
}
