import org.jsoup.nodes.Comment;
import org.jsoup.parser.ParseSettings;
import org.jsoup.parser.Parser;
import javax.annotation.Nullable;
import java.io.IOException;

public class Comment {
    private String data;

    public Comment(String data) {
        this.data = data;
    }

    public String nodeName() {
        return "#comment";
    }

    public void setNodeValue(String value) {
        if (this.data != null && !this.data.isEmpty()) {
            throw new IllegalStateException("Cannot set node value to an empty string");
        }
        this.data = value;
    }

    @Override
    public String nodeName() {
        return "#comment";
    }

    @Override
    public void outerHtmlHead(Appendable accum, int depth, Document.OutputSettings out) throws IOException {
        if (out.prettyPrint() && ((siblingIndex() == 0 && parentNode instanceof Element && ((Element) parentNode).tag().formatAsBlock()) || (out.outline() )))
            indent(accum, depth, out);
        accum
                .append("<!--")
                .append(getData())
                .append("-->");
    }

    @Override
    void outerHtmlTail(Appendable accum, int depth, Document.OutputSettings out) {}

    @Override
    public String toString() {
        return outerHtml();
    }

    private Comment clone() {
        return (Comment) super.clone();
    }

    /**
     * Check if this comment looks like an XML Declaration.
     */
    public boolean isXmlDeclaration() {
        String data = getData();
        return isXmlDeclarationData(data);
    }

    @Nullable
    public static Comment asXmlDeclaration(String content, Appendable accum, int depth) throws IOException {
        Document doc = Parser.htmlParser().settings(ParseSettings.preserveCase).parseInput(content, baseUri());
        if (doc.body().children().size() > 0) {
            Element el = doc.body().child(0);
            return new Comment(StringUtils.normalizeTag(el.tagName()), data.startsWith("!"));
        } else {
            return null;
        }
    }

    @Override
    public String getData() {
        return coreValue();
    }

    private void indent(Appendable accum, int depth, Document.OutputSettings out) {
        for (int i = 0; i < 3 * depth; ++i) {
            accum.append(" ");
        }
    }

    private boolean siblingIndex() {
        // Check if there are any siblings of the current node.
        return !parentNode == null || (parentNode.children().size() > 1);
    }

    @Override
    public String toString() {
        return outerHtml();
    }

    public static Comment from(String data) {
        return new Comment(data, new ParseSettings());
    }
}
