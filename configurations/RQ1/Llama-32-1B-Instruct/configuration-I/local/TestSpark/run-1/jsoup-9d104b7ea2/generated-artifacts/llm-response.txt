public class TextNode {

    private String value;

    @Override
    public String nodeName() {
        return "#text";
    }

    /**
     * Get the text content of this text node.
     * @return Unencoded, normalised text.
     */
    public String text() {
        return StringUtil.normaliseWhitespace(getWholeText());
    }

    /**
     * Set the text content of this text node.
     * @param text unencoded text
     * @return this, for chaining
     */
    public TextNode text(String text) {
        coreValue(text);
        return this;
    }

    /**
     Get the (unencoded) text of this text node, including any newlines and spaces present in the original.
     @return text
     */
    public String getWholeText() {
        return coreValue();
    }

    /**
     Test if this text node is blank -- that is, empty or only whitespace (including newlines).
     @return true if this document is empty or only whitespace, false if it contains any text content.
     */
    public boolean isBlank() {
        return StringUtil.isBlank(coreValue());
    }

    /**
     * Split this text node into two nodes at the specified string offset. After splitting, this node will contain
     * the original text up to the offset, and will have a new text node sibling containing the text after the offset.
     * @param offset string offset point to split node at.
     * @return the newly created text node containing the text after the offset.
     */
    public TextNode splitText(int offset) {
        final String text = coreValue();
        Validate.isTrue(offset >= 0, "Split offset must be not be negative");
        Validate.isTrue(offset < text.length(), "Split offset must not be greater than current text length");

        String head = text.substring(0, offset);
        String tail = text.substring(offset);
        text(head);
        TextNode tailNode = new TextNode(tail);
        if (parentNode != null)
            parentNode.addChildren(siblingIndex()+1, tailNode);

        return tailNode;
    }

    @Override
    public void outerHtmlHead(Appendable accum, int depth, Document.OutputSettings out) throws IOException {
        final boolean prettyPrint = out.prettyPrint();
        final Element parent = parentNode instanceof Element ? ((Element) parentNode) : null;
        final boolean normaliseWhite = prettyPrint && !Element.preserveWhitespace(parentNode);
        final boolean trimLikeBlock = parent != null && (parent.tag().isBlock() || parent.tag().formatAsBlock());
        boolean trimLeading = false, trimTrailing = false;

        if (normaliseWhite) {
            trimLeading = (trimLikeBlock && siblingIndex == 0) || parentNode instanceof Document;
            trimTrailing = trimLikeBlock && nextSibling() == null;

            // if this text is just whitespace, and the next node will cause an indent, skip this text:
            Node next = nextSibling();
            Node prev = previousSibling();
            boolean isBlank = isBlank();
            boolean couldSkip = (next instanceof Element && ((Element) next).shouldIndent(out)) // next will indent
                || (next instanceof TextNode && (((TextNode) next).isBlank())) // next is blank text, from re-parenting
                || (prev instanceof Element && ((Element) prev).isBlock())
                ;
            if (couldSkip && isBlank) return;

            if (
                (siblingIndex == 0 && lastCharIsWhitespace(sandbox)) ||
                (lastCharIsWhitespace(sandbox) && parent != null)
            )
                indent(accum, depth, out);
        }

        Entities.escape(accum, coreValue(), out, false, normaliseWhite, trimLeading, trimTrailing);
    }

    @Override
    public void outerHtmlTail(Appendable accum, int depth, Document.OutputSettings out) throws IOException {}

    /**
     * Create a new TextNode from HTML encoded (aka escaped) data.
     * @param encodedText Text containing encoded HTML (e.g. {@code &lt;})
     */
    public static TextNode createFromEncoded(String encodedText) {
        String text = Entities.unescape(encodedText);
        return new TextNode(text);
    }

    public static void testCreateFromEncoded() {
        TextNode createdNode1 = TextNode.createFromEncoded("Hello, ");
        TextNode createdNode2 = TextNode.createFromEncoded("World!");
        assert createdNode1.value.equals("Hello, ") && createdNode2.value.equals("World!");
    }
}

    private static void printNodes(Appendable appendable, String indent, Document outputSettings, boolean prettyPrint) {
        if (indent.isEmpty()) {
            return;
        }
        appendable.append("\n");
        appendable.append(indent).append("  ");
        if (prettyPrint) {
            System.out.print(outputSettings.getIndent());
        }
        for (int i = 0; i < outputSettings.getDefaultIndent(); i++) {
            appendable.append("  ");
        }
        for (Node node : document.childNodes()) {
            printNodes(appendable, indent + "\n", outputSettings, prettyPrint);
        }
    }

    private static void printNodes(Appendable appendable) {
        appendable.append("\n");
        appendable.append("  ");
        if (prettyPrint) {
            System.out.print(outputSettings.getIndent());
        }
        for (int i = 0; i < outputSettings.getDefaultIndent(); i++) {
            appendable.append("  ");
        }
    }

    private static String indent(String string, int spacesToInsert) {
        StringBuilder builder = new StringBuilder();
        int level = 0;
        while (level > 0) {
            builder.insert(0, " ".repeat(spacesToInsert * -1));
            level--;
        }
        return builder.toString() + string;
    }

    private static String truncate(String string, int maxLength) {
        if (string.length() <= maxLength) {
            return string;
        } else {
            return string.substring(0, maxLength);
        }
    }

class TextNode {
    private Appendable appendable;
    private int siblingIndex;

    public TextNode(Appendable appendable) {
        this.appendable = appendable;
        this.siblingIndex = -1;
    }

    public void text(String value) {
        if (value.length() > 100) {
            String truncatedValue = truncate(value, 99);
            appendable.append("\n");
            appendable.append("  ");
            if (prettyPrint) {
                System.out.print(outputSettings.getIndent());
            }
            for (int i = 0; i < outputSettings.getDefaultIndent(); i++) {
                appendable.append("  ");
            }
            appendable.append(truncatedValue);
        } else {
            appendable.append(value);
        }
    }

    public Appendable createFromEncoded(String encodedText) {
        return new TextNode(encodedText, this, siblingIndex + 1);
    }

    public void addChildren(int offset, Node node) {
        if (offset < 0 || offset > document.childNodes.length) {
            throw new IndexOutOfBoundsException();
        }
        int childIndex = document.childNodes.length - offset;
        for (int i = 0; i < childIndex; i++) {
            document.childNodes[i].removeChild(document.childNodes[offSet + i]);
        }
        document.childNodes[offSet + i] = node;
    }

    public void splitText(int offset) {
        if (offset >= 0 && offset < document.childNodes.length) {
            int childIndex = document.childNodes.length - offset;
            for (int i = 0; i < childIndex; i++) {
                document.childNodes[i].removeChild(document.childNodes[offSet + i]);
            }
            document.childNodes[offSet + offset] = new TextNode(doc.createElement("p"));
        }
    }

    @Override
    public String toString() {
        Appendable appendable = new StringBuilder();
        appendable.append(this.appendable.toString());
        return appendable.toString();
    }
}

class Document {
    private List<Node> nodes;

    public Document() {
        this.nodes = new ArrayList<>();
    }

    public void addNode(Node node) {
        this.nodes.add(node);
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        for (Node node : nodes) {
            sb.append(node.toString()).append("\n");
        }
        return sb.toString();
    }
}

class OutputSettings {
    private int indentLevel;

    public OutputSettings(int indentLevel) {
        this.indentLevel = indentLevel;
    }

    public int getIndentLevel() {
        return indentLevel;
    }
}
