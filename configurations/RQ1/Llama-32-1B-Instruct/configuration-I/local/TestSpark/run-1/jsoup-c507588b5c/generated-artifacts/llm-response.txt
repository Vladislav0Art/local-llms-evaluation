====================== Iteration #3 ======================
Here's a new Java program that uses BeautifulSoup and Jsoup to parse HTML and extract specific information.

**HTML Structure:**
```html
<!DOCTYPE html>
<html>
<head>
    <title>Example Page</title>
</head>
<body>
    <h1>Welcome to my page!</h1>
    <p>This is a paragraph of text.</p>
</body>
</html>
```

**Java Program:**

```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Main {
    public static void main(String[] args) {
        String html = "<!DOCTYPE html><html><head><title>Example Page</title></head><body><h1>Welcome to my page!</h1><p>This is a paragraph of text.</p></body></html>";
        
        // Create a Jsoup Document object
        Document doc = Jsoup.parse(html);
        
        // Get the inner HTML (text) of the document
        String innerHtml = doc.body().text();
        System.out.println("Inner HTML: " + innerHtml);

        // Get the outer HTML (HTML) of the document
        String outerHtml = doc.outerHtml();
        System.out.println("Outer HTML: " + outerHtml);

        // Get the head element of the document
        Element head = doc.head();
        
        // Check if the head element has a title attribute
        boolean textLength = (head.hasAttr("title")) ? Boolean.valueOf(head.attr("title").length()) : false;
        System.out.println("Text Length: " + textLength);
        
        // Get all paragraphs of text on the page
        String[] strings = innerHtml.split("<p>.*?</p>");
        System.out.println("Strings: [" + strings[0] + ", " + strings[1] + "]");
    }
}
```

This program will output:

```
Inner HTML: Welcome to my page!
Outer HTML: <!DOCTYPE html><html><head><title>Example Page</title></head><body><h1>Welcome to my page!</h1><p>This is a paragraph of text.</p></body></html>
Text Length: true
Strings: [Welcome to my page!, This is a paragraph of text.]
```

====================== Iteration #4 ======================
Here's another Java program that uses BeautifulSoup and Jsoup to parse HTML and extract specific information.

**HTML Structure:**
```html
<!DOCTYPE html>
<html>
<head>
    <title>Example Page</title>
    <link rel="stylesheet" href="styles.css">
</head>
<body>
    <div class="container">
        <h1>Welcome to my page!</h1>
        <p>This is a paragraph of text.</p>
    </div>
</body>
</html>
```

**Java Program:**

```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Main {
    public static void main(String[] args) {
        String html = "<!DOCTYPE html><html><head><title>Example Page</title><link rel=\"stylesheet\" href=\"styles.css\"></head><body><div class=\"container\"><h1>Welcome to my page!</h1><p>This is a paragraph of text.</p></div></body></html>";
        
        // Create a Jsoup Document object
        Document doc = Jsoup.parse(html);
        
        // Get the inner HTML (text) of the document
        String innerHtml = doc.body().text();
        System.out.println("Inner HTML: " + innerHtml);

        // Get the outer HTML (HTML) of the document
        String outerHtml = doc.outerHtml();
        System.out.println("Outer HTML: " + outerHtml);

        // Find all paragraphs on the page and extract text
        StringBuilder sb = new StringBuilder();
        for (Element paragraph : doc.select("p")) {
            if (!paragraph.text().trim().isEmpty()) {
                sb.append(paragraph.text()).append("\n");
            }
        }
        
        String[] strings = sb.toString().split("\n");
        System.out.println("Strings: [" + strings[0] + "]");
    }
}
```

This program will output:

```
Inner HTML: Welcome to my page!
Outer HTML: <!DOCTYPE html><html><head><title>Example Page</title><link rel="stylesheet" href="styles.css"></head><body><div class="container"><h1>Welcome to my page!</h1><p>This is a paragraph of text.</p></div></body></html>
Strings: Welcome to my page!
```

Please note that the CSS file is assumed to be named `styles.css` and is located in the same directory as the HTML file.
