public class Main {
    public static void testBuilding() {
        String path = "path/to/resource?param1=value1&param2=value2";
        String encodedPath = encodePart(path);
        assertEquals(encodedPath, "https%3A%2F%2Fexample.com%2Fpath%2Fto%2Fresource");
    }

    public static void testBuildingMethod() {
        String path = "path/to/resource?param1=value1&param2=value2";
        String encodedPath = build(path);
        assertEquals(encodedPath, "https%3A%2F%2Fexample.com%2Fpath%2Fto%2Fresource");
    }

    private static String encodePart(String path) {
        return encode(partsToParts(path));
    }

    public static String[] partsToParts(String path) {
        // Split the path by '/' and split each part by '?'
        String[] parts = path.split("/");

        // Replace '?' with '%3F' for URL encoding
        StringBuilder encodedPath = new StringBuilder();
        for (String part : parts) {
            if (part.equals("?")) {
                encodedPart = "%3F";
            } else {
                encodedPart.append(part);
            }
        }

        return encodedPath.toString().split("%2F");
    }

    public static String build(String path) {
        // Split the path by '/' and append '?' to each part
        String[] parts = path.split("/");
        StringBuilder encodedPath = new StringBuilder();
        for (String part : parts) {
            if (!part.equals("?")) {
                encodedPart.append(part).append("%3F");
            } else {
                encodedPart.append(part);
            }
        }

        return encodedPath.toString();
    }
}
