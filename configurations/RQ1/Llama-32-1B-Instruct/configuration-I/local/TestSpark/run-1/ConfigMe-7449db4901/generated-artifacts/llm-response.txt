public class CommentsConfigurationTest {

    private CommentsConfiguration testObj;

    @Before
    public void setup() {
        MockitoAnnotations.initMocks(this);
    }

    public void testGetAllComments() {
        Map<String, List<String>> comments = new HashMap<>();
        comments.put("path", Collections.singletonList("comment1"));
        comments.put("path2", Collections.singletonList("comment2"));

        testObj = new CommentsConfiguration(comments);

        when(testObj.getAllComments()).thenReturn(Collections.unmodifiableMap(testObj.comments));

        assertEquals(Collections.unmodifiableMap(testObj.comments), testObj.getAllComments());
    }

    public void testSetComment() {
        Map<String, List<String>> comments = new HashMap<>();
        comments.put("path", Collections.singletonList("comment1"));
        comments.put("path2", Collections.singletonList("comment2"));

        testObj = new CommentsConfiguration(comments);

        when(testObj.setComment("path", "newLine comment")) {
            @Mock
            String getCommentLines = mock(String.class);
            when(getCommentLines).thenReturn(Arrays.asList(new String[]{"comment1"}));
            when(getCommentLines).thenAnswer(invocation -> Arrays.asList(new String[]{"newLine comment"}));

            testObj.setComment("path", "newLine comment");
        }

        assertEquals(Collections.unmodifiableList(Arrays.asList("newLine comment")), testObj.getCommentsForPath("path"));
    }

    public void testGetAllCommentsNoNewlineLines() {
        Map<String, List<String>> comments = new HashMap<>();
        comments.put("path", Collections.singletonList("comment1"));

        testObj = new CommentsConfiguration(comments);

        when(testObj.getAllComments()).thenReturn(Collections.unmodifiableMap(testObj.comments));

        assertEquals(Collections.unmodifiableMap(testObj.comments), testObj.getAllComments());
    }

    public void testGetAllCommentsEmptyMap() {
        Map<String, List<String>> comments = new HashMap<>();

        testObj = new CommentsConfiguration(comments);

        when(testObj.getAllComments()).thenReturn(Collections.unmodifiableMap(null));

        assertEquals(Collections.emptySet(), testObj.getAllComments());
    }
}

public class CommentsConfigurationMockedTest {

    private CommentsConfiguration testObj;

    @Before
    public void setup() {
        MockitoAnnotations.initMocks(this);
    }

    public void testGetAllCommentsNoNewlineLines() {
        Map<String, List<String>> comments = new HashMap<>();
        comments.put("path", Collections.singletonList("comment1"));

        when(testObj.getAllComments()).thenReturn(Collections.unmodifiableMap(map -> map.entrySet().stream()
                .collect(Collectors.toMap(Map.Entry::getKey, Map.Entry::getValue)));
    }

    public void testGetAllComments() {
        String expected = "comment1";
        Map<String, List<String>> actual = testObj.getAllComments();
        assertEquals(expected, actual.get("path").get(0));
    }
}
