```java
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class TextNodeTest {

    public static Node createTextNode(String text) {
        return new TextNode(text);
    }

    @Test
    public void testIsEmpty() {
        assertNull(createTextNode("").splitText(0).text());
        assertNull(createTextNode("   ").splitText(-1));
    }

    @Test
    public void testIsBlank() {
        assertTrue(createTextNode("a").blank());
        assertFalse(createTextNode(" a").blank(null));
    }

    @Test
    public void testSplitText_SingleCharacter() {
        TextNode text = createTextNode("a");
        assertNotNull(text.splitText(0).text());
    }

    @Test
    public void testSplitText_EmptyString() {
        TextNode text = new TextNode("");
        assertNull(text.splitText(-1));
    }

    @Test
    public void testSplitText_OneCharacterString() {
        TextNode text = createTextNode("a");
        assertNotNull(text.splitText(0).text());
    }

    @Test
    public void testSplitText_Whitespace() {
        TextNode text = new TextNode("   ");
        assertNotNull(text.splitText(-1));
    }

    @Test
    public void testSplitText_MultipleCharactersString() {
        TextNode text = createTextNode("hello world");
        assertNotNull(text.splitText(7).text());
    }

    @Test
    public void testOuterHtmlHead_NestedNodes() {
        Elements nodes = new Elements();
        Document doc = new Document();
        for (int i = 0; i < 10; i++) {
            Element element = new Element("div");
            node = createTextNode(String.valueOf(i));
            element.appendChild(node);
            nodes.add(element);
        }
        doc.appendChild(nodes);
        TextNode text = createTextNode(doc.body().text());
        System.out.println(text.text());
    }

    @Test
    public void testOuterHtmlTail_NestedNodes() {
        Elements nodes = new Elements();
        Document doc = new Document();
        for (int i = 0; i < 10; i++) {

            element.appendChild(node);
            nodes.add(element);
        }
        doc.appendChild(nodes);
        TextNode text = createTextNode(doc.body().text());

    }
}
```

```java
public class Element {
    public String name;
    public String text;

    public Element(String name, String text) {
        this.name = name;
        this.text = text;
    }

    public void appendChild(Element node) {
        System.out.println(name + ": " + text);
    }
}

class Document {
    public String content = "";
    public List<Element> elements = new ArrayList<>();

    public Element createElement(String name, String text) {
        return new Element(name, text);
    }

    public void appendChild(Element element) {
        elements.add(element);
    }

    public void print() {
        for (Element element : elements) {
            System.out.println("Element: " + element.name + ": " + element.text);
        }
    }
}

public class TestElement extends Document {

    @Override
    public Element createElement(String name, String text) {
        return new Element(name, text);
    }

    @Override
    public void appendChild(Element element) {
        elements.add(element);
    }

    @Override
    public void print() {
        System.out.println("Content: " + content);
    }
}

public class TestNode extends Document {

    private String name;
    private String text;

    public TestNode(String name, String text) {
        super(name, text);
        this.name = name;
        this.text = text;
    }

    @Override
    public Element createElement(String name, String text) {
        return new Element(name, text);
    }

    @Override
    public void appendChild(Element element) {
        elements.add(element);
    }

    @Override
    public void print() {
        System.out.println("Content: " + content + "\n" + "Elements: \n" + "\n".join(elements.stream().map(E -> E.name + ": " + E.text).collect(Collectors.joining("\n")));
    }
}

public class TestElementTree extends Document {

    private List<TestNode> nodes = new ArrayList<>();

    public TestElementTree(String name, String text) {
        super(name, text);
    }

    @Override
    public Element createElement(String name, String text) {
        return new Element(name, text);
    }

    @Override
    public void appendChild(Element element) {
        nodes.add(element);
    }

    @Override
    public void print() {
        System.out.println("Content: " + content + "\n" + "Elements:\n" + "\n".join(nodes.stream().map(E -> E.name + ": " + E.text).collect(Collectors.joining("\n")));
    }
}

public class TestDocumentTree extends Document {

    private List<TestElement> elements = new ArrayList<>();

    public TestDocumentTree(String name, String text) {
        super(name, text);
    }

    @Override
    public Element createElement(String name, String text) {
        return new Element(name, text);
    }

    @Override
    public void appendChild(Element element) {
        elements.add(element);
    }

    @Override
    public void print() {
        System.out.println("Content: " + content + "\n" + "Elements:\n" + "\n".join(elements.stream().map(E -> E.name + ": " + E.text).collect(Collectors.joining("\n")));
    }
}

public class Main {

    public static void main(String[] args) {
        Document doc = new Document();

        Element root = doc.createElement("root", "Root");
        TestNode node1 = doc.createElement("node1", "Hello World!");
        TestNode node2 = doc.createElement("node2", "This is another element.");
        TestNode node3 = doc.createElement("node3", "And this one too!");

        root.appendChild(node1);
        root.appendChild(node2);
        root.appendChild(node3);

        Element childElement = doc.createElement("child", "Child");
        node1.appendChild(childElement);

        Document treeDoc = new TestDocumentTree("", "");

        treeDoc.appendChild(root);
        treeDoc.appendChild(doc);

        treeDoc.print();
    }
}
```
