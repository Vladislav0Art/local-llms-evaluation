public class TagParser {

    private static final Map<String, Tag> tags = new HashMap<>();

    public static void register(Tag tag) {
        tags.put(tag.tagName, tag);
    }

    @Test
    public void testRegister() {
        // Test that the tag is registered correctly
        Tag tag = new Tag("span");
        register(tag);
        assert tags.containsKey(tag.tagName);

        tag = new Tag("img");
        register(tag);
        assert tags.containsKey(tag.tagName);
    }

    public static class Tag {

        private final String tagName;
        private boolean isBlock;
        private boolean formatAsBlock;
        private boolean preserveWhitespace;

        public Tag(String tagName) {
            this.tagName = tagName;
        }

        public void setBlock(boolean block) {
            isBlock = block;
        }

        public void setFormatAsBlock(boolean formatAsBlock) {
            formatAsBlock = formatAsBlock;
        }

        public void setPreserveWhitespace(boolean preserveWhitespace) {
            preserveWhitespace = preserveWhitespace;
        }
    }

    @Test
    public void testRegisterTag() {
        // Test that the tag name and its corresponding Tag object are registered correctly
        register(new Tag("span"));
        register(new Tag("img"));

        Tag tag = new Tag("span");
        assertTrue(tags.containsKey(tag.tagName));

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class TagParser {

    private static final Map<String, Tag> tags = new HashMap<>();

    public static void register(Tag tag) {
        if (tag.tagName == null || tag.tagName.isEmpty()) {
            throw new IllegalArgumentException("Tag name cannot be empty");
        }
        tags.put(tag.tagName.toLowerCase(), tag);
    }

    @Test
    public void testRegister() {
        // Test that the tag is registered correctly
        Tag tag = new Tag("div");
        register(tag);
        Document doc = Jsoup.parse("<html><body><div id='test'></div></body></html>");
        for (Map.Entry<String, Tag> entry : tags.entrySet()) {
            Element element = doc.getElementById(entry.getKey());
            if (element != null) {
                System.out.println("Tag found: " + entry.getKey() + " has tag " + entry.getValue().getTag());
            }
        }
    }

    public static class Tag {
        private String tagName;

        public Tag(String tagName) {
            this.tagName = tagName;
        }

        public String getTagName() {
            return tagName;
        }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (o == null || getClass() != o.getClass()) return false;
            Tag tag = (Tag) o;
            return Objects.equals(tag.tagName, this.tagName);
        }

        @Override
        public int hashCode() {
            return Objects.hash(tagName);
        }
    }

    public static void main(String[] args) {
        register(new Tag("div"));
        register(new Tag("span"));
    }
}
