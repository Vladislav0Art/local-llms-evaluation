import java.io.ByteArrayOutputStream;
import java.io.PrintStream;

public class TestUrlBuilder {
    private static final String input = "https://example.com/path?query=valid#ref";
    private static final ByteArrayOutputStream outContent = new ByteArrayOutputStream();
    private static final PrintStream originalOut = System.out;

    public static void main(String[] args) throws Exception {
        System.setOut(new PrintStream(outContent));

        TestUrlBuilder.urlBuilder(input);
        System.out.println("Test 1:");
        System.out.println(outContent.toString());
        System.out.println();

        outContent.reset();
        System.setOut(originalOut);

        TestUrlBuilder.urlBuilder2(input);
        System.out.println("Test 2:");
        System.out.println(outContent.toString());
        System.out.println();

        TestUrlBuilder.urlBuilder3(input);
        System.out.println("Test 3:");
        System.out.println(outContent.toString());
        System.out.println();
    }

    public static void urlBuilder(String input) {
        outContent.write("https://example.com/path?query=" + input + " & ref=" + System.identityHashCode(this) + "\n");
    }

    public static void urlBuilder2(String input) throws Exception {
        ByteArrayOutputStream outContent = new ByteArrayOutputStream();
        outContent.write(input);
        outContent.flush();

        TestUrlBuilder.outContent.reset();
        System.setOut(new PrintStream(outContent));
    }

    public static void urlBuilder3(String input) throws Exception {
        ByteArrayOutputStream outContent = new ByteArrayOutputStream();
        outContent.write(input);

        for (int i = 0; i < 10000; i++) {
            outContent.write("https://example.com/path?query=" + input + " & ref=" + System.identityHashCode(this));
        }
        outContent.flush();

        TestUrlBuilder.outContent.reset();
        System.setOut(new PrintStream(outContent));
    }
}
