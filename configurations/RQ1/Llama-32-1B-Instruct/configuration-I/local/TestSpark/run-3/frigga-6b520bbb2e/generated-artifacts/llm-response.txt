import com.netflix.frigga.NameConstants;
import org.junit.AfterEach;
import org.junit.Before;
import org.junit.Test;
import static org.mockito.Mockito.*;

public class AppVersionTest {

    private AppVersion appVersion;

    @Before
    public void setup() {
        appVersion = new AppVersion();
    }

    @Test
    public void testParseName() {
        String amiName = "subscriberha-1.0.0-586499";
        when(appVersion.getParser().parseName(amiName)).thenReturn(new AppVersion());
        AppVersion parsedVersion = appVersion.parseName(amiName);
        assertEquals("subscriberha-1.0.0-586499", parsedVersion.getPackageName());
    }

    @Test
    public void testParseNameNull() {
        when(appVersion.getParser().parseName(null)).thenReturn(new AppVersion());
        try {
            appVersion.parseName(null);
            fail();
        } catch (Exception e) {}

        when(appVersion.getParser().parseName("invalidamiName")).thenReturn(null);

        try {
            appVersion.parseName("invalidamiName");
            fail();
        } catch (Exception e) {}

        AppVersion parsedVersion = appVersion.parseName("");
        assertEquals(0, parsedVersion.getVersion());
    }

    @Test
    public void testNullSafeStringComparator() {
        String one = "hello";
        String two = "world";

        when(appVersion.nullSafeStringComparator(one)).thenReturn(1);
        when(appVersion.nullSafeStringComparator(two)).thenReturn(-1);

        assertEquals(1, appVersion.nullSafeStringComparator(one));
        assertEquals(-1, appVersion.nullSafeStringComparator(two));

        assertNotEquals(one, appVersion.nullSafeStringComparator(two));
    }

    @Test
    public void testEquals() {
        AppVersion appVersion1 = new AppVersion();
        AppVersion appVersion2 = new AppVersion();

        when(appVersion1.getBuilder().getPackageName()).thenReturn("rpm");
        when(appVersion2.getBuilder().getPackageName()).thenReturn("rpm");

        assertTrue(appVersion1.equals(appVersion2));

        when(appVersion1.getBuilder().getVersion()).thenReturn("1.0.0-586499");
        when(appVersion2.getBuilder().getVersion()).thenReturn("1.0.0-586499");

        assertFalse(appVersion1.equals(appVersion2));
    }

    @Test
    public void testGetBuildJobName() {
        AppVersion appVersion = new AppVersion();

        when(appVersion.getBuilder().getBuildJobName()).thenReturn(null);

        assertEquals(null, appVersion.getBuildJobName());
    }

    @AfterEach
    public void tearDown() {
        // No need to tear down anything in this test
    }
}
