public class Solution {
    private int[][] dp;
    private boolean[] visited;

    public int find(int i, int j) {
        if (i < 0 || j < 0 || i == dp.length || j == dp[0].length)
            return -1;
        if (visited[i][j])
            return dp[i][j];
        else
            visited[i][j] = true;
        if (dp[i][j] != -1) {
            return dp[i][j];
        }
        int[] directions = {{-1, 0}, {1, 0}};
        for (int direction : directions) {
            int ni = i + direction[0], nj = j + direction[1];
            if (ni >= 0 && ni < dp.length && nj >= 0 && nj < dp[0].length)
                return dp[i][j] = Math.max(dp[i][j], find(ni, nj) + 1);
        }
        return -1;
    }

    public int maximumDistance(int[][] grid) {
        int n = grid.length, m = grid[0].length;

            Arrays.fill(visited, false);

        return find(i, j);
    }
}
