import org.jsoup.Jsoup;

        String html = "<html><body>Hello <b>World!</b></body></html>";

        Document doc = Jsoup.parse(html);

        System.out.println("Parsed HTML:");

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class Main {
    public static void main(String[] args) {
        String html = "<html><body>Hello <b>World!</b></body></html>";

        // Create a new HTML document from scratch
        Document doc = Jsoup.parse(html);

        // Get all elements that are tags and their text content
        Elements allElements = doc.getAllElements();
        for (Element element : allElements) {
            if (!element.tagName().equals("script") && !element.tagName().equals("style")) {
                System.out.println(element.text());
            }
        }
    }
}
