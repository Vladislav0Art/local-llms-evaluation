    }

    public static void printTable(int[][] table, String[] headers) {
        System.out.println("  " + headers[0]);
        for (int i = 1; i < headers.length; i++) {
            System.out.print(headers[i] + " ");
        }
        System.out.println();

        for (int i = 0; i < table.length; i++) {
            for (int j = 0; j < table[i].length; j++) {
                if (i == j) {
                    System.out.printf("%5d", table[i][j]);
                } else {
                    System.out.print("%8s ", headers[j] + " ");
                }
            }

            System.out.println();
        }

public class MyTest {

            throw new IllegalArgumentException("Side length cannot be negative");
        }
        double perimeter = 2 * Math.PI * sideLength;
        return (int) perimeter;
    }

    public static String printHelloName(String name) {
        return "Hello, " + name + "!";
    }

    public static int calculateSquare(int number) {
        if (number < 0) {
            throw new IllegalArgumentException("Number cannot be negative");
        }
        double square = Math.pow(number, 2);
        return (int) square;
    }

    public static void main(String[] args) {
        System.out.println(printHelloWorld());
        System.out.println(calculatePerimeter(5));
        int perimeter = calculatePerimeter(-1);
        if (perimeter != -inf) {
            System.out.println("Actual Perimeter: " + perimeter);
        } else {
            System.out.println("Error calculating perimeter");
        }

        try {
            calculateSquare(-1);
        } catch (IllegalArgumentException e) {
            System.out.println(e.getMessage());
        }
        int square = calculateSquare(10);
        if (square != 100) {
            System.out.println("Actual Square: " + square);
        } else {
            System.out.println("Error calculating square");
        }

        printHelloName("John");

        int sideLength = 5;
        double perimeter = calculatePerimeter(sideLength);
        System.out.println("Expected Perimeter: " + (2 * Math.PI * sideLength));
        if (perimeter == (2 * Math.PI * sideLength)) {
            System.out.println("Actual Perimeter is correct");
        } else {
            System.out.println("Error in calculation");
        }

        int number = 10;
        double square = calculateSquare(number);
        if (square != 100) {
            System.out.println("Expected Square: " + 100);
        } else {

        }
    }
}
