package ch.jalu.configme.configurationdata;

import ch.jalu.configme.SettingsHolder;
import org.jetbrains.annotations.NotNull;
import org.junit.jupiter.api.Test;
import org.mockito.Mockito;

public class CommentsConfigurationTest {

    @Test
    public void testConstructor_MultiplePaths() {
        Map<String, List<String>> comments = new HashMap<>();
        comments.put("path1", Collections.singletonList("line1"));
        comments.put("path2", Collections.singletonList("line2"));

        CommentsConfiguration configuration = new CommentsConfiguration(comments);
        Mockito.when(SettingsHolder.registerComments(anyString())).thenReturn(true);

        configuration.setComment("path1", "line3");
        Mockito.verify(SettingsHolder.class).registerComments(anyString());
    }

    @Test
    public void testConstructor_SinglePath() {
        Map<String, List<String>> comments = new HashMap<>();
        comments.put("path", Collections.singletonList("line1"));

        CommentsConfiguration configuration = new CommentsConfiguration(comments);
        Mockito.when(SettingsHolder.registerComments(anyString())).thenReturn(true);

        configuration.setComment("path", "line2");
        Mockito.verify(SettingsHolder.class).registerComments(anyString());
    }

    @Test
    public void testGetAllComments_SinglePath() {
        Map<String, List<String>> comments = new HashMap<>();
        comments.put("path1", Collections.singletonList("line1"));

        CommentsConfiguration configuration = new CommentsConfiguration(comments);
        Mockito.when(SettingsHolder.registerComments(anyString())).thenReturn(true);

        Mockito.verifyAll(() -> {
            @NotNull @UnmodifiableView Map<String, @UnmodifiableView List<String>> result = configuration.getAllComments();
            MockitoassertThat(result.get("path1"), @NotNull @UnmodifiableView List<String>()::new);
        });
    }

    @Test
    public void testGetAllComments_SinglePathNoComment() {
        CommentsConfiguration configuration = new CommentsConfiguration(new HashMap<>());
        Mockito.when(SettingsHolder.registerComments(anyString())).thenReturn(true);

        Mockito.verifyAll(() -> {
            @NotNull @UnmodifiableView Map<String, @UnmodifiableView List<String>> result = configuration.getAllComments();
            MockitoassertThat(result.get("path1"), @NotNull @UnmodifiableView List<String>()::new);
        });
    }
}
