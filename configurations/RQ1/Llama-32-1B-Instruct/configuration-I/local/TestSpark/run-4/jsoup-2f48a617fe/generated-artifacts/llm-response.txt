        assertEquals("This is a sample comment.", comment.getData());
    }

    public void testSetData() {
        Comment comment = new Comment("#comment");
        comment.setData("New comment content.");
        assertEquals("New comment content.", comment.getData());
    }

    public void testOuterHtmlHead() throws IOException {
        String data = "#Comment is a sample comment.";
        Comment comment = new Comment(data);
        Appendable accum = new StringBuilder();
        comment.outerHtmlHead(accum, 0, null).append("\n").append(accum.toString()).append("\n");
        assertEquals("<!--\n#Comment is a sample comment.\n-->", accum.toString());
    }

    @Test
    public void testOuterHtmlTail() {
        String data = "#Comment is a sample comment.";
        Comment comment = new Comment(data);
        Appendable accum = new StringBuilder();
        comment.outerHtmlTail(accum, 0, null).append("\n").append(accum.toString()).append("\n");
        assertEquals("<!--\n#Comment is a sample comment.\n-->", accum.toString());
    }

    @Test
    public void testIsXmlDeclaration() {
        String data = "This is a sample XML Declaration.";
        Comment comment = new Comment(data);
        assertTrue(comment.isXmlDeclaration());
    }

    public void testAsXmlDeclaration() throws Exception {
        String data = "#Comment is a sample comment.";
        Comment comment = new Comment(data);
        XmlDeclaration decl = comment.asXmlDeclaration();

import org.jsoup.nodes.Comment;
import org.jsoup.nodes.Node;
import org.jsoup.nodes.Text;

public class CommentTest {

    public static String nodeName(String comment) {
        return comment.nodeName();
    }

    @Test
    public void testnodeName() {
        assertEquals("comment", nodeName("<!-- This is the first part of a comment --><p>This is a sample text.</p>"));
    }

    public static String getData(Node node) {
        if (node instanceof Text) {
            return ((Text) node).getData();
        } else {
            return null;
        }
    }

    @Test
    public void testgetData() {
        Node node = new Tag("", 1);
        node.appendChild(new Text(""));
        assertEquals("This is a sample comment.", getData(node));
    }

    public static class Tag implements Comparable<Tag> {
        private String name;

        public Tag(String name) {
            this.name = name;
        }

        @Override
        public int compareTo(Tag other) {
            return this.name.compareTo(other.name);
        }
    }

    @Test
    public void testXmlDeclaration() {
        Node node = new Tag("", 1);
        node.appendChild(new Text(""));
        assertEquals(false, node.isXmlDeclaration());
    }
}
