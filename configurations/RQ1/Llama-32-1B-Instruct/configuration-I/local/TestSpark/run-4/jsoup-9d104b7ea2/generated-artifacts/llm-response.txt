        System.out.println(output);

            return "";
        }
        return doc.toString();
    }

    public void appendTo(Appendable accum) {
        super.appendTo(accum);
    }

    public void splitText(int count, Appendable output) {

            throw new IllegalArgumentException("Output must be a string");

        int index = 0;

            append(output);
            if (index + i > output.length() - 1) {
                break;
            }
            appendTo(output, index);
            index += i;
        }
    }

    @Override
    public void outerHtmlAppend(Appendable accum, int depth) {
        super.outerHtmlAppend(accum, depth);
    }

    @Override
    public String toString() {
        return getWholeText();
    }

    @Override
    protected Object clone() throws CloneNotSupportedException {
        if (this == super.clone()) {
            throw new UnsupportedOperationException("clone());
        }
        TextNodeImpl result = (TextNodeImpl) super.clone();
        result.value = StringUtil.normaliseWhitespace(getWholeText());
        return result;
    }
}

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class TextNodeImpl extends LeafNode {
    private final String value;
    private final StringBuilder builder = new StringBuilder();

    public TextNodeImpl(String text) {
        super(text);
        this.value = StringUtil.normaliseWhitespace(getWholeText());
    }

    @Override
    public String nodeName() {
        return "#text";
    }

    public static Document createDocument() {
        return MockDocument.create();
    }

    public static void outerHtmlHead(Appendable accum, int depth) {
        Element root = TextNodeImpl.outerHtmlHead(accum, depth);
        // Assuming we have a method to get the output of the document
        String output = getOutput(root);
        System.out.println(output);
    }

    public static void outerHtmlTail(Appendable accum, int depth) {
        Document doc = TextNodeImpl.outerHtmlTail(accum, depth);
        // Assuming we have a method to get the output of the document
        String output = getOutput(doc);
        System.out.println(output);
    }

    public static boolean clone(TextNode node) {
        return CloneUtil.clone(node);
    }

    private static Document getOutput(Element element) {
        if (element == null || !element.exists()) {
            return "";
        }
        for (Element child : element.children()) {
            String output = getOutput(child);
            builder.append(output).append("\n");
        }
        return builder.toString();
    }

    @Override
    public void appendTo(Appendable accum) {
        super.appendTo(accum);
    }

    @Override
    protected Object clone() throws CloneNotSupportedException {
        if (this == super.clone()) {
            throw new UnsupportedOperationException("clone());
        }
        TextNodeImpl result = (TextNodeImpl) super.clone();
        result.value = value;
        return result;
    }

    public static void main(String[] args) {
        String text = "Hello World!";
        TextNode node = new TextNodeImpl(text);
        System.out.println(node.toString());

        TextNode clonedNode = clone(node);
        System.out.println(clonedNode.toString());
    }
}
