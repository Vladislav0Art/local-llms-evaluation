import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class TagParserTest {

    private Document document;

    @Before
    public void setup() {
        document = Jsoup.parse("<div><base href=\"/base\"><object type=\"application/x-srf1/xhtml\"><img src=\"image.jpg\"></object></div>");
    }

    public boolean validateTag(String tag) {
        return validateAttribute(tag);
    }

    private boolean validateAttribute(String attribute) {
        Elements tags = document.getElementsByAttributeValue("class", attribute);
        if (tags.isEmpty()) {
            return false;
        }
        Element firstTag = tags.first();
        if (!firstTag.hasAttr("style") || !firstTag.attr("style").contains(attribute)) {
            return false;
        }
        return true;
    }

    public void testValidateAttribute() {
        System.out.println(validateTag("class"));
        System.out.println(validateTag("style"));
        System.out.println(validateTag("script"));
        System.out.println(validateTag(""));
    }

    public void testGetAttribute(String element, String attribute) {
        Element tag = document.getElementById(element);
        if (tag != null && tag.hasAttr(attribute)) {
            System.out.println(tag.attr(attribute));
        } else {
            System.out.println("Attribute not found");
        }
    }

    public void testGetAttributeValue(String element, String attribute) {
        Element tag = document.getElementById(element);
        if (tag != null) {
            String value = tag.attr(attribute);
            System.out.println(value);
        } else {
            System.out.println("Element not found");
        }
    }

    public static void main(String[] args) {
        TagParserTest test = new TagParserTest();
        test.testValidateAttribute();
        test.testGetAttribute("img", "src");
        test.testGetAttributeValue("div", "class");
    }
}

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class TagParserTest {

    private Document document;

    @Before
    public void setup() {
        String html = "<html><head></head><body><div><base href=\"/base\"><object type=\"application/x-srf1/xhtml\"><img src=\"image.jpg\"></object></div></body></html>";
        document = Jsoup.parse(html);
    }

    public boolean validateTag(String tag) {
        return validateAttribute(tag, "class");
    }

    private boolean validateAttribute(String attribute, String tag) {
        Elements tags = document.getElementsByTag(tag);
        if (tags.isEmpty()) {
            return false;
        }
        Element firstTag = tags.first();
        if (!firstTag.hasAttr(attribute)) {
            return false;
        }
        return true;
    }

    public void testValidateAttribute(String tag, String attribute) {
        System.out.println(validateTag(tag));
        System.out.println(validateAttribute(attribute, tag));
        System.out.println(validateAttribute("", tag));
    }

    public void testGetAttribute(String element, String attribute) {
        Element tag = document.getElementById(element);
        if (tag != null && tag.hasAttr(attribute)) {
            System.out.println(tag.attr(attribute));
        } else {
            System.out.println("Attribute not found");
        }
    }

    public void testGetAttributeValue(String element, String attribute) {
        Element tag = document.getElementById(element);
        if (tag != null) {
            String value = tag.attr(attribute);
            System.out.println(value);
        } else {
            System.out.println("Element not found");
        }
    }

    public static void main(String[] args) {
        TagParserTest test = new TagParserTest();
        test.testValidateAttribute("img", "src");
        test.testGetAttribute("div", "class");
        test.testGetAttributeValue("body", "style");
    }
}
