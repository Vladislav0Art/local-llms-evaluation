import java.io.*;
import java.net.*;
import java.util.*;

public class DocumentServer {
    public static void main(String[] args) throws Exception {
        // Create a directory to store documents
        File dir = new File("documents");
        if (!dir.exists()) {
            dir.mkdir();
        }

        // List of documents
        Map<String, Document> documents = new HashMap<>();

        // Listen for GET requests on port 8080
        ServerSocket serverSocket = new ServerSocket(8080);
        Socket socket = serverSocket.accept();

        System.out.println("GET request received");

        while (true) {
            InputStream inputStream = socket.getInputStream();
            OutputStream outputStream = socket.getOutputStream();

            byte[] buffer = new byte[1024];
            int bytesRead;
            while ((bytesRead = inputStream.read(buffer)) != -1) {
                String request = new String(buffer, 0, bytesRead);
                System.out.println("Request: " + request);

                if (request.startsWith("/")) {
                    String path = request.substring(1);
                    Document document = documents.get(path);
                    if (document != null) {
                        outputStream.write(document.getTitle().getBytes());
                        outputStream.flush();
                        System.out.println("Document '" + path + "' returned.");
                    } else {
                        System.out.println("Document not found: " + path);
                    }
                } else if (request.startsWith("/add")) {
                    String[] parts = request.substring(3).split(",");
                    Document document = new Document(parts[0], parts[1]);
                    documents.put(path, document);
                    outputStream.write(document.getTitle().getBytes());
                    System.out.println("Document '" + path + "' added.");
                } else if (request.startsWith("/remove")) {
                    String[] parts = request.substring(4).split(",");
                    Document document = documents.get(parts[0]);
                    if (document != null) {
                        documents.remove(path);
                        outputStream.write(document.getTitle().getBytes());
                        System.out.println("Document '" + path + "' removed.");
                    } else {
                        System.out.println("Document not found: " + parts[0]);
                    }
                } else if (request.startsWith("/list")) {
                    List<String> list = new ArrayList<>();
                    for (Map.Entry<String, Document> entry : documents.entrySet()) {
                        list.add(entry.getKey());
                    }
                    outputStream.write(list.toString().getBytes());
                    System.out.println("Document list: " + list);
                }

                socket.close();
            }
        }
    }

    private static void createDirectory(String directory) throws IOException {
        File file = new File(directory);
        if (!file.exists()) {
            file.mkdirs();
        }
    }
}

class Document {
    private String id;
    private String title;

    public Document(String id, String title) {
        this.id = id;
        this.title = title;
    }

    public String getId() {
        return id;

        return title;
