public class OnlineStockSpanTest {

    @Test
    public void test_next_test() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        int price = 7;
        int result = onlineStockSpan.next(price);
        assertEquals(1, result);

        price = 10;
        result = onlineStockSpan.next(price);
        assertEquals(2, result);
    }

    @Test
    public void test_calculateSpans_test() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        int[] prices = {7, 9, 4, 3, 20};
        int[] expected = {1, 1, 2, 1, 1};
        int[] result = onlineStockSpan.calculateSpans(prices);
        assertEquals(expected, result);

        prices = new int[]{10, 20, 30, 40, 50, 100};
        OnlineStockSpan.ScenarioTest scenarioTest = new OnlineStockSpan.ScenarioTest(prices);
        int[] actualResult = scenarioTest.getExpectedResults();
        assertArrayEquals(new int[]{3, 5, 7, 9, 11}, result);

        prices = new int[]{50, 0};
        int[] actualResult2 = onlineStockSpan.calculateSpans(prices);
        assertEquals(1, actualResult2[0]);
    }

    public static class OnlineStockSpan.ScenarioTest {
        private int[] prices;

        public OnlineStockSpan.ScenarioTest(int[] prices) {
            this.prices = prices;
        }

        public int[] getExpectedResults() {
            List<Integer> result = new ArrayList<>();
            for (int i = 0; i < prices.length - 1; i++) {
                if (prices[i + 1] > prices[i]) {
                    result.add(i + 1);
                }
            }
            return result.toArray(new int[0]);
        }

        public int calculateSpans(int[] prices) {
            List<Integer> list = new ArrayList<>();
            for (int i = 0; i < prices.length; i++) {
                if (!list.isEmpty()) {
                    int index = list.get(list.size() - 1);
                    while (!prices[index].equals(prices[i])) {
                        index--;
                    }
                    list.add(index + 1);
                } else {
                    list.add(0);
                }
            }
            return list.stream().mapToInt(Integer::intValue).sum();
        }
    }

}
