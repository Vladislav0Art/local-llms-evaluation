import org.jsoup.nodes.Document;
import org.jsoup.nodes.TextNode;
import org.jsoup.nodes.Appendable;
import org.junit.Test;

public class TextNodeTest {

    @Test
    public void isBlank() {
        // Arrange
        String text = "   Hello World";
        Appendable parent = mock(Appendable.class);
        Document document = mock(Document.class);

        when(parent.isBlank()).thenReturn(true);

        // Act and Assert
        assertTrue(TextNode.isBlank(text));
    }

    @Test
    public void splitText() {
        // Arrange
        int offset = 10;
        String text = "Hello World";
        Appendable accum = mock(Appendable.class);
        Document document = mock(Document.class);

        when(document.isOutline()).thenReturn(true);

        // Act
        TextNode result = new TextNode().splitText(offset, accum, document, null);

        // Assert
        verify(accum).append("Hello World".substring(0, offset));
    }

    @Test
    public void stripLeadingWhitespace() {
        // Arrange
        StringBuilder sb = new StringBuilder();
        sb.append("   Hello World");

        when(parent.stripLeadingWhitespace(BuilderStringBuilder.newStringBuilder(sb))).thenReturn(false);

        // Act and Assert
        String text = TextNode.stripLeadingWhitespace(sb.toString());
        assertTrue(text.isEmpty());
    }

    @Test
    public void stripLastCharWhitespace() {
        // Arrange
        StringBuilder sb = new StringBuilder("   Hello World");
        sb.append('W');

        when(parent.lastCharIsWhitespace(BuilderStringbuilder.newStringBuilder(sb))).thenReturn(true);

        // Act and Assert
        String text = TextNode.lastCharIsWhitespace(sb.toString());
        assertTrue(text.equals("Hello"));
    }

    @Test
    public void lastCharIsWhitespace() {
        // Arrange
        StringBuilder sb = new StringBuilder("   Hello World");
        sb.append('W');

        when(parent.lastCharIsWhitespace(BuilderStringbuilder.newStringBuilder(sb))).thenReturn(true);

        // Act and Assert
        String text = TextNode.lastCharIsWhitespace(sb.toString());
        assertTrue(text.equals("Hello"));
    }

    @Test
    public void isBlankWithEmptyString() {
        // Arrange
        String text = "";

        // Act and Assert
        assertTrue(TextNode.isBlank(text));
    }
}
