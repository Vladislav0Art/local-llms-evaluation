            settings.setDefaultParser(new DefaultParser());
            settings.remove("http://jsoup.org/skip");

            TestSpark other = (TestSpark) spark.outerHtmlHead(settings).getData();

class DefaultParser implements ParseSettings.DefaultParser {
    @Override
    public Object parse(String tag) {
        if (tag.equals("comment")) {
            String comment = spark.outerHtmlHead(settings).getData();
            // Convert the comment to a string of hexadecimal IDs separated by "#" and return it.
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < comment.length(); i++) {
                char c = comment.charAt(i);
                if (c == '#') continue;
                int hexId = Integer.parseInt(c, 16);
                if (hexId > 255) hexId -= 256;
                sb.append(hexId).append(":");
            }
            return Objects.equals(comment, sb.toString());
        } else {
            return null;
        }
    }
