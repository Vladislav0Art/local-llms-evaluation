    }

    }

    public static void appendKeyValUsingStringBuilder(String urlBuilder) {

        for (String part : urlBuilder.toString().split("#")) {

```java
import java.util.HashMap;
import java.util.Map;

public class Main {
    public static void main(String[] args) throws Exception {
        String url = "https://example.com#key1=value1";
        
        Document doc = Jsoup.parse(url);
        
        System.out.println("Original URL: " + url);
        
        System.out.println("Normalized URL: " + normalizeUrl(doc, url));
        
        System.out.println("Append key-value pairs using StringBuilder");
        appendKeyValUsingStringBuilder(url);
        
        System.out.println("Append key-value pairs using JavaScript");
        appendKeyValUsingJavaScript(url);
    }
    
    public static String normalizeUrl(Document doc, String builder) throws Exception {
        // Remove any existing query string
        builder = builder.replaceFirst("(?i)^(https?:\\/\\/[^#]+)([?#][^\\"]*|[^#]*)$", "");
        
        // Create a new query string
        Map<String, String> params = new HashMap<>();
        for (String part : builder.split("#")) {
            if (!part.isEmpty()) {
                int index = part.indexOf('=');
                if (index != -1) {
                    params.put(part.substring(0, index), part.substring(index + 1));
                }
            }
        }
        
        // Create a new URL
        return url + "?" + java.net.URLEncoder.encode(String.join(";", params.entrySet()), "UTF-8");
    }

    public static void appendKeyValUsingStringBuilder(String urlBuilder) {
        StringBuilder sb = new StringBuilder();
        for (String part : urlBuilder.split("#")) {
            if (!part.isEmpty()) {
                int index = part.indexOf('=');
                if (index != -1) {
                    sb.append(part.substring(0, index)).append("=").append(value()).append("&");
                } else {
                    sb.append(part).append("=").append(value());
                }
            }
        }
        
        String normalizedUrl = sb.substring(0, sb.length() - 1);
        System.out.println("Normalized URL: " + normalizedUrl);
    }

    public static void appendKeyValUsingJavaScript(String urlBuilder) throws Exception {
        document.body.innerHTML += "<script>var builder = new URL('" + urlBuilder + "');</script>";
        
        Document doc = Jsoup.parse(builder.toString());
        
        Element element = doc.select(".url-builder").first();
        System.out.println("Normalized URL: " + element.attr("value"));
    }

    public static String value() {
        return "/api/v1/user";
    }
}
```

This program uses the JavaScript `URL` API to normalize a URL by removing any existing query string and creating a new query string with key-value pairs. It also demonstrates how to append key-value pairs using a `StringBuilder` and JavaScript. The `value()` method is used to get the base path of the API, which is "/api/v1/user".
