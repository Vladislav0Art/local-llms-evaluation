        InetAddress address = socket.getInetAddress();

    }

        Socket socket = new Socket("http://example.com", 80);
        return socket.getRemotePort();
    }

    public static String getProtocol() throws Exception {
        Socket socket = new Socket("http://example.com");
        Enumeration<Protocol> protocols = socket.getAvailableProtocols();
        while (protocols.hasMoreElements()) {
            Protocol protocol = protocols.nextElement();
            if (protocol.getName().equals("HTTP")) {
                return "HTTP";
            }
        }
        throw new Exception("Unknown protocol");
    }

    public static boolean isFileSystemPath() throws Exception {
        return new File(new URL("https://example.com").toString()).isFile();
    }

    public static boolean hasScheme() throws Exception {
        String[] schemes = {"http", "https"};
        for (String scheme : schemes) {
            if (new URL("https://example.com").getProtocol().equals(scheme)) {
                return true;
            }
        }
        return false;
    }

    public static boolean hasHostname() throws Exception {
        String[] hosts = {new URL("http://example.com").toString()};
        for (String host : hosts) {
            if (!host.startsWith("http")) {
                return true;
            }
        }
        return false;
    }

    public static String getPath() throws Exception {
        Socket socket = new Socket("http://example.com");

        while ((line = in.readLine()) != null) {
            if (line.startsWith("/")) {
                return line.substring(1);
            }
        }
        throw new Exception("Unknown path");
    }

    public static boolean isSchemeLess() throws Exception {

                return true;
