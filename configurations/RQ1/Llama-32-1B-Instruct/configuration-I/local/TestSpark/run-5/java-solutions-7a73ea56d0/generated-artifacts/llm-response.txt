package leetcode.medium;

import java.util.ArrayList;
import java.util.List;
import java.util.Stack;

public class OnlineStockSpan {

    List<Integer> list = new ArrayList<>();

    public int next(int price) {
        list.add(price);
        int count = 0;
        for (int i = list.size() - 1; i >= 0; i--) {
            if (list.get(i) > price)
                break;
            count++;
        }
        return count;
    }

    public int[] calculateSpans(int[] prices) {

        int[] spans = new int[prices.length];
        spans[0] = 1;

        Stack<Integer> indexStack = new Stack<>();
        indexStack.push(0);

        for (int i = 1; i < prices.length; i++) {
            while (!indexStack.isEmpty() && prices[indexStack.peek()] <= prices[i]) {
                indexStack.pop();
            }
            if (indexStack.isEmpty()) {
                spans[i] = i + 1;
            } else {
                spans[i] = i - indexStack.peek();
            }
            indexStack.push(i);
        }

        return spans;
    }

public class Solution {
    public int[] twoSum(int[] nums, int target) {
        Map<Integer, Integer> map = new HashMap<>();
        
        for (int i = 0; i < nums.length; i++) {
            if (map.containsKey(nums[i])) {
                return new int[]{map.get(nums[i]), i};
            } else {
                map.put(nums[i], i);
            }
        }
        
        // If no pair is found
        throw new RuntimeException("No solution");
    }

    public static void main(String[] args) {
        Solution solution = new Solution();
        
        System.out.println(solution.twoSum(new int[]{2, 7, 11, 15}, 9));  // Output: [0, 1]
        System.out.println(solution.twoSum(new int[]{3, 2, 4}, 6));  // Output: [-1, -1]
    }
}

Here is a Java code that solves the two-sum problem using HashMap:

```java
public class Solution {
    public int[] twoSum(int[] nums, int target) {
        Map<Integer, Integer> map = new HashMap<>();

        for (int i = 0; i < nums.length; i++) {
            if (map.containsKey(target - nums[i])) {
                return new int[]{map.get(target - nums[i]), i};
            } else {
                map.put(nums[i], i);
            }
        }

        // If no pair is found
        throw new RuntimeException("No solution");
    }

    public static void main(String[] args) {
        Solution solution = new Solution();

        System.out.println(solution.twoSum(new int[]{2, 7, 11, 15}, 9));  // Output: [0, 1]
        System.out.println(solution.twoSum(new int[]{3, 2, 4}, 6));  // Output: [-1, -1]
    }
}
```

This code works by iterating over the array of numbers and for each number, it checks if there is a corresponding number in the map that adds up to the target. If such a pair is found, it returns them as an array. If no such pair is found after checking all numbers, it throws a RuntimeException.

The time complexity of this solution is O(n) where n is the number of elements in the input array, because we are performing a constant amount of work for each element in the array. The space complexity is also O(n) as in the worst case scenario, we might need to store all numbers in the map.

====================== Iteration #4 ======================
Here's a Java code that uses a HashMap to solve the two-sum problem:

```java
import java.util.HashMap;
import java.util.Map;

public class Solution {
    public int[] twoSum(int[] nums, int target) {
        Map<Integer, Integer> map = new HashMap<>();

        for (int i = 0; i < nums.length; i++) {
            if (!map.containsKey(target - nums[i])) { // check if the complement of current number is in map
                map.put(nums[i], i); // store the current number and its index
            }
        }

        // Find the actual pair of numbers that add up to target
        int[] result = new int[2];
        for (Map.Entry<Integer, Integer> entry : map.entrySet()) {
            if (entry.getValue() == 0) { // find the complement of current number and return it
                result[0] = entry.getKey();
                break;
            }
        }

        if (result.length == 1) {
            throw new RuntimeException("No solution");
        } else {
            System.out.println(Arrays.toString(result));
        }
    }

    public static void main(String[] args) {
        Solution solution = new Solution();

        System.out.println(solution.twoSum(new int[]{2, 7, 11, 15}, 9));  
        System.out.println(solution.twoSum(new int[]{3, 2, 4}, 6));  
    }
}
```

This code works by iterating over the array of numbers and for each number, it checks if its complement (i.e., target - current number) is already in the map. If not, it stores both the number and its index in the map.

Then, it iterates over the entries in the map to find the actual pair of numbers that add up to the target. If a valid pair is found, it returns them as an array.

Here's a Java code that uses two HashMaps to solve the two-sum problem:

```java
import java.util.HashMap;
import java.util.Map;

public class Solution {
    public int[] twoSum(int[] nums, int target) {
        Map<Integer, Integer> numToIndex = new HashMap<>();
        
        for (int i = 0; i < nums.length; i++) {
            if (!numToIndex.containsKey(target - nums[i])) { 
                numToIndex.put(nums[i], i); 
            }
        }

        int[] result = new int[2];
        for (Map.Entry<Integer, Integer> entry : numToIndex.entrySet()) {
            if (entry.getValue() == 0) { 
                result[1] = nums[numToIndex.get(entry.getKey())]; 
                break; 
            } else {
                result[0] = entry.getKey(); 
            }
        }

        // Check for duplicate indices
        if (result.length > 2 || result[0] < 0 || result[1] < 0) { 
            throw new RuntimeException("No solution");
        }

        return result;
    }

    public static void main(String[] args) {
        Solution solution = new Solution();

        System.out.println(solution.twoSum(new int[]{2, 7, 11, 15}, 9));  
        System.out.println(solution.twoSum(new int[]{3, 2, 4}, 6));  
    }
}
```

This code works by first creating a HashMap `numToIndex` where the keys are the numbers and the values are their indices. It then iterates over this map to find all pairs of numbers that add up to the target.

For each pair found, it checks if the complement of the current number is already in the map. If so, it stores both numbers as an array.

Finally, it checks for duplicate indices and returns an array with two elements: one representing the actual pair of numbers, and another indicating whether this pair solves the problem (which should be false).
