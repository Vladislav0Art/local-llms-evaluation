import org.junit.Test;
import static org.junit.Assert.*;

public class NodeUtils {

    public static String normalizeWhitespace(String text) {
        text = StringUtil.normalizeWhitespace(text);
        return text;
    }

    public static String stripLeadingWhitespace(StringBuilder sb) {
        if (sb == null || sb.length() == 0) {
            return "";
        }
        return sb.toString().trim();
    }

    public static boolean lastCharIsWhitespace(StringBuilder sb) {
        if (sb == null || sb.length() == 0) {
            return false;
        }
        return sb.charAt(sb.length() - 1) == ' ';
    }

    @Test
    public void testSplitText_1() {
        TextNode node = new TextNode("<p>Hello <span>World!</span></p>");
        StringBuilder result = normalizeWhitespace(node.getText());
        assertEquals("Hello World!", result.toString());
    }

    @Test
    public void testGetWholeText_2() {
        TextNode node = new TextNode("<p>Hello <span>World!</span></p>");
        assertEquals("", stripLeadingWhitespace(node.getText().toString()));
    }

import org.junit.Test;
import static org.junit.Assert.*;

public class NodeUtils {

    public static String normalizeWhitespace(String text) {
        return text.trim();
    }

    public static StringBuilder stripLeadingWhitespace(StringBuilder sb) {
        if (sb == null || sb.length() == 0) {
            return new StringBuilder();
        }
        int index = sb.indexOfFirst(" ");
        if (index != -1) {
            return sb.substring(index + 1);
        } else {
            return sb.toString();
        }
    }

    @Test
    public void testSplitText_1() {
        TextNode node = new TextNode("<p>Hello <span>World!</span></p>");
        String result = normalizeWhitespace(node.getText());
        assertEquals("Hello World!", result);
    }

    @Test
    public void testGetWholeText_2() {
        TextNode node = new TextNode("<p>Hello <span>World!</span></p>");
        StringBuilder result = stripLeadingWhitespace(node.getText().toString());
        assertEquals("", result.toString());
    }
}
