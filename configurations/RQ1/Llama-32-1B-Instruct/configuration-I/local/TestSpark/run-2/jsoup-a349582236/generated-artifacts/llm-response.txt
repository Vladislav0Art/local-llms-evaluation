/**

public abstract class Tag {
    private String name;
    private boolean open;
    private int level;

    public Tag(String name) {
        this.name = name;
        this.open = false;
        this.level = 0;
    }

    /**
     * Returns the tag name.
     *
     * @return the tag name
     */
    public String getName() {
        return name;
    }

    /**
     * Sets the tag name.
     *
     * @param name the new tag name
     */
    public void setName(String name) {
        this.name = name;
    }

    /**
     * Returns whether the tag is open or closed.
     *
     * @return true if the tag is open, false otherwise
     */
    public boolean isOpen() {
        return open;
    }

    /**
     * Sets whether the tag is open or closed.
     *
     * @param open whether the tag is open or closed
     */
    public void setOpen(boolean open) {
        this.open = open;
    }

    /**
     * Returns the opening level of the tag.
     *
     * @return the opening level
     */
    public int getLevel() {
        return level;
    }

    /**
     * Sets the opening level of the tag.
     *
     * @param level the new opening level
     */
    public void setLevel(int level) {
        this.level = level;
    }

    /**
     * Returns a string representation of the tag.
     *
     * @return a string representation of the tag
     */
    @Override
    public String toString() {
        if (open) {
            return "<" + name + ">";
        } else {
            return "</" + name + ">";
        }
    }

    /**
     * Recursively adds indentation to the tag.
     *
     * @param indent the number of spaces to use for indentation
     */
    public void addIndent(int indent) {
        String indentStr = "  ".repeat(indent);
        if (open) {
            System.out.print(indentStr + "<");
        } else {
            System.out.print(indentStr + "</");
        }
    }

    /**
     * Prints the tag to the console.
     */
    public void print() {
        addIndent(getLevel());
        System.out.println(name);
    }

    /**
     * Returns a string representation of the tag with HTML tags.
     *
     * @return a string representation of the tag
     */
    @Override
    public String toStringHtml() {
        return "<" + name;
    }
}
