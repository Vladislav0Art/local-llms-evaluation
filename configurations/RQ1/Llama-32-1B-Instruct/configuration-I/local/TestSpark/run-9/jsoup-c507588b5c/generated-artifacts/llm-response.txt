public class TextNodeImpl {

}

public static void textNodeBlankCheck(TextNode node) {
    if (!node.isBlank()) {
        throw new RuntimeException("Text node should be blank");
    }
}

public static void textNodeSplitCheck(TextNode node, int offset) {
    if (offset > 0 && offset < node.getText().length()) {
        throw new RuntimeException("Invalid split offset");
    }
}

public class TestTextNode {

    @Test
    public void testBlankText() {
        TextNode node = new TextNode("");
        textNodeBlankCheck(node);
    }

    @Test
    public void testSplitText() {
        TextNode node = new TextNode("<a></a>");
        int offset = 0;
        textNodeSplitCheck(node, offset);
        offset = 4;
        node = node.splitText(offset);
        textNodeSplitCheck(node, offset - 1);
    }

    @Test
    public void testCreateFromEncoded() {
        TextNode node = TestTextNode.createFromEncoded("test");
        textNodeBlankCheck(node);
        textNodeSplitCheck(node, 0);
        textNodeSplitCheck(node, 10);
    }
}

import org.jsoup.nodes.Document;
import static org.junit.Assert.*;
import org.jsoup.nodes.Element;

public class TextNodeTest {

    public static void main(String[] args) {
        TestTextNode testTextNode = new TestTextNode();
        System.out.println("Text node is blank: " + testTextNode.isBlank());
        
        Document document = new Document();
        String text = "<a></a>";
        Element element = document.createElement("a");
        element.text(text);
        document.appendChild(element);
        
        testTextNode = new TextNode(document);
        assertEquals(text, testTextNode.getText());

        TestTextNode result1 = testTextNode.splitText(0);
        assertEquals(text, result1.getText());
        
        TestTextNode result2 = testTextNode.splitText(4);
        assertEquals(text.substring(0, 4), result2.getText());

        document.clear();
        text = "test";
        element = document.createElement("a");
        element.text(text);
        document.appendChild(element);

        testTextNode = new TextNode(document);
        assertEquals(text, testTextNode.getText());
    }

    public static class TestTextNode {
        private Document document;

        public void createDocument() {
            this.document = new Document();
        }

        public String getText() {
            return this.document.text();
        }

        public void appendText(String text) {
            Element element = document.createElement("a");
            element.text(text);
            document.appendChild(element);
        }
    }

}

import org.jsoup.nodes.Document;
import static org.junit.Assert.*;

public class TextNodeTest {

    public void testBlankText() {
        TestTextNode node = new TestTextNode();
        assertNotNull(node);
        assertFalse(node.isBlank());
    }

    public void testSplitText() {
        Document doc = new Document();
        String text = "<a></a>";
        doc.appendChild(doc.createTextNode(text));
        TextNode result1 = new TestTextNode().splitText(0);
        assertEquals(text, result1.getText());
        
        TestTextNode result2 = new TestTextNode().splitText(4);
        assertEquals(text.substring(0, 4), result2.getText());

        String textToAppend = "test";
        doc.clear();
        testTextNode = new TextNode(doc);
        testTextNode.appendText(textToAppend);

        Document finalDoc = doc.clone();
        
        document = new Document();
        String finalText = "<a>" + textToAppend + "</a>";
        Element element = document.createElement("a");
        element.text(finalText);
        finalDoc.appendChild(element);

        testTextNode = new TextNode(document);
        assertEquals(textToAppend, testTextNode.getText());
    }

    public void testCreateDocument() {
        TestTextNode testTextNode = new TestTextNode();
        
        testTextNode.createDocument();

        Document document = new Document();
        String text = "test";
        testTextNode.appendText(text);

        assertTrue(document.getElementById("a"));
    }
}
