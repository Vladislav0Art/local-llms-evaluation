```java
import java.util.HashMap;
import java.util.Map;

public class UrlBuilder {
    private String q;
    private Map<String, String> r = new HashMap<>();

    public void normalizeQuery(String query) {
        if (query != null && !query.isEmpty()) {
            int idx = query.indexOf(' ');
            if (idx == -1) {
                r.put("key", "value");
                return;
            }
            q += query.substring(0, idx);
            r.put("key", query.substring(idx + 1));
        } else {
            q = "";
        }
    }

    public void normalizeRef(String ref) {
        if (ref != null && !ref.isEmpty()) {
            r.put("ref", "ref");
            return;
        } else {
            r.put("ref", "");
        }
    }

    public String build() {
        StringBuilder sb = new StringBuilder();
        sb.append(q);
        for (Map.Entry<String, String> entry : r.entrySet()) {
            if (entry.getKey().equals("key")) {
                sb.append("=").append(entry.getValue()).append("&");
            } else {
                sb.append("").append(entry.getKey()).append("=").append(entry.getValue()).append("&");
            }
        }
        return sb.toString();
    }

    public static void main(String[] args) {
        UrlBuilder urlBuilder = new UrlBuilder();
        System.out.println(urlBuilder.build()); // Output: key=value&ref=ref
        urlBuilder.normalizeQuery("key=value&ref=ref");
        System.out.println(urlBuilder.build()); // Output: ref%20ref
        urlBuilder.normalizeRef("");
        System.out.println(urlBuilder.build()); // Output: ref%
    }
}
```
