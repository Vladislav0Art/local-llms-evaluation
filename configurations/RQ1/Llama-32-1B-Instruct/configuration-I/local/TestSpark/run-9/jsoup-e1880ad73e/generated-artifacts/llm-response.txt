        Collections.sort(query, Comparator.comparing(UrlField::getValue));
        for (String str : query) {
            String encodedStr = getEncodedKey(str);

    private static List<String> encodePath(String path) {
        Map<String, String> urlMap = new HashMap<>();
        for (String part : path.split("/")) {
            if (!part.startsWith("#")) {
                urlMap.put(part, null);
            } else {
                urlMap.remove(part);
            }
        }
        return new ArrayList<>(urlMap.values());
    }

    private static String decodePath(String path) {
        Map<String, String> urlMap = new HashMap<>();
        for (String part : path.split("/")) {
            if (!part.startsWith("#")) {
                urlMap.put(part, null);
            } else {
                urlMap.remove(part);
            }
        }
        return urlMap.values().stream()
                .map(UrlField::getValue)
                .collect(Collectors.joining("/"));
    }

    private static String getEncodedKey(String str) {
        Set<String> keyParts = new HashSet<>();
        for (int i = 0; i < str.length(); i++) {
            if (str.charAt(i) == '@') {
                break;
            }
            keyParts.add(str.substring(0, i + 1));
        }
        return String.join("@", keyParts);
    }

    private static String decodePart(String encodedStr) {
        Set<String> decodedKeyParts = new HashSet<>();
        for (int i = 1; i <= encodedStr.length(); i++) {
            if (!encodedStr.startsWith("#")) {
                decodedKeyParts.add(encodedStr.substring(0, i));
            } else {
                decodedKeyParts.add(encodedStr.substring(i, encodedStr.length()));
            }
        }
        return String.join("=", decodedKeyParts);
    }

    public static class UrlField implements Comparable<UrlField> {
        private final int valueLength;
        private final boolean hasValue;

        public UrlField(int valueLength, boolean hasValue) {
            this.valueLength = valueLength;
            this.hasValue = hasValue;
        }

        @Override
        public int compareTo(UrlField other) {
            return Integer.compare(other.valueLength, valueLength);
        }
    }
}

        assertEquals("https://test.example.co.uk/path?query=value#anchor", urlBuilder.build());

    public void testNormalizeInputWithQuery() {
        String query = "param1=value1&another=2";
        URI uri = new URI("http://example.com/path");
        List<String> params = parseQueryString(uri.getQuery());
        Set<String> encodedParams = new HashSet<>();
        for (String param : params) {
            String[] parts = param.split("=");
            if (parts.length == 2) {
                encodedParams.add(decodeParam(parts[0], StandardCharsets.UTF_8));
            }
        }
        StringBuilder builder = new StringBuilder();
        for (int i = 0; i < encodedParams.size(); i++) {
            if (i > 0) {
                builder.append("&");
            }
            builder.append(encodedParams.get(i));
        }
        assertEquals("https://example.com/path?param1=value1&another=2", urlBuilder.build());
    }

    private static List<String> parseQueryString(String query) {
        String[] parts = query.split("&");
        Set<String> params = new HashSet<>();
        for (String part : parts) {
            if (!part.isEmpty()) {
                int equalsIndex = part.indexOf("=");
                if (equalsIndex > 0 && equalsIndex < part.length() - 1) {
                    params.add(decodeParam(part.substring(0, equalsIndex), StandardCharsets.UTF_8));
                    params.add(decodeParam(part.substring(equalsIndex + 1, part.length()), StandardCharsets.UTF_8));
                }
            }
        }
        return new ArrayList<>(params);
    }

    private static String decodeParam(String param, char[] charset) {
        Set<String> decodedParams = new HashSet<>();
        for (int i = 0; i < param.length(); i++) {
            if (param.charAt(i) == '@') {
                break;
            }
            decodedParams.add(param.substring(0, i + 1), charset);
        }
        return String.join("=", decodedParams);
    }

    public static class UrlField implements Comparable<UrlField> {
        private final int valueLength;

        public UrlField(int valueLength) {
            this.valueLength = valueLength;
        }

        @Override
        public int compareTo(UrlField other) {
            if (valueLength != other.valueLength) {
                return Integer.compare(valueLength, other.valueLength);
            } else {
                return 0;
            }
        }
    }

    private static String decodeParam(String param, byte[] charset) {
        Set<String> decodedParams = new HashSet<>();
        for (int i = 0; i < param.length(); i++) {
            if (param.charAt(i) == '@') {
                break;
            }
            decodedParams.add(param.substring(0, i + 1), charset);
        }
        return String.join("=", decodedParams);
    }

    public static class UrlBuilder {
        private final URI uri;

        public UrlBuilder(URI uri) {
            this.uri = uri;
        }

        public String build() {
            StringBuilder builder = new StringBuilder();
            for (String part : uri.getPath().split("/")) {
                if (!part.isEmpty()) {
                    try {
                        builder.append(part).append("?");
                    } catch (Exception e) {
                        throw new RuntimeException(e);
                    }
                }
            }
            return builder.toString();
        }

        public String normalizeHost(String host) {
            try {
                String[] netlocParts = host.split(":");
                if (netlocParts.length == 2 && netlocParts[0].equals("www.")) {
                    host = "http://" + netlocParts[1];
                } else {
                    host = new URI(host).getHost();
                }
            } catch (Exception e) {
                // Ignore
            }
            return host;
        }
    }

    public static void main(String[] args) {
        UrlBuilder urlBuilder = new UrlBuilder(new URI("http://example.com/path"));
        System.out.println(urlBuilder.build());
        urlBuilder.setQuery("param1=value1&another=2");
        System.out.println(urlBuilder.build());
    }
}
