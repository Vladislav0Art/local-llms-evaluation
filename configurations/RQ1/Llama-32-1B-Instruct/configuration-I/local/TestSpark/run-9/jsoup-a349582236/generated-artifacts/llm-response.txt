import java.util.Scanner;

public class InputTag extends Input {
    public InputTag(String tagName) {
        super(tagName);
    }
}

class KeygenTag extends InputTag {
    private String secret;
    private String input;

    public KeygenTag(String tagName) {
        super(tagName);
        this.secret = "secretkey";
    }

    @Override
    public String getTagName() {
        return super.getTagName() + ":SecretKey:" + input;
    }
}

class ButtonTag extends InputTag {
    private String buttonAction;

    public ButtonTag(String tagName) {
        super(tagName);
    }

    @Override
    public String getTagName() {
        return super.getTagName() + ":ButtonAction";
    }
}

public class TestInputTags {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.println("Enter the tag name for an input:");
        String tagName = scanner.nextLine();

        InputTag inputTag = new InputTag(tagName);
        KeygenKeygen keygenKeygen = new KeygenTag(tagName);
        Button buttonButton = new ButtonTag(tagName);

        System.out.println("Enter the secret key for a keygen tag (press enter to skip):");
        String secret = scanner.nextLine();

        if (!secret.isEmpty()) {
            keygenKeygen.secret = secret;
        }

        System.out.println("Enter the input value for an input tag:");
        String inputValue = scanner.nextLine();
        inputTag.inputValue = inputValue;

        // Verify the tag names
        System.out.println(inputTag.getTagName());
        System.out.println(keygenKeygen.getTagName());

        Button testButton = new Button("button");
        buttonTestAgain = new Button("Click Here!");

        System.out.println(buttonTestAgain.getTagName());
    }
}

class Input {
    private String tagName;

    public Input(String tagName) {
        this.tagName = tagName;
    }

    public String getTagName() {
        return tagName;
    }
}

This program creates three types of tags: `Input`, `KeygenTag`, and `Button`. The `KeygenTag` class includes a secret key for the tag, while the `Input` class simply contains a tag name. The `Button` class also has a button action.

When you run this program, it prompts you to enter the tag names and input values. If you don't want to add an input value or secret key, you can press enter. Otherwise, it will create the corresponding tags with the provided information.
