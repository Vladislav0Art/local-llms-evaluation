    }

    public void splitText_returnsNewNode_withTextAfterOffset() {
        String text = "Hello World!";
        TextNode node = createTextNode(splitText(-1));
        assertEquals("World!", node.text());

    private void assertTextIsBlank(Document document) {
        for (TextNode child : document.childNodes()) {
            if (!child.text().isEmpty()) return;
        }
        assertTrue(document.isDocumentEmpty());
    }

    private String stripLeadingWhitespace(StringBuilder sb) {
        while (sb.length() > 0 && sb.charAt(sb.length() - 1) == ' ') {
            sb.deleteCharAt(sb.length() - 1);
        }
        return sb.toString();
    }

    private boolean lastCharIsWhitespace(StringBuilder sb) {
        if (sb != null && !sb.isEmpty()) {
            if (sb.charAt(sb.length() - 1) == ' ') return true;
        }
        return false;
    }

    @Test
    public void stripLeadingWhitespace_returnsCorrect_result_for_test_string() {
        String testString = "Hello World!";
        assertEquals("World", stripLeadingWhitespace(new StringBuilder(testString)));
    }

    private TextNode createTextNode(String text) {
        return new TextNode(text);
    }

import org.junit.Test;
import static org.junit.Assert.assertEquals;

public class Main {
    @Test
    public void testEqual() {
        assertEquals("Hello", stripLeadingWhitespace(new StringBuilder("Hello\nWorld")));
        assertEquals("World", stripLeadingWhitespace(new StringBuilder("Hello\nWorld")));
        assertEquals("World", stripLeadingWhitespace(new StringBuilder("\nWorld\nHello")));
    }

    @Test
    public void testNotEqual() {
        assertEquals("Goodbye", stripLeadingWhitespace(new StringBuilder("Goodbye\nWorld")));
        assertEquals("Goodbye", stripLeadingWhitespace(new StringBuilder("\nGoodbye\nWorld")));
        assertEquals("Goodbye", stripLeadingWhitespace(new StringBuilder("\nGoodbyeHello")));
    }

    @Test
    public void testEmpty() {
        String str = "";
        assertEquals("Hello", stripLeadingWhitespace(new StringBuilder(str)));
        assertEquals("", stripLeadingWhitespace(new StringBuilder ""));
    }

    private String stripLeadingWhitespace(StringBuilder sb) {
        while (!sb.isEmpty() && Character.isWhitespace(sb.charAt(0))) {
            sb.deleteCharAt(0);
        }
        return sb.toString();
    }
}
