```java
import java.util.HashMap;
import java.util.Map;

public class TagFactory {

    private static final Map<String, Tag> tags = new HashMap<>();

    private static void register(Tag tag) {
        if (!tags.containsKey(tag.tagName)) {
            tags.put(tag.tagName, tag);
        }
    }

    public static Tag create(String tagName) {
        return tags.get(tagName);
    }

    @Test
    public void testRegister() {
        // Test register method
        Tag tag = new Tag("div");
        register(tag);
        validateTags();
        System.out.println(tags.size());
    }

    private static void validateTags() {
        // Check if all tags have a name property
        for (String tagName : tags.keySet()) {
            if (!tags.get(tagName).getName() != null) {
                fail("Missing name property in " + tagName);
            }
        }
    }
}
```

```java
public class Tag {

    private String tagName;
    private boolean isBlock;
    private boolean formatAsBlock;

    public Tag(String tagName) {
        this.tagName = tagName;
        this.isBlock = false;
        this.formatAsBlock = false;
    }

    public String getName() {
        return tagName;
    }

    public void setFormatAsBlock(boolean formatAsBlock) {
        this.formatAsBlock = formatAsBlock;
    }
}
```

```java
public class Element {

    private Map<String, Tag> tags;

    public Element(Map<String, Tag> tags) {
        this.tags = tags;
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        for (Map.Entry<String, Tag> entry : tags.entrySet()) {
            sb.append(entry.getKey()).append(": ").append(entry.getValue().getName());
            if (!entry.getValue().formatAsBlock && !entry.getValue().isBlock) {
                sb.append(" ");
            }
        }
        return sb.toString();
    }

    @Test
    public void testToString() {
        // Test toString method
        Map<String, Tag> tags = new HashMap<>();
        tags.put("div", new Tag("div"));
        Element element = new Element(tags);
        System.out.println(element.toString());
    }
}
```

```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Tag {
    private String tagName;
    public Tag(String tagName) {
        this.tagName = tagName;
    }
}

class Element {
    private Map<String, Tag> tags;

    public Element(Map<String, Tag> tags) {
        this.tags = tags;
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj) return true;
        if (obj == null || getClass() != obj.getClass()) return false;
        Element element = (Element) obj;
        return Objects.equals(tagName, element.tagName);
    }

    @Override
    public int hashCode() {
        return Objects.hash(tagName);
    }
}

class TagBuilder extends StringBuilder {
    private String tagName;

    public TagBuilder(String tagName) {
        this.tagName = tagName;
    }

    public Element addTag() {
        return new Element(this);
    }

    @Override
    public String toString() {
        return super.toString() + tagName;
    }

    public void println() {
        System.out.println("[" + toString() + "]");
    }
}

public class TestSpark {

    private static Map<String, Tag> tags = new HashMap<>();

    public static Element register(Tag tag) {
        if (!tags.containsKey(tag.tagName)) {
            tags.put(tag.tagName, tag);
        }
        return new Element(tags.get(tag.tagName));
    }

    @Test
    public void testToString() {
        // Test toString method
        Element element1 = register(new Tag("div"));
        Element element2 = register(new Tag("span"));

        System.out.println(element1.toString());
        System.out.println(element2.toString());

        tags.put("div", new TagBuilder("div").addTag().build());
        tags.put("span", new TagBuilder("span").addTag().build());

        for (Map.Entry<String, Element> entry : tags.entrySet()) {
            Element element = entry.getValue();
            System.out.println(entry.getKey() + ": " + element.toString());
        }
```
