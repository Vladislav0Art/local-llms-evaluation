import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class GeneratedTest {
    public String test5() throws URISyntaxException {
        URI u1 = new URI("https://example.com/path?query=123");
        String uri1 = u1.toString();

        try {
            Document doc = Jsoup.connect(u1).get();
            Element link1 = doc.selectFirst("#link").first();
            if (link1 == null) {
                throw new URISyntaxException(u1.toString());
            }
            link1.attr("href", "https://sub.example.com");
            String uri2 = urlFromLink(link1);
            // assert uri2.toString().contains("?key2=");
            
            Document doc2 = Jsoup.connect(uri2).get();
            Element link2 = doc2.selectFirst("#link").first();
            if (link2 == null) {
                throw new URISyntaxException(uri2.toString());
            }
            link2.attr("href", "https://example.com/path?ref=123");
        } catch (URISyntaxException e) {
            // handle exception
        }

        return "Test 5 passed.";
    }

    private URI urlFromLink(Element link) throws URISyntaxException {
        String href = link.attr("href");
        if (href != null && !href.isEmpty()) {
            return new URI(href);
        } else {
            throw new URISyntaxException(link.attr("href"));
        }
    }
}
