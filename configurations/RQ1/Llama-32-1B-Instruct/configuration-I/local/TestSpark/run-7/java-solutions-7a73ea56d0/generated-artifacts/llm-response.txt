        Stock[] result = new Stock[stock.length + 1];
        result[0] = s1;

            result[i] = stock[i - 1];
        }
        System.arraycopy(s2, 0, result, stock.length + 1, stock.length);
        return result;
    }

        for (int i = 0; i < stock.length; i++) {
            if (stock[i].getPrice() > s1.getPrice()) {
                return stock[i];
            }
        }
        throw new RuntimeException("No matching price");
    }

    static public Stock getStockAverage(Stock[] stock) {
        int sum = 0;
        for (Stock s : stock) {
            sum += s.getPrice();
        }
        return new Stock(sum / stock.length, "Average", 1);
    }
