    }

import org.junit.jupiter.api.Test;
import static org.mockito.Mockito.*;

public class TestOnlineStockSpan {

    @Test
    public void test_next() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();

        when(list.add(anyInt())).thenReturn(0);
        onlineStockSpan.list = new ArrayList<>();
        int result = onlineStockSpan.next(prices[0]);
        assertEquals(2, result);

        onlineStockSpan.list = new ArrayList<>();
        prices = new int[]{3, 2, 6, 5, 0, 1};
        when(list.add(anyInt())).thenReturn(0);
        result = onlineStockSpan.next(prices[0]);
        assertEquals(4, result);
    }

    @Test
    public void test_calculateSpans() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        int[] prices = {7, 6, 5, 4, 3, 2};
        when(list.add(anyInt())).thenReturn(0);
        onlineStockSpan.list = new ArrayList<>();
        int[] expectedSpans = {1, 1, 2, 2, 3};
        onlineStockSpan.calculateSpans(prices);
        verifyAll();
        assertArrayEquals(expectedSpans, list.stream().mapToInt(Integer::intValue).toArray());

        prices = new int[]{10, 22, 9, 33, 21, 50, 41, 60, 80};
        when(list.add(anyInt())).thenReturn(0);
        onlineStockSpan.list = new ArrayList<>();
        result[] = onlineStockSpan.calculateSpans(prices);
        verifyAll();
        assertEquals(new int[]{1, 2, 3, 4, 5}, result);

    }
}
