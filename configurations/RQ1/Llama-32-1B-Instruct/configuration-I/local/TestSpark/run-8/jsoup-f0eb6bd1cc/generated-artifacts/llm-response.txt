```java
    public void testUrlBuilder_5ScenarioTest() {
```

```java
            // expected
```

```java
import java.net.URI;
import java.net.URISyntaxException;

public class UrlParserTest {

    @Before
    public void init() {
        String url = "https://example.com/path?query=value#anchor";
    }

    public void testUrlParser_1ScenarioTest(String url) {
        try {
            URL parsedUrl = new URI(url);
            assertEquals("https://example.com/path", parsedUrl.getScheme());
            assertEquals("https://example.com/path", parsedUrl.getHost());
            assertEquals("#anchor", parsedUrl.getPath(), "Path is not correctly normalized");
        } catch (URISyntaxException e) {
            // expected
        }
    }

    public void testUrlParser_2ScenarioTest(String url) {
        try {
            UrlParser parser = new UrlParser();
            String result = parser.parse(url);
            assertEquals("https://example.com/path", result.getScheme());
            assertEquals("https://example.com/path", result.getHost());
            assertEquals("#anchor", result.getPath(), "Path is not correctly normalized");
        } catch (UnsupportedEncodingException e) {
            // expected
        }
    }

    @After
    public void tearDown() {}

}

class UrlParser {

    public String parse(String url) throws URISyntaxException {
        URI parsedUrl = new URI(url);
        return normalizedPath(parsedUrl.getPath());
    }

    private String normalizedPath(String path) {
        if (path == null || path.isEmpty()) {
            return "";
        }
        path = normalizePath(path);
        int index = path.indexOf("#");
        if (index != -1) {
            path = path.substring(0, index);
        }
        return path;
    }

}

public class StringUtil {

    public static String normalizePath(String path) {
        return path.replaceAll("^/|\\\\|/\\$(?!$", "");
    }
}
```
