    private static void append(Node node, int depth, String tree) {
        if (node == null) return;

        // Append the new node and a space
        tree = appendHelper(node, depth + 1, tree);

        // Insert a newline at each level except the last one
        for (int i = 0; i < depth - 1; i++) {
            if (tree.length() > 0) tree += "\n";
        }

        // Append a space and a closing bracket
        tree = appendHelper(node, depth + 2, tree);
    }

    private static String appendHelper(Node node, int level, String tree) {
        // If the node has text, append it to the tree string
        if (node.getText() != null && node.getText().length() > 0) {

        } else if (tree.length() > 0) {

        String newText = tree.substring(0, startIndex);
        if (newText.isEmpty()) {
            root.setText(null);
            return;

        new BinaryTreeGenerator().generateTree(root.left, tree.substring(startIndex + 2));

        if (root.getText() == null) {
            root.setText(newText);

class Node {

    public Node(String text) {

        this.left = null;
        this.right = null;
    }

    public void addNode(Node node, String value) {
        if (node == null || !node.getText().equals(value)) return;

        node.addNode(value, value);
    }
}

    public static void main(String[] args) {
        Node root = new Node("root");

        left.addNode(right, "y->b");

        printTree(root);
    }

    private static void printTree(Node node) {
        if (node == null) return;

        System.out.println(node.getText());

        for (Node child : node.children) {
            printTree(child);
        }
    }

    }
