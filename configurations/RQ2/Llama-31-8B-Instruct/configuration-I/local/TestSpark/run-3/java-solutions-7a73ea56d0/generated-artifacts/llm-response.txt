import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.InjectMocks;
import org.mockito.junit.MockitoJUnitRunner;

import static org.junit.Assert.assertArrayEquals;
import static org.junit.Assert.assertEquals;

@RunWith(MockitoJUnitRunner.class)
public class OnlineStockSpanTest {

    @InjectMocks
    private OnlineStockSpan onlineStockSpan;

    @Test
    public void next_EmptyStackTest() {
        assertEquals(1, onlineStockSpan.next(10));
    }

    @Test
    public void next_PushToStackTest() {
        onlineStockSpan.next(10);
        assertEquals(1, onlineStockSpan.next(20));
    }

    @Test
    public void next_PopFromStackTest() {
        onlineStockSpan.next(10);
        onlineStockSpan.next(20);
        assertEquals(1, onlineStockSpan.next(15));
    }

    @Test
    public void next_PopFromStackTwiceTest() {
        onlineStockSpan.next(10);
        onlineStockSpan.next(20);
        onlineStockSpan.next(15);
        assertEquals(1, onlineStockSpan.next(25));
    }

    @Test
    public void calculateSpans_EmptyArrayTest() {
        assertArrayEquals(new int[]{}, onlineStockSpan.calculateSpans(new int[]{}));
    }

    @Test
    public void calculateSpans_SingleElementArrayTest() {
        assertArrayEquals(new int[]{1}, onlineStockSpan.calculateSpans(new int[]{10}));
    }

    @Test
    public void calculateSpans_IncreasingArrayTest() {
        assertArrayEquals(new int[]{1, 1, 2, 3, 4}, onlineStockSpan.calculateSpans(new int[]{10, 7, 20, 15, 30}));
    }

    @Test
    public void calculateSpans_DecreasingArrayTest() {
        assertArrayEquals(new int[]{1, 1, 1, 1, 1}, onlineStockSpan.calculateSpans(new int[]{30, 25, 20, 15, 10}));
    }

    @Test
    public void calculateSpans_MixedArrayTest() {
        assertArrayEquals(new int[]{1, 1, 2, 1, 1, 4, 2}, onlineStockSpan.calculateSpans(new int[]{10, 7, 20, 15, 30, 25, 20}));
    }
