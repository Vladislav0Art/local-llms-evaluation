import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import static org.junit.Assert.*;
import static org.mockito.Mockito.*;

@RunWith(MockitoJUnitRunner.class)
public class UrlBuilderTest {

    @Mock
    private Connection.KeyVal kv;

    private final UrlBuilder urlBuilder = new UrlBuilder("https://example.com");

    @Test
    public void appendKeyVal_NoException_KeysAreAdded() {
        // Arrange
        when(kv.getKey()).thenReturn("key1");
        when(kv.getValue()).thenReturn("value1");

        // Act
        urlBuilder.appendKeyVal(kv);

        // Assert
        verify(kv, times(1)).getKey();
        verify(kv, times(1)).getValue();
    }

    @Test
    public void appendKeyVal_KeysAndValuesAreAdded() {
        // Arrange
        when(kv.getKey()).thenReturn("key1");
        when(kv.getValue()).thenReturn("value1");

        // Act
        urlBuilder.appendKeyVal(kv);

        // Assert
        verify(kv, times(2)).getKey();
        verify(kv, times(2)).getValue();
    }

    @Test
    public void appendKeyVal_KeysAndValuesAreAdded WithoutKV() {
        // Arrange
        when(kv.getKey()).thenReturn("key1");
        when(kv.getValue()).thenReturn("value1");

        // Act
        urlBuilder.appendKeyVal(null);

        // Assert
        verify(kv, times(2)).getKey();
        verify(kv, times(2)).getValue();
    }

    @Test
    public void build_NoException_UrlIsReturned() {
        // Arrange
        when(urlBuilder.build()).thenReturn("https://example.com");

        // Act
        String url = urlBuilder.build();

        // Assert
        assertEquals("https://example.com", url);
    }
}
