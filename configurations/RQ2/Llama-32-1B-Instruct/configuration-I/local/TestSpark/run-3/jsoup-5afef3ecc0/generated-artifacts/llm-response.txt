import java.util.HashMap;

    public static void main(String[] args) throws Exception {
        System.out.println(getUrl("https://example.com"));
        System.out.println(getUrl("http://localhost:8080/test?key1=value1&key2=value2"));
    }

    private static String getUrl(String url) throws Exception {
        URL obj = new URL(url);
        HttpURLConnection con = (HttpURLConnection) obj.openConnection();
        con.setRequestMethod("GET");
        int responseCode = con.getResponseCode();

        if (responseCode == 200) {
            StringBuilder sb = new StringBuilder();
            if (con.getHeaderFields().get("Content-Type").equals("text/html")) {
                // Get the HTML content of the URL
                try (InputStream in = con.getInputStream()) {
                    byte[] buffer = new byte[1024];
                    int len;
                    while ((len = in.read(buffer)) != -1) {
                        sb.append(new String(buffer, 0, len));
                    }
                } catch (Exception e) {
                    System.out.println("Error reading the URL: " + e.getMessage());
                }
            }

            return sb.toString();
        } else {
            throw new Exception("Failed to retrieve URL: " + url);
        }
    }
}
