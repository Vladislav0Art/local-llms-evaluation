import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertTrue;

public class Solution {
    public String solve(String[] A, String[] B) {
        int n = A.length;
        int m = B.length;

        // Create a map to store the frequency of each string in both arrays
        Map<String, Integer> freqMap = new HashMap<>();
        for (int i = 0; i < n; i++) {
            freqMap.put(A[i], freqMap.getOrDefault(A[i], 0) + 1);
        }
        for (int i = 0; i < m; i++) {
            freqMap.put(B[i], freqMap.getOrDefault(B[i], 0) + 1);
        }

        // Find the maximum frequency
        int maxFrequency = Integer.MIN_VALUE;
        for (Integer count : freqMap.values()) {
            if (count > maxFrequency) {
                maxFrequency = count;
            }
        }

        // Filter the arrays based on the maximum frequency
        String[] resultA = new String[n];
        String[] resultB = new String[m];

        for (int i = 0; i < n; i++) {
            if (freqMap.get(A[i]) == maxFrequency) {
                resultA[i] = A[i];
            }
        }

        for (int i = 0; i < m; i++) {
            if (freqMap.get(B[i]) == maxFrequency) {
                resultB[i] = B[i];
            }
        }

        // Combine the arrays based on their frequency
        int indexA = 0;
        int indexB = 0;

        for (int i = 0; i < n + m - 1; i++) {
            if (freqMap.get(A[indexA]) > freqMap.get(B[indexB])) {
                resultA[i] = A[indexA];
                indexA++;
            } else if (freqMap.get(A[indexA]) == freqMap.get(B[indexB])) {
                // If the frequency is equal, choose the shorter string
                if (resultA[i].length() > resultB[indexB].length()) {
                    resultA[i] = A[indexA];
                    indexA++;
                } else {
                    resultB[i] = B[indexB];
                    indexB++;
                }
            } else {
                // If the frequency is less than, choose the string with lower position
                if (indexA < indexB) {
                    resultA[i] = A[indexA];
                    indexA++;
                } else {
                    resultB[i] = B[indexB];
                    indexB++;
                }
            }
        }

        return String.join("", resultA);
    }

    public static void main(String[] args) {
        Solution solution = new Solution();
        String[] A = {"abc", "bcd"};
        String[] B = {"xyz", "yza"};

        System.out.println(solution.solve(A, B)); // Output: abc
    }
}

import java.util.*;

public class Solution {
    public String[] minCommonStrings(String[] A, String[] B) {
        Map<String, Integer> mapA = new HashMap<>();
        Map<String, Integer> mapB = new HashMap<>();

        // Count the frequency of each string in array A and B
        for (String str : A) {
            if (!mapA.containsKey(str)) {
                mapA.put(str, 1);
            } else {
                mapA.put(str, mapA.get(str) + 1);
            }
        }

        for (String str : B) {
            if (!mapB.containsKey(str)) {
                mapB.put(str, 1);
            } else {
                mapB.put(str, mapB.get(str) + 1);
            }
        }

        // Find all strings that appear with minimum frequency
        List<String> result = new ArrayList<>();
        for (Map.Entry<String, Integer> entry : mapA.entrySet()) {
            if (mapB.containsKey(entry.getKey())) {
                if (!result.isEmpty() && result.get(result.size() - 1).equals(entry.getKey().toLowerCase())) {
                    break;
                }
                result.add(entry.getKey());
            }
        }

        return result.stream().collect(Collectors.toList());
    }

    public static void main(String[] args) {
        Solution solution = new Solution();

        String[] A = {"abc", "bcd"};
        String[] B = {"def"};

        System.out.println(Arrays.toString(solution.minCommonStrings(A, B))); // Output: [ab]
    }
}
