import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class TestUtil {

    private Document document = Jsoup.connect("https://www.example.com").get();

    public String getHTML() {
        return document.body().outerHtml();
    }

    public void printText(Node textNode) {
        System.out.println(textNode.toString());
    }

    public void assertString(String str, Node node) {
        if (!str.equals(node.toString())) {
            throw new RuntimeException("Expected '" + str + "' but got '" + node.toString() + "'");
        }
    }

    public Document cloneDocument(Node root) {
        return document.cloneNode(true);
    }

    public String getFirstTextNode(String text, String pattern) {
        Elements elements = document.select(pattern);
        for (Element element : elements) {
            if (element.text().trim().isEmpty()) {
                continue;
            }
            System.out.println(element.text());
            assertString(text, element);
        }
        return null;
    }

    public void printTextNodes(Node textNode) {
        Elements elements = document.select("[type='text']");
        for (Element element : elements) {
            if (element.text().trim().isEmpty()) {
                continue;
            }
            System.out.println(element.text());
        }
    }

    public static void main(String[] args) {
        TestUtil testUtil = new TestUtil();

        // Retrieve the HTML content
        String htmlContent = testUtil.getHTML();
        System.out.println("HTML Content: " + htmlContent);

        // Print all text nodes in the document
        testUtil.printTextNodes(testUtil.document.select("[type='text']"));

        // Clone the document
        Document clonedDocument = testUtil.cloneDocument(testUtil.document.select("[id='header']"));
        System.out.println("Cloned Document: " + clonedDocument.outerHtml());

        // Get the first text node in a specific pattern
        String pattern = "[type='text' and @id='header']";
        String textContent = testUtil.getFirstTextNode("Hello, World!", pattern);
        System.out.println("First Text Content: " + textContent);

        // Test nodes
        Element element1 = testUtil.document.select("[href='#top']").first();
        System.out.println(element1.attr("id"));  // Should print 'top'

        Element element2 = testUtil.document.select("[class='nav navbar navbar-fixed-top nav-header pull-left']").first();
        System.out.println(element2.text());  // Should print an empty string

        Element element3 = testUtil.document.select("[href='#footer']").first();
        System.out.println(element3.attr("id"));  // Should print 'footer'

        Element element4 = testUtil.document.select("[data-toggle='modal']").first();
        System.out.println(element4.text());  // Should print an empty string
    }
}
