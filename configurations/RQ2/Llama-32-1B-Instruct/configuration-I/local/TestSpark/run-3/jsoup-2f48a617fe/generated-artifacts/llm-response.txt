```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class Main {

    public static String generateHtml(String htmlString, Element parentElement) {
        if (parentElement == null) {
            return htmlString;
        }

        Document doc = Jsoup.parse(htmlString);
        Elements tags = doc.getAllElements();
        for (Element tag : tags) {
            if (tag.tagName().equals("h1")) {
                tag.append(parentElement.text());
            } else {
                parentElement.append(tag.clone()).html(generateHtml(tag.html(), tag));
            }
        }

        return doc.toString();
    }

    public static void main(String[] args) {
        String htmlString = "<html><body><h1>Hello World!</h1></body></html>";
        Element parentElement = Jsoup.parse(htmlString).selectFirst("body").parent();
        System.out.println(generateHtml(htmlString, parentElement));
    }
}
```

Here's a new Java solution for generating HTML from a string, using regular expressions to extract the HTML tags.

This solution uses a simple regular expression to find and replace all occurrences of HTML tags in the input string.

```java
import java.util.regex.Matcher;
import java.util.regex.Pattern;

public class Main {

    public static String generateHtml(String inputString) {
        // Define a regular expression pattern that matches most common HTML tags.
        Pattern pattern = Pattern.compile("<[^>]*>");
        
        // Use Matcher to find and replace all occurrences of the pattern in the input string.
        Matcher matcher = pattern.matcher(inputString);
        StringBuffer stringBuffer = new StringBuffer();
        
        while (matcher.find()) {
            String match = matcher.group();
            
            if (!match.isEmpty()) {
                // If the tag is not already enclosed in angle brackets, append it with opening and closing tags to the result string.
                if (Pattern.quote(match).matches("^[\\w:]+$")) {
                    stringBuffer.append("<").append(match).append(">");
                    
                    // Remove extra space at the end of the tag
                    matcher.appendReplacement(stringBuffer, "");
                } else {
                    matcher.appendReplacement(stringBuffer, "<" + match + ">");
                }
            }
        }
        
        matcher.appendTail(stringBuffer);
        
        return stringBuffer.toString();
    }

    public static void main(String[] args) {
        String inputString = "Hello < World!>";
        System.out.println(generateHtml(inputString));
    }
}
```

Here's a Java solution that uses regular expressions to find and replace all occurrences of HTML tags in the input string.

This solution assumes that the input string does not contain any JavaScript code or other special characters that might interfere with the parsing process.

```java
import java.util.regex.Matcher;
import java.util.regex.Pattern;

    public static String generateHtml(String inputString) {
        // Define a regular expression pattern that matches most common HTML tags.
        Pattern pattern = Pattern.compile("<[^>]*>");
        
        // Use Matcher to find and replace all occurrences of the pattern in the input string.
        Matcher matcher = pattern.matcher(inputString);
        StringBuffer stringBuffer = new StringBuffer();
        
        while (matcher.find()) {
            String match = matcher.group();
            
            if (!match.isEmpty()) {
                // If the tag is not already enclosed in angle brackets, append it with opening and closing tags to the result string.
                if (Pattern.quote(match).matches("^[\\w:]+$")) {
                    matcher.appendReplacement(stringBuffer, "<").append(match).append(">");  // Removed extra space at the end of the tag
                } else {
                    matcher.appendReplacement(stringBuffer, match);  // No replacement for non-angle-bracketed tags
                }
            }
        }
        
        matcher.appendTail(stringBuffer);
        
        return stringBuffer.toString();
    }
```
