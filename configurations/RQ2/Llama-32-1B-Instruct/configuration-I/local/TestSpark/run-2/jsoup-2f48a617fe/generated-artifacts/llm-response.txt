public class Comment extends LeafNode {
    /**
     * Create a new comment node.
     * @param data The contents of the comment
     */
    public Comment(String data) {
        super(data);
    }

    /**
     * Get the contents of the comment.
     * @return comment content
     */
    public String getNodeName() { return null; }
}

public class CommentTest extends TestCase {
    private static final String COMMENT_DATA = "This is a test comment";

    public void testCommentCreation() {
        Comment comment = new Comment(COMMENT_DATA);
        assertEquals(COMMENT_DATA, comment.getNodeData());
        assertNotNull(comment.getNodeName());
    }

    public void testGetNodeName() {
        Comment comment = new Comment(COMMENT_DATA);
        assertEquals("comment", comment.getNodeName());
    }

    public void testGetCommentContent() {
        Comment comment = new Comment(COMMENT_DATA);
        assertEquals(COMMENT_DATA, comment.getData());
        assertNotNull(comment.getNodeName());
    }

    public void testSetData() {
        Comment comment = new Comment(COMMENT_DATA);
        comment.setData("New content");
        assertEquals("New content", comment.getData());
    }

    @Test
    public void testOuterHtmlHead() {
        Comment comment = new Comment(COMMENT_DATA);
        when(comment.outerHtmlHead(any(Appendable.class), any(int.class), any(Document.OutputSettings.class)).thenReturn("", "");
        assertTrue(comment.outerHtmlHead(new Appendable(), 1, Document.OutputSettings.OUTPUT_TYPE_HTML));
    }

    public void testOuterHtmlTail() {
        Comment comment = new Comment(COMMENT_DATA);
        when(comment.outerHtmlTail(any(Appendable.class), any(int.class), any(Document.OutputSettings.class)).thenReturn("", "");
        assertTrue(comment.outerHtmlTail(new Appendable(), 1, Document.OutputSettings.OUTPUT_TYPE_HTML));
    }

    @Test
    public void testToString() {
        Comment comment = new Comment(COMMENT_DATA);
        assertEquals("This is a test comment", comment.toString());
    }

    public void testClone() {
        Comment comment = new Comment(COMMENT_DATA);
        Comment clonedComment = comment.clone();
        assertEquals(comment.getNodeName(), clonedComment.getNodeName());
        assertTrue(clonedComment.getNodeData().equals(comment.getNodeData()));
        assertNotNull(clonedComment.getNodeName());
    }

    @Test
    public void testIsXmlDeclaration() {
        Comment comment = new Comment(COMMENT_DATA);
        assertTrue(comment.isXmlDeclaration());
    }
}
