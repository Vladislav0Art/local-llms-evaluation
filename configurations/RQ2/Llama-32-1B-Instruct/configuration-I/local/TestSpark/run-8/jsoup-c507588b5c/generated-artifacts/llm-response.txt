public class TextNodeTest {

    @Test
    public void testTextNodeBlank() {
        TextNode node = new TextNode("");
        assertEquals(true, node.isBlank());
    }

    @Test
    public void testTextNodeNoText() {
        TextNode node = new TextNode("Hello World");
        assertEquals("", node.getText());
    }

    @Test
    public void testTextNodeWithSpaces() {
        TextNode node = new TextNode("   Hello   World  ");
        assertEquals("Hello World", node.getText());
    }

    @Test
    public void testTextNodeSplitting() {
        String text = "Hello World";
        TextNode node1 = node1 = new TextNode(text);
        TextNode node2 = node2 = node1.splitText(7);

        assertEquals(text.substring(0, 7), node2.getText());
        assertEquals("World", node2.getText().substring(node1.getText().length()));
    }

    @Test
    public void testTextNodeConcatenation() {
        String text = "Hello World";
        TextNode node = new TextNode(text);
        StringBuilder builder = new StringBuilder();
        node.getText(builder);

        assertEquals("Hello World", builder.toString());
    }

    @Test
    public void testTextNodeToString() {
        TextNode node = new TextNode("Hello World");
        assertEquals("Hello World", node.toString());
    }

    @Test
    public void testTextNodeClone() {
        TextNode node1 = new TextNode("Hello World");
        TextNode node2 = node2 = node1.clone();

        assertEquals(node1, node2);
    }

    @Test
    public void testTextNodeToString Clone() {
        TextNode node = new TextNode("Hello World");
        assertEquals("Hello World", node.toString());
    }
}

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class NodeAssert {

    private static Document node1 = new Document();
    private static Document node2 = new Document();

    @Test
    public void testTextNodeBlank() {
        node1.html("");
        assertEquals(true, node1.isBlank());
    }

    @Test
    public void testTextNodeNoText() {
        node1.html("Hello World");
        assertEquals("", node1.getText());
    }

    @Test
    public void testTextNodeWithSpaces() {
        node1.html(" Hello   World ");
        assertEquals(" Hello   World ", node1.getText());
    }

    @Test
    public void testTextNodeEmptyString() {
        node1.html("");
        node2.html("");
        assertEquals("", node2.getText());
    }

    @Test
    public void testTextNodeSubstring() {
        node1.html("<p> Hello <span>World</span></p>");
        node2.html("");
        assertEquals("Hello World", node2.getText().replaceAll("^<|>$", ""));
    }

    @Test
    public void testTextNodeSubstringIndex() {
        node1.html("<p> Hello <span>World</span></p>");
        node2.html("");
        Elements spanTags = node1.querySelectorAll("span");
        assertEquals(3, spanTags.size());
        assertEquals("Hello World", spanTags.get(0).text());
    }

    public static void main(String[] args) {
        NodeAssert nodeAssert = new NodeAssert();
        System.out.println(nodeAssert.testTextNodeBlank());
        System.out.println(nodeAssert.testTextNodeNoText());
        System.out.println(nodeAssert.testTextNodeWithSpaces());
        System.out.println(nodeAssert.testTextNodeEmptyString());
        System.out.println(nodeAssert.testTextNodeSubstring());
        System.out.println(nodeAssert.testTextNodeSubstringIndex());
    }
}
