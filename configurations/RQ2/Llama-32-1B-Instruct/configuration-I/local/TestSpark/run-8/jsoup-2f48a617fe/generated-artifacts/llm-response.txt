public class Comment extends LeafNode {
    /**
     * Create a new comment node.
     * @param data The contents of the comment
     */
    public Comment(String data) { this.data = data; }

    /**
     * Get the contents of the comment.
     * @return comment content
     */
    public String getData() { return data; }

    public Comment setData(String data) { this.data = data; return this; }

    @Override
    void outerHtmlHead(Appendable accum, int depth, Document.OutputSettings out) throws IOException {
        StringBuilder builder = new StringBuilder();
        builder.append("<comment>\n");
        builder.append("  <data>").append(data).append("</data>\n");
        builder.append("</comment>\n");
        appendTo(accum, builder.toString(), depth);
    }

    @Override
    void outerHtmlTail(Appendable accum, int depth, Document.OutputSettings out) { /* empty */ }

    @Override
    public String toString() {
        return "<Comment>" + getData() + "</Comment>";
    }

    @Override
    public Comment clone() {
        return (Comment) super.clone();
    }

    /**
     * Check if this comment looks like an XML Declaration.
     * @return true if it looks like, maybe, it's an XML Declaration.
     */
    public boolean isXmlDeclaration() { return data.startsWith("<?xml"); }

    /**
     * Attempt to cast this comment to an XML Declaration node.
     * @return an XML declaration if it could be parsed as one, null otherwise.
     */
    public @Nullable XmlDeclaration asXmlDeclaration() {
        String content = getData();
        if (content.startsWith("<?xml")) { return new XmlDeclaration(content); }
        return null;
    }

    private void appendTo(Appendable accum, String content, int depth) throws IOException {
        if (!depth > 0) { // Only add newline after each line
            append(accum, "  \n");
        } else {
            appendTo(accum, "<comment>\n" + content + "\n</comment>");
        }
    }

    private void append(Appendable accum, String content) throws IOException {
        accum.append(content);
    }
}

public class XmlDeclaration extends Comment {
    /**
     * Create a new XML declaration node.
     * @param data The contents of the XML declaration
     */
    public XmlDeclaration(String data) { this.data = data; }

    // Overridden methods

    @Override
    boolean isXmlDeclaration() {
        return true;
    }

    @Override
    @Nullable
    XmlDeclaration asXmlDeclaration() {
        return null;
    }
}
