import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class UrlBuilder {

    public static String urlBuilder(String inputUrl) {
        try {
            Document document = Jsoup.parse(inputUrl);
            StringBuilder sb = new StringBuilder();
            for (org.jsoup.nodes.Element element : document.children()) {
                if (element.tagName().equals("a")) {
                    sb.append(element.attr("href")).append("\n");
                }
            }
            return sb.toString();
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }

    public static void main(String[] args) {
        String url = "https://example.com";
        System.out.println(urlBuilder(url));

        String link = "https://example.com/link";
        System.out.println(urlBuilder(link));
    }
}

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class UrlParser {

    public static String parseUrl(String url) {
        try {
            Document document = Jsoup.parse(url);
            return document.text();
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }

    public static void main(String[] args) {
        String url = "https://example.com";
        System.out.println(parseUrl(url));

        String link = "https://example.com/link";
        System.out.println(parseUrl(link));
    }
}
