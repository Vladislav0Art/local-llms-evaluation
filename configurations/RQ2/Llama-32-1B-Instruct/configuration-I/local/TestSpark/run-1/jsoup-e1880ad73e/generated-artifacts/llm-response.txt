import org.jsoup.nodes.Document;

public class Main {

    public static void main(String[] args) {
        System.out.println(checkUrl("https://www.google.com"));
    }

    public static boolean checkUrl(String url) {
        try {
            Document doc = Jsoup.connect(url).get();
            if (doc.select("body").isEmpty()) {
                return false;
            }
            for (int i = 0; i < doc.select("h1, h2, h3, h4, h5, h6, p, span, div").size(); i++) {
                String text = doc.select("h" + i + ", h" + (i + 1)).first().text();
                if (text.isEmpty()) {
                    return false;
                }
            }

            for (int i = 0; i < doc.select("#nav").size(); i++) {
                String text = doc.select("#nav" + i).first().text();
                if (!text.isEmpty() && !text.contains("<a")) {
                    return false;
                }
            }

            return true;

        } catch (Exception e) {
            System.out.println(e.getMessage());
            return false;
        }
    }
}

public class Main {
    public static void main(String[] args) {
        // Check if a URL is valid
        String url = "https://www.google.com";
        System.out.println(checkUrl(url));
    }

    public static boolean checkUrl(String url) {
        try {
            // Connect to the URL and get the HTML response
            Document doc = Jsoup.connect(url).get();
            
            // Check if there is any content in the <body> tag
            if (doc.select("body").isEmpty()) {
                return false;
            }
            
            // Iterate over all headings with text content
            for (int i = 0; i < doc.select("h1, h2, h3, h4, h5, h6, p, span, div").size(); i++) {
                String text = doc.select("h" + i + ", h" + (i + 1)).first().text();
                
                // Check if the heading is empty
                if (text.isEmpty()) {
                    return false;
                }
            }
            
            // Iterate over anchor tags with no content
            for (int i = 0; i < doc.select("#nav").size(); i++) {
                String text = doc.select("#nav" + i).first().text();
                
                // Check if the anchor tag has any content
                if (!text.isEmpty() && !text.contains("<a")) {
                    return false;
                }
            }
            
            // If no issues were found, return true
            return true;

        } catch (Exception e) {
            System.out.println(e.getMessage());
            return false;
        }
    }
}
