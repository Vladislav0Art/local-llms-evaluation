public class TextNodeTests {

    private LeafNode document;
    private String text;

    @Test
    public void testCreateFromEncoded() {
        text = "test";
        document = TextNode.createFromEncoded(text);
        assertEquals("test", document.text());
    }

    @Test
    public void testNodeName() {
        document = new TextNode("hello");
        assertEquals("TextNode", document.getNodeName());
    }

    @Test
    public void testText() {
        text = "   hello   ";
        document = new TextNode(text);
        assertEquals("hello ", document.text());
    }

    @Test
    public void testGetWholeText() {
        text = "   hello   ";
        document = new TextNode(text);
        assertEquals("hello", document.getText());
    }

    @Test
    public void testIsBlank() {
        document = new TextNode("");
        assertTrue(document.isBlank());
    }

    @Test
    public void testSplitText() {
        text = "hello world";
        document = new TextNode(text);
        TextNode result = document.splitText(7);
        assertEquals("hello", result.text());
        assertEquals("world", document.getText().substring(8));
    }

    @Test
    public void testOuterHtmlHead() throws IOException {
        String html = "<p>Hello</p>";
        Appendable accum = new StringBuilder();
        Document.OutputSettings out = new Document.OutputSettings();
        document.outerHtmlHead(accum, 1, out);
        assertEquals(html, accum.toString());
    }

    @Test
    public void testOuterHtmlTail() throws IOException {
        String html = "<p>Hello</p>";
        Appendable accum = new StringBuilder();
        document.outerHtmlTail(accum, 0, null);
        assertEquals(html, accum.toString());
    }
}
