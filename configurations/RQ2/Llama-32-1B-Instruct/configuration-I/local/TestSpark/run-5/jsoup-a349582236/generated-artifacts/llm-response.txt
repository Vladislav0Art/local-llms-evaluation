public class TagImpl implements Cloneable {
    
    private String name;
    private String normalizedName;

    @Override
    protected String clone() {
        return name != null ? "TagImpl" + (normalizedName != null ? "_" + normalizedName : "") : "Tag";
    }

    public void setName(String name) { this.name = name; }
    public String getName() { return name; }
    public String normalName() { return name == null ? null : name.toLowerCase(); }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;

        TagImpl that = (TagImpl) o;

        if (!name.equals(that.name)) return false;
        return normalizedName.equals(that.normalizedName);
    }

    @Override
    public int hashCode() {
        int result = name.hashCode();
        result = 31 * result + normalizedName.hashCode();
        return result;
    }
}
