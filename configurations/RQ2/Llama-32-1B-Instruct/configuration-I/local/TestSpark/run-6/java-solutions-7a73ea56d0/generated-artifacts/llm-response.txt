import org.junit.jupiter.api.Test;
import static org.mockito.Mockito.*;

public class OnlineStockSpanTest {

    private OnlineStockSpan instance = new OnlineStockSpan();

    @Test
    public void test_max_line_coverage_empty_prices() {
        int[] prices = {};
        int[] result = instance.calculateSpans(prices);
        verifyAll().next(0);
        assertEquals(result.length, 1);
        Arrays.asList(result).set(0, 0);
        assertArrayEquals(new int[]{1}, result);
    }

    @Test
    public void test_max_line_coverage_all_prices() {
        int[] prices = {100, 200, 300};
        int[] result = instance.calculateSpans(prices);
        verifyAll().next(3);
        Arrays.asList(result).set(0, 1);
        Arrays.asList(result).set(1, 2);
        Arrays.asList(result).set(2, 4);
        assertArrayEquals(new int[]{12, 8, 4}, result);
    }

    @Test
    public void test_max_line_coverage_with_mixed_prices() {
        int[] prices = {100, 200, 300};
        int[] result = instance.calculateSpans(prices);
        verifyAll().next(3);
        Arrays.asList(result).set(0, 1);
        Arrays.asList(result).set(1, 2);
        Arrays.asList(result).set(2, 4);
        assertArrayEquals(new int[]{12, 8, 4}, result);
    }

    @Test
    public void test_max_line_coverage_with_multiple_spans() {
        OnlineStockSpan.Span[] spans = new OnlineStockSpan.Span[]{
                OnlineStockSpan.Span.builder(10).build(),
                OnlineStockSpan.Span.builder(20).build()
        };
        int[] prices = {3, 5};
        int[] result = instance.calculateSpans(prices);
        verifyAll().next(9);
        Arrays.asList(result).set(0, spans[0].value());
        Arrays.asList(result).set(1, spans[1].value());
    }

    @Test
    public void test_max_line_coverage_with_single_span() {
        int[] prices = {10};
        int[] result = instance.calculateSpans(prices);
        verifyAll().next(1);
        assertArrayEquals(new int[]{1}, result);
    }
}
