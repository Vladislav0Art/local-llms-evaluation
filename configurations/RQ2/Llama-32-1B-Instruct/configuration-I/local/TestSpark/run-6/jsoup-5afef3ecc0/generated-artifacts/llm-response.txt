====================== Iteration #1 ======================
```java
import org.junit.jupiter.api.Test;
import org.jsoup.helper.UrlBuilder;
import org.mockito.Mockito;

public class UrlBuilderTest {

    @Test
    public void testBuild_ReturnsUrl() {
        // Arrange
        URL inputUrl = new URL("https://example.com");
        Mockito.when(inputUrl.getScheme()).thenReturn("https");

        // Act
        String actualUrl = UrlBuilder.build().toString();

        // Assert
        assertEquals("https://example.com", actualUrl);
    }

    @Test
    public void testAppendKeyVal appendKeysAndValues() throws UnsupportedEncodingException {
        // Arrange
        Connection.KeyVal kv1 = new Connection.KeyVal("key1", "value1");
        Connection.KeyVal kv2 = new Connection.KeyVal("key2", "value2");

        Mockito.when(inputUrl.getQuery()).thenReturn(null);
        Mockito.when(inputUrl.getFragment()).thenReturn(null);

        UrlBuilder urlBuilder = new UrlBuilder(new URL("https://example.com"));
        String actual = urlBuilder.appendKeyVal(kv1).appendKeyVal(kv2).toString();

        // Assert
        assertEquals("key1=value1&key2=value2", actual);
    }

    @Test
    public void testAppendKeyVal_ThrowsUnsupportedEncodingException() {
        // Arrange
        Connection.KeyVal kv = new Connection.KeyVal("key", "value");

        Mockito.when(inputUrl.getQuery()).thenReturn(null);
        Mockito.when(inputUrl.getFragment()).thenReturn(null);

        UrlBuilder urlBuilder = new UrlBuilder(new URL("https://example.com"));
        String actual = urlBuilder.appendKeyVal(kv).toString();

        // Assert
        assertEquals(0, actual.length()); // Should not throw UnsupportedEncodingException
    }

    @Test
    public void testAppendKeyVal_ThrowsUnsupportedEncodingException_SingleValue() {
        // Arrange
        Connection.KeyVal kv = new Connection.KeyVal("key", "value");

        Mockito.when(inputUrl.getQuery()).thenReturn(null);
        Mockito.when(inputUrl.getFragment()).thenReturn(null);

        UrlBuilder urlBuilder = new UrlBuilder(new URL("https://example.com"));
        try {
            urlBuilder.appendKeyVal(kv);
            fail("Expected UnsupportedEncodingException");
        } catch (UnsupportedEncodingException e) {
            // Expected
        }

        // Assert
        assertEquals(0, actual.length()); // Should not throw UnsupportedEncodingException
    }
}
```
