```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Tag {

    public static void main(String[] args) {
        testTagHasChildren();
        testTagHasAttributes();
        testTagHasBaseUri();
        testTagHasData();
    }

    public static void testTagHasChildren() {
        Document doc = Jsoup.create("<div>Hello World!</div>");
        assertElementHasChildNodes(doc).size() == 1;
    }

    public static void testTagHasAttributes() {
        Document doc = Jsoup.create("<span id='my-span'>Hello World!</span>");
        assertElementHasAttributes(doc);
    }

    public static void testTagHasBaseUri() {
        Document doc = Jsoup.create("<div>https://example.com/</div>");
        assertNodeHasAttribute(doc, "baseuri", "https://example.com/");
    }

    public static void testTagHasData() {
        Document doc = Jsoup.create("<span>Hello World!</span>");
        assertEquals("", doc.getData());
        doc.appendText("World!");
        assertEquals("Hello World!", doc.getData());
    }
}

public class Element {

    public static boolean assertElementHasChildNodes(Element node) {
        return node.children().size() > 0;
    }

    public static void assertNodeHasAttribute(Node node, String name, String value) {
        if (!node.hasAttr(name)) {
            throw new RuntimeException("Element does not have attribute '" + name + "'");
        }
        assertEquals(value, node.attr(name));
    }

    public static boolean assertElementHasAttributes(Element node) {
        return node.attributes().size() > 0;
    }
}
```
