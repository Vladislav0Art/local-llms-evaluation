import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class QuirksMode {
    public static String quirksMode(String html) {
        Document doc = Jsoup.parse(html);
        return doc.select("meta[name=quirks-mode]").first().attr("content");
    }

    public static void setQuirksMode(String html, String quirkMode) {
        Document doc = Jsoup.parse(html);
        Element metaTag = doc.selectFirst("meta[name=quirks-mode]");
        if (metaTag != null) {
            metaTag.attr("content", quirkMode);
        } else {
            Element newMetaTag = doc.createElement("meta");
            newMetaTag.setName("quirks-mode");
            newMetaTag.attr("content", quirkMode);
            doc.appendElement(newMetaTag);
        }
    }

    public static void main(String[] args) {
        String html = "<html><head></head><body>Hello, World!</body></html>";
        String quirksMode = "console";
        System.out.println(quirksMode);

        setQuirksMode(html, "firefox");
        System.out.println(quirksMode);
    }
}

Quirks mode is a browser setting that allows the user's default browser settings to override those set by the operating system. This is typically used for testing or development purposes, where the developer wants to test their website without the standard rendering engine.

**Setting Quirks Mode:**

To set quirks mode, you need to change your browser's rendering engine. Here are the steps:

1. Open your browser and go to the "Tools" menu (usually three horizontal lines or a gear icon).
2. Click on "Options" or "Preferences".
3. Click on the "Advanced" tab.
4. Scroll down to the "User Agent" section.
5. Select "Use custom user agent string" and enter a unique string that identifies your browser (e.g., "quirksmode").
6. Set the "Rendering engine" to "Quirks Mode".

Once you've set quirks mode, you can control its behavior by adjusting various options:

* Scripting language: You can select a different scripting language for your website (e.g., JavaScript, HTML).
* Image compression: You can enable or disable image compression.

**Quirks Mode Limitations:**

Keep in mind that quirks mode has some limitations:

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.select.Elements;

public class QuirksMode {
    public static String quirksMode(String html) {
        Document doc = Jsoup.parse(html);
        Elements metaTags = doc.select("meta[name=quirks-mode]");
        if (metaTags.isEmpty()) {
            return "Unknown";
        }
        for (Element tag : metaTags) {
            String content = tag.attr("content");
            if (!content.equals("console")) {
                return "Using " + tag.attr("name") + " instead of console.";
            }
        }
        return "Using console";
    }

    public static void setQuirksMode(String html, String quirkMode) {
        Document doc = Jsoup.parse(html);
        Elements metaTags = doc.select("meta[name=quirks-mode]");
        if (metaTags.isEmpty()) {
            Element newMetaTag = doc.createElement("meta");
            newMetaTag.setName("quirks-mode");
            newMetaTag.attr("content", quirkMode);
            doc.appendChild(newMetaTag);
        } else {
            for (Element tag : metaTags) {
                String content = tag.attr("content");
                if (!content.equals(quirkMode)) {
                    Element oldTag = doc.createElement(tag.tagName());
                    oldTag.attr(tag.name, "name", quirkMode);
                    doc.replaceWith(oldTag);
                }
            }
        }
    }

    public static void main(String[] args) {
        String html = "<html><head></head><body>Hello, World!</body></html>";
        System.out.println(quirksMode(html));

        setQuirksMode(html, "console");
        System.out.println(quirksMode(html));
    }
}
