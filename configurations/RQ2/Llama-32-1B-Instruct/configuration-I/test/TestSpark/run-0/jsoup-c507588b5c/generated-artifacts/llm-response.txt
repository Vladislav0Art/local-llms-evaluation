import org.jsoup.nodes.Document;

public class Node {

    private String text;
    private int id;

        return this.text;
    }

    public int getId() {
        return this.id;
    }

    public static Node createNode(String text) {
        Document document = new Document();
        element = document.createElement("p");
        element.setText(text);
        document.appendChild(element);
        node = new Node(document.toString());
        id = (int) Math.random() * 1000000 + 1000000;
        return node;
    }

    public static void main(String[] args) {

        this.id = 0;
    }

        return this.text;
    }

    public void setText(String text) {
        this.text = text;
    }

    public int getId() {
        return this.id;
    }

    public static Node createNode(String text) {
        Document document = new Document();
        element = document.createElement("p");
        element.setText(text);
        document.appendChild(element);
        node = new Node(document.toString());
        id = (int) Math.random() * 1000000 + 1000000;
        return node;
    }

    public static void main(String[] args) {
        Node node1 = createNode("Hello, World!");
        System.out.println(node1.getText().trim());

        Document document2 = new Document();
        element = document2.createElement("p");
        element.setText("This is a test.");
