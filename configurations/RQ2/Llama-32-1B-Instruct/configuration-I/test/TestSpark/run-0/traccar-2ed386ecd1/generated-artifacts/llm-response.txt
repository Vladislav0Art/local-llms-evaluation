public class GeneratedTest {
    public enum Direction {
        NORTH,
        SOUTH,
        EAST,
        WEST
    }

    public static Direction getDirection(double lat, double lon) throws Exception {
        if (lat < 0 || lat > 90) {
            throw new IllegalArgumentException("Latitude must be between -90 and 90");
        } else if (lon < -180 || lon >= 180) {
            throw new IllegalArgumentException("Longitude must be between -180 and 180");
        }

        int degrees = (int) Math.round(Math.toRadians(lat));

        switch (Direction.valueOf(degrees)) {
            case NORTH:
                return Direction.NORTH;
            case SOUTH:
                return Direction.SOUTH;
            case EAST:
                return Direction.EAST;
            case WEST:
                return Direction.WEST;
        }
    }

    private static final Position POSITION = newPosition(42.1234L);

    public static void testGetDirection() throws Exception {
        assertEquals(Direction.NORTH, getDirection(45.0L, -110.1234L));
        assertEquals(Direction.SOUTH, getDirection(-180.4567L, 90));
        assertTrue(newPosition(-1).hasLength);
        assertFalse(newPosition(1).hasLength);
    }

    public static void testGetDirection2() throws Exception {
        Position position = newPosition(45.1234L);
