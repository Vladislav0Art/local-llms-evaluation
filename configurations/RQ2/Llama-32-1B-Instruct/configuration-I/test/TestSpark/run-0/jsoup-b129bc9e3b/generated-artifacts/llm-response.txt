import java.util.HashMap;
import java.util.Map;

public class Document {
    private Map<String, String> data;

    public void put(String key, String value) {
        if (data == null) {
            data = new HashMap<>();
        }
        data.put(key, value);
    }

    public String get(String key) {
        return data.getOrDefault(key, "");
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        for (Map.Entry<String, String> entry : data.entrySet()) {
            sb.append(entry.getKey()).append(": ").append(entry.getValue()).append("\n");
        }
        return sb.toString();
    }
}

public class PublicRequest extends Request {
    private Document document;

    public PublicRequest(Document document) {
        this.document = document;
    }

    @Override
    public Response post(String dataKeyval) throws Exception {
        Map<String, String> requestData = new HashMap<>();
        for (String key : dataKeyval.keySet()) {
            requestData.put(key, document.get(key));
        }
        return new Response(document().doc());
    }
}

public class PublicMethod {
    private Document document;

    public PublicMethod(Document document) {
        this.document = document;
    }

    @Override
    public String doc() {
        return document.toString();
    }
}
