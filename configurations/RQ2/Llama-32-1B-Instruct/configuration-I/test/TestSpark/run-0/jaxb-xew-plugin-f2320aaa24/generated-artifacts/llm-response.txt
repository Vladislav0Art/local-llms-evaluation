public class TestClass {
    private String input;

    public String getMethod2() {
        return "Input is : " + input;
    }

    public void setMethod1(String value) {
        this.input = value;
    }
}

public class Main {
    public static void main(String[] args) {
        // Create an instance of TestClass
        TestClass test = new TestClass();

        // CallgetMethod2() and METHOD1() on the same object
        System.out.println("\nBefore method call:");
        System.out.println(test.getMethod2()); 
        System.out.println(test.getMethod1());

        // CallMETHOD1() with different input arguments
        String output = "Tested successfully";
        test.setMethod1(output);
        System.out.println("After method call and updating the object's value");

        // CallgetMethod2() again to verify changes made by METHOD1()
        System.out.println("\nOutput after second method call:");
        System.out.println(test.getMethod2()); 
    }
}
