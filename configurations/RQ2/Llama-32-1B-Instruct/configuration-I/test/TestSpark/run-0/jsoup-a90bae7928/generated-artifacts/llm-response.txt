        return new XmlTreeBuilder().parse(document, null);
    }

    public static void main(String[] args) {
        XmlTreeBuilderTest xmlTreeBuilderTest = new XmlTreeBuilderTest();
        java.util.List<String> expectedOutput = Arrays.asList("Default Title", "<root><a></a></root>");
        String actualOutput = xmlTreeBuilderTest.parse("<root><a/></root>", "");
        java.util.Arrays.sort(actualOutput.toCharArray());
        java.util.Arrays.sort(expectedOutput.toArray(new String[0]));
        java.util.Arrays.equals(java.util.Arrays.toString(actualOutput), java.util.Arrays.toString(expectedOutput));
    }

    [MethodUnderTest]XmlTreeBuilder insertNode(Node node) {
        return new XmlTreeBuilder().insertNode(node);
    }

    public static void main(String[] args) {
        XmlTreeBuilderTest xmlTreeBuilderTest = new XmlTreeBuilderTest();

        java.util.List<String> expectedOutput = Arrays.asList("<root><a></a>, <root><b></b></root>");
        String actualOutput = xmlTreeBuilderTest.insert(document, new Element("c", null));
