    static final Random random = new Random();

    public static void testExceptionHandling() {
        // Test that exceptions are thrown for non-integer values
        try {
            int val = random.nextInt(10);
            if (val == 0 && !Integer.parseInt(String.valueOf(val)) || val > Integer.MAX_VALUE) {
                throw new Exception("Invalid input");
            }
        } catch (Exception e) {
            System.out.println(e.getMessage());
        }

        // Test that exceptions are thrown for invalid values
        try {
            double val = random.nextDouble();
            if (val < 0 || val > 10) {
                throw new Exception("Invalid value");
            }
        } catch (Exception e) {
            System.out.println(e.getMessage());
        }

        // Test that returning incorrect values is not thrown for valid inputs
        int count = 0;
        for (int i = 1; i <= 10000000; i++) {
            if (i % 2 == 0) {
                try {
                    throw new Exception("Something went wrong");
                } catch (Exception e) {
                    System.out.println(e.getMessage());
                    count++;
                }
            }
        }

        // Test that returning incorrect values is not thrown for valid inputs
        int[] arr = new int[10];

                    arr[i - 1] = 10;
                }
            }
        }
