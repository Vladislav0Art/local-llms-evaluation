import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class Response {
    private String status_code;
    private String body;

    public Response(String status_code, String body) {
        this.status_code = status_code;
        this.body = body;
    }

    public String getStatus_code() {
        return status_code;
    }

    public void setStatus_code(String status_code) {
        this.status_code = status_code;
    }

    public String getBody() {
        return body;
    }

    public void setBody(String body) {
        this.body = body;
    }
}

public class TestResponse extends org.jsoup.Jsoup<Request> {
    @Override
    public Response parse(String html) {
        Document doc = Jsoup.parse(html);
        Request request = new Request(doc.select("title").first().text(), doc.text());
        return new Response(request.getStatus_code(), request.getBody());
    }

    @Test
    public void testGet() {
        String html = "<html><body>Hello World!</body></html>";
        TestResponse response = parse(html);
        System.out.println(response.getStatus_code());
        assertEquals(200, response.getStatus_code());
        System.out.println(response.getBody());
        assertEquals("<html><body>Hello World!</body></html>", response.getBody());
    }

    @Test
    public void testPost() {
        String html = "<html><body>Hello World!</body></html>";
        TestResponse response = parse(html);
        System.out.println(response.getStatus_code());
        assertEquals(201, response.getStatus_code());
        System.out.println(response.getBody());
        assertEquals("<html><body>Hello World!</body></html>", response.getBody());
    }
}
