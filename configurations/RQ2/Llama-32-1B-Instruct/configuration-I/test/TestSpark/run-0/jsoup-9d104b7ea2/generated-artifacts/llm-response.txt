        // Test outerHtmlHead() and outerHtmlTail() methods
        appendStringBuilder();
        DocumentOutputSettings out = new DocumentOutputSettings(appendStringBuilder(), true);
        node.outerHtmlHead(out);
        assertEquals("Hello World", (String) appendStringBuilder.toString());
        appendStringBuilder.clear();

        // Test toString() method
        appendStringBuilder();
        String result = "Hello World";
        assertEquals(result, node.toString());

        appendStringBuilder();
    }

    }

        throw new UnsupportedOperationException();
    }

    public static TextNode createFromEncoded(String encodedText) {
        Element node = new Element("text");
        node.text(encodedText);
        return new TextNode(node);
    }
}

public class DocumentOutputSettings {

    private Appendable accum;
    private boolean encode;

    public DocumentOutputSettings(Appendable accum, boolean encode) {
        this.accum = accum;
        this.encode = encode;
    }

    public static void appendAppendable(Appendable appendable) {
        DocumentOutputSettings out = new DocumentOutputSettings(appendable, false);
        out.accumulate.appendText(String.valueOf(out.accum));
    }
}
