import org.junit.Test;
import static org.mockito.Mockito.*;
import java.util.Arrays;

public class OnlineStockSpanTest {

    @Test
    public void test_next_withNoPrices_Returns0() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        when(onlineStockSpan.next()).thenReturn(1);
        int result = onlineStockSpan.next(100);
        assertEquals(1, result);
    }

    @Test
    public void test_nextWithOnePrice_Returns1() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        when(onlineStockSpan.next()).thenReturn(1);
        int result = onlineStockSpan.next(50);
        assertEquals(1, result); // [1]
    }

    @Test
    public void test_nextWithMultiplePrices_ReturnsCorrectSum() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        when(onlineStockSpan.next()).thenReturn(2);
        when(onlineStockSpan.next()).thenReturn(3);
        int[] prices = new int[]{50, 150};
        onlineStockSpan.setPrices(prices);
        int result = onlineStockSpan.calculateSpans(new int[]{});
        assertEquals(Arrays.asList(1, 2, 3).size(), result); // [1, 2, 3]
    }

    @Test
    public void test_calculateSpans_withNoPrices_Returns() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        when(onlineStockSpan.calculateSpans(new int[]{})).thenReturn(new int[]{0});
        OnlineStockSpan result = onlineStockSpan;
        int[] resultArray = result.calculateSpans(new int[]{});
        assertEquals(resultArray, new int[]{0});
    }

    @Test
    public void test_calculateSpans_withOnlyOnePrice_Returns1() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        when(onlineStockSpan.next()).thenReturn(1);
        onlineStockSpan.setPrices(new int[]{100});
        int result = onlineStockSpan.calculateSpans(new int[]{});
        assertEquals(Arrays.asList(1, 0).size(), result); // [1, 0]
    }

    @Test
    public void test_calculateSpans_withMultiplePrices_ReturnsCorrectSum() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        when(onlineStockSpan.next()).thenReturn(2);
        when(onlineStockSpan.next()).thenReturn(3);
        onlineStockSpan.setPrices(new int[]{50, 150});
        int result = onlineStockSpan.calculateSpans(new int[]{});
        assertEquals(Arrays.asList(1, 2).size(), result); // [1, 2]
    }

    public void setPrices(int[] prices) {
        this.prices = prices;
    }

    @Test
    public void test_calculateSpans_withMultiplePrices_Sorted() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        when(onlineStockSpan.next()).thenReturn(2);
        when(onlineStockSpan.next()).thenReturn(1);
        onlineStockSpan.setPrices(new int[]{50, 150});
        onlineStockSpan.setPrices(new int[]{'a', 'b'});
        int result = onlineStockSpan.calculateSpans(new int[]{});
        assertEquals(Arrays.asList(1, 2).size(), result); // [1, 2]
    }
}
