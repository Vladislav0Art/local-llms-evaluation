import org.jsoup.nodes.Document;
import org.jsoup.select.Elements;

public class CommentTest {

    public static Document createDocument() {
        return new Document();
    }

    public static void main(String[] args) {
        Document document = createDocument();

        // Create a comment node and add it to the document.
        Element comment = new Element("comment");
        comment.addAttribute("data", "This is another comment.");

        // Add the comment node to the document.
        document.appendChild(comment);

        // Remove the comment node from the document after use.
        comment.remove();
    }

    @Test
    public void nodeNameShouldReturnCommentNode() {
        Element comment = new Element("comment");
        assertEquals(Comment.class, comment.nodeName().getClass());
    }

    @Test
    public void getDataShouldReturnCommentData() {
        String data = "This is a comment.";
        assertEquals(data, comment.getData());
    }

    @Test
    public void setDataShouldSetCommentData() {
        comment.setData("New data.");
        assertEquals("New data.", comment.getData());
    }

    @Test
    public void outerHtmlHeadShouldAddCommentNodeToDocument() throws IOException {
        Document document = createDocument();

        // Create a head node and add it to the document.
        Element head = new Element("head");
        document.appendChild(head);

        // Add the comment node to the head node in the output stream.
        Appendable output = document.getOutput();
        output.writeHead(0, 1, null); // Replace 1 with the length of the string
        output.writeComment(0, 2, new CharacterData(comment));
    }

    @Test
    public void outerHtmlTailShouldAddCommentNodeToDocument() throws IOException {
        Document document = createDocument();

        // Create a body node and add it to the document.
        Element body = new Element("body");
        document.appendChild(body);

        // Add the comment node to the body node in the output stream.
        Appendable output = document.getOutput();
        output.writeBody(0, 1, null); // Replace 1 with the length of the string
        output.writeComment(0, 2, new CharacterData(comment));
    }

    @Test
    public void toStringShouldReturnCommentString() {
        String data = "This is a comment.";
        assertEquals(data, comment.toString());
    }

    @Test
    public void cloneShouldReturnCloneOfCommentNode() {
        Element comment = new Element("comment");

        assertEquals(comment, cloneComment);
    }

    @Test
    public void isXmlDeclarationShouldReturnFalse() {
        assertEquals(false, comment.isXmlDeclaration());
