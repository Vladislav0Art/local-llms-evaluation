import org.junit.Test;
import org.mockito.Mockito;

public class GalileoProtocolDecoderTest {

    @Test
    public void testDecode_GalileoMessage() throws Exception {
        // Arrange
        Channel channel = Mockito.mock(Channel.class);
        SocketAddress remoteAddress = Mockito.mock(SocketAddress.class);
        Object msg = "Hello, World!".getBytes();

        // Act
        Object decodedMsg = new GalileoProtocolDecoder().decode(channel, remoteAddress, msg);

        // Assert
        assertEquals("Hello, World!", decodedMsg);
    }

    @Test
    public void testDecode_GalileoMultipleMessages() throws Exception {
        // Arrange
        Channel channel = Mockito.mock(Channel.class);
        SocketAddress remoteAddress1 = Mockito.mock(SocketAddress.class);
        SocketAddress remoteAddress2 = Mockito.mock(SocketAddress.class);
        Object msg1 = "Hello, World!".getBytes();
        Object msg2 = "Hi, There!".getBytes();

        // Act
        Object[] decodedMsgs = new GalileoProtocolDecoder().decode(channel, remoteAddress1, msg1);
        Object[] decodedMsgs2 = new GalileoProtocolDecoder().decode(channel, remoteAddress2, msg2);

        // Assert
        assertEquals(2, decodedMsgs.length);
        Mockito.verify(decodedMsgs).equals(msg1);
        Mockito.verify(decodedMsgs2).equals(msg2);
    }

    @Test
    public void testDecode_GalileoEmptyMessage() throws Exception {
        // Arrange
