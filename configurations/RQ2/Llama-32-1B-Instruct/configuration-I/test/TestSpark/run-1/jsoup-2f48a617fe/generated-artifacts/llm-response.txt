    public static final String TAG = "comment";
    public static final String XML_DECLARATION_TAG = "<" + TAG + ">";

    protected Document asXmlDeclaration() {
        return new Document();
    }

    @Override
    public void toString() {
        Element element = getRoot();
        if (element instanceof TextNode) {
            TextNode text = (TextNode) element;
            String textContent = text.text().trim();

            assertTrue(textContent.equals("Comment"));
        } else {
            throw new RuntimeException("Unexpected node type");
        }

        return "";
    }
}

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Comment extends LeafNode {

    public static final String TAG = "comment";

    @Override
    public Document asXmlDeclaration() {
        return new Document();
    }

    @Override
    protected String toString() {
        Element element = getRoot();
        if (element instanceof TextNode) {
            TextNode text = (TextNode) element;
            return "<" + TAG + ">" + text.text() + "</" + TAG + ">";
        } else {
            throw new RuntimeException("Unexpected node type");
        }
    }

    public static void main(String[] args) {
        Document doc = Comment.asXmlDeclaration();
        System.out.println(doc.toString());

        assertTrue(Comment Clown().toString());
        assertFalse(Comment Clones().isXmlDeclaration());
        assertTrue(Comment Clone().asXmlDeclaration().isXmlDeclaration());
    }

    private static class Clown extends LeafNode {

        @Override
        protected String toString() {
            return "<" + TAG + ">" + "</" + TAG + "}";
        }
    }

    private static class Clones extends LeafNode {

        @Override
        public boolean isXmlDeclaration() {
            return false;
        }

        @Override
        protected Document asXmlDeclaration() {
            return null;
        }
    }

    private static class Clone extends LeafNode {

        @Override
        protected String toString() {
            return "<" + TAG + ">" + "</" + TAG + "}";
        }

        public static void main(String[] args) {
            Clone clone = new Clone();
            System.out.println(clone.toString());
        }
    }
}

class LeafNodeTest {

    public static void main(String[] args) {
        Document doc = Comment.asXmlDeclaration();
        System.out.println(doc.toString());

        assertTrue(Comment Clown().toString());
        assertFalse(Comment Clones().isXmlDeclaration());
        assertTrue(Comment Clone().asXmlDeclaration().isXmlDeclaration());
    }
}
