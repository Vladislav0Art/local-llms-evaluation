import org.jsoup.nodes.*;
import org.jsoup.select.Elements;
import static org.mockito.Mockito.mock;

public class ElementTest {

    private Node node;
    private Document document;
    private String tagName;
    private String tagValue;

    public void setup() {
        document = new Document();
        node = mock(Node.class);
        tagName = "div";
        tagValue = "Hello World!";

        assertEquals(tagName, element.tagName());
        assertEquals(tagValue, element.text());
    }

    public void testText() {
        String text = "Hello World!";
        Element element = new Element(tagName, text, null);
        assertEquals(text, element.text());
    }

    @Test
    public void testHasText() {
        Element element = new Element(tagName, tagValue, null);
        assertTrue(element.hasText());
    }

    public void testData() {
        String data = "Hello World!";
        Element element = new Element(tagName, data, null);
        assertEquals(data, element.data());
    }

    @Test
    public void testHasData() {
        Element element = new Element(tagName, tagValue, data);
        assertTrue(element.hasText());
        assertTrue(element.hasData());
    }

    public void testGetElementsByTag() {
        Elements elements = document.select(".test");
        assertEquals(1, elements.size());
        assertNotNull(elements.get(0));
    }

    @Test
    public void testGetElementsByClass() {
        Element element = new Element(tagName + ":tag", tagValue, null);
        assertEquals(element, document.select("." + tagName).first());
    }

    @Test
    public void testGetElementsByAttribute() {
        Elements elements = document.select("a[href]");
        assertEquals(1, elements.size());
        assertNotNull(elements.get(0));
    }

    @Test
    public void testGetElementsByAttributeValue() {
        Elements elements = document.select("p[attr-length='6']");
        assertEquals(1, elements.size());
        assertNotNull(elements.get(0));
    }

    @Test
    public void testGetElementsByAttributeValueMatching() {
        Element element = new Element(tagName + ":tag", tagValue, null);
        assertTrue(document.select("a[href^=https://").first().hasText(element.text()));
    }

    @Test
    public void testGetElementsContainingOwnText() {
        Elements elements = document.select("p.text");
        assertEquals(1, elements.size());
        assertNotNull(elements.get(0));
    }

    @Test
    public void testGetAllElements() {
        Elements elements = document.getAllElements();
        assertEquals(2, elements.size());
        assertNotNull(elements.get(0));
        assertNotNull(elements.get(1));
    }

    public void testPreserveWhitespace() {
        Element element = new Element(tagName + ":tag", tagValue);
        preserveWhitespace(element).addNode(new Text("Hello World!"));
        assertEquals("Hello World!", element.text());
    }
}
