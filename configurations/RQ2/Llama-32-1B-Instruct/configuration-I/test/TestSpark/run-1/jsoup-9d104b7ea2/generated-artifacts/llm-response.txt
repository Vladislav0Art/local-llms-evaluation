import org.junit.jupiter.api.Test;
import static org.mockito.Mockito.mock;
import static org.mockito.Mockito.when;

public class TextNodeTest {

    @Test
    public void testnodeName() {
        String text = "Hello World";
        TextNode node = new TextNode(text);
        assertEquals("TextNode", node.getNodeName());
        assertEquals("HelloWorld", node.getText());
    }

    @Test
    public void testText() {
        String text = "Hello World";
        TextNode node = new TextNode(text);
        assertEquals(text, node.getText());
        assertEquals("", node.getWholeText());
        assertEquals(10, node.isBlank().length());
        assertNotNull(node.splitText(0));
    }

    @Test
    public void testGetWholeText() {
        String text = "Hello World";
        TextNode node = new TextNode(text);
        assertEquals("Hello World", node.getWholeText());
    }

    @Test
    public void testIsBlank() {
        String text = "Hello World";
        TextNode node = new TextNode(text);
        assertTrue(node.isBlank());
        assertFalse(node.isBlank());
    }

    @Test
    public void testSplitText() {
        String text = "Hello World";
        TextNode node = new TextNode(text);
        when(node.splitText(0)).thenReturn(new StringBuffer());
        assertEquals("HelloWorld", ((StringBuffer) node.splitText(0)).toString());
    }

    @Test
    public void testOuterHtmlHead() {
        String text = "Hello World";
        TextNode node = new TextNode(text);
        when(node.outerHtmlHead(any(Appendable.class), any(int.class), any(Document.OutputSettings.class))).thenReturn("Hello World");
    }

    @Test
    public void testOuterHtmlTail() {
        String text = "Hello World";
        TextNode node = new TextNode(text);
        when(node.outerHtmlTail(any(Appendable.class), any(int.class), any(Document.OutputSettings.class))).thenReturn("World");
    }

    @Test
    public void testToString() {
        String text = "Hello World";
        TextNode node = new TextNode(text);
        assertEquals("TextNode", node.toString());
    }

    @Test
    public void testClone() {
        String text = "Hello World";
        TextNode node = new TextNode(text);
        TextNode clonedNode = (TextNode) node.clone();
        assertEquals(text, clonedNode.getText());
        assertEquals("", clonedNode.getWholeText());
        assertEquals(10, clonedNode.isBlank().length());
    }

    @Test
    public void testcreateFromEncoded() {
        String encodedText = "Hello World";
        TextNode node = new TextNode(createFromEncoded(encodedText));
        assertEquals(encodedText, node.getText());
    }

    @Test
    public void testNormaliseWhitespace() {
        String text = "   Hello   World  ";
        TextNode node = new TextNode(text);
        assertEquals("  HelloWorld  ", node.stripLeadingWhitespace().toString());
    }

    @Test
    public void testStripLeadingWhitespace() {
        String text = "Hello World";
        TextNode node = new TextNode(text);
        assertEquals("Hello World", node.stripLeadingWhitespace().toString());
    }

    @Test
    public void testLastCharIsWhitespace() {
        StringBuilder sb = new StringBuilder();
        sb.append('a');
        sb.append('b');
        sb.append('c');
        sb.append('d');
        TextNode node = new TextNode(createFromEncoded(sb.toString()));
        assertTrue(node.lastCharIsWhitespace(sb));
    }

    @Test
    public void testNormaliseWhitespaceEmpty() {
        TextNode node = new TextNode("");
        assertEquals("", node.stripLeadingWhitespace().toString());
    }
}
