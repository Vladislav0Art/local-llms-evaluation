import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.params.ParameterizedTest;
import org.junit.jupiter.params.provider.MethodSource;

import java.util.Locale;
import java.util.function.Function;

public class Gt06ProtocolDecoderTests {

    private BaseProtocolDecoder decoder;

    @BeforeEach
    public void init() {
        decoder = new Gt06ProtocolDecoder(new Protocol());
    }

    @ParameterizedTest
    @MethodSource("getScenarios")
    public <T> void testDecodeGps(T position, ByteBuf buf, boolean hasLength, TimeZone timezone) throws Exception {
        Function<Position, Boolean> func = getFunction(position);
        assertTrue(decoder.decodeGps(position, buf, hasLength, true, false, timezone).equals(func.apply(position)));
    }

    @ParameterizedTest
    @MethodSource("getScenarios")
    public <T> void testDecodeGpsNoSatellites(T position, ByteBuf buf, boolean hasLength, TimeZone timezone) throws Exception {
        assertTrue(decoder.decodeGps(position, buf, hasLength, false, false, timezone).equals(true));
    }

    @ParameterizedTest
    @MethodSource("getScenarios")
    public <T> void testDecodeSpeed(T position, ByteBuf buf, boolean hasLength, TimeZone timezone) throws Exception {
        assertTrue(decoder.decodeGps(position, buf, hasLength, false, true, timezone).equals(true));
    }

    @ParameterizedTest
    @MethodSource("getScenarios")
    public <T> void testDecodeGpsNoSpeed(T position, ByteBuf buf, boolean hasLength, TimeZone timezone) throws Exception {
        assertTrue(decoder.decodeGps(position, buf, hasLength, true, false, timezone).equals(true));
    }

    @ParameterizedTest
    @MethodSource("getScenarios")
    public <T> void testDecodeGpsNoTimezone(T position, ByteBuf buf, boolean hasLength, boolean hasSatellites) throws Exception {
        assertTrue(decoder.decodeGps(position, buf, hasLength, false, true, false).equals(true));
    }

    @ParameterizedTest
    @MethodSource("getScenarios")
    public <T> void testDecodeNoPosition(T position, ByteBuf buf, boolean hasLength, boolean hasSatellites) throws Exception {
        assertTrue(decoder.decodeGps(null, buf, hasLength, false, true, false).equals(true));
    }

    @ParameterizedTest
    @MethodSource("getScenarios")
    public <T> void testDecodeNoByteBuf(T position, Boolean hasLength, boolean hasSatellites, boolean hasSpeed) throws Exception {
        assertTrue(decoder.decodeGps(position, null, hasLength, false, true, false).equals(true));
    }

    private Function<Position, Boolean> getFunction(Position position) {
        return msg -> ((Position) msg).getLatitude() == position.getLatitude();
    }
}
