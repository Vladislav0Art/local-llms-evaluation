====================== Iteration #3 ======================
Here is a Java program that demonstrates how to create and use classes, variables, methods, and constructors:

```java
public class Main {
    public static void main(String[] args) {
        TestClass testInstance1 = new TestClass();
        System.out.println("Test Instance 1: " + testInstance1.testMethod());

        AnotherTestClass anotherTestInstance2 = new AnotherTestClass();
        System.out.println("Another Test Instance 2: " + anotherTestInstance2.testMethod());
    }
}

class TestClass {
    public String testMethod() {
        return "This is a test method";
    }

    public int publicVariable = 10;
}

class AnotherTestClass extends TestClass {
    public String anotherTestMethod() {
        return "This is an another test method";
    }
}
```

In this code:

1. We define two classes: `TestClass` and `AnotherTestClass`.
2. In the `Main` class, we create instances of both classes.
3. We use the `.testMethod()` and `.anotherTestMethod()` methods to call the corresponding methods from each instance.

This will output:

```
Test Instance 1: This is a test method
Another Test Instance 2: This is an another test method
```

**Encapsulation**

```java
public class BankAccount {
    private double balance;

    public void deposit(double amount) {
        this.balance += amount;
    }

    public void withdraw(double amount) {
        if (this.balance >= amount) {
            this.balance -= amount;
        } else {
            System.out.println("Insufficient funds");
        }
    }

    public static void main(String[] args) {
        BankAccount account = new BankAccount();
        account.deposit(100);
        System.out.println(account.balance); // Print the balance

        account.withdraw(50);
        System.out.println(account.balance); // Print the updated balance
    }
}
```

**Composition**

```java
public class Car {
    private String brand;
    private int year;

    public Car(String brand, int year) {
        this.brand = brand;
        this.year = year;
    }

    public void displayInfo() {
        System.out.println("Brand: " + brand + ", Year: " + year);
    }
}

public class Motorcycle extends Car {
    public Motorcycle(String brand, int year, String color) {
        super(brand, year);
        this.color = color;
    }

    @Override
    public void displayInfo() {
        System.out.println("Brand: " + brand + ", Year: " + year + ", Color: " + color);
    }
}

public class Main {
    public static void main(String[] args) {
        Car car = new Car("Toyota", 2015);
        Motorcycle motorcycle = new Motorcycle("Honda", 2020, "Red");
        car.displayInfo();
        motorcycle.displayInfo(); // Call the method to display information on the motorcycle
    }
}
```
