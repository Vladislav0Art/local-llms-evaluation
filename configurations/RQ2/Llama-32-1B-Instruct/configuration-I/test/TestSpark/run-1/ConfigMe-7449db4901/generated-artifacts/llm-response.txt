import org.junit.jupiter.api.Test;
import org.mockito.Mockito;

public class CommentsConfigurationTest {

    @Test
    public void commentsConfigurationNewlineOnlySetComment() {
        // Arrange
        String path = "path";
        String commentLines = Mockito.mock(String.class);

        // Act
        CommentsConfiguration configuration = new CommentsConfiguration();
        configuration.setComment(path, commentLines);

        // Assert
        Mockito.verify(configuration, Mockito.times(1)).setComment(Mockito.anyString(), Mockito.any(String[].class));
    }

    @Test
    public void commentsConfigurationMultipleCommentsSetComment() {
        // Arrange
        String path = "path";
        String[] commentLines = {"line1\n", "line2\n"};
        Map<String, List<String>> comments = new HashMap<>();
        comments.put(path, commentLines);

        // Act
        CommentsConfiguration configuration = new CommentsConfiguration();
        configuration.setComment(path, Mockito.mock(String.class), commentLines);

        // Assert
        Mockito.verify(configuration, Mockito.times(1)).setComment(Mockito.anyString(), Mockito.any(String[].class));
    }

    @Test
    public void commentsConfigurationAllCommentsSetComment() {
        // Arrange
        String path = "path";
        String[] commentLines = {"line1\n", "line2\n"};
        Map<String, List<String>> comments = new HashMap<>();
        comments.put(path, commentLines);

        // Act
        CommentsConfiguration configuration = new CommentsConfiguration();
        Mockito.when(configuration.getAllComments()).thenReturn(comments);
        configuration.setComment(path, Mockito.mock(String.class), commentLines);

        // Assert
        Mockito.verifyAll Mockito.times(2).setComment(Mockito.anyString(), Mockito.any(String[].class));
    }

    @Test
    public void commentsConfigurationNoCommentSetComment() {
        // Arrange
        String path = "path";
        String[] commentLines = {"line1\n", "line2\n"};
        Map<String, List<String>> comments = new HashMap<>();
        comments.put(path, commentLines);

        // Act
        CommentsConfiguration configuration = new CommentsConfiguration();
        Mockito.when(configuration.getAllComments()).thenReturn(comments);
        configuration.setComment(path, Mockito.mock(String.class), commentLines);

        // Assert
        Mockito.verifyAll Mockito.times(1).setComment(Mockito.anyString(), Mockito.any(String[].class));
    }
}
