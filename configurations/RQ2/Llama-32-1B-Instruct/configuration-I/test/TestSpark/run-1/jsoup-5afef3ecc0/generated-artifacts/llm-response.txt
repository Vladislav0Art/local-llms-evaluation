import org.junit.jupiter.api.Test;
import org.jsoup.helper.UrlBuilder;
import org.jsoup.nodes.Document;

public class UrlBuilderTest {

    @Test
    public void testBuildWithEmptyInputUrl() {
        final Document document = new Document();
        String urlString = "https://example.com";
        UrlBuilder urlBuilder = new UrlBuilder(urlString);
        URL actualUrl = urlBuilder.build();
        Assert.assertEquals("https://example.com", actualUrl.toString());
    }

    @Test
    public void testBuildWithInvalidKeyVal() {
        final Document document = new Document();
        String urlString = "https://example.com";
        UrlBuilder urlBuilder = new UrlBuilder(urlString);
        try {
            urlBuilder.appendKeyVal(null);
            Assert.fail("Expected UnsupportedOperationException");
        } catch (UnsupportedEncodingException e) {
            // expected
        }
    }

    @Test
    public void testAppendKeyValue() throws Exception {
        final Document document = new Document();
        String urlString = "https://example.com";

        urlBuilder.appendKeyVal("key", "value");
        URL actualUrl = urlBuilder.build();
        Assert.assertEquals("https://example.com?key=value", actualUrl.toString());

public class Test {

    public static void main(String[] args) {
        System.out.println("This is the main method");
        
        // Variables and Data Types
        int age = 20;
        String name = "John Doe";
        boolean isAdmin = true;

        // Conditional Statements (if-else)
        if (age >= 18 && isAdmin) {
            System.out.println(name + " is eligible to vote.");
        } else {
            System.out.println(name + " is not eligible to vote.");
        }

        // Loops
        for (int i = 0; i < 5; i++) {
            System.out.println("Iteration " + i);
        }
        
        // Arrays and Lists
        String[] colors = {"Red", "Green", "Blue"};
        System.out.println(Arrays.toString(colors));

        // Object-Oriented Programming (Classes, Objects)
        Person person = new Person();
        person.name = "Jane Doe";
        person.age = 25;
        System.out.println(person.getName() + ", " + person.getAge());

        // Methods
        greetPerson(greetName);

        // Method Overloading
        public void greet(String name) {
            System.out.println("Hello, " + name);
        }
        public void greet(int age) {
            System.out.println("Hi there, " + age + " years old!");
        }

        greet("John");
    }

    public static void greetName() {
        System.out.println("Hello! My name is John.");
    }

    // Method Overriding
    public interface Animal {
        void sound();
    }

    public class Dog implements Animal {
        @Override
        public void sound() {
            System.out.println("Woof!");
        }
    }

    public static String greet(String name) {
        return "Hello, " + name + "!";
    }
}

- Loops: For loop to iterate through an array or range of numbers
- Arrays and Lists: Java's built-in arrays and lists data structures
- Object-Oriented Programming (Classes, Objects): Classes and objects in Java
- Methods: Defining methods with parameters and return types
- Method Overloading: Defining multiple methods with the same name but different parameters
- Method Overriding: Defining a subclass to override a method from its superclass
