    }

    @Test
    public void test-parseFragment() {
        String inputFragment = "<p>This is a paragraph.</p>";
        Element context = null;

        builder.parseFragment(inputFragment, context, baseUri, parser).assertNotNull(context);
    }

    public void test-process() {

        assertEquals(true, result);
