        assertEqual(0, calculateYearDifference(date, forTest));
        assertEquals(1, calculateMonthDifference(date, forTest));
        assertEquals(5, calculateDayDifference(date, forTest));

        forTest = new LocalDate(2022, 1, 15);
    }

    public double calculateYearDifference(LocalDate date1, LocalDate date2) {
        return Math.abs((date1.getYear() - date2.getYear()));
    }

    public int calculateMonthDifference(LocalDate date1, LocalDate date2) {
        int diff = Math.abs((date1.getMonthValue() - date2.getMonthValue()));
        if (diff == 12) {
            return 0;
        }
        if (diff % 12 == 3 || diff % 12 == 4) {
            return 1;
        }
        if (diff % 12 == 6 || diff % 12 == 9 || diff % 12 == 0) {
            return 2;
        }
        return -1; // month difference
    }

    public int calculateDayDifference(LocalDate date1, LocalDate date2) {
        int diff = Math.abs((date1.getDayOfMonth() - date2.getDayOfMonth()));
        if (diff == 31 || diff == 30 || diff == 29) {
            return 0;
        }
        if ((diff % 4 != 0 && diff % 100 == 0) || diff % 400 == 0) {
            return -1; // leap year difference
        } else if (diff % 4 == 3 || diff % 4 == 2) {
            return -2; // century and month difference
        }
        return 1;
    }

    public void assertEqual(int expected, int actual) {
        if (actual != expected) {
            throw new AssertionError("Expected " + expected + ", got " + actual);
        }
    }
}

    public double calculateYearDifference(LocalDate date1, LocalDate date2) {
        return Math.abs((date1.getYear() - date2.getYear()));
    }

    public int calculateMonthDifference(LocalDate date1, LocalDate date2) {
        if (date2.getMonthValue() < date1.getMonthValue()) {
            return 6;
        } else if (date2.getMonthValue() == date1.getMonthValue() && date2.getDayOfMonth() > date1.getDayOfMonth()) {
            return 1;
        } else if (date2.getMonthValue() == date1.getMonthValue() && date2.getDayOfMonth() < date1.getDayOfMonth()) {
            return -1;
        } else if (date2.getMonthValue() != date1.getMonthValue()) {
            return -1;
        }
    }

    public int calculateDayDifference(LocalDate date1, LocalDate date2) {
        if ((date1.getDayOfMonth() == date2.getDayOfMonth()) && (date1.getMonthValue() == date2.getMonthValue())) {
            return 0;
        } else if (date1.getYear() != date2.getYear()) {
            return -4;
        } else if (date1.getMonthValue() < date2.getMonthValue()) {
            return 1;
        } else if (date1.getMonthValue() > date2.getMonthValue()) {
            return -1;
        }

package com.example.myapp;

import static org.junit.Assert.*;

public class MyClass {

    public String myMethod() {
        return "Hello World";
    }

    public static void main(String[] args) {
        MyClass obj = new MyClass();
        System.out.println(obj.myMethod());
    }
}

package com.example.myapp2;

import static org.junit.Assert.*;

public class MyClass2 {

    public void myMethod() {
        System.out.println("Hello World");
    }

    public static void main(String[] args) {

        obj.myMethod();
    }
}
