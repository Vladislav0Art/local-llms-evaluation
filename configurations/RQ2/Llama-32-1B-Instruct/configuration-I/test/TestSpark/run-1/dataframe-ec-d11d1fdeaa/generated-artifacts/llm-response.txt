import java.util.*;

public class DataFrame {
    private Map<String, Object[]> data;

    public DataFrame() {
        this.data = new HashMap<>();
    }

    // Public methods
    public void addColumn(String column) {
        this.data.put(column, new Object[][]{{null}, {null}});
    }

    public void setData(Object[] data) {
        for (Object row : data) {
            if (!this.data.containsKey(row)) {
                this.data.put(row, new Object[]{});
            }
        }
        for (Object[] row : data) {
            addColumn(row[0]);
        }
    }

    public void printColumns() {
        System.out.println("Columns: " + this.columns);
    }

    public void printData() {
        for (Map.Entry<String, Object[]> entry : data.entrySet()) {
            System.out.print(entry.getKey() + ": ");
            for (Object value : entry.getValue()) {
                if (value != null) {
                    System.out.print(value + ", ");
                } else {
                    System.out.print("null, ");
                }
            }
            System.out.println();
        }
    }

    // Private methods
    private void addData(Object[] row) {
        Object[] newRow = new Object[data.size() + 1];
        System.arraycopy(data.get(row[0]), 0, newRow, 0, data.size());
        newRow[newRow.length - 1] = row;
        this.data.put(row[0], newRow);
    }

    private void printDataBeforeIndex(int index) {
        for (int i = 0; i < index; i++) {
            System.out.print(this.columns.get(i) + ": ");
            for (Object value : this.data.getOrDefault(i, new Object[]{})) {
                if (value != null) {
                    System.out.print(value + ", ");
                } else {
                    System.out.print("null, ");
                }
            }
            System.out.println();
        }
    }

    private void printDataAfterIndex(int index) {
        for (int i = index; i < data.size(); i++) {
            System.out.print(this.columns.get(i) + ": ");
            for (Object value : this.data.get(i).getValue()) {
                if (value != null) {
                    System.out.print(value + ", ");
                } else {
                    System.out.print("null, ");
                }
            }
            System.out.println();
        }
    }

    private void checkIndex(int index) {
        if (!this.data.containsKey(index)) {
            throw new RuntimeException("Index out of bounds: " + index);
        }
    }

    // Other methods
    public String toString() {
        StringBuilder sb = new StringBuilder();
        for (String column : columns) {
            sb.append(column).append(": ");
        }
        for (Object[] row : data.values()) {
            for (Object value : row) {
                if (value != null) {
                    sb.append(value).append(", ");
                } else {
                    sb.append("null, ");
                }
            }
            sb.append("\n");
        }
        return sb.toString();
    }

        df.addColumn("Age");
