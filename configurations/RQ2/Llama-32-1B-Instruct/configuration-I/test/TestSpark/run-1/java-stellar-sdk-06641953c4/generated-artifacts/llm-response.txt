import javax.crypto.Mac;
import javax.crypto.spec.SecretKeySpec;
import java.security.KeyPair;
import java.security.PrivateKey;
import java.security.PublicKey;

public class TestKeyPair {

    private static final String ALGORITHM = "HmacSHA256";

    public static KeyPair generatePublicKey() {
        try {
            // Generate a new key pair
            KeyPair keyPair = KeyPairGenerator.getInstance(ALGORITHM).generateKeyPair();

            // Get the private and public keys
            PrivateKey privateKey = (PrivateKey) keyPair.getPrivate();
            PublicKey publicKey = (PublicKey) keyPair.getPublic();

            return new KeyPair(publicKey, privateKey);
        } catch (Exception e) {
            throw new RuntimeException("Failed to generate key pair", e);
        }
    }

    public static String generatePrivateKey(String secretSeed) throws Exception {
        try {
            // Derive a private key from the secret seed
            SecretKeySpec secretKeySpec = new SecretKeySpec(secretSeed.getBytes(), ALGORITHM);
            Mac mac = Mac.getInstance(ALGORITHM);

            // Derive a signature using HmacSHA256
            byte[] digest = mac.doFinal(secretKeySpec.getEncoded());

            return "privateKey" + (new String(digest));
        } catch (Exception e) {
            throw new RuntimeException("Failed to derive private key", e);
        }
    }

    public static void main(String[] args) throws Exception {
        KeyPair publicKey = generatePublicKey();
        PublicKey publicKey2 = publicKey.getPublic();

        System.out.println("Generated Public Key: " + publicKey.toString());
        System.out.println("Generated Private Key 1:");
        System.out.println(publicKey2);

        String secretSeed = "my_secret_seed";
        String privateKey1 = generatePrivateKey(secretSeed);
        System.out.println("Generated Private Key 2: " + privateKey1);

        assertEquals(publicKey, new KeyPair(publicKey.getPublic(), publicKey.getPrivate()));
    }
}
