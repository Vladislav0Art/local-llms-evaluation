import org.junit.jupiter.api.Test;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;

public class DirectivesBuilderTest {

    @Mock
    private AnnotatedElement annotatedElement;

    @Mock
    private ProcessingElementsContainer processingElementsContainer;

    @Test
    public void build_returns_empty_array_when_no_directives() {
        // Arrange
        when(annotatedElement.getDirectives()).thenReturn(new GraphQLDirective[0]);

        // Act
        DirectivesBuilder builder = new DirectivesBuilder(annotatedElement, processingElementsContainer);

        // Assert
        assertEquals([], builder.build());
    }

    @Test
    public void build_returns_directive_when_directive_found() {
        // Arrange
        when(annotatedElement.getDirectives()).thenReturn(new GraphQLDirective[] {new GraphQLDirective()});

        // Act
        DirectivesBuilder builder = new DirectivesBuilder(annotatedElement, processingElementsContainer);
        Directives[] directives = builder.build();

        // Assert
        assertEquals(Arrays.asList(new GraphQLDirective()), directives);
    }

    @Test
    public void build_returns_directive_when_multiple_directives_found() {
        // Arrange
        when(annotatedElement.getDirectives()).thenReturn(new GraphQLDirective[]{new GraphQLDirective1(), new GraphQLDirective2()});

        // Act
        DirectivesBuilder builder = new DirectivesBuilder(annotatedElement, processingElementsContainer);
        Directives[] directives = builder.build();

        // Assert
        assertEquals(Arrays.asList(new GraphQLDirective1(), new GraphQLDirective2()), directives);
    }

    @Test
    public void build_returns_directive_with_error_when_invalid_type() {
        // Arrange
        when(annotatedElement.getDirectives()).thenReturn(new GraphQLDirective[]{new GraphQLDirectiveInvalidType()});

        // Act
        DirectivesBuilder builder = new DirectivesBuilder(annotatedElement, processingElementsContainer);
        Directives[] directives = builder.build();

        // Assert
        assertNoNullPointerException();
    }

    @Test
    public void build_returns_directive_with_error_when_invalid_args() {
        // Arrange
        when(annotatedElement.getDirectives()).thenReturn(new GraphQLDirective[]{new GraphQLDirectiveInvalidArgs()});

        // Act
        DirectivesBuilder builder = new DirectivesBuilder(annotatedElement, processingElementsContainer);
        Directives[] directives = builder.build();

        // Assert
        assertNoNullPointerException();
    }

    @Test
    public void build_returns_directive_with_error_when_invalid_type_argument() {
        // Arrange
        when(annotatedElement.getDirectives()).thenReturn(new GraphQLDirective[]{new GraphQLDirectiveInvalidTypeError("invalid argument")});

        // Act
        DirectivesBuilder builder = new DirectivesBuilder(annotatedElement, processingElementsContainer);
        Directives[] directives = builder.build();

        // Assert
        assertNoNullPointerException();
    }

    @Test
    public void build_returns_directive_with_error_when_non_existent_directive_found() {
        // Arrange
        when(annotatedElement.getDirectives()).thenReturn(new GraphQLDirective[]{new GraphQLDirectiveInvalidType()});

        // Act
        DirectivesBuilder builder = new DirectivesBuilder(annotatedElement, processingElementsContainer);
        Directives[] directives = builder.build();

        // Assert
        assertNoNullPointerException();
    }

    private void assertNoNullPointerException() {
        MockitoAnnotations.initMocks(this);
        try {
            DirectivesBuilder builder = new DirectivesBuilder(annotatedElement, processingElementsContainer);
            Directives directives = builder.build();
            assertThrows(NullPointerException.class, () -> directives.getDirectives());
        } catch (NullPointerException e) {
            // Expected
        }
    }
}

import org.junit.Test;
import static org.junit.Assert.assertEquals;

public class TestClass {

    @Test
    public void testGetPublicVariable() {
        // Arrange
        String variableName = "public_variable";
        
        // Act
        Object result1 = TestClass.publicMethod(variableName);
        Object result2 = TestClass.publicMethod(variableName, 10);
        
        // Assert
        assertEquals("public_variable", result1);
        assertEquals(10, result2);
    }

    @Test
    public void testGetPublicVariableMultipleTimes() {
        // Arrange
        String variableName = "public_variable";
        
        // Act
        Object result1 = TestClass.publicMethod(variableName);
        Object result2 = TestClass.publicMethod(variableName, 10);
        Object result3 = TestClass.publicMethod(variableName, 20);

        // Assert
        assertEquals(10, result1);
        assertEquals("public_variable", result2);
        assertEquals(20, result3);
    }

    @Test
    public void testGetPublicVariableMultipleTimesWithSameArguments() {
        // Arrange
        String variableName = "public_variable";
        
        // Act
        Object result1 = TestClass.publicMethod(variableName, 10);
        Object result2 = TestClass.publicMethod(variableName, 20);

        // Assert
        assertEquals(10, result1);
        assertEquals(20, result2);
    }

    public static class TestClass {
        private String publicVariable;

        public TestClass() {
            this.publicVariable = "public_variable";
        }

        public Object publicMethod(String variableName) {
            if (variableName.equals("public_variable")) {
                return publicVariable;
            } else {
                return null;
            }
        }
    }
}
