import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;

public class JacksonJsonTransformerImpl {

    private ObjectMapper objectMapper = new ObjectMapper();

    public String parse(String json, Class<?> clazz) throws JsonProcessingException {
        return objectMapper.writeValueAsString(clazz.getClass().getConstructor().newInstance(json));
    }

    public void convert(Object obj, String strType, Class<?> clazz) throws JsonProcessingException {
        if (obj == null || strType.equals(String.class)) {
            throw new NullPointerException("Input object cannot be null and the type is String");
        }
        try {
            objectMapper.writeValue(strType.newInstance(), obj);
        } catch (JsonProcessingException e) {
            // ignore the exception
        }
    }

    public void convert(Object obj, Class<?> clazz, Object[] args) throws JsonProcessingException {
        if (obj == null || !clazz.isAssignableFrom(obj.getClass())) {
            throw new NullPointerException("Input object cannot be null and the type is not a subclass of " + clazz.getName());
        }
        try {
            objectMapper.writeValue(clazz.newInstance(), obj);
        } catch (JsonProcessingException e) {
            // ignore the exception
        }
    }

    public void convert(Object obj, Class<?>[] args, Object[] values) throws JsonProcessingException {
        if (obj == null || !args.length.equals(obj.getClass().getDeclaredFields().length)) {
            throw new NullPointerException("Input object cannot be null and the number of type arguments is not equal to " + args.length);
        }
        try {
            for (int i = 0; i < values.length && i < obj.getClass().getDeclaredFields().length; i++) {
                objectMapper.writeValue(values[i], obj.getClass().getDeclaredField(i).getType(obj));
            }
        } catch (JsonProcessingException e) {
            // ignore the exception
        }
    }

}
