import java.io.UnsupportedEncodingException;
import java.nio.charset.StandardCharsets;

public class Main {

    public static void main(String[] args) {
        String input = "Hello, World!";
        
        // Base64 encoded string
        try {
            byte[] encodedBytes = input.getBytes(StandardCharsets.UTF_8);
            String base64EncodedString = Base64.getEncoder().encodeToString(encodedBytes);
            System.out.println("Base64 encoded: " + base64EncodedString);
        } catch (UnsupportedEncodingException e) {
            System.out.println(e.getMessage());
        }

        // Decoded string
        try {
            byte[] decodedBytes = Base64.getDecoder().decode(base64EncodedString);
            String decodedString = new String(decodedBytes, StandardCharsets.UTF_8);
            System.out.println("Decoded: " + decodedString);
        } catch (UnsupportedEncodingException e) {
            System.out.println(e.getMessage());
        }

        // Decoding without decoding
        try {
            byte[] encodedBytes2 = input.getBytes(StandardCharsets.UTF_8);
            String base64Encoded2 = Base64.getEncoder().encodeToString(encodedBytes2);
            String decodedString2 = new String(Base64.getDecoder().decode(base64Encoded2), StandardCharsets.UTF_8);
            System.out.println("Base64 decoded (no decoding): " + decodedString2);
        } catch (UnsupportedEncodingException e) {
            System.out.println(e.getMessage());
        }

        // Decoding without encoding
        try {
            String base64Encoded3 = Base64.getEncoder().encodeToString(input.getBytes(StandardCharsets.UTF_8));
            byte[] decodedBytes3 = new byte[base64Encoded3.length()];
            for (int i = 0; i < base64Encoded3.length(); i++) {
                decodedBytes3[i] = (byte) (base64Encoded3.charAt(i) & 0xff);
            }
            String decodedString3 = new String(decodedBytes3, StandardCharsets.UTF_8);
            System.out.println("Base64 encoded string (no decoding): " + decodedString3);
        } catch (UnsupportedEncodingException e) {
            System.out.println(e.getMessage());
        }

        // Decoding with incorrect encoding
        try {
            byte[] encodedBytes4 = input.getBytes(StandardCharsets.UTF_8);
            String base64Encoded4 = Base64.getEncoder().encodeToString(encodedBytes4);
            String decodedString4;
            for (int i = 0; i < base64Encoded4.length(); i++) {
                if ((base64Encoded4.charAt(i) & 0x80) != 0) {
                    decodedString4 += ((char) (Base64.getDecoder().decode(base64Encoded4).charAt(i + 1)));
                } else {
                    decodedString4 += ((char) (Base64.getDecoder().decode(base64Encoded4).charAt(i + 1)) | 0x80);
                }
            }
            System.out.println("Decoded: " + decodedString4);
        } catch (UnsupportedEncodingException e) {
            System.out.println(e.getMessage());
        }

        // Decoding with incorrect decoding
        try {
            String base64Encoded5 = Base64.getEncoder().encodeToString(input.getBytes(StandardCharsets.UTF_8));
            byte[] decodedBytes5;
            for (int i = 0; i < base64Encoded5.length(); i++) {
                if ((base64Encoded5.charAt(i) & 0x80) != 0) {
                    decodedBytes5[i] = (byte) (Base64.getDecoder().decode(base64Encoded5).charAt(i + 1));
                } else {
                    decodedBytes5[i] = (byte) (Base64.getDecoder().decode(base64Encoded5).charAt(i + 1)) | 0x80;
                }
            }
            System.out.println("Decoded string (no decoding): " + new String(decodedBytes5, StandardCharsets.UTF_8));
        } catch (UnsupportedEncodingException e) {
            System.out.println(e.getMessage());
        }

        // Base64 encoded integer
        int num = 1234567890;
        byte[] encodedIntegers = Integer.toByteArray(num);
        String base64EncodedInteger = Base64.getEncoder().encodeToString(encodedIntegers);
        System.out.println("Base64 encoded integer: " + base64EncodedInteger);

        // Decoded integer
        try {
            int decodedNum = Base64.getDecoder().decode(base64EncodedInteger);
            System.out.println("Decoded integer: " + decodedNum);
        } catch (UnsupportedEncodingException e) {
            System.out.println(e.getMessage());
        }

        // Try to decode with incorrect parameters
        try {
            byte[] encodedIntegers2 = new byte[num];
            Base64.getDecoder().decode(encodedIntegers2); // This should fail
            num = 1234567890;
            System.out.println("Decoded integer (incorrect decoding): " + decodedNum);
        } catch (UnsupportedEncodingException e) {
            System.out.println(e.getMessage());
        }
    }
}
