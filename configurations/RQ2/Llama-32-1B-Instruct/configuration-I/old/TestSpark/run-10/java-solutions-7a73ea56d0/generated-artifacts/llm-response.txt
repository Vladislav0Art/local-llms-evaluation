import java.lang.reflect.Method;
import java.util.Arrays;

public class Main {
    private static Object instance = null;

    public static void main(String[] args) throws Exception {
        Method method = getMethod("calculateSpans", int.class, boolean.class);
        if (method != null && !instance.getClass().equals(method.getDeclaringClass())) {
            instance = method.invoke(instance, 1);
        }

        while (instance == null) {
            method = Main.this.getClass().getDeclaredMethods()[0];
            try {
                instance = method.invoke(null);
            } catch (Exception e) {
                System.out.println(e.getMessage());
            }
        }

        Method[] methods = Main.class.getMethods();
        for (Method method : methods) {
            if (method.getName().equals("testCalculateSpans") || method.getName().equals("testSetPrices")) {
                testMethod(method);
            } else if (method.getName().startsWith("calculate_")) {
                calculateMethod(method);
            }
        }
    }

    private static void testMethod(Method method) {
        int[] result = (int[]) method.invoke(instance, 1);
        System.out.println(result[0] + " " + result[1]);
    }

    private static void calculateMethod(Method method) {
        if (!method.getName().startsWith("calculate_")) {
            throw new RuntimeException("Method name does not start with 'calculate_'");
        }

        int[] result = (int[]) method.invoke(instance, 2);
        System.out.println(result[0] + " " + result[1]);
    }
}
