                sb.append(" ").append(System.lineSeparator());
            }
        }

        return sb.toString();
    }

        int port = 80;
        String method = "GET";

        generateRandomUrl(baseUrl, port, method);

    public static void generateRandomUrl(String baseUrl, int port, String method) throws Exception {
        Random random = new SecureRandom();

        // Generate a random path
        List<String> pathParts = new ArrayList<>();
        for (int i = 0; i < 1000; i++) {
            String part = "/".repeat(i);
            pathParts.add(part);
        }

        StringBuilder sb = new StringBuilder(baseUrl + "/" + pathParts.stream().map(String::valueOf).collect(Collectors.joining("/")));

        // Generate random query parameters
        List<String> queryParams = new ArrayList<>();
        for (int i = 0; i < 10; i++) {
            int paramLength = random.nextInt(3) + 1;
            String paramName = "param_" + i;
            Random paramNameGen = new SecureRandom();
            byte[] paramValue = new byte[paramLength * 2];
            for (int j = 0; j < paramLength; j++) {
                if (j % 2 == 0) {
                    paramValue[j] = random.nextByte() & 0xFF;
                } else {
                    paramValue[j] = random.nextByte() & 0xFF ^ 128;
                }
            }

            queryParams.add(paramName + "=" + System.lineSeparator().concat(String.format("%02x%02x", paramValue)));
        }

        // Combine everything into a single query string
        StringBuilder queryString = new StringBuilder();
        for (int i = 0; i < queryParams.size(); i++) {
            if (i > 0) {
                queryString.append("&");
            }
            queryString.append(queryParams.get(i));
        }

        sb.append(baseUrl).append("/").append(pathParts.toString()).append("?").append(queryString.toString());

        System.out.println(sb);
    }
}
