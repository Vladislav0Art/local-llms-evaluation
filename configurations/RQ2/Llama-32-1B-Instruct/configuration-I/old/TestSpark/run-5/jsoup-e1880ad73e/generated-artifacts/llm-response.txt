import org.junit.Test;
import org.jsoup.helper.UrlBuilder;
import org.jsoup.helper.KeyValue;
import org.junit.AfterClass;
import org.junit.BeforeClass;

import java.net.URLEncoder;
import java.nio.charset.StandardCharsets;
import java.util.Map;

public class UrlBuilderTest {

    @Mock
    private Connection.KeyVal kv;

    @BeforeClass
    public static void setup() {
        MockitoAnnotations.initMocks(this);
    }

    @AfterClass
    public static void tearDown() {
        Mockito.resetMocks(this);
    }

    @Test
    public void testAppendKeyVal_ConvertToMap_ConvertedMap() throws UnsupportedEncodingException {
        // Arrange
        String key = "key";
        String value = "value";
        UrlBuilder underTest = new UrlBuilder(new URL("https://example.com/path"));
        Map<String, String> expectedMap = Map.of(key, value);
        String separator = "-";
        int num = 0;

        // Act and Assert
        underTest.appendKeyVal(kv);
        assertUnderTestIsMap(underTest, expectedMap);
    }

    @Test
    public void testAppendKeyVal_URLEncoder() throws UnsupportedEncodingException {
        // Arrange
        String key = "key";
        String value = "value";
        UrlBuilder underTest = new UrlBuilder(new URL("https://example.com/path"));
        Map<String, String> expectedMap = Map.of(key, URLEncoder.encode(value, StandardCharsets.UTF_8));
        String separator = "-";
        int num = 0;

        // Act and Assert
        underTest.appendKeyVal(kv);
        assertUnderTestIsMap(underTest, expectedMap);
    }

    private void assertUnderTestIsMap(UrlBuilder underTest, Map<String, String> expectedMap) {
        Mockito.verify(underTest).appendKeyVal(kv);
        assertEquals(expectedMap.size(), underTest.build().size());
        for (Map.Entry<String, String> entry : expectedMap.entrySet()) {
            String key = entry.getKey();
            String value = URLEncoder.encode(entry.getValue(), StandardCharsets.UTF_8);
            Mockito.verify(underTest).appendKeyVal(entry -> underTest.appendKeyVal(entry));
            assertEquals(key, underTest.build().get(key));
            assertEquals(value, underTest.build().getOrDefault(key, "").replaceAll(".*", ""));
        }
    }

}

import org.jsoup.helper.KeyValue;
import org.jsoup.helper.UrlBuilder;
import org.junit.BeforeClass;
import org.junit.Test;

public class GeneratedTest {
    private static final String URL = "https://example.com/path";
    private static final String EXPECTED_MESSAGE = "Key-value pairs are not equal";

    @BeforeClass
    public static void init() {
        MockitoAnnotations.initMocks(this);
    }

    @Test
    public void testAppendKeyValue() {
        UrlBuilder builder = new UrlBuilder(URL);
        Map<String, String> expectedMap = Map.of("key1", "value1");
        String result = builder.appendKeyVal(KeyValue.of("key1", "value1")).toString();
        assertEquals(EXPECTED_MESSAGE, result);
    }

    @Test
    public void testAppendKeyValueMultiplePairs() {
        UrlBuilder builder = new UrlBuilder(URL);
        Map<String, String> expectedMap = Map.of("key1", "value1", "key2", "value2");
        String result = builder.appendKeyVal(KeyValue.of("key1", "value1")).appendKeyVal(KeyValue.of("key2", "value2")).toString();
        assertEquals(EXPECTED_MESSAGE, result);
    }

    @Test
    public void testAppendKeyValueEmptyMap() {
        UrlBuilder builder = new UrlBuilder(URL);
        Map<String, String> expectedMap = Map.of();
        String result = builder.appendKeyVal(KeyValue.of("key1", "value1")).toString();
        assertEquals(EXPECTED_MESSAGE, result);
    }

    @Test
    public void testAppendKeyValueNoPairs() {
        UrlBuilder builder = new UrlBuilder(URL);
        Map<String, String> expectedMap = Map.of();
        String result = builder.appendKeyVal(KeyValue.of()).toString();
        assertEquals(EXPECTED_MESSAGE, result);
    }
}

    }
}

    public static void testMain() throws Exception {

        final String expectedOutput = "abc";

        Main.main(input);
