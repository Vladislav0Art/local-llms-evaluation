```java
import org.junit.Test;
import static org.junit.Assert.assertTrue;

public class UrlBuilderTest {
    private UrlBuilder urlBuilder;

    @Before
    public void setup() {}

    @Test
    public void testBuildUrl() throws UnsupportedEncodingException {
        URL inputUrl = "https://www.example.com/path?query=value";
        urlBuilder = new UrlBuilder(inputUrl);
        String actualUrl = urlBuilder.build();
        assertTrue(actualUrl != null && !actualUrl.isEmpty());
    }

    @Test
    public void testAppendKeyVal() throws UnsupportedEncodingException {
        String key = "key";
        KeyVal keyVal = new KeyVal(key, new byte[]{'k'}); // Mocking KeyVal with an empty byte array for testing purposes only

        urlBuilder.appendKeyVal(keyVal);

        assertTrue(!key.isEmpty());
    }

    @Test
    public void testAppendKeyValUnsupportedEncodingException() throws UnsupportedEncodingException {
        String key = "key";
        KeyVal keyVal = new KeyVal(key, new byte[]{'k'}); // Mocking KeyVal with an empty byte array for testing purposes only

        urlBuilder.appendKeyVal(keyVal);

        assertTrue(!key.isEmpty());
    }

    @Test
    public void testBuildUrlWithEmptyKey() throws UnsupportedEncodingException {
        URL inputUrl = null;
        urlBuilder = new UrlBuilder(inputUrl);
        String actualUrl = urlBuilder.build();

        assertTrue(actualUrl != null && !actualUrl.isEmpty());
    }
}

class UrlBuilder {
    private URL inputUrl;

    public UrlBuilder(URL inputUrl) {
        this.inputUrl = inputUrl;
    }

    public String build() throws UnsupportedEncodingException {
        if (inputUrl != null) {
            return new URI(inputUrl.toString()).toURL();
        } else {
            return null;
        }
    }

    public void appendKeyVal(Connection.KeyVal kv) throws UnsupportedEncodingException {
        if (!kv == null || !kv.getKey().isEmpty()) {
            String keyValStr = kv.getKey() + "=" + kv.getValue();
            inputUrl = new URI(keyValStr);
        } else {
            // Remove empty string from the URL
            inputUrl = new URI(inputUrl.toString().replaceAll("^\\s+|\\s+$", ""));
        }
    }

    public static class KeyVal {
        private String key;
        private byte[] val;

        public KeyVal(String key, byte[] val) {
            this.key = key;
            this.val = val;
        }

        @Override
        public String toString() {
            return "KeyVal{" +
                    "key='" + key + '\'' +
                    ", val=" + new String(val) +
                    '}';
        }
    }
}
```
