import org.jsoup.nodes.TextNode;

public class TextNode {

    public static boolean isBlank() {
        return this.text().isEmpty();
    }
    
    @Override
    public String toString() {
        return this.toString();
    }

    private String text() {
        return this.html().trim();
    }

    private String html() {
        Element element = getFirstElement(this);
        if (element != null) {
            return element.outerHtml();
        } else {
            return "";
        }
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof TextNode)) return false;

        TextNode textNode = (TextNode) o;

        if (!text().equals(textNode.text())) return false;
        return this.html() != null ? this.html().equals(textNode.html()) : textNode.html() == null;
    }

    @Override
    public int hashCode() {
        return 31 * java.util.Collections.unaryCombining(this.text(), this.html());
    }

    private static TextNode getFirstElement(TextNode node) {
        for (Element element : node.elements()) {
            if (!element.text().isEmpty()) {
                return element;
            }
        }
        return null;
    }

}
