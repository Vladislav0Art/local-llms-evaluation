class CommentNodeTest {

    public static void main(String[] args) {
        TestCommentNodeCommentData testComment = new TestCommentNodeCommentData("Example comment");

        outerHtmlHead(testComment.getData(), 0, null);

        TestCommentNodeXmlDeclarationCommentData testXmlData = new TestCommentNodeXmlDeclarationCommentData();
        
        assertEquals(1L, testXmlData.getDepth(), 10);
        
        assertEquals(true, testXmlData.isPublished());
        
        assertEquals(false, testXmlData.isDraft());

        outerHtmlHead(testXmlData.getData(), 0, null);
    }

    public static void outerHtmlHead(TestCommentNodeCommentData comment, int depth, Appendable appendable) {
        if (depth == 0) {
            appendable.append("<comment>Example comment</comment>");
        } else {
            for (int i = 0; i < depth - 1; i++) {
                appendable.append("</comment>");
            }
            appendable.append("<comment>Example comment</comment>");
        }
    }

    public static class CommentNode extends CommentNodeTest {

        public static void main(String[] args) {
            TestCommentNodeCommentData testComment = new TestCommentNodeCommentData("Example comment");

            outerHtmlHead(testComment.getData(), 0, null);
        }
    }
}
