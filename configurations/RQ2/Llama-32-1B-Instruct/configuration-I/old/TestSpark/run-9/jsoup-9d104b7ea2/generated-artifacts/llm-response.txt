public class TextNodeTest {

    @Test
    public void testBlank() {
        TextNode node = new TextNode("");
        assertTrue(node.isBlank());
    }

    @Test
    public void testNoTextContent() {
        TextNode node = new TextNode("<p>Hello, World!</p>");
        assertFalse(node.text("Hello, World!"));
    }

    @Test
    public void testNormalisedWhitespace() {
        String text = "   <p>Hello, World!</p>";
        TextNode node = new TextNode(text);
        assertEquals("Hello, World!", node.text());
    }

    @Test
    public void testStripLeadingWhitespace() {
        StringBuilder sb = new StringBuilder();
        sb.append("   <p>Hello, World!</p>");
        TextNode node = new TextNode(sb.toString());
        assertTrue(node.stripLeadingWhitespace().text().isEmpty());
    }

    @Test
    public void testLastCharIsWhitespace() {
        String text = "Hello<BR>World!";
        TextNode node = new TextNode(text);
        assertTrue(node.lastCharIsWhitespace(new StringBuilder()));
    }

    @Test
    public void testSplitText() {
        TextNode node1 = new TextNode("<p>Hello, World!</p>");
        TextNode node2 = new TextNode("&lt;p&gt;Hello, World!&lt;/p&gt;");
        Node child1 = node1.splitText(0);
        Node child2 = node2.splitText(4);
        assertTrue(child1.text().equals("Hello, World!"));
        assertTrue(child2.text().equals("<p>Hello, World!</p>"));
    }

    @Test
    public void testFromEncoded() {
        String encodedText = "&lt;p&gt;Hello, World!&lt;/p&gt;";
        TextNode node = TextNode.createFromEncoded(encodedText);
        assertEquals("p", node.nodeName());
        assertTrue(node.text().isEmpty());
    }
}

import org.jsoup.nodes.Element;

public class TextNodeTest {

    public static String stripLeadingWhitespace(TextNode node) {
