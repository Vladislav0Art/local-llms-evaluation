```java
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;

import java.net.URLEncoder;
import java.nio.charset.StandardCharsets;
import java.util.HashMap;
import java.util.Map;

public class UrlBuilderTest {

    private Map<String, String> urlMap = new HashMap<>();

    @BeforeEach
    public void setUp() {
        for (int i = 0; i < 10; i++) {
            urlMap.put(i + "key", URLEncoder.encode("value_" + i, StandardCharsets.UTF_8));
        }
    }

    @Test
    public void build_url_uses_key_val_from_map() throws UnsupportedEncodingException {
        UrlBuilder builder = new UrlBuilder(new URL("https://example.com"));
        String expectedUrl = "https://example.com/value_0";
        assertEquals(expectedUrl, builder.build().toURI());
    }

    @Test
    public void append_key_val_sets_url_key_and_value_in_same_connection() throws UnsupportedEncodingException {
        UrlBuilder builder = new UrlBuilder(new URL("https://example.com"));
        String key = "value";
        int index = 1;
        String[] expectedUrlParts = {"https://example.com", key, URLEncoder.encode(String.valueOf(index), StandardCharsets.UTF_8)};
        assertEquals(expectedUrlParts[0], builder.appendKeyVal(Map.of(key, URLEncoder.encode(String.valueOf(index), StandardCharsets.UTF_8)).toMap()));
    }

    @Test
    public void append_key_val_throws UnsupportedEncodingException_for_non_string_value() {
        UrlBuilder builder = new UrlBuilder(new URL("https://example.com"));
        Map<String, String> kv = Map.of(0, "value_0");
        try {
            builder.appendKeyVal(kv);
            fail("Expected UnsupportedEncodingException to be thrown");
        } catch (UnsupportedEncodingException e) {
            // expected
        }
    }

    @Test
    public void append_key_val_throws UnsupportedEncodingException_for_string_value() {
        UrlBuilder builder = new UrlBuilder(new URL("https://example.com"));
        String value = "value_0";
        int index = 1;
        Map<String, String> kv = Map.of(0, URLEncoder.encode(value, StandardCharsets.UTF_8), index.toString());
        try {
            builder.appendKeyVal(kv);
            fail("Expected UnsupportedEncodingException to be thrown");
        } catch (UnsupportedEncodingException e) {
            // expected
        }
    }

    @Test
    public void append_key_val_sets_url_key_in_url_string() throws UnsupportedEncodingException {
        UrlBuilder builder = new UrlBuilder(new URL("https://example.com"));
        String key = "value";
        Map<String, String> kv = Map.of(0, URLEncoder.encode(String.valueOf(key), StandardCharsets.UTF_8));
        int index = 1;
        String[] expectedUrlParts = {"https://example.com", key, URLEncoder.encode(String.valueOf(index), StandardCharsets.UTF_8)};
        assertEquals(expectedUrlParts[0], builder.appendKeyVal(kv).toURI().toString());
    }
}
```
