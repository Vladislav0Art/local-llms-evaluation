import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.assertEquals;

public class MockTextNode {
    private TextNode appender = new TextNode();

    @Test
    public void testToString() {
        assertEquals("Hello World", this.toString());
    }
}

class TestMockTextNode extends MockTextNode {

    @Override
    public String toString() {
        return "Hello World";
    }

    public static class TextNode {
        private String data;

        public TextNode(String data) {
            this.data = data;
        }

        public void append(String text) {
            if (this.appender != null && !this.appender.isEmpty()) {
                this.appender.append(text);
            }
        }

        @Override
        public String toString() {
            return "MockTextNode{" +
                    "data='" + data + '\'' +
                    ", appender=" + appender +
                    '}';
        }
    }
}

    private TextNode appender;

    public void testAppend() {

        appender.append(" World");
        assertEquals("Hello World", appender.toString());
    }

    @Test
    public void testClearAndAppendAgain() {
        mockText = new TestMockTextNode();
        mockText.append("Hello");
        appender = mockText.appender;
        appender.append(" World");
        appender.clear();
        assertEquals("Hello", appender.toString());
    }
}

public class TextNodeAppenderTest {

    public static void main(String[] args) {
        TestMockTextNode mockText = new TestMockTextNode();
        expectedData = "Hello";
        
        // Append to empty text
        mockText.clear();
        mockText.append(" World");
        assertEquals(expectedData, mockText.text.data);
        
        // Append a string to an empty text
        mockText = new TestMockTextNode();
        mockText.clear();
        mockText.append(expectedData);
        assertEquals(expectedData, mockText.text.data);

        // Append a non-existent child node as text
        mockText.setChildNode(new String(null)); // This will throw a RuntimeException
        try {
            mockText.text.data = expectedData;
        } catch (RuntimeException e) {
            // Ignore this exception
        }
        
        // Append to an empty text again
        mockText = new TestMockTextNode();
        mockText.clear();
        mockText.append(" World");
        assertEquals(expectedData, mockText.text.data);
    }

}

class MockTextNode extends Node {
    private String data;

    public void append(String text) {
        this.data += text;
    }

    public String text() { return data; }
    
    @Override
    public String toString() {
        return "MockTextNode{" +
                "data='" + data + '\'' +
                '}';
    }

    public static MockTextNode newNode() {
        return new MockTextNode();
    }
}

class TestMockTextNode extends Node {
    private String text;

    public void clear() {
        this.text = null;
    }

    @Override
    public String toString() {
        return "TestMockTextNode{" +
                "text='" + text + '\'' +
                '}';
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;

        TestMockTextNode that = (TestMockTextNode) o;

        if (text == null ? !that.text!!.equals(null) : text.equals(that.text)) return true;
        return false;
    }

    @Override
    public int hashCode() {
        return (text != null ? text.hashCode() : 0);
    }
}
