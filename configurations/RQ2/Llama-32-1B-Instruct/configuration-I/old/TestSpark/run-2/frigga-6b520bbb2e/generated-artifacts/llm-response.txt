            return -1;
        } else if (this.commitHash == null && other.commitHash != null) {
            return 1;
        } else {
            return 0;
        }
    }

        // implement appversion pattern here
        return null;

public class AppVersion {
    private String amiName;
    private int buildNumber;
    private String commitHash;

    public void setAmiName(String amiName) { this.amiName = amiName; }
    public String getAmiName() { return amiName; }

    public void setBuildNumber(int buildNumber) { this.buildNumber = buildNumber; }
    public int getBuildNumber() { return buildNumber; }

    public void setCommitHash(String commitHash) { this.commitHash = commitHash; }
    public String getCommitHash() { return commitHash; }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        if (this.amiName != null && !this.amiName.isEmpty()) {
            sb.append(this.amiName).append(",");
        } else {
            sb.append("null, ");
        }
        if (this.buildNumber != 0) {
            sb.append(this.buildNumber).append(",");
        } else {
            sb.append("0");
        }
        if (this.commitHash != null && !this.commitHash.isEmpty()) {
            sb.append(",").append(this.commitHash);
        } else {
            sb.append(",null");
        }

        return sb.toString();
    }

    @Override
    public int hashCode() { // implementation of hashCode method
        return this.amiName == null ? 0 : this.amiName.hashCode();
    }

    @Override
    public boolean equals(Object obj) { // implementation of equals method
        if (this == obj)
            return true;
        if (obj == null || getClass() != obj.getClass())
            return false;

        AppVersion that = (AppVersion) obj;

        if (!Objects.equals(amiName, that.amiName))
            return false;

        if (!(that instanceof AppVersion))
            return false;

        if (buildNumber != 0 && not that.buildNumber)
            return false;
        if (commitHash == null || !Objects.equals(commitHash, that.commitHash))
            return false;

        return true;
    }

    public static class builder {
        private String amiName = null;
        private int buildNumber = 0;
        private String commitHash = null;

        public AppVersionBuilder withAmiName(String amiName) { 
            this.amiName = amiName; 
            return this;
        }
        
        public AppVersionBuilder withBuildNumber(int buildNumber) {
            this.buildNumber = buildNumber;
            return this;
        }

        public AppVersionBuilder withCommitHash(String commitHash) {
            this.commitHash = commitHash;
            return this;
        }

        public AppVersion build() { 
            return new AppVersion(this.amiName, this.buildNumber, this.commitHash);
        }
    }

    public static class builder {
        private String amiName = null;
        private int buildNumber = 0;
        private String commitHash = null;

        public AppVersionBuilder withAmiName(String amiName) { 
            this.amiName = amiName; 
            return this;
        }
        
        public AppVersionBuilder withBuildNumber(int buildNumber) {
            this.buildNumber = buildNumber;
            return this;
        }

        public AppVersionBuilder withCommitHash(String commitHash) {
            this.commitHash = commitHash;
            return this;
        }

        public AppVersion build() { 
            AppVersion version = new AppVersion(this.amiName, this.buildNumber, this.commitHash);
            if (this.amiName != null && !this.amiName.isEmpty()) {
                version.setAmiName(this.amiName);
            }
            if (this.buildNumber != 0) {
                version.setBuildNumber(this.buildNumber);
            }
            return version;
        }
    }

    public static AppVersion create(String amiName, int buildNumber, String commitHash) {
        builder with = new builder();
        with.withAmiName(amiName).withBuildNumber(buildNumber).withCommitHash(commitHash).build();

        return with.build();
    }
}
