    public void testBuildWithoutException() throws Exception {

        int value = 10;
        
        // Act
        int result = Main.class.getMethod("buildValue", int.class).invoke(null, value);
        
        // Assert
        assertThrows(IllegalArgumentException.class, () -> Main.class.getMethod("buildValue", int.class).invoke(null, null));
    }

    public int buildValue(int value) {
        return Integer.parseInt(String.valueOf(value));
    }

    public String appendValue(String value, int index) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < value.length(); i++) {
            if (i == index) {
                sb.append('*');
            } else {
                sb.append(value.charAt(i));
            }
        }
        return sb.toString();
    }

    public String buildValue(String value) {
        return appendValue(value, value.length());
    }

    public static void main(String[] args) {
        Main test = new Main();
        System.out.println(test.buildValue(10));
        System.out.println(test.appendValue("Hello", 0));
        assertThrows(IllegalArgumentException.class, () -> test.appendValue(null, null));
        test.buildWithoutException();
    }
