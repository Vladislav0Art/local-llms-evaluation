```java
import org.junit.Test;
import static org.mockito.Mockito.when;
import static org.hamcrest.Matchers.is;
import static org.junit.Assert.*;

public class UrlBuilderTest {

    @Test
    public void testBuildWithoutAppendKeyVal() {
        String inputUrl = "https://example.com";
        URL url = new UrlBuilder(inputUrl).build();
        assertEquals("https://example.com", url);
    }

    @Test
    public void testBuildWithAppendKeyVal() throws UnsupportedEncodingException {
        String inputUrl = "https://example.com";
        URL url = new UrlBuilder(inputUrl).build();
        assertNotNull(url);

        when(inputUrl.getKeyValue()).thenReturn("key1=value1");
        when(inputUrl.getOtherValue()).thenReturn("otherValue");

        URL expectedUrl = new UrlBuilder("https://example.com").build();
        assertEquals(expectedUrl, url);
    }

    @Test
    public void testAppendKeyVal() {
        String inputUrl = "https://example.com";
        URL url = new UrlBuilder(inputUrl).build();

        when(inputUrl.getKeyValue()).thenReturn("key1=value1");
        when(inputUrl.getOtherValue()).thenReturn("otherValue");

        assertEquals("key1=value1", url.getKeyValue());
        assertEquals("otherValue", url.getOtherValue());
    }

    @Test
    public void testAppendKeyValUnsupportedEncoding() {
        String inputUrl = "https://example.com";
        URL url = new UrlBuilder(inputUrl).build();

        when(inputUrl.getKeyValue()).thenReturn("key1=value1");
        when(inputUrl.getOtherValue()).thenReturn("otherValue");

        try {
            assertEquals(null, url.getKeyValue());
            assertEquals(null, url.getOtherValue());
        } catch (UnsupportedEncodingException e) {}

        when(inputUrl.getKeyValue()).thenReturn("key1=value1");
        when(inputUrl.getOtherValue()).thenThrow(UnsupportedEncodingException.class);

        try {
            assertEquals(null, url.getKeyValue());
            assertEquals(null, url.getOtherValue());
        } catch (UnsupportedEncodingException ex) {}

        when(inputUrl.getOtherValue()).thenReturn("otherValue");

        try {
            assertEquals(null, url.getKeyValue());
            assertEquals("otherValue", url.getOtherValue());
        } catch (UnsupportedEncodingException ex) {}
    }

    @Test
    public void testAppendKeyValNonString() {
        String inputUrl = "https://example.com";
        URL url = new UrlBuilder(inputUrl).build();

        when(inputUrl.getKeyValue()).thenReturn(123);
        when(inputUrl.getOtherValue()).thenThrow(NumberFormatException.class);

        try {
            assertEquals(null, url.getKeyValue());
            assertEquals(null, url.getOtherValue());
        } catch (UnsupportedEncodingException | NumberFormatException ex) {}

        when(inputUrl.getOtherValue()).thenReturn("otherValue");

        try {
            assertEquals(null, url.getKeyValue());
            assertEquals("otherValue", url.getOtherValue());
        } catch (UnsupportedEncodingException ex) {}
    }
}
```
