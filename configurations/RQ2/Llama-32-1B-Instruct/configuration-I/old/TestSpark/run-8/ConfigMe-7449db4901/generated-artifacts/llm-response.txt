import java.util.*;

class CommentsConfiguration {
    private Map<String, List<String>> commentsMap;

    public void setComment(String path, String comment) {
        if (path == null || path.isEmpty()) {
            throw new IllegalArgumentException("Path cannot be empty");
        }

        if (!commentsMap.containsKey(path)) {
            commentsMap.put(path, new ArrayList<>());
        }
        commentsMap.get(path).add(comment);
    }

    public List<String> getAllComments() {
        return new ArrayList<>(commentsMap.values());
    }

    @Override
    public String toString() {
        return "CommentsConfiguration{" +
                "commentsMap=" + commentsMap +
                '}';
    }
}

class CommentsConfigurationTest {
    private CommentsConfiguration config;

    public void setUp() throws Exception {
        config = new CommentsConfiguration();
        Map<String, List<String>> commentsMap = new HashMap<>();
        commentsMap.put("path1", Arrays.asList("comment1", "comment2"));
        commentsMap.put("path2", Collections.singletonList(""));
        config.setComment("path1", "first comment");
        config.setComment("path2", "");
        config.commentsMap = commentsMap;
    }

    public void testCommentLinesSetter_SingleNewLineMultiple() {
        assertEquals(3, config.getAllComments().size());
        assertEquals(Arrays.asList("comment1\ncomment2"), config.getCommentsForPath("path1"));
        assertEquals(Collections.singletonList(""), config.getCommentsForPath("path2"));
    }

    public void testGetAllComments() {
        assertEquals(1, config.getAllComments().size());
        assertEquals(Arrays.asList("first comment"), config.getAllComments());
    }
}
