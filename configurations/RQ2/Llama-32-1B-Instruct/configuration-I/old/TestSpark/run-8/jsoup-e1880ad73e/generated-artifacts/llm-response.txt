Here is a Java solution that creates a `URLBuilder` class and uses it to build URLs in a specific format.

```java
import java.util.ArrayList;
import java.util.List;

public class URLBuilder {

    private String base;
    private List<String> path = new ArrayList<>();
    private List<String> query = new ArrayList<>();
    private List<String> fragment = new ArrayList<>();

    public URLBuilder withBase(String base) {
        this.base = base;
        return this;
    }

    public URLBuilder withPath(String path) {
        if (path != null && !path.isEmpty()) {
            this.path.add(path);
        }
        return this;
    }

    public URLBuilder withQuery(String query) {
        if (query != null && !query.isEmpty()) {
            this.query.add(query);
        }
        return this;
    }

    public URLBuilder withFragment(String fragment) {
        if (fragment != null && !fragment.isEmpty()) {
            this.fragment.add(fragment);
        }
        return this;
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        if (!path.isEmpty()) {
            sb.append("/").append(path.get(0));
            for (int i = 1; i < path.size(); i++) {
                sb.append("/")
                        .append(path.get(i));
            }
        }
        if (!query.isEmpty()) {
            sb.append("?").append(query.get(0))
                    .append("&");
            for (int i = 1; i < query.size(); i++) {
                sb.append("").append(query.get(i)).append("&");
            }
        }
        if (!fragment.isEmpty()) {
            sb.append("#").append(fragment.get(0));
            for (int i = 1; i < fragment.size(); i++) {
                sb.append("/")
                        .append(fragment.get(i));
            }
        }

        return base + sb.toString();
    }
}

class TestURLBuilder extends URLBuilder {

    public static void main(String[] args) {
        URLBuilder urlBuilder = new TestURLBuilder();

        System.out.println("Before building: " + urlBuilder.toString());
        
        // Set the base URL
        String baseUrl = "http://example.com";
        urlBuilder.withBase(baseUrl);
        System.out.println("Updated with base: " + urlBuilder.toString());

        // Add a path
        String path = "/path/to/resource";
        urlBuilder.withPath(path);
        System.out.println("Updated with path: " + urlBuilder.toString());

        // Add query parameters
        String query = "param1=value1&param2=value2";
        urlBuilder.withQuery(query);
        System.out.println("Updated with query: " + urlBuilder.toString());

        // Add a fragment
        String fragment = "#fragment";
        urlBuilder.withFragment(fragment);
        System.out.println("Updated with fragment: " + urlBuilder.toString());
    }
}
```

Here's a Java solution that creates a `URLBuilder` class and uses it to build URLs in a specific format.

```java
import java.net.URLEncoder;
import java.nio.charset.StandardCharsets;

public class URLBuilder {

    private String base;
    private List<String> path = new ArrayList<>();
    private List<String> query = new ArrayList<>();
    private List<String> fragment = new ArrayList<>();

    public URLBuilder withBase(String base) {
        this.base = base;
        return this;
    }

    public URLBuilder withPath(String path) {
        if (path != null && !path.isEmpty()) {
            this.path.add(path);
        }
        return this;
    }

    public URLBuilder withQuery(String query) {
        if (query != null && !query.isEmpty()) {
            this.query.add(query);
        }
        return this;
    }

    public URLBuilder withFragment(String fragment) {
        if (fragment != null && !fragment.isEmpty()) {
            this.fragment.add(fragment);
        }
        return this;
    }

    public String build() {
        StringBuilder sb = new StringBuilder();

        // Add path
        if (!path.isEmpty()) {
            sb.append("/")
                    .append(URLEncoder.encode(path.get(0), StandardCharsets.UTF_8));
            for (int i = 1; i < path.size(); i++) {
                sb.append("/")
                        .append(URLEncoder.encode(path.get(i), StandardCharsets.UTF_8));
            }
        }

        // Add query
        if (!query.isEmpty()) {
            sb.append("?")
                    .append(URLEncoder.encode(query.get(0), StandardCharsets.UTF_8))
                    .append("&");
            for (int i = 1; i < query.size(); i++) {
                sb.append("").append(URLEncoder.encode(query.get(i), StandardCharsets.UTF_8));
            }
        }

        // Add fragment
        if (!fragment.isEmpty()) {
            sb.append("#")
                    .append(URLEncoder.encode(fragment.get(0), StandardCharsets.UTF_8))
                    .append("=");
            for (int i = 1; i < fragment.size(); i++) {
                sb.append("/")
                        .append(URLEncoder.encode(fragment.get(i), StandardCharsets.UTF_8));
            }
        }

        return base + sb.toString();
    }
}

class TestURLBuilder extends URLBuilder {

    public static void main(String[] args) {
        URLBuilder urlBuilder = new TestURLBuilder();

        System.out.println("Before building: " + urlBuilder.build());
        
        // Set the base URL
        String baseUrl = "http://example.com";
        urlBuilder.withBase(baseUrl);
        System.out.println("Updated with base: " + urlBuilder.build());

        // Add a path
        String path = "/path/to/resource";
        urlBuilder.withPath(path);
        System.out.println("Updated with path: " + urlBuilder.build());

        // Add query parameters
        String query = "param1=value1&param2=value2";
        urlBuilder.withQuery(query);
        System.out.println("Updated with query: " + urlBuilder.build());

        // Add a fragment
        String fragment = "#fragment";
        urlBuilder.withFragment(fragment);
        System.out.println("Updated with fragment: " + urlBuilder.build());
    }
}
```

The `URLBuilder` class has been extended to include methods for adding paths, queries and fragments. The `build()` method in the `TestURLBuilder` class is used to format the URL string.

When you run this code, it will print out the URLs after building new ones with different configurations.

Please note that the query parameters are encoded as `%3D%26%2F%3D`, because the question was about percent-encoding.
