public class TagImpl {
    private String name;
    private boolean selfClosing;
    private boolean formSubmittable;

    public TagImpl(String name, boolean selfClosing) {
        this.name = name;
        this.selfClosing = selfClosing;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public boolean isSelfClosing() {
        return selfClosing;
    }

    public void setSelfClosing(boolean selfClosing) {
        this.selfClosing = selfClosing;
    }

    public boolean isFormSubmittable() {
        return formSubmittable;
    }

    public void setFormSubmittable(boolean formSubmittable) {
        formSubmittable = formSubmittable;
    }

    @Override
    public String toString() {
        if (name == null && selfClosing == false && formSubmittable == false) {
            return name + "!";
        } else if (selfClosing || formSubmittable) {
            return "<" + name + ">" + (selfClosing ? "/>" : "") + "</" + name + "> ";
        } else {
            return name;
        }
    }

    public static Tag value(String name, boolean selfClosing, boolean formSubmittable) {
        if (name == null && selfClosing == false && formSubmittable == false) {
            return new TagImpl(name, selfClosing);
        } else {
            return new TagImpl(name, selfClosing);
        }
    }

    public static String getValueByName(String name) {
        for (Tag tag : values()) {
            if (tag.getName().equals(name)) {
                return tag.toString();
            }
        }
        throw new RuntimeException("Tag " + name + " not found");
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;

        TagImpl tagImpl = (TagImpl) o;

        if (!Objects.equals(name, tagImpl.name)) return false;
        if (!Objects.equals(selfClosing, tagImpl.selfClosing)) return false;
        if (!Objects.equals(formSubmittable, tagImpl.formSubmittable)) return false;

        return true;
    }

    @Override
    public int hashCode() {
        return Objects.hash(name, selfClosing, formSubmittable);
    }
}

import java.util.HashMap;
import java.util.Map;

public class TagImpl {
    private String name;
    private boolean selfClosing;
    private boolean formSubmittable;
    private Map<String, Object> attributes = new HashMap<>();

    public TagImpl(String name, boolean selfClosing) {
        this.name = name;
        this.selfClosing = selfClosing;
    }

    public void addAttribute(String key, Object value) {
        attributes.put(key, value);
    }

    public String getAttribute(String key) {
        return (String) attributes.get(key);
    }

    public boolean isSelfClosing() {
        return selfClosing;
    }

    public boolean isFormSubmittable() {
        return formSubmittable;
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        if (name != null && !selfClosing) {
            sb.append("<").append(name).append(">");
        } else if (!name || selfClosing) {
            sb.append(name);
        }
        sb.append("> ");

        for (Map.Entry<String, Object> entry : attributes.entrySet()) {
            sb.append(entry.getKey()).append(": ").append(entry.getValue().toString()).append("\n");
        }

        return sb.toString();
    }

    public static Tag value(String name, boolean selfClosing, boolean formSubmittable) {
        if (name == null && selfClosing == false && formSubmittable == false) {
            return new TagImpl(name, selfClosing);
        } else {
            Map<String, Object> attributesMap = new HashMap<>();
            for (String key : name.split(", ")) {
                attributesMap.put(key.trim(), "true");
            }
            return new TagImpl(name, true);
        }
    }

    public static String getValueByName(String name) {
        for (Tag tag : values()) {
            if (tag.getName().equals(name)) {
                return tag.toString();
            }
        }
        throw new RuntimeException("Tag " + name + " not found");
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;

        TagImpl tagImpl = (TagImpl) o;

        if (!attributes.equals(tagImpl.attributes)) return false;
        for (Map.Entry<String, Object> entry : attributes.entrySet()) {
            if (!entry.getValue().equals(tagImpl.attributes.get(entry.getKey()))) return false;
        }

        return true;
    }

    @Override
    public int hashCode() {
        int result = name != null ? name.hashCode() : 0;
        for (Map.Entry<String, Object> entry : attributes.entrySet()) {
            if (entry.getValue().hashCode() > 0) {
                result = 31 * result + entry.getValue().hashCode();
            }
        }
        return result;
    }

    public static Tag[] values() {
        TagImpl tagImpl = new TagImpl("p", true);
        Tag tag2 = new TagImpl("input", false, false);
        Tag tag3 = new TagImpl("button", false, false);

        TagImpl value1 = value("p", true);
        TagImpl value2 = value("form", true, false);
        TagImpl value3 = value("img", false, false);

        Tag[] tags = {value1, value2, value3};
        return tags;
    }

    public static void main(String[] args) {
        Tag pTag = new TagImpl("p");
        Tag formTag = new TagImpl("form");

        System.out.println(pTag.toString());
        System.out.println(formTag.toString());

        System.out.println(values()[0].toString());
    }
}

In this implementation, `TagImpl` is the base class for all tags. It has attributes like `name`, `selfClosing`, and `formSubmittable`. The `addAttribute` method allows you to add or update attributes for a tag.

The `equals` method compares two tags based on their attribute values. This ensures that tags with the same name and attributes are considered equal.

The `values` method returns an array of all possible tags, including some example usage in the `main` method.
