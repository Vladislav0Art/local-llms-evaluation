import org.junit.Test;

public class Text {

    public String text(String originalString) {
        return originalString.replace("---", "-").replace("\"", "\"\"").trim();
    }

    public String[] splitString(String originalString, String delimiter) {
        return originalString.split(delimiter);
    }
}

@Test
public void testTextWithMultipleSpaces() {
