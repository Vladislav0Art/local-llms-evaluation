public class Comment extends LeafNode {
    private String data;

    public Comment(String data) {
        this.data = data;
    }

    @Override
    public String nodeName() {
        return "Comment";
    }

    @Override
    public String getData() {
        return data;
    }

    public Comment setData(String data) {
        this.data = data;
        return this;
    }

    @Override
    void outerHtmlHead(Appendable accum, int depth, Document.OutputSettings out) throws IOException {
        accum.appendLine();
        accum.appendTag("Comment", null);
        accum.appendText(data);
        if (depth > 0) {
            accum.appendBreakPoint();
        }
    }

    @Override
    void outerHtmlTail(Appendable accum, int depth, Document.OutputSettings out) throws IOException {
        if (depth < 1 || data.isEmpty()) {
            return;
        }
        accum.appendLine();
        accum.appendTag("Comment", null);
        accum.appendText(data.substring(0, Math.min(data.length(), 100))); // truncate if needed
        accum.appendBreakPoint();
    }

    @Override
    public String toString() {
        return "<" + nodeName() + ">" + getData() + "</" + nodeName() + ">";
    }

    @Override
    public Comment clone() {
        return new Comment(data);
    }

    @Override
    public boolean isXmlDeclaration() {
        // This method should be implemented by subclasses to check if they look like an XML Declaration.
        // For now, assume it's always false.
        return false;
    }
}

public class XmlDeclaration extends Comment {
    private String type;

    public XmlDeclaration(String data) {
        super(data);
        this.type = "xml";
    }

    @Override
    public boolean isXmlDeclaration() {
        return true; // For now, assume it's always true.
    }

    @Override
    @Nullable
    public XmlDeclaration asXmlDeclaration() {
        if (this.type.equals("xml")) {
            return this;
        } else {
            throw new UnsupportedOperationException();
        }
    }
}

public class Comment {
    private String data;

    public Comment(String data) {
        this.data = data;
    }

    }
}

public class XmlDeclaration extends Comment {

    private String type;

    public XmlDeclaration(String data, String type) {
        super(data);
        this.type = type;
    }

    @Nullable
    public XmlDeclaration asXmlDeclaration() {
        if (this.type.equals("xml")) {
            return this;
        } else {
            throw new UnsupportedOperationException();
        }
    }

    @Override
    @Nullable
    public String toString() {
        StringBuilder builder = new StringBuilder("<Comment");
        builder.append(": ").append(this.data);
        if (!this.type.equals("xml")) {
            builder.append("</Comment>");
        }
        return builder.toString();
    }
}
