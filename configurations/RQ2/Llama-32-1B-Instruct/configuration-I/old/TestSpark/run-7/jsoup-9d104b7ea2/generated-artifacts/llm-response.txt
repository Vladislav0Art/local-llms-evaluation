import java.util.Arrays;
import java.util.Comparator;

public class TextNode {
    private String text;
    private int index;
    private boolean isTrimmed;

    public TextNode(String text, int index, boolean isTrimmed) {
        this.text = text;
        this.index = index;
        this.isTrimmed = isTrimmed;
    }

    @Override
    public String toString() {
        return "TextNode{" +
                "text='" + text + '\'' +
                ", index=" + index +
                ", isTrimmed=" + isTrimmed +
                '}';
    }
}

public class Main {
    public static void main(String[] args) {
        TextNode[] textNodes = new TextNode[5];
        textNodes[0] = new TextNode("Hello World", 0, true);
        textNodes[1] = new TextNode("", 1, false);
        textNodes[2] = new TextNode("   Hello   ", 2, true);
        textNodes[3] = new TextNode("  Test   ", 3, false);
        textNodes[4] = new TextNode("   Testing   ", 4, true);

        Arrays.sort(textNodes, new Comparator<TextNode>() {
            @Override
            public int compare(TextNode o1, TextNode o2) {
                if (o1.isTrimmed && !o2.isTrimmed) return -1;
                else if (!o1.isTrimmed && o2.isTrimmed) return 1;

                // Fixing the issues in the provided tests
                int lengthComparison = Math.abs(o1.text.length() - o2.text.length());
                if (lengthComparison != 0) {
                    return Integer.compare(lengthComparison, 0);
                }

                return o1.index - o2.index;
            }
        });

        for (TextNode textNode : textNodes) {
            System.out.println(textNode.toString());
        }
    }
}
