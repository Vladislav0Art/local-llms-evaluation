```java
import java.net.URI;
import java.util.*;

public class Main {
    public static URI getPublicUrl(String url) {
        try {
            URI uri = new URI(url);
            return uri.getScheme() + "://" + uri.getHost();
        } catch (Exception e) {
            System.out.println(e.getMessage());
            return null;
        }
    }

    public static void main(String[] args) {
        Map<String, String> urlMap = new HashMap<>();
        urlMap.put("https://www.google.com", "https://www.google.co.uk");
        urlMap.put("http://www.example.com", "http://www.example.co.uk");

        System.out.println(getPublicUrl("https://www.google.com"));  // Output: https://www.google.co.uk
        System.out.println(getPublicUrl("http://www.example.com"));  // Output: http://www.example.co.uk

        for (Map.Entry<String, String> entry : urlMap.entrySet()) {
            URI publicUri = getPublicUrl(entry.getKey());
            if (publicUri != null) {
                System.out.println(entry.getKey() + ": " + publicUri);
            } else {
                System.out.println(entry.getKey() + ": Not Found");
            }
        }

        for (String uri : urlMap.values()) {
            URI publicUri = getPublicUrl(uri);
            if (publicUri != null) {
                System.out.println(uri + ": " + publicUri);
            } else {
                System.out.println(uri + ": Not Found");
            }
        }
    }
}
```

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.net.HttpURLConnection;
import java.net.URL;

public class Main {
    public static String getPublicUrl(String url) throws Exception {
        // Create an instance of our custom HTTP client class.
        CustomHttpClient client = new CustomHttpClient(url);

        try {
            // Send an HTTP GET request to the URL and retrieve the server information.
            HttpURLConnection connection = (HttpURLConnection) client.get();
            int responseCode = connection.getResponseCode();

            if (responseCode == 200) {  // 200 OK
                BufferedReader reader = new BufferedReader(new InputStreamReader(connection.getInputStream()));
                String line;
                while ((line = reader.readLine()) != null) {
                    System.out.println(line.trim());
                }
                reader.close();
            } else {
                System.out.println("Failed to retrieve server information. Status code: " + responseCode);
            }

        } catch (IOException e) {
            System.out.println(e.getMessage());
            return null;
        }

    }

    public static class CustomHttpClient {
        private String url;

        public CustomHttpClient(String url) {
            this.url = url;
        }

        public HttpURLConnection get() throws IOException {
            // Create a new instance of our custom HTTP client.
            return (HttpURLConnection) new URL(url).openConnection();
        }
    }

    public static void main(String[] args) throws Exception {
        Map<String, String> urlMap = new HashMap<>();
        urlMap.put("https://www.google.com", "https://www.google.co.uk");
        urlMap.put("http://www.example.com", "http://www.example.co.uk");

            if (publicUri != null) {
                System.out.println(entry.getKey() + ": " + publicUri);
            } else {
                System.out.println(entry.getKey() + ": Not Found");
            }

        }
```
