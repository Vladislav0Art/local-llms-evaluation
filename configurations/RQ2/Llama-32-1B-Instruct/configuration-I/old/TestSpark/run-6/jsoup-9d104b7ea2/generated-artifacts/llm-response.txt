import java.util.ArrayList;
import java.util.List;

public class Document {
    public String htmlContent;
    private List<String> textBlocks;
    private int currentLineIndex;

    /**
     * Initializes a new instance of the Document class.
     */
    public Document() {
        this.htmlContent = "";
        this.textBlocks = new ArrayList<>();
        this.currentLineIndex = 0;
    }

    /**
     * Adds text to the document's HTML content and accumulates it.
     *
     * @param content The text to add.
     * @return True if no new text was added, false otherwise.
     */
    public boolean addText(String content) {
        // Check if we've reached the end of this line
        if (this.currentLineIndex == 0 || !isLastBlock(content)) {
            htmlContent += "<p>" + content + "</p>";
            textBlocks.add(content);
            return true;
        } else {
            addText(content, null);
        }
    }

    /**
     * Checks if a given line contains only block-level text (i.e., not HTML tags).
     *
     * @param content The text to check.
     * @return True if the content is block-level, false otherwise.
     */
    private boolean isLastBlock(String content) {
        for (String block : textBlocks) {
            if (block.equals(content)) {
                return true;
            }
        }
        return false;
    }

    /**
     * Prints the document's HTML content to the console.
     *
     * @return A string representation of the HTML content.
     */
    public String toString() {
        StringBuilder output = new StringBuilder();
        for (int i = 0; i < textBlocks.size(); i++) {
            output.append(textBlocks.get(i)).append("\n");
            if (i < textBlocks.size() - 1) {
                output.append("<p>");
            }
        }

        // Append the remaining HTML content
        if (textBlocks.size() > 0) {
            output.append(htmlContent);
        } else {
            output.append("");
        }

        return output.toString();
    }

    public static void main(String[] args) {
        Document document = new Document();

        document.addText("This is the first line.");
        document.addText("<p>This is a paragraph with <b>bold</b> text.</p>");
        document.addText("And this is the second line.");

        System.out.println(document.toString());
    }
}

import java.util.*;

public class Document {
    private Map<String, List<Integer>> lineNumbers;
    public String htmlContent;

    /**
     * Initializes a new instance of the Document class.
     */
    public Document() {
        this.lineNumbers = new HashMap<>();
        this.htmlContent = "";
    }

    /**
     * Adds text to the document's HTML content and accumulates it.
     *
     * @param content The text to add.
     * @return True if no new text was added, false otherwise.
     */
    public boolean addText(String content) {
        // Get the current line number
        int lineNumber = getLineNumber();

        // Check if we've reached the end of this line
        if (lineNumber == 0 || !isLastBlock(content)) {
            htmlContent += "<p>" + content + "</p>";
            List<Integer> blocks = getBlocks(lineNumber);
            addTextToLine(content, blocks);
            return true;
        } else {
            addText(content, null);
        }
    }

    /**
     * Checks if a given line contains only block-level text (i.e., not HTML tags).
     *
     * @param content The text to check.
     * @return True if the content is block-level, false otherwise.
     */
    private boolean isLastBlock(String content) {
        for (List<Integer> blocks : lineNumbers.values()) {
            if (!blocks.contains(content)) {
                return true;
            }
        }
        return false;
    }

    /**
     * Returns a list of line numbers where the given text was added.
     *
     * @param content The text to look up.
     * @return A list of line numbers or an empty list if not found.
     */
    private List<Integer> getBlocks(String content) {
        for (int lineNumber = 1; lineNumber <= Math.min(lineNumbers.size(), lineNumbers.get(0).size() + 1); lineNumber++) {
            String blockText = htmlContent.substring(lineNumber - 1, lineNumber);
            if (blockText.equals(content)) {
                return Arrays.asList(blockText.split("\n"));
            }
        }

        // If the text is not found, return an empty list
        return Collections.emptyList();
    }

    /**
     * Adds text to the document's HTML content and accumulates it.
     *
     * @param content The text to add.
     * @param blocks  A list of line numbers where the text was added.
     */
    private void addTextToLine(String content, List<Integer> blocks) {
        for (int block : blocks) {
            htmlContent += "<p>" + content + "</p>";
        }
    }

    /**
     * Prints the document's HTML content to the console.
     *
     * @return A string representation of the HTML content.
     */
    public String toString() {
        StringBuilder output = new StringBuilder();
        for (int i = 0; i < lineNumbers.size(); i++) {
            if (i > 0) {
                output.append("\n");
            }
            output.append(lineNumbers.get(i).get(0));
            output.append(": ").append(htmlContent.substring(lineNumbers.get(i).get(0), lineNumbers.get(i).get(1)));
        }

        // Append the remaining HTML content
        if (!htmlContent.isEmpty()) {
            output.append("\n").append(htmlContent);
        } else {
            output.append("");
        }

        return output.toString();
    }

    public static void main(String[] args) {
        Document document = new Document();

        String text = "This is the first line.\n\nThis is a paragraph with <b>bold</b> text.";
        document.addText(text);

        System.out.println(document.toString());
    }
}

*   The `addText` method checks if the new text belongs to the last block and adds it accordingly, using a list of line numbers to keep track of the blocks.
*   In the `main` method, I demonstrate how to use the `addText` method with test cases.
