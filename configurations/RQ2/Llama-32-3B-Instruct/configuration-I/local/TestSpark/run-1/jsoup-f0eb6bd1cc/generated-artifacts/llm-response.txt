import org.junit.Test;
import java.net.URL;
import java.util.Map;

public class UrlBuilderTest {

    @org.junit.Test
    public void testUrlBuilder() {
        UrlBuilder urlBuilder = new UrlBuilder();
        Map<String, String> keyValue1 = Map.of("key1", "value1");
        Map<String, String> keyValue2 = Map.of("key2", "value2");

        urlBuilder.appendKeyValue("key1", "value1");
        urlBuilder.appendKeyValue("key2", "value2");

        URL result = urlBuilder.build();
        System.out.println(result);
    }
}

public class UrlBuilder {

    private StringBuilder url = new StringBuilder();

    public void appendKeyValue(String key, String value) {
        if (url.length() > 0) {
            url.append("&");
        }
        url.append(key).append("=").append(value);
    }

    public URL build() {
        return new URL(url.toString());
    }
}

public class UrlBuilder {

    public void appendKeyVal(KeyVal keyValue) {
        if (keyValue.key != null && keyValue.value != null) {
            if (this.url.toString().length() > 0) {
                this.url.append("&");
            }
            this.url.append(keyValue.key).append("=").append(keyValue.value);
        }
    }

    public URL build() {
        return new URL(this.url.toString());
    }
}
