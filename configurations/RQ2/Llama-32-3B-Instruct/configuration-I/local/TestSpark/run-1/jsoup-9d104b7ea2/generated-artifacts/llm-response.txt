        assertTrue(node.isBlank());
    }

    public void isBlankNotEmptyStringTest() {

        assertFalse(node.isBlank());
    }

        assertEquals(null, node.splitText(-1));

        assertEquals("", ba.toString());
    }

    public void outerHtmlTailEmptyStringTest() throws IOException {
        ByteArrayOutputStream ba = new ByteArrayOutputStream();
        Document.OutputSettings out = new Document.OutputSettings(ba);
        TextNode node = new TextNode("");
        node.outerHtmlTail(ba, 0, out);
        assertEquals("", ba.toString());

    public void outerHtmlHeadNotEmptyStringTest() throws IOException {

        Document.OutputSettings out = new Document.OutputSettings(ba);
        TextNode node = new TextNode("Hello");
        node.outerHtmlHead(ba, 0, out);

    @Test
    public void outerHtmlTailNotEmptyStringTest() throws IOException {
        ByteArrayOutputStream ba = new ByteArrayOutputStream();

        node.outerHtmlTail(ba, 0, out);
        assertEquals("</p>", ba.toString());
    }

    public void cloneTextEqualsOriginalTest() {
        TextNode node1 = new TextNode("Hello");
