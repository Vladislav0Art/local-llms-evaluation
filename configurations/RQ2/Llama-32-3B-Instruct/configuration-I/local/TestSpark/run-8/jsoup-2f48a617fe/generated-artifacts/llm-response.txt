import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class CommentTest {

    @Test
    public void nodeName() {
        Document doc = new Document();
        Element comment = doc.createElement("comment");
        assertEquals("comment", comment.tagName());
    }

    @Test
    public void writeCharacterArrayAndInts() throws Exception {
        StringBuilder sb = new StringBuilder();
        sb.append('a');
        sb.append(1);
        char[] chars = {'b', 'c'};
        sb.append(chars, 0, 2);

        assertEquals("abc", sb.toString());
    }

    @Test
    public void appendCharacterToStringBuilder() {
        StringBuilder sb = new StringBuilder();
        sb.append('a');

        Appendable accum = sb;
        accum.append('b');
        accum.append('c');

        assertEquals("abc", accum.toString());
    }

    @Test
    public void cloneElement() {
        Document doc = new Document();
        Element comment = doc.createElement("comment");
        comment.appendChild(doc.createElement("child"));
        
        Element clonedComment = (Element) comment.cloneNode(false);
        
        assertNotNull(clonedComment);
        assertEquals(1, clonedComment.elementNodes().size());
    }

    @Test
    public void compareStringObjects() {
        String str1 = "Hello";
        String str2 = "World";

        assertNotEquals(str1, str2);
    }
}

class Document {
    private Element element;

    public Document() {}

    public Element createElement(String tag) {
        element = new Element(tag);
        return element;
    }

    public void appendElement(Element child) {
        element.appendChild(child);
    }
}

class Element {
    private String tag;
    private List<Element> children = new ArrayList<>();

    public Element(String tag) {
        this.tag = tag;
    }

    public void appendChild(Element child) {
        children.add(child);
    }

    public boolean hasChild() {
        return !children.isEmpty();
    }
}
