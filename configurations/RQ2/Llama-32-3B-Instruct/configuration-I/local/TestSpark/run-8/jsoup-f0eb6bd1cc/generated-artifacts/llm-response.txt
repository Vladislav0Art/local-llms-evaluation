**Explanation of JavaScript Url**

```javascript
function buildUrl(base, path, params) {
  let url = base;
  
  if (path) {
    url += '/' + path.replace(/^\/+|\/+$/g, '');
  }
  
  if (params && Object.keys(params).length > 0) {
    url += '?' + Object.keys(params).map(key => key + '=' + encodeURIComponent(params[key])).join('&');
  }
  
  return url;
}

console.log(buildUrl('https://example.com', '/path/to/resource', { query: 'param' }));
// Output: "https://example.com/path/to/resource?query=param"

console.log(buildUrl('https://example.com', '/path/to/resource', { query: 'param', anotherParam: 'value' }));
// Output: "https://example.com/path/to/resource?query=param&anotherParam=value"
```

```python
import urllib.parse

def build_url(base, path, params):
    url = base
    
    if path:
        url += '/' + path.lstrip('/')
    
    if params:
        query_params = urllib.parse.urlencode(params)
        if query_params:
            url += '?' + query_params
    
    return url

print(build_url('https://example.com', '/path/to/resource', {'query': 'param'}))
# Output: "https://example.com/path/to/resource?query=param"

print(build_url('https://example.com', '/path/to/resource', {'query': 'param', 'anotherParam': 'value'}))
# Output: "https://example.com/path/to/resource?query=param&anotherParam=value"
```
