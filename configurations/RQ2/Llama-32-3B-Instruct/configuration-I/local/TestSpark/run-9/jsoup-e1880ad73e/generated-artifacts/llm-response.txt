public class URLBuilder {
    private String url = "";

    public void appendKeyVal(Connection.KeyVal kv) {
        if (kv == null || kv.getValue() == null) return;
        String value = kv.getValue();
        if (url.isEmpty()) url += "?" + kv.getKey() + "=" + value;
        else url += "&" + kv.getKey() + "=" + value;
    }

    public String build() {
        return url;
    }
}

public class Connection {
    public static class KeyVal {
        private String key;
        private String value;

        public KeyVal(String key, String value) {
            this.key = key;
            this.value = value;
        }

        public String getKey() {
            return key;
        }

        public String getValue() {
            return value;
        }
    }
}

public class URLBuilderTest {

    @org.junit.Test
    public void testAppendKeyVal_withValue() {
        Connection.KeyVal kv1 = new Connection.KeyVal("key", "value");
        URLBuilder urlBuilder = new URLBuilder();
        urlBuilder.appendKeyVal(kv1);
        assertEquals("?key=value", urlBuilder.build());
    }

    @org.junit.Test
    public void testAppendKeyVal_withNullValue() {
        Connection.KeyVal kv1 = new Connection.KeyVal("key", null);
        URLBuilder urlBuilder = new URLBuilder();
        urlBuilder.appendKeyVal(kv1);
        assertEquals("?key=", urlBuilder.build());
    }

    @org.junit.Test
    public void testBuild_emptyUrl() {
        URLBuilder urlBuilder = new URLBuilder();

    }
}

public class URLBuilder {
    private StringBuilder url = new StringBuilder();

    public void append(String key, String value) {
        if (key == null || value == null) return;
        if (!url.toString().isEmpty()) {
            url.append("&");
        }
        url.append(key).append("=").append(value);
    }

    public String build() {
        if (url.toString().isEmpty()) {
            return "http://example.com";
        }
        return "http://example.com?" + url.toString();
    }
}

import java.net.URI;

public class URLBuilder {
    private String url = "http://example.com";

    public void appendQuery(String key, String value) {
        if (!url.contains("?")) {
            url += "?";
        } else {
            url += "&";
        }
        url += key + "=" + value;
    }

    public String build() throws URISyntaxException {
        URI uri = new URI(url);
        return uri.toString();
    }
}

            this.key = key;
