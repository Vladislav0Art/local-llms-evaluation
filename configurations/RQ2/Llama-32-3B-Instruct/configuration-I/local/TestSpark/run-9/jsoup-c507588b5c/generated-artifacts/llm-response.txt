import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Main {
    public static String normalizeWhitespace(String input) {
        Document doc = Jsoup.parse(input);
        return doc.body().text();
    }

    public static String stripLeadingWhitespace(String input) {
        Element body = Jsoup.parse(input).body();
        return body.text().trim();
    }

    public static char lastCharIsWhitespace(CharSequence input) {
        int length = input.length();
        if (length == 0 || Character.isWhitespace(input.charAt(length - 1))) {
            return '\u200B'; // Zero-width space
        }
        return '\u0000'; // Null character
    }

    public static void main(String[] args) {
        System.out.println(normalizeWhitespace("<html>Hello World\nThis is a test</html>"));
        System.out.println(stripLeadingWhitespace("   Hello World  "));
        char result = lastCharIsWhitespace("a");
        System.out.println(result);
    }
}
