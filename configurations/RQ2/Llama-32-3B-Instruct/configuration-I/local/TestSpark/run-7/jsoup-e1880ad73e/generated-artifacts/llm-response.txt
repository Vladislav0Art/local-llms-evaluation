import org.jsoup.helper.UrlBuilder;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNotNull;
import static org.junit.Assert.assertNull;
import static org.junit.Assert.assertTrue;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class UrlBuilderTest {

    @Mock
    private URL inputUrl;

    @Mock
    private Connection.KeyVal kv;

    @Test
    public void newUrlBuilder_fromValidUrl_returnsNewUrl() {
        when(inputUrl.toString()).thenReturn("https://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        assertNotNull(urlBuilder.build());
    }

    @Test
    public void newUrlBuilder_fromInvalidUrl_throwsException() {
        when(inputUrl.toString()).thenReturn(null);
        assertThrows(NullPointerException.class, () -> new UrlBuilder(inputUrl).build());
    }

    @Test
    public void appendKeyValue_toValidUrl_appendsKeyVal() {
        when(inputUrl.toString()).thenReturn("https://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        urlBuilder.appendKeyVal(kv);
        assertNotNull(urlBuilder.build().toString());
        assertTrue(urlBuilder.build().toString().contains(kv.getKey()));
    }

    @Test
    public void appendKeyValue_toInvalidUrl_doesNotAppend() {
        when(inputUrl.toString()).thenReturn(null);
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        assertNull(urlBuilder.appendKeyVal(kv));
    }

    @Test
    public void appendKeyValue_toEmptyString_appendsKeyVal() {
        String inputStr = "";
        UrlBuilder urlBuilder = new UrlBuilder(inputStr);
        urlBuilder.appendKeyVal(kv);
        assertNotNull(urlBuilder.build().toString());
        assertTrue(urlBuilder.build().toString().contains(kv.getKey()));
    }
}
