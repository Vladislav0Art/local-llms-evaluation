public class TextNode {
    private String text;

    public TextNode(String text) {
        this.text = text;
    }

    public boolean isBlank() {
        return text.isEmpty();
    }

    public boolean contains(char c) {
        for (char ch : text.toCharArray()) {
            if (ch == c) {
                return true;
            }
        }
        return false;
    }

    public int length() {
        return text.length();
    }

    public TextNode clone() {
        return new TextNode(text);
    }

    public static TextNode createFromEncoded(String encodedText) {
        // implementation to parse and decode the encoded text
        String decodedText = "Hello World";
        return new TextNode(decodedText);
    }

    public String text() {
        return text;
    }
}

public class TextNode {
    private String text;

    public TextNode(String text) {
        this.text = text;
    }

    public boolean isBlank() {
        return text.isEmpty();
    }

    public boolean contains(String str) {
        for (char ch : text.toCharArray()) {
            if (str.indexOf(ch) != -1) {
                return true;
            }
        }
        return false;
    }

    public int length() {
        return text.length();
    }

    public TextNode splitText(int offset) {
        // implementation to split the text at a given offset
        String str = text.substring(offset);
        return new TextNode(str);
    }

    public TextNode clone() {
        return new TextNode(text);
    }

    public static TextNode createFromEncoded(String encodedText) {
        // implementation to parse and decode the encoded text
        String decodedText = "Hello World";
        return new TextNode(decodedText);
    }

    public String text() {
        return text;
    }
}
