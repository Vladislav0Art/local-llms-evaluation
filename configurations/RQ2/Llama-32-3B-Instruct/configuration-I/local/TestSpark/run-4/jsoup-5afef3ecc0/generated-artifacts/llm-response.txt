import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;

import static org.junit.Assert.assertEquals;
import static org.mockito.Mockito.when;

import org.jsoup.helper.UrlBuilder;
import org.jsoup.helper.Connection.KeyVal;
import java.net.URL;
import java.util.Map;

@RunWith(org.junit.runner.JUnit4ClassRunner.class)
public class UrlBuilderTest {

    @Mock
    private URL inputUrl;

    public void setup() {
        MockitoAnnotations.initMocks(this);
        when(inputUrl.toString()).thenReturn("http://test.com");
    }

    @Test
    public void build_EmptyUrl_ReturnsNull() {
        UrlBuilder builder = new UrlBuilder((URL) null);
        assertEquals(null, builder.build());
    }

    @Test
    public void appendKeyVal_NoKeys_ThrowsUnsupportedEncodingException() {
        UrlBuilder builder = new UrlBuilder(inputUrl);
        when(builder.appendKeyVal(Mockito.any(Connection.KeyVal.class))).thenThrow(UnsupportedEncodingException.class);
        try {
            builder.appendKeyVal(null);
        } catch (UnsupportedEncodingException e) {
            assertEquals("Unsupported encoding", e.getMessage());
        }
    }

    @Test
    public void appendKeyVal_SingleKeyVal_BuildsCorrectUrl() {
        UrlBuilder builder = new UrlBuilder(inputUrl);
        Connection.KeyVal kv = Connection.KeyVal.of("a", "b");
        when(builder.appendKeyVal(Mockito.any(Connection.KeyVal.class))).thenReturn(kv);
        Map<String, String> params = builder.build().getParams();
        assertEquals(1, params.size());
        assertEquals("b", params.get("a"));
    }

    @Test
    public void appendMultipleKeyVal_BuildsCorrectUrl() {
        UrlBuilder builder = new UrlBuilder(inputUrl);
        Connection.KeyVal kv1 = Connection.KeyVal.of("c", "d");
        Connection.KeyVal kv2 = Connection.KeyVal.of("e", "f");
        when(builder.appendKeyVal(Mockito.any(Connection.KeyVal.class))).thenReturn(kv1).thenReturn(kv2);
        Map<String, String> params = builder.build().getParams();
        assertEquals(2, params.size());
        assertEquals("d", params.get("c"));
        assertEquals("f", params.get("e"));
    }

    @Test
    public void appendKeyVal_DuplicateKeys_ThrowsUnsupportedEncodingException() {
        UrlBuilder builder = new UrlBuilder(inputUrl);
        Connection.KeyVal kv1 = Connection.KeyVal.of("a", "b");
        Connection.KeyVal kv2 = Connection.KeyVal.of("a", "c");
        when(builder.appendKeyVal(Mockito.any(Connection.KeyVal.class))).thenReturn(kv1).thenThrow(UnsupportedEncodingException.class);
        try {
            builder.appendKeyVal(kv2);
        } catch (UnsupportedEncodingException e) {
            assertEquals("Duplicate key value", e.getMessage());
        }
    }

}
