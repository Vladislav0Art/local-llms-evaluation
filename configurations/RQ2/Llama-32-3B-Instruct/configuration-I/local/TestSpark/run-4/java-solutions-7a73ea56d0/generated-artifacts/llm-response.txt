public class OnlineStockSpan {

    private int[] stack = new int[0];

    public int next(int price) {
        if (price > stack[stack.length - 1]) {
            stack = Arrays.copyOf(stack, stack.length + 1);
            stack[stack.length - 1] = price;
        } else {
            while (!isEmpty() && price <= stack[stack.length - 1]) {
                int top = pop();
                if (price > top) {
                    push(price);
                    break;
                }
            }
            if (isEmpty()) {
                push(price);
            }
        }
        return stack.length;
    }

    private void push(int value) {
        stack = Arrays.copyOf(stack, stack.length + 1);
        stack[stack.length - 1] = value;
    }

    private int pop() {
        int last = stack[stack.length - 1];
        stack = Arrays.copyOfRange(stack, 0, stack.length - 1);
        return last;
    }

    private boolean isEmpty() {
        return stack.length == 0;
    }

    public int[] calculateSpans(int[] prices) {
        int[][] result = new int[prices.length][];
        for (int i = 0; i < prices.length; i++) {
            result[i] = new int[]{onlineStockSpan.next(prices[i])};
        }
        return result;
    }

}
