public class Tag {

    public static final String TAG = "tag";

}

public class Element {
    private String tagName;
    private java.util.List<Attribute> attributes;

    public void setTagName(String tagName) {
        this.tagName = tagName;
    }

    public String getTagName() {
        return tagName;
    }

    public java.util.List<Attribute> getAttributes() {
        return attributes;
    }

    public Element(String tagName) {
        this.tagName = tagName;
        this.attributes = new java.util.ArrayList<>();
    }

    public Element Tag(String tagName) {
        this(tagName);
        return this;
    }

    public Element addAttr(String key, Object value) {
        Attribute attribute = new Attribute(key, value);
        this.attributes.add(attribute);
        return this;
    }

    public void clearAttributes() {
        attributes.clear();
    }

    public java.util.List<Attribute> removeAttrStartingWith(String attrName) {
        int index = -1;
        for (int i=0; i < attributes.size(); i++) {
            if (attributes.get(i).getKey().startsWith(attrName)) {
                index = i;
                break;
            }
        }

        if (index != -1) {
            Attribute attributeToRemove = attributes.remove(index);
            return new java.util.ArrayList<>(attributes);
        } else {
            return new java.util.ArrayList<>();
        }
    }

    public static class Attribute {
        private String key;
        private Object value;

        public Attribute(String key, Object value) {
            this.key = key;
            this.value = value;
        }

        public String getKey() {
            return key;
        }

        public Object getValue() {
            return value;
        }
    }

    public static class GeneratedTest {

        @Test
        public void testRemoveAttrStartingWith() {
            Element element = new Element(Tag.TAG).addAttr("attr1", "value1").addAttr("attr2", "value2");
            assertEquals(1, element.removeAttrStartingWith("attr").size());
        }

        @Test
        public void testClearAttributes() {
            Element element = new Element(Tag.TAG).addAttr("attr1", "value1").addAttr("attr2", "value2");
            assertEquals(0, element.clearAttributes().getAttributeCount());
        }

        @Test
        public void testGetElementsContainingText() {
            Element element = new Element(Tag.TAG);
            assertEquals(1, element.getElementsContainingText("text").size());
        }
    }
}
