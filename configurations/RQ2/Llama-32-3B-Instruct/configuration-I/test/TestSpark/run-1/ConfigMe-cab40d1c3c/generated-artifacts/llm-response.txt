import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;

import java.util.ArrayList;
import java.util.List;
import java.util.Map;

import static org.junit.Assert.*;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class PropertyListBuilderTest {

    @Mock
    private List<Property<?>> mockProperties;

    @Mock
    private Map<String, Object> mockRootEntries;

    @Mock
    private Property<?> mockProperty;

    public PropertyListBuilderTest() {
        MockitoAnnotations.initMocks(this);
    }

    @Test
    public void add_PropertyIsNotNull() {
        when(mockProperty != null).thenReturn(true);
        PropertyListBuilder builder = new PropertyListBuilder();
        builder.add(mockProperty);
        assertNotNull(builder.getRootEntries());
    }

    @Test
    public void add_PropertyIsNull() {
        when(mockProperty == null).thenReturn(false);
        PropertyListBuilder builder = new PropertyListBuilder();
        try {
            builder.add(null);
            fail("Expected NullPointerException");
        } catch (NullPointerException e) {
            // expected
        }
    }

    @Test
    public void create_ReturnsNotNullList() {
        when(mockProperties != null).thenReturn(new ArrayList<>());
        PropertyListBuilder builder = new PropertyListBuilder();
        List<Property<?>> result = builder.create();
        assertNotNull(result);
        assertTrue(result.isEmpty());
    }

    @Test
    public void create_ReturnsNotEmptyList() {
        List<Property<?>> mockList = new ArrayList<>();
        mockProperties = mockList;
        when(mockProperties).thenReturn(new ArrayList<>());
        PropertyListBuilder builder = new PropertyListBuilder();

        assertNotNull(result);
        assertTrue(result.size() > 0);
    }

    @Test
    public void getRootEntries_ReturnsNotNullMap() {

        PropertyListBuilder builder = new PropertyListBuilder();
        Map<String, Object> result = builder.getRootEntries();
        assertNotNull(result);
    }

    @Test
    public void getRootEntries_ReturnsEmptyMapIfNoPropertiesAdded() {
        when(mockProperties == null).thenReturn(true);
        PropertyListBuilder builder = new PropertyListBuilder();
        Map<String, Object> result = builder.getRootEntries();
        assertTrue(result.isEmpty());
    }
}
