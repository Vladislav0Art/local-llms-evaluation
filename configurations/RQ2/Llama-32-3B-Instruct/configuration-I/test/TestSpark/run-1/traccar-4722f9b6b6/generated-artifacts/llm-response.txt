import org.junit.Test;
import static org.junit.Assert.*;
import java.util.Arrays;
import org.traccar.protocol.Protocol;
import org.traccar.protocol.WatchProtocolDecoder;
import org.traccar.protocol.BaseProtocolDecoder;
import org.traccar.model.Message;
import org.traccar.model.Device;
import org.traccar.model.Channel;
import org.traccar.model.SocketAddress;
import org.mockito.Mockito;

public class WatchProtocolDecoderTest {

    @Test
    public void getHasIndex_WithValidMessage_ReturnsTrue() {
        Message message = new Message("message", Arrays.asList("key"));
        assertTrue(WatchProtocolDecoder.getHasIndex(message));
    }

    @Test
    public void getHasIndex_WithoutMessage_ReturnsFalse() {
        assertFalse(WatchProtocolDecoder.getHasIndex(null));
    }

    @Test
    public void getManufacturer_MatchesExpectedValue() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(new Protocol("protocol"));
        assertEquals("manufacturer", decoder.getManufacturer());
    }

    @Test
    public void decode_DecodesMessageCorrectly() throws Exception {
        Channel channel = Mockito.mock(Channel.class);
        SocketAddress remoteAddress = Mockito.mock(SocketAddress.class);
        Message message = new Message("message", Arrays.asList("key"));
        Object result = WatchProtocolDecoder.decode(channel, remoteAddress, message);
        assertNotNull(result);
    }

    @Test
    public void decode_WithoutMessage_ReturnsEmptyString() throws Exception {
        Channel channel = Mockito.mock(Channel.class);
        SocketAddress remoteAddress = Mockito.mock(SocketAddress.class);
        Message message = null;
        Object result = WatchProtocolDecoder.decode(channel, remoteAddress, message);
        assertEquals("", result.toString());
    }

    @Test
    public void decode_WithUnsupportedMessage_ReturnsUnknown() throws Exception {
        Channel channel = Mockito.mock(Channel.class);
        SocketAddress remoteAddress = Mockito.mock(SocketAddress.class);
        Message message = new Message("message", Arrays.asList("unknown"));
        Object result = WatchProtocolDecoder.decode(channel, remoteAddress, message);
        assertEquals("unknown", result.toString());
    }

    @Test
    public void decode_WithInvalidMessage_ReturnsUnknown() throws Exception {
        Channel channel = Mockito.mock(Channel.class);
        SocketAddress remoteAddress = Mockito.mock(SocketAddress.class);
        Message message = new Message("message", Arrays.asList("key", "invalid"));
        Object result = WatchProtocolDecoder.decode(channel, remoteAddress, message);
        assertEquals("unknown", result.toString());
    }

    @Test
    public void decode_WithEmptyMessage_ReturnsUnknown() throws Exception {
        Channel channel = Mockito.mock(Channel.class);
        SocketAddress remoteAddress = Mockito.mock(SocketAddress.class);
        Message message = new Message("", Arrays.asList("key"));
        Object result = WatchProtocolDecoder.decode(channel, remoteAddress, message);
        assertEquals("unknown", result.toString());
    }

    @Test
    public void getDevice_WithMessage_ReturnsCorrectDevice() throws Exception {
        Channel channel = Mockito.mock(Channel.class);
        SocketAddress remoteAddress = Mockito.mock(SocketAddress.class);
        Message message = new Message("message", Arrays.asList("device"));
        Device device = WatchProtocolDecoder.getDevice(message);
        assertNotNull(device);
    }

    @Test
    public void getDevice_WithoutMessage_ReturnsNull() throws Exception {
        Channel channel = Mockito.mock(Channel.class);
        SocketAddress remoteAddress = Mockito.mock(SocketAddress.class);
        Message message = null;
        Device device = WatchProtocolDecoder.getDevice(message);
        assertNull(device);
    }
}
