public class KeyPair {
    private byte[] accountId;
    private byte[] secretSeed;
    private EdDSAPublicKey publicKey;

    public static class KeyPairBuilder {
        private byte[] accountId;
        private byte[] secretSeed;

        public KeyPairBuilder setAccountId(byte[] accountId) {
            this.accountId = accountId;
            return this;
        }

        public KeyPairBuilder setSecretSeed(byte[] secretSeed) {
            this.secretSeed = secretSeed;
            return this;
        }

        public KeyPair build() {
            return new KeyPair();
        }
    }

    public KeyPair(KeyPairBuilder builder) {
        this.accountId = builder.accountId;
        this.secretSeed = builder.secretSeed;
        this.publicKey = EdDSAPublicKey.fromXdr(builder.generatePublicKey());
    }

    public byte[] getAccountId() {
        return accountId;
    }

    public byte[] getSecretSeed() {
        return secretSeed;
    }
