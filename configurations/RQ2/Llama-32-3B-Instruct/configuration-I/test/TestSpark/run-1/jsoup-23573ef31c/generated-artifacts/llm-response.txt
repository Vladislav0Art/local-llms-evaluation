public class JsoupTest {
    public static Document document(String url) {
        return new org.jsoup.nodes.Document(url);
    }

    public static String location(Document doc) {
        return doc.location();
    }

    public static void titleSetString(Element element, String value) {
        element.text(value);
    }

    public static Element createElement(String tagName) {
        return new org.jsoup.nodes.Element(tagName);
    }

    public static String outerHtml(Document document) {
        StringBuilder html = new StringBuilder();
        html.append(document.outerHtml());
        return html.toString();
    }

    public static String nodeName(Element element) {
        return element.nodeName();
    }

    public static void charsetSetCharset(Charset charset, Document document) {
        document.charset(charset);
    }
}
