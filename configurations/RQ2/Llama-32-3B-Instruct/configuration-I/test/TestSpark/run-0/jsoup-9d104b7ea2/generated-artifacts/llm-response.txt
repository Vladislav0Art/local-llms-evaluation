import org.jsoup.nodes.Document;
import org.junit.Test;
import static org.junit.Assert.*;
import java.io.ByteArrayOutputStream;
import java.util.Arrays;

public class TextNodeTest {

    @Test
    public void getWholeText_EmptyString_ReturnsEmptyString() {
        String text = "";
        TextNode node = new TextNode(text);
        assertEquals("", node.getWholeText());
    }

    @Test
    public void getWholeText_NonEmptyString_ReturnsNonEmptyString() {
        String text = "Hello, World!";
        TextNode node = new TextNode(text);
        assertEquals("Hello, World!", node.getWholeText());
    }

    @Test
    public void isBlank_WhitespaceOnly_ReturnsTrue() {
        String text = "\t\n\r";
        TextNode node = new TextNode(text);
        assertTrue(node.isBlank());
    }

    @Test
    public void isBlank_NoWhitespace_ReturnsFalse() {
        String text = "Hello, World!";
        TextNode node = new TextNode(text);
        assertFalse(node.isBlank());
    }

    @Test
    public void splitText_SplitAtZeroOffset_ReturnsOriginalString() {
        String text = "Hello, World!";
        TextNode node = new TextNode(text);
        assertEquals("Hello, World!", node.splitText(0));
    }

    @Test
    public void splitText_SplitAtNonZeroOffset_ReturnsSubstring() {
        String text = "Hello, World!";
        TextNode node = new TextNode(text);
        assertEquals("World", node.splitText(6));
    }

    @Test
    public void clone_CloneReturnsCloneOfOriginal() {
        String text = "Hello, World!";
        TextNode original = new TextNode(text);
        TextNode clone = original.clone();
        assertEquals(original, clone);
    }

    @Test
    public void createFromEncoded_EncodedTextReturnsTextNode() throws IOException {
        String encodedText = "text";
        TextNode node = TextNode.createFromEncoded(encodedText);
        assertNotNull(node);
    }

    @Test
    public void toString_ReturnsStringRepresentation() {
        String text = "Hello, World!";
        TextNode node = new TextNode(text);
        assertEquals("text", node.toString());
    }
}

public class AppendableTest {

    @Test
    public void append_WritesStringToAppendable() {
        Appendable target = new StringBuilder();
        target.append("Hello, World!");
        assertTrue(target instanceof String);
    }

    @Test
    public void appendAll_WritesStringArrayToAppendable() {
        Appendable target = new StringBuilder();
        char[] array = "Hello, World!".toCharArray();
        target.appendAll(array);
        assertTrue(target instanceof String);
    }
}
