import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.TextNode;
import org.junit.Test;
import static org.junit.Assert.*;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

    @Mock
    private Appendable appendable;

        TextNode textNode = new TextNode(text);
        assertEquals(text, textNode.text());

        TextNode textNode = new TextNode(text);
        assertEquals("text-node", textNode.nodeName());
    }

    public void clone_GivesSameTextNode() {
        String text = "Hello World";
        TextNode textNode = new TextNode(text);
        TextNode clonedTextNode = textNode.clone();
        assertNotNull(clonedTextNode);
        assertEquals(text, clonedTextNode.text());
    }

    @Test
    public void getWholeText_ReturnsCorrectText() {
        String text = "Hello World";
        TextNode textNode = new TextNode(text);
        assertEquals(text, textNode.getWholeText());
    }

    public void isBlank_ReturnsTrueIfTextIsEmptyOrContainsOnlyWhitespace() {

        assertTrue(TextNode.isBlank(text));
        text = "   ";
        assertTrue(TextNode.isBlank(text));
    }

    public void splitText_SplitsCorrectly() {

    }

    @Test
    public void outerHtmlTail_PutsEndTagIntoHTML() {
        String html = "<html><body>";
        appendable.write(html);
        new TextNode("Hello World").outerHtmlTail(appendable, 0, document_OUTPUT_SETTINGS);
        assertEquals("<div> Hello World </div></body></html>", appendable.toString());
    }

    @Test
    public void normaliseWhitespace_ConvertsTabsToSpaces() {
        String text = "  Hello World";
        assertEquals("Hello World", TextNode.normaliseWhitespace(text));
    }

    @Test
    public void stripLeadingWhitespace_StripsOnlyLeadingWhitespace() {
        String text = "   Hello World";
        assertEquals("Hello World", TextNode.stripLeadingWhitespace(text));
    }

    @Test
    public void lastCharIsWhitespace_ReturnsTrueIfLastCharacterIsWhitespace() {
        StringBuilder sb = new StringBuilder();
        assertTrue(TextNode.lastCharIsWhitespace(sb.append(' ').toString()));
        assertFalse(TextNode.lastCharIsWhitespace(sb.append('a').toString()));
    }
}

public interface TextNode {
    String text();
    String nodeName();
    void outerHtmlHead(Appendable appendable, int offset, String document_OUTPUT_SETTINGS);
    void outerHtmlTail(Appendable appendable, int offset, String document_OUTPUT_SETTINGS);

    static boolean isBlank(String text) {
        return text == null || text.isEmpty() || text.trim().isEmpty();
    }

    static String normaliseWhitespace(String text) {
        return text.replace("\t", " ");
    }

    static String stripLeadingWhitespace(String text) {
        int leadingWhitespace = 0;
        while (leadingWhitespace < text.length() && Character.isWhitespace(text.charAt(leadingWhitespace))) {
            leadingWhitespace++;
        }
        if (leadingWhitespace == text.length()) {
            return "";
        } else {
            return text.substring(leadingWhitespace);
        }
    }

    static boolean lastCharIsWhitespace(StringBuilder sb) {
        char lastChar = sb.charAt(sb.length() - 1);
        return Character.isWhitespace(lastChar);
    }

    public class TextNodeImpl implements TextNode {

        private String text;

        public TextNodeImpl(String text) {
            this.text = text;
        }

        @Override
        public String text() {
            return text;
        }

        @Override
        public String nodeName() {
            return "text-node";
        }

        @Override
        public void outerHtmlHead(Appendable appendable, int offset, String document_OUTPUT_SETTINGS) {
            // Implementation
        }

        @Override
        public void outerHtmlTail(Appendable appendable, int offset, String document_OUTPUT_SETTINGS) {
            // Implementation
        }
    }
}
