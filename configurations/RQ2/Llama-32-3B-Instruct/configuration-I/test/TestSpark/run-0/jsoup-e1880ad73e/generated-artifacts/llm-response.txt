import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertTrue;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class UrlBuilderTest {

    @Mock
    private URL inputUrl;

    @Mock
    private Connection.KeyVal kv;

    public UrlBuilder urlBuilderInstance() {
        return new UrlBuilder(inputUrl);
    }

    @Test
    public void buildUrlWithCorrectInput() {
        when(inputUrl.toString()).thenReturn("http://example.com");
        UrlBuilder builder = urlBuilderInstance();
        URL result = builder.build();
        assertEquals("http://example.com", result.toString());
    }

    @Test
    public void buildUrlWithoutInput() {
        UrlBuilder builder = urlBuilderInstance();
        URL result = builder.build();
        assertTrue(result == null || result.toString().isEmpty());
    }

    @Test
    public void appendKeyValWithValidKeyAndValue() throws UnsupportedEncodingException {
        when(kv.getKey()).thenReturn("key");
        when(kv.getValue()).thenReturn("value");
        UrlBuilder builder = urlBuilderInstance();
        builder.appendKeyVal(kv);
        assertEquals("http://example.com?key=value", builder.build().toString());
    }

    @Test
    public void appendKeyValWithoutValidValue() {
        Connection.KeyVal invalidKV = new Connection.KeyVal(null, "invalidValue");
        UrlBuilder builder = urlBuilderInstance();
        try {
            builder.appendKeyVal(invalidKV);
            assert false : "Expected UnsupportedEncodingException to be thrown";
        } catch (UnsupportedEncodingException e) {}
    }

    @Test
    public void appendKeyValWithNullKey() {
        Connection.KeyVal invalidKV = new Connection.KeyVal(null, null);
        UrlBuilder builder = urlBuilderInstance();
        try {
            builder.appendKeyVal(invalidKV);
            assert false : "Expected NullPointerException to be thrown";
        } catch (NullPointerException e) {}
    }
}

    }

    }

public class UrlBuilder {

    private final java.util.Map<String, String> keyValue = new java.util.HashMap<>();

    public UrlBuilder(MockingBean mockingBean) {
        this.keyValue.put("key", (String) mockingBean.getMocking().get("http://example.com"));

        return keyValue;
