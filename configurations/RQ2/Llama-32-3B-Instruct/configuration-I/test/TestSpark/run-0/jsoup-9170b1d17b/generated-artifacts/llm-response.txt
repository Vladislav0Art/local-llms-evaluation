import org.jsoup.nodes.Attribute;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Attributes {

    private java.util.Map<String, String> map = new java.util.HashMap<>();

    public void put(String key, String value) {
        this.map.put(key, value);
    }

    public boolean remove(String key) {
        if (this.map.containsKey(key)) {
            String result = this.map.remove(key);
            return true;
        }
        return false;
    }

    public boolean hasKey(String key) {
        return this.map.containsKey(key);
    }

    public boolean hasDeclaredValueForKey(String key) {
        return this.map.containsKey(key);
    }

    public void add(String key, String value) {
        this.put(key, value);
    }

    public String get(String key) {
        if (this.hasKey(key)) {
            return this.map.get(key);
        }
        throw new RuntimeException("Key not found: " + key);
    }

    public boolean getUserData(String key, Object data) {
        if (this.hasDeclaredValueForKey(key)) {
            return true;
        } else {
            return false;
        }
    }

    public void putUserDate(String key, String value) {
        // currently does nothing
    }

    public static class Attribute {

        private String key;
        private String value;

        public Attribute(String key, String value) {
            this.key = key;
            this.value = value;
        }

        public String getKey() {
            return key;
        }

        public String getValue() {
            return value;
        }
    }
}
