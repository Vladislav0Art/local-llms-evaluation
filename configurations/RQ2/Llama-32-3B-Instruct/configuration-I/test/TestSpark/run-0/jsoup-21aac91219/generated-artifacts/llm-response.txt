        HtmlTreeBuilder htmlTreeBuilder = new HtmlTreeBuilder();

        Document document = new Document();
        htmlTreeBuilder.initialiseParse(document, parser);
        System.out.println(parser.getAllowTags());
        System.out.println(parser.getAllowEntities());
        System.out.println(parser.getAllowComments());

        Element element = (Element) document.body().get(0);

        Node node = element childNode(0);
        if (node instanceof TextNode) {
            System.out.println(((TextNode) node).getData());
        }

        Tag tag = (Tag) element.tagName();
        System.out.println(tag.getValue());

        parser.setAllowTags(true);
        parser.setAllowEntities(false);
        parser.setAllowComments(true);

        htmlTreeBuilder.defaultSettings(parser);

        NodeContainer container = document.body().get(0);
        if (container instanceof NodeVisitor) {
            ((NodeVisitor) container).accept(node);
        }
    }

    public static class HtmlTreeBuilder {

            return new Parser();
