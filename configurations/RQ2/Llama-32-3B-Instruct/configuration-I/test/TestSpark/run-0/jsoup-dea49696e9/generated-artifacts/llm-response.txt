import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import static org.hamcrest.MatcherAssert.assertThat;
import static org.hamcrest.Matchers.is;
import static org.junit.Assert.assertEquals;
import static org.mockito.Mockito.when;

public class HtmlTreeBuilderTest {

    @Mock
    private TreeBuilder treeBuilderMock;

    @Mock
    private ParseSettings parseSettingsMock;

    @Mock
    private Parser parserMock;

    @Mock
    private Reader inputReaderMock;

    @Mock
    private Element elementMock;

    public static class Token {
        public int type;
        public String value;

        public Token(int type, String value) {
            this.type = type;
            this.value = value;
        }
    }

    public static class HtmlTreeBuilderState {
        public Node node;
    }

    @Test
    public void givenDefaultSettingsAndNoInputWhenParseFragmentThenReturnEmptyList() {
        when(treeBuilderMock.defaultSettings()).thenReturn(parseSettingsMock);
        List<Node> result = HtmlTreeBuilder.parseFragment("", null, "baseUri", parserMock);
        assertThat(result, is(null));
    }

    @Test
    public void givenInputFragmentAndNoContextWhenParseFragmentThenReturnEmptyList() {
        when(treeBuilderMock.defaultSettings()).thenReturn(parseSettingsMock);
        List<Node> result = HtmlTreeBuilder.parseFragment("<div></div>", null, "baseUri", parserMock);
        assertThat(result, is(null));
    }

    @Test
    public void givenInputFragmentAndContextWhenParseFragmentThenReturnCorrectNode() {
        when(treeBuilderMock.defaultSettings()).thenReturn(parseSettingsMock);
        List<Node> result = HtmlTreeBuilder.parseFragment("<div></div>", elementMock, "baseUri", parserMock);
        assertThat(result.get(0).value, is("div"));
    }

    @Test
    public void givenProcessAndTokenWhenProcessThenReturnCorrectBoolean() {
        when(parserMock.process(any(Token.class))).thenReturn(true);
        boolean result = HtmlTreeBuilder.process(new Token(1, "tokenValue"));
        assertThat(result, is(true));
    }

    @Test
    public void testProcessWithNullToken() {
        when(parserMock.process(any(Token.class))).thenReturn(false);
        boolean result = HtmlTreeBuilder.process(null);
        assertThat(result, is(false));
    }
}
