    public void createKeyPairFromAccountIdAndSeedShouldReturnTrue() {

    }

    public void getAccountIdEmptySeedShouldReturnNull() {
        KeyPair keyPair = KeyPair.fromSecretSeed(new char[0]);
        assertNull(keyPair.getAccountId());
