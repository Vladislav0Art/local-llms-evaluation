import org.junit.Test;
import static org.hamcrest.CoreMatchers.is;
import static org.hamcrest.MatcherAssert.assertThat;
import static org.jsoup.select.Elements.select;
import static org.mockito.Mockito.when;
import static org.mockito.Mockito.mock;

public class ElementTest {

    @Test
    public void givenElementTagReturnsTagName() {
        when("tag").thenReturn("div");
        Element element = new Element("tag");
        assertThat(element.tagName(), is("div"));
    }

    @Test
    public void givenElementTagReturnsNormalName() {
        when("tag").thenReturn("div");
        Element element = new Element("tag");
        assertThat(element.normalName(), is("div"));
    }

    @Test
    public void givenElementTextReturnsText() {
        when("text").thenReturn("Hello World");
        Element element = new Element("text");
        assertThat(element.text(), is("Hello World"));
    }

    @Test
    public void givenElementHasClassReturnsTrueWhenClassNameIsPresent() {
        when("className").thenReturn("div-class");
        Element element = new Element("className");
        assertThat(element.hasClass("class"), is(true));
    }

    @Test
    public void givenElementHasClassReturnsFalseWhenClassNameIsAbsent() {
        when("className").thenReturn("");
        Element element = new Element("className");
        assertThat(element.hasClass("class"), is(false));
    }

    @Test
    public void givenElementRemoveAttrRemovesAttributeFromHtml() {
        when("key").thenReturn("data-foo");
        when("value").thenReturn("bar");
        Element element = new Element("value");
        element.removeAttr("data-foo");
        assertThat(element.html(), is("<p>bar</p>"));
    }

    @Test
    public void givenElementRootReturnsElementRoot() {
        when("parent").thenReturn(mock(Element.class));
        Element element = new Element("parent");
        assertThat(element.root(), is(element));
    }

    @Test
    public void givenElementAddClassAddsCssClassToElement() {
        when("className").thenReturn("div-class");
        Element element = new Element();
        element.addClass("class");
        assertThat(select(element, "div.class").size(), is(1));
    }
}

// ElementTest.java
public class ElementTest {

    private Element element;

    @Before
    public void setUp() {
        element = new Element("div");
    }

    @Test
    public void givenElementTagReturnsTagName() {
        String expectedTag = "div";
        assertThat(element.tagName(), is(expectedTag));
    }

    @Test
    public void givenElementTextReturnsText() {
        String expectedText = "";
        when("text").thenReturn(expectedText);
        element.setText(expectedText);
        assertThat(element.textValue(), is(expectedText));
    }

    @Test
    public void givenElementHasClassReturnsTrueWhenClassNameIsPresent() {
        String className = "class";
        when("className").thenReturn(className);
        when("className", () -> true).when("hasClass", (s) -> s.equals(className));
        element.hasClass(className);
        boolean expectedResult = true;
        assertThat(element.hasClass(className), is(expectedResult));
    }

    @Test
    public void givenElementHasClassReturnsFalseWhenClassNameIsAbsent() {
        String className = "class";
        when("className").thenReturn("");
        when("hasClass", (s) -> !s.equals(className));
        element.hasClass(className);
        boolean expectedResult = false;
        assertThat(element.hasClass(className), is(expectedResult));
    }

    @Test
    public void givenElementAddsClassReturnsAddedClass() {
        String className = "class";
        Element addedElement = new Element("div");
        when("addClass", (s) -> addedElement).when("hasClass", (s) -> s.equals(className));
        element.addClass(className);
        boolean expectedResult = true;
        assertThat(addedElement.hasClass(className), is(expectedResult));
    }
}
