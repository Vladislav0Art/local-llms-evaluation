    }

    public void setCommentForNewPathAddsNewEntry() {
        Map<String, List<String>> comments = new HashMap<>();

        assertThat(configuration.getAllComments().get("new path"), is(Arrays.asList("comment")));

    public void setCommentForNewPathAddsEmptyLineIfNoMarker() {
        Map<String, List<String>> comments = new HashMap<>();
        when(comments).thenReturn(comments);

        try {
            configuration.getAllComments().put("key", null);
            assert false;
        } catch (UnsupportedOperationException e) {
        }
    }

}

import org.junit.Test;
import static org.hamcrest.MatcherAssert.assertThat;
import static org.hamcrest.Matchers.is;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;

public class CommentsConfiguration {

    private Map<String, List<String>> comments = new HashMap<>();

    public void setComment(String path, String commentLine) {
        if (!comments.containsKey(path)) {
            comments.put(path, new ArrayList<>());
        }
        comments.get(path).add(commentLine);
    }

    public Map<String, List<String>> getAllComments() {
        return Collections.unmodifiableMap(comments);
    }

}

public class CommentsConfigurationTest {

    @Test
    public void defaultConstructorIsCreated() {
        CommentsConfiguration configuration = new CommentsConfiguration();
        assertThat(configuration, is(null));
    }

    @Test
    public void constructorNoArgumentsIsCreated() {
        CommentsConfiguration configuration = new CommentsConfiguration();
        assertThat(configuration.getAllComments(), is(Map.of()));
    }

    @Test
    public void setCommentForExistingPathOverwritesOldValue() {
        CommentsConfiguration configuration = new CommentsConfiguration();
        String commentLine = "new comment";
        configuration.setComment("path", commentLine);
        Map<String, List<String>> comments = configuration.getAllComments();
        assertThat(comments.get("path"), is(List.of(commentLine)));
    }

    @Test
    public void setCommentForNewPathAddsNewEntry() {
        CommentsConfiguration configuration = new CommentsConfiguration();
        String commentLine = "comment";
        configuration.setComment("new path", commentLine);
        Map<String, List<String>> comments = configuration.getAllComments();
        assertThat(comments.get("new path"), is(List.of(commentLine)));
    }

    @Test
    public void setCommentForNewPathAddsEmptyLineIfNoMarker() {
        CommentsConfiguration configuration = new CommentsConfiguration();
        String[] commentLines = {"--", "", "--"};
        String emptyLine = "";
        for (String line : commentLines) {
            if (!line.equals("--")) {
                continue;
            }
            configuration.setComment("new path", emptyLine);
        }
        Map<String, List<String>> comments = configuration.getAllComments();
        assertThat(comments.get("new path"), is(List.of(emptyLine)));
    }

    @Test
    public void getAllCommentsReturnsUnmodifiableMap() {
        CommentsConfiguration configuration = new CommentsConfiguration();
        String commentLine = "comment";
        configuration.setComment("path", commentLine);
        Map<String, List<String>> comments = configuration.getAllComments();
        assertThat(comments.get("path"), is(List.of(commentLine)));
        try {
            comments.put("key", null);
            assert false;
        } catch (UnsupportedOperationException e) {
        }
    }

}
