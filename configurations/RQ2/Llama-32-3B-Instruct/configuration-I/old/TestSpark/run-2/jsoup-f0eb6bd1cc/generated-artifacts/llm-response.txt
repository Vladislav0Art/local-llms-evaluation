import org.jsoup.helper.UrlBuilder;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import java.net.URL;
import java.util.Arrays;
import java.util.Map;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertThrows;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class UrlBuilderTest {

    @Mock
    private URL inputUrl;

    @Test
    public void build_URL_BasedOnInputUrl() {
        when(inputUrl.toExternalForm()).thenReturn("https://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        assertEquals("https://example.com", urlBuilder.build().toExternalForm());
    }

    @Test
    public void appendKeyVal_AppendsKeyValueToUrl() {
        Connection.KeyVal kv = Connection.KeyVal.builder().key("test").value("value").build();
        when(inputUrl.toExternalForm()).thenReturn("https://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        urlBuilder.appendKeyVal(kv);
        assertEquals("https://example.com?test=value", urlBuilder.build().toExternalForm());
    }

    @Test
    public void appendKeyVal_AppendsMultipleKeyValueToUrl() {
        Connection.KeyVal kv1 = Connection.KeyVal.builder().key("test1").value("value1").build();
        Connection.KeyVal kv2 = Connection.KeyVal.builder().key("test2").value("value2").build();
        when(inputUrl.toExternalForm()).thenReturn("https://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        urlBuilder.appendKeyVal(kv1);
        urlBuilder.appendKeyVal(kv2);
        assertEquals("https://example.com?test1=value1&test2=value2", urlBuilder.build().toExternalForm());
    }

    @Test
    public void appendKeyVal_AppendsKeyValueToEmptyUrl() {
        when(inputUrl.toExternalForm()).thenReturn("");
        Connection.KeyVal kv = Connection.KeyVal.builder().key("test").value("value").build();
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        urlBuilder.appendKeyVal(kv);
        assertEquals("https://example.com?test=value", urlBuilder.build().toExternalForm());
    }

    @Test
    public void appendKeyVal_ThrowsUnsupportedEncodingException() {
        Connection.KeyVal kv = Connection.KeyVal.builder().key("test").value("value").build();
        when(inputUrl.toExternalForm()).thenReturn("");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        assertThrows(UnsupportedEncodingException.class, () -> urlBuilder.appendKeyVal(kv));
    }

    @Test
    public void appendKeyVal_MultipleConnections() {
        Connection.KeyVal kv1 = Connection.KeyVal.builder().key("test1").value("value1").build();
        Connection.KeyVal kv2 = Connection.KeyVal.builder().key("test2").value("value2").build();
        when(inputUrl.toExternalForm()).thenReturn("");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        urlBuilder.appendKeyVal(kv1);
        urlBuilder.appendKeyVal(kv2);
        assertEquals("https://example.com?test1=value1&test2=value2", urlBuilder.build().toExternalForm());
    }
}
