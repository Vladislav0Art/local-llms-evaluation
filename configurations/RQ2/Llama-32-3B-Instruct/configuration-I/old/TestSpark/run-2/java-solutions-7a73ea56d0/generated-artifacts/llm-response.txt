public class OnlineStockSpan {

    private int[] spans;

    public OnlineStockSpan() {
        this.spans = new int[0];
    }

    public int next(int price) {
        return calculateSpan(price);
    }

    public int[] calculateSpans(int[] prices) {
        int[] newSpans = new int[prices.length + 1];
        System.arraycopy(spans, 0, newSpans, 0, spans.length);

        for (int i = 0; i < prices.length; i++) {
            if (prices[i] > price) {
                newSpans[i + 1] = newSpans[i] + 1;
            } else {
                int j = findIndex(prices, price);
                newSpans[j + 1] = newSpans[j] + 1;
            }
        }

        spans = newSpans;
        return spans;
    }

    private int findIndex(int[] array, int element) {
        for (int i = 0; i < array.length; i++) {
            if (array[i] == element) {
                return i;
            }
        }
        throw new RuntimeException();
    }

    private int calculateSpan(int price) {
        int result = 1;
        while (!pricesArray.isEmpty() && pricesArray.get(0) <= price) {
            result++;
            pricesArray.remove(0);
        }
        return result;
    }
}

    }

    public int next(int price) {
        while (index < prices.length && prices[index] <= price) {
            index++;
        }
        int span = index == prices.length ? 1 : index - pricesArray.get(index - 1);

import org.junit.Test;
import org.junit.runner.RunWith;
import java.util.Arrays;

public class OnlineStockSpan {

    private int[] pricesArray;
    private int index;
    public int totalSpans = 0;

    public OnlineStockSpan() {
        this.pricesArray = new int[0];
        this.index = 0;
    }

    public int next(int price) {
        while (index < pricesArray.length && pricesArray[index] <= price) {
            index++;
        }
        int span = index == pricesArray.length ? 1 : index - pricesArray[index - 1];
        pricesArray = Arrays.copyOf(pricesArray, index + 1);
        totalSpans += span;
        pricesArray[pricesArray.length-1] = span;
        return span;
    }

    public int[] calculateSpans(int[] prices) {
        this.pricesArray = prices;
        return new int[]{totalSpans};
    }
}
