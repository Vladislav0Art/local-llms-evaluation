import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;

import java.util.List;

public class UrlBuilderTest {

    @Mock
    private Connection connection;

    @Test
    public void ofNullUrl_ThrowsNullPointerException() {
        assertThrows(NullPointerException.class, () -> new UrlBuilder(null));
    }

    @Test
    public void build_HtmlUrl() throws MalformedURLException {
        URL url = Mockito.mock(URL.class);

        Mockito.verify(url).toExternalForm();
    }

    public void appendKeyVal_AddsKeyValueParam() throws UnsupportedEncodingException {
        Connection.KeyVal kv = Mockito.mock(Connection.KeyVal.class);
        URL url = Mockito.mock(URL.class);

        builder.appendKeyVal(kv);
        Mockito.verify(url).setPath(Mockito.anyString());
    }

    @Test
    public void appendKeyVal_DoesNotChangeExistingQueryParams() throws UnsupportedEncodingException {
        Connection.KeyVal kv1 = Mockito.mock(Connection.KeyVal.class);
        URL url = Mockito.mock(URL.class);
        UrlBuilder builder = new UrlBuilder(url);
        builder.appendKeyVal(kv1);
        Mockito.verify(url).setPath(Mockito.anyString());
    }

    @Test
    public void appendKeyVal_Multiplereturns() {
        Connection.KeyVal kv1 = Mockito.mock(Connection.KeyVal.class);

        UrlBuilder builder = new UrlBuilder(url);
        builder.appendKeyVal(kv1);

    }

    @Test
    public void appendKeyVal_ThrowsUnsupportedEncodingException() {

}

import org.junit.Test;
import org.mockito.Mock;
import org.mockito.Mockito;

public class UrlBuilderTest {

    @Mock
    private Connection connection;

    public static class UrlBuilder {
        public UrlBuilder(URL inputUrl) { /* implementation */ }

        public URL build() { /* implementation */ }

        public void appendKeyVal(Connection.KeyVal kv) throws UnsupportedEncodingException { /* implementation */ }
    }

    public static class Connection {
        public interface KeyVal {}

        public static class KeyValImpl implements KeyVal {}
    }

    @Test
    public void ofNullUrl_ThrowsNullPointerException() {
        assertThrows(NullPointerException.class, () -> new UrlBuilder(null));
    }

    @Test
    public void build_HtmlUrl() throws MalformedURLException {
        URL url = new URL("http://example.com");
        UrlBuilder builder = new UrlBuilder(url);
        Mockito.when(builder.build()).thenReturn(url);
        String builtUrl = builder.build().toString();
        Mockito.verify(url).toExternalForm();
        System.out.println(builtUrl);
    }

    @Test
    public void appendKeyVal_AddsKeyValueParam() throws UnsupportedEncodingException {
        Connection.KeyVal kv = new Connection.KeyValImpl();
        URL url = new URL("http://example.com");
        UrlBuilder builder = new UrlBuilder(url);
        Mockito.when(builder.appendKeyVal(kv)).thenReturn(url);
        String builtUrl = builder.appendKeyVal(kv).toString();
        System.out.println(builtUrl);
    }

    @Test
    public void appendKeyVal_DoesNotChangeExistingQueryParams() throws UnsupportedEncodingException {
        Connection.KeyVal kv1 = new Connection.KeyValImpl();
        URL url = new URL("http://example.com?param=value");
        UrlBuilder builder = new UrlBuilder(url);
        Mockito.when(builder.appendKeyVal(kv1)).thenReturn(url);
        String builtUrl = builder.appendKeyVal(kv1).toString();
        System.out.println(builtUrl);
    }

    @Test
    public void appendKeyVal_Multiplereturns() {
        Connection.KeyVal kv1 = new Connection.KeyValImpl();
        URL url = new URL("http://example.com");
        UrlBuilder builder = new UrlBuilder(url);
        Mockito.when(builder.appendKeyVal(kv1)).thenReturn(url);
        String builtUrl = builder.appendKeyVal(kv1).toString();
        System.out.println(builtUrl);
    }

    @Test
    public void appendKeyVal_ThrowsUnsupportedEncodingException() {
        Connection.KeyVal kv = new Connection.KeyValImpl();
        URL url = new URL("http://example.com");
        UrlBuilder builder = new UrlBuilder(url);
        assertThrows(UnsupportedEncodingException.class, () -> builder.appendKeyVal(kv));
    }
}
