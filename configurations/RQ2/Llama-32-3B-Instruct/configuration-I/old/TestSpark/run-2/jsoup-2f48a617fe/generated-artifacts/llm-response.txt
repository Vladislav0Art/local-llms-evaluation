import static org.mockito.Mockito.when;

    private Document document;

    private Appendable appendable;

    public void nodeName_Test() {
        when(document.nodeName()).thenReturn("comment");
        assertEquals("comment", comment.nodeName());
    }

    @Test
    public void getData_Test() {
        comment.setData("test");
        assertEquals("test", comment.getData());
    }

    @Test
    public void setData_Test() {
        comment.setData(null);
        assertNull(comment.getData());
    }

    @Test
    public void isXmlDeclaration_Test() {
        // This test case should return false because there is no XML declaration in the document.
        when(document.toString()).thenReturn("<html>test</html>");
        assertFalse(comment.isXmlDeclaration());
    }

    @Test
    public void asXmlDeclaration_Test() {
        Comment xmlDeclaration = new XmlDeclaration();
        // This test case should return an XmlDeclaration object with the correct data.
        when(document.toString()).thenReturn("<?xml version=\"1.0\"?><declaration>test</declaration>");
        assertNotNull(comment.asXmlDeclaration());
        assertTrue(comment.asXmlDeclaration() instanceof XmlDeclaration);
    }

    @Test
    public void clone_Test() {
        Comment clonedComment = comment.clone();
        assertEquals(comment, clonedComment);
