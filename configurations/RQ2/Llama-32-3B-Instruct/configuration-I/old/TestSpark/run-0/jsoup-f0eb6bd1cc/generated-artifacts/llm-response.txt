import org.jsoup.helper.UrlBuilder;
import org.junit.BeforeClass;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;

import java.net.URL;
import java.util.Map;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertThrows;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class UrlBuilderTest {

    @Mock
    private URL inputUrl;

    @Mock
    private Connection.KeyVal kv;

    private UrlBuilder urlBuilder;

    @BeforeClass
    public static void setup() {
        MockitoAnnotations.initMocks(new UrlBuilderTest());
    }

    public UrlBuilder getURLBuilder() {
        return new UrlBuilder(inputUrl);
    }

    @Test
    public void buildUrl_ValidInput_ReturnsValidUrl() {
        String url = "https://example.com";
        when(inputUrl.toString()).thenReturn(url);
        UrlBuilder builder = getURLBuilder();
        URL result = builder.build();
        assertEquals(url, result.toString());
    }

    @Test
    public void buildUrl_InvalidInput_ThrowsException() {
        assertThrows(UnsupportedEncodingException.class, () -> getURLBuilder().build());
    }

    @Test
    public void appendKeyVal_ValidKV_AddsKeyValueToUrl() {
        String url = "https://example.com";
        when(inputUrl.toString()).thenReturn(url);
        UrlBuilder builder = getURLBuilder();
        builder.appendKeyVal(kv);
        URL result = builder.build();
        assertEquals("https://example.com?key=value", result.toString());
    }

    @Test
    public void appendKeyVal_NullKV_ThrowsException() {
        assertThrows(NullPointerException.class, () -> getURLBuilder().appendKeyVal(null));
    }
}

public class UrlBuilder {
    private URL inputUrl;

    public UrlBuilder(URL inputUrl) {
        this.inputUrl = inputUrl;
    }

    public void appendKeyVal(Connection.KeyVal kv) {
        // Implementation to add key-value pair to the url
    }

    public URL build() {
        return new URL(inputUrl.toString());
    }
}

public class Connection {
    public static class KeyVal {
        private String key;
        private String value;

        public KeyVal(String key, String value) {
            this.key = key;
            this.value = value;
        }

        public static KeyVal of(String key, String value) {
            return new KeyVal(key, value);
        }
    }
}
