import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import java.util.ArrayList;
import java.util.List;
import java.util.Map;

import static org.junit.Assert.*;
import static org.mockito.Mockito.*;

@RunWith(MockitoJUnitRunner.class)
public class CommentsConfigurationTest {

    @Mock
    private Map<String, List<String>> commentsMap;

    @InjectMocks
    private CommentsConfiguration commentsConfiguration;

    @Test
    public void newCommentsConfiguration_isNotNull() {
        assertNotNull(commentsConfiguration);
    }

    @Test
    public void newCommentsConfiguration_withMapIsInitializedFromMap() {
        when(commentsMap).anyArguments().thenReturn(commentsMap);
        assertNotNull(commentsConfiguration.getAllComments());
    }

    @Test
    public void setComment_emptyLineShouldNotLeaveEmptyLineWithoutMarker() {
        commentsConfiguration.setComment("path", "", "comment");
        assertTrue(!commentsConfiguration.getAllComments().get("path").isEmpty());
    }

    @Test
    public void setComment_singleLineCommentShouldBeAddedAsSingleElementList() {
        commentsConfiguration.setComment("path", "comment");
        assertEquals(1, commentsConfiguration.getAllComments().get("path").size());
    }

    @Test
    public void setComment_multipleLinesShouldAddAllElementsToPath() {
        List<String> lines = new ArrayList<>();
        lines.add("comment1");
        lines.add("comment2");
        commentsConfiguration.setComment("path", "line1", "line2", "comment", "line3", "comment4");
        assertEquals(5, commentsConfiguration.getAllComments().get("path").size());
    }

    @Test
    public void getAllComments_isUnmodifiableView() {
        Map<String, List<String>> map = new HashMap<>();
        map.put("path", new ArrayList<>());
        when(commentsMap).thenReturn(map);
        assertTrue(!commentsConfiguration.getAllComments().containsValue(null));
    }
}

import org.junit.Test;
import java.util.*;

public class CommentConfigurationTests {

    @Test
    public void testSetComment() {
        // Arrange
        CommentConfiguration comments = new CommentConfiguration();
        
        // Act
        comments.setComment("path", "line1");
        comments.setComment("path", "line2");

        // Assert
        assertEquals(2, comments.getAllComments().get("path").size());
    }

    @Test
    public void testGetAllComments() {
        CommentConfiguration comments = new CommentConfiguration();
        comments.setComment("path", "line1");
        comments.setComment("path", "line2");

        Map<String, List<String>> map = new HashMap<>();
        map.put("path", Arrays.asList("line1", "line2"));
        
        assertEquals(map, comments.getAllComments());
    }

    @Test
    public void testSetLines() {
        // Arrange
        CommentConfiguration comments = new CommentConfiguration();
        List<String> lines = new ArrayList<>();

        // Act
        comments.setLines(lines);
        comments.setComment("path", "line1");

        // Assert
        assertEquals(2, comments.getAllComments().get("path").size());
    }

    @Test
    public void testGetSizeOfLines() {
        CommentConfiguration comments = new CommentConfiguration();
        List<String> lines = new ArrayList<>();
        
        comments.setLines(lines);

        assertEquals(0, comments.getSizeOfLines());
    }
}

class CommentConfiguration {
    private Map<String, List<String>> allComments;

    public CommentConfiguration() {
        this.allComments = new HashMap<>();
    }

    public void setComment(String key, String value) {
        if (!allComments.containsKey(key)) {
            allComments.put(key, new ArrayList<>());
        }
        allComments.get(key).add(value);
    }

    public Map<String, List<String>> getAllComments() {
        return allComments;
    }

    public void setLines(List<String> lines) {
        this.allComments = lines;
    }

    public int getSizeOfLines() {
        return allComments.size();
    }
}
