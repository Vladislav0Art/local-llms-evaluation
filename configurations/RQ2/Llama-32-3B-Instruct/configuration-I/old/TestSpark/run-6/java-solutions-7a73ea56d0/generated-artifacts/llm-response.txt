    }

public class OnlineStockSpan {

    private int nextPrice;
    private int currentSpan;

    public OnlineStockSpan() { /* implementation */ }

    public int next(int price) {
        if (price == null) {
            throw new NullPointerException("Null input is not allowed");
        }
        return price / currentSpan + 1;
    }

    public int[] calculateSpans(int[] prices) {
        int[] result = new int[prices.length];
        for (int i = 0; i < prices.length; i++) {
            result[i] = next(prices[i]);
        }
        return result;
    }
}

    }

    @Test
    public void next_DividesByOne_ReturnsCorrectValue() {

    public void next_IsZeroForNoPriceProvided_ThrowsNullPointerException() {
        assertThrows(NullPointerException.class, () -> onlineStockSpan.next(null));
    }

    @Test
    public void next_ReturnsCorrectValueWhenNull() {
        assertEquals(0, (int)onlineStockSpan.next(null));
    }

    @Test
    public void next_ReturnsCorrectValueForPositiveNumber() {
        assertEquals(1, (int)onlineStockSpan.next(100));
    }

    @Test
    public void next_ReturnsCorrectValueForNegativeNumber() {
        assertEquals(-1, (int)onlineStockSpan.next(-100));
    }

    @Test
    public void next_ReturnsCorrectValueWhenPriceIsZero() {
        assertEquals(0, (int)onlineStockSpan.next(0));
    }
}

public class OnlineStockSpan {

    private int price;

    public void update(int p) {
        this.price = p;
    }

    public int next() {
        if(price == null) return 0;
        return price == -1 ? 0 : price;
    }
}

public class OnlineStockSpan {

    private int count = 1;

    public int next(int price) {
        if (price == null) return 0;
        count += 1;
        return count - 1;
    }

    public void reset() {
        count = 1;
    }
}

public class GeneratedTest {

    @Test
    public void testNextPriceIncrease() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        int expectedCount = 2;

        for (int price : new int[]{10, 7, 5, 8, 11}) {
            onlineStockSpan.next(price);
            assertEquals(expectedCount, onlineStockSpan.next());
            expectedCount += 1;
        }

        reset(onlineStockSpan);
    }

    @Test
    public void testNextPriceDecrease() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        int expectedCount = -2;

        for (int price : new int[]{-100, -7, -5, -8, -11}) {
            onlineStockSpan.next(price);
            assertEquals(expectedCount, onlineStockSpan.next());
            expectedCount += 1;
        }

        reset(onlineStockSpan);
    }

    @Test
    public void testNextPriceNull() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        int expectedCount = 0;

        for (int price : new int[]{null, 10, 7, 5, 8, 11}) {
            onlineStockSpan.next(price);
            assertEquals(expectedCount, onlineStockSpan.next());
            if (price != null) {
                expectedCount += 1;
            }
        }

        reset(onlineStockSpan);
    }

    @Test
    public void testReset() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        int expectedCount = 1;

        for (int price : new int[]{10, 7, 5, 8, 11}) {
            onlineStockSpan.next(price);
        }

        assertEquals(expectedCount, onlineStockSpan.next());
    }
}
