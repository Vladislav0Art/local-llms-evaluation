import org.jsoup.nodes.TextNode;
import org.junit.Test;
import static org.junit.Assert.*;
import java.io.ByteArrayOutputStream;
import java.io.IOException;

public class TextNodeTest {

    @Test
    public void newTextNode_textIsBlankTest() {
        TextNode textNode = new TextNode("   ");
        assertTrue(textNode.isBlank());
    }

    @Test
    public void newTextNode_textIsNotBlankTest() {
        TextNode textNode = new TextNode("Hello World");
        assertFalse(textNode.isBlank());
    }

    @Test
    public void createFromEncoded_emptyStringTest() {
        String encodedText = "";
        TextNode textNode = TextNode.createFromEncoded(encodedText);
        assertNotNull(textNode);
    }

    @Test
    public void createFromEncoded_nonEmptyStringTest() {
        String encodedText = "&lt;&gt;";
        TextNode textNode = TextNode.createFromEncoded(encodedText);
        assertEquals("&lt;", textNode.text());
    }

    @Test
    public void getWholeText_emptyTextNodeTest() {
        TextNode textNode = new TextNode("");
        assertEquals("", textNode.getWholeText());
    }

    @Test
    public void getWholeText_nonEmptyTextNodeTest() {
        String text = "Hello World";
        TextNode textNode = new TextNode(text);
        assertEquals(text, textNode.getWholeText());
    }

    @Test
    public void clone_originalNodeTest() throws IOException {
        ByteArrayOutputStream outContent = new ByteArrayOutputStream();
        Appendable accum = outContent;
        TextNode originalNode = new TextNode("Hello World");
        Document.OutputSettings settings = null;
        originalNode.outerHtmlHead(accum, 0, settings);
        assertTrue(outContent.toString().equals("<p>Hello World</p>"));
        TextNode clonedNode = originalNode.clone();
        assertSame(originalNode, clonedNode);
    }

    @Test
    public void clone_cloneCreatedTextuallyTest() throws IOException {
        ByteArrayOutputStream outContent = new ByteArrayOutputStream();
        Appendable accum = outContent;
        Document.OutputSettings settings = null;
        TextNode originalNode = new TextNode("Hello World");
        TextNode clonedNode = (TextNode) originalNode.clone();
        assertTrue(outContent.toString().equals("<p>Hello World</p>"));
    }

    @Test
    public void splitText_atStartTest() throws IOException {
        ByteArrayOutputStream outContent = new ByteArrayOutputStream();
        Appendable accum = outContent;
        Document.OutputSettings settings = null;
        TextNode textNode = new TextNode("Hello");
        TextNode node1 = (TextNode) textNode.splitText(0);
        TextNode node2 = (TextNode) textNode.splitText(textNode.text().length());
        assertTrue(outContent.toString().equals("<p>Hello</p><p></p>"));
    }

    @Test
    public void splitText_atMiddleTest() throws IOException {
        ByteArrayOutputStream outContent = new ByteArrayOutputStream();
        Appendable accum = outContent;
        Document.OutputSettings settings = null;
        TextNode textNode = new TextNode("Hello World");
        TextNode node1 = (TextNode) textNode.splitText(0);
        TextNode node2 = (TextNode) textNode.splitText(textNode.text().length());
        assertTrue(outContent.toString().equals("<p>Hello</p><p>World</p>"));
    }

    @Test
    public void stripLeadingWhitespace_testStringWithMultipleSpacesTest() {
        String testString = "   ";
        assertEquals(" ", TextNode.stripLeadingWhitespace(testString));
    }

    @Test
    public void stripLeadingWhitespace_testStringWithoutLeadingSpacesTest() {
        String testString = "Hello World";
        assertEquals("Hello World", TextNode.stripLeadingWhitespace(testString));
    }
}
