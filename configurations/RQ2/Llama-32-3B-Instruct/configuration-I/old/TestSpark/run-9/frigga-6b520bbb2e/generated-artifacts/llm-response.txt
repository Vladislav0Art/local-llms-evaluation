import org.junit.Test;
import static org.hamcrest.MatcherAssert.assertThat;
import static org.hamcrest.Matchers.is;

public class AppVersionTest {

    @Test
    public void testParse() {
        AppVersion appVersion = new AppVersion("1.0-2.0");
        assertThat(appVersion.getBuildJobName(), is("build"));
        assertThat(appVersion.getBuildNumber(), is("2"));
    }

    public static class AppVersion {
        private String version;

        public AppVersion(String version) {
            this.version = version;
        }

        public String getBuildJobName() {
            return version.split("-")[0];
        }

        public int getBuildNumber() {
            return Integer.parseInt(version.split("-")[1]);
        }
    }
}

public class AppVersion {
    private String version;

    public AppVersion(String version) {
        this.version = version;
    }

    public String getBuildJobName() {
        return version.split("-")[0];
    }

    public int getBuildNumber() {
        try {
            return Integer.parseInt(version.split("-")[1]);
        } catch (NumberFormatException e) {
            return 0;
        }
    }
}

public class AppVersionTest {

    @Test
    public void testParseValidFormat() {
        AppVersion appVersion = new AppVersion("build-2");
        assertEquals("build", appVersion.getBuildJobName());
        assertEquals(2, appVersion.getBuildNumber());
    }

    @Test
    public void testParseInvalidFormat() {
        AppVersion appVersion = new AppVersion("invalid-format");
        assertNull(appVersion.getBuildJobName());
        assertEquals(0, appVersion.getBuildNumber());
    }
}
