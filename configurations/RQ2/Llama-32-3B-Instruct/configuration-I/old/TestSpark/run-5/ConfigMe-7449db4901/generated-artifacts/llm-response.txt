import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import java.util.ArrayList;
import java.util.List;
import java.util.Map;

@RunWith(MockitoJUnitRunner.class)
public class CommentsConfigurationTest {

    @Mock
    private Map<String, List<String>> commentsMap;

    @InjectMocks
    public CommentsConfiguration commentsConfiguration;

    @Test
    public void constructor_test() {
        Mockito.verify(commentsConfiguration).commentsMap = commentsMap;
    }

    @Test
    public void constructor_with_comments_map_test() {
        commentsConfiguration = new CommentsConfiguration(commentsMap);
        Mockito.verify(commentsMap).equals(commentsMap);
    }

    @Test
    public void setComment_no_comment_lines_test() {
        commentsConfiguration.setComment("path", "");
        List<String> lines = new ArrayList<>();
        Mockito.verify(lines).add("");
    }

    @Test
    public void setComment_single_line_with_comment_marker_test() {
        String commentLine = "# Comment";
        commentsConfiguration.setComment("path", commentLine);
        List<String> lines = new ArrayList<>();
        lines.add(commentLine);
        Mockito.verify(lines).add(commentLine);
    }

    @Test
    public void setComment_multiple_lines_with_comments_test() {
        String[] commentLines = {"# Comment1", "# Comment2"};
        commentsConfiguration.setComment("path", commentLines[0], commentLines[1]);
        List<String> lines = new ArrayList<>();
        lines.add(commentLines[0]);
        lines.add(commentLines[1]);
        Mockito.verify(lines).add(commentLines[0]).add(commentLines[1]);
    }

    @Test
    public void getAllComments_empty_comments_map_test() {
        commentsConfiguration.commentsMap = null;
        Map<String, @UnmodifiableView List<String>> allComments = commentsConfiguration.getAllComments();
        Mockito.verify(allComments).isEmpty();
    }

    @Test
    public void getAllComments_with_comments_map_test() {
        List<String> commentLines = new ArrayList<>();
        commentLines.add("# Comment1");
        commentLines.add("# Comment2");
        commentsConfiguration.commentsMap.put("path", commentLines);
        Map<String, @UnmodifiableView List<String>> allComments = commentsConfiguration.getAllComments();
        Mockito.verify(allComments).containsKey("path").verify(allComments.get("path"), Mockito.equals(commentLines)).isEmpty();
    }

    @Test
    public void getAllComments_with_unmodifiable_comments_map_test() {
        commentsConfiguration.commentsMap = new HashMap<>(new HashMap<>());
        commentsConfiguration.commentsMap.put("path", new ArrayList<>());
        Map<String, @UnmodifiableView List<String>> allComments = commentsConfiguration.getAllComments();
        Mockito.verify(allComments).containsKey("path").verify(allComments.get("path"), Mockito.equals(new ArrayList<>())).isEmpty();
    }

}
