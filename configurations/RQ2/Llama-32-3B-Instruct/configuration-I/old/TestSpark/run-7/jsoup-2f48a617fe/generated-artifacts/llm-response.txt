import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Main {

    public static String getOuterHtmlHead(String html, int version) {
        Document doc = Jsoup.parse(html);
        Element headElement = doc.select("head").first();
        if (headElement == null || headElement.children().isEmpty()) {
            return "";
        }
        StringBuilder outerHtml = new StringBuilder();
        for (Element child : headElement.children()) {
            outerHtml.append(child.outerHtml()).append("\n");
        }
        return version > 0 ? "<html " + doc.html() + ">" + "\n" + outerHtml.toString().trim() + "\n</html>" : outerHtml.toString().trim();
    }

    public static class Comment {
        private String value;

        public Comment(String value) {
            this.value = value;
        }

        @Override
        public boolean equals(Object o) {
            return getClass() == o.getClass();
        }
    }

    public static void cloneComment(Comment comment, StringBuilder value) {
        comment.value = value.toString().trim();
    }

    public static class GeneratedTest extends Comment {
        private String test;

        public GeneratedTest(String test) {
            super(test);
        }

        @Override
        public boolean equals(Object o) {
            return getClass() == o.getClass();
        }

        @Override
        public void clone( StringBuilder value ) {
            super.clone(value);
        }
    }
}
