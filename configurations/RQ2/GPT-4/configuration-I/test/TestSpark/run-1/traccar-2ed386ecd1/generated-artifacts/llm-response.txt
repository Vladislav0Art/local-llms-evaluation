import org.jboss.netty.buffer.ChannelBuffers;
import org.jboss.netty.channel.Channel;
import org.jboss.netty.channel.ChannelHandlerContext;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.runners.MockitoJUnitRunner;
import org.traccar.model.Position;

import java.net.InetSocketAddress;
import java.net.SocketAddress;
import java.nio.ByteOrder;
import java.nio.charset.StandardCharsets;
import java.util.TimeZone;

import static org.mockito.Mockito.verify;

@RunWith(MockitoJUnitRunner.class)
public class Gt06ProtocolDecoderTest {
    
    @Mock
    private Protocol protocol;

    @Mock
    private Channel channel;
    
    @Mock
    private ChannelHandlerContext context;

    private Position position = new Position();

    @Test
    public void decodeGpsHasLengthTrueTimeZoneNotNullTest() {
        ByteBuf buf = ChannelBuffers.copiedBuffer(ByteOrder.LITTLE_ENDIAN, "buffer data", StandardCharsets.UTF_8);
        Gt06ProtocolDecoder.decodeGps(position, buf, true, TimeZone.getDefault());
        verify(buf).readableBytes();
    }

    @Test
    public void decodeGpsHasLengthFalseTimeZoneNotNullTest() {
        ByteBuf buf = ChannelBuffers.copiedBuffer(ByteOrder.LITTLE_ENDIAN, "buffer data", StandardCharsets.UTF_8);
        Gt06ProtocolDecoder.decodeGps(position, buf, false, TimeZone.getDefault());
        verify(buf).readableBytes();
    }

    @Test
    public void decodeGpsWithParametersHasLengthTrueHasSatellitesTrueHasSpeedTrueTimeZoneNotNullTest() {
        ByteBuf buf = ChannelBuffers.copiedBuffer(ByteOrder.LITTLE_ENDIAN, "buffer data", StandardCharsets.UTF_8);
        Gt06ProtocolDecoder.decodeGps(position, buf, true, true, true, TimeZone.getDefault());
        verify(buf).readableBytes();
    }

    @Test
    public void decodeGpsWithParametersHasLengthFalseHasSatellitesFalseHasSpeedFalseTimeZoneNotNullTest() {
        ByteBuf buf = ChannelBuffers.copiedBuffer(ByteOrder.LITTLE_ENDIAN, "buffer data", StandardCharsets.UTF_8);
        Gt06ProtocolDecoder.decodeGps(position, buf, false, false, false, TimeZone.getDefault());
        verify(buf).readableBytes();
    }

    @Test
    public void decodeTest() throws Exception {
        Gt06ProtocolDecoder decoder = new Gt06ProtocolDecoder(protocol);
        InetSocketAddress remoteAddress = new InetSocketAddress("localhost", 0);
        ByteBuf msg = ChannelBuffers.copiedBuffer(ByteOrder.LITTLE_ENDIAN, "buffer data", StandardCharsets.UTF_8);
        decoder.decode(context, channel, remoteAddress, msg);
        verify(context).channel();
    }
}
