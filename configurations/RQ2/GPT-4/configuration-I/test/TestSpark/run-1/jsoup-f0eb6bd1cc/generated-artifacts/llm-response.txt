import org.junit.Test;
import org.junit.Assert;
import org.jsoup.Connection;
import org.mockito.Mockito;

import java.net.URL;
import java.net.MalformedURLException;
import java.io.UnsupportedEncodingException;

public class UrlBuilderTest {

    @Test
    public void urlBuilderConstructorTest() throws MalformedURLException {
        URL inputUrl = new URL("http://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        Assert.assertNotNull(urlBuilder);
    }

    @Test
    public void buildMethodTest() throws MalformedURLException {
        URL inputUrl = new URL("http://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        URL outputUrl = urlBuilder.build();
        Assert.assertEquals(inputUrl, outputUrl);
    }

    @Test
    public void appendKeyValTest() throws MalformedURLException, UnsupportedEncodingException {
        URL inputUrl = new URL("http://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        Connection.KeyVal kv = Mockito.mock(Connection.KeyVal.class);
        Mockito.when(kv.toString()).thenReturn("key=value");
        urlBuilder.appendKeyVal(kv);
        URL expectedUrl = new URL("http://example.com?key=value");
        Assert.assertEquals(expectedUrl, urlBuilder.build());
    }

    @Test(expected = IllegalArgumentException.class)
    public void appendKeyValNullTest() throws MalformedURLException, UnsupportedEncodingException {
        URL inputUrl = new URL("http://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        urlBuilder.appendKeyVal(null);
    }

    @Test(expected = MalformedURLException.class)
    public void urlBuilderInvalidUrlTest() throws MalformedURLException {
        new UrlBuilder(new URL("invalid_url"));
    }
}
