import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import java.io.*;
import java.nio.charset.Charset;
import java.util.HashSet;

import static org.junit.Assert.*;
import static org.mockito.Mockito.*;

@RunWith(MockitoJUnitRunner.class)
public class WordWrapTest {

    @Mock
    Reader reader;

    @Mock
    Writer writer;

    @Test
    public void fromReaderTest() {
        assertNotNull(WordWrap.from(reader));
    }

    @Test
    public void fromClasspathUtf8Test() {
        assertNotNull(WordWrap.fromClasspathUtf8("testResource.txt"));
    }

    @Test
    public void fromClasspathTest() {
        assertNotNull(WordWrap.fromClasspath("testResource.txt", Charset.defaultCharset()));
    }

    @Test
    public void fromTextTest() {
        assertNotNull(WordWrap.from("This is a test text"));
    }

    @Test
    public void fromUtf8Test() throws FileNotFoundException {
        assertNotNull(WordWrap.fromUtf8(new FileInputStream(new File("testFile.txt"))));
    }

    @Test
    public void fromInputStreamTest() throws FileNotFoundException {
        assertNotNull(WordWrap.from(new FileInputStream(new File("testFile.txt")), Charset.defaultCharset()));
    }

    @Test
    public void fromFileTest() {
        assertNotNull(WordWrap.from(new File("testFile.txt"), Charset.defaultCharset()));
    }

    @Test
    public void closeReaderTest() {
        WordWrap.close(reader);
        verify(reader, times(1)).close();
    }

    @Test
    public void rightTrimTest() {
        CharSequence text = " This is a test text ";
        assertEquals(" This is a test text", WordWrap.rightTrim(text).toString());
    }

    @Test
    public void isWhitespaceTest() {
        assertTrue(WordWrap.isWhitespace(" "));
        assertFalse(WordWrap.isWhitespace("A"));
    }

    @Test
    public void wordWrapStreamToStreamTest() throws IOException {
        WordWrap.wordWrap(reader, writer, "\n", 80, CharSequence::length, new HashSet<>(), false, false);
        verify(reader, times(1)).read();
        verify(writer, times(1)).write(anyString());
    }
}
