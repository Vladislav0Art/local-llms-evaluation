        @Override
        public KeyPair getKeypair() {
            return sourceAccount;
        }

        @Override
        public Long getIncrementedSequenceNumber() {
            return Long.valueOf(SEQUENCE_NUMBER);
        }

        @Override
        public BaseFee getBaseFee() {
            return new BaseFee(FEE);
        }

        @Override
        public void incrementSequenceNumber() { /*do nothing*/ }
    };

    @Test
    public void getSourceAccountTest() {
        Transaction transaction = new Transaction.Builder(SOURCE_ACCOUNT, NETWORK).build();
        assertEquals(transaction.getSourceAccount(), sourceAccount.getAccountId());
    }

    @Test
    public void getSequenceNumberTest() {
        Transaction transaction = new Transaction.Builder(SOURCE_ACCOUNT, NETWORK).build();
        assertEquals(transaction.getSequenceNumber(), SEQUENCE_NUMBER);
    }

    public void getMemoTest() {
        Transaction transaction = new Transaction.Builder(SOURCE_ACCOUNT, NETWORK).addMemo(MEMO).build();
        assertEquals(transaction.getMemo(), MEMO);
    }

    public void getFeeTest() {

import static org.junit.Assert.*;
import org.junit.Test;
import org.mockito.Mockito;
import org.stellar.sdk.*;

public class TransactionTest {

    // Example values for constructing Transaction
    private final static KeyPair sourceAccount = KeyPair.random();
    private final static Network NETWORK = Network.TESTNET;
    private final static long FEE = 100;
    private final static long SEQUENCE_NUMBER = 0;
    private final static Operation[] OPERATIONS = new Operation[0];
    private final static Memo MEMO = Memo.none();
    private final static long TIMEOUT = 10;

    @Test
    public void getSourceAccountTest() {
        Transaction transaction = createDefaultTransaction(sourceAccount, NETWORK);
        assertEquals(transaction.getSourceAccount(), sourceAccount.getAccountId());
    }

    @Test
    public void getSequenceNumberTest() {
        Transaction transaction = createDefaultTransaction(sourceAccount, NETWORK);
        assertEquals(transaction.getSequenceNumber(), SEQUENCE_NUMBER);
    }

    @Test
    public void getMemoTest() {
        Transaction transaction = createDefaultTransaction(sourceAccount, NETWORK);
        assertTrue(transaction.getMemo() instanceof MemoNone);
    }

    @Test
    public void getOperationsTest() {
        Transaction transaction = createDefaultTransaction(sourceAccount, NETWORK);
        assertEquals(0, transaction.getOperations().length);
    }

    @Test
    public void hashCodeTest() {
        Transaction transaction1 = createDefaultTransaction(sourceAccount, NETWORK);
        Transaction transaction2 = createDefaultTransaction(sourceAccount, NETWORK);
        assertEquals(transaction1.hashCode(), transaction2.hashCode());
    }

    @Test
    public void equalsTest() {
        Transaction transaction1 = createDefaultTransaction(sourceAccount, NETWORK);
        Transaction transaction2 = createDefaultTransaction(sourceAccount, NETWORK);
        assertTrue(transaction1.equals(transaction2));
    }

    private Transaction createDefaultTransaction(KeyPair account, Network network) {
        Account.Builder accountBuilder = new Account.Builder(account, SEQUENCE_NUMBER);
        Transaction.Builder txnBuilder = new Transaction.Builder(accountBuilder.build(), network);
        txnBuilder.addMemo(MEMO);
        return txnBuilder.build();
    }
}
