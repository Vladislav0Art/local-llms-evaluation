import org.jsoup.nodes.Document;
import org.junit.Assert;
import org.junit.Test;
import java.io.IOException;

import static org.mockito.Mockito.mock;

public class TextNodeTest {

    @Test
    public void TextNodeConstructTest() {
        TextNode node = new TextNode("hello world");
        Assert.assertEquals("hello world", node.text());
    }

    @Test
    public void NodeNameTest() {
        TextNode node = new TextNode("hello world");
        Assert.assertEquals("#text", node.nodeName());
    }

    @Test
    public void TextTest() {
        TextNode node = new TextNode("hello world");
        node.text("new text");
        Assert.assertEquals("new text", node.text());
    }

    @Test
    public void GetWholeTextTest() {
        TextNode node = new TextNode("hello world");
        Assert.assertEquals("hello world", node.getWholeText());
    }


    @Test
    public void IsBlankTest() {
        TextNode node = new TextNode("");
        Assert.assertTrue(node.isBlank());
    }

    @Test
    public void SplitTextTest() {
        TextNode node = new TextNode("hello world");
        TextNode newNode = node.splitText(5);
        Assert.assertEquals("hello", node.text());
        Assert.assertEquals(" world", newNode.text());
    }

    @Test(expected = IOException.class)
    public void OuterHtmlheadThrowsExceptionTest() throws IOException {
        Appendable appendable = mock(Appendable.class);
        TextNode node = new TextNode("");
        node.outerHtmlHead(appendable, 1, new Document.OutputSettings());
    }

    @Test(expected = IOException.class)
    public void OuterHtmlTailThrowsExceptionTest() throws IOException {
        Appendable appendable = mock(Appendable.class);
        TextNode node = new TextNode("");
        node.outerHtmlTail(appendable, 1, new Document.OutputSettings());
    }

    @Test
    public void CreateFromEncodedTest() {
        TextNode node = TextNode.createFromEncoded("hello world", false);
        Assert.assertEquals("hello world", node.text());
    }

    @Test
    public void NormaliseWhitespaceTest() {
        String result = TextNode.normaliseWhitespace("hello  world");
        Assert.assertEquals("hello world", result);
    }

    @Test
    public void StripLeadingWhitespaceTest() {
        String result = TextNode.stripLeadingWhitespace(" hello world");
        Assert.assertEquals("hello world", result);
    }

    @Test
    public void LastCharIsWhitespaceTest() {
        boolean result = TextNode.lastCharIsWhitespace(new StringBuilder("hello world "));
        Assert.assertTrue(result);
    }
}

    }

    @Test
    public void OuterHtmlHeadTest() throws IOException {
        Appendable appendable = new StringBuilder();
        TextNode node = new TextNode("test node");
        node.outerHtmlHead(appendable, 0, new OutputSettings());
        Assert.assertEquals("test node", appendable.toString());
    }

    @Test
    public void OuterHtmlTailTest() {
        Appendable appendable = new StringBuilder();
        TextNode node = new TextNode("test node");
        node.outerHtmlTail(appendable, 0, new OutputSettings());
