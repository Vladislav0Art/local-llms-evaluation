import org.jsoup.Connection;
import org.jsoup.helper.UrlBuilder;
import org.junit.Test;

import java.io.UnsupportedEncodingException;
import java.net.URL;

import static org.mockito.Mockito.mock;
import static org.mockito.Mockito.when;

public class UrlBuilderTest {

    @Test
    public void UrlBuilderValidUrlTest() throws MalformedURLException {
        URL url = new URL("https://www.google.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);
    }

    @Test(expected = NullPointerException.class)
    public void UrlBuilderNullUrlTest() throws MalformedURLException {
        URL url = null;
        UrlBuilder urlBuilder = new UrlBuilder(url);
    }

    @Test
    public void buildValidUrlTest() throws MalformedURLException {
        URL url = new URL("https://www.google.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        urlBuilder.build();
    }

    @Test(expected = MalformedURLException.class)
    public void buildInvalidUrlTest() throws MalformedURLException {
        URL url = new URL("invalid-url");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        urlBuilder.build();
    }

    @Test
    public void appendKeyValValidKeyValTest() throws UnsupportedEncodingException {
        URL url = new URL("https://www.google.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);

        Connection.KeyVal keyVal = mock(Connection.KeyVal.class);
        when(keyVal.key()).thenReturn("key");
        when(keyVal.value()).thenReturn("value");

        urlBuilder.appendKeyVal(keyVal);
    }

    @Test(expected = UnsupportedEncodingException.class)
    public void appendKeyValUnsupportedEncodingKeyValTest() throws UnsupportedEncodingException {
        URL url = new URL("https://www.google.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);

        Connection.KeyVal keyVal = mock(Connection.KeyVal.class);
        when(keyVal.key()).thenReturn("key\u0019");
        when(keyVal.value()).thenReturn("value\u0019");

        urlBuilder.appendKeyVal(keyVal);
    }
}
