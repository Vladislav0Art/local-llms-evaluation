import org.jsoup.nodes.Element;
import org.junit.Test;
import org.jsoup.parser.Tag;
import java.util.Collections;

import static org.junit.Assert.*;

public class ElementTests {
    @Test
    public void testConstructor() {
        Element element = new Element(Tag.valueOf("p"), "http://www.test.com");
        assertNotNull(element);
    }

    @Test
    public void testBaseUri() {
        Element element = new Element(Tag.valueOf("p"), "http://www.test.com");
        assertEquals("http://www.test.com", element.baseUri());
    }

    @Test
    public void tagNameTest() {
        Element element = new Element(Tag.valueOf("p"), "http://www.test.com");
        assertEquals("p", element.tagName());
    }

    @Test
    public void normalNameTest() {
        Element element = new Element(Tag.valueOf("P"), "http://www.test.com");
        assertEquals("p", element.normalName());
    }

    @Test
    public void attrStringAttributeKeyBooleanAttributeValueTest() {
        Element element = new Element(Tag.valueOf("p"), "http://www.test.com");
        Element result = element.attr("bool", true);
        assertEquals("true", result.attr("bool"));
        assertEquals(element, result);
    }

    @Test
    public void attrStringAttributeKeyStringAttributeValueTest() {
        Element element = new Element(Tag.valueOf("p"), "http://www.test.com");
        Element result = element.attr("attr", "value");
        assertEquals("value", result.attr("attr"));
        assertEquals(element, result);
    }

    @Test
    public void attrStringAttributeKeyAbsentAttributeValueTest() {
        Element element = new Element(Tag.valueOf("p"), "http://www.test.com");
        String result = element.attr("absent");
        assertEquals("", result);
    }

    @Test
    public void appendTest() {
        Element element = new Element(Tag.valueOf("p"), "http://www.test.com");
        Element result = element.append("<span>Test</span>");
        assertEquals("<p><span>Test</span></p>", result.toString());
    }

    @Test
    public void prependTest() {
        Element element = new Element(Tag.valueOf("p"), "http://www.test.com");
        Element result = element.prepend("<span>Test</span>");
        assertEquals("<p><span>Test</span></p>", result.toString());
    }

    @Test
    public void insertChildrenTest() {
        Element parent = new Element(Tag.valueOf("div"), "http://www.test.com");
        Element child1 = new Element(Tag.valueOf("p"), "http://www.test.com");
        Element child2 = new Element(Tag.valueOf("span"), "http://www.test.com");
        parent.insertChildren(0, Collections.singletonList(child1));
        parent.insertChildren(0, Collections.singletonList(child2));
        assertEquals("<div><span></span><p></p></div>", parent.toString());
    }

    @Test
    public void htmlTest() {
        Element element = new Element(Tag.valueOf("div"), "http://www.test.com");
        element.html("<p>Test</p>");
        assertEquals("<p>Test</p>", element.html());
    }
}
