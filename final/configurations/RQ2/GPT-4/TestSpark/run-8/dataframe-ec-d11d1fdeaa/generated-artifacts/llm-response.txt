import io.github.vmzakharov.ecdataframe.dataframe.DataFrame;
import io.github.vmzakharov.ecdataframe.dsl.value.Value;
import io.github.vmzakharov.ecdataframe.dsl.value.ValueType;
import org.eclipse.collections.api.factory.Lists;
import org.eclipse.collections.api.list.ListIterable;
import org.junit.Test;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.Arrays;

import static org.junit.Assert.*;

public class DataFrameTest {
    @Test
    public void addStringColumnTest() {
        DataFrame dataFrame = new DataFrame("testing");
        dataFrame.addStringColumn("Test_Column");
        assertEquals("Test_Column", dataFrame.getColumnNamed("Test_Column").getName());
    }
    @Test
    public void addLongColumnTest() {
        DataFrame dataFrame = new DataFrame("testing");
        dataFrame.addLongColumn("Test_Column", Arrays.asList(1L, 2L, 3L));
        assertEquals("Test_Column", dataFrame.getColumnNamed("Test_Column").getName());
    }

    @Test
    public void addDateColumnTest() {
        DataFrame dataFrame = new DataFrame("testing");
        dataFrame.addDateColumn("Test_Column", Arrays.asList(LocalDate.now(), LocalDate.now()));
        assertEquals("Test_Column", dataFrame.getColumnNamed("Test_Column").getName());
    }

    @Test
    public void rowCountTest() {
        DataFrame dataFrame = new DataFrame("testing");
        dataFrame.addRow(Arrays.asList(new Value[] {Value.of(1), Value.of("test")}));
        assertEquals(1, dataFrame.rowCount());
    }

    @Test
    public void columnCountTest() {
        DataFrame dataFrame = new DataFrame("testing");
        dataFrame.addStringColumn("Column1");
        dataFrame.addLongColumn("Column2");
        assertEquals(2, dataFrame.columnCount());
    }

    @Test
    public void getNameTest() {
        DataFrame dataFrame = new DataFrame("testing");
        assertEquals("testing", dataFrame.getName());
    }

    @Test
    public void addColumnTest() {
        DataFrame dataFrame = new DataFrame("testing");
        dataFrame.addColumn("TestColumn", ValueType.STRING);
        assertEquals("TestColumn", dataFrame.getColumnNamed("TestColumn").getName());
    }

    @Test
    public void isFlaggedTest() {
        DataFrame dataFrame = new DataFrame("testing");
        dataFrame.addRow(Arrays.asList(new Value[] {Value.of(1), Value.of("test")}));
        dataFrame.setFlag(0);
        assertTrue(dataFrame.isFlagged(0));
    }

    @Test
    public void isFlaggedTest() {
        DataFrame dataFrame = new DataFrame("testing");
        assertFalse(dataFrame.isFlagged(0));
    }

    @Test
    public void addRowTest() {
        DataFrame dataFrame = new DataFrame("testing");
        dataFrame.addRow(new Object[]{"1", "test"});
        assertEquals(1, dataFrame.rowCount());
    }

    @Test
    public void getStringTest() {
        DataFrame dataFrame = new DataFrame("testing");
        dataFrame.addRow(new Object[]{"1", "test"});
        assertEquals("test", dataFrame.getString("Column2", 0));
    }
}
