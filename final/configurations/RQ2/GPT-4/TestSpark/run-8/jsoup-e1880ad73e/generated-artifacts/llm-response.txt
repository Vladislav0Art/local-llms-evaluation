import org.junit.Assert;
import org.jsoup.Connection;
import java.net.URL;
import java.net.MalformedURLException;
import java.io.UnsupportedEncodingException;

public class UrlBuilderTest {

    @Test
    public void UrlBuilderConstructor_WithValidURLTest() throws MalformedURLException {
        URL inputUrl = new URL("http://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        Assert.assertNotNull(urlBuilder);
    }

    @Test(expected = NullPointerException.class)
    public void UrlBuilderConstructor_WithNullURLTest() {
        UrlBuilder urlBuilder = new UrlBuilder(null);
    }

    @Test
    public void build_WithValidURLTest() throws MalformedURLException {
        URL inputUrl = new URL("http://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        URL outputUrl = urlBuilder.build();
        Assert.assertNotNull(outputUrl);
    }

    @Test
    public void appendKeyVal_WithValidKeyValTest() throws MalformedURLException, UnsupportedEncodingException {
        URL inputUrl = new URL("http://example.com");

        URL outputUrl = urlBuilder.build();
        Assert.assertTrue(outputUrl.toString().contains("key=value"));

    public void appendKeyVal_WithInvalidKeyValTest() throws MalformedURLException, UnsupportedEncodingException {

        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);

        urlBuilder.appendKeyVal(kv);
    }

}
