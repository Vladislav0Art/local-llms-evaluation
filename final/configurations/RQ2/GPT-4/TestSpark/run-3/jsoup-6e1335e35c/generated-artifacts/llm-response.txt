import org.jsoup.helper.StringUtil;
import org.jsoup.nodes.Attributes;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.Node;
import org.jsoup.parser.Tag;
import org.jsoup.select.Elements;
import org.jsoup.select.Evaluator;
import org.junit.Assert;
import org.junit.Test;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Set;
import static org.junit.Assert.*;
import static org.mockito.Mockito.*;

public class ElementTest {

    @Test
    public void nodeNameTest() {
        Element element = new Element("div");
        assertEquals("div", element.nodeName());
    }

    @Test
    public void tagTest() {
        Element element = new Element("a");
        assertTrue(element.tag() instanceof Tag);
    }

    @Test
    public void isBlockTest() {
        Element element = new Element("p");
        assertTrue(element.isBlock());
    }

    @Test
    public void idTest() {
        Element element = new Element("div");
        element.id("testId");
        assertEquals("testId", element.id());
    }

    @Test
    public void parentTest() {
        Element parent = new Element("div");
        Element child = new Element("span");
        parent.appendChild(child);
        assertEquals(parent, child.parent());
    }

    @Test
    public void appendElementTest() {
        Element element = new Element("div");
        Element child = element.appendElement("p");
        assertEquals("p", child.nodeName());
    }

    @Test
    public void textTest() {
        Element element = new Element("p");
        element.text("content");
        assertEquals("content", element.text());
    }

    @Test
    public void setDataTest() {
        Element element = new Element("p");
        element.attr("data-test", "test");
        assertTrue(element.dataset().containsKey("test"));
    }

    @Test
    public void selectAttrValueTest() {
        Element element = new Element("div");
        Element child = new Element("p");
        child.attr("class", "test");
        element.appendChild(child);
        Elements result = element.select("p.class");
        assertFalse(result.isEmpty());
    }

    @Test
    public void removeAttrTest() {
        Element element = new Element("p");
        element.attr("class", "test");
        assertFalse(element.attributes().isEmpty());

        element.removeAttr("class");
        assertTrue(element.attributes().isEmpty());
    }

    @Test
    public void htmlTest() {
        Element element = new Element("div");
        element.html("<p>Text</p>");
        assertEquals("<p>Text</p>", element.html());
    }

    @Test
    public void cloneTest() {
        Element element = new Element("div");
        Node cloned = element.clone();

        assertNotEquals(System.identityHashCode(element), System.identityHashCode(cloned));
    }
}
