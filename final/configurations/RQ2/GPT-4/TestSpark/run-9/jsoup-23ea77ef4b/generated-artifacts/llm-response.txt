import org.jsoup.helper.W3CDom;
import org.jsoup.nodes.Document;
import org.junit.Assert;
import org.junit.Test;
import org.w3c.dom.Node;
import org.w3c.dom.NodeList;

import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Properties;
import static org.junit.Assert.*;

public class W3CDomTest {

    @Test
    public void namespaceAwareTest() {
        W3CDom dom = new W3CDom();
        assertFalse(dom.namespaceAware());
        dom.namespaceAware(true);
        assertTrue(dom.namespaceAware());
    }

    @Test
    public void fromJsoupDocumentTest() {
        W3CDom dom = new W3CDom();
        Document document = new Document("");
        assertNotNull(dom.fromJsoup(document));
    }

    @Test
    public void fromJsoupElementTest() {
        W3CDom w3cDom = new W3CDom();
        org.jsoup.nodes.Element element = new org.jsoup.nodes.Element("");
        assertNotNull(w3cDom.fromJsoup(element));
    }

    @Test
    public void contextNodeTest() {
        W3CDom w3cDom = new W3CDom();
        Document wDoc = new Document("");
        assertNotNull(w3cDom.contextNode(wDoc));
    }

    @Test
    public void selectXpathTest() {
        W3CDom w3cDom = new W3CDom();
        Document document = new Document("");
        NodeList nodeList = w3cDom.selectXpath("", document);
        assertNotNull(nodeList);
    }

    @Test
    public void selectXpathContextNodeTest() {
        W3CDom w3cDom = new W3CDom();
        Node node = new org.jsoup.nodes.Element("").childNode(0);
        NodeList nodeList = w3cDom.selectXpath("", node);
        assertNotNull(nodeList);
    }

    @Test
    public void sourceNodesTest() {
        W3CDom w3cDom = new W3CDom();
        NodeList nodeList = w3cDom.selectXpath("", new Document(""));
        List<org.jsoup.nodes.Node> nodes = w3cDom.sourceNodes(nodeList, org.jsoup.nodes.Node.class);
        assertNotNull(nodes);
    }

    @Test
    public void asStringWithPropertiesTest() {
        W3CDom w3cDom = new W3CDom();
        Document document = new Document("");
        Map<String, String> properties = new HashMap<>();
        properties.put("indent", "yes");
        String str = W3CDom.asString(document, properties);
        assertNotNull(str);
    }

    @Test
    public void asStringTest() {
        W3CDom w3cDom = new W3CDom();
        Document document = new Document("");
        String str = w3cDom.asString(document);
        assertNotNull(str);
    }

    @Test
    public void convertDocumentTest() {
        Document in = new Document("");
        org.w3c.dom.Document out =
                javax.xml.parsers.DocumentBuilderFactory.newInstance().newDocumentBuilder().newDocument();
        new W3CDom().convert(in, out);
        assertEquals(in.charset().displayName(), out.getXmlEncoding());
    }

    @Test
    public void convertElementTest() {
        org.jsoup.nodes.Element in = new org.jsoup.nodes.Element("");
        org.w3c.dom.Document out =
                javax.xml.parsers.DocumentBuilderFactory.newInstance().newDocumentBuilder().newDocument();
        new W3CDom().convert(in, out);
        assertNotNull(out.getDocumentElement());
    }

    @Test
    public void propertiesFromMapTest() {
        Map<String, String> map = new HashMap<>();
        map.put("key","value");
        Properties properties = W3CDom.propertiesFromMap(map);
        assertNotNull(properties);
        assertEquals(properties.getProperty("key"), "value");
    }

    @Test
    public void outputHtmlTest() {
        HashMap<String, String> outputHtml = W3CDom.OutputHtml();
        assertNotNull(outputHtml);
        assertEquals(outputHtml.get("method"), "html");
    }

    @Test
    public void outputXmlTest() {
        HashMap<String, String> outputXml = W3CDom.OutputXml();
        assertNotNull(outputXml);
        assertEquals(outputXml.get("method"), "xml");
    }
}

        assertNotNull(dom.namespaceAware(true));

        assertNotNull(W3CDom.convert(document));

        org.jsoup.nodes.Document document = Jsoup.parse("<html></html>");
        org.w3c.dom.Document w3cDoc = W3CDom.convert(document);
        assertNotNull(W3CDom.asString(w3cDoc, W3CDom.OutputHtml()));
    }

    @Test
    public void propertiesFromMapTest() {
        assertNotNull(W3CDom.propertiesFromMap(W3CDom.OutputHtml()));
    }

    @Test
    public void outputHtmlTest() {
        assertNotNull(W3CDom.OutputHtml());
        assertTrue(W3CDom.OutputHtml().size() > 0);
    }

    @Test
    public void outputXmlTest() {
        assertNotNull(W3CDom.OutputXml());
        assertTrue(W3CDom.OutputXml().size() > 0);
    }

    @Test
    public void fromJsoupDocumentTest() {
        W3CDom dom = new W3CDom();
        org.jsoup.nodes.Document document = Jsoup.parse("<html></html>");
        assertNotNull(dom.fromJsoup(document));
    }

    @Test
    public void fromJsoupElementTest() {
        W3CDom dom = new W3CDom();
        org.jsoup.nodes.Document document = Jsoup.parse("<html><head><title>Test</title></head></html>");
        assertNotNull(dom.fromJsoup(document.head()));
    }

    @Test
    public void asStringTest() {
        W3CDom dom = new W3CDom();
        org.jsoup.nodes.Document document = Jsoup.parse("<html></html>");
        org.w3c.dom.Document w3cDoc = dom.fromJsoup(document);
        assertNotNull(dom.asString(w3cDoc));
    }

    @Test
    public void convertInAndOutTest() {
        W3CDom dom = new W3CDom();
        org.jsoup.nodes.Document in = Jsoup.parse("<html></html>");

        dom.convert(in, out);
        assertNotNull(out);
    }

    @Test
    public void convertElementInAndOutTest() {
        W3CDom dom = new W3CDom();
        org.jsoup.nodes.Document document = Jsoup.parse("<html><body><div>Test</div></body></html>");
        org.w3c.dom.Document out = dom.fromJsoup(document);
        dom.convert(document.body(), out);
        assertNotNull(out);
    }
}
