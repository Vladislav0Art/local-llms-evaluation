import org.mockito.Mockito;

        }
    }

    public void runInternalWithOutlineTest() throws ClassNotFoundException, IOException {

        try {
            xmlElementWrapperPlugin.runInternal(outline);
        } catch (Exception e) {
            Assert.fail("Expected exception to be thrown");
        }
