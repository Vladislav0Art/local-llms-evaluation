package org.jsoup.nodes;

import org.jsoup.nodes.Document;
import org.junit.Assert;
import org.junit.Before;
import org.junit.Test;
import java.io.IOException;

public class TextNodeTest {

    @Test
    public void createFromEncodedTest() {
        TextNode result = TextNode.createFromEncoded("&amp;");
        Assert.assertEquals("&", result.getWholeText());
    }

    @Test
    public void cloneTest() {
        TextNode original = new TextNode("Sample");
        TextNode cloned = original.clone();
        Assert.assertEquals(original.text(), cloned.text());
    }

    @Test
    public void toStringTest() {
        TextNode node = new TextNode("Sample");
        Assert.assertEquals("Sample", node.toString());
    }

    @Test
    public void nodeNameTest(){
        TextNode node = new TextNode("Sample");
        Assert.assertEquals("#text", node.nodeName());

    }

    @Test
    public void textSingleNodeTest(){
        TextNode node = new TextNode("Sample");
        Assert.assertEquals("Sample", node.text());
    }

    @Test
    public void textSetTest(){
        TextNode node = new TextNode("Sample");
        node.text("New Sample");
        Assert.assertEquals("New Sample", node.text());
    }

    @Test
    public void getWholeTextTest(){
        TextNode node = new TextNode("Sample Text Node");
        Assert.assertEquals("Sample Text Node", node.getWholeText());
    }

    @Test
    public void isBlankTrueTest(){
        TextNode node = new TextNode("");
        Assert.assertTrue(node.isBlank());
    }

    @Test
    public void isBlankFalseTest(){
        TextNode node = new TextNode("Not blank");
        Assert.assertFalse(node.isBlank());
    }

    @Test
    public void splitTextTest(){
        TextNode node = new TextNode("Sample Text Node");
        TextNode splitNode = node.splitText(6);
        Assert.assertEquals("Sample ", node.text());
        Assert.assertEquals("Text Node", splitNode.text());
    }

    @Test
    public void outerHtmlHeadTest() throws IOException {
        TextNode node = new TextNode("Sample Text Node");
        StringBuilder sb = new StringBuilder();
        node.outerHtmlHead(sb, 0, new Document("").outputSettings());
        Assert.assertEquals("Sample Text Node", sb.toString());
    }

    @Test
    public void outerHtmlTailTest() throws IOException {
        TextNode node = new TextNode("Sample Text Node");
        StringBuilder sb = new StringBuilder();
        node.outerHtmlTail(sb, 0, new Document("").outputSettings());
        Assert.assertEquals("", sb.toString());
    }

    @Test
    public void normaliseWhitespaceTest() throws IOException {
        String result = TextNode.normaliseWhitespace("No   whitespace");
        Assert.assertEquals("No whitespace", result);
    }

    @Test
    public void stripLeadingWhitespaceTest() {
        String result = TextNode.stripLeadingWhitespace("  No leading whitespace");
        Assert.assertEquals("No leading whitespace", result);
    }

    @Test
    public void lastCharIsWhitespaceTrueTest() {
        boolean result = TextNode.lastCharIsWhitespace(new StringBuilder("Trailing space "));
        Assert.assertTrue(result);
    }

    @Test
    public void lastCharIsWhitespaceFalseTest() {
        boolean result = TextNode.lastCharIsWhitespace(new StringBuilder("NoTrailingSpace"));
        Assert.assertFalse(result);
    }
}
