package com.sun.tools.xjc.addon.xew;

import com.sun.tools.xjc.outline.Outline;
import com.sun.tools.xjc.reader.Ring;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import java.io.IOException;

@RunWith(MockitoJUnitRunner.class)
public class XmlElementWrapperPluginTest {

    @Mock
    private Outline outline;
    @Mock
    private Ring ring;

    @Test
    public void runInternalValidOutlineTest() throws IOException, ClassNotFoundException {
        XmlElementWrapperPlugin xmlElementWrapperPlugin = new XmlElementWrapperPlugin();
        xmlElementWrapperPlugin.runInternal(outline);
        Mockito.verify(outline, Mockito.times(1)).getAllClasses();
    }

    @Test(expected = IOException.class)
    public void runInternalOutlineThrowsIOExceptionTest() throws IOException, ClassNotFoundException {
        Mockito.doThrow(IOException.class).when(outline).getAllClasses();
        XmlElementWrapperPlugin xmlElementWrapperPlugin = new XmlElementWrapperPlugin();
        xmlElementWrapperPlugin.runInternal(outline);
    }

    @Test(expected = ClassNotFoundException.class)
    public void runInternalOutlineThrowsClassNotFoundExceptionTest() throws IOException, ClassNotFoundException {
        Mockito.doThrow(ClassNotFoundException.class).when(outline).getAllClasses();
        XmlElementWrapperPlugin xmlElementWrapperPlugin = new XmlElementWrapperPlugin();
        xmlElementWrapperPlugin.runInternal(outline);
    }

}
