import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.TextNode;
import org.jsoup.select.Elements;
import org.junit.Assert;
import org.junit.Test;

public class ElementTest {

    @Test
    public void tagNameTest() {
        Element element = new Element("p");
        Assert.assertEquals("p", element.tagName());
    }

    @Test
    public void idTest() {
        Element element = new Element("p").id("myId");
        Assert.assertEquals("myId", element.id());
    }

    @Test
    public void attrTest() {
        Element element = new Element("p").attr("key", "value");
        Assert.assertEquals("value", element.attr("key"));
    }

    @Test
    public void childTest() {
        Element parent = new Element("div");
        Element child = new Element("p");
        parent.appendChild(child);
        Assert.assertEquals(child, parent.child(0));
    }

    @Test
    public void childrenSizeTest() {
        Element parent = new Element("div");
        parent.appendChild(new Element("p"));
        parent.appendChild(new Element("span"));
        Assert.assertEquals(2, parent.childrenSize());
    }

    @Test
    public void textNodesTest() {
        Element element = new Element("p");
        TextNode textNode = new TextNode("Hello, world!");
        element.appendChild(textNode);
        Assert.assertEquals(1, element.textNodes().size());
        Assert.assertEquals(textNode, element.textNodes().get(0));
    }

    @Test
    public void isTest() {
        Element element = new Element("p");
        Assert.assertTrue(element.is("p"));
    }

    @Test
    public void closestTest() {
        Element parent = new Element("div");
        Element child = new Element("p");
        parent.appendChild(child);
        Assert.assertEquals(parent, child.closest("div"));
    }

    @Test
    public void appendChildTest() {
        Element parent = new Element("div");
        Element child = new Element("p");
        parent.appendChild(child);
        Assert.assertEquals(child, parent.child(0));
    }

    @Test
    public void prependChildTest() {
        Element parent = new Element("div");
        Element firstChild = new Element("p");
        Element secondChild = new Element("span");
        parent.appendChild(firstChild);
        parent.prependChild(secondChild);
        Assert.assertEquals(secondChild, parent.child(0));
    }

    @Test
    public void appendTextTest() {
        Element element = new Element("p");
        element.appendText("Hello, world!");
        Assert.assertEquals("Hello, world!", element.text());
    }

    @Test
    public void siblingElementsTest() {
        Element parent = new Element("div");
        Element firstChild = new Element("p");
        Element secondChild = new Element("span");
        parent.appendChild(firstChild);
        parent.appendChild(secondChild);
        Elements siblings = firstChild.siblingElements();
        Assert.assertEquals(1, siblings.size());
        Assert.assertEquals(secondChild, siblings.get(0));
    }

    @Test
    public void nextElementSiblingTest() {
        Element parent = new Element("div");
        Element firstFirstChild = new Element("p");
        Element secondChild = new Element("span");
        parent.appendChild(firstChild);
        parent.appendChild(secondChild);
        Assert.assertEquals(secondChild, firstChild.nextElementSibling());
    }

    @Test
    public void previousElementSiblingTest() {
        Element parent = new Element("div");
        Element firstChild = new Element("p");
        Element secondChild = new Element("span");
        parent.appendChild(firstChild);
        parent.appendChild(secondChild);
        Assert.assertEquals(firstChild, secondChild.previousElementSibling());
    }

    @Test
    public void htmlTest() {
        Element element = new Element("p");
        element.html("<span>Hello</span>");
        Assert.assertEquals("<span>Hello</span>", element.html());
    }

    @Test
    public void classNamesTest() {
        Element element = new Element("p").addClass("class1").addClass("class2");
        Assert.assertEquals(2, element.classNames().size());
        Assert.assertTrue(element.classNames().contains("class1"));
        Assert.assertTrue(element.classNames().contains("class2"));
    }

    @Test
    public void removeClassTest() {
        Element element = new Element("p").addClass("class1").removeClass("class1");
        Assert.assertFalse(element.hasClass("class1"));
    }

    @Test
    public void insertChildrenTest() {
        Element parent = new Element("div");
        Element firstChild = new Element("p");
        Element secondChild = new Element("span");
        parent.insertChildren(0, Arrays.asList(firstChild, secondChild));
        Assert.assertEquals(firstChild, parent.child(0));
        Assert.assertEquals(secondChild, parent.child(1));
    }
}
