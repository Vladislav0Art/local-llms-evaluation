import io.github.vmzakharov.ecdataframe.dsl.*;
import io.github.vmzakharov.ecdataframe.util.PrinterFactory;
import org.junit.Test;
import static org.junit.Assert.assertEquals;

public class PrettyPrintVisitorTests {

    @Test
    public void visitAssignExprTest() {
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(PrinterFactory.newPrinter());

        AssingExpr assingExpr = new AssingExpr(new VarExpr("x"), new DecimalExpr(5));
        assingExpr.accept(visitor);

        assertEquals("x = 5", assingExpr.toString());
    }

    @Test
    public void visitBinaryExprTest() {
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(PrinterFactory.newPrinter());

        BinaryExpr binaryExpr = new BinaryExpr(new VarExpr("x"), Op.ADD, new DecimalExpr(5));
        binaryExpr.accept(visitor);

        assertEquals("x + 5", binaryExpr.toString());
    }

    @Test
    public void visitUnaryExprTest() {
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(PrinterFactory.newPrinter());

        UnaryExpr unaryExpr = new UnaryExpr(new VarExpr("x"), Op.NEGATE);
        unaryExpr.accept(visitor);

        assertEquals("-x", unaryExpr.toString());
    }

    @Test
    public void visitConstExprTest() {
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(PrinterFactory.newPrinter());

        Value constExpr = new DecimalExpr(5);
        constExpr.accept(visitor);

        assertEquals("5", constExpr.toString());
    }

    @Test
    public void visitFunctionCallExprTest() {
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(PrinterFactory.newPrinter());

        Value constExpr = new DecimalExpr(5);
        FunctionCallExpr functionCallExpr = new FunctionCallExpr("sqrt", constExpr);
        functionCallExpr.accept(visitor);

        assertEquals("sqrt(5)", functionCallExpr.toString());
    }

    // Add similar test methods for all other visit methods
}
