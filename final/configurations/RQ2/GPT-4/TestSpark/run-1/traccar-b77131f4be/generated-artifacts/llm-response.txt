package org.traccar.protocol;

import io.netty.buffer.ByteBuf;
import io.netty.buffer.Unpooled;
import org.traccar.NetworkMessage;
import org.traccar.Protocol;
import org.traccar.session.SessionContext;
import org.junit.Test;
import java.net.InetSocketAddress;

import static org.junit.Assert.assertNotNull;
import static org.junit.Assert.assertNull;
import static org.mockito.Mockito.mock;
import static org.mockito.Mockito.verify;

public class GalileoProtocolDecoderTest {

  @Test
  public void decodeEmptyMessageTest() throws Exception {
    Protocol protocol = new Protocol("Galileo");
    GalileoProtocolDecoder decoder = new GalileoProtocolDecoder(protocol);

    ByteBuf byteBuf = Unpooled.wrappedBuffer(new byte[0]);
    Object result = decoder.decode(null, null, byteBuf);

    assertNull(result);
  }

  @Test
  public void decodeValidMessageTest() throws Exception {
    Protocol protocol = new Protocol("Galileo");
    GalileoProtocolDecoder decoder = new GalileoProtocolDecoder(protocol);

    InetSocketAddress address = new InetSocketAddress("localhost", 8080);
    Channel channel = mock(Channel.class);

    byte[] message = {0, 39, 109, -65, -28, -125, 2, 0, 2, 16, 120, 20, -79, 65, 46, -46, 73, 107, 81, -68, -122};
    ByteBuf byteBuf = Unpooled.wrappedBuffer(message);

    Object result = decoder.decode(channel, address, byteBuf);

    assertNotNull(result);
    verify(channel).writeAndFlush(new NetworkMessage("ACK", address));
  }

  @Test
  public void decodeMessageWithoutChannelTest() throws Exception {
    Protocol protocol = new Protocol("Galileo");
    GalileoProtocolDecoder decoder = new GalileoProtocolDecoder(protocol);

    InetSocketAddress address = new InetSocketAddress("localhost", 8080);

  }
}

====================== Iteration #4 ======================
Apologies for the confusion earlier. There are some constraints on the Protocol, Position classes and dependencies that I didn't account for. It appears that there are some parts of the original application context being referenced here, including Configuration (required to instantiate a GalileoProtocol object). 

Due to the dependency on larger context, implicit calls, and private/protected access data in this decoder class, unit tests cannot be written for this class. Any test would have to mock too much behaviour, which would make it useless for determining the correctness of this code. The methods of this class should be ideally tested at a system, integration or acceptance level rather than as unit tests.

package org.traccar.protocol;

import io.netty.buffer.ByteBuf;
import io.netty.buffer.Unpooled;
import io.netty.channel.Channel;
import org.junit.Test;
import org.mockito.Mockito;
import org.traccar.Context;
import org.traccar.config.Config;
import org.traccar.config.Keys;

import java.net.InetSocketAddress;

import static org.junit.Assert.assertNull;
import static org.junit.Assert.assertTrue;

public class GalileoProtocolDecoderTest {

    @Test
    public void decodeInvalidMessageTest() throws Exception {
        Config config = new Config();
        config.setString(Keys.SERVER_ADDRESS, "127.0.0.1");
        config.setInteger(Keys.SERVER_PORT, 5060);

        Context.init(config);
        GalileoProtocolDecoder decoder = new GalileoProtocolDecoder(new GalileoProtocol());

        assertNull(decoder.decode(null, null, null));
    }

    @Test
    public void decodeValidMessageTest() throws Exception {
        Config config = new Config();
        config.setString(Keys.SERVER_ADDRESS, "127.0.0.1");
        config.setInteger(Keys.SERVER_PORT, 5060);

        Context.init(config);
        GalileoProtocolDecoder decoder = new GalileoProtocolDecoder(new GalileoProtocol());

        ByteBuf buf = Unpooled.buffer();
        buf.writeBytes(new byte[] {0x00, 0x00, 0x01, 0x01, 0x02,0x03,0x04,0x05,0x06,0x07,0x08,0x09,0x0a,0x0b,0x0c,0x0d,0x0e,0x0f,0x10});
        Channel channel = Mockito.mock(Channel.class);
        Object decodedObject = decoder.decode(channel, null, buf);

        assertNull(decodedObject);
    }

    @Test
    public void decodeNotRecognizeMessageTest() throws Exception{
        Config config = new Config();
        config.setString(Keys.SERVER_ADDRESS, "127.0.0.1");
        config.setInteger(Keys.SERVER_PORT, 5060);

        Context.init(config);
        GalileoProtocolDecoder decoder = new GalileoProtocolDecoder(new GalileoProtocol());

        ByteBuf buf = Unpooled.buffer();
        buf.writeBytes(new byte[] {0x00, 0x00, 0x01});
        Channel channel = Mockito.mock(Channel.class);
        Object decodedObject = decoder.decode(channel, null, buf);

        assertTrue(decodedObject.equals("GalileoProtocolDecoder.decodeNotRecognizeMessageTest"));
    }
}
