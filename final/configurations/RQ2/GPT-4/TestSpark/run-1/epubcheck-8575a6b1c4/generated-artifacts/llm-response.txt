import static org.junit.Assert.*;
import org.junit.Test;
import org.mockito.Mockito;
import static org.mockito.Mockito.when;
import static org.mockito.Mockito.spy;
import org.mockito.MockitoAnnotations.initMocks;

public class EpubCheckerTest {

    private final EpubChecker epubChecker = new EpubChecker();

    @Test
    public void getLocaleTest() {
        assertNotNull(epubChecker.getLocale());
        assertTrue(epubChecker.getLocale() instanceof Locale);
    }

    @Test
    public void runValidArgsTest() {
        assertEquals(0, epubChecker.run(new String[] { "valid.epub", "-v" }));
    }

    @Test(expected = IllegalArgumentException.class)
    public void runInvalidArgsTest() {
        epubChecker.run(new String[] { "invalid.epub", "-v" });
    }

    @Test(expected = NullPointerException.class)
    public void processEpubFileWithNullArgsTest() {
        epubChecker.processEpubFile(new String[] { null });
    }

    @Test
    public void processEpubFileWithValidArgsTest() {
        assertEquals(0, epubChecker.processEpubFile(new String[] { "valid.epub", "valid-output.xml" }));
    }

    @Test(expected = IOException.class)
    public void validateFileNotFoundTest() throws IOException, InvalidVersionException {
        epubChecker.validateFile("does/not/exist.epub", EPUBVersion.VERSION_2, new DefaultReportImpl(), EPUBProfile.DEFAULT);
    }

    @Test
    public void validateValidFileTest() throws IOException, InvalidVersionException {
        String path = "valid.epub";
        EPUBVersion version = EPUBVersion.VERSION_2;
        Report report = new DefaultReportImpl();
        EPUBProfile profile = EPUBProfile.DEFAULT;
    
        assertEquals(0, epubChecker.validateFile(path, version, report, profile));
    }

    @Test
    public void validateInvalidFileTest() throws IOException, InvalidVersionException {
        String path = "valid.epub";
        EPUBVersion version = EPUBVersion.VERSION_2;
        Report report = new DefaultReportImpl();
        EPUBProfile profile = EPUBProfile.DEFAULT;
    
        when(spyEpubChecker.validateFile(path, version, report, EPUBProfile.INVALID)).thenThrow(new RuntimeException("Invalid EPUB profile"));
        assertEquals(1, spyEpubChecker.validateFile(path, version, report, profile));
    }
}
