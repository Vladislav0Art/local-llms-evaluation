import org.jsoup.parser.Tag;
import org.jsoup.parser.ParseSettings;
import org.junit.Test;
import static org.junit.Assert.*;

public class TagTest {

    @Test
    public void getNameTest() {
        Tag tag = Tag.valueOf("html");
        assertEquals("html", tag.getName());
    }

    @Test
    public void normalNameTest() {
        Tag tag = Tag.valueOf("HTML");
        assertEquals("html", tag.normalName());
    }

    @Test
    public void valueOfWithNameAndSettingsTest() {
        ParseSettings settings = new ParseSettings(true, true);
        Tag tag = Tag.valueOf("div", settings);
        assertEquals("DIV", tag.normalName());
        assertEquals("DIV", tag.getName());
    }

    @Test
    public void isBlockTest() {
        Tag blockTag = Tag.valueOf("div");
        assertTrue(blockTag.isBlock());
        Tag nonBlockTag = Tag.valueOf("span");
        assertFalse(nonBlockTag.isBlock());
    }

    @Test
    public void formatAsBlockTest() {
        Tag blockTag = Tag.valueOf("div");
        assertTrue(blockTag.formatAsBlock());
    }

    @Test
    public void isInlineTest() {
        Tag inlineTag = Tag.valueOf("span");
        assertTrue(inlineTag.isInline());
        Tag nonInlineTag = Tag.valueOf("table");
        assertFalse(nonInlineTag.isInline());
    }

    @Test
    public void isEmptyTest() {
        Tag emptyTag = Tag.valueOf("img");
        assertTrue(emptyTag.isEmpty());
        Tag nonEmptyTag = Tag.valueOf("p");
        assertFalse(nonEmptyTag.isEmpty());
    }

    @Test
    public void isSelfClosingTest() {
        Tag selfClosingTag = Tag.valueOf("img");
        assertTrue(selfClosingTag.isSelfClosing());
        Tag nonSelfClosingTag = Tag.valueOf("p");
        assertFalse(nonSelfClosingTag.isSelfClosing());
    }

    @Test
    public void isKnownTagTest() {
        assertTrue(Tag.isKnownTag("div"));
        assertFalse(Tag.isKnownTag("custom-tag"));
    }

    @Test
    public void preserveWhitespaceTest() {
        Tag tagNeedsWhitespace = Tag.valueOf("pre");
        assertTrue(tagNeedsWhitespace.preserveWhitespace());
        Tag tagNotNeedWhitespace = Tag.valueOf("div");
        assertFalse(tagNotNeedWhitespace.preserveWhitespace());
    }

    @Test
    public void isFormListedTest() {
        Tag tag = Tag.valueOf("button");
        assertTrue(tag.isFormListed());
        Tag nonListedTag = Tag.valueOf("div");
        assertFalse(nonListedTag.isFormListed());
    }

    @Test
    public void isFormSubmittableTest() {
        Tag tag = Tag.valueOf("input");

    }

    @Test
    public void setSelfClosingTest() {

        assertTrue(tag.isSelfClosing());

    public void equalsTest() {
        Tag tag1 = Tag.valueOf("div");

        assertTrue(tag1.equals(tag2));

    public void hashCodeTest() {
        Tag tag = Tag.valueOf("div");

    }

    public void toStringTest() {
        Tag tag = Tag.valueOf("div");
        assertEquals("div", tag.toString());
    }

    public void cloneTest() throws CloneNotSupportedException {
        Tag originalTag = Tag.valueOf("div");
        Tag cloneTag = originalTag.clone();
        assertEquals(originalTag, cloneTag);
    }

}
