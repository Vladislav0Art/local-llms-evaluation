import graphql.annotations.processor.exceptions.GraphQLAnnotationsException;
import graphql.annotations.processor.retrievers.fieldBuilders.DirectivesBuilder;
import graphql.schema.GraphQLDirective;
import graphql.annotations.processor.ProcessingElementsContainer;
import org.junit.Assert;
import org.junit.Test;
import org.mockito.Mockito;

import java.lang.reflect.Field;

public class DirectivesBuilderTest {

    @Test
    public void buildNullAnnotationTest() {
        ProcessingElementsContainer container = new ProcessingElementsContainer();
        Field field = getClass().getDeclaredFields()[0];
        DirectivesBuilder builder = new DirectivesBuilder(field, container);
        GraphQLDirective[] directives = builder.build();

        Assert.assertNotNull(directives);
        Assert.assertEquals(0, directives.length);
    }

    @Test(expected = GraphQLAnnotationsException.class)
    public void buildWithInvalidAnnotationTest() {
        ProcessingElementsContainer container = Mockito.mock(ProcessingElementsContainer.class);
        Field field = getClass().getDeclaredFields()[0];
        DirectivesBuilder builder = new DirectivesBuilder(field, container);

        builder.build();
    }

    @Test
    public void buildWithValidAnnotationTest() {
        ProcessingElementsContainer container = new ProcessingElementsContainer();
        GraphQLDirective annotation = Mockito.mock(GraphQLDirective.class);
        Field field = Mockito.mock(Field.class);
        Mockito.when(field.getAnnotations()).thenReturn(new Annotation[]{annotation});

        DirectivesBuilder builder = new DirectivesBuilder(field, container);
        GraphQLDirective[] directives = builder.build();

        Assert.assertNotNull(directives);
        Assert.assertEquals(1, directives.length);
    }

    @Test
    public void buildWithMultipleValidAnnotationsTest() {
        ProcessingElementsContainer container = new ProcessingElementsContainer();
        GraphQLDirective annotation1 = Mockito.mock(GraphQLDirective.class);
        GraphQLDirective annotation2 = Mockito.mock(GraphQLDirective.class);
        Field field = Mockito.mock(Field.class);
        Mockito.when(field.getAnnotations()).thenReturn(new Annotation[]{annotation1, annotation2});

        DirectivesBuilder builder = new DirectivesBuilder(field, container);
        GraphQLDirective[] directives = builder.build();

        Assert.assertNotNull(directives);
        Assert.assertEquals(2, directives.length);
    }
}
