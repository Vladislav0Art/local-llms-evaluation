import com.sun.codemodel.JCodeModel;
import com.sun.tools.xjc.addon.xew.XmlElementWrapperPlugin;
import com.sun.tools.xjc.Options;
import com.sun.tools.xjc.outline.Outline;
import org.junit.Test;

import java.io.IOException;

import static org.mockito.Mockito.mock;
import static org.mockito.Mockito.verify;
import static org.mockito.Mockito.when;

public class XmlElementWrapperPluginTest {

    @Test
    public void runInternalTest() throws Exception {
        XmlElementWrapperPlugin wrapperPlugin = new XmlElementWrapperPlugin();
        
        Options options = mock(Options.class);
        JCodeModel codeModel = new JCodeModel();
        when(options.getCodeModel()).thenReturn(codeModel);
        
        Outline outline = mock(Outline.class);
        when(outline.getCodeModel()).thenReturn(codeModel);

        wrapperPlugin.initOutline(outline, options);
        wrapperPlugin.runInternal(outline);
        
        verify(options).getCodeModel();
    }
    
    @Test(expected = IOException.class)
    public void runInternalThrowsExceptionTest() throws Exception {
        XmlElementWrapperPlugin wrapperPlugin = new XmlElementWrapperPlugin();
        
        Options options = mock(Options.class);
        JCodeModel codeModel = new JCodeModel();
        when(options.getCodeModel()).thenReturn(codeModel);

        when(outline.getCodeModel()).thenThrow(IOException.class);

        wrapperPlugin.initOutline(outline, options);
        wrapperPlugin.runInternal(outline);
    }

    @Test(expected = ClassNotFoundException.class)
    public void runInternalThrowsClassNotFoundExceptionTest() throws Exception {
        XmlElementWrapperPlugin wrapperPlugin = new XmlElementWrapperPlugin();

        Options options = mock(Options.class);
        JCodeModel codeModel = new JCodeModel();
        when(options.getCodeModel()).thenReturn(codeModel);

        when(outline.getCodeModel()).thenThrow(ClassNotFoundException.class);

        wrapperPlugin.initOutline(outline, options);
        wrapperPlugin.runInternal(outline);
    }
}
