import io.github.vmzakharov.ecdataframe.dsl.*;
import io.github.vmzakharov.ecdataframe.dsl.visitor.PrettyPrintVisitor;
import io.github.vmzakharov.ecdataframe.util.CollectingPrinter;
import org.eclipse.collections.impl.factory.Lists;
import org.junit.Test;

import static org.junit.Assert.assertEquals;

public class PrettyPrintVisitorTest {
    @Test
    public void exprToStringTest() {
        Expression expr = new VarExpr("testVar");
        String result = PrettyPrintVisitor.exprToString(expr);
        assertEquals("testVar", result);
    }

    @Test
    public void visitAssignExprTest() {
        PrettyPrintVisitor visitor = new PrettyPrintVisitor();
        AssingExpr expr = new AssingExpr(new VarExpr("x"), new DecimalExpr(5));
        expr.accept(visitor);
        assertEquals("x <== 5", visitor.toString());
    }

    @Test
    public void visitBinaryExprTest() {
        PrettyPrintVisitor visior = new PrettyPrintVisitor();
        BinaryExpr expr = new BinaryExpr(new VarExpr("x"), new VarExpr("y"), BinaryOp.PLUS);
        expr.accept(visitor);
        assertEquals("x + y", visitor.toString());
    }

    @Test
    public void visitUnaryExprTest() {
        PrettyPrintVisitor visitor = new PrettyPrintVisitor();
        UnaryExpr expr = new UnaryExpr(new VarExpr("x"), UnaryOp.MINUS);
        expr.accept(visitor);
        assertEquals("-x", visitor.toString());
    }

    @Test
    public void visitVectorExprTest() {
        PrettyPrintVisitor visitor = new PrettyPrintVisitor();
        VectorExpr vectorExpr = new VectorExpr(Lists.mutable.of(new DecimalExpr(1), new DecimalExpr(2), new DecimalExpr(3)));
        vectorExpr.accept(visitor);
        assertEquals("[1, 2, 3]", visitor.toString());
    }

    @Test
    public void visitProjectionExprTest() {
        PrettyPrintVisitor visitor = new PrettyPrintVisitor();
        ProjectionExpr expr = new ProjectionExpr(Lists.mutable.with(new VarExpr("variable")), "projection");
        expr.accept(visitor);
        assertEquals("variable.projection", visitor.toString());
    }
}
