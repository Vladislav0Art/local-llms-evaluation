public class W3CDomTest {

    public void namespaceAwareTest() {
        W3CDom w3CDom = new W3CDom();
        assertTrue(w3CDom.namespaceAware());
    }

    @Test
    public void namespaceAwareParameterTest() {
        W3CDom w3CDom = new W3CDom().namespaceAware(false);
        assertFalse(w3CDom.namespaceAware());
    }

    @Test
    public void convertDocumentInTest() {
        Document document = new Document("");
        assertNotNull(W3CDom.convert(document));
    }

    @Test
    public void asStringDocumentPropertiesTest() {
        Map<String, String> properties = new HashMap<>();
        properties.put("INDENT", "yes");
        properties.put("ENCODING", "UTF-8");
        properties.put("METHOD", "xml");
        assertNotNull(W3CDom.asString(document, properties));
    }

    @Test
    public void propertiesFromMapTest() {
        Map<String, String> map = new HashMap<>();
        map.put("INDENT", "yes");
        map.put("ENCODING", "UTF-8");
        map.put("METHOD", "xml");
        Properties properties = W3CDom.propertiesFromMap(map);
        assertNotNull(properties);
        assertEquals("yes", properties.get("INDENT"));
    }

    @Test
    public void outputHtmlTest() {
        assertNotNull(W3CDom.OutputHtml());
    }

    @Test
    public void outputXmlTest() {
        assertNotNull(W3CDom.OutputXml());
    }

    @Test
    public void fromJsoupDocumentInTest() {
        Document document = new Document("");
        W3CDom w3CDom = new W3CDom();
        assertNotNull(w3CDom.fromJsoup(document));
    }

    @Test
    public void fromJsoupElementInTest() {
        Element element = new Element(Tag.valueOf("p"), "");
        W3CDom w3CDom = new W3CDom();
        assertNotNull(w3CDom.fromJsoup(element));
    }

    @Test
    public void convertDocumentInDocumentOutTest() {
        Document document = new Document("");
        W3CDom w3CDom = new W3CDom();
        w3CDom.convert(document, document);
        assertNotNull(document);
    }

    @Test
    public void convertElementInDocumentOutTest() {
        Document document = new Document("");
        Element element = new Element(Tag.valueOf("p"), "");
        W3CDom w3CDom = new W3CDom();
        w3CDom.convert(element, document);
        assertNotNull(document.body());
    }

    @Test
    public void selectXpathStringDocumentTest() {
        Document document = new Document("");
        W3CDom w3CDom = new W3CDom();
        NodeList nodeList = w3CDom.selectXpath("/html/body", document);
        assertEquals(0, nodeList.getLength());
    }

    @Test
    public void selectXpathStringNodeTest() {
        Node node = new Node("p");
        W3CDom w3CDom = new W3CDom();
        NodeList nodeList = w3CDom.selectXpath("/p", node);
        assertNull(nodeList);
    }

    @Test
    public void sourceNodesNodeListClassTest() {
        NodeList nodeList = new NodeList();
        W3CDom w3CDom = new W3CDom();
        List<org.jsoup.nodes.Node> nodes = w3CDom.sourceNodes(nodeList, org.jsoup.nodes.Node.class);
        assertTrue(nodes.isEmpty());
    }

    @Test
    public void contextNodeDocumentTest() {
        Document document = new Document("");
        W3CDom w3CDom = new W3CDom();
        assertNotNull(w3CDom.contextNode(document));
    }

    @Test
    public void asStringDocumentTest() {
        Document document = new Document("");
        assertNotNull(W3CDom.asString(document));
    }
}
