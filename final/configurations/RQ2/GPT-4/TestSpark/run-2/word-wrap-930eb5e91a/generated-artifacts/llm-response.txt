package org.davidmoten.text.utils;

import org.junit.Test;
import org.mockito.Mockito;

import java.io.*;
import java.nio.charset.Charset;
import java.nio.charset.StandardCharsets;
import java.util.Arrays;
import java.util.HashSet;

import static org.junit.Assert.*;

public class WordWrapTest {

    @Test
    public void fromReaderTest() {
        Reader reader = new StringReader("Test Data");
        WordWrap.Builder builder = WordWrap.from(reader);
        assertNotNull(builder);
    }

    @Test
    public void fromClasspathUtf8Test() {
        WordWrap.Builder builder = WordWrap.fromClasspathUtf8("TextResource.txt");
        assertNotNull(builder);
    }

    @Test
    public void fromClasspathCharsetTest() {
        WordWrap.Builder builder = WordWrap.fromClasspath("TextResource.txt", StandardCharsets.UTF_8);
        assertNotNull(builder);
    }

    @Test
    public void fromCharSequenceTest() {
        WordWrap.Builder builder = WordWrap.from("Test String");
        assertNotNull(builder);
    }

    @Test
    public void fromUtf8InputStreamTest() {
        InputStream in = new ByteArrayInputStream("Test Data".getBytes());
        WordWrap.Builder builder = WordWrap.fromUtf8(in);
        assertNotNull(builder);
    }

    @Test
    public void fromInputStreamCharsetTest() {
        InputStream in = new ByteArrayInputStream("Test Data".getBytes());
        WordWrap.Builder builder = WordWrap.from(in, StandardCharsets.UTF_8);
        assertNotNull(builder);
    }

    @Test
    public void fromFileCharsetTest() throws FileNotFoundException {
        File file = new File("src/test/resources/testFile.txt");
        WordWrap.Builder builder = WordWrap.from(file, StandardCharsets.UTF_8);
        assertNotNull(builder);
    }

    @Test(expected = NullPointerException.class)
    public void rightTrimNullTest() {
        WordWrap.rightTrim(null);
    }

    @Test
    public void rightTrimTest() {
        CharSequence result = WordWrap.rightTrim("   Test   ");
        assertEquals("   Test", result);
    }

    @Test
    public void isWhitespaceTest() {
        assertTrue(WordWrap.isWhitespace("   "));
    }

    @Test
    public void isNotWhitespaceTest() {
        assertFalse(WordWrap.isWhitespace("Test"));
    }

    @Test
    public void wordWrapFirstScenarioTest() throws IOException {
        Reader in = new StringReader("Test Data");
        Writer out = new StringWriter();
        Set<Character> extraWordChars = new HashSet<>();
        extraWordChars.add('-');
        WordWrap.wordWrap(in, out, "\n", 10, input -> input.length(), extraWordChars, true, true);
        assertEquals("Test Data\n", out.toString());
    }

    @Test
    public void wordWrapSecondScenarioTest() throws IOException {
        Reader in = new StringReader("TestDataExample");
        Writer out = new StringWriter();
        HashSet<Character> extraWordChars = new HashSet<>();

        WordWrap.wordWrap(in, out, "\n", 5, CharSequence::length,
                extraWordChars, true, false);

        assertEquals("TestD\nataEx\nample\n", out.toString());
    }
}
