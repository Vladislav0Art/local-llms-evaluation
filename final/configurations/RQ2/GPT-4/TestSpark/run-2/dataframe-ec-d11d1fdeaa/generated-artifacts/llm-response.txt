```java
import io.github.vmzakharov.ecdataframe.dataframe.DataFrame;
import io.github.vmzakharov.ecdataframe.dataframe.DfColumn;
import io.github.vmzakharov.ecdataframe.dataframe.DfColumnSortOrder;
import io.github.vmzakharov.ecdataframe.dsl.value.ValueType;
import org.eclipse.collections.impl.factory.Lists;
import org.junit.Assert;
import org.junit.Test;

public class DataFrameUnitTest {

    @Test
    public void DataFrameConstructorTest() {
        DataFrame df = new DataFrame("newDataFrame");
        Assert.assertEquals("newDataFrame", df.getName());
    }

    @Test(expected = RuntimeException.class)
    public void addStringColumnTwiceTest() {
        DataFrame df = new DataFrame("testFrame");
        df.addStringColumn("name");
        df.addStringColumn("name");
    }

    @Test
    public void addStringColumnTest() {
        DataFrame df = new DataFrame("testFrame");
        df.addStringColumn("name");
        Assert.assertEquals(1, df.columnCount());
    }

    @Test
    public void addRowTest() {
        DataFrame df = new DataFrame("testFrame");
        df.addStringColumn("name");
        df.addRow("John");
        Assert.assertEquals(1, df.rowCount());
    }

    @Test
    public void rowCountTest() {
        DataFrame df = new DataFrame("testFrame");
        df.addStringColumn("name");
        df.addRow("John");
        df.addRow("Alex");
        Assert.assertEquals(2, df.rowCount());
    }

    @Test(expected = RuntimeException.class)
    public void getColumnAtNegativeIndexTest() {
        DataFrame df = new DataFrame("testFrame");
        df.addStringColumn("name");
        df.getColumnAt(-1);
    }

    @Test
    public void isPoolingEnabledTest() {
        DataFrame df = new DataFrame("testPoolFrame");
        Assert.assertFalse(df.isPoolingEnabled());
    }

    @Test
    public void getColumnsTest() {
        DataFrame df = new DataFrame("testFrame");
        df.addStringColumn("name");
        df.addStringColumn("address");
        Assert.assertEquals(2, df.getColumns().size());
    }

    @Test
    public void addColumnTest() {
        DataFrame df = new DataFrame("testFrame");
        df.addColumn("salary", ValueType.LONG);
        Assert.assertTrue(df.hasColumn("salary"));
    }

    @Test(expected = RuntimeException.class)
    public void addColumnTwiceTest() {
        DataFrame df = new DataFrame("testFrame");
        df.addColumn("salary", ValueType.LONG);
        df.addColumn("salary", ValueType.STRING);
    }

    @Test
    public void newColumnTest() {
        DataFrame df = new DataFrame("testFrame");
        DfColumn column = df.newColumn("age", ValueType.LONG);
        Assert.assertEquals("age", column.getName());
        Assert.assertEquals(ValueType.LONG, column.getType());
    }

    @Test
    public void aggregateTest() {
        DataFrame df = new DataFrame("testFrame");
        df.addStringColumn("name");
        df.addLongColumn("salary");
        df.addRow(Lists.mutable.of("John", 50000L));
        df.aggregate(Lists.mutable.of(t -> t.count()));
        Assert.assertEquals(1, df.getColumns().size());
    }

    @Test
    public void aggregateByTest() {
        DataFrame df = new DataFrame("testFrame");
        df.addStringColumn("name");
        df.addLongColumn("salary");
        df.addRow(Lists.mutable.of("John", 50000L));
        df.addRow(Lists.mutable.of("Alex", 60000L));
        df.aggregateBy(Lists.mutable.of(t -> t.count()), Lists.mutable.of("name"));
        Assert.assertEquals(2, df.getColumns().size());
    }

    @Test
    public void distinctTest() {
        DataFrame df = new DataFrame("testFrame");
        df.addStringColumn("name");
        df.addRow("John");
        df.addRow("John");
        Assert.assertEquals(2, df.rowCount());
        df.distinct();
        Assert.assertEquals(1, df.rowCount());
    }

    @Test
    public void sortByTest() {
        DataFrame df = new DataFrame("testFrame");
        df.addStringColumn("name");
        df.addLongColumn("age");
        df.addRow(Lists.mutable.of("John", 25L));
        df.addRow(Lists.mutable.of("Alex", 50L));
        df.sortBy(Lists.mutable.of("name"), Lists.mutable.of(DfColumnSortOrder.ASC));
        Assert.assertEquals("Alex", df.getObject("name", 1));
    }
}
```
