```java
import org.junit.Assert;
import org.junit.Test;
import org.jsoup.Connection;
import org.jsoup.helper.UrlBuilder;

import java.io.UnsupportedEncodingException;
import java.net.MalformedURLException;
import java.net.URL;

public class UrlBuilderTest {

    @Test
    public void appendKeyValValidInputTest() throws UnsupportedEncodingException {
        UrlBuilder urlBuilder = new UrlBuilder(new URL("http://example.com"));
        Connection.KeyVal kv = Connection.KeyVal.create("key", "value");
        urlBuilder.appendKeyVal(kv);
        URL url = urlBuilder.build();
        Assert.assertEquals("http://example.com?key=value", url.toString());
    }

    @Test(expected = UnsupportedEncodingException.class)
    public void appendKeyValUnsupportedEncodingTest() throws UnsupportedEncodingException {
        UrlBuilder urlBuilder = new UrlBuilder(new URL("http://example.com"));
        Connection.KeyVal kv = Connection.KeyVal.create("key", "\u0080");
        urlBuilder.appendKeyVal(kv);
    }

    @Test
    public void buildEmptyUrlTest() throws MalformedURLException {
        UrlBuilder urlBuilder = new UrlBuilder(new URL("http://"));
        URL url = urlBuilder.build();
        Assert.assertEquals("http://", url.toString());
    }

    @Test
    public void buildValidUrlTest() throws MalformedURLException {
        UrlBuilder urlBuilder = new UrlBuilder(new URL("http://example.com"));
        URL url = urlBuilder.build();
        Assert.assertEquals("http://example.com", url.toString());
    }

    @Test
    public void buildUrlWithQueryParamsTest() throws MalformedURLException, UnsupportedEncodingException {
        UrlBuilder urlBuilder = new UrlBuilder(new URL("http://example.com"));
        urlBuilder.appendKeyVal(Connection.KeyVal.create("key", "value"));
        URL url = urlBuilder.build();
        Assert.assertEquals("http://example.com?key=value", url.toString());
    }
}
```
