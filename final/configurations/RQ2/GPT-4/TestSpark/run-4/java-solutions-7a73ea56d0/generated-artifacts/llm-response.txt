import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.junit.MockitoJUnitRunner;
import leetcode.medium.OnlineStockSpan;

import static org.junit.Assert.assertEquals;

@RunWith(MockitoJUnitRunner.class)
public class OnlineStockSpanTest {

    @Test
    public void nextEmptyStackTest() {
        OnlineStockSpan span = new OnlineStockSpan();
        int result = span.next(100);
        assertEquals(1, result);
    }

    @Test
    public void nextNonEmptyStackTest() {
        OnlineStockSpan span = new OnlineStockSpan();
        span.next(80);
        int result = span.next(100);
        assertEquals(2, result);
    }

    @Test
    public void nextLessThanStackTopTest() {
        OnlineStockSpan span = new OnlineStockSpan();
        span.next(100);
        int result = span.next(80);
        assertEquals(1, result);
    }

    @Test
    public void calculateSpansEmptyArrayTest() {
        OnlineStockSpan span = new OnlineStockSpan();
        int[] result = span.calculateSpans(new int[0]);
        assertEquals(0, result.length);
    }

    @Test
    public void calculateSpansNonEmptyArrayTest() {
        OnlineStockSpan span = new OnlineStockSpan();
        int[] result = span.calculateSpans(new int[]{100, 80, 60, 70, 60, 75, 85});
        int[] expected = new int[]{1, 1, 1, 2, 1, 4, 6};
        assertEquals(expected.length, result.length);
        for (int i = 0; i < expected.length; i++) {
            assertEquals(expected[i], result[i]);
        }
    }
}
