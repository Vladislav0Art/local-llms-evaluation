import io.netty.channel.Channel;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.junit.MockitoJUnitRunner;
import org.mockito.Mock;
import org.traccar.Protocol;
import org.traccar.helper.UnitsConverter;
import org.traccar.model.Position;
import org.traccar.protocol.GalileoProtocolDecoder;
import org.traccar.session.DeviceSession;

import java.net.SocketAddress;
import java.nio.charset.StandardCharsets;

import static org.hamcrest.CoreMatchers.is;
import static org.hamcrest.MatcherAssert.assertThat;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class GalileoProtocolDecoderTest {
	
    @Mock
    private Protocol protocol;
	
    @Mock
    private Channel channel;
	
    @Mock
    private SocketAddress remoteAddress;
	
    @Test
    public void GalileoProtocolDecoderTest() {
        GalileoProtocolDecoder decoder = new GalileoProtocolDecoder(protocol);
        assertThat(decoder, is(GalileoProtocolDecoder.class));
    }

    @Test
    public void decodeTest() throws Exception {
        GalileoProtocolDecoder decoder = new GalileoProtocolDecoder(protocol);
        DeviceSession deviceSession = new DeviceSession("deviceId");
		String hex = "Byte data as hex str";
        byte[] bytes = hex.getBytes(StandardCharsets.UTF_8);
        when(channel.write(new NetworkMessage(bytes, remoteAddress))).thenReturn(deviceSession);

        Object result = decoder.decode(channel, remoteAddress, Unpooled.wrappedBuffer(bytes));
        assertThat(result, is(Position.class));
    }

    @Test
    public void decodeDifferentDataTest() throws Exception {
        GalileoProtocolDecoder decoder = new GalileoProtocolDecoder(protocol);
        DeviceSession deviceSession = new DeviceSession("differentDeviceId");
		String hex = "Different byte data as hex str";
        byte[] bytes = hex.getBytes(StandardCharsets.UTF_8);
        when(channel.write(new NetworkMessage(bytes, remoteAddress))).thenReturn(deviceSession);

        Object result = decoder.decode(channel, remoteAddress, Unpooled.wrappedBuffer(bytes));
        assertThat(result, is(Position.class));
    }
}
