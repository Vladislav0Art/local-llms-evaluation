import org.junit.Test;
import static org.junit.Assert.*;

public class JacksonJsonTransformerTest {

    @Test
    public void parseNullJsonTest() {
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        try {
            transformer.parse(null, Project.class);
            fail("Expected an HttpBadRequestException to be thrown");
        } catch (HttpBadRequestException e) {
            assertEquals("Could not parse null JSON", e.getMessage());
        }
    }

    @Test
    public void parseEmptyJsonTest() {
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        Project result = transformer.parse("", Project.class);
        assertNotNull(result);
    }

    @Test
    public void parseValidJsonTest() {
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        String validJson = "{\"name\":\"Test Project\"}";
        Project result = transformer.parse(validJson, Project.class);
        assertEquals("Test Project", result.getName());
    }

    @Test
    public void parseInvalidJsonTest() {
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        try {
            transformer.parse("{InvalidJson}", Project.class);
            fail("Expected an HttpBadRequestException to be thrown");
        } catch (HttpBadRequestException e) {
            assertEquals("Could not parse JSON", e.getMessage());
        }
    }

    @Test
    public void convertNullObjectTest() {
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        try {
            transformer.convert(null);
            fail("Expected an HttpBadRequestException to be thrown");
        } catch (HttpBadRequestException e) {
            assertEquals("Cannot convert null object", e.getMessage());
        }
    }

    @Test
    public void convertValidObjectTest() {
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        Project project = new Project();
        project.setName("Test Project");
        String result = transformer.convert(project);
        assertTrue(result.contains("\"name\":\"Test Project\""));
    }

    @Test
    public void convertInvalidObjectTest() {
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        try {
            transformer.convert(new Object());
            fail("Expected an HttpBadRequestException to be thrown");
        } catch(HttpBadRequestException e) {
            assertEquals("Could not convert object to JSON", e.getMessage());
        }
    }
}
