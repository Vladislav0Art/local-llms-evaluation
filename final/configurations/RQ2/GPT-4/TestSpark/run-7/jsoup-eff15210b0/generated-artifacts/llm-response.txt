import org.jsoup.Connection;
import org.jsoup.helper.HttpConnection;
import org.junit.Test;
import org.mockito.Mockito;

import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.io.InputStream;
import java.net.Proxy;
import java.net.URL;
import java.util.HashMap;
import java.util.Map;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertTrue;

public class HttpConnectionTest {

    @Test
    public void connectStringTest() {
        Connection connection = HttpConnection.connect("https://jsoup.org/");
        assertEquals("https://jsoup.org/", connection.request().url().toString());
    }

    @Test
    public void connectURLTest() throws Exception {
        URL url = new URL("https://jsoup.org/");
        Connection connection = HttpConnection.connect(url);
        assertEquals(url, connection.request().url());
    }

    @Test
    public void urlTest() throws Exception {
        URL url = new URL("https://jsoup.org/");
        HttpConnection connection = new HttpConnection();
        connection.url(url);

        assertEquals(url, connection.request().url());
    }

    @Test
    public void proxyTest() {
        Proxy proxy = Mockito.mock(Proxy.class);
        HttpConnection connection = new HttpConnection();
        connection.proxy(proxy);

        assertEquals(proxy, connection.request().proxy());
    }

    @Test
    public void dataStringTest() {
        HttpConnection connection = new HttpConnection();
        connection.data("key", "value");

        assertEquals("value", connection.request().data().get(0).value());
    }

    @Test
    public void dataInputStreamTest() {
        InputStream stream = new ByteArrayInputStream("test".getBytes());
        HttpConnection connection = new HttpConnection();
        connection.data("key", "filename.ext", stream);

        assertEquals("test", new String(connection.request().data().get(0).inputStreamValue()));
    }

    @Test
    public void dataMapTest() {
        Map<String, String> data = new HashMap<>();
        data.put("key", "value");

        HttpConnection connection = new HttpConnection();
        connection.data(data);

        assertEquals("value", connection.request().data().get(0).value());
    }

    @Test
    public void cookiesTest() {
        Map<String, String> cookies = new HashMap<>();
        cookies.put("session", "randomsessionid");

        HttpConnection connection = new HttpConnection();
        connection.cookies(cookies);

        assertEquals("randomsessionid", connection.request().cookies().get("session"));
    }

    @Test(expected = IOException.class)
    public void postTest() throws IOException {
        HttpConnection connection = new HttpConnection();
        connection.post(); // Should throw an IOException because url is null.
    }

    @Test
    public void parserTest() {
        HttpConnection connection = new HttpConnection();
        connection.parser(org.jsoup.parser.Parser.xmlParser());

        assertEquals(org.jsoup.parser.Parser.xmlParser(), connection.request().parser());
    }

    @Test
    public void maxBodySizeTest() {
        HttpConnection connection = new HttpConnection();
        connection.maxBodySize(1024 * 1024);

        assertEquals(1024 * 1024, connection.request().maxBodySize());
    }

    @Test
    public void ignoreHttpErrorsTest() {
        HttpConnection connection = new HttpConnection();
        connection.ignoreHttpErrors(true);

        assertTrue(connection.request().ignoreHttpErrors());
    }
}
