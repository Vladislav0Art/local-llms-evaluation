    }

        Assert.assertEquals(expectedUrl, actualUrl);
    }

        Connection connection = new HttpConnection();
        Connection.Proxy expectedProxy = new Connection.Proxy("localhost", 8000);

        Assert.assertEquals(expectedProxy, actualProxy);
    }
