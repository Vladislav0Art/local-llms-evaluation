import com.adobe.epubcheck.api.EPUBProfile;
import com.adobe.epubcheck.api.Report;
import com.adobe.epubcheck.tool.EpubChecker;
import com.adobe.epubcheck.util.EPUBVersion;
import com.adobe.epubcheck.util.DefaultReportImpl;
import junit.framework.TestCase;
import org.junit.Rule;
import org.junit.Test;
import org.junit.rules.ExpectedException;

public class EpubCheckerTests extends TestCase {

    @Rule
    public ExpectedException exceptionRule = ExpectedException.none();

    @Test
    public void getLocaleTest() {
        EpubChecker checker = new EpubChecker();
        Locale result = checker.getLocale();
        assertNotNull(result);
    }

    @Test
    public void runTestWithNullArgs() {
        EpubChecker checker = new EpubChecker();
        int result = checker.run(null);
        assertEquals(-1, result);
    }

    @Test
    public void processEpubFileTestWithNullArgs() {
        EpubChecker checker = new EpubChecker();
        int result = checker.processEpubFile(null);
        assertEquals(-1, result);
    }

    @Test
    public void validateFileTestWithInvalidPath() {
        EpubChecker checker = new EpubChecker();
        Report report = new DefaultReportImpl("filename");
        int result = checker.validateFile("invalid path", EPUBVersion.VERSION_2_0, report, EPUBProfile.DEFAULT);
        assertEquals(-1, result);
    }

    @Test
    public void validateFileTestWithValidPath(){
        try {
            File tempFile = File.createTempFile("tempFile", ".tmp");
            EpubChecker checker = new EpubChecker();
            Report report = new DefaultReportImpl("filename");
            int result = checker.validateFile(tempFile.getAbsolutePath(), EPUBVersion.VERSION_2_0, report, EPUBProfile.DEFAULT);
            assertEquals(0, result);
            tempFile.deleteOnExit();
        } catch (IOException e) {
            fail("Failed to create temporary file.");
        }
    }
}
