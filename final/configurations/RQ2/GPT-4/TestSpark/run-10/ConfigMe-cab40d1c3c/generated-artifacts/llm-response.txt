import ch.jalu.configme.configurationdata.PropertyListBuilder;
import ch.jalu.configme.exception.ConfigMeException;
import ch.jalu.configme.properties.Property;
import org.junit.Test;

import static org.junit.Assert.*;

public class PropertyListBuilderTest {

    @Test
    public void addNullPropertyTest() {
        PropertyListBuilder builder = new PropertyListBuilder();

        try {
            builder.add(null);
            fail("Expected ConfigMeException when adding null Property.");
        } catch (ConfigMeException e) {
            assertEquals("Property may not be null", e.getMessage());
        }
    }

    @Test
    public void addPropertyTest() {
        PropertyListBuilder builder = new PropertyListBuilder();
        Property<Object> mockProperty = Mockito.mock(Property.class);

        builder.add(mockProperty);
        List<Property<?>> createdProperties = builder.create();

        assertTrue("Created properties should contain added property.", createdProperties.contains(mockProperty));
    }

    @Test
    public void createEmptyTest() {
        PropertyListBuilder builder = new PropertyListBuilder();
        List<Property<?>> createdProperties = builder.create();

        assertTrue("Created properties list should be empty.", createdProperties.isEmpty());
    }

    @Test
    public void getRootEntriesEmptyTest() {
        PropertyListBuilder builder = new PropertyListBuilder();
        Map<String, Object> rootEntries = builder.getRootEntries();

        assertTrue("Root entries map should be empty.", rootEntries.isEmpty());
    }

    @Test
    public void getRootEntriesWithAddedPropertyTest() {
        PropertyListBuilder builder = new PropertyListBuilder();
        Property<Object> mockProperty = Mockito.mock(Property.class);
        Object mockValue = new Object();
        Mockito.when(mockProperty.getDefaultValue()).thenReturn(mockValue);

        builder.add(mockProperty);
        Map<String, Object> rootEntries = builder.getRootEntries();

        assertTrue("Root entries map should contain added property's default value.", rootEntries.containsValue(mockValue));
    }
}
