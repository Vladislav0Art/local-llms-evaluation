        assertNotNull(builder);
    }

        assertNotNull(builderUtf8);

        WordWrap.Builder builder = WordWrap.from("example text");
        assertNotNull(builder);
    }

    public void fromUtf8Test() {

        WordWrap.Builder builder = WordWrap.fromUtf8(in);
        assertNotNull(builder);
    }
    
    @Test
    public void fromInputStreamCharsetTest() {
        ByteArrayInputStream in = new ByteArrayInputStream("example text".getBytes());
        WordWrap.Builder builder = WordWrap.from(in, StandardCharsets.UTF_8);
        assertNotNull(builder);
    }
    
    @Test
    public void fromFileCharsetTest() {
        File file = new File("src/test/resources/sample.txt");
        WordWrap.Builder builder = WordWrap.from(file, StandardCharsets.UTF_8);
        assertNotNull(builder);
    }

    @Test(expected = IOException.class)
    public void wordWrapReaderTest() throws IOException {
        BufferedReader in = new BufferedReader(new StringReader("test"));
        StringWriter out = new StringWriter();
        WordWrap.wordWrap(in, out, "\n", 80, WordWrap.DEFAULT_CHAR_WIDTH_PROVIDER, null, false, false);
    }

    @Test
    public void isWhiteSpaceTest() {
        String whiteSpace = "   ";
        assertTrue(WordWrap.isWhitespace(whiteSpace));
    }
