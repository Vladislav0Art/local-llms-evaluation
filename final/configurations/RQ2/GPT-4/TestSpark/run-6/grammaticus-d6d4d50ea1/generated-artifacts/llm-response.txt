package com.force.i18n.grammar.impl;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertTrue;
import static org.junit.Assert.assertFalse;
import static org.hamcrest.MatcherAssert.assertThat; 
import static org.hamcrest.Matchers.hasItem;

import org.junit.Test;
import org.mockito.Mockito;

import java.util.HashMap;
import java.util.Map;

import com.force.i18n.HumanLanguage;
import com.force.i18n.grammar.GrammaticalTerm;

public class GrammaticalTermMapImplTest {
    
    @Test
    public void equalsSameObjectTest() {
        GrammaticalTermMapImpl<GrammaticalTerm> termMap = new GrammaticalTermMapImpl<GrammaticalTerm>();
        assertTrue(termMap.equals(termMap));
    }

    @Test
    public void equalsNullTest() {
        GrammaticalTermMapImpl<GrammaticalTerm> termMap = new GrammaticalTermMapImpl<GrammaticalTerm>();
        assertFalse(termMap.equals(null));
    }

    @Test
    public void isSkinnyTest() {
        Map<String, GrammaticalTerm> mapTest = new HashMap<String, GrammaticalTerm>();
        GrammaticalTermMapImpl<GrammaticalTerm> termMap = new GrammaticalTermMapImpl<GrammaticalTerm>(mapTest, false);
        assertFalse(termMap.isSkinny());
    }

    @Test
    public void putTest() {
        GrammaticalTerm term = Mockito.mock(GrammaticalTerm.class);
        GrammaticalTermMapImpl<GrammaticalTerm> termMap = new GrammaticalTermMapImpl<GrammaticalTerm>();
        termMap.put("hello", term);
        assertEquals(term, termMap.get("hello"));
    }

    @Test
    public void putAllTest() {
        GrammaticalTermMapImpl<GrammaticalTerm> termMap1 = new GrammaticalTermMapImpl<GrammaticalTerm>();
        GrammaticalTermMapImpl<GrammaticalTerm> termMap2 = new GrammaticalTermMapImpl<GrammaticalTerm>();
        GrammaticalTerm term = Mockito.mock(GrammaticalTerm.class);
        termMap2.put("hello", term);
        termMap1.putAll(termMap2);
        assertEquals(term, termMap1.get("hello"));
    }

    @Test
    public void isEmptyTest() {
        GrammaticalTermMapImpl<GrammaticalTerm> termMap = new GrammaticalTermMapImpl<GrammaticalTerm>();
        assertTrue(termMap.isEmpty());
    }

    public void notIsEmptyTest() {
        GrammaticalTerm term = Mockito.mock(GrammaticalTerm.class);
        GrammaticalTermMapImpl<GrammaticalTerm> termMap = new GrammaticalTermMapImpl<GrammaticalTerm>();
        termMap.put("hello", term);
        assertFalse(termMap.isEmpty());
    }

    @Test
    public void keySetTest() {
        GrammaticalTerm term = Mockito.mock(GrammaticalTerm.class);
        GrammaticalTermMapImpl<GrammaticalTerm> termMap = new GrammaticalTermMapImpl<GrammaticalTerm>();
        termMap.put("hello", term);
        assertThat(termMap.keySet(), hasItem("hello"));
    }
}
