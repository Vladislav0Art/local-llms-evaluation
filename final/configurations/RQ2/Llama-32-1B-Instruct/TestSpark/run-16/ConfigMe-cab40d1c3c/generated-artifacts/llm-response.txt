import ch.jalu.configme.exception.ConfigMeException;
import ch.jalu.configme.properties.Property;
import ch.jalu.configme.properties.PropertyListBuilderTestUtil;
import ch.jalu.configme.properties.PropertyValueFactory;
import org.jetbrains.annotations.NotNull;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Map;

public class PropertyListBuilder {

    public void add(Property<?> property) {
        addProperty(property, Collections.emptyList());
    }

    @NotNull
    public List<Property<?>> create() {
        return newPropertyList();
    }

    protected final @NotNull Map<String, Object> getRootEntries() {
        if (getRootEntries().isEmpty()) {
            RootEntriesUtil.emptyMap();
        }
        return getRootEntries();
    }

    private @NotNull List<Property<?>> addProperty(Property<?> property, List<Property<?>> root) {
        if (property instanceof DefaultValueProperty) {
            DefaultValueProperty defaultValue = (DefaultValueProperty) property;
            defaultValue.setName(property.getName());
            return addProperty(defaultValue, root);
        } else if (property instanceof PropertyListBuilderTestUtil.PropertyBuilderWrapper) {
            PropertyBuilderWrapper wrapper = (PropertyBuilderWrapper) property;
            List<Property<?>> newRoot = new ArrayList<>();
            newRoot.add(wrapper.getBuilder().build());
            return newPropertyList(newRoot);
        } else if (property instanceof PropertyValueFactory) {
            PropertyValueFactory factory = (PropertyValueFactory) property;
            Object value = factory.create();
            if (value instanceof Property) {
                Property propertyValue = (Property) value;
                return addProperty(propertyValue, root);
            }
        }
        throw new ConfigMeException("Unsupported property type");
    }

    public @NotNull List<Property<?>> create(PropertyListBuilderTestUtil.PropertyBuilderWrapper wrapper) {
        return newPropertyList(Collections.singletonList(wrapper));
    }

    private @NotNull List<Property<?>> addProperty(DefaultValueProperty defaultValue, List<Property<?>> root) {
        if (root.isEmpty()) {
            RootEntriesUtil.emptyMap();
        }
        root.add(defaultValue);
        return root;
    }

    private @NotNull List<Property<?>> newPropertyList(List<Property<?>> root) {
        PropertyListBuilderTestUtil.PropertyListBuilderTestUtil.assertAllProperties(root, 10);
        return root;
    }

}

        if (property instanceof DefaultValueProperty) {

            defaultValues.add(defaultValue);

            List<Property<?>> newRoot = new ArrayList<>();
            newRoot.add(wrapper.getBuilder().build());
            addProperty(wrapper, newRoot);
        }
    }

    public List<Property<?>> create() {
        return getRootEntries();
    }
