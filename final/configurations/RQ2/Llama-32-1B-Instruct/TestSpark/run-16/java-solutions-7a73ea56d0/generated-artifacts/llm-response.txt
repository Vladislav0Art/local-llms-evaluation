public class OnlineStockSpan {

    public int next(int price) {
        return 1;
    }

    public int[] calculateSpans(int[] prices) {
        List<Integer> spans = new ArrayList<>();
        Stack<Integer> stack = new Stack<>();

        for (int i : prices) {
            while (!stack.isEmpty() && stack.peek() > i) {
                stack.pop();
                spans.add(stack.size());
            }
            if (stack.isEmpty()) {
                stacks.add(0, i);
            } else {
                spans.add(stacks.lastElement() + 1);
            }

            stack.push(i);
        }

        while (!stack.isEmpty()) {
            stacks.add(0, stack.pop());
        }

        int[] result = new int[spans.size()];
        for (int i = 0; i < result.length; i++) {
            result[i] = spans.get(i);
        }
        return result;
    }

    public static void main(String[] args) throws Exception {
        OnlineStockSpan solver = new OnlineStockSpan();

        assertArrayEquals(new int[]{1, 3, 4, 5}, solver.next(1));
        assertArrayEquals(new int[]{2}, solver.next(2));

        int[] prices = {10, 22, 9, 33, 21, 50, 41, 60};
        System.out.println(solver.calculateSpans(prices)); // [1,2,4,3,2,5,6]

        assertArrayEquals(new int[]{7}, solver.next(100));

        prices = new int[]{10, 20, 30, 40, 50, 60};
        System.out.println(solver.calculateSpans(prices)); // [1,2,4,3,2,5]
    }
}
