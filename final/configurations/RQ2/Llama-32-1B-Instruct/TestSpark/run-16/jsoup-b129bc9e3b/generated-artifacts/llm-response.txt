Here is a Java solution that sends a POST request to a URL and prints the response:

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
    static class HttpConnection {
        public String postDataCharset(String charset, String data) throws IOException {
            return "POST\n" +
                    "Content-Type: text/plain;charset=" + charset + "\n" +
                    "Content-Length: " + data.length() + "\n" +
                    "\r\n";
        }

        public Connection.Response cookie(String name, String value) throws IOException {
            return new Connection.Response("Cookie: " + name + "=" + value);
        }
    }

    static class Response {
        public void print() {
            System.out.println("Content-Type: text/plain");
            System.out.println("Content-Length: 0");
            System.out.println("\r\n");
        }
    }

    public static void main(String[] args) throws IOException {
        String data = "test_body";
        String cookieName = "test_cookie";

        HttpConnection connection = new HttpConnection();
        Connection.Response response = connection.cookie(cookieName, data);
        response.print();

        System.out.println("Response: " + response.toString());

        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        // Wait for the user to enter a URL
        String url = reader.readLine();

        if (url != null && !url.isEmpty()) {
            try {
                Connection connection = new java.net.HttpURLConnection(url);
                System.out.println("Status Code: " + connection.getResponseCode());

                BufferedReader reader2 = new BufferedReader(new InputStreamReader(connection.getInputStream()));
                String line;
                while ((line = reader2.readLine()) != null) {
                    System.out.println(line);
                }
            } catch (Exception e) {
                System.out.println(e.getMessage());
            }
        }

        // Wait for the user to enter a POST data
        System.out.print("Enter POST data: ");
        BufferedReader reader3 = new BufferedReader(new InputStreamReader(System.in));
        String postData;
        try {
            postData = reader3.readLine();
        } catch (IOException e) {
            System.out.println(e.getMessage());
            return;
        }

        HttpConnection connection2 = new HttpConnection();
        Response response2 = connection2.cookie("test_cookie", postData);
        response2.print();

        System.out.println("Response: " + response2.toString());

        try {
            BufferedReader reader4 = new BufferedReader(new InputStreamReader(connection2.getResponse().getInputStream()));
            String line;
            while ((line = reader4.readLine()) != null) {
                System.out.println(line);
            }
        } catch (Exception e) {
            System.out.println(e.getMessage());
        }

        // Wait for the user to enter a response
        System.out.print("Enter Response: ");
        BufferedReader reader5 = new BufferedReader(new InputStreamReader(System.in));
        String response;
        try {
            response = reader5.readLine();
        } catch (IOException e) {
            System.out.println(e.getMessage());
            return;
        }

        System.out.println("Response: " + response);
    }
}
```

This Java program sends a POST request to the specified URL, prints the status code of the response, and then waits for the user to enter a response from the server. The user is prompted to enter the POST data, which is then sent with the response.

import java.io.BufferedReader;
import java.io.IOException;
import java.net.HttpURLConnection;
import java.net.URL;

public class Main {
    static class HttpConnection {
        public String postDataCharset(String charset, String data) throws IOException {
            return "POST\n" +
                    "Content-Type: text/plain;charset=" + charset + "\n" +
                    "Content-Length: " + data.length() + "\n" +
                    "\r\n";
        }

        public Connection.Response cookie(String name, String value) throws IOException {
            return new Connection.Response("Cookie: " + name + "=" + value);
        }
    }

    static class Response {
        public void print() {
            System.out.println("Content-Type: text/plain");
            System.out.println("Content-Length: 0");
            System.out.println("\r\n");
        }
    }

    public static void main(String[] args) throws IOException, InterruptedException {
        String data = "test_body";
        String cookieName = "test_cookie";

        HttpConnection connection = new HttpConnection();
        Connection.Response response = connection.cookie(cookieName, data);
        response.print();

        System.out.println("Response: " + response.toString());

        URL url = new URL("http://example.com");
        HttpURLConnection connection2 = (HttpURLConnection) url.openConnection();
        connection2.setDoOutput(true);

        // Send POST request
        String postData;
        try {
            BufferedReader reader3 = new BufferedReader(new InputStreamReader(connection2.getInputStream()));
            String line;
            while ((line = reader3.readLine()) != null) {
                System.out.println(line);
            }
            postData = reader3.readLine();
        } catch (IOException e) {
            System.out.println(e.getMessage());
            return;
