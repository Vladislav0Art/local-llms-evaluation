public class GeneratedTest {
    public static String ArraysToString(int[] nums) {
        StringBuilder sb = new StringBuilder();
        for (int num : nums) {
            sb.append(num).append(",");
        }
        return sb.toString().trim();
    }

    public static void main(String[] args) {
        System.out.println(ArraysToString(new int[]{1, 2, 3})); // Expected output: "1,2,3"
        System.out.println(ArraysToString(new int[]{4, 5, 6, 7})); // Expected output: "4,5,6,7"

        int[] nums = {1, 2, 3};
        System.out.println(ArraysToString(nums)); // Expected output: "1,2,3"
    }

    public static String solution(int[] nums) {
        return ArraysToString(nums);
    }
}
