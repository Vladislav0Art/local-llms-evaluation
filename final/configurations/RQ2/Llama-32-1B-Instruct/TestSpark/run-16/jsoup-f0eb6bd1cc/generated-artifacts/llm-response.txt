        public AppendKeyValRequest(Map<String, Object> kv) {

        }

        @FunctionalInterface
        public interface Method {
            String summary();

        public String appendKeyVal(UrlBuilder.AppendKeyValRequest request) throws Exception {

            if (request.getKv() != null && !request.getS().isEmpty()) {

                return sb.toString();
            } else {
                throw new Exception("Invalid request");
            }
        }

    }

    }

public class TestUtils {
    public static String joinPath(String path1, String path2) {
        return path1 + "." + path2;
    }

    public static String normalizeUrl(String url) {
        // Remove trailing slash if present
        if (url.endsWith("/")) {
            return url.substring(0, url.length() - 1);
        }
        return url;
    }

    public static String normalizePath(String path) {
        // Split the path into components and join them with a forward slash
        String[] components = path.split("/");
        StringBuilder normalizedPath = new StringBuilder();
        for (String component : components) {
            if (!component.isEmpty()) {
                normalizedPath.append("/").append(component);
            }
        }
        return normalizedPath.toString();
    }

    public static void main(String[] args) {
        testJoinPath("test", "path");
        testNormalizeUrl("https://example.com/test");
        testNormalizePath("/home/user");
    }

    private static void testJoinPath(String path1, String path2) {
        System.out.println(joinPath(path1, path2));
    }

    private static void testNormalizeUrl(String url) {
        System.out.println(normalizeUrl(url));
    }

    private static void testNormalizePath(String path) {
        System.out.println(normalizePath(path));
    }
}
