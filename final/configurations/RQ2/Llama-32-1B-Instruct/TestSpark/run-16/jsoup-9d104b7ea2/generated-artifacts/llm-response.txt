```java
        }
    }

    public void outerHtmlHead() throws Exception {

        String expected = "<text>text</text>";

        assertEquals(expected, builder.toString());

    public void outerHtmlTail() throws Exception {

        assertEquals(expected, builder.toString());
    }

        Elements elements = document.getElementsByTagName("text");
        StringBuilder builder = new StringBuilder();
        for (Element element : elements) {
            builder.append(element.getText()).append("\n");
        }
        assertEquals("<text>Hello, world!</text>", builder.toString());
```

```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class TextNodeTest {

    public static void main(String[] args) {
        String html = "<html><body>Hello, world!</body></html>";
        Document doc = Jsoup.parse(html);
        
        System.out.println(doc.text());  // prints: Hello, world!
        System.out.println(doc.text("Hello, world!"));  // prints: Hello, world!

        String text = "This is a test";
        Element element = doc.selectFirst("[text]");
        if (element != null) {
            System.out.println(element.text());  // prints: This
            System.out.println(element.attr("name"));  // prints: text
        } else {
            System.out.println("Element not found");
        }

        String text2 = "Hello, world!";
        Element element2 = doc.selectFirst("[text='" + text2 + "']");
        if (element2 != null) {
            System.out.println(element2.text());  // prints: Hello, world!
        } else {
            System.out.println("Element not found");
        }

        int count = 0;
        for (int i = 1; i <= elementCount(); i++) {
            Element element3 = doc.selectFirst("[text='" + i + "']");
            if (element3 != null) {
                System.out.println(element3.text());  // prints: , 
                count++;
            }
        }

        Element head = doc.selectFirst("head");
        String expectedHead = "<style>body { text-align: center; }</style><title>This is a test</title>";
        Document newDoc = Jsoup.parse(expectedHead);
        System.out.println(doc.html());
        System.out.println(newDoc.text());  // prints: This is a test
    }

    public static int elementCount() {
        return doc.select("html, head").size();
    }
}
```

This code uses the `Jsoup` library to parse an HTML document and manipulate its text content. The `text()` method returns the entire text of the page, while `text("Hello, world!")` extracts a specific piece of text.

The code also demonstrates how to select elements by attribute (e.g., `[text='Hello, world!']`) or selector (e.g., `[text='"Hello, world!"']`).
