import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class DocumentTools {

    public static Element getBody(Element document) {
        return document.body();
    }

    public static String getTitle(Document document) {
        return document.title().trim();
    }

    public static List<Element> getForms(Element document) {
        return (List<Element>) document.select("form");
    }

    public static String getTextElements(Document document, String nodeName) {
        Element textElement = document.selectFirst(nodeName);
        if (textElement != null) {
            return textElement.text();
        } else {
            return "";
        }
    }

    public static String getCharset(Document document) {
        return document.charset().name();
    }

    public static String getNodeType(Document document) {
        Element body = document.body();

            return "DocumentType.BODY";
        } else {
            return "";
        }
    }
}

    public static void main(String[] args) {

Here's a complete Java program that demonstrates how to use the `DocumentTools` class to extract various information from a HTML document.

```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class DocumentTools {

    public static Element getBody(Element document) {
        return document.body();
    }

    public static String getTitle(Document document) {
        return document.title().trim();
    }

    public static List<Element> getForms(Element document) {
        return (List<Element>) document.select("form");
    }

    public static String getTextElements(Document document, String nodeName) {
        Element textElement = document.selectFirst(nodeName);
        if (textElement != null) {
            return textElement.text();
        } else {
            return "";
        }
    }

    public static String getCharset(Document document) {
        return document.charset().name();
    }

    public static String getNodeType(Document document) {
        Element body = document.body();
        if (body != null && body.tagName().equals("BODY")) {
            return "DocumentType.BODY";
        } else {
            return "";
        }
    }

    public static void main(String[] args) {
        Document document = Jsoup.connect("https://www.example.com")
                .get()
                .body();

        System.out.println("Title: " + DocumentTools.getTitle(document));
        System.out.println("Body: " + DocumentTools.getBody(document).text());

        List<Element> forms = DocumentTools.getForms(document);
        System.out.println("\nForm Tags:");
        for (Element form : forms) {
            System.out.println(form.tagName());
        }

        String textElements = DocumentTools.getTextElements(document, "h1");
        if (!textElements.isEmpty()) {
            System.out.println("Text Elements:\n" + textElements);
        } else {
            System.out.println("No text elements found.");
        }

        System.out.println("\nCharset: " + DocumentTools.getCharset(document));
        System.out.println("Node Type: " + DocumentTools.getNodeType(document));

        String title = DocumentTools.getTitle(document);
        System.out.println("\nTitle: " + title);

        String charset = DocumentTools.getCharset(document);
        System.out.println("Charset: " + charset);

        String body = DocumentTools.getBody(document).text();
        System.out.println("\nBody:\n" + body);

        List<Element> formsWithText = DocumentTools.getForms(document).filter(form -> form.tagName().contains("input") || form.tagName().contains("textarea"));
        if (!formsWithText.isEmpty()) {
            System.out.println("Form Tags with Text Elements:");
            for (Element form : formsWithText) {
                System.out.println(form.tagName() + ": " + form.text());
            }
        } else {
            System.out.println("No form tags with text elements found.");
        }

        String text = DocumentTools.getTextElements(document, "h2");
        if (!text.isEmpty()) {
            System.out.println("Text Elements:\n" + text);
        } else {
            System.out.println("No text elements found.");
        }
    }
}
```

This program creates a simple HTML document using the `Jsoup` library. It then uses the `DocumentTools` class to extract various information from this document, including the title, body text, form tags with text elements, and charset. The results are printed to the console for easy viewing.
