            throw new RuntimeException(e);

        return document;
    }

    @Test

        String documentString = convert(doc).outerHtml();
        System.out.println(documentString);
    }

    public Document fromJsoup(Element element) {
        return convert(elementDocument);
    }

    private static org.jsoup.nodes.Document elementDocument;

    @Test
    public void testFromJsoup() {

        Element body = new org.jsoup.nodes.Element("body");
