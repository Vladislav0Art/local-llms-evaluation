import javax.swing.*;
import java.awt.*;
import java.util.ArrayList;
import java.util.List;

public class Main {
    public static void main(String[] args) {
        // Create an empty document
        Document doc = new Document();

        // Create a root element
        Element root = doc.createElement("html");
        doc.addElement(root);

        // Create a head element
        Element head = doc.createElement("head");
        root.appendChild(head);
        head.appendChild(doc.createElement("title"));

        // Create a body element
        Element body = doc.createElement("body");
        root.appendChild(body);

        // Create a main element and add it to the body
        Element main = doc.createElement("main");
        body.appendChild(main);

        // Create some text elements
        TextImpl text1 = new TextImpl("This is the first paragraph of our HTML document.");
        Element text2 = doc.createElement("p");
        text2.appendChild(text1);
        main.appendChild(text2);

        TextImpl text3 = new TextImpl("This is the second paragraph of our HTML document.");
        Element text4 = doc.createElement("p");
        text4.appendChild(text3);
        main.appendChild(text4);

        // Create a form element
        Element form = doc.createElement("form");
        root.appendChild(form);

        // Create some input elements
        TextImpl inputText1 = new TextImpl("Enter your name: ");
        InputFieldImpl input1 = new InputFieldImpl();
        Element input2 = doc.createElement("input");
        input2.appendChild(input1);
        form.appendChild(input2);

        TextImpl label1 = new TextImpl("Name:");
        LabelImpl label2 = new LabelImpl();
        Element label3 = doc.createElement(label1.tag);
        label3.appendChild(label2);
        main.appendChild(label3);

        // Create a button element
        ButtonImpl button = new ButtonImpl("Click me!");
        Element buttonElement = doc.createElement("button");
        buttonElement.appendChild(button);
        form.appendChild(buttonElement);

        // Add the elements to the document
        doc.addElement(root);
    }
}

class InputFieldImpl extends JTextField {
    public InputFieldImpl() {
        super();
    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        Graphics2D g2d = (Graphics2D) g;
        // set the drawing color and style
        g2d.setColor(Color.BLACK);
        g2d.setStroke(new BasicStroke(5));
        g2d.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

        // draw the input field
        int x = 10;
        int y = 100;
        g2d.fillOval(x, y + 40, 280, 20);
    }
}

class LabelImpl extends JLabel {
    public LabelImpl() {
        super();
    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        Graphics2D g2d = (Graphics2D) g;
        // set the drawing color and style
        g2d.setColor(Color.BLACK);
        g2d.setStroke(new BasicStroke(5));
        g2d.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

        // draw the label text
        int x = 10;
        int y = 120;
        g2d.setFont(new Font("Arial", Font.BOLD, 14));
        g2d.drawString(this.getText(), x + (this.getWidth() - this.getCharsIn()) / 2, y);
    }
}
