Here's a high-level overview of how to implement a word wrapping system:

1.  **Split the input string into words**: Split the input string into individual words using whitespace as the delimiter.
2.  **Calculate the maximum allowed width for each line**: Calculate the total width required for all lines and then divide it by the number of words in the last line to determine the maximum allowed width per line.
3.  **Build the output with word wrapping**: Use a loop to build the output string, adding spaces as necessary to wrap the input text into lines while ensuring each line does not exceed the calculated maximum width.

Here's an example implementation:

```java
public class WordWrap {
    public static void main(String[] args) {
        String input = "Lorem ipsum dolor sit amet, consectetur adipiscing elit.";
        int maxWidth = 20;

        WordWrap wordWrap = new WordWrap();
        StringBuilder output = new StringBuilder();

        // Split the input string into words
        String[] words = input.split("\\s+");

        // Calculate the maximum allowed width for each line
        int maxLineWidth = getMaxLineWidth(words, maxWidth);

        // Build the output with word wrapping
        buildOutput(maxLineWidth, words, output);

        System.out.println(output.toString());
    }

    /**
     * Calculates the maximum allowed width for each line.
     *
     * @param words The input words.
     * @param maxWidth The maximum allowed width for each line.
     * @return The maximum allowed width per line.
     */
    private static int getMaxLineWidth(String[] words, int maxWidth) {
        if (words.length == 0) {
            return 0;
        }

        // Calculate the length of the first word
        int lineWidth = getLineWidth(words[0]);

        for (int i = 1; i < words.length; i++) {
            lineWidth = Math.min(lineWidth + 2, maxWidth - 4); // add spaces and subtract 4 for the delimiter
        }

        return lineWidth;
    }

    /**
     * Calculates the length of a single word.
     *
     * @param word The input word.
     * @return The length of the word.
     */
    private static int getLineWidth(String word) {
        // Calculate the length of the word, including spaces
        int length = 0;
        for (char c : word.toCharArray()) {
            if (!Character.isWhitespace(c)) {
                length++;
            }
        }

        return length + 1; // add one for the delimiter space
    }

    /**
     * Builds the output with word wrapping.
     *
     * @param maxWidth The maximum allowed width per line.
     * @param words The input words.
     * @param output The output string builder.
     */
    private static void buildOutput(int maxWidth, String[] words, StringBuilder output) {
        if (output.length() > 0 && maxWidth <= getMaxLineWidth(words, maxWidth)) {
            // Add the first word to the output
            output.append(words[0]);
            for (int i = 1; i < words.length; i++) {
                int lineWidth = getMaxLineWidth(words, maxWidth);
                while (words[i].length() > lineWidth) {
                    output.append(" ").append(words[i].substring(0, lineWidth));
                    words[i] = words[i].substring(lineWidth);
                }
                output.append(" ");
                output.append(words[i]);
            }

            // Add spaces to fill the line
            for (int i = 1; i < words.length; i++) {
                while (words[i].length() > maxWidth) {
                    output.append(" ").append(words[i].substring(0, maxWidth));
                    words[i] = words[i].substring(maxWidth);
                }
            }

            // Append the last word to the output
            if (!output.toString().isEmpty()) {
                output.append(" ");
                output.append(words[words.length - 1]);
            } else {
                output.append(words[words.length - 1]);
            }

            System.out.println(output.toString());
        }
    }
}
```
