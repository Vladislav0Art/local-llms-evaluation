import org.junit.Test;
import static org.junit.Assert.*;

public class Main {

    @Test
    public void testParse() {
        String input = "<a>";
        String expected = "a";
        assertNotEquals(input, expected);
        assertTrue(Main.parse(input));
    }

    @Test
    public void testParseWithMultipleTags() {
        String input = "<a><b>Hello</b> World!";
        String expected = "Hello World!";
        assertNotEquals(input, expected);
        assertTrue(Main.parse(input));
    }

    @Test
    public void testEmptyInput() {
        String input = "";
        String expected = null;
        assertNotEquals(input, expected);
        assertTrue(Main.parse(input) == null);
    }

    @Test
    public void testParseWithSingleTag() {
        String input = "<a>";
        String expected = "a";
        assertNotEquals(input, expected);
        assertTrue(Main.parse(input));
    }

    @Test
    public void testMultipleTags() {
        String input = "<a><b>Hello</b> World!";
        String expected = "Hello World!";
        assertEquals(expected, Main.parse(input));
    }
}
