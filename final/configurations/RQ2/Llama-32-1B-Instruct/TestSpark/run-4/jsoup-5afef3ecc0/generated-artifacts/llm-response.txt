import org.jsoup.Jsoup;

public class TestURLBuilder {

    public static String urlBuilder(String str) {
        Document doc = Jsoup.parse(str);
        return doc.toString();
    }

    public static boolean isAbsoluteUrl(String url) {
        try {
            URL parsedUrl = new URL(url);
            return !parsedUrl.getProtocol().isEmpty();
        } catch (Exception e) {
            return false;
        }
    }

    public static void test1() {
        String url = "https://example.com/path?query=value";

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class TestURLBuilder {

    public static String urlBuilder(String str) {
        return Jsoup.parse(str).toString();
    }

    public static boolean isAbsoluteUrl(String url) {
        try {
            Document doc = Jsoup.parse(url);
            // Check if the URL has a protocol (http/https)
            if (!doc.select("meta[name=viewport]").isEmpty() || !doc.select("script[src]").isEmpty()) {
                return true;
            }
            return false;
        } catch (Exception e) {
            return false;
        }
    }

    public static void main(String[] args) {
        String url = "https://example.com/path?query=value";
        Document doc = Jsoup.parse(url);
        System.out.println(isAbsoluteUrl(urlBuilder(doc.toString())));
        System.out.println(urlBuilder("http://example.com/path"));
        try {
            urlBuilder("https://example.com/absolute/path?query=value");
        } catch (Exception e) {
            System.out.println(e.getMessage());
        }

        String url2 = "https://example.com/path";
        Document doc2 = Jsoup.parse(url2);
        System.out.println(isAbsoluteUrl(urlBuilder(doc2.toString())));
    }
}
