import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class Main {

    public static String test5(String input) {
        Document doc = Jsoup.parse(input);
        Element element = doc.select("h1").first(); // selects the first h1 tag
        System.out.println(element.text());
        return null;
    }

    public static void test6() {
        document = Jsoup.parse("<html><body>This is a <b>test</b> page.</body></html>");
        Element element = document.select("b");
        for (Element e : element) {
            System.out.println(e.text());
        }
    }

    public static void main(String[] args) {
        String input1 = "<html><body>This is a <b>test</b> page.</body></html>";
        System.out.println(test5(input1));
        test6();
    }
}
