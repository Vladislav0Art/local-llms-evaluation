import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertTrue;

public class TextNodeTest {

    @Test
    void testnodeName() {
        TextNode node = new TextNode("Hello World");
        assertEquals("Hello World", node.nodeName());
    }

    @Test
    void testText() {
        TextNode node = new TextNode("Hello World");
        assertEquals("Hello World", node.text());
    }

    @Test
    void testText_createFromEncoded() {
        String encodedText = "Hello World";
        TextNode createdNode = TextNode.createFromEncoded(encodedText);
        assertEquals(encodedText, createdNode.text());
    }

    @Test
    void testNormaliseWhitespace() {
        String text = "   Hello   World  ";
        String normalisedText = normalizeWhitespace(text);
        assertEquals("Hello World", normalisedText);
    }

    @Test
    void testStripLeadingWhitespace() {
        String text = "   Hello   World  ";
        StringBuilder sb = new StringBuilder();
        stripLeadingWhitespace(sb, text);
        assertEquals("Hello World", sb.toString());
    }

    @Test
    void testLastCharIsWhitespace() {
        StringBuilder sb = new StringBuilder();
        lastCharIsWhitespace(sb);
        assertTrue(sb.length() > 0 && sb.charAt(sb.length() - 1) == ' ');
    }

    public static String stripLeadingWhitespace(StringBuilder sb, String text) {
        int index = 0;
        while (index < text.length() && !Character.isWhitespace(text.charAt(index))) {
            ++index;
        }
        if (index > 0) {
            return text.substring(0, index);
        } else {
            return text;
        }
    }

    public static boolean lastCharIsWhitespace(StringBuilder sb) {
        int index = sb.length() - 1;
        while (index >= 0 && !Character.isWhitespace(sb.charAt(index))) {
            --index;
        }
        return index < 0;
    }

    private static String normalizeWhitespace(String text) {
        StringBuilder sb = new StringBuilder();
        for (char c : text.toCharArray()) {
            if (!Character.isWhitespace(c)) {
                sb.append(c);
            } else {
                break;
            }
        }
        return sb.toString();
    }

}

import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class TestStringBuilder {

    public static void testnodeName() {
        Document doc = new Document();
        Element node = doc.createElement("test");
        node.setAttribute("node-name", "Hello World");
        assertEquals("test", node.nodeName());
    }

    public static void testText() {
        Document doc = new Document();
        Element node = doc.createElement("test");
        node.setTextContent("Hello World");
        assertEquals("Hello World", node.text());
    }

    public static void testText_createFromEncoded() {
        String encodedText = "Hello World";
        TestStringBuilder.createTestNode(encodedText);
        System.out.println(encodedText);
    }

    public static void normalizeWhitespace(String text) {
        StringBuilder sb = new StringBuilder();
        for (char c : text.toCharArray()) {
            if (c == ' ') {
                sb.append(c);
            }
        }
        assertEquals(text, sb.toString());
    }

    public static void stripLeadingWhitespace(StringBuilder sb, String text) {
        int i = 0;
        while (i < text.length() && sb.charAt(i) != ' ') {
            i++;
        }
        if (i > 0) {
            for (; i < text.length(); i++) {
                sb.deleteCharAt(i);
            }
        }
    }

    public static boolean lastCharIsWhitespace(StringBuilder sb) {
        return sb.charAt(sb.length() - 1) == ' ';
    }

    public static void createTestNode(String encodedText) {
        TestStringBuilder.testText();
        String text = "Hello World";
        Document doc = new Document();
        Element node = doc.createElement("test");
        node.setAttribute("node-text", text);
        assertEquals(encodedText, node.getAttribute("node-text"));
        testText();
    }
}
