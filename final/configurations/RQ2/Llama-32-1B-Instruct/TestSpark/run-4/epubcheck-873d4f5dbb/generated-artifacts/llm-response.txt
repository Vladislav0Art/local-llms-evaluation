```java
import java.util.Set;

public class Book {
    private String title;
    private String author;
    private Set<String> formats = Set.of("audio/mpeg", "ebook");

    public Book(String title, String author) {
        this.title = title;
        this.author = author;
    }

    public String getTitle() {
        return title;
    }

    public void setTitle(String title) {
        this.title = title;
    }

    public String getAuthor() {
        return author;
    }

    public void setAuthor(String author) {
        this.author = author;
    }

    public Set<String> getFormats() {
        return formats;
    }
}

public class Bookshelf {
    private Map<String, Book> books;

    public Bookshelf() {
        this.books = new HashMap<>();
    }

    public void addBook(Book book) {
        if (books.containsKey(book.getTitle())) {
            System.out.println("Book already exists in the shelf");
        } else {
            books.put(book.getTitle(), book);
        }
    }

    public Book getBook(String title) {
        return books.get(title);
    }

    public boolean removeBook(String title) {
        return books.remove(title) != null;
    }

    public void printBooks() {
        for (Map.Entry<String, Book> entry : books.entrySet()) {
            System.out.println(entry.getKey() + ": " + entry.getValue().getTitle());
        }
    }
}

public class Main {
    public static void main(String[] args) {
        Bookshelf bookshelf = new Bookshelf();
        
        // Create some books

        // Add the books to the shelf
        bookshelf.addBook(audiobook1);
        bookshelf.addBook(audiobook2);

        // Print all books on the shelf
        bookshelf.printBooks();

        // Try to add an existing book

            bookshelf.addBook(audiobook1);  // This should print "Book already exists in the shelf"
        } catch (Exception e) {
            System.out.println(e.getMessage());

        // Remove a book
```

```java
import java.util.Map;

        return title;

    public void setTitle(String title) {

        return price;
    }

        this.price = price;
    }

    public boolean isAvailable() {
        return isAvailable;
    }

    public void setAvailability(boolean availability) {
        isAvailable = availability;
    }
}

class Bookshelf {
    private Map<String, Book> books;

    public Bookshelf() {
        this.books = new HashMap<>();
    }

    public void addBook(Book book) {
        if (books.containsKey(book.getTitle())) {
            System.out.println("Book already exists in the shelf");
        } else {
            books.put(book.getTitle(), book);
            System.out.println("Book added successfully");
        }
    }

    public Book getBook(String title) {
        return books.get(title);
    }

    public boolean removeBook(String title) {
        if (books.containsKey(title)) {
            books.remove(title);
            System.out.println("Book removed successfully");
            return true;
        } else {
            System.out.println("No book found with this title");
            return false;
        }
    }

    public void printBooks() {
        for (Map.Entry<String, Book> entry : books.entrySet()) {
            System.out.println(entry.getKey() + ": " + entry.getValue().getTitle());
        }
    }

        // Create a bookshelf
        Bookshelf bookshelf = new Bookshelf();

        // Print all books on the shelf
        System.out.println("\nAll Books:");

            bookshelf.addBook(new Book("Pride and Prejudice", "Jane Austen", 8.99));

            bookshelf.printBooks();

    }
```

Here's a Java program that simulates a simple text-based book store management system:

```java
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

class Book {
    private String title;
    private double price;
    private int quantity;

    public Book(String title, double price) {
        this.title = title;
        this.price = price;
        this.quantity = 0;
    }

    public void setQuantity(int quantity) {
        if (quantity <= 0) {
            System.out.println("Quantity must be greater than zero.");
        } else {
            this.quantity = quantity;
        }
    }

    public String getTitle() {
        return title;
    }

    public double getPrice() {
        return price;
    }

    public int getQuantity() {
        return quantity;
    }
}

class BookStore {
    private Map<String, Book> books;

    public BookStore() {
        this.books = new HashMap<>();
    }

    public void addBook(Book book) {
        if (books.containsKey(book.getTitle())) {
            System.out.println("Book already exists in the store.");
        } else {
            books.put(book.getTitle(), book);
            System.out.println("Book added successfully");
        }
    }

    public Book getBook(String title) {
        return books.get(title);
    }

    public void removeBook(String title) {
        if (books.containsKey(title)) {
            books.remove(title);
            System.out.println("Book removed successfully");
            return;
        } else {
            System.out.println("No book found with this title.");
            return;
        }
    }

    public void printBooks() {
        for (Map.Entry<String, Book> entry : books.entrySet()) {
            System.out.println(entry.getKey() + ": " + entry.getValue().getTitle());
        }
    }

    public static Scanner scanner = new Scanner(System.in);

    public static void main(String[] args) {
        BookStore bookstore = new BookStore();
        boolean runProgram = true;

        while (runProgram) {
            System.out.println("\nBook Store Management System");
            System.out.println("1. Add Book");
            System.out.println("2. Remove Book");
            System.out.println("3. Print All Books");
            System.out.println("4. Exit");

            System.out.print("Choose an option: ");
            int choice = scanner.nextInt();

            switch (choice) {
                case 1:
                    addBook(bookstore);
                    break;
                case 2:
                    removeBook(bookstore);
                    break;
                case 3:
                    printBooks(bookstore);
                    break;
                case 4:
                    runProgram = false;
                    System.out.println("Exiting the program.");
                    break;
                default:
                    System.out.println("Invalid option. Please choose a valid option.");
            }
        }

        scanner.close();
    }

    private static void addBook(BookStore bookstore) {
        System.out.print("\nEnter title: ");
        String title = scanner.next();

        double price;
        while (true) {
            System.out.print("Enter price (or 'done' to finish): ");
            if (scanner.nextLine().equals("done")) {
                break;
            } else {
                try {
                    price = Double.parseDouble(scanner.nextLine());
                    bookstore.addBook(new Book(title, price));
                    break;
                } catch (NumberFormatException e) {
                    System.out.println("Invalid price. Please enter a number.");
                }
            }
        }

        System.out.print("\nEnter quantity: ");
        int quantity;
        while (true) {
            try {
                quantity = Integer.parseInt(scanner.nextLine());
                bookstore.setQuantity(quantity);
                break;
            } catch (NumberFormatException e) {
                System.out.println("Invalid quantity. Please enter a number.");
            }
        }
    }

    private static void removeBook(BookStore bookstore) {
        System.out.print("\nEnter title of the book to remove: ");
        String title = scanner.next();

        if (bookstore.getBook(title).getQuantity() > 0) {
            bookstore.removeBook(title);
            System.out.println("Book removed successfully");
        } else {
            System.out.println("No book found with this title.");
        }
    }

    private static void printBooks(BookStore bookstore) {
        System.out.println("\nAll Books:");
        bookstore.printBooks();
    }
}
```

This program provides a simple text-based interface for managing books in a bookstore. Users can add, remove, and print books. The program also checks for invalid input and handles exceptions accordingly.

Here's a Java program that simulates a simple online shopping system:

```java
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

class Product {
    private String id;
    private String name;
    private double price;

    public Product(String id, String name, double price) {
        this.id = id;
        this.name = name;
        this.price = price;
    }

    public String getId() {
        return id;
    }

    public void setId(String id) {
        this.id = id;
    }

    public String getName() {
        return name;
    }

    public double getPrice() {
        return price;
    }
}

class Customer {
    private String id;
    private String name;
    private Map<Product, Integer> purchases;

    public Customer(String id, String name) {
        this.id = id;
        this.name = name;
        this.purchases = new HashMap<>();
    }

    public void addPurchase(Product product, int quantity) {
        if (purchases.containsKey(product.getId())) {
            purchases.put(product.getId(), purchases.get(product.getId()) + quantity);
        } else {
            purchases.put(product.getId(), quantity);
        }
    }

    public int getPurchaseCount(String id) {
        return purchases.getOrDefault(id, 0);
    }
}

class Shopper {
    private String id;
    private Customer customer;

    public Shopper(String id) {
        this.id = id;
    }

    public void buyProduct(Product product) {
        if (customer != null) {
            int quantity = Integer.parseInt(scanner.nextLine());
            if (!product.get Purchases().containsKey(product.getId())) {
                System.out.println("Quantity already in stock.");
            } else {
                customer.addPurchase(product, quantity);
                System.out.println("You purchased " + quantity + " of " + product.getName() + ". Your new balance is $" + (customer.purchases.get(product.getId()) - quantity) + ".");
            }
        } else {
            System.out.println("No account created yet. Create an account to purchase products.");
        }
    }

    public String getName() {
        return customer.getName();
    }
}

public class Main {
    public static void main(String[] args) {
        Map<Product, Integer> products = new HashMap<>();
        products.put(new Product("1", "Product A", 10.99), 5);
        products.put(new Product("2", "Product B", 9.99), 3);

        Scanner scanner = new Scanner(System.in);

        Shopper shopper = new Shopper("1");
        Customer customer = new Customer("2");

        while (true) {
            System.out.println("\nOnline Shopping System");
            System.out.println("1. Buy Product");
            System.out.println("2. Check Balance");
            System.out.println("3. Exit");

            System.out.print("Choose an option: ");
            int choice = scanner.nextInt();

            switch (choice) {
                case 1:
                    buyProduct(shopper, customer);
                    break;
                case 2:
                    printBalance(customer);
                    break;
                case 3:
                    System.exit(0);
                default:
                    System.out.println("Invalid option. Please choose a valid option.");
            }
        }

        scanner.close();
    }

    private static void buyProduct(Shopper shopper, Customer customer) {
        while (true) {
            System.out.print("\nEnter product ID: ");
            String id = scanner.next();

            if (products.containsKey(new Product(id, "", 0))) {
                for (int i : products.get(id)) {
                    customer.addPurchase(new Product(id, "", 0), i);
                }
                break;
            } else {
                System.out.println("Product not found. Please add the product first.");
            }
        }

        System.out.print("\nEnter quantity: ");
        int quantity = scanner.nextInt();
        scanner.nextLine();

        for (int i = 1; i <= quantity; i++) {
            customer.buyProduct(new Product(id, "", 0));
        }
    }

    private static void printBalance(Customer customer) {
        if (customer != null && !customer.purchases.isEmpty()) {
            System.out.println("Your balance is $" + String.join(" + ", customer.purchases.values()));
        } else {
            System.out.println("No purchases.");
        }
    }
}
```

This program simulates an online shopping system where customers can purchase products. The program uses a simple menu-based interface to allow users to buy products and check their balances.
