import ch.jalu.configme.configurationdata CommentsConfiguration;
import ch.jalu.configme.configurationdata.CommentsConfigurationBuilder;
import ch.jalu.configme.settings.SettingsHolder;
import org.jetbrains.annotations.NotNull;
import org.junit.jupiter.api.Test;

public class CommentsConfigurationTest {

    private SettingsHolder settings = new SettingsHolder();
    private CommentsConfiguration configuration;

    @Test
    public void commentsConfigurationSetsNewCommentLines(@NotNull String path, @NotNull String... commentLines) {
        // Arrange
        StringBuilder sb = new StringBuilder();
        for (String line : commentLines) {
            sb.append(line).append("\n");
        }

        // Act
        configuration.setComment(path, sb.toString());

        // Assert
        assertEquals(commentLines.length, configuration.getAllComments().size());
    }

    @Test
    public void commentsConfigurationSetsNewLineOnlyCommentLines(@NotNull String path, @NotNull String... commentLines) {
        // Arrange
        StringBuilder sb = new StringBuilder();
        for (String line : commentLines) {
            sb.append(line).append("\n");
        }
        for (int i = 0; i < commentLines.length - 1; i++) {
            sb.append("\n");
        }

        // Act
        configuration.setComment(path, commentLines);

        // Assert
        assertEquals(2, configuration.getAllComments().size());
    }

    @Test
    public void commentsConfigurationSetsExistingCommentLine(@NotNull String path, @NotNull String oldComment) {
        // Arrange
        StringBuilder sb = new StringBuilder();
        sb.append(oldComment).append("\n");

        // Act
        configuration.setComment(path, oldComment);

        // Assert
        assertEquals(1, configuration.getAllComments().size());
    }

    @Test
    public void commentsConfigurationSetsEmptyStringCommentLine(@NotNull String path) {
        // Arrange
        StringBuilder sb = new StringBuilder();

        // Act
        configuration.setComment(path, "");

        // Assert
        assertEquals(0, configuration.getAllComments().size());
    }
}

        configuration.setAllComments(sb.toString());

            sb.append("\n");
        }
        
        CommentsConfiguration configuration = getCommentsConfiguration(path);
        configuration.setAllComments(sb.toString());
    }

    @Test
    public void commentsSetsExistingCommentLine(@NotNull String path, @NotNull String oldComment) {
        StringBuilder sb = new StringBuilder();
        sb.append(oldComment).append("\n");

        CommentsConfiguration configuration = getCommentsConfiguration(path);
        configuration.setAllComments(sb.toString());
    }

    @Test
    public void commentsSetsEmptyStringCommentLine(@NotNull String path) {

    }
}
