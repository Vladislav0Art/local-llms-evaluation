import static org.hamcrest.MatcherAssert.assertThat;
import static org.hamcrest.Matchers.is;

public class SolarPositionTest {

    public void testSolarPosition() {
        double northernHeemspheric = 0;
        int daysInYear = 365;
        for (int day = 1; day <= daysInYear; ++day) {
            double sunAngle = getSunAngle(day, northEquatorLatitude, southEquatorLatitude);
            if (sunAngle >= 0 && sunAngle <= Math.PI) northernHeemspheric += sunAngle;
        }
        double southernmostPointLatitude = -90 + northernHeemspheric / daysInYear * 180;
        assertThat(southernmostPointLatitude, is(-90));
    }

    public void testSouthernmostPoint() {
        int southernmostIndex = findSouthernmostIndex();
        double southernmostPointLatitude = getSunAngle(0, northEquatorLatitude, southEquatorLatitude);
        assertThat(southernmostPointLatitude, is(-90 + southernmostIndex * 180 / 360));
    }

    public void testSolarDeclination() {
        int solarDeclination = 23.44;
        double northernHeemspheric = 0;
        for (int day = 1; day <= daysInYear; ++day) {
            double sunAngle = getSunAngle(day, northEquatorLatitude, southEquatorLatitude);
            if (solarDeclination >= -sunAngle && solarDeclination <= Math.PI - sunAngle) northernHeemspheric += solarDeclination;
        }
        assertThat(northernHeemspheric, is(23.44));
    }

    public double getSunAngle(int day, double northEquatorLatitude, double southEquatorLatitude) {
        int latitude = 0;
        while (true) {
            double angle = Math.toRadians(getTimeInDecimalHours(day + 1)) - Math.toRadians(latitude);
            if (Math.abs(angle) > 5 * Math.PI / 180 && !areAnglesEqual(angle, Math.PI - angle)) return angle;
            latitude += 1;
        }
    }

    public int findSouthernmostIndex() {
        double southernmostAngle = Math.PI + getSunAngle(0, northEquatorLatitude, southEquatorLatitude);
        for (int i = 1; i <= daysInYear / 2; ++i) {
            double angle = Math.toRadians(getTimeInDecimalHours(i + 1)) - Math.toRadians(southernmostAngle);
            if (Math.abs(angle) < 0.5 * Math.PI && !areAnglesEqual(angle, Math.PI - angle)) return i;
        }
        return daysInYear / 2;
    }

    public double getSunTime() {
        double latitude = northEquatorLatitude;
        int solarDeclination = 23.44;
        int day = findSouthernmostIndex();
        for (int hour = 1; hour <= hoursPerDay; ++hour) {
            double sunAngle = Math.toRadians(getTimeInDecimalHours(hour + 1)) - Math.toRadians(latitude);
            if (Math.abs(sunAngle) < 0.5 * Math.PI && !areAnglesEqual(sunAngle, Math.PI - sunAngle)) return hour;
        }
        return hoursPerDay / 2;
    }

    public double getTimeInDecimalHours(int minute) {
        // Convert minutes to decimal hours using the Earth's sidereal day
        int siderealDays = daysInYear / daysOfSiderealYear * getEarthTropicalLongitude();
        int hour = (minute + 0.5) / 60;
        return siderealDays * 24 + hour;
    }

    public double getElevationFromLatitude(double latitude, double altitude) {
        // This method is not implemented
        throw new UnsupportedOperationException("Not supported");
    }

    private boolean areAnglesEqual(double angle1, double angle2) {
        if (angle1 > Math.PI || angle2 < -Math.PI) return false;
        return Math.abs(angle1 - angle2) < 0.00001;
    }
}
