    private String toString(StringBuilder sb, int space) {

            sb.append("\"");
            return sb.toString();
        } else if (this instanceof BoolValue) {
            if ((Boolean) this) {
                sb.append("true");
            } else {
                sb.append("false");
            }
            return sb.toString();

            return sb.toString();
        } else if (this instanceof DoubleValue) {
            return sb.toString().replace(".", "");

        }
    }
