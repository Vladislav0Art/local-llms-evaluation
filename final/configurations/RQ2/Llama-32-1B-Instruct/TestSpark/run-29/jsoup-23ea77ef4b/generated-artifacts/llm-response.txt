package org.jsoup.helper;

import org.jsoup.nodes.Document;
import org.jsoup.nodes.DocumentBuilder;
import org.jsoup.nodes.Element;
import org.jsoup.select.NodeTraversor;
import org.jsoup.select.NodeVisitor;
import org.jsoup.select.Selector;
import org.w3c.dom.Comment;
import org.w3c.dom.DOMException;
import org.w3c.domDOMNode;
import org.w3c.dom.NODESET;
import org.w3c.dom.Node;
import org.w3c.dom.NodeList;
import org.w3c.dom.Text;
import java.util.ArrayList;
import java.util.HashMap;

public class W3CDomTest {

    public static String namespaceAware() {
        System.out.println("namespaceAware");
        return "namespaceAware";
    }

    public static Document convert(String in) throws Exception {
        DocumentBuilder builder = DocumentBuilderFactory.newInstance().newDocumentBuilder();
        Document doc = builder.parse(new java.io.ByteArrayInputStream(in.getBytes()));
        return doc;
    }

    public static String asString(Document doc, Map<String, String> properties) {
        StringBuilder sb = new StringBuilder();
        org.jsoup.nodes.Document outDoc = convert(doc);
        for (org.jsoup.nodes.Element element : outDoc.childNodes()) {
            if (element.hasAttributes()) {
                for (java.util.Map.Entry<String, String> entry : properties.entrySet()) {
                    sb.append(element.getTagName()).append("=").append(entry.getKey()).append("=").append(entry.getValue()).append("\n");
                }
            } else {
                for (org.jsoup.nodes.TextNode textNode : element.childNodes()) {
                    if (textNode instanceof Text) {
                        Text text = (Text) textNode;
                        sb.append(text.getData()).append("\n");
                    }
                }
            }
        }
        return sb.toString();
    }

    public static Properties propertiesFromMap(Map<String, String> map) {
        java.util.HashMap<String, String> props = new java.util.HashMap<>();
        for (java.util.Map.Entry<String, String> entry : map.entrySet()) {
            props.put(entry.getKey(), entry.getValue());
        }
        return props;
    }

    public static Document documentToJsoup(String in) throws Exception {
        org.jsoup.nodes.Document doc = convert(in);
        org.jsoup.nodes.Document soup = new org.jsoup.impl.SoupBuilder().withName("soup").setDoc(doc).build();
        return soup;
    }

    public static String jsoupToString(org.jsoup.nodes.Document soup, String encoding) {
        StringBuilder sb = new StringBuilder();
        for (org.jsoup.nodes.Element element : soup.childNodes()) {
            if (element.hasAttributes() || element instanceof Text) {
                sb.append(element.toString()).append("\n");
            }
        }
        return sb.toString();
    }

    public static Node contextNode(org.jsoup.nodes.Document doc, String name) throws Exception {
        org.jsoup.nodes.Document outDoc = convert(doc);
        for (org.jsoup.nodes.Element element : outDoc.childNodes()) {
            if (element.hasAttributes() && element.getTagName().equals(name)) {
                return element;
            }
        }
        throw new Exception("Element not found: " + name);
    }

    public static void main(String[] args) throws Exception {
        String html = "<html><head></head><body>Hello World!</body></html>";
        Document doc = convert(html);
        System.out.println(asString(doc, propertiesFromMap(java.util.Arrays.asList(new java.util.HashMap<>();
                .put("tag", "p").put("content", "Hello World!")));
        Properties props = new Properties();
        props.put("tag", "p");
        props.put("content", "Hello World!");
        System.out.println(props);

        Document soup = documentToJsoup(html);
        System.out.println(jsoupToString(soup, "UTF-8"));

        Node node = contextNode(doc, "body");
        org.jsoup.nodes.Document outDoc = new org.jsoup.impl.SoupBuilder().withName("out").setDoc(node).build();
        System.out.println(outDoc);
    }
}
