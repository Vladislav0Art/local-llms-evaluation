import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;

public class BufferedReader2 implements Runnable {
    private InputStream input;
    private BufferedWriter output;

    public BufferedReader2(InputStream in) throws IOException {
        this.input = in;
        this.output = new BufferedWriter(new OutputStreamWriter(System.out));
    }

    @Override
    public void run() {
        try {
            String line;
            while ((line = input.readLine()) != null) {
                if (line.length() > 0 && !isSpace(line)) {
                    output.write(line);
                    output.newLine();
                }
            }
        } catch (IOException e) {
            // handle exception
        }
    }

    public void flush() throws IOException {
        output.flush();
    }

    private boolean isSpace(String str) {
        return str.length() == 0 || Character.isWhitespace(str.charAt(0));
    }

            reader.readLine();

        thread.start();

            while ((line = input.readLine()) != null) {

import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.util.ArrayList;

public class BufferedReader2 implements Runnable {
    private InputStream input;
    private ArrayList<String> lines;

    public static void main(String[] args) throws IOException {
        Thread thread = new Thread(new BufferedReader2());
        thread.start();
    }

    @Override
    public void run() {
        try {
            StringBuilder outputBuilder = new StringBuilder();

            while (input.ready()) {
                String line;
                if (!line.isEmpty()) {
                    lines.add(line);
                    outputBuilder.append(line).append("\n");
                } else {
                    flush(outputBuilder.toString());
                }
            }

            if (!lines.isEmpty()) {
                outputBuilder.deleteCharAt(outputBuilder.length() - 1); // remove trailing newline
            }

            String finalOutput = outputBuilder.toString();
            System.out.println(finalOutput);
        } catch (IOException e) {
            // handle exception
        }
    }

    private void flush(String output) throws IOException {
        if (!output.isEmpty()) {
            System.out.print(output);
        }
    }

    @Test
    public void testWordWrapFromReader() {
        String[] lines = {"This is a very long text that needs to be wrapped.", "Short message"};
        for (String line : lines) {
            WordWrap.fromClasspathUtf8(line);
        }
    }

    @Test
    public void testWordWrapFromClasspathUtf8() {
        String[] lines = {"This is a very long text that needs to be wrapped.", "Short message"};
        for (String line : lines) {
            WordWrap.fromClasspathUtf8(line);
        }
    }

    @Test
    public void testWordWrapFromClasspath() {
        String[] lines = {"This is a very long text that needs to be wrapped.", "Short message"};
        for (String line : lines) {
            WordWrap.fromClasspath("test.txt");
        }
    }

    @Test
    public void testIsWhitespace() {
        ArrayList<String> lines = new ArrayList<>();
        for (int i = 0; i < 10000; i++) {
            lines.add("   ");
        }
        for (String line : lines) {
            WordWrap.isWhitespace(line);
        }
    }

    @Test
    public void testLeftTrim() {
        while (true) {
            try {
                input.readLine();
                if (!input.ready()) {
                    break;
                }
            } catch (IOException e) {
                // handle exception
            }
        }
        System.out.println(input.readLine());
    }
}
