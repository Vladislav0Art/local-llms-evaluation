import org.jsoup.nodes.Comment;
import org.jsoup.nodes.Document;
import org.jsoup.parser.Parser;
import org.junit.Assert;

public class CommentTest {
    private Document document;

    @Before
    public void setup() throws Exception {
        Parser parser = new Parser();
        ParseSettings parseSettings = parser.getSettings();
        parseSettings.setFormMode(Parser.FORM_MODE_COMMENT);
        document = new Document("comment.html", parseSettings, parser);
    }

    @Test
    public void nodeName_returnsCommentNodeName() {
        Assert.assertEquals("comment", Comment.class.getName());
    }

    @Test
    public void getData_returnsCommentData() {
        String data = "Some comment data";
        Comment comment = new Comment(data);
        Assert.assertEquals(data, comment.getData());
    }

    @Test
    public void setDataSetsCommentData() {
        String data = "Some updated comment data";
        Comment comment = new Comment(data);
        comment.setData(data);
        Assert.assertEquals(data, comment.getData());
    }

    @Test
    public void outerHtmlHead_setsOuterHtmlHeader() throws Exception {
        Comment comment = new Comment("Initial comment");
        comment.outerHtmlHead(document, 0, document.OutputSettings());
        Assert.assertEquals("<comment>", comment.outerHtml());
    }

    @Test
    public void outerHtmlTail_setsOuterHtmlTail() throws Exception {
        Comment comment = new Comment("Initial comment");
        comment.outerHtmlTail(document, 0, document.OutputSettings());
        Assert.assertEquals("<comment></comment>", comment.outerHtml());
    }

    @Test
    public void toString_returnsCommentString() {
        Comment comment = new Comment("Some comment data");
        Assert.assertEquals("Some comment data", comment.toString());
    }

    @Test
    public void clone_setsCommentClone() throws Exception {
        Comment comment = new Comment("Initial comment");
        Comment clonedComment = comment.clone();
        Assert.assertEquals(comment, clonedComment);
    }

    @Test
    public void isXmlDeclaration_returnsfalse() {
        Comment comment = new Comment("");
        Assert.assertFalse(comment.isXmlDeclaration());
    }

    @Test
    public void asXmlDeclaration_setsAsXmlDeclaration() throws Exception {
        Comment comment = new Comment("");
        XmlDeclaration xmlDeclaration = comment.asXmlDeclaration();
        Assert.assertEquals("<comment>", xmlDeclaration.getValue());
    }
}

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.parser.Parser;

public class CommentTest {
    private Document document;
    private Comment comment1, comment2;

    @Before
    public void setup() throws Exception {
        Parser parser = new Parser();
        ParseSettings parseSettings = parser.getSettings();
        parseSettings.setFormMode(Parser.FORM_MODE_COMMENT);
        comment1 = new Comment();
        Element element = createCommentElement("comment1");
        document = new Document(), comment2 = new Comment();
        comment2.outerHtml = "comment2";
        addElementToDocument(document, element, comment1);
    }

    private void addElementToDocument(Document document, Element element, Comment comment) {
        element.appendChild(comment.outerHtmlElement());
    }

    public static class Comment extends Element {
        private String text;

        public Comment() {}

        public String getText() {
            return text;
        }

        public void setText(String text) {
            this.text = text;
        }
    }

    @Test
    public void test_outerHtmlHead() throws Exception {
        document.selectFirst("comment1").outerHtmlElement().setText("test");
        Assert.assertEquals("<comment1>test</comment1>", document.selectFirst("comment1").outerHtmlElement().toString());
    }

    @Test
    public void test_outerHtmlTail() throws Exception {
        comment1.outerHtmlElement().setText("new text");
        comment2.outerHtmlElement().setText("");
        Assert.assertEquals("<comment1>New text</comment1>", comment1.outerHtmlElement().toString());
    }
}

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.parser.Parser;

public class CommentTest {
    private Document document, comment1;
    private Comment comment2;

    @Before
    public void setup() throws Exception {
        Parser parser = new Parser();
        ParseSettings parseSettings = parser.getSettings();
        parseSettings.setFormMode(Parser.FORM_MODE_COMMENT);
        document = new Document(), comment1 = new Comment(), comment2 = new Comment();
    }

    @Test
    public void test_outerHtmlHead() throws Exception {
        addElementToDocument(document, createCommentElement("comment1"), comment1);
        Assert.assertEquals("<comment1>test</comment1>", document.selectFirst("comment1").outerHtmlElement().toString());
    }

    @Test
    public void test_outerHtmlTail() throws Exception {
        comment1.outerHtmlElement().setText("new text");
        Assert.assertEquals("<comment1>New text</comment1>", comment1.outerHtmlElement().toString());
        addElementToDocument(document, comment2);
        Assert.assertEquals("<comment1>New text</comment1>", document.selectFirst("comment1").outerHtmlElement().toString());
    }

    private void addElementToDocument(Document document, Element element) {
        if (document.hasChildNodes()) {
            for (int i = 0; i < element.childNodes.length; i++) {
                Element child = element.childNodes.item(i);
                if (child instanceof Document && ((Document) child).hasChildNodes() && !((Document) child).isEmpty()) {
                    addElementToDocument((Document) child, element);
                }
            }
        } else {
            document.appendChild(element);
        }
    }

    private Element createCommentElement(String text) {
        Element comment = new Element("comment");
        comment.setAttribute("text", text);
        return comment;
    }
}

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.parser.Parser;

public class CommentTest {
    private Document document, comment1;
    private Comment comment2;

    @Before
    public void setup() throws Exception {
        Parser parser = new Parser();
        ParseSettings parseSettings = parser.getSettings();
        parseSettings.setFormMode(Parser.FORM_MODE_COMMENT);
        document = new Document(), comment1 = new Comment(), comment2 = new Comment();
        comment2.outerHtmlElement().setText("");
    }

    @Test
    public void test_outerHtmlHead() throws Exception {
        addElementToDocument(document, comment1, "comment1");
        Assert.assertEquals("<comment1>test</comment1>", document.selectFirst("comment1").outerHtmlElement().toString());
    }

    @Test
    public void test_outerHtmlTail() throws Exception {
        comment1.outerHtmlElement().setText("new text");

    }

        if (!document.hasChildNodes()) {
            element.appendChild(document.createElement("br"));

                Element child = element.childNodes.item(i);
                if (child instanceof Document && ((Document) child).hasChildNodes() && !((Document) child).isEmpty()) {
                    addElementToDocument((Document) child, element);
                }
            }
        }
    }

    private static void assertNothing() {
        System.out.println("Test case passed");
    }
}
