import org.junit.jupiter.api.Test;

public class Documentofiscal {
    public int codigoInterno(int n) {
        return n;
    }

    public String placa(String numero) {
        return numero;
    }

    public String renavam(String nome) {
        return nome + " Renavam";
    }

    public String tara(String marca) {
        return marca + " Tara";
    }

    public double capacidadeKG(int peso) {
        if (peso < 50) {
            return 100;
        } else if (peso >= 50 && peso <= 150) {
            return 200.0;
        } else if (peso > 150) {
            return 300.0;
        }
    }

    public double capacidadeM3(int volume) {
        if (volume < 100) {
            return 1.0;
        } else if (volume >= 100 && volume <= 500) {
            return 2.5;
        } else if (volume > 500) {
            return 4.0;
        }
    }
}

@Test
public void testCodigoInterno() {
    assertEquals(100, Documentofiscal.codigoInterno(100));
}

@Test
public void testPlaca() {
    assertEquals("ABC001", Documentofiscal.placa("001"));
}

@Test
public void testRenavam() {
    assertEquals("Fernando Renavam", Documentofiscal.renavam("Fernando"));
}

@Test
public void testTara() {
    assertEquals("Toyota Tara", Documentofiscal.tara("Toyota"));
}

@Test
public void testCapacidadeKG() {
    assertEquals(100, Documentofiscal.capacidadeKG(50));
}

@Test
public void testCapacidadeM3() {
    assertEquals(200.0, Documentofiscal.capacidadeM3(150), 1);
}
