public enum Manufacturer {
    MY("My Manufacturer"),
    OTHER("Other Manufacturer");

    private final String name;

        this.name = name;
    }

    public String getName() {
        return name;
    }
}

class Network {
    private Manufacturer manufacturer;
    private int id;

    public Network(Manufacturer manu, int id) {
        this.manufacturer = manu;
        this.id = id;
    }

    public Manufacturer getManufacturer() {
        return manufacturer;
    }

    public void setManufacturer(Manufacturer manu) {
        this.manufacturer = manu;
    }

    public int getId() {
        return id;
    }

    public void setId(int id) {
        this.id = id;
    }
}

public class NetworkTest extends TestCase {
    private Manufacturer[] manufacturers;

    protected void setUp() throws Exception {
        super.setUp();
        manufacturers = new Manufacturer[]{Manufacturer.MY, Manufacturer.OTHER};
    }

    protected void tearDown() throws Exception {
        super.tearDown();
        manufacturers = null;
    }

    public void testGetManufacturer() {
        assertEquals(Manufacturer.MY, Network.getManufacturer(0));
        assertEquals(Manufacturer.OTHER, Network.getManufacturer(1));

}
