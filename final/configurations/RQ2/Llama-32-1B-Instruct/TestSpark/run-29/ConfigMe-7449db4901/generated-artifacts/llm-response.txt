public class CommentsConfigurationImpl {
    private final Map<String, List<String>> comments;

    public CommentsConfigurationImpl(@NotNull Map<String, List<String>> comments) {
        this.comments = comments;
    }

    @MethodUnderTest(setComment)
    public void setComment(String path, String... commentLines) {
        if (commentLines.length > 1 && !commentLines[0].isEmpty()) {
            throw new IllegalArgumentException("Only one single-line comment is allowed");
        }
        for (String line : commentLines) {
            comments.get(path).add(line);
        }
    }

    @MethodUnderTest(getAllComments)
    public Map<String, List<String>> getAllComments() {
        return Collections.unmodifiableMap(new HashMap<>(comments));
    }

    @MethodUnderTest(getAllComments)
    public boolean hasComment(String path) {
        return !comments.isEmpty();
    }
}

public class CommentsConfigurationTest {
    private CommentsConfigurationImpl commentsConfiguration;

    @Before
    public void setup() {}

    @MethodUnderTest(setComment)
    public void test_setComment_SingleLineComment() {
        String path = "path";
        List<String> commentLines = Arrays.asList("");
        commentsConfiguration.setComment(path, commentLines);
        assertEquals(Collections.emptyList(), commentsConfiguration.getAllComments().values());
        assertTrue(commentsConfiguration.hasComment(path));
    }

    @MethodUnderTest(setComment)
    public void test_setComment_SingleLineCommentWithNoSpace() {
        String path = "path";
        List<String> commentLines = Arrays.asList("comment");
        commentsConfiguration.setComment(path, commentLines);
        assertEquals(Collections.emptyList(), commentsConfiguration.getAllComments().values());
        assertTrue(commentsConfiguration.hasComment(path));
    }

    @MethodUnderTest(setComment)
    public void test_setComment_NoComment() {
        String path = "path";
        List<String> commentLines = Arrays.asList("comment");
        commentsConfiguration.setComment(path, commentLines);
        assertEquals(Collections.emptyList(), commentsConfiguration.getAllComments().values());
        assertTrue(commentsConfiguration.hasComment(path));
    }

    @MethodUnderTest(getAllComments)
    public void test_getAllComments() {
        String path = "path";
        List<String> commentLines = Arrays.asList("comment1", "comment2");
        commentsConfiguration.setComment(path, commentLines);
        assertEquals(Arrays.asList(new ArrayList<>(Arrays.asList("comment1", "comment2"))), commentsConfiguration.getAllComments().values());
    }

    @MethodUnderTest(getAllComments)
    public void test_getAllCommentsNoPath() {
        Map<String, List<String>> map = new HashMap<>();
        map.put("", Collections.emptyList());
       _commentsConfiguration.setComment("", map);
        assertEquals(Collections.emptyList(), commentsConfiguration.getAllComments().values());
    }
}
