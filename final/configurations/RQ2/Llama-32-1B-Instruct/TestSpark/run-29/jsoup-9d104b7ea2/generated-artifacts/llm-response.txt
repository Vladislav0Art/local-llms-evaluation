    public String stripLeadingWhitespace(String str) {
        return str.trim();
    }

    public boolean isEmpty(String str) {

        if (str == null || str.isEmpty()) {
            return true;
        } else {
            return false;
        }
    }

    public int normaliseWhitespace(String str, int length) {
        String s = stripLeadingWhitespace(str);
        // Normalizing the remaining string by removing any leading whitespace
        return s.length() - s.trim().length();
    }

    public boolean lastCharIsWhitespace(String str) {
        if (str == null || str.isEmpty()) {
            throw new AssertionError("Input should not be empty");
        }
        String s = stripLeadingWhitespace(str);
        // Check if the string ends with a whitespace
        return s.charAt(s.length() - 1).equals(' ');
