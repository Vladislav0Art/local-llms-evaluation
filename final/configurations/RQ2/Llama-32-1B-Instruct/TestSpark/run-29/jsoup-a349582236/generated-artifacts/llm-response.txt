    @Test
    public void should_return_true_for_public_method() {
        assertThat(Public.publicMethod(), is(true));
    }

    @Test
    public void should_return_false_for_nonPublicMethod() {
        assertNotSame(Public.publicMethod, null);
        assertNotSame(Public.publicMethod, "Hello");
    }

    @Test
    public void should_return_the_same_value_from_public_method_if_it_is_a_static_variable() {
        System.out.println(Public.staticVariable);
        assertThat(Public.staticVariable, is(42));
    }

    @Test
    public void should_not_return_the_same_value_from_nonPublicMethod_if_it_is_a_static_variable() {
        assertNotSame(Public.staticVariable, null);
        assertNotSame(Public.staticVariable, "Hello");
