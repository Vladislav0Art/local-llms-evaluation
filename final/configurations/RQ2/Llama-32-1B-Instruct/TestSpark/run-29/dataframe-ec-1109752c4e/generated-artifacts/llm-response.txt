class DataFrame {
    private String name;
    private int columns;

    public DataFrame(String name, int columns) {
        this.name = name;
        this.columns = columns;
    }

    public String getName() {
        return name;
    }

    public int getColumns() {
        return columns;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;

        DataFrame df = (DataFrame) o;

        if (name != null ? !name.equals(df.name) : df.name != null)
            return false;

        return columns == df.columns;
    }

    @Override
    public int hashCode() {
        int result = name != null ? name.hashCode() : 0;
        result = 31 * result + (columns != 0 ? columns : 0);

        return result;
    }
}

class Solution {
    public DataFrame printDataFrame() {

        return columns == solution.columns;
    }

    @Override
    public int hashCode() {
        int result = name != null ? name.hashCode() : 0;

        result = 31 * result + (columns != 0 ? columns : 0);

    }
