import java.util.*;

public class TagService {
    private Map<String, Document> documents;
    private List<Tag> tags;

    public void saveTag(String id, String name) {
        if (id != null && !name.isEmpty()) {
            if (!documents.containsKey(id)) {
                Document document = new Document();
                document.setContent("");
                documents.put(id, document);
            }
            Tag tag = new Tag(name);
            if (!tags.contains(tag)) {
                tags.add(tag);
            }
            ((Document) documents.get(id)).addTag(tag);
        } else {
            System.out.println("Invalid tag or document id not found");
        }
    }

    public List<Tag> getAllTags() {
        return tags;
    }

    public void deleteTag(String id, String name) {
        if (id != null && !name.isEmpty()) {
            Document document = documents.get(id);
            for (int i = 0; i < document.getTags().size(); i++) {
                if (document.getTags().get(i).getName().equals(name)) {
                    ((Document) documents.get(id)).getTags().remove(i);
                }
            }
        } else {
            System.out.println("Invalid tag or document id not found");
        }
    }

    public void updateTag(String id, String name) {
        if (id != null && !name.isEmpty()) {
            Document document = documents.get(id);
            Tag tag = new Tag(name);
            for (int i = 0; i < document.getTags().size(); i++) {
                if (document.getTags().get(i).getName().equals(tag.getName())) {
                    ((Document) documents.get(id)).getTags().set(i, tag);
                    System.out.println("Tag updated successfully");
                    break;
                }
            }
        } else {
            System.out.println("Invalid tag or document id not found");
        }
    }

    public void updateDocumentName(String id, String newName) {
        if (id != null && !newName.isEmpty()) {
            Document document = documents.get(id);
            ((Document) documents.get(id)).setContent(newName);
            System.out.println("Document name updated successfully");
        } else {
            System.out.println("Invalid document id not found");
        }
    }

    public void updateTagName(String id, String newName) {
        if (id != null && !newName.isEmpty()) {
            Tag tag = new Tag(newName);
            for (Tag t : documents.get(id).getTags()) {
                if (!t.getName().equals(tag.getName())) {
                    ((Document) documents.get(id)).addTag(t);
                    System.out.println("Tag name updated successfully");
                    break;
                }
            }
        } else {
            System.out.println("Invalid tag id not found");
        }
    }

    public void saveDocuments() {
        Map<String, Document> map = new HashMap<>();
        for (Document doc : documents.values()) {
            map.put(doc.getId(), doc);
        }
        return map;
    }
}
