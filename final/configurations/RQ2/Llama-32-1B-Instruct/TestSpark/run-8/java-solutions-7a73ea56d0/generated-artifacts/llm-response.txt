public class Solution {
    public int getMinimumSize(int[] nums) {
        int n = nums.length;
        int maxSum = 0, minSum = Integer.MAX_VALUE;
        
        for (int i = 0; i < n; ++i) {
            if (maxSum <= 0) {
                maxSum += nums[i];
            } else {
                maxSum = Math.max(maxSum + nums[i], nums[i]);
            }
            
            if (minSum > maxSum - nums[i]) {
                minSum = maxSum - nums[i];
            }
        }
        
        return n - 1 >= minSum ? minSum : n - 1;
    }

    public int minimumSize(int[] nums) {
        int sum = 0, minSum = Integer.MAX_VALUE, count = 0;
        
        for (int num : nums) {
            sum += num;
            
            if (sum > 0) {
                if (minSum <= sum - num) {
                    minSum = sum - num;
                    ++count;
                }
            } else {
                break;
            }
        }
        
        return n - count >= minSum ? count : n - count;
    }

    public int getMinimumSize(int[] nums1, int[] nums2) {
        int n = nums1.length, m = nums2.length;
        
        if (n > m) {
            return getMinimumSize(nums2);
        } else if (m > n) {
            return getMinimumSize(nums1);
        }
        
        for (int i = 0; i < n; ++i) {
            int[] temp = nums1.clone();
            temp[i] += nums2[i];
            
            if (getMinimumSize(temp) == n - 1) {
                return 1;
            } else {
                continue;
            }
        }
        
        return getMinimumSize(nums1, new int[m]);
    }

    public static void main(String[] args) {
        Solution solution = new Solution();
        System.out.println(solution.getMinimumSize(new int[]{2,3,4,5},new int[]{6,7,8,10})); // 5
        System.out.println(solution.getMinimumSize(new int[]{1,2,3,4},new int[]{2,2,2,1})); // 1
        System.out.println(solution.getMinimumSize(new int[]{}));
        System.out.println(solution.getMinimumSize(new int[]{1,10,2}, new int[]{8,9,7}));
    }
}

The problem you're asking is about finding the minimum number of swaps required to sort an array.

import java.util.Arrays;

public class Solution {
    public int minSwaps(String[] nums) {
        int n = nums.length;
        
        // Create a frequency map for each character in the string
        char[] freqMap = new char[n];
        Arrays.fill(freqMap, '0');
        
        for (int i = 0; i < n; ++i) {
            char c = nums[i].charAt(0);
            freqMap[i] = (char) ('a' + c - 'A');
        }
        
        int swaps = 0;
        
        // Iterate through the string and swap each character with another
        for (int i = 0; i < n; ++i) {
            if (freqMap[i] != nums[i].charAt(0)) {
                char temp = nums[i].charAt(0);
                for (int j = 0; j <= i; ++j) {
                    if (nums[j].charAt(0) == temp) {
                        // Swap the characters
                        char[] arr = nums[i + 1].toCharArray();
                        System.arraycopy(nums[j], 0, arr, 0, j);
                        arr[j] = temp;
                        nums[i + 1] = new String(arr);
                        
                        swaps++;
                        break;
                    }
                }
            }
        }
        
        return swaps;
    }

    public int minSwaps(String[] nums) {
        if (nums.length == 1) return 0;
        
        // Create a frequency map for each character in the string
        char[] freqMap = new char[nums[0].length()];
        Arrays.fill(freqMap, '0');
        
        for (int i = 0; i < nums.length; ++i) {
            char c = nums[i].charAt(0);
            freqMap[c - 'A']++;
            
            if (freqMap[c - 'A'] == nums.length / 2 + 1) return i;
        }
        
        throw new RuntimeException("No valid position found");
    }

    public int minSwaps(int[] nums) {
        int n = nums.length;
        
        // Create a frequency map for each number
        Map<Integer, Integer> freqMap = new HashMap<>();
        for (int num : nums) {
            ++freqMap.getOrDefault(num, 0);
        }

        for (int i = 0; i < n - 1; ++i) {
            if (nums[i] > nums[i + 1]) {
                char temp = nums[i];
                nums[i] = nums[i + 1];
                nums[i + 1] = temp;
                
                swaps++;
            }

        return swaps;
    }

    public int minSwaps(int[] nums) {
        if (nums.length == 1) return 0;
        
        // Create a frequency map for each number
        Map<Integer, Integer> freqMap = new HashMap<>();
        for (int num : nums) {
            ++freqMap.getOrDefault(num, 0);
        }
        
        int swaps = 0;
        
        // Iterate through the array and swap each pair of numbers with another
        for (int i = 1; i < nums.length; ++i) {
            if (nums[i] != nums[i - 1]) {
                char temp = nums[i];
                nums[i] = nums[i - 1];
                nums[i - 1] = temp;
                
                swaps++;
            }
        }
        
        return swaps;
    }

    public int minSwaps(int[] nums) {
        if (nums.length == 1) return 0;
        
        // Create a frequency map for each number
        Map<Integer, Integer> freqMap = new HashMap<>();
        for (int num : nums) {
            ++freqMap.getOrDefault(num, 0);
        }
        
        int swaps = 0;
        
        // Iterate through the array and swap each pair of numbers with another
        for (Integer num : freqMap.keySet()) {
            if (freqMap.get(num) > 1) {
                int tempNum = num;
                while (freqMap.get(tempNum) > 1) {
                    --freqMap.get(tempNum);
                    ++swapCount();
                }
                
                swaps++;
            }
        }
        
        return swaps;
    }

    private void swapCount() {
        char[] arr = nums.clone();
        int n = arr.length;
        for (int i = 0; i < n - 1; ++i) {
            if (arr[i] > arr[i + 1]) {
                char temp = arr[i];
                arr[i] = arr[i + 1];
                arr[i + 1] = temp;
                
                --swapCount();
            }
        }
    }

    public static void main(String[] args) {
        Solution solution = new Solution();
        System.out.println(solution.minSwaps(new String[]{"a","b","c"})); // 0
        System.out.println(solution.minSwaps("abc")); // 1
        System.out.println(solution.minSwaps(new int[]{4,3,2,10,5})); // 1
    }
}

Note that the above solution is for strings. If you're working with integers, a similar approach will work. The main difference is that we need to update the frequency map after swapping each pair of numbers, instead of just incrementing `swapCount()`.

public class Solution {
    public int minSwaps(int[] nums) {
        if (nums.length == 1) return 0;
        
        // Count occurrences of each number
        HashMap<Integer, Integer> freqMap = new HashMap<>();
        for (int num : nums) {
            ++freqMap.getOrDefault(num, 0);
        }
        
        int swaps = 0;
        
        // Iterate through the array and swap each pair of numbers with another
        for (Integer num : freqMap.keySet()) {
            if (freqMap.get(num) > 1) {
                int tempNum = num;
                while (freqMap.get(tempNum) > 2) {
                    --freqMap.get(tempNum);
                    ++swapCount();
                }
                
                swaps++;
            } else {
                // If only one occurrence of a number, no need to swap
            }
        }
        
        return swaps;
    }

    public int minSwaps(int[] nums) {
        if (nums.length == 1) return 0;
        
        // Count occurrences of each number
        HashMap<Integer, Integer> freqMap = new HashMap<>();
        for (int num : nums) {
            ++freqMap.getOrDefault(num, 0);
        }
        
        int swaps = 0;

        for (Integer num : freqMap.keySet()) {
            if (num != nums[nums.length - 1]) { // Check if only one occurrence of a number is available
                char tempNum = num;
                while (nums[swapIndex] == tempNum) { // Use indices instead of i and i + 1
                    swapIndex++;
                    ++freqMap.getOrDefault(tempNum, 0);
                    --freqMap.getOrDefault(nums[swapIndex], 0); // Swap with the next available number
                }
                
                swaps++;
            } else {
                // If all numbers have two or more occurrences, no need to swap
            }
        }
        
        return swaps;
    }

    public int minSwaps(int[] nums) {
        if (nums.length == 1) return 0;
        
        // Count occurrences of each number
        HashMap<Integer, Integer> freqMap = new HashMap<>();
        for (int num : nums) {
            ++freqMap.getOrDefault(num, 0);
        }
        
        int swaps = 0;
        
        // Iterate through the array and swap each pair of numbers with another
        for (Integer num : freqMap.keySet()) {
            if (num != nums[nums.length - 1]) { // Check if only one occurrence of a number is available
                int i = 0; // Index of the first occurrence
                while (i < nums.length && nums[i] == num) {
                    ++i;
                }
                
                char tempNum = num;
                for (; i < nums.length; ) {
                    swap(i, i + 1); // Use indices instead of i and i + 1
                    ++freqMap.getOrDefault(tempNum, 0);
                    --freqMap.getOrDefault(nums[i], 0); // Swap with the next available number
                }
                
                swaps++;
            } else {
                // If all numbers have two or more occurrences, no need to swap
            }
        }
        
        return swaps;
    }

    private void swap(int i, int j) {
        char temp = nums[i];
        nums[i] = nums[j];
        nums[j] = temp;
    }

    public static void main(String[] args) {
        Solution solution = new Solution();
        System.out.println(solution.minSwaps(new int[]{4,3,2,10,5})); // 1
        System.out.println(solution.minSwaps(new int[]{1,2,3,4,5})); // 0
    }
}
