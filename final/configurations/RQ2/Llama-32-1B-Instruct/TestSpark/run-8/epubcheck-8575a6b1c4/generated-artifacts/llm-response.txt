```java
public class PubSubMessageService {

    public String processPubSub(String message) {
        return "Received: " + message;
    }

    public String publishPubSub(String channel, String message) {
        System.out.println("Publishing to " + channel);
        return "Published to " + channel + ": " + message;
    }
}

public class PubSubProcessor {

    public String processPubSub(PubSubMessageService pubSubService, String channel, String message) {
        return pubSubService.publishPubSub(channel, message);
    }

    public static void main(String[] args) {
        PubSubProcessor processor = new PubSubProcessor();
        PubSubMessageService pubSubService = new PubSubMessageService();

        System.out.println("Testing processPubSub with channel 'test' and message 'Hello World!'");
        String result = processor.processPubSub(pubSubService, "test", "Hello World!");
        System.out.println(result);  // Output: Published to test: Hello World!

        System.out.println("\nTesting processPubSub with channel 'test' and message 'Goodbye World!'");
        result = processor.processPubSub(pubSubService, "test", "Goodbye World!");
        System.out.println(result);  // Output: Published to test: Goodbye World!
    }
}
```

**PubSubProcessor.java**
```java
import java.util.Scanner;

public class PubSubProcessor {

    private PubSubMessageService pubSubService;

    public PubSubProcessor(PubSubMessageService pubSubService) {
        this.pubSubService = pubSubService;
    }

    public String processPubSub(String channel, String message) {
        return pubSubService.processPubSub(message);
    }
}
```

**Main.java**
```java
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        PubSubMessageService pubSubService = new PubSubMessageService();

        Scanner scanner = new Scanner(System.in);

        System.out.println("1. Publish message to channel 'test'");
        System.out.println("2. Process received message");
        System.out.println("3. Exit");

        while (true) {
            System.out.print("Choose an option: ");
            int option = scanner.nextInt();
            scanner.nextLine(); // Consume newline left-over

            switch (option) {
                case 1:
                    System.out.print("Enter channel to publish message: ");
                    String channel = scanner.nextLine();
                    System.out.print("Enter message: ");
                    String message = scanner.nextLine();

                    System.out.println(pubSubService.publishPubSub(channel, message));
                    break;
                case 2:
                    processPubSub(scanner);
                    break;
                case 3:
                    System.out.println("Exiting...");
                    return;
                default:
                    System.out.println("Invalid option. Please choose again.");
            }
        }
    }

    public static void processPubSub(Scanner scanner) {
        PubSubProcessor processor = new PubSubProcessor(pubSubService);

        while (true) {
            System.out.print("Enter channel to publish message: ");
            String channel = scanner.nextLine();
            System.out.print("Enter message: ");
            String message = scanner.nextLine();

            System.out.println(processor.processPubSub(channel, message));
        }
    }
}
```

This implementation provides a simple command-line interface for publishing messages to channels and processing received messages. The `processPubSub` method in the `PubSubProcessor` class calls the `publishPubSub` method of the `PubSubMessageService`.

Here is a Java code that implements a Simple Pub/Sub system using Apache Kafka and Netty:

**KafkaProducer.java**
```java
import org.apache.kafka.clients.producer.KafkaProducer;
import org.apache.kafka.clients.producer.ProducerConfig;
import org.apache.kafka.common.serialization.StringSerializer;

public class KafkaProducer {
    private final KafkaProducer<String, String> producer;

    public KafkaProducer() {
        Properties props = new Properties();
        props.put(ProducerConfig.BOOTSTRAP_SERVERS_CONFIG, "localhost:9092");
        props.put(ProducerConfig.KEY_SERIALIZER_CLASS_CONFIG, StringSerializer.class.getName());
        props.put(ProducerConfig.VALUE_SERIALIZER_CLASS_CONFIG, StringSerializer.class.getName());

        producer = new KafkaProducer<>(props);
    }

    public void publish(String topic, String message) {
        producer.send(new ProducerRecord<>(topic, message));
    }
}
```

**KafkaConsumer.java**
```java
import org.apache.kafka.clients.consumer.ConsumerConfig;
import org.apache.kafka.clients.consumer.KafkaConsumer;
import org.apache.kafka.common.serialization.StringDeserializer;

public class KafkaConsumer {
    private final KafkaConsumer<String, String> consumer;

    public KafkaConsumer() {
        Properties props = new Properties();
        props.put(ConsumerConfig.BOOTSTRAP_SERVERS_CONFIG, "localhost:9092");
        props.put(ConsumerConfig.GROUP_ID_CONFIG, "my-group");

        consumer = new KafkaConsumer<>(props);
        consumer.subscribe(Collections.singleton("test"));
    }

    public void consume() {
        while (true) {
            ConsumerRecords<String, String> records = consumer.poll(100);

            for (ConsumerRecord<String, String> record : records) {
                System.out.println(record.value());
            }
        }
    }
}
```

**Main.java**
```java
import org.apache.kafka.clients.consumer.ConsumerConfig;
import org.apache.kafka.common.serialization.StringDeserializer;

public class Main {
    public static void main(String[] args) {
        KafkaProducer<String, String> producer = new KafkaProducer<>();
        KafkaConsumer<String, String> consumer = new KafkaConsumer<>();

        System.out.println("Subscribing to topic 'test'");
        consumer.subscribe(Collections.singleton("test"));

        while (true) {
            System.out.println("Publishing message: Hello World!");
            producer.publish("test", "Hello World!".getBytes());

            System.out.println("\nConsuming messages");
            consumer.poll();
        }
    }
}
```

This code creates a Simple Pub/Sub system with two classes:

*   `KafkaProducer`: This class provides methods to publish and consume messages. It uses the Apache Kafka client library.
*   `KafkaConsumer`: This class provides methods to subscribe, poll for new records, and print consumed records.

In the `Main` class, we create instances of these classes, subscribe to a topic, publish a message, and then read and print consumed records.
