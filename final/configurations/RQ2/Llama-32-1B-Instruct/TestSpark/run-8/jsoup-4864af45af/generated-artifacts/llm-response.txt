```java
}
```

```java
public static boolean isSafeTag(String tag, String[] allowedTags) {
```

```java
        }
    }
```

```java
import java.util.*;

public class Safelist {

    public static void addProtocol(String protocol) {

        return Arrays.stream(allowedTags).anyMatch(t -> t.equals(tag));

    public static Set<String> getProtocols() {
        return ALLOWED_PROTOCOLS;
    }
}

        String safeTag = Safelist.isSafeTag("img", new String[]{"image/jpeg"});
        System.out.println(safeTag); // prints: true

        Set<String> protocols = Safelist.getProtocols();
        System.out.println(protocols); // prints: [image/jpeg, video/mp4]
    }
}
```
