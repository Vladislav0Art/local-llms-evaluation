public class Main {

    public static double estimate(LocalDate forDate) {
        String today = LocalDate.now().format(DateTimeFormatter.ISO_DATE);
        return getEstimatedValue(forDate, today);
    }

    private static double getEstimatedValue(LocalDate forDate, String dateStr) {
        // Implementation of temperature calculation based on the date
        if (dateStr.equals("2022-01-01")) {
            return 20;
        } else if (dateStr.equals("2022-06-15")) {
            return 22;
        } else {
            double temp = getTemperatureForDate(dateStr);
            if (!temp.equals(null)) {
                return temp;
            }
            return 21; // Default value for other dates
        }
    }

    private static double getTemperatureForDate(String dateStr) {
        String[] temperatureParts = dateStr.split("-");
        LocalDate year = LocalDate.parse(temperatureParts[0]);
        int month = Integer.parseInt(temperatureParts[1]);
        int day = Integer.parseInt(temperatureParts[2]);

        if (month < 3 || (month == 4 && day >= 20)) {
            return 21; // Winter
        } else {
            LocalDate date = year.atMonth(month - 1).withDayOfMonth(day);
            double temperature = calculateTemperatureForDate(date.format(DateTimeFormatter.ISO_DATE));
            if (temperature != null) {
                return temperature;
            }
            return 21; // Default value for other months
        }
    }

        String date = "2022-07-15";

    }

import java.time.LocalDate;
import java.time.format.DateTimeFormatter;

public class Main {

    public static double estimate(LocalDate forDate) {
        String today = LocalDate.now().format(DateTimeFormatter.ISO_DATE);
        return getEstimatedValue(forDate, today);
    }

    private static double getEstimatedValue(LocalDate forDate, String dateStr) {
        // Implementation of temperature calculation based on the date
        if (dateStr.equals("2022-01-01")) {
            return 20;

            return 22;
        } else if (isSummerDate(dateStr)) {

            double temp = getTemperatureForDate(dateStr);
            if (!temp.equals(null)) {
                return temp;
            }
            return 21; // Default value for other dates
        }
    }

    private static boolean isSummerDate(String dateStr) {
        String[] temperatureParts = dateStr.split("-");
        LocalDate year = LocalDate.parse(temperatureParts[0]);
        int month = Integer.parseInt(temperatureParts[1]);
        int day = Integer.parseInt(temperatureParts[2]);

        if (month < 6 || (month == 7 && day >= 20)) {
            return true;
        }
        return false;
