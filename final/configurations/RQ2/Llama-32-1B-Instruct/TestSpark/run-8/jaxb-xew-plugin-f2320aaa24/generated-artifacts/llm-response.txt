import java.lang.reflect.Method;

public class TestGenerator {
    private static final String PUBLIC_METHODS = "public final void $1($2) { $3(); }";
    private static final String PRIVATE_METHODS = "private final void $1() { $2(); }";

    public static void main(String[] args) throws Exception {
        for (String methodName : new String[] {"foo", "bar"}) {
            String methodDefinition = generateMethod(methodName);
            System.out.println(methodDefinition);

            try (Method declaration = getMethod("test.$1($2), TestGenerator") throws NoSuchMethodException, SecurityException) {
                declaration.invoke(null, args);
            }
        }

        for (String methodName : new String[] {"foo", "bar"}) {
            String methodDefinition = generatePrivateMethod(methodName);
            System.out.println(methodDefinition);

            try (Method declaration = getMethod("test.$1($2), TestGenerator") throws NoSuchMethodException, SecurityException) {
                declaration.invoke(null, args);
            }
        }
    }

    private static String generatePublicMethod(String methodName) {
        return PUBLIC_METHODS.replace("$1", "foo").replace("$2", "bar").replace("$3", "$4");
    }

    private static String generatePrivateMethod(String methodName) {
        return PRIVATE_METHODS.replace("$1", "foo").replace("$2", "bar");
    }
}

class TestGeneratorTest {
    @Test
    public void testFooBarPublicMethod() throws Exception {
        TestGenerator.main(new String[]{"test.foo($1, $2)", new Object[]{"bar"}});
    }

    @Test
    public void testFooBarPrivateMethod() throws Exception {
        TestGenerator.main(new String[]{"test.foo($1(), $2)", new Object[]{"bar"}});

        Method declaration = TestGenerator.getTestElementWrapper().getMethod("foo", java.util.List.class, java.lang.Object.class);
        declaration.invoke(null, null, null);
    }

    private static Method getMethod(String methodName) throws NoSuchMethodException, SecurityException {
        return TestGenerator.testClass.getMethod(methodName);
    }
}

class TestClass {
    public String foo() { return "bar"; }
}
