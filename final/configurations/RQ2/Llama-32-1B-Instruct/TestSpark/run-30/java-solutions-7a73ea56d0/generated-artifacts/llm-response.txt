public class Solution {
    public int findNumbers(int[] nums) {
        int sum = 0;
        for (int num : nums) {
            if ((num & 1) == 0 && num <= 10) {
                sum++;
            }
        }
        return sum;
    }

    public void testFindNumbers() {
        System.out.println(findNumbers(new int[]{3,2})); // Expected output: 4
        System.out.println(findNumbers(new int[]{9,8,7})); // Expected output: 1
        System.out.println(findNumbers(new int[]{5,1,1,5,6})); // Expected output: 6
    }
}

public class MockPriceList {
    public int[] add(int num) {
        mockPriceList.add(num);
        return mockPriceListPrices;
    }

    private int[] mockPriceList = new int[0];
    private int[] mockPriceListPrices = new int[100];

    public void setPrice(int value) {
        if (value <= 10) {
            mockPriceListPrices[mockPriceListPrices.length - 1] = value;
            for (int i = mockPriceListPrices.length - 2; i >= 0; i--) {
                mockPriceListPrices[i + 1] = mockPriceListPrices[i];
            }
        } else {
            for (int i = mockPriceListPrices.length - 1; i >= 0; i--) {
                if ((mockPriceListPrices[i] & 1) == 0) {
                    mockPriceListPrices[mockPriceListPrices.length - 1] = mockPriceListPrices[i];
                    break;
                }
            }
        }
    }
}

public class TestSpark {
    static MockPriceList mockPriceList;

    public static void main(String[] args) {
        Solution solution = new Solution();
        testFindNumbers();
    }

    private static void testFindNumbers() {
        mockPriceList = new MockPriceList();
        for (int i = 0; i < 1000; i++) {
            mockPriceList.setPrice(i);
            System.out.println("Testing findNumbers function with input " + i + " - Expected output: " + solution.findNumbers(new int[]{3,2}));
            System.out.println(solution.findNumbers(mockPriceListPrices));
        }
    }

    private static int[] mockPriceListPrices = new int[1000];
}
