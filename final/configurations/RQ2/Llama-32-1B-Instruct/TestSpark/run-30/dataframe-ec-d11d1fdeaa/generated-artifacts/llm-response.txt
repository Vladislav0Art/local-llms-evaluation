import java.util.*;

public class MyTable {
    private List<String> names;
    private int ages;

    public MyTable(List<String> names, int ages) {
        this.names = names;
        this.ages = ages;
    }

    public String getName() {
        return names.get(0);
    }

    public int getAge() {
        return ages;
    }
}

public class DfCursor {
    private List<MyTable> rows;

    public void add(MyTable row) {
        rows.add(row);
    }

    public boolean next() {
        if (rows.isEmpty()) {
            throw new NoSuchElementException("No more rows");
        }
        MyTable row = rows.remove(0);
        return true;
    }

    public int getSize() {
        return rows.size();
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder("[\n");
        for (MyTable row : rows) {
            sb.append("  ").append(row.getName()).append(": ").append(row.getAge()).append("\n");
        }
        sb.append("]\n");
        return sb.toString();
    }

    public MyTable getRow(int index) {
        if (index >= getSize()) {
            throw new IndexOutOfBoundsException("Index out of bounds");
        }
        return rows.get(index);
    }
}

public class MyJoinExample {
    private DfCursor joinCursor;

    public void join(MyTable table, List<String> otherNames) {
        if (otherNames == null || otherNames.isEmpty()) {
            throw new NullPointerException("Other names cannot be empty");
        }

        for (String name : otherNames) {
            int age = Integer.parseInt(name.split("_")[1]);
            MyTable row1 = new MyTable(Arrays.asList(table.getName(), String.valueOf(age)), age);
            joinCursor.add(row1);
        }
    }

    public DfResult getDfResult() {
        return joinCursor.getCursor();
    }

    public void printRows(DfResult dfResult) {
        for (MyTable row : dfResult.rows()) {
            System.out.println("Name: " + row.getName());
            for (int i = 0; i < row.getAge(); i++) {
                System.out.print("Age: " + row.getAge() - i);
                if (i < row.getAge() - 1) {
                    System.out.print(", ");
                }
            }
            System.out.println();
        }
    }

    public static void main(String[] args) {
        List<String> names = new ArrayList<>();
        for (int i = 0; i < 10; i++) {
            names.add("Name_" + i);
        }
        int ages1_1 = 25;
        int ages1_2 = 30;

        MyTable table1 = new MyTable(names, ages1_1);
        DfJoin join1 = new DfJoin("MyTable1", "join", Arrays.asList(names, ages1_1));
        MyTable row1 = new MyTable(Arrays.asList(table1.getName(), String.valueOf(ages1_1)), ages1_1);

        List<String> names2 = new ArrayList<>();
        for (int i = 0; i < 10; i++) {
            names2.add("Name_" + i);
        }
        int ages2_2 = 20;
        int ages2_3 = 35;

        MyTable table2 = new MyTable(names2, ages2_2);
        row1 = new MyTable(Arrays.asList(table1.getName(), String.valueOf(ages1_1)), ages1_1);

        DfJoin join2 = new DfJoin("MyTable2", "join", Arrays.asList(names2, ages2_2));
        MyTable row2 = new MyTable(Arrays.asList(table2.getName(), String.valueOf(ages2_2)), ages2_2);

        join1.join(table1, names2);
        join2.join(table2, names2);

        DfResult dfResult = join1.getDfResult();
        System.out.println("Join 1 Results:");
        dfResult.printRows(dfResult.rows());

        System.out.println("\nJoin 2 Results:");
        dfResult.printRows(dfResult.rows());
    }
}
