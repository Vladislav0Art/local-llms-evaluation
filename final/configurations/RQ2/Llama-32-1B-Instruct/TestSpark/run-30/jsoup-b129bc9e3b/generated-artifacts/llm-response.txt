            Document document = Jsoup.parse(new URL(url));
            if (args.length == 0) {
                return new Response(200, "Hello World!", null);
            } else {
                return new Response(200, "Hello World!" + args[0], null);
            }
        }

        public Method method(Method method) {
            return mockConnection.method(method).getCode();
        }

        public boolean ignoreHttpErrors(boolean ignore) {
            return (boolean) mockConnection.ignoreHttpErrors(ignore).getCode();
        }

        public boolean ignoreContentType(boolean ignore) {
            return (boolean) mockConnection.ignoreContentType(ignore).getCode();
        }

        public int data(String key, String... args) {
            if (args.length == 0) {
                return 1024 * 8;
            } else {
                return 1024 * 8 + args[0].length();
            }
        }

        public boolean followRedirects(boolean follow) {
            return (boolean) mockConnection.followRedirects(follow).getCode();
        }

        public String getBody() throws MalformedURLException, URISyntaxException {
            Document document = Jsoup.parse(new URL("http://example.com"));
            return document.text().trim();
        }
    }

    public static class Response {
        private int code;
        private String message;

        public Response(int code, String message) {
            this.code = code;
            this.message = message;
        }

        public int getCode() {
            return code;
        }

        public String getMessage() {
            return message;
        }
    }

    public static class Method {
        private String name;

        public Method(String name) {
            this.name = name;
        }

        public String getName() {
            return name;
        }
    }

import java.util.HashMap;
import java.util.Map;

public class Main {
    private Map<String, Request> requests = new HashMap<>();

    public void setRequest(String url, String path, Request request) {
        requests.put(url, request);
    }

    public Request getRequest(String url) {
        return requests.getOrDefault(url, null);
    }

    public static class Request {
        private String path;

        public Request(String path) {
            this.path = path;
        }

        public String getPath() {
            return path;
        }
    }

    public void testGetRequest() {
        // Test case 1
        Set<String> expectedUrls = new HashSet<>();
        expectedUrls.add("/path");
        setRequest("/", "/path", Request.create("GET", "/path"));

        Request response = getRequest("/path");
        Assert.assertTrue(response != null && response.getPath().equals("GET /path"));
        Assert.assertTrue(expectedUrls.contains("/path"));

        // Test case 2
        Set<String> expectedUrls = new HashSet<>();
        expectedUrls.add("/");
        setRequest("", "", Request.create("GET", ""));

        Response response = getRequest("");
        Assert.assertTrue(response != null && response.getPath().equals(""));
        Assert.assertTrue(expectedUrls.contains("/"));

        // Test case 3
        Set<String> expectedUrls = new HashSet<>();
        expectedUrls.add("path");
        setRequest("/path", "path", Request.create("GET", "path"));

        Response response = getRequest("/path");
        Assert.assertTrue(response != null && response.getPath().equals("GET /path"));
        Assert.assertTrue(expectedUrls.contains("/path"));

        // Test case 4
        Set<String> expectedUrls = new HashSet<>();
        expectedUrls.add("/ignored");
        setRequest("", "", Request.create("GET", "/ignored"));

        Response response = getRequest("");
        Assert.assertTrue(response != null);
    }

    public static class Response {
        private String method;
        private String path;

        public Response(String method, String path) {
            this.method = method;
            this.path = path;
        }

        public String getMethod() {
            return method;
        }

        public String getPath() {
            return path;
        }
    }

    public static void main(String[] args) {
        Main main = new Main();
        main.testGetRequest();
    }

    private static Request create(String method, String path) {
        if (method.equals("GET")) {
            return new Response(method, path);
        } else if (method.equals("POST")) {
            return new Response(method, path);
        } else if (method.equals("PUT")) {
            return new Response(method, path);
        } else if (method.equals("DELETE")) {
            return new Response(method, path);
        }
        return null;
    }
}

You can add or remove methods as per your requirements by modifying the `create` method in the `Main` class.
