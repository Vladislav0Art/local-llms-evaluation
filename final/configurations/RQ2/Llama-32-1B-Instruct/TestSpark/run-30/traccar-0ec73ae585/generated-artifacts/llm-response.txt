    public static void addTag(String tag) {
        if (tag != null && !tag.isEmpty()) {
            tags.add(tag);
        }
    }

    public static void removeTag(String tag) {
        if (!tags.remove(tag)) {
            System.out.println("Tag not found in the list.");
        }
    }

    public static List<String> getTags() {
        return tags;
    }
}
