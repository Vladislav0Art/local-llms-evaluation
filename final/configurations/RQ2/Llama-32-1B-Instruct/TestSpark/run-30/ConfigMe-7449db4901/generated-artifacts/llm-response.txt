import java.io.FileNotFoundException;
import java.util.ArrayList;
import java.util.List;

public class CommentsConfigurationImpl {

    private SettingsHolder settingsHolder;
    private Map<String, List<String>> comments;

    public CommentsConfigurationImpl(SettingsHolder settingsHolder) {
        this.settingsHolder = settingsHolder;
        this.comments = new HashMap<>();
    }

    /**
     * Sets a comment for the specified path.
     *
     * @param path         the path to set the comment for
     * @param commentLines the lines of text that form the comment
     */
    public void setComment(String path, String[] commentLines) {
        if (settingsHolder.settings.containsKey(path)) {
            throw new RuntimeException("The setting '" + path + "' already exists.");
        }
        settingsHolder.settings.put(path, commentLines);
        this.comments.put(path, new ArrayList<>(Arrays.asList(commentLines)));
    }

    /**
     * Retrieves all comments for the specified paths.
     *
     * @return a list of strings containing all comments
     */
    public List<String> getAllComments() {
        return new ArrayList<>();
    }
}

public class SettingsHolder {

    private Map<String, Object> settings;

    public void setPath(String path) {
        if (settings.containsKey(path)) {
            throw new RuntimeException("The setting '" + path + "' already exists.");
        }
        settings.put(path, null);
    }

    public void addSetting(String key, String value) {
        if (!settings.containsKey(key)) {
            throw new RuntimeException("The setting '" + key + "' does not exist.");
        }
        settings.put(key, value);
    }
}

public class ConsoleApp {

    public static void main(String[] args) throws FileNotFoundException {
        SettingsHolder settingsHolder = new SettingsHolder();
        File file = new File("test.txt");
        for (int i = 0; i < 5; i++) {
            settingsHolder.addSetting(i + "_key", "value_" + i);
        }
        settingsHolder.setPath("settings");

        CommentsConfigurationImpl commentsConfig = new CommentsConfigurationImpl(settingsHolder);

        String path1 = "path1";
        String[] commentLines1 = {"line1_1", "line1_2"};
        commentsConfig.setComment(path1, commentLines1);

        List<String> allComments = commentsConfig.getAllComments();
        for (String line : allComments) {
            System.out.println(line);
        }
    }
}
