import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class JsoupSafetyTest {

    public static void main(String[] args) {
        Document doc = Jsoup.parse("https://www.example.com");
        System.out.println(doc.title());
        System.out.println(doc.body().text());
    }

    public static String generateString(int length) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < length; i++) {
            sb.append((char) ('a' + (int) Math.floor(Math.random() * 26)));
        }
        return sb.toString();
    }

    public static void testGenerateString() {
        String string = generateString(10);
        System.out.println(string.length());
    }

    public static boolean checkDomStructure(String html) {
        Document doc = Jsoup.parse(html);
        if (doc.body().tagName() != "body") {
            return false;
        }
        for (int i = 0; i < 5; i++) {
            String childTag = doc.select("h1").get(0).tagName();
            if (!childTag.equals("p")) {
                return false;
            }
        }
        return true;
    }

    public static void testCheckDomStructure() {
        String html = "<html><head></head><body><h1>Hello</h1><h2>World!</h2></body></html>";
        System.out.println(checkDomStructure(html));
    }

    public static int getRandomNumber(int min, int max) {
        return (int) Math.random() * (max - min + 1) + min;
    }

    public static void testGetRandomNumber() {
        System.out.println(getRandomNumber(1, 100));
    }
}
