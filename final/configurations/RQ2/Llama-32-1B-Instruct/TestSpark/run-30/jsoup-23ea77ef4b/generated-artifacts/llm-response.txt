            String title = body.attr("title");

            // Check if the page has a class attribute
            if (body.hasAttr("class")) {
                String[] classes = body.attr("class").split("\\s+");
                
                for (String class_ : classes) {
                    System.out.print(class_);
                }
                System.out.println();
            }

            // Call the testBody method
            sparkTest.testBody(body);
        } else {
            System.out.println("No body tag found");
        }

            String[] classes = body.attr("class").split("\\s+");
            
            for (String class_ : classes) {
                System.out.print(class_);
            }

        }
