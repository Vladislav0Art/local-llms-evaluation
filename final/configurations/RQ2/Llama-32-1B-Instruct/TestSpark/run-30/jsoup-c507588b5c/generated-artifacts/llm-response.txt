import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class TextNode {
    private String text;

    public TextNode(String text) {
        this.text = text;
    }

    public static String nodeName() {
        return "text";
    }

    public static String text() {
        return text;
    }

    public Document clone() {
        return new Document();
    }

    public boolean isBlank() {
        return !this.text().trim().isEmpty();
    }

    public String splitText(StringBuilder sb) {
        return this.text().split("\\s+");
    }

    public String[] splitTextWithOneElement(StringBuilder sb) {
        if (this.text().isEmpty()) {
            return new String[]{sb.toString()};
        }
        int lastIndex = 0;
        while (lastIndex < sb.length() && !isBlank(sb.charAt(lastIndex))) {
            lastIndex++;
        }
        StringBuilder result = new StringBuilder();
        for (int i = 0; i <= lastIndex; i++) {
            if (!isBlank(sb.charAt(i))) {
                result.append(sb.charAt(i));
            }
        }
        return new String[]{result.toString()};
    }

    public String outerHtmlHead() {
        Document doc = new Document();
        Elements elements = doc.getElementById("root").children;
        for (int i = 0; i < elements.size(); i++) {
            Element element = (Element) elements.get(i);
            if ("text".equals(element.attr("data-name"))) {
                element.attr("class", "outer-text");
            }
        }
        return doc.outerHtml();
    }

    public String outerHtmlTail() {
        Document doc = new Document();
        Elements elements = doc.getElementById("root").children;
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < elements.size(); i++) {
            Element element = (Element) elements.get(i);
            if ("text".equals(element.attr("data-name"))) {
                element.attr("class", "inner-text");
            }
        }
        for (int i = 0; i < elements.size() - 1; i++) {
            sb.append("<p>").append(elements.get(i)).append("</p>");
        }
        return doc.outerHtml().replaceFirst("<body>.*</body>", "<body><p>" + sb.toString() + "</p></body>");
    }

    public String innerText() {
        Elements elements = doc.getElementById("text").children;
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < elements.size(); i++) {
            Element element = (Element) elements.get(i);
            if ("text".equals(element.attr("data-name"))) {
                sb.append(element.text());
            }
        }
        return sb.toString();
    }

    public static String join(StringBuilder sb, String sep) {
        while (sb.length() > 0 && !isBlank(sb.charAt(sb.length() - 1))) {
            sb.append(sep);
        }
        return sb.toString();
    }
}
