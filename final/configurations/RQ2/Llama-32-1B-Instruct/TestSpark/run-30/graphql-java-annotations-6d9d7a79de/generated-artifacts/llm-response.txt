import org.junit.Assert;
import org.junit.Test;

public class Directives {
    public static int checkEqual(int[] array1, int[] array2) {
        for (int i : array1) {
            if (i != array2[i]) {
                return 0;
            }
        }
        return 1;
    }

    @Test
    public void testCheckEqual() {
        Assert.assertEquals(1, checkEqual(array1, array2));
    }

    public static int[] generateArray(int size) {
        int[] array = new int[size];
        for (int i = 0; i < size; i++) {
            array[i] = (int) (Math.random() * 100);
        }
        return array;
    }

    @Test
    public void testCheckEqual2() {
        Assert.assertEquals(1, checkEqual(generateArray(10), generateArray(20)));
    }
}
