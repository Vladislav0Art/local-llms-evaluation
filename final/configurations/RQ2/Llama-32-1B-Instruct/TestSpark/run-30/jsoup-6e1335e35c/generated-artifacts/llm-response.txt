class Node {

    private String text;
    private String className;

    public Node(String text, String className) {
        this.text = text;
        this.className = className;
    }

    @Override
    public String toString() {
        return "Node{" +
                "text='" + text + '\'' +
                ", className='" + className + '\'' +
                '}';
    }
}

// NodeList class implementation

class NodeList {

    private List<Node> nodes;

    public NodeList(List<Node> nodes) {
        this.nodes = nodes;
    }

    public String text() {
        return nodes.stream().map(Node::text).collect(String.join(", ", Arrays.asList()));
    }

    public void clearAttributes() {
        for (Node node : nodes) {
            node.removeAttr("class");
        }
    }

    @Override
    public boolean equals(Object other) {
        if (this == other)
            return true;
        if (other == null || getClass() != other.getClass())
            return false;

        NodeList that = (NodeList) other;

        return nodes.size() == that.nodes.size() &&
               nodes.stream().allMatch(n -> n.equals(that.nodes.get(n)));
    }

    @Override
    public int hashCode() {
        return Objects.hash(nodes);
    }
}

public abstract class Node {
    private String name;

    public Node(String name) {
        this.name = name;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        Node node = (Node) o;
        return Objects.equals(name, node.name);
    }

    @Override
    public int hashCode() {
        return Objects.hash(name);
    }
}

public class NodeList extends Node {

    private List<Node> nodes;

    public NodeList(List<Node> nodes) {
        super("node1");
        this.nodes = nodes;
    }

    public void addNode(Node node) {
        if (nodes == null) nodes = new ArrayList<>();
        nodes.add(node);
    }
}

public class GeneratedTest extends TestSparkBase {

    @Override
    protected String generateInput(String[] expectedOutput) throws Exception {
        return "node1\n";
    }

    @Override
    protected String generateOutput() {
        StringBuilder sb = new StringBuilder();
        for (Node node : nodes) {
            sb.append(node.toString()).append("\n");
        }
        return sb.toString().trim();
    }

    public void testAddNode() throws Exception {
        NodeList nodeList = new NodeList(Arrays.asList(new Node("node1"), new Node("node2")));
        runner.addNode(nodeList);
        String output = runner.generateOutput();
        assertEquals("node1, class1\nnode2, class2", output);
    }
}

public class NodeFactory {

    public static NodeList createNodeList(List<Node> nodes) {
        return new NodeList(nodes);
    }

    public interface NodeInput {
        List<String> getExpectedOutput();
        String[] generateInput(String[] expectedOutput);
    }

    public static class NodeData implements NodeInput {

        @Override
        public List<String> getExpectedOutput() {
            return Arrays.asList("node1, class1", "node2, class2");
        }

        @Override
        public String[] generateInput(String[] expectedOutput) throws Exception {
            return expectedOutput;
        }
    }

    public static void main(String[] args) throws Exception {
        NodeData data = new NodeData();
        NodeList nodeList = NodeFactory.createNodeList(data.generateInput(data.getExpectedOutput()));
        runner.addNode(nodeList);
        String output = runner.generateOutput();
        System.out.println(output);
    }
}
