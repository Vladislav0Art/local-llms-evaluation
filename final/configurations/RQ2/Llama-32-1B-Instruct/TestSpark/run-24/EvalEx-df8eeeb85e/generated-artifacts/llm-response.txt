public class TokenizerTest {

    private Tokenizer tokenizer;

    @Before
    public void setup() {
        String expressionString = "1 + 2";
        ExpressionConfiguration configuration = new DefaultExpressionConfiguration();
        tokenizer = new Tokenizer(expressionString, configuration);
    }

    @Test
    public void testParseStringLiteral() {
        String expressionString = "1 + 2";
        List<Token> expectedTokens = new ArrayList<>();
        expectedTokens.add(new SingleNumberToken(1));
        expectedTokens.add(new BinaryPlusToken(" + ", 2));
        assertEquals(expectedTokens, tokenizer.parseStringLiteral());
    }

    @Test
    public void testParseIntegerLiteral() {
        String expressionString = "123";
        List<Token> expectedTokens = new ArrayList<>();
        expectedTokens.add(new IntegerNumberToken(123));
        assertEquals(expectedTokens, tokenizer.parseIntegerLiteral(expressionString));
    }

    @Test
    public void testParseFloatLiteral() {
        String expressionString = "123.456";
        List<Token> expectedTokens = new ArrayList<>();
        expectedTokens.add(new FloatNumberToken(123.456));
        assertEquals(expectedTokens, tokenizer.parseFloatLiteral(expressionString));
    }

    @Test
    public void testParseIdentifier() {
        String expressionString = "_x";
        List<Token> expectedTokens = new ArrayList<>();
        expectedTokens.add(new IdentifierToken("_x"));
        assertEquals(expectedTokens, tokenizer.parseIdentifier(expressionString));
    }

    @Test
    public void testParseVariableReference() {
        String expressionString = "x";
        List<Token> expectedTokens = new ArrayList<>();
        expectedTokens.add(new VariableReferenceToken("x"));
        assertEquals(expectedTokens, tokenizer.parseVariableReference(expressionString));
    }

    @Test
    public void testParseFunctionCall() {
        String expressionString = "F(x)";
        List<Token> expectedTokens = new ArrayList<>();
        expectedTokens.add(new FunctionCallToken(F));
        expectedTokens.add(new VariableReferenceToken("x"));
        assertEquals(expectedTokens, tokenizer.parseFunctionCall(expressionString));
    }

    @Test
    public void testParseOperator() {
        String expressionString = "+";
        List<Token> expectedTokens = new ArrayList<>();
        expectedTokens.add(new OperatorToken("+", " + "));
        assertEquals(expectedTokens, tokenizer.parseOperator(expressionString));
    }

    @Test
    public void testParseParentheses() {
        String expressionString = "(1)";
        List<Token> expectedTokens = new ArrayList<>();
        expectedTokens.add(new LeftParenthesisToken("("));
        expectedTokens.add(new NumberToken(1));
        expectedTokens.add(new RightParenthesisToken(")"));
        assertEquals(expectedTokens, tokenizer.parseParentheses(expressionString));
    }

    @Test
    public void testParseExpression() {
        String expressionString = "(1 + 2)";
        List<Token> expectedTokens = new ArrayList<>();
        expectedTokens.add(new LeftParenthesisToken("("));
        expectedTokens.add(new BinaryPlusToken(" + ", 2));
        expectedTokens.add(new NumberToken(1));
        expectedTokens.add(new RightParenthesisToken(")"));
        assertEquals(expectedTokens, tokenizer.parseExpression(expressionString));
    }
}

    }

        assertTrue(tokenizer.parse("").isEmpty());
        assertEquals(0, tokenizer.getTokens().size());
    }

        assertTrue(tokenizer.parse("null").isEmpty());
        assertEquals(0, tokenizer.getTokens().size());
    }

    @Test
    public void testParseNoneLiteral() {
        assertTrue(tokenizer.parse("None").isEmpty());
        assertEquals(0, tokenizer.getTokens().size());
    }

    @Test
    public void testParseEmptyExpression() {
        assertTrue(tokenizer.parse("(a) b").isEmpty());
        assertEquals(1, tokenizer.getTokens().size());
        assertEquals("(", 0);
        assertEquals(",", 1);
        assertEquals(")", 2);
    }

    @Test
    public void testParseStringWithPunctuation() {
        assertTrue(tokenizer.parse("a,b,c").isEmpty());
        assertEquals(3, tokenizer.getTokens().size());
        assertEquals(" ", 0);
        assertEquals(",", 1);
        assertEquals(")", 2);
    }

    @Test
    public void testParseStringWithoutPunctuation() {
        assertTrue(tokenizer.parse("a,b").isEmpty());
        assertEquals(3, tokenizer.getTokens().size());

        assertEquals(")", 2);
    }

    public void testParseFloats() {
