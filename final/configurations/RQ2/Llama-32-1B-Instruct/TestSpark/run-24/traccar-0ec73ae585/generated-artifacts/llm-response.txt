class CellTower {

    private List<Network> networks;

    public CellTower(List<Network> networks) {
        this.networks = networks;
    }

    public int getCellTowerIndex() {
        return networksWithoutCellTower().size();
    }

    private List<Network> networksWithoutCellTower() {
        return networks.stream()
            .filter(n -> n.cellTower == null)
            .collect(Collectors.toList());
    }

class Network {

    private SocketAddress address;
    private CellTower cellTower;

    public Network(SocketAddress address, CellTower cellTower) {
        this.address = address;
        this.cellTower = cellTower;
    }

    public void setCellTowerIndex(int index) {
        if (cellTower != null && cellTower.getCellTowerIndex() != index) {
            throw new RuntimeException("A cell tower's index cannot be changed after it is created.");
        }
        cellTower.setCellTowerIndex(index);
    }

    public int getCellTowerIndex() {
        return cellTower.getCellTowerIndex();
    }

    @Override
    public String toString() {
        return "Network{" +
                "address=" + address +
                ", cellTower=" + cellTower +
