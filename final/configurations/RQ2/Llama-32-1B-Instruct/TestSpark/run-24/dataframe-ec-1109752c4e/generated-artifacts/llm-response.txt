import java.util.ArrayList;
import java.util.List;

public class Main {

    public static String addStrings(String s1, String s2) {
        List<Integer> list = new ArrayList<>();
        for (int i = 0; i < Math.min(s1.length(), s2.length()); i++) {
            int num1 = Integer.parseInt(s1.substring(i, i + 1));
            int num2 = Integer.parseInt(s2.substring(i, i + 1));
            list.add(num1 + num2);
        }
        if (s1.length() > s2.length()) {
            return String.join("", list.subList(list.size(), list.size() + s1.length()));
        } else if (s2.length() > s1.length()) {
            return String.join("", list.subList(0, list.size() - Math.min(s2.length(), s1.length())));
        } else {
            return String.join("", list);
        }
    }

    public static String findSecondMax(String s) {
        List<Integer> list = new ArrayList<>();
        for (char c : s.toCharArray()) {
            if ("0123456789".indexOf(c) != -1) {
                list.add(Character.getNumericValue(c));
            }
        }
        Collections.sort(list);
        return list.size() > 2 ? String.valueOf(list.get(1)) + " " + list.get(0) : null;
    }

    public static boolean checkEqual(String s1, String s2) {
        if (s1.equals(s2)) {
            return true;
        }
        for (int i = 0; i < Math.min(s1.length(), s2.length()); i++) {
            if (s1.charAt(i) != s2.charAt(i)) {
                return false;
            }
        }
        return true;
    }

    public static boolean checkEqual(String s1, String s2, int n) {
        for (int i = 0; i < Math.min(s1.length(), s2.length()); i++) {
            if (!checkEqual(s1.substring(0, i + 1), s2.substring(0, i + 1), n)) {
                return false;
            }
        }
        return true;
    }

    public static void main(String[] args) {
        System.out.println(addStrings("123", "456")); // Output: "579"
        System.out.println(findSecondMax("abc")); // Output: "c" or null if there is no second max
        System.out.println(checkEqual("1234", "5678")); // True
        System.out.println(checkEqual("abcde", "cdefg", 3)); // False
    }
}

====================== Iteration #3 ======================
Here's a new solution for your original question:

The given strings `s1` and `s2` can be concatenated to form a single string `s`. This is because concatenation is associative, meaning that the order in which we concatenate two strings does not change the result.

Therefore, if `s1` and `s2` are two different strings, then `addStrings(s1, s2)` will always return an empty string.

Here's a sample code snippet to demonstrate this:

```java
public static void main(String[] args) {
    System.out.println(addStrings("123", "456")); // Output: ""
}
```

This solution has a time complexity of O(n), where n is the length of `s1` and `s2`, since we need to iterate over all characters in both strings.

import java.util.HashMap;
import java.util.Map;

public class Main {

    public static String addStrings(String s1, String s2) {
        Map<Character, Integer> map = new HashMap<>();
        
        // Count the occurrence of each character in both strings
        for (char c : s1.toCharArray()) {
            map.put(c, map.getOrDefault(c, 0) + 1);
        }
        for (char c : s2.toCharArray()) {
            map.put(c, map.getOrDefault(c, 0) + 1);
        }

        // Initialize the result string and its length
        StringBuilder sb = new StringBuilder();
        int i = 0;

        while (i < Math.min(s1.length(), s2.length())) {
            char c1 = s1.charAt(i++);
            char c2 = s2.charAt(i++);
            if (map.containsKey(c1) && map.containsKey(c2)) {
                // If both characters are present in both strings, find the smaller one
                int num1 = Math.min(map.get(c1), map.get(c2));
                sb.append(num1);
                map.put(c1, map.get(c1) - num1);
                map.put(c2, map.get(c2) - num1);
            } else if (map.containsKey(c1)) {
                // If only one character is present in the first string, append it
                sb.append(map.get(c1));
                map.put(c1, 0);
            } else {
                // If only one character is present in the second string, append it
                sb.append(map.get(c2));
                map.put(c2, 0);
            }
        }

        return sb.toString();
    }

    public static String findSecondMax(String s) {
        int[] arr = new int[128];
        
        // Count the occurrence of each character in the string
        for (char c : s.toCharArray()) {
            if ('a' <= c && c <= 'z') {
                arr[c] += 1;
            }
        }

        // Find the second maximum number
        String max = "";
        int secMax = Integer.MIN_VALUE;
        for (int i = 0; i < 128; i++) {
            if (arr[i] != 0) {
                if (arr[i] > secMax) {
                    secMax = arr[i];
                } else if (arr[i] == secMax) {
                    // If the current number is equal to the second maximum, increment it
                    secMax += 1;
                }
            }
        }

        return max + " " + secMax;
    }

    public static boolean checkEqual(String s1, String s2) {
        return s1.equals(s2);
    }

    public static boolean checkEqual(String s1, String s2, int n) {
        return checkEqual(s1, s2) && n <= 10;
    }

    public static void main(String[] args) {
        System.out.println(addStrings("123", "456")); // Output: "579"
        System.out.println(findSecondMax("abc")); // Output: "c" or null if there is no second max
        System.out.println(checkEqual("1234", "5678")); // True
        System.out.println(checkEqual("abcde", "cdefg", 3)); // False
    }
}

import java.util.*;

public class Main {

    public static String findSecondMax(String s) {
        int[] arr = new int[256];
        
        // Count the occurrence of each character in the string
        for (char c : s.toCharArray()) {
            if ('a' <= c && c <= 'z') {
                arr[c]++;
            }
        }

        // Find the second maximum number
        String max = "";
        int secMax = Integer.MIN_VALUE;
        for (int i = 0; i < 256; i++) {
            if (arr[i] != 0) {
                if (arr[i] > secMax) {
                    secMax = arr[i];
                } else if (arr[i] == secMax) {
                    // If the current number is equal to the second maximum, increment it
                    secMax += 1;
                }
            }
        }

        return max + " " + secMax;
    }

    public static int binarySearch(int[] arr, int target) {
        int low = 0;
        int high = arr.length - 1;

        while (low <= high) {
            int mid = low + (high - low) / 2;

            if (arr[mid] == target) {
                return mid;
            } else if (arr[mid] < target) {
                low = mid + 1;
            } else {
                high = mid - 1;
            }
        }

        return -1;
    }

    public static String findFirstOccurrence(String s, char c) {
        int i = s.indexOf(c);
        while (i != -1) {
            if (s.substring(i, i + 1).equals(new StringBuilder().reverse().toString())) {
                return s.substring(0, i + 1);
            }
            i = s.indexOf(c, i + 1);
        }

        return "";
    }

    public static boolean checkStringHasAllCharacters(String s) {
        for (char c : "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789".toCharArray()) {
            if (!s.contains(c)) {
                return false;
            }
        }
        return true;
    }

    public static List<String> findStringsWithSpecificChars(String s, String chars) {
        List<String> result = new ArrayList<>();
        for (char c : chars.toCharArray()) {
            result.add(s + c);
        }
        return result;
    }

    public static int factorial(int n) {
        if (n == 0 || n == 1) {
            return 1;
        } else {
            return n * factorial(n - 1);
        }
    }

    public static boolean isPrime(int num) {
        if (num <= 1) {
            return false;
        }

        for (int i = 2; i <= Math.sqrt(num); i++) {
            if (num % i == 0) {
                return false;
            }
        }

        return true;
    }

    public static void main(String[] args) {
        System.out.println(findSecondMax("abcde")); // Output: "c" or null if there is no second max
        System.out.println(checkStringHasAllCharacters("abcabc")); // Output: true
        System.out.println(findStringsWithSpecificChars("hello world", "l")); // Output: ["hehllo"]
    }
}
