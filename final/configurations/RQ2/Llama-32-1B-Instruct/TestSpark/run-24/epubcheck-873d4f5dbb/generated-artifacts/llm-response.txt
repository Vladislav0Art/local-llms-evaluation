import java.util.Map;

public class Main {
    public static String getPreferredMediaType(String prefix, String type) {
        Map<String, String> types = new HashMap<>();
        types.put("audio", "audio");
        types.put("video", "video");
        types.put("font", "font");
        types.put("script", "script");

        return types.get(prefix).equals(type) ? prefix + "/" + type : getPreferredMediaType(prefix + "/"+type, type);
    }

    public static void testGetPreferredMediaType() {
        System.out.println(getPreferredMediaType("audio", "video")); // Should print 'audio/video'
        System.out.println(getPreferredMediaType("prefix:audio video/test", "video")); // Should print 'audio/video/test'
        System.out.println(getPreferredMediaType(prefix: "coremedia-type", type: "audio")); // Should print 'coremedia-type/audio'
    }
