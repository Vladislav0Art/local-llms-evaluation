```java
import java.util.HashMap;
import java.util.Map;

public class Main {

    public static void main(String[] args) {
        Map<String, Object> data1 = new HashMap<>();
        data1.put("name", "John");
        data1.put("age", 30);

        Map<String, Object> data2 = new HashMap<>();
        data2.put("name", null);
        data2.put("age", 25);

        System.out.println(buildUrl(data1));
        System.out.println(buildUrl(data2));

        String query = buildQuery(data1);
        System.out.println("Build Query: " + query);

        String url = buildUrl(data2);
        System.out.println("Build URL: " + url);
    }

    public static String buildQuery(Map<String, Object> data) {
        StringBuilder query = new StringBuilder();
        for (Map.Entry<String, Object> entry : data.entrySet()) {
            if (entry.getValue() != null) {
                query.append("=").append(String.valueOf(entry.getKey())).append("&");
            }
        }
        return query.length() > 0 ? query.toString().trim() + "&" : "";
    }

    public static String buildUrl(Map<String, Object> data) {
        Map<String, Object> params = new HashMap<>();
        for (Map.Entry<String, Object> entry : data.entrySet()) {
            if (!entry.getValue().equals(null)) {
                params.put(entry.getKey(), entry.getValue());
            }
        }

        StringBuilder url = new StringBuilder("http://example.com/api");
        if (!params.isEmpty()) {
            String queryPart = buildQuery(params);
            if (queryPart.length() > 0) {
                url.append("?").append(queryPart).append("&");
            } else {
                url.append("?").append(buildQuery(params));
            }
        }
        return url.toString().trim();
    }
}
```
