public class TestTransformer {

    private String date;

    public void setDate(String date) {
        this.date = date;
    }

    public boolean shouldConvertToPublicClass() {
        return !date.isEmpty();
    }

    public String toString() {
        if (shouldConvertToPublicClass()) {
            return "TestTransformer: " + date;
        } else {
            return "TestTransformer: No Date";
        }
    }
}

class PublicClass {

    private TestTransformer transformer;

    public void setTransformer(TestTransformer transformer) {
        this.transformer = transformer;
    }

    @Override
    public String toString() {
        return transformer.toString();
    }
}

public class Main {

    public static void main(String[] args) {
        TestTransformer testTransformers = new TestTransformer();
        testTransformers.setDate("2022-01-01");
        PublicClass publicClass = new PublicClass();
        publicClass.setTransformer(testTransformers);
        System.out.println(publicClass.toString());
    }
}

public class TestTransformer {

    private String date;

    public void setDate(String date) {
        this.date = date;
    }

    public boolean shouldConvertToPublicClass() {
        return !date.isEmpty();
    }

    public String toString() {
        if (shouldConvertToPublicClass()) {
            return "TestTransformer: " + date;
        } else {
            return "TestTransformer: No Date";
        }
    }

    public static void main(String[] args) {
        TestTransformer testTransformers = new TestTransformer();
        testTransformers.setDate("2022-01-01");
        System.out.println("Should convert to public class: " + (testTransformers.shouldConvertToPublicClass()));
        
        // create an instance of the public class
        PublicClass publicClass = new PublicClass();

        // set the transformer in the public class
        publicClass.setTransformer(testTransformers);

        // print the public class
        System.out.println("Public Class: " + (publicClass.toString()));
    }
}

class PublicClass {

    private TestTransformer transformer;

    public void setTransformer(TestTransformer transformer) {
        this.transformer = transformer;
    }

    @Override
    public String toString() {
        return transformer.toString();
    }
}
