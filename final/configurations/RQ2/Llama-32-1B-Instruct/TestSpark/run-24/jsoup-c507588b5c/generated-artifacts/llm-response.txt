import org.jsoup.nodes.TextNode;
import org.junit.jupiter.api.AfterEach;
import org.junit.jupiter.api.BeforeEach;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.TextNodeBuilder;
import org.jsoup.select.Elements;

import java.io.IOException;

public class TextNodeTest {
    private Document document;
    private Elements elements;
    private Element element;

    @BeforeEach
    public void setup() throws IOException {
        document = new Document();
        elements = document.getAllElements();

        // Create text node with some content
        String content = "Hello World";
        TextNodeBuilder builder = TextNodeBuilder.create(content);
        for (int i = 0; i < 10; i++) {
            builder.append("\n").append(element).append("test");
        }
        element = document.createElementText("text");

        // Append the text node to the document
        document.appendChild(element);

        // Add some other elements to the document
        for (int i = 0; i < 20; i++) {
            Element elem = document.createElementText(i + "test");
            elements.add(elem);
        }
    }

    @AfterEach
    public void tearDown() throws IOException {
        if (elements.size() > 1) {
            // Remove all text nodes from the document
            for (int i = 0; i < elements.size(); i++) {
                document.removeChild(elements.get(i));
            }
        } else {
            document.clear();
        }
    }

    @Test
    public void test nodeName() throws IOException {
        TextNode node = createTextNode("Hello World");
        assertEquals("text", node.nodeName());
    }

    @Test
    public void test text() throws IOException {
        TextNode node = createTextNode("Hello World");
        assertEquals("Hello World", node.text());
    }

    @Test
    public void test text(String text) throws IOException {
        TextNode node = createTextNode(text);
        assertEquals(text, node.text());
    }

    @Test
    public void test getWholeText() throws IOException {
        TextNode node = createTextNode("Hello World");
        assertEquals("Hello World", node.getWholeText());
    }

    @Test
    public void test isBlank() throws IOException {
        TextNode node = createTextNode("");
        assertTrue(node.isBlank());
    }

    @Test
    public void test splitText(int offset) throws IOException {
        TextNode node = createTextNode("Hello World");
        assertEquals(5, node.splitText(0));
    }
}

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class TextNodeTest {

    public static void main(String[] args) {
        // Create a document
        Document document = new Document();

        // Add some elements to the document
        for (int i = 0; i < 20; i++) {
            Element elem = document.createElement("text");
            elem.setTextContent(i + "test");
            Elements childElements = document.getChildElements();
            childElements.add(elem);
        }

        // Create a text node with some content
        String content = "Hello World";
        TextNodeBuilder builder = TextNodeBuilder.create(content);

        // Append the text node to the document
        Element parentElement = document.getElementById("root");
        if (parentElement != null) {
            parentElement.appendChild(builder.append("\n").append(parentElement).append("test"));
        } else {
            parentElement = document.createElementText("text");
            parentElement.appendText(content);
            Elements childElements = document.getChildElements();
            childElements.add(parentElement);

            // Add the text node to the document
            Element element = document.createElementText("text");
            for (int i = 0; i < content.length(); i++) {
                element.appendChild(builder.append("\n").append(element).append(content.substring(0, i + 1)));
            }
            parentElement.appendChild(element);

        }

        // Get the text node from the document
        TextNode textNode = createTextNode(content);
        String text = textNode.getTextContent();

        // Test the nodeName()
        System.out.println("nodeName(): " + textNode.nodeName());

        // Test the text()
        System.out.println("text(): " + text);

        // Test the text(String)
        System.out.println("text(String): " + textNode.text(text));

        // Test the getWholeText()
        System.out.println("getWholeText(): " + textNode.getWholeText());
    }

    public static TextNode createTextNode(String content) {
        return new TextNode(content);
    }
}
