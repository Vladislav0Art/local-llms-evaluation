public class Test {
    private static final int COUNT = 100;

    public static void main(String[] args) {
        doTest();
    }

    public static void doTest() {
        for (int i = 0; i < COUNT; i++) {
            if (i % 2 == 0) {
                System.out.println("Even number: " + i);
            } else {
                System.out.println("Odd number: " + i);
            }
        }

        if (COUNT % 3 == 1) {
            System.out.println("Conditionally executed method");
            doConditionallyExecutedMethod();
        }

        for (int j = 0; j < COUNT; j++) {
            if (j % 2 == 0 && j < 50) {
                System.out.println("Even and less than 50: " + j);
            } else {
                System.out.println("Odd or greater than 50: " + j);
            }
        }

        doConditionallyExecutedMethod();
    }

    public static void doConditionallyExecutedMethod() {
        if (COUNT % 3 == 1) {
            System.out.println("Conditionally executed method");
        } else {
            // This will be skipped
        }
    }

    public static int add(int a, int b) {
        return a + b;
    }
}
