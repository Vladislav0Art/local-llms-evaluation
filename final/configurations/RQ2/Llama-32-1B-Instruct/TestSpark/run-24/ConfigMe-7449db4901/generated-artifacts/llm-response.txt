#### Method 1: Using Euclidean Algorithm
```java
public class Test7 {
    public static void main(String[] args) {
        System.out.println(myMethod3());
    }

    public static String myMethod3() {
        int num1 = 12;
        int num2 = 18;
        return "GCD: " + findGcd(num1, num2);
    }

    public static int findGcd(int a, int b) {
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
```
