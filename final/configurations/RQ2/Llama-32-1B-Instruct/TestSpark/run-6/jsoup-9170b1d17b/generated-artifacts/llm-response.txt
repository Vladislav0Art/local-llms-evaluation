import org.jsoup.Jsoup;

public class TestGenerator {
    public static void main(String[] args) throws Exception {

        System.out.println("Name present: " + namePresent);

        System.out.println("Alt present: " + altPresent);

        // Test values for non-existent attributes
        boolean idNonPresent = isAttributePresent(element, "nonexistent-id");
        System.out.println("ID not found: " + idNonPresent);
        String classNonPresent = isAttributePresent(element, "class-nonexistent");
        System.out.println("Class not found: " + classNonPresent);

        boolean styleNonPresent = isAttributePresent(element, "style-nonexistent");
        System.out.println("Style not found: " + styleNonPresent);

        // Test values for attributes with multiple values
        String[] attributes = {"id", "class"};
        printAttributeValues(element, attribute -> {
            System.out.print(attribute + ": ");
            return element.attr(attribute).split("; ").length > 1 ? element.attr(attribute) : "";
        }, "value1; value2");

        // Test values for attributes without space
        String[] attributes2 = {"name", "alt"};
        printAttributeValues(element, attribute -> {
            System.out.print(attribute + ": ");
            return element.attr(attribute);
        }, attribute -> (element.attr(attribute).equals("image of") ? null : element.attr(attribute)));
    }

    public static void printAttributeValues(Element element, Function<String, String> func) {
        for (String attribute : element.attributes()) {
            if (!attribute.equals("id")) {
                try {
                    System.out.print(func.apply(element.attr(attribute)) + ": ");
                } catch (Exception e) {
                    System.out.println(e.getMessage());
                }
            }
        }
    }

    private static boolean isAttributePresent(Element element, String attribute) {
        try {
            return element.hasAttr(attribute);
        } catch (Exception e) {
            return false;
        }
    }
}
