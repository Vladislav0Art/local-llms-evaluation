public class OnlineStockSpan {

    private List<Integer> record;
    private Stack<Integer> stack;

    public OnlineStockSpan() {
        this.record = new ArrayList<>();
        this.stack = new Stack<>();
    }

    public int next(int price) {
        while (!stack.isEmpty()) {
            if (record.get(stack.peek()) <= price) {
                stack.pop();
            } else {
                break;
            }
        }
        record.add(price);
        return record.size() - 1;
    }

    public int[] calculateSpans(int[] prices) {
        OnlineStockSpan instance = new OnlineStockSpan();
        for (int i = 0; i < prices.length; i++) {
            if (!instance.stack.isEmpty()) {
                instance.next(prices[i]);
                if (prices[i] > prices[record.get(instance.stack.peek())]) {
                    instance.record.add(record.get(instance.stack.pop()));
                } else {
                    instance.record.add(i + 1);
                }
            }
        }

        int[] spans = new int[prices.length];
        for (int i = 0; i < prices.length; i++) {
            spans[i] = record.get(i) - i;
        }

        return spans;
    }
}

public class OnlineStockSpan {

    private int[] record;
    private Stack<Integer> stack;

    public OnlineStockSpan() {
        this.record = new int[10000];
        this.stack = new Stack<>();
    }

    public int next(int price) {
        while (!stack.isEmpty()) {
            if (record[stack.peek()] <= price) {
                stack.pop();
            } else {
                break;
            }
        }
        record[stack.size()] = price;

        return stack.size();
    }

    public int[] calculateSpans(int[] prices) {
        OnlineStockSpan instance = new OnlineStockSpan();

        for (int i = 0; i < prices.length; i++) {
            if (!instance.stack.isEmpty()) {
                int span = next(prices[i]);
                if (prices[i] > prices[record.get(instance.stack.peek())]) {
                    instance.record[i + 1] = record[span];
                } else {
                    instance.record[i + 1] = i + 1;
                }
            }

            instance.record[i + 1] = i + 1;
        }

        return instance.record;
    }
}
