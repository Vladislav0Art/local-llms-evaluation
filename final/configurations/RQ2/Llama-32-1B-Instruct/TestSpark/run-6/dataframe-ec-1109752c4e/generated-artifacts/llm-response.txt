public class ECDslVisitor {
    public static class DataFrame {
        private List<Map<String, Object>> rows;
        private Map<Integer, Integer> columnIndex;

        public DataFrame(List<Map<String, Object>> rows, Map<Integer, Integer> columnIndex) {
            this.rows = rows;
            this.columnIndex = columnIndex;
        }

        public boolean isNumeric() {
            for (Map.Entry<String, Object> entry : columnIndex.entrySet()) {
                if (!(entry.getValue() instanceof Number)) {
                    return false;
                }
            }
            return true;
        }

        public List<Map<String, Object>> toList() {
            List<Map<String, Object>> result = new ArrayList<>();
            for (Map<String, Object> row : rows) {
                Map<String, Object> newRow = new HashMap<>(row);
                for (Map.Entry<Integer, Integer> entry : columnIndex.entrySet()) {
                    newRow.put(entry.getKey(), row.get(entry.getKey()));
                }
                result.add(newRow);
            }
            return result;
        }

        public static DataFrame parse(String str) throws Exception {
            List<Map<String, Object>> rows = new ArrayList<>();
            String[] columnsStr = str.split(",");
            for (String col : columnsStr) {
                if (!col.isEmpty()) {
                    Map<String, Object> row = new HashMap<>();
                    String[] valuesStr = col.substring(1).split(",");
                    for (int i = 0; i < valuesStr.length; i++) {
                        try {
                            row.put(i, Integer.parseInt(valuesStr[i]));
                        } catch (NumberFormatException e) {
                            throw new NumberFormatException("Invalid value: " + valuesStr[i]);
                        }
                    }
                    rows.add(row);
                }
            }
            Map<Integer, Integer> columnIndex = new HashMap<>();
            for (int i = 0; i < columnsStr.length; i++) {
                if (!columnsStr[i].isEmpty()) {
                    columnIndex.put(i, Integer.parseInt(columnsStr[i].substring(1)));
                }
            }
            return new DataFrame(rows, columnIndex);
        }

        public static void main(String[] args) throws Exception {
            String str = "A,B,C\n1,2,3";
            DataFrame df = ECDslVisitor.parse(str);
            System.out.println(df.toList());
            System.out.println(df.isNumeric());
        }
    }

    public interface ECDsl {
        DataFrame parse(String str);
    }

    public static class ECSVVisitor implements Ecdsl {
        @Override
        public DataFrame parse(String str) throws Exception {
            return DataFrame.parse(str);
        }
    }

    public static void main(String[] args) throws Exception {
        ECSVVisitor csv = new ECSVVisitor();
        String str = "A,B,C\n1,2,3";
        System.out.println(csv.parse(str));
    }
}
