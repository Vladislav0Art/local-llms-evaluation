    }

    public void outerHtmlTailTests() {

        StringBuilder builder = new StringBuilder();
        TextNode textNode = new TextNode("Hello");
        int depth = 0;
        textNode.outerHtmlHead(accum, depth, null);
        textNode.outerHtmlTail(accum, depth, null);
        assertEquals("<p>Hello</p>", accum.toString());
    }

    public void toStringTests() {

        assertEquals("Hello", textNode.toString());

        assertNotNull(textNode2);

    }

    public void normaliseWhitespaceTests() {

        assertEquals("\n", normaliseWhitespace("\n\r\n"));

        assertEquals("", stripLeadingWhitespace(""));
        assertEquals("Hello World", stripLeadingWhitespace("Hello World"));

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class TestUtils {

    public static String normaliseWhitespace(String str) {
        return str.replaceAll("\\s+", " ");
    }

    public static void stripLeadingWhitespace(StringBuilder str) {
        while (!str.isEmpty() && str.charAt(0) == ' ') {
            str.deleteCharAt(0);
        }
    }

    public static boolean lastCharIsWhitespace(StringBuilder sb) {
        if (sb.length() > 1 && sb.charAt(sb.length() - 1) != '\n' && sb.charAt(sb.length() - 1) != '\r') {
            return true;
        } else {
            return false;
        }
    }

    public static String[] lastCharIsWhitespace(String str) {
        StringBuilder sb = new StringBuilder(str);
        return lastCharIsWhitespace(sb.toString());
    }

    public static boolean trimLeadingSpaces(String str) {
        if (str.length() > 0 && !Character.isWhitespace(str.charAt(0))) {
            int i = 1;
            while (i < str.length()) {
                if (!Character.isWhitespace(str.charAt(i))) {
                    break;
                }
                i++;
            }
            return true;
        } else {
            return false;
        }
    }

    public static String[] trimLeadingSpaces(StringBuilder sb) {
        while (!sb.isEmpty() && !trimLeadingSpaces(sb.toString())) {
            sb.deleteCharAt(0);
        }
        if (sb.length() > 1 && sb.charAt(sb.length() - 1) != '\n' && sb.charAt(sb.length() - 1) != '\r') {
            return new String[] {sb.toString()};
        } else {
            return null;
        }
    }

    public static String[] normaliseWhitespace(String str) {
        if (str == null || str.isEmpty()) {
            return new String[0];
        }
        if (!trimLeadingSpaces(str)) {
            str = trimLeadingSpaces(str);
        }
        StringBuilder sb1 = new StringBuilder();
        for (char c : str.toCharArray()) {
            if (" \t\n\r\f\v".indexOf(c) != -1 || c == ' ') {
                sb1.append(c);
            } else {
                break;
            }
        }
        return new String[]{sb1.toString()};
    }

    public static void main(String[] args) {
        TestUtils normaliseWhitespace = new TestUtils();
        System.out.println(normaliseWhitespace("Hello World")); // Expected output: Hello World
        System.out.println(normaliseWhitespace(null)); // Expected output:
        System.out.println(normaliseWhitespace(""));
        System.out.println(trimLeadingSpaces(new StringBuilder("   "))); // Expected output: 
        System.out.println(lastCharIsWhitespace(new StringBuilder("   "))); // Expected output: true
    }
}
