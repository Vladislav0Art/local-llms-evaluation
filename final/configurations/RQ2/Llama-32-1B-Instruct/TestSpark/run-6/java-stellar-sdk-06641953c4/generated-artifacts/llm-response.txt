import static org.junit.Assert.assertTrue;

public class ExampleClass {
    private byte[] verifiedSignature;

    @Test
    public void testSign() throws Exception {
        String data = "Hello, World!";
        byte[] decoratedData = ExampleClass.getExampleData(data);
        KeyPair keyPair = new PublicKey();
        boolean signed = ExampleClass.sign(keyPair, decoratedData);

        if (signed) {
            byte[] signature = ExampleClass.getSignature(keyPair, signed);
            assertTrue(verifySignature(signature));
            verifiedSignature = signature;
        } else {
            throw new AssertionError("Signature was not generated");
        }
    }

    private byte[] getExampleData(String data) {
        return data.getBytes();
    }

    private byte[] getSignature(KeyPair keyPair, boolean signed) throws Exception {
        byte[] signature = (byte[])keyPair.getEncoded().length > 0 ? new byte[keyPair.getEncoded().length] : null;
        if (!signed) {
            signature = generateSignature(keyPair);
        }
        return signature;
    }

    private byte[] generateSignature(KeyPair keyPair) throws Exception {
        return ExampleClass.sign(keyPair, (byte[])new byte[0]);
    }

    public static boolean verifySignature(byte[] signature) {
        if (signature == null) {
            throw new AssertionError("Invalid signature");
        }
        try {
            VerifyTrustRequest.verify(signature);
            return true;
        } catch (VerifyTrustRequest.VerifyException e) {
            // This is a dummy exception for demonstration purposes
            System.out.println("Signature verification failed: " + e.getMessage());
            return false;
        }
    }

    public static String toString(byte[] data) throws Exception {
        StringBuilder builder = new StringBuilder();
        for (byte b : data) {
            builder.append(String.format("%02x", b));
        }
        return builder.toString();
    }
}

import org.stellar.sdk.Crypto;
import org.stellar.sdk.KeyPair;
import org.stellar.sdk.PrivateKey;

public class GeneratedTest {

    public static byte[] generateSignature(KeyPair keyPair) {
        String data = "test";
        return Crypto.sign(keyPair, data.getBytes());
    }

    public static boolean sign(byte[] signature, KeyPair privateKey) {
        try {
            // Convert private key to public
            KeyPair publicKey = new PublicKey(privateKey);
            byte[] signatureBytes = generateSignature(publicKey);

            // Compare the signed data with the expected signature
            return Crypto.verify(signatureBytes, signature);
        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());
            return false;
        }
    }

    public static boolean verifyTrustRequest(String signature) {
        try {
            // Convert private key to public
            KeyPair privateKey = new PublicKey(new PrivateKey(signature));
            String data = new String(Crypto.sign(privateKey, "test"));
            return Crypto.verify(signature, data.getBytes());
        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());
            return false;
        }
    }

    public static void main(String[] args) {
        // Generate public key
        KeyPair publicKey = new PublicKey(new PrivateKey("YOUR_PUBLIC_KEY_HERE"));
        String data = new String(Crypto.sign(publicKey, "test"));

        // Verify trust request signature
        boolean signed = sign(data.getBytes(), publicKey);
        System.out.println(signed);

        // Get public key from private key
        KeyPair privateKey = new KeyPair(new PrivateKey("YOUR_PRIVATE_KEY_HERE"));
        try {
            String data = new String(Crypto.sign(privateKey, "test"));
            verifyTrustRequest(data);
        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());
        }
    }
}
