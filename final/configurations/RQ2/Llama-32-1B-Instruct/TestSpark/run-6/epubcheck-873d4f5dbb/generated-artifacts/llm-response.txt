public class OPFChecker {
    public static int getPreferredMediaType(String mediaType, String path) throws Exception {
        if (mediaType.equals("audio") || mediaType.equals("video")) {
            return 0; // audio/opus
        } else if (mediaType.equals("common video")) {

        } else {
            return 4;
        }

    public static int getPreferredFileExtension(String expected) throws Exception {

            case "audio":
                return 0;
            case "video":
                return 4;
            default:
                throw new IllegalArgumentException("Unknown media type");
        }
    }

    public static String constructPreferredPath(int fileExtension, String mediaType, String path) throws Exception {
        if (fileExtension != getPreferredFileExtension(mediaType)) {
            throw new RuntimeException("Invalid file extension");
