class InputSource extends java.io.StringReader {
    private String input;

    public InputSource(String input) {
        this.input = input;
    }

    @Override
    public java.io.InputStream read() throws Exception {
        return new java.io.BufferedInputStream(new java.io.FileInputStream(input));
    }

    @Override
    public int read() throws Exception {
        if (input.length() < 1) {
            throw new Exception("EOF");
        }
        return input.charAt(0);
    }
}

class InputSourceReader extends Reader {
    private InputSource source;

    public InputSourceReader(InputSource source) {
        this.source = source;
    }

    @Override
    public java.io.InputStream read() throws Exception {
        return source.read();
    }
}
