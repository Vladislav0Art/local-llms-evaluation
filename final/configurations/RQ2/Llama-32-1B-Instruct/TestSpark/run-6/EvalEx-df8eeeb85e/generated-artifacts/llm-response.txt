    }

    private List<Token> parse(String expression, ExpressionConfiguration configuration) {
        // Split the expression into tokens
        String[] tokens = splitExpression(expression);

        // Initialize a list to store the parsed tokens
        List<Token> tokensList = new ArrayList<>();

        for (int i = 0; i < tokens.length; i++) {
            int type = parseType(tokens[i]);
            if (type != -1) {
                String literal = tokens[i].str;
                tokensList.add(new Literal(literal));
            } else {
                String tokenName = tokens[i].tokenName;
                if (configuration.isIdentify(tokenName)) {
                    // Token is an identifier
                    tokensList.add(new Identifier(tokenName));
                } else {
                    throw new ParseException("Unknown token: " + tokens[i]);
                }
            }

            if (currentPosition < expression.length()) {
                // Check for arithmetic operators and add the corresponding token to the list
                int operatorPos = currentPosition;
                while (operatorPos < expression.length() && Character.isDigit(expression.charAt(operatorPos))) {
                    operatorPos++;
                }
                
                if (operatorPos >= expression.length() || expression.charAt(operatorPos) != '+') {
                    throw new ParseException("Invalid arithmetic operation: " + expression);
                }

                String tokenValue = "";
                while (currentPosition < expression.length()) {
                    if (Character.isDigit(expression.charAt(currentPosition))) {
                        tokenValue += expression.charAt(currentPosition++);
                    } else {
                        break;
                    }
                }

                tokensList.add(new BinaryOperation(tokenValue, expression.charAt(operatorPos + 1), tokens[i]));
            }
        }

        return tokensList;
    }

    private int parseType(Token token) {
        if (token instanceof IntegerLiteral || token instanceof FloatLiteral) {
            return 0; // Token type (integer, float)
        } else if (token instanceof Identifier) {
            return 1; // Token type (identifier)
        } else if (token instanceof Operation) {
            return 2; // Token type (operation)
        } else {
            throw new ParseException("Unknown token type: " + token);
        }
    }

}

This `Tokenizer` class splits the input expression into tokens and adds them to a list of parsed tokens. It supports arithmetic operations (`+`, `-`, `*`, `/`) and literal values (integers). The `parse()` method returns a list of `BinaryOperation` objects, depending on whether it encounters an identifier or not.
