public class TestUtil {
    public static String hash(String str) {
        return System.identityHashCode(str);
    }

    public static int compareHashes(int hash1, int hash2) {
        return hash1 - hash2;
    }
}

public class AppTest {

    @Test
    public void testHelloWorld() {
        // Test 1: assertEquals
        String expected = "Hello World!";
        String actual = TestUtil.hash("Hello World!");
        assertNotEquals(0, TestUtil.compareHashes(TestUtil.hash(expected), actual));
        System.out.println("Test passed");

        // Test 2: assertNotNull
        assertNotNull(TestUtil.hash("Hello"));
        System.out.println("Test passed");
    }

    @Test
    public void testNullString() {
        // Test 3: assertNotEquals
        String expected = null;
        String actual = TestUtil.hash(null);
        System.out.println("Test passed");
    }

    @Test
    public void testEmptyString() {
        // Test 4: assertNotNull
        String expected = "";
        String actual = TestUtil.hash("");
        System.out.println("Test passed");
    }
}
