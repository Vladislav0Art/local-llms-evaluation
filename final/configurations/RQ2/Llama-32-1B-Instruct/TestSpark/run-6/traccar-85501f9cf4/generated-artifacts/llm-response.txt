```java
public class Network {
    private int id;
    private String name;
    private String ip;

    public Network(int id, String name, String ip) {
        this.id = id;
        this.name = name;
        this.ip = ip;
    }

    // Getters and Setters
    public void setId(int id) {
        this.id = id;
    }

    public int getId() {
        return id;
    }

    public void setName(String name) {
        this.name = name;
    }

    public String getName() {
        return name;
    }

    public void setIp(String ip) {
        this.ip = ip;
    }

    public String getIpAddress() {
        return ip;
    }
}

public class NetworkPropertiesTest {

    @Test
    public void testNetworkCreation() {
        // Test creating a network with default values
        Network network = new Network(1, "example.com", "192.168.0.100");
        assertNotNull(network);
        assertEquals(1, network.getId());
        assertEquals("example.com", network.getName());
        assertEquals("192.168.0.100", network.getIpAddress());

        // Test creating a network with custom values
        network.setId(2, "test.com", "192.168.0.200");
        assertNotNull(network);
        assertEquals(2, network.getId());
        assertEquals("test.com", network.getName());
        assertEquals("192.168.0.200", network.getIpAddress());

        // Test getting and setting properties
        network.setName("newName");
        System.out.println("Network Name: " + network.getName());
    }

    @Test
    public void testGetAndSetProperties() {
        Network network = new Network(1, "example.com", "192.168.0.100");

        // Test getting and setting id
        network.setId(2, "test.com", "192.168.0.200");
        assertEquals(2, network.getId());
        System.out.println("Network ID: " + network.getId());

        // Test getting and setting name
        network.setName("newName");
        assertEquals("newName", network.getName());
        System.out.println("Network Name: " + network.getName());

        // Test getting and setting ip
        network.setIp("newIpAddress");
        assertEquals("newIpAddress", network.getIpAddress());
    }
}
```
