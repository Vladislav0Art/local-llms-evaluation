package net.e175.klaus.solarpositioning;

import java.time.LocalDate;
import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertThrows;
import static org.mockito.Mockito.mock;
import static org.mockito.Mockito.when;

public class DeltaTPersonalTest {

    private final LocalDate forDate = LocalDate.of(2022, 1, 1);

    @Test
    public void testEstimate_NoException() {
        double result = DeltaT.estimate(forDate);
        assertEquals(0, result);
    }

    @Test
    public void testEstimate_SmallValue() {
        when(DeltaT.estimate(forDate)).thenReturn(-2); // small value to minimize calculation
        double result = DeltaT.estimate(forDate);
        assertEquals(-2, result);
    }

    @Test
    public void testEstimate_MediumValue() {
        when(DeltaT.estimate(forDate)).thenReturn(10.0); // medium value for demonstration purposes
        double result = DeltaT.estimate(forDate);
        assertEquals(10.0, result);
    }

    @Test
    public void testEstimate_LargeValue() {
        when(DeltaT.estimate(forDate)).thenReturn(-20.0); // large value to test the estimate function
        double result = DeltaT.estimate(forDate);
        assertEquals(-20.0, result);
    }

    @Test
    public void testEstimate_Scenario_InvalidInput() {
        when(DeltaT.estimate(any(LocalDate))).thenThrow(Exception.class);

        assertThrows(Exception.class, () -> DeltaT.estimate(null));
    }

    @Test
    public void testEstimate_Scenario_TooManyCalls() {
        when(DeltaT.estimate(forDate)).thenReturn(0); // set the return value to 0

        for (int i = 1; i <= 10000; i++) {
            DeltaT.estimate(forDate);
        }

        assertEquals(10000, DeltaT.estimate(forDate));
    }
}
