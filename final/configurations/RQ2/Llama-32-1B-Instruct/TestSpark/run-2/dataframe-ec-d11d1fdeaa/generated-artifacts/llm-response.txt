class Dataframe {
    private Object[][] data;
    private int rowCount;

    public Dataframe(String[] columnNames) {
        this.data = new Object[rowCount()][columnNames.length];
        this.rowCount = 0;
        for (String columnName : columnNames) {
            data[rowCount] = new Object[columnName.length()];
        }
    }

    public void setRowData(int index, Object[] row) {
        if (index >= rowCount) {
            throw new IndexOutOfBoundsException("Index out of range");
        } else {
            System.arraycopy(row, 0, data[index], 0, columnNames.length);
        }
    }

    public Dataframe addRow(Object[] newRow) {
        Object[][] newData = new Object[rowCount()][columnNames.length];
        for (int i = 0; i < rowCount; i++) {
            for (int j = 0; j < columnNames.length; j++) {
                newData[i][j] = data[i][j];
            }
        }
        data[rowCount] = newData;
        return this;
    }

    public void printTable() {
        if (data == null) {
            System.out.println("Dataframe is empty");
            return;
        }

            System.out.print(String.join(", ", data[i]));
            if (i != rowCount - 1) {
                System.out.println();
            }
        }
    }

    public int getRowCount() {
        return rowCount;
    }

    public Object[] getData(int index) {
        if (index >= rowCount) {
            throw new IndexOutOfBoundsException("Index out of range");
        } else {
            return data[index];
        }
    }
}
