import org.apache.commons.lang3.tuple.Pair;
import java.nio.ByteBuffer;

public class Gt06ProtocolDecoderTest {

    public static void main(String[] args) {
        String protocolVersion = "3";
        int channelId = 0;
        String hostAddress = "127.0.0.1";
        int portNumber = 12345;

        String testData = ByteBuffer.allocate(100).putInt(123).putShort(456).putLong(789).putDouble(1012).putBoolean(true).putString("test", 13).toString();

        Gt06ProtocolDecoder decoder = new Gt06ProtocolDecoder(protocolVersion, channelId, hostAddress, portNumber);
        decoder.decode(testData);

        String decodedString = decoder.getDecodedString();
        System.out.println(decodedString);
    }

    public static void testDecodeGpsPosition() {
        Position position = new Position();
        position.setLat(37.7749);
        position.setLon(-122.4194);
        position.setAlt(0);

        String testPosition1 = "lat:37.7749,lon:-122.4194,alt=0";
        ByteBuf buf1 = encodePosition(position, Protocol.GT06);
        System.out.println(decodeGps(buf1));
    }

    public static void testDecodeGpsWithSatellites() {
        Position position = new Position();
        position.setLat(37.7749);
        position.setLon(-122.4194);
        position.setAlt(0);
        position.setSatelliteCount(6);

        String testPosition2 = "lat:37.7749,lon:-122.4194,alt=0,sat=6";
        ByteBuf buf2 = encodePosition(position, Protocol.GT06);
        System.out.println(decodeGps(buf2));
    }

    public static void testDecodeSpeed() {
        Position position = new Position();
        position.setLat(37.7749);
        position.setLon(-122.4194);
        position.setAlt(0);
        position.setSpeedInKmh(100);

        String testPosition3 = "lat:37.7749,lon:-122.4194,alt=0,speed=100km/h";
        ByteBuf buf3 = encodePosition(position, Protocol.GT06);
        System.out.println(decodeGps(buf3));
    }

    public static Pair<ByteBuf, String> decodeGps(ByteBuf buf) {
        if (buf instanceof LongBuffer) {
            return Pair.of((LongBuffer) buf, "test string");
        } else {
            int channel = ((LongBuffer) buf).get(0);
            int dataFormat = ((LongBuffer) buf).get(1);
            long data = ((LongBuffer) buf).get(2);

            String[] parts = new String[3];
            if (dataFormat == 1) { // latitude
                parts[0] = "lat:".concat(ByteUtils.toHexString(data)).trim();
            } else if (dataFormat == 2) { // longitude
                parts[0] = "lon:".concat(ByteUtils.toHexString(data)).trim();
            } else if (dataFormat == 3) { // altitude
                parts[0] = "alt=(".concat(ByteUtils.toHexString(data)).trim() + ".";
            }

            return Pair.of(buf, String.join("\n", parts));
        }
    }

    public static ByteBuf encodePosition(Position position, Protocol protocol) {
        if (protocol == Protocol.GT06) {
            return ByteBuffer.allocate(100).putInt(position.getChannelId()).putShort(position.getHostAddress().hashCode()).putLong(position.getPortNumber()).putDouble(position.getSpeedInKmh()).putBoolean(position.isSatelliteCount());
        } else {
            throw new UnsupportedOperationException("Unsupported protocol version");
        }
    }
}
