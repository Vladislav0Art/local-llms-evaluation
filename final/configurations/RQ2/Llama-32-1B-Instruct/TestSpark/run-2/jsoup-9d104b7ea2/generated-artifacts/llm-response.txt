public class Main {
    public static void main(String[] args) {

        Document document = Jsoup.parse(html);

        // Creating an empty HTML body
        String emptyBodyHtml = createEmptyHtmlBody(document, 0);
        System.out.println(outerHtmlHead(document, 0).text() + "\n" + outerHtmlHead(document, 1).text());
    }

    public static String createEmptyHtmlBody(Document document, int offset) {
        return "<body></body>";
    }

    public static Document outerHtmlHead(Document document, int offset) {
        if (offset < 0 || offset > document.children().size()) {
            return document;
        }
        
        // Simulating the outer HTML head
        String headContent = createHeadContent(document, offset);
        return document.outerHtml().replaceFirst("<head>.*?</head>", headContent + "</head>");
    }

    public static String createHeadContent(Document document, int offset) {
        Elements elements = document.children();
        StringBuilder htmlBuilder = new StringBuilder();
        for (int i = 0; i < Math.min(offset + 1, elements.size()); i++) {
            if (!elements.get(i).isEmpty()) {
                htmlBuilder.append(elements.get(i).text()).append(" ");
            }
        }
        return htmlBuilder.toString().trim();
    }

    public static String isBlank(String input) {
        return input.trim().isEmpty();
    }
}
