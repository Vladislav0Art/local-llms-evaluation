import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.select.Elements;

public class Main {
    public static void main(String[] args) {
        // Create a new Jsoup Document object from an HTML string.
        String htmlString = "<html><body><h1>Hello World!</h1><p>This is a paragraph.</p></body></html>";
        Document document = Jsoup.parse(htmlString);

        System.out.println("Parsed HTML:");
        System.out.println(document.toString());

        System.out.println("\nExtracting all h1 tags...");
        Elements elements = document.select("h1");
        for (Element element : elements) {
            System.out.println(element.text());
        }

        System.out.println("\nExtracting text from paragraph with id 'head' and p tags with id 'paragraph'...");
        String headText = document.select("h1[id='head']").attr("id") + document.select("p[id='paragraph']").first().text();
        System.out.println(headText);
    }
}

    private String text;

    public String getText() {
        return text;
    }

    public void setText(String text) {
        this.text = text;
    }

    public boolean equals(Object obj) {

            return false;

    }

    @Override
    public int hashCode() {
        return Objects.hash(text);
    }
}

class Document {
    private String body;

        return body;
    }

    public void setBody(String body) {
        this.body = body;
    }
}
