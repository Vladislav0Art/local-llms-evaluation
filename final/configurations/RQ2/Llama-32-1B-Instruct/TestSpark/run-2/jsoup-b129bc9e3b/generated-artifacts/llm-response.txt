import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.net.HttpURLConnection;
import java.net.URL;
import java.util.Scanner;

public class TestHttpConnection {
    public static void main(String[] args) throws Exception {
        new TestHttpConnection().testConnect();
    }

    private void testConnect() throws MalformedURLException, URISyntaxException {
        String url = "https://example.com";
        URL parsedUrl = new URL(url);
        Request request = new Request();
        request.method(Method.HEAD);
        request.from(parsedUrl.getHost());
        httpConnection = HttpConnection.connect(url);
        testGetResponse(httpConnection);
    }

    private void testGetResponse(HttpConnection httpConnection) throws IOException {
        Scanner scanner = new Scanner(new BufferedReader(new InputStreamReader(httpConnection.getInputStream())));
        System.out.println(scanner.nextLine());
        scanner.close();
    }
}

class ConnectionTest extends TestHttpConnection {
    @Override
    public HttpConnection connect(String url) throws MalformedURLException, URISyntaxException {
        URL parsedUrl = new URL(url);
        HttpURLConnection connection = (HttpURLConnection) parsedUrl.openConnection();
        connection.setRequestMethod(Method.HEAD);
        connection.setConnectTimeout(10000); // 10 seconds
        return new HttpConnection(connection);
    }
}
