import ch.jalu.configme.configurationdata.CommentsConfiguration;
import ch.jalu.configme.configurationdata.CommentsConfigurationBuilder;
import ch.jalu.configme.configurationdata.CommentType;
import ch.jalu.configme.configurationdata.SettingsHolder;
import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.UnmodifiableView;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;

public class CommentsConfigurationBuilderTest {

    private CommentsConfiguration commentsConfiguration = new CommentsConfigurationBuilder();

    public void testCommentsSet() {
        String path = "path/to/comments";
        List<String> commentLines = Arrays.asList("line 1\n", "line 2");
        commentsConfiguration.setComment(path, commentLines);
        assertEquals(3, commentsConfiguration.getAllComments().size());
        assertEquals(2, commentsConfiguration.getAllComments().get(path).size());
    }

    public void testAllComments() {
        String path = "path/to/all/comments";
        List<String> commentLines = new ArrayList<>();
        commentLines.add("");
        commentLines.add("line 1\n");
        commentLines.add("line 2\n");
        commentsConfiguration.setComment(path, commentLines);
        assertEquals(4, commentsConfiguration.getAllComments().size());
        assertEquals(3, commentsConfiguration.getAllComments().get(path).size());
    }

    public void testCommentType() {
        CommentsConfigurationBuilder builder = new CommentsConfigurationBuilder();
        String path = "path/to/comment-type";
        List<String> commentLines = Arrays.asList("line 1\n", "line 2");
        CommentType type = CommentType.LINE;
        builder.setComment(path, type, commentLines);
        assertEquals(3, commentsConfiguration.getAllComments().size());
        assertEquals(2, commentsConfiguration.getAllComments().get(path).size());
    }

    public void testAllCommentTypes() {
        CommentsConfigurationBuilder builder = new CommentsConfigurationBuilder();
        String path = "path/to/all/comment-types";
        List<CommentType> commentTypes = Arrays.asList(CommentType.LINE, CommentType.LOCALE);
        for (CommentType type : commentTypes) {
            builder.setComment(path, type, commentLines -> Arrays.asList(type.toString()));
        }
        assertEquals(4, commentsConfiguration.getAllComments().size());
        assertEquals(3, commentsConfiguration.getAllComments().get(path).size());
    }

    public void testSettingsHolder() {
        CommentsConfigurationBuilder builder = new CommentsConfigurationBuilder();
        String path = "path/to/settings";
        SettingsHolder settingsHolder = new SettingsHolder();
        builder.setComment(path, "", settingsHolder);
        assertEquals(1, commentsConfiguration.getAllComments().get(path).size());
    }
}

public class Config {
    public Map<String, String> settings = new HashMap<>();

    public void setSetting(String key, String value) {
        this.settings.put(key, value);
    }

    public static ConfigBuilder builder() {
        return new ConfigBuilder();
    }
}

public class ConfigBuilder {
    private Config config;

    public ConfigBuilder() {
        this.config = new Config();
    }

    public Config get() {
        return this.config;
    }

    public void setSetting(String key, String value) {
        this.config.settings.put(key, value);
    }
}

class TestClass {
    public static void main(String[] args) {
        Config config = ConfigBuilder.builder().build();

        System.out.println(config.get().settings());

        config.setSetting("setting1", "value1");
        System.out.println(config.get().settings());
    }
}
