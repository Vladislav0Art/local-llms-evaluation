public class Solution {
    public int[] numsSameConsecPermutation(int n, int[] arr) {
        if (n == 1) return new int[]{arr[0]};
        int[][] dp = new int[n][n];
        dp[0][0] = 1;
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j <= i; ++j) {
                if (dp[i][j] != 0 && arr[j] == arr[i]) {
                    dp[i + 1][j] += dp[i][j];
                    if (arr[j] != arr[i]) {
                        dp[i + 1][i] += dp[i][j];
                    }
                }
            }
        }
        int[] result = new int[n];
        for (int i = n - 1; i > 0; --i) {
            for (int j = 0; j <= i; ++j) {
                if (dp[i][j] != 0 && arr[j] == arr[i]) {
                    result[i] += dp[i][j];
                    if (arr[j] != arr[i]) {
                        result[i] += dp[i][j];
                    }
                } else {
                    break;
                }
            }
        }
        return result;
    }

    public void test() {
        int[] a = {1, 2};
        for (int i : numsSameConsecPermutation(4, a)) {
            System.out.print(i + " ");
        }
        System.out.println();
    }
}

public class TestSolution {
    public static void main(String[] args) {
        TestSolution solution = new TestSolution();
        solution.test();
    }
}
