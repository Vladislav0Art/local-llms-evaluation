public class XmlTreeBuilderImpl extends XmlTreeBuilder {
    private String result;

    @Override
    protected void process(Token token) {
        if (token instanceof Token.StartTag) {
            insert((Token.StartTag) token);
        } else if (token instanceof Token.EndTag) {
            popStackToClose((Token.EndTag) token);
        }
    }

    public void insertNode(Node node, String uri) {
        if (node instanceof Element) {
            ((Element) node).append(node);
        }
    }

    @Override
    protected Document parse(Reader input, String baseUri) {
        return parse(new StringReader(input.toString()), baseUri);
    }

    @Override
    protected void insertToken(Token token) {
        if (token instanceof Token.StartTag || token instanceof Token.EndTag) {
            // Do nothing, this is already handled in the process() method
        } else if (token instanceof Token.Text) {
            result += ((Token.Text) token).text();
        }
    }

    public void insertComment(String text) {
        if (!isEmpty()) {
            result += " <!-- " + text + " -->";
        }
    }

    @Override
    protected Document parseFragment(String inputFragment, String baseUri, Parser parser) {
        return parseFragment(inputFragment, null, baseUri, parser);
    }

    public void insertFragment(String inputFragment, Document document, String baseUri, Parser parser) {
        List<Node> nodes = new ArrayList<>();
        for (Token token : parseFragment(inputFragment, inputFragment, baseUri, parser).nodes()) {
            if (token instanceof Token.Text || token instanceof Token.Comment || token instanceof Node.CDataNode) {
                insert(node -> node.append(token));
            } else {
                insertNode(document.createElement(token.tagName().toString()), token.uri());
            }
        }
    }

    @Override
    protected void popStackToClose(Token endTag) {
        if (endTag.getTagName().equals("END")) {
            result = null;
        } else {
            result += " - " + endTag.getText();
        }
    }

    public Document parseFragment(String inputFragment, Element context, String baseUri, Parser parser) {
        return parseFragment(inputFragment, context.toString(), baseUri, parser);
    }

    private void insertNode(Element element) {
        if (element.getTagName().equals("END")) {
            popStackToClose(element);
        } else {
            result += " - " + element.getText();
        }
    }

    public String getresult() {
        return result;
    }

    @Override
    protected Document parse(String input, String baseUri) {
        return parse(new StringReader(input), baseUri);
    }

    private void insertToken(Token token) {}

    @Override
    protected void initialiseParse(Reader input, String baseUri, Parser parser) {
        // Do nothing for ParseSettings defaultSettings()
    }
}
