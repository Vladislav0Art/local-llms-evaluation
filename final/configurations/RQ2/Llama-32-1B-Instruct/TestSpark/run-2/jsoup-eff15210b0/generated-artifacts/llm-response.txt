public class HttpRequest {
    private String method;
    private int status;

    public void setMethod(String method) {
        this.method = method;
    }

    public void setStatus(int status) {
        this.status = status;
    }
}

class HttpRequestHandlerTest {

    @Before
    public void setUp() {
        HttpRequest request = new HttpRequest();
    }

    @Test
    public void testGetSuccess() {
        HttpRequestHandler handler = new HttpRequestHandler();
        Request request = new Request("GET", "http://example.com");
        Response response = handler.handleRequest(request);
        Assert.assertTrue("Request did not set response correctly", response.getMethod().equals(request.getMethod()));
        Assert.assertFalse("Response was not set to HTTP status code 200", response.getStatus() != 200);
    }

    @Test
    public void testPostFailure() {
        HttpRequestHandler handler = new HttpRequestHandler();
        Request request = new Request("POST", "http://example.com");
        Response response = handler.handleRequest(request);
        Assert.assertFalse("Response was not set to HTTP status code 500", response.getStatus());
    }
}

class Response {
    private int statusCode;
    private String message;

    public void setStatus(int statusCode) {
        this.statusCode = statusCode;
    }

    public int getStatusCode() {
        return statusCode;
    }

    @Override
    public String toString() {
        return "Response{" +
                "statusCode=" + statusCode +
                ", message='" + message + '\'' +
                '}';
    }
}

class Request {
    private String method;
    private String url;

    public void setMethod(String method) {
        this.method = method;
    }

    public void setUrl(String url) {
        this.url = url;
    }

    @Override
    public String toString() {
        return "Request{" +
                "method='" + method + '\'' +
                ", url='" + url + '\'' +
                '}';
    }
}

class HttpRequestHandler {
    private Response response;

    public void handleRequest(Request request) {
        if (request.getMethod().equals("GET")) {
            response = new Response(200, "");
            this.response = new Response(200, "");
        } else {
            this.response = new Response(500, "Internal Server Error");
        }
    }

    public Response getResponse() {
        return response;
    }
}
