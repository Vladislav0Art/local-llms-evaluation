import ch.jalu.configme.configurationdata CommentsConfiguration;
import ch.jalu.configme.configurationdata.CommentsConfigurationException;
import ch.jalu.configme.settings.SettingsHolder;
import org.jetbrains.annotations.NotNull;
import org.jetbrains.annotations.UnmodifiableView;
import java.util.Arrays;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class CommentsConfigurationTest {

    public void testCreateCommentsConfiguration() {
        CommentsConfiguration config = new CommentsConfiguration();
        Assertions.assertTrue(config.comments.isEmpty());
        config.setComment("path", "line1\nline2");
        config.setComment("path", "");
        Assertions.assertFalse(config.comments.isEmpty());
    }

    public void testGetAllComments() {
        SettingsHolder settings = new SettingsHolder();
        Map<String, List<String>> comments = new HashMap<>();
        comments.put("path1", Arrays.asList("comment1"));
        comments.put("path2", Arrays.asList("comment2", "comment3"));
        CommentsConfiguration config = new CommentsConfiguration(comments);
        Assertions.assertTrue(config.getAllComments().containsKey("path1"));
        Assertions.assertFalse(config.getAllComments().containsKey("path2"));
    }

    public void testCommentSetter() {
        SettingsHolder settings = new SettingsHolder();
        Map<String, List<String>> comments = new HashMap<>();
        comments.put("path", Arrays.asList("comment1\ncomment2"));
        CommentsConfiguration config = new CommentsConfiguration(comments);
        String path = "path";
        Assertions.assertTrue(config.setComment(path, "newline1\nnewline2"));
        Assertions.assertEquals(Arrays.asList(new String[]{"newline1\nnewline2"}), config.getAllComments().get(path));
    }

    public void testGetAllCommentsException() {
        SettingsHolder settings = new SettingsHolder();
        Map<String, List<String>> comments = new HashMap<>();
        comments.put("path", Collections.singletonList("");
        CommentsConfiguration config = new CommentsConfiguration(comments);
        Assertions.assertFalse(config.getAllComments().containsKey("path"));
    }

    public void testCommentSetterNullPath() {
        SettingsHolder settings = new SettingsHolder();
        Map<String, List<String>> comments = new HashMap<>();
        CommentsConfiguration config = new CommentsConfiguration(comments);
        String path = null;
        Assertions.assertTrue(config.setComment(path, "newline1\nnewline2").isEmpty());
    }

    public void testCommentSetterNullComment() {
        SettingsHolder settings = new SettingsHolder();
        Map<String, List<String>> comments = new HashMap<>();
        CommentsConfiguration config = new CommentsConfiguration(comments);
        String path = "path";
        Assertions.assertTrue(config.setComment(path, null).isEmpty());
    }
}

import org.junit.Test;

public class ConfigMe {

    @Test
    public void testGetSettings() {
        // Given
        String[] keys = {"key1", "key2"};

        // When
        SettingsHolder settings = new SettingsHolder();
        for (String key : keys) {
            settings.set(key, null);
        }

        Object settingsValue = settings.get("key1");
        if (settingsValue instanceof String) {
            System.out.println(settingsValue);
        } else {
            throw new RuntimeException("Settings value is not a string");
        }
    }
}

class SettingsHolder {
    public void set(String key, Object value) {
        this.keys.put(key, value);
    }

    public Object get(String key) {
        return keys.get(key);
    }
}
