### Using Jsoup to Parse HTML and Extract Data

The following Java code uses Jsoup to parse the given HTML, extract all elements that contain specific text or data, and then convert these elements into a JSON string.

```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class JsoupParser {

    public static String parseHtml(String html) {
        Document document = Jsoup.parse(html);
        List<Map<String, String>> data = new ArrayList<>();

        // Extract elements that contain specific text or data
        for (Element element : document.select("table tr td[style*='font-size:12px;']")) {
            Map<String, String> map = new HashMap<>();
            map.put("id", element.attr("id"));
            map.put("name", element.text());
            data.add(map);
        }

        // Convert the list of maps into a JSON string
        StringBuilder json = new StringBuilder();
        for (Map<String, String> map : data) {
            json.append("{");
            for (Map.Entry<String, String> entry : map.entrySet()) {
                json.append("\"").append(entry.getKey()).append("\": \"").append(entry.getValue()).append("\"");
                if (json.length() != 0) {
                    json.append(",");
                }
            }
            json.append("}");
        }

        return json.toString();
    }

    public static void main(String[] args) {
        String html = "<html><body><table id='example'><tr><td>John</td></tr><tr><td>Doe</td></tr></table></body></html>";
        System.out.println(parseHtml(html));
    }
}
```

### Using Jsoup to Parse HTML, Extract Data, and Convert to JSON String

The following Java code uses Jsoup to parse the given HTML, extract all elements that contain specific text or data, and then convert these elements into a JSON string.

```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.select.Elements;

import java.util.HashMap;
import java.util.Map;

public class JsoupParser {

    public static String parseHtml(String html) {
        Document document = Jsoup.parse(html);
        Map<String, Object> data = new HashMap<>();

        // Extract elements that contain specific text or data
        Elements tableElements = document.select("table tr td[style*='font-size:12px;']");
        for (Element element : tableElements) {
            String value = element.text();
            if (!value.isEmpty()) {
                data.put(element.attr("id"), value);
            }
        }

        // Extract elements that contain specific text or data
        Elements tableRows = document.select("table tr");
        for (Elements rows : tableRows) {
            for (Element rowElement : rows) {
                String value = rowElement.text();
                if (!value.isEmpty()) {
                    Map<String, Object> map = new HashMap<>();
                    map.put("id", rowElement.attr("id"));
                    map.put("name", value);
                    data.put(map.get("id"), map);
                }
            }
        }

        // Convert the map into a JSON string
        StringBuilder json = new StringBuilder();
        for (Map.Entry<String, Object> entry : data.entrySet()) {
            if (!json.toString().isEmpty()) {
                json.append(",");
            }
            json.append("\"").append(entry.getKey()).append("\": ").append(toJson(entry.getValue()));
        }

        return json.toString();
    }

    public static String toJson(Object obj) {
        try {
            return org.json.JSONObject((org.json.JsonObject)obj).toString();
        } catch (Exception e) {
            System.out.println("Error parsing object: " + e.getMessage());
            return "";
        }
    }

    public static void main(String[] args) {
        String html = "<html><body><table id='example'><tr><td>John</td></tr><tr><td>Doe</td></tr></table></body></html>";
        System.out.println(parseHtml(html));
    }
}
```
