import java.io.IOException;

import java.net.URL;

        String url = "http://example.com";
        connect(url);

        // Connect without cookies
        url = "http://example.com";
        connectWithoutCookies(url);

        // Connect with cookie store
        url = "http://example.com";
        Map<String, String> data = new java.util.HashMap<>();
        data.put("session_id", "1234567890");
        connectWithCookieStore(url, data);

        // Connect without cookie store

        connectWithoutCookieStore(url);

        // Connect with parser

        connectWithParser(parser);

    public static void connect(String url) throws Exception {
        URL obj = new URL(url);
        HttpURLConnection con = (HttpURLConnection) obj.openConnection();
        con.setRequestMethod("GET");
        int responseCode = con.getResponseCode();

        if (responseCode == 200) {
            System.out.println("Connected to " + url);
        } else {
            System.out.println("Failed to connect to " + url + ". Status code: " + responseCode);
        }
    }

        URL obj = new URL(url);

import java.io.IOException;
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;

public class ConnectionTest {

    public static void main(String[] args) {
        // Connect to a URL with cookies
        String url = "http://example.com";
        connect(url);

        // Connect without cookies
        url = "http://example.com";
        connectWithoutCookies(url);

        // Connect with cookie store
        Map<String, String> data = new java.util.HashMap<>();
        data.put("session_id", "1234567890");
        connectWithCookieStore(url, data);

        // Connect without cookie store
        url = "http://example.com";
        connectWithoutCookieStore(url);

        // Connect with parser
        Parser parser = new Parser();
        String url2 = "http://example.com";
        connectWithParser(parser);
    }

    public static void connect(String url) throws Exception {
        HttpClient client = HttpClient.newHttpClient();
        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(url))
                .GET()
                .build();

        HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());

        if (response.statusCode() == 200) {
            System.out.println("Connected to " + url);
        } else {
            System.out.println("Failed to connect to " + url + ". Status code: " + response.statusCode());
        }
    }

    public static void connectWithoutCookies(String url) throws Exception {
        HttpClient client = HttpClient.newHttpClient();
        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(url))
                .GET()
                .build();

        HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());

        if (response.statusCode() == 200) {
            System.out.println("Connected to " + url);
        } else {
            System.out.println("Failed to connect to " + url + ". Status code: " + response.statusCode());
        }
    }

    public static void connectWithCookieStore(String url, Map<String, String> data) throws Exception {
        HttpClient client = HttpClient.newHttpClient();
        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(url))
                .header("Authorization", "Bearer " + data.get("session_id"))
                .GET()
                .build();

        HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());

        if (response.statusCode() == 200) {
            System.out.println("Connected to " + url + ". Status code: " + response.statusCode());
        } else {
            System.out.println("Failed to connect to " + url + ". Status code: " + response.statusCode());
        }
    }

    public static void connectWithoutCookieStore(String url, Map<String, String> data) throws Exception {
        HttpClient client = HttpClient.newHttpClient();
        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(url))
                .header("Authorization", data.get("session_id"))
                .GET()
                .build();

        HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());

        if (response.statusCode() == 200) {
            System.out.println("Connected to " + url + ". Status code: " + response.statusCode());
        } else {
            System.out.println("Failed to connect to " + url + ". Status code: " + response.statusCode());
        }
    }

    public static void connectWithParser(String url, Parser parser) throws Exception {
        HttpClient client = HttpClient.newHttpClient();
        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(url))
                .GET()
                .build();

        HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());

        if (response.statusCode() == 200 && response.body().contains(parser.toString())) {
            System.out.println("Parsed " + url + ". Status code: " + response.statusCode());
        } else {
            System.out.println("Failed to parse " + url + ". Status code: " + response.statusCode());
        }
    }

    public static void parse(String url, Parser parser) throws Exception {
        HttpClient client = HttpClient.newHttpClient();
        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(url))
                .GET()
                .build();

        HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());

        if (response.statusCode() == 200 && response.body().contains(parser.toString())) {
            System.out.println("Parsed " + url + ". Status code: " + response.statusCode());
        } else {
            System.out.println("Failed to parse " + url + ". Status code: " + response.statusCode());
        }
    }

    public static void main(String[] args) {
        Parser parser = new Parser();
        String url2 = "http://example.com";
        connectWithParser(parser, url2);

        URL obj = new URI("http://example.com");
        HttpClient client = HttpClient.newHttpClient();
        HttpRequest request = HttpRequest.newBuilder()
                .uri(obj)
                .GET()
                .build();

        HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());

        parse(url2, parser);
    }
