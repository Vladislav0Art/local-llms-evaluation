```java
public class TestNetwork {
    public static String testConnect(String networkId) throws Exception {
        // Connect to mock server
        Socket sock = new Socket("127.0.0.1", 12345);
        
        // Send a message
        byte[] message = "Hello, World!".getBytes();
        sock.getOutputStream().write(message);
        
        // Receive data
        int bytesRead;
        byte[] buffer = new byte[1024];
        while ((bytesRead = sock.getInputStream().read(buffer)) != -1) {
            System.out.println(new String(buffer, 0, bytesRead));
        }
        
        // Close the socket
        sock.close();
        
        return "Connected to network with id " + networkId;
    }

    public static void testDisconnect(String networkId) throws Exception {
        // Disconnect from mock server

        assertEquals("Disconnected from network with id " + networkId, testConnect(networkId));
    }

    public static void main(String[] args) throws Exception {
        String networkId = "test-network-123";
        System.out.println(testConnect(networkId));

        testDisconnect(networkId);
    }
```

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.net.Socket;

public class Main {

    public static void main(String[] args) {
        try (Socket client = new Socket("127.0.0.1", 12345)) {
            System.out.println("Connected to network with id " + client.getRemoteSocketAddress());

            // Send a message
            String message = "Hello, World!";
            byte[] buffer = message.getBytes();
            client.getOutputStream().write(buffer);
            System.out.println("Sent: " + new String(buffer));

            // Receive data
            int bytesRead;
            StringBuilder response = new StringBuilder();
            while ((bytesRead = client.getInputStream().read(buffer)) != -1) {
                response.append(new String(buffer, 0, bytesRead));
            }
            System.out.println("Received: " + response.toString());

        } catch (Exception e) {
            System.out.println("Error connecting to network: " + e.getMessage());
        }
    }

    public static void testDisconnect(String networkId) throws Exception {
        // Test that the error message is not empty
        try (Socket client = new Socket("127.0.0.1", 12345)) {
            System.out.println("Connected to network with id " + client.getRemoteSocketAddress());

            client.close();
            System.out.println(testConnect(networkId));
        } catch (Exception e) {
            System.out.println("Error disconnecting from network: " + e.getMessage());
        }
    }

    public static void testDisconnectUsingMockServer(String networkId) throws Exception {
        // Create a mock server
        try (Socket client = new Socket("127.0.0.1", 12345)) {
            System.out.println("Connected to network with id " + client.getRemoteSocketAddress());

            String response = "Welcome to the mock server!";
            byte[] buffer = response.getBytes();
            client.getOutputStream().write(buffer);
            System.out.println("Sent: " + new String(buffer));

            // Receive data
            int bytesRead;
            StringBuilder responseFromServer = new StringBuilder();
            while ((bytesRead = client.getInputStream().read(buffer)) != -1) {
                responseFromServer.append(new String(buffer, 0, bytesRead));
            }
            System.out.println("Received from server: " + responseFromServer.toString());

        } catch (Exception e) {
            System.out.println("Error connecting to mock server: " + e.getMessage());
        }

        // Test that the error message is not empty
        try (Socket client = new Socket("127.0.0.1", 12345)) {
            testDisconnect(networkId);
        } catch (Exception e) {
            System.out.println("Error disconnecting from mock server: " + e.getMessage());
        }
    }
}
```
