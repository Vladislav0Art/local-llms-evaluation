```java
import java.io.IOException;
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.util.Scanner;

public class Main {
    private static final String URL = "https://example.com";

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        while (true) {
            System.out.println("Enter a command:");
            System.out.println("1. GET");
            System.out.println("2. POST");
            int choice = scanner.nextInt();
            scanner.nextLine();

            switch (choice) {
                case 1:
                    get(url);
                    break;
                case 2:
                    post(url, scanner.nextLine());
                    break;
                default:
                    System.out.println("Invalid choice.");
            }

            System.out.println("Do you want to continue? (y/n)");
            String response = scanner.next().toLowerCase();
            while (!response.equals("y") && !response.equals("n")) {
                System.out.println("Invalid input. Please enter y for yes or n for no.");
                response = scanner.next().toLowerCase();
            }

            if (!response.equals("y")) {
                break;
            }
        }
    }

    private static void get(String url) {
        HttpClient client = HttpClient.newHttpClient();
        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(url))
                .GET()
                .build();

        try {
            client.sendAsync(request, HttpResponse.BodyHandlers.ofString())
                    .thenApply(HttpResponse::body)
                    .map(String::trim)
                    .forEach(System.out::println);
        } catch (IOException e) {
            System.err.println("Error: " + e.getMessage());
        }
    }

    private static void post(String url, String data) {
        HttpClient client = HttpClient.newHttpClient();
        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(url))
                .POST(HttpRequest.BodyPublishers.ofString(data).build())
                .build();

        try {
            client.sendAsync(request, HttpResponse.BodyHandlers.ofString())
                    .thenApply(HttpResponse::body)
                    .map(String::trim)
                    .forEach(System.out::println);
        } catch (IOException e) {
            System.err.println("Error: " + e.getMessage());
        }
    }
}
```

```java
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        System.out.println("Choose an action:");
        System.out.println("1. GET");
        System.out.println("2. POST");

        int choice = scanner.nextInt();
        scanner.nextLine(); // Consume newline left-over

        String response;
        switch (choice) {
            case 1:
                get(scanner);
                break;
            case 2:
                post(scanner);
                break;
            default:
                System.out.println("Invalid choice.");
        }

        while (true) {
            System.out.println("Do you want to continue? (y/n)");
            response = scanner.next().toLowerCase();
            while (!response.equals("y") && !response.equals("n")) {
                System.out.println("Invalid input. Please enter y for yes or n for no.");
                response = scanner.next().toLowerCase();
            }

            if (!response.equals("y")) {
                break;
            }
        }
    }

    private static void get(Scanner scanner) {
        try {
            System.out.println("Enter the URL:");
            String url = scanner.nextLine();

            HttpClient client = HttpClient.newHttpClient();
            HttpRequest request = HttpRequest.newBuilder()
                    .uri(URI.create(url))
                    .GET()
                    .build();

            HttpResponse<String> response = client.sendAsync(request, HttpResponse.BodyHandlers.ofString())
                    .thenApply(HttpResponse::body)
                    .map(String::trim)
                    .forEach(System.out::println);

            System.out.println("Response: " + response.body());
        } catch (IOException e) {
            System.err.println("Error: " + e.getMessage());
        }
    }

    private static void post(Scanner scanner) {
        try {
            System.out.println("Enter the URL:");
            String url = scanner.nextLine();

            System.out.println("Enter the data:");
            String data = scanner.nextLine();
            System.out.println("Enter a message (optional):");
            if (scanner.hasNextLine()) {
                data = scanner.nextLine();
            }

            HttpClient client = HttpClient.newHttpClient();
            HttpRequest request = HttpRequest.newBuilder()
                    .uri(URI.create(url))
                    .POST(HttpRequest.BodyPublishers.ofString(data).build())
                    .build();

            HttpResponse<String> response = client.sendAsync(request, HttpResponse.BodyHandlers.ofString())
                    .thenApply(HttpResponse::body)
                    .map(String::trim)
                    .forEach(System.out::println);

            System.out.println("Response: " + response.body());
        } catch (IOException e) {
            System.err.println("Error: " + e.getMessage());
        }
    }
}
```

```java
import java.io.IOException;
import java.net.URI;
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {
    private static final String URL = "https://example.com";

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        Map<String, Map<String, String>> actions = new HashMap<>();
        actions.put("GET", createActionMap("get"));
        actions.put("POST", createActionMap("post"));

        while (true) {
            System.out.println("Choose an action:");
            for (String key : actions.keySet()) {
                System.out.println(key + ": " + actions.get(key).size());
            }

            String choice = scanner.next();
            if (actions.containsKey(choice)) {
                Map<String, String> actionMap = actions.get(choice);
                System.out.println("Enter the URL:");
                String url = scanner.nextLine();

                try {
                    System.out.println("Enter the data:");
                    String data = scanner.nextLine();

                    for (Map.Entry<String, String> entry : actionMap.entrySet()) {
                        if (!entry.getValue().equals("")) {
                            System.out.println(entry.getKey() + ": " + entry.getValue());
                            break;
                        }
                    }

                    HttpClient client = HttpClient.newHttpClient();
                    HttpRequest request = HttpRequest.newBuilder()
                            .uri(URI.create(url))
                            .POST(HttpRequest.BodyPublishers.ofString(data).build())
                            .build();

                    HttpResponse<String> response = client.sendAsync(request, HttpResponse.BodyHandlers.ofString())
                            .thenApply(HttpResponse::body)
                            .map(String::trim)
                            .forEach(System.out::println);
                } catch (IOException e) {
                    System.err.println("Error: " + e.getMessage());
                }
            } else {
                System.out.println("Invalid choice.");
            }

            System.out.println("Do you want to continue? (y/n)");
            String response = scanner.next().toLowerCase();
            while (!response.equals("y") && !response.equals("n")) {
                System.out.println("Invalid input. Please enter y for yes or n for no.");
                response = scanner.next().toLowerCase();
            }

            if (!response.equals("y")) {
                break;
            }
        }
    }

    private static Map<String, String> createActionMap(String action) {
        Map<String, String> actionMap = new HashMap<>();

            case "get":
                return createGetActionMap();
            case "post":
                return createPostActionMap();
            default:
                System.out.println("Invalid action.");
                return null;
        }
    }

        map.put("data", "");

        return map;
```
