    }

        }

    public Row selectRow(String tableName, String condition) throws Exception {

        }
        for (List<Row> rows : this.tables.get(currentTable)) {
            if (rows.stream().anyMatch(row -> row.getKey().equals(condition))) {
                return rows.get(rows.size() - 1);
            }
        }
        throw new Exception("No matching table found or no match in the table");
    }

    // Update a row in the database
    public void updateRow(String tableName, String condition, Row row) throws Exception {
        if (currentTable == null || currentTable.equals(tableName)) {
            throw new Exception("Updating from an existing table is not allowed");
        }
        for (List<Row> rows : this.tables.get(currentTable)) {
            if (rows.stream().anyMatch(row -> row.getKey().equals(condition))) {

                return;
            }
        }
        throw new Exception("No matching table found or no match in the table");
    }

    public void deleteRow(String tableName, String condition) throws Exception {
        if (currentTable == null || currentTable.equals(tableName)) {
            throw new Exception("Deleting from an existing table is not allowed");
        }
        for (List<Row> rows : this.tables.get(currentTable)) {
            if (rows.stream().anyMatch(row -> row.getKey().equals(condition))) {
                rows.remove(rows.size() - 1);
                return;
            }
        }
        throw new Exception("No matching table found or no match in the table");
    }

    // Add a new row to the database
    public void addNewRow(String tableName, String key, String value) throws Exception {
        if (currentTable == null || currentTable.equals(tableName)) {
            throw new Exception("Cannot add multiple rows or an empty table is not allowed");
        }
        List<Row> rows = this.tables.get(currentTable);
        rows.add(new Row(key, value));
    }

    // Retrieve existing row values by key
    public Map<String, String> getRowValues(String tableName, String condition) throws Exception {
        if (currentTable == null || currentTable.equals(tableName)) {
            throw new Exception("Selecting from an existing table or an empty table is not allowed");
        }
        for (List<Row> rows : this.tables.get(currentTable)) {
            if (rows.stream().anyMatch(row -> row.getKey().equals(condition))) {
                Map<String, String> map = new HashMap<>();
                map.putAll(rows.get(rows.size() - 1));
                return map;
            }
        }
        throw new Exception("No matching table found or no match in the table");
    }

    // Print the database
    public void printDatabase() {
        for (Map.Entry<String, List<Row>> entry : this.tables.entrySet()) {
            System.out.println(entry.getKey());
            for (Row row : entry.getValue()) {
                System.out.println(row);
            }
        }
    }
}
