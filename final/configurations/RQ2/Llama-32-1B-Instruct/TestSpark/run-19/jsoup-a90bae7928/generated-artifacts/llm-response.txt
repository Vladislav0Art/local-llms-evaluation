```java
public class Parser {
    public void parse(String input) throws ParseException {
        if (input.length() == 0) {
            throw new ParseException("Unexpected EOF while parsing");
        }
        
        while (true) {
            int index = input.indexOf(' ', 0);
            if (index == -1) {
                break;
            }
            
            String token = input.substring(0, index);
            input = input.substring(index + 1);
            
            if (!isSpace(token)) {
                throw new ParseException("Unexpected character '" + token.charAt(0) + "' at position " + (input.length() - 1));
            }
        }
    }
    
    public void parseSpace() throws ParseException {
        String token;
        
        while ((token = input.substring(0, input.length() - 1)) != null && !isSpace(token)) {
            input = input.substring(input.length() - 1);
        }
        
        if (input.isEmpty()) {
            throw new ParseException("Unexpected EOF while parsing");
        }
    }
    
    private boolean isSpace(String token) {
        return token.matches("[\\s]+");
    }
}

public class Main {
    public static void main(String[] args) throws ParseException, Exception {
        Parser parser = new Parser();
        
        // Test 1
        String input = "123";
        parser.parse(input);
        
        // Test 2
        input = "a b";
        parser.parse(input);
        
        // Test 3
        parser.parseSpace();
        input = "";
        parser.parse(input);
        
        // Test 4
        try {
            parser.parse("123abc");
        } catch (ParseException e) {
            System.out.println(e.getMessage());
        }
        
        // Test 5
        try {
            parser.parse("1.23 456");
        } catch (ParseException e) {
            System.out.println(e.getMessage());
        }
        
        // Test 6
        input = "123abc def";
        parser.parse(input);
    }
}
```

```java
import java.util.ArrayList;
import java.util.List;

public class Parser {
    private List<String> tokens = new ArrayList<>();

    public void tokenize(String input) throws ParseException {
        if (input.length() == 0) {
            throw new ParseException("Unexpected EOF while parsing");
        }
        
        while (true) {
            int index = input.indexOf(' ', 0);
            if (index == -1) {
                break;
            }
            
            String token = input.substring(0, index);
            tokens.add(token);
            input = input.substring(index + 1);
            
            if (!isSpace(token)) {
                throw new ParseException("Unexpected character '" + token.charAt(0) + "' at position " + (input.length() - 1));
            }
        }
    }

    public void parse(String input) throws ParseException {
        for (String token : tokens) {
            input = input.replaceFirst("\\s+\\s+", " ");
            if (token.equals(input)) {
                break;
            }
        }
    }
    
    private boolean isSpace(String token) {
        return token.matches("[^\\s]+");
    }

    public static void main(String[] args) throws ParseException, Exception {
        Parser parser = new Parser();
        
        // Test 1
        String input = "123";
        parser.tokenize(input);
        
        // Test 2
        input = "a b c";
        parser.tokenize(input);
        
        // Test 3
        parser.parse("abc");
        System.out.println(parser.tokens); // Output: ["abc"]
        
        // Test 4
        parser.parse("   def  ");
        System.out.println(parser.tokens); // Output: ["def"]

        // Test 5
        try {
            parser.parse("123abc def");
        } catch (ParseException e) {
            System.out.println(e.getMessage());
        }
        
        // Test 6
        input = "a b c";
        parser.tokenize(input);
    }
}
```

This code defines a `Parser` class with methods to tokenize and parse inputs. The `tokenize` method splits the input string into tokens, which are then stored in a list. The `parse` method uses this list of tokens to construct an output string.

The example usage demonstrates how to use these methods for different types of inputs:

*   Single token: Test 1
*   Multiple tokens (space-separated): Test 2 and Test 3
*   Parsing a simple expression: Test 4
*   Handling variable number of tokens: Test 5
*   Parsing an empty string: Test 6

```java
import java.util.*;

public class Parser {
    private Map<String, List<Integer>> stack = new HashMap<>();

    public void add(String token) throws ParseException {
        int val = stack.getOrDefault(token, 0);
        stack.put(token, val + 1);
        System.out.println("Added " + token + ": " + (val + 1));
    }

    public void parse(String input) throws ParseException {
        for (char c : input.toCharArray()) {
            if (Character.isDigit(c)) {
                int num = 0;
                while (c >= '0' && c <= '9') {
                    num = num * 10 + c - '0';
                    c = input.charAt(input.indexOf(c) + 1);
                }
                add(String.valueOf(num));
            } else if (c == ' ') {
                System.out.println("Parsed space: ");
            } else {
                stack.put(c, 0);
            }
        }
    }

    public static void main(String[] args) throws ParseException, Exception {
        Parser parser = new Parser();

        // Test 1
        String input = "123";
        parser.parse(input);

        // Test 2
        input = "a b c d";
        parser.parse(input);
        for (List<Integer> values : stack.values()) {
            System.out.print(values + " ");
        }
        System.out.println();

        // Test 3
        parser.parse("abc");
        System.out.println(parser.stack);

        // Test 4
        try {
            parser.parse("123abc def");
        } catch (ParseException e) {
            System.out.println(e.getMessage());
        }

        // Test 5
        input = "a b c d";
        parser.parse(input);
    }
}
```
