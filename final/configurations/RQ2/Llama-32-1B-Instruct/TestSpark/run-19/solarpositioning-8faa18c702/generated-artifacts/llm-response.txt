public class Solution {
    public boolean isPalindrome(String s) {
        int left = 0;
        int right = s.length() - 1;

        while (left < right) {
            if (!s.charAt(left).equals(s.charAt(right))) {
                return false;
            }
            left += 1;
            right -= 1;
        }

        return true;
    }

    public boolean isPalindrome(String s, int start, int end) {
        while (start <= end) {
            if (!s.charAt(start).equals(s.charAt(end))) {
                return false;
            }
            start++;
            end--;
        }

        return true;
    }

class Solution {
    public boolean isPalindrome(String s) {
        int n = s.length();
        
        // Convert string to array of characters for easier manipulation
        char[] arr = s.toCharArray();
        
        // Initialize two pointers, one at the start and one at the end of the array
        int left = 0;
        int right = n - 1;
        
        // Loop through the array until the two pointers meet
        while (left < right) {
            if (arr[left] != arr[right]) {
                return false;
            }
            left++;
            right--;
        }
        
        return true;
    }

    public boolean isPalindrome(String s) {
        int n = s.length();
        
        // Convert string to array of characters for easier manipulation
        char[] arr = s.toCharArray();
        
        // Initialize two pointers, one at the start and one at the end of the array
        int left = 0;
        int right = n - 1;
        
        // Loop through the array until the two pointers meet
        while (left < right) {
            if (!Character.equals(arr[left], arr[right])) {
                return false;
            }
            left++;
            right--;
        }
        
        return true;
    }

    public boolean isPalindrome(int[] nums) {
        int n = nums.length;
        
        // Convert array of integers to array of characters for easier manipulation
        char[] arr = new char[n];
        System.arraycopy(nums, 0, arr, 0, n);
        
        // Initialize two pointers, one at the start and one at the end of the array
        int left = 0;
        int right = n - 1;
        
        // Loop through the array until the two pointers meet
        while (left < right) {
            if (!Character.equals(arr[left], arr[right])) {
                return false;
            }
            left++;
            right--;
        }
        
        return true;
    }

    public boolean isPalindrome(int[] nums, int start, int end) {
        // If the start index is greater than or equal to the end index,
        // it means there are no characters in the array to compare

            return true;
        }
        
        while (start <= end) {
            if (!Character.equals(nums[start], nums[end])) {
                return false;
            }
            start++;
            end--;
        }
        
        return true;
    }
