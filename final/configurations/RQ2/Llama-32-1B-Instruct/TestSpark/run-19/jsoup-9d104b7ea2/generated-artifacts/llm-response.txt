package org.jsoup.nodes;

import java.io.IOException;
import java.util.function.Function;

public class TextNodeTest {
    public static void createTextNode(String text) {
        return new TextNode(text);
    }

    public static void normalizeWhitespace(Function<String, String> func) {
        return func.apply(normalizeWhitespace);
    }

    public static Function<String, String> stripLeadingWhitespace(Function<String, String> func) {
        return func.apply(stripLeadingWhitespace);
    }

    public static boolean lastCharIsWhitespace(StringBuilder sb) {
        return sb.length() > 0 && Character.isWhitespace(sb.charAt(sb.length() - 1));
    }

    public static TextNode textNodeText(String text) {
        return new TextNode(text);
    }

    public static void testNormaliseWhitespace() throws IOException {
        Function<String, String> func = normalizeWhitespace((s) -> s.trim());
        String text = "   Hello   World!  ";
        assertEquals("Hello World", func.apply(text));
    }

    public static void testStripLeadingWhitespace() throws IOException {
        Function<String, String> func = stripLeadingWhitespace((s) -> s.replaceFirst("^\\s+", ""));
        String text = "   Hello   World!";
        assertEquals("Hello World!", func.apply(text));
    }

    public static void testLastCharIsWhitespace() throws IOException {
        String text = "a";
        assertTrue(lastCharIsWhitespace(new StringBuilder(text)));
        text = "Hello";
        assertFalse(lastCharIsWhitespace(new StringBuilder(text)));
    }

    public static void testTextNodeText() throws IOException {
        TextNode textNode = createTextNode("Test text");
        assertEquals("Test text", textNode.text());
    }
}
