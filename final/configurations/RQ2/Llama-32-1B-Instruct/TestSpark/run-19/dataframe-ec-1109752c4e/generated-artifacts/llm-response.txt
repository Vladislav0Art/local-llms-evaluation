import org.junit.Test;
import static org.junit.Assert.assertArrayEquals;

public class Main {
    @Test
    public void testPublicClass() {
        // Create a new public class called "HelloWorld"
        HelloWorld helloWorld = new HelloWorld();

        // Call the constructor with some arguments
        helloWorld.newMethod(5, 7);

        // Print the value of a field
        System.out.println(helloWorld.fieldValue());

        // Call another method that calls the first one
        helloWorld.method1();
    }

    @Test
    public void testPublicMethod() {
        // Create two public methods in "HelloWorld"
        HelloWorld helloWorld = new HelloWorld();

        // Test if calling a private method from the constructor returns correct value.
        assertArrayEquals(new int[]{5, 7}, helloWorld.method1());

        // Test if calling another method that calls the first one works as expected
        MethodHelper helper = new MethodHelper();
        helper.method2(helloWorld);
    }

    @Test
    public void testPublicMethodReturnsCorrectValue() {
        // Create a new public method in "HelloWorld"
        HelloWorld helloWorld = new HelloWorld();

        // Call the method with some arguments and store the returned value
        int result = helloWorld.myMethod(5, 7);

        // Print the stored value
        System.out.println(result);
    }

    private class HelloWorld {
        private String fieldValue;
        private MethodHelper myMethod;

        public HelloWorld() {
            this.fieldValue = "Hello";
            this.myMethod = new MethodHelper();
            this.myMethod.myMethod2(this);
        }

        public int method1() {
            return 5;
        }

        public void method2(HelloWorld helloWorld) {
            System.out.println("Method2 called with value: " + helloWorld.fieldValue);
            // Test if calling a private method from the constructor returns correct value.
            assertArrayEquals(new int[]{5, 7}, helloWorld.method1());
        }
    }

    private static class MethodHelper {
        public void myMethod(int x, int y) {
            System.out.println("myMethod called with values: " + x + ", " + y);
        }
    }
}
