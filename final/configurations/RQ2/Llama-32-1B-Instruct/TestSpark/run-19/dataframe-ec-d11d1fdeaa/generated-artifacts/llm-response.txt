const fs = require('fs');
const { createReadStream, readFile } = require('fs-promises');

// Define the schema of the database
class DatabaseSchema {
    constructor() {
        this.columns = {};
    }

    addColumn(name, type) {
        this.columns[name] = type;
    }

    deleteColumn(name) {
        if (this.columns[name]) {
            delete this.columns[name];
        }
    }

    addRow(id, data) {
        const row = Object.keys(data).map(key => ({ [key]: data[key] })).reduce((acc, curr) => acc.concat(curr), []);
        this.id = id;
        this.rows[id] = row;
    }

    getRow(id) {
        if (this.rows[id]) {
            return this.rows[id];
        }
        return null;
    }

    close() {
        // Close the database connection

        try {
            fs.unlinkSync(dbPath);
        } catch (err) {
            console.error(err);
        }
    }
}

class Database {
    constructor(schema) {
        this.schema = schema;
        this.dbPath = './database.json';
    }

    addRow(id, data) {
        const row = Object.keys(data).map(key => ({ [key]: data[key] })).reduce((acc, curr) => acc.concat(curr), []);
        this.rows[id] = row;
    }

    getRow(id) {
        return this.rows[id];
    }

// Function to load the database schema
function loadSchema(dbPath) {
    const db = new Database(new DatabaseSchema());
    try {
        const jsonFileContent = fs.readFileSync(dbPath, 'utf8');
        const data = JSON.parse(jsonFileContent);
        for (const tableName in data.columns) {
            if (!db.schema.addColumn(tableName, data.columns[tableName])) {
                console.error(`Column "${tableName}" does not exist.`);
            }
        }
    } catch (err) {
        console.error(err);
    }
    return db;
}

// Function to save the database schema
function saveSchema(dbPath, db) {
    const jsonFileContent = JSON.stringify(db.schema.columns, null, 2);
    fs.writeFileSync(dbPath, jsonFileContent);
}

// Main function
async function main() {
    // Load the database schema from a file
    const db = await loadSchema('./database.json');

    // Create a new row with id=1 and data={'name': 'John', 'age': 25}
    const rowData = { 'name': 'John', 'age': 25 };
    db.addRow(1, rowData);
