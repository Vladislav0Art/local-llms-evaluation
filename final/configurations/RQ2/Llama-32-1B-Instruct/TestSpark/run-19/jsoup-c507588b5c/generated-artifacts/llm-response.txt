import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.junit.jupiter.api.Test;
import org.mockito.Mockito;

public class TextNodeTest {

    @Test
    public void testNodeName() {
        String text = "Hello, World!";
        TextNode node = new TextNode(text);
        assertEquals("Hello, World!", node.nodeName());
    }

    @Test
    public void testText() {
        String text = "Hello, World!";
        Element element = Mockito.mock(Element.class);
        TextNode node = new TextNode(text);
        Mockito.when(element.getText()).thenReturn(text);
        assertEquals(text, node.text(text));
    }

    @Test
    public void testGetWholeText() {
        String text = "Hello, World!";
        Document document = Mockito.mock(Document.class);
        Element element = Mockito.mock(Element.class);
        TextNode node = new TextNode(text);
        Mockito.when(element.getWholeText()).thenReturn(text);
        assertEquals(text, node.getText());
        Mockito.verify(document).getWholeText().withAnyMatch((wholeText) -> wholeText.equals(text));
    }

    @Test
    public void testIsBlank() {
        String text = "   Hello  ";
        TextNode node = new TextNode(text);
        assertTrue(node.isBlank());
    }

    @Test
    public void testSplitText() {
        String text = "Hello, World!";
        int offset = 10;
        Element element = Mockito.mock(Element.class);
        TextNode node = new TextNode(text);
        Mockito.when(element.getWholeText()).thenReturn(text);
        assertEquals(3, node.splitText(offset).getWholeText().length());
    }

    @Test
    public void testOuterHtmlHead() {
        String text = "Hello, World!";
        Document document = Mockito.mock(Document.class);
        Element element = Mockito.mock(Element.class);
        TextNode node = new TextNode(text);
        Mockito.when(element.getOuterHtml()).thenReturn("Hello, World!");
        assertEquals("Hello, World!", node.outerHtmlHead(document, 0, null));
    }

    @Test
    public void testOuterHtmlTail() {
        String text = "Hello, World!";
        Document document = Mockito.mock(Document.class);
        Element element = Mockito.mock(Element.class);
        TextNode node = new TextNode(text);
        Mockito.when(element.getOuterHtml()).thenReturn("Hello, World!");
        assertEquals("Hello, World!", node.outerHtmlTail(document, 0, null));
    }

    @Test
    public void testToString() {
        String text = "Hello, World!";
        Element element = Mockito.mock(Element.class);
        TextNode node = new TextNode(text);
        Mockito.when(element.toString()).thenReturn("Hello, World!");
        assertEquals("Hello, World!", node.toString());

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.junit.Test;

public class GeneratedTest {

    @Test
    public void testNodeToString() {
        Document document = createMockDocument();
        Element node = newElement(document);
        String result = (String) node.toString();
        assertEquals("HEllo, WorlD", result);
    }

    private Document createMockDocument() {
        return createMockHTMLDocument("Hello, World!");
    }

    @Test
    public void testNodeTextLength() {
        Document document = createMockDocument();
        Element node = newElement(document);
        String result1 = (String) node.text().replaceAll("\\n", "");
        assertEquals(3, Integer.parseInt(result1.splitText(0).getWholeText().length()));
        String result2 = (String) node.text().replaceAll("\\n", "");
        assertEquals("Hello, World!", result2);
    }

    @Test
    public void testNodeIsBlank() {
        Document document = createMockDocument();
        Element node = newElement(document);
        assertTrue(node.isBlank());
    }

    private Document createMockHTMLDocument(String content) {
        Document document = new Document();
        Elements elements = document.createElement("html").appendChild(createElement("head"));
        Document head = document.createElement("head");
        head.appendChild(createElement("title", "Hello, World!"));
        elements.appendChild(head);
        return document;
    }

    private Element createElement(String tag, String content) {
        return new Element(tag).text(content);
    }
}

import org.jsoup.nodes.Document;

public class GeneratedTest {

    public static Document createMockDocument() {
        return new Document();
