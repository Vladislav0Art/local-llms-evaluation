import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Main {

    public static String parseHtml(String html) {
        Document doc = Jsoup.parse(html);
        return doc.html();
    }

    public static Element createElement(String tagName, String attrs, String text) {
        Element elem = new Element(tagName, attrs);
        if (text != null) {
            elem.text(text);
        }
        return elem;
    }

    @Test
    public void testParseHtml() {
        String html = "<p>This is a paragraph of text.</p>";
        System.out.println(parseHtml(html));
    }

    @Test
    public void testCreateElement() {
        Element elem = createElement("b", "", "Bold");
        System.out.println(elem.html());
    }

    @Test
    public void testCreateMultipleElements() {
        Element elem1 = createElement("span", "", "First span");
        Element elem2 = createElement("em", "", "Second em");
        Element elem3 = createElement("i", "", "Third i");
        System.out.println(String.join("\n", elem1.html(), elem2.html(), elem3.html()));
    }

    @Test
    public void testEmptyText() {
        String html = "<p>This is a paragraph of text.</p>";
        System.out.println(parseHtml(html).text());
    }

    @Test
    public void testRemoveClass() {
        Element elem = createElement("span", "", "TestClass");
        elem.addClass("TestClass2");
        elem.removeClass("TestClass3");
        System.out.println(elem.html());
    }
}

        System.out.println(tag1.html());
    }

    public void testCreateMultipleElements() {
        Document doc = Jsoup.parse("<html><body><p>This is a paragraph of text.</p><br><span>Hello World!</span></body></html>");

        Tag tag2 = parseElement("span", "", "");
        System.out.println(tag2.html());
    }

    @Test
    public void testParseHtmlWithEmptyString() {

        System.out.println(parseHtml(html));

    public void testCreateElementWithEmptyString() {

        Tag tag = parseElement("", "", "");
        System.out.println(tag.html());
    }
}

    public String text;

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Main {

    public static String parseHtml(String html) {
        Document doc = Jsoup.parse(html);
        return doc.html();
    }

    public static Tag parseElement(String tagName, String attrs, String text) {
        Element elem = new Element(tagName, attrs, text);
        if (text != null && !text.isEmpty()) {
            elem.text(text);
        }
        return elem;
    }

    @Test
    public void testParseHtml() {
        String html = "<p>This is a paragraph of text.</p>";
        Tag tag = parseElement("p", new java.util.HashMap<>(), html);
        System.out.println(tag.toString());
    }

    @Test
    public void testParseEmptyHtml() {
        String html = "";
        Tag tag = parseElement("p", new java.util.HashMap<>(), html);
        System.out.println(tag.toString());
    }

    @Test
    public void testCreateElementWithEmptyString() {
        Document doc = Jsoup.parse("<html><body><span>Hello World!</span></body></html>");
        Element elem = doc.body().child("span");
        Tag tag = parseElement("", "", elem.text());
        System.out.println(tag.toString());
    }

    @Test
    public void testCreateString() {
        String html = "";
        Tag tag = parseElement("p", new java.util.HashMap<>(), html);
        System.out.println(tag.toString());
    }
}

class Tag extends Element {
    public String text;
    public Object attrs;

    @Override
    public String toString() {
        return "Tag{" +
                "text='" + text + '\'' +
                ", attrs=" + attrs +
                '}';
    }

    public void setAttributes(java.util.Map<String, String> attrs) {
        this.attrs = attrs;
    }
}
