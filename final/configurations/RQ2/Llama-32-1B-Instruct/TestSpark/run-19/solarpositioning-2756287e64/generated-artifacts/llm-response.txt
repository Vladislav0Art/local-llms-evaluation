    public static double calculateSolarPosition(ZonedDateTime date, double latitude, double longitude, double deltaT, double pressure, double temperature) {

        double latRad = Math.toRadians(latitude);
        double lonRad = Math.toRadians(longitude);

        // Calculate the solar altitude angle using trigonometry
        double solarAltitudeAngle = Math.asin(Math.sin(latRad) * Math.cos(deltaT) / Math.sin(latRad));

        // Calculate the solar declination angle in degrees
        double declinationAngleInDegrees = 23.45 * deltaT - 2;

        // Convert the solar altitude and declination angles from radians to degrees
        double solarAltitudeAngleInDegrees = Math.toDegrees(solarAltitudeAngle);
        double declinationAngleInDegrees = Math.toDegrees(declinationAngle);

        // Calculate the angle between the sun's position and the observer's location
        double observerLocationAngleInRadians = Math.asin(Math.sin(latRad) * Math.cos(deltaT) / Math.sin(latRad));
        
        // Calculate the solar altitude angle in degrees using trigonometry
        double solarAltitudeAngleInDegreesFromTrigonometric = Math.toDegrees(Math.atan2(Math.sin(solarAltitudeAngleInRadians), Math.cos(solarAltitudeAngleInRadians)));

        // Combine the declination angle, observer location angle, and solar altitude angle to get the final result
        double result = 90 - Math.min(declinationAngleInDegrees, Math.abs(observerLocationAngleInRadians - solarAltitudeAngleInDegreesFromTrigonometric));

    }
