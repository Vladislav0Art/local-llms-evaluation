import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class TestJsoup {
    private Document doc;

    public void testAsString() {
        String xpath = "test";
        Element element = new Element(xpath);
        document(doc).append(element);

        List<String> nodes = new ArrayList<>();
        try {
            node(nodes, "test");
            assertEquals(1, nodes.size());
        } catch (Exception e) {
            System.out.println(e.getMessage());
        }

        String result = W3CDom.asString(doc);
        System.out.println(result);
    }

    public void testSourceNodes() {
        String xpath = "test";
        try {
            W3CDom.sourceNodes(doc, xpath);
            assertEquals(2, doc nodeList().size());
        } catch (Exception e) {
            System.out.println(e.getMessage());
        }
    }

    private void node(List<String> nodes, String element) {
        Element node = new Element(element);
        document(doc).append(node);
        nodes.add(node.outerHtml());
    }

    public static void main(String[] args) {
        TestJsoup testJsoup = new TestJsoup();
        testJsoup.testAsString();
        testJsoup.testSourceNodes();
    }
}

        System.out.println(W3CDom.namespaceAware(true));
    }

    @Test
    public void namespaceAware_false() {
        Document doc = new Document();
        System.out.println(W3CDom.namespaceAware(false));
    }

    @Test
    public void namespaceAwareConstructor_implementation() {
        W3CDomHelper helper = new W3CDomHelper();
        Assert.assertTrue(helper.isNamespaceAware());
    }

    @Test
    public void convertDocument_conversion() {
        Document doc = new Document("<html><body>Hello World!</body></html>");
        String result = W3CDom.convert(doc, null);
        System.out.println(result);
    }

    @Test
    public void convertDocToString_conversion() throws Exception {
        Document doc = new Document("<html><body>Hello World!</body></html>");
        String result = W3CDom.convert(doc, null);
        Assert.assertTrue(result.contains("Hello"));
    }

    public static class Document {
        public Node body;

        public static Document create(String html) {
            Document document = new Document();
            document.body = parse(html).body;
            return document;
        }
    }

    public static class W3CDomHelper {
        public boolean isNamespaceAware() {
            return false;
        }
    }

    public static void main(String[] args) throws Exception {
        Document doc = new Document("<html><body>Hello World!</body></html>");
        String result = W3CDom.convert(doc, null);
        System.out.println(result);

        Assert.assertTrue(doc.body.getTextContent().contains("Hello"));
    }
}
