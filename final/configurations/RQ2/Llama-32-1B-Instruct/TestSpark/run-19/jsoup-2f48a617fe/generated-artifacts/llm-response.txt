        assertEquals("This is a test comment", comment.getData());

    public void outerHtmlHeadTest() throws IOException {
        ParseSettings settings = new ParseSettings();
        settings.setDocumentMode(0);
        Comment comment = new Comment("This is a test comment");

        assertEquals(expectedOuterHtmlHead, comment.outerHtmlHead(settings, 1, null));
    }

    public void outerHtmlTailTest() {
        ParseSettings settings = new ParseSettings();
        Comment comment = new Comment("This is a test comment");

        assertEquals(expectedToString, comment.toString());

    public void cloneTest() {

        Comment clonedComment = originalComment.clone();

Here's a Java solution that uses Jsoup library to parse HTML and compare elements:

```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Main {
    public static void main(String[] args) {
        // Test string
        String htmlString = "<html><head><title>Test</title></head><body><h1>Hello World!</h1><p>This is a test.</p></body></html>";

        // Parse the HTML string to get the parsed tree
        Document document = Jsoup.parse(htmlString);

        // Find all head and tail elements in the parsed tree
        Element head = document.head();
        Element tail = document.tail();

        // Compare the actual and expected elements
        for (Element element : head.getAsElements()) {
            if (!element.equals(document.getElementById(0))) {
                System.out.println(element + " does not match");
            }
        }

        for (Element element : tail.getAsElements()) {
            if (!element.equals(document.getId(1))) {
                System.out.println(element + " does not match");
            }
        }

        // Parse the HTML string to get the parsed tree
        Document newDocument = Jsoup.parse(htmlString);

        // Find all head and tail elements in the new parsed tree
        Element actualHead = newDocument.head();
        Element actualTail = newDocument.tail();

        // Compare the actual and expected elements
        for (Element element : actualHead.getAsElements()) {
            if (!element.equals(document.getElementById(0))) {
                System.out.println(element + " does not match");
            }
        }

        for (Element element : actualTail.getAsElements()) {
            if (!element.equals(document.getId(1))) {
                System.out.println(element + " does not match");
            }
        }

        // Parse the HTML string to get the parsed tree
        Document document2 = Jsoup.parse(htmlString);

        // Find all head and tail elements in the new parsed tree
        Element actualHead2 = document2.head();
        Element actualTail2 = document2.tail();

        // Compare the actual and expected elements
        for (Element element : actualHead2.getAsElements()) {
            if (!element.equals(document2.getElementById(0))) {
                System.out.println(element + " does not match");
            }
        }

        for (Element element : actualTail2.getAsElements()) {
            if (!element.equals(document2.getId(1))) {
                System.out.println(element + " does not match");
            }
        }
    }
}
```

This solution uses Jsoup library to parse the HTML string and compare elements. It finds all head and tail elements in both parsed trees and compares their values. The expected elements for each comparison are retrieved using `document.getElementById(0)` and `document.getId(1)`.
