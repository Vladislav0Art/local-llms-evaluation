import com.github.davidmoten.guavamini.Preconditions;
import com.github.davidmoten.guavamini.annotations.VisibleForTesting;
import org.junit.jupiter.api.Test;

public class WordWrap {

    public static Builder from(Reader reader) { 
        return new Builder();
    }

    public static Builder from(String input, String encoding) {
        return new Builder(input, encoding);
    }

    public static class Builder {
        private StringBuilder sb = new StringBuilder();
        
        @VisibleForTesting
        public Builder withSeparator(char separator) {
            sb.append(separator);
            return this;
        }
        
        @VisibleForTesting
        public Builder addWord(String word) {
            sb.append(word).append(" ");
            return this;
        }
        
        @VisibleForTesting
        public String toString() {
            return sb.toString();
        }

        private static int minMaxWidth(int min, int max) {
            return Math.min(min, max);
        }

    }
}
