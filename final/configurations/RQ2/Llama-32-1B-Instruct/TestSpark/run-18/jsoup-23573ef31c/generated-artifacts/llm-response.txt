public class DocumentTest {
    private Document document;
    private String baseUri;

    @Before
    public void setup() {
        baseUri = "http://example.com";
        // Create a Jsoup test server for testing connections
        Jsoup.connect("http://" + baseUri).get().text();
    }

    [Document][createShell]Test createShellTest() {
        String expectedDocType = "html5";
        document = new Document(baseUri);
        return this;
    }

    [Document][location]Test locationTest() {
        assertEquals(expectedDocType, document.location());
        return this;
    }

    [Document][connection]Test connectionTest() {
        Connection connection = document.connection();
        assertEquals("http://example.com", connection.url().toString());
        return this;
    }

    [Document][documentType]Test documentTypeTest() {
        DocumentType documentType = document.documentType();
        assertEquals(expectedDocType, documentType.getType());
        return this;
    }

    [Document][head]Test headTest() {
        Element head = document.head();
        assertNotNull(head);
        assertEquals("DOCTYPE html", head.outerHtml().trim());
        return this;
    }

    [Document][body]Test bodyTest() {
        Element body = document.body();
        assertNotNull(body);
        assertEquals("<html><head></head><body></body></html>", body.outerHtml().trim());
        return this;
    }

    [Document][forms]Test formsTest() {
        List<FormElement> forms = document.forms();
        assertNotNull(forms);
        for (FormElement form : forms) {
            assertNotNull(form);
            assertEquals("form", form.tagName());
        }
        return this;
    }

    [Document][expectForm]Test expectFormTest() {
        String cssQuery = "input[name=\"test\"]";
        FormElement form = document.expectForm(cssQuery);
        assertNotNull(form);
        assertEquals("form", form.tagName());
        return this;
    }

    [Document][title]Test titleTest() {
        document.title("Example Title");
        return this;
    }

    [Document][text]Test textTest() {
        document.text("Example Text");
        return this;
    }

    @After
    public void tearDown() {}

    [Document][nodeName]Test nodeNameTest() {
        assertEquals("html", document.nodeName());
        return this;
    }
}
