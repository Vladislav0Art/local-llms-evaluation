import com.google.protobuf.ByteString;
import com.google.protobuf.ByteStringBytesBuilder;
import com.google.protobuf.ByteValue;
import org.junit.Test;

public class XdrPublicKeyTest {
    @Test
    public void testFromXdrPublic() throws Exception {
        String publicKey = "Hello, World!";
        ByteString xdrData = Bytes.Builder().applyTag("tag").appendString(publicKey).build();
        EdDSAPublicKey publicKeyData = fromXdrBytes(xdrData);
        assertNotNull(publicKeyData);
    }

    @Test
    public void testEquals() throws Exception {
        String publicKey1 = "Hello, World!";
        String publicKey2 = "Hello, World!";
        EdDSAPublicKey publicKey1Data = fromXdrBytes(new ByteString("tag", BytesBuilder().appendString(publicKey1).build()));
        EdDSAPublicKey publicKey2Data = fromXdrBytes(new ByteString("tag", BytesBuilder().appendString(publicKey2).build()));
        assertTrue(Objects.equals(publicKey1, publicKey2));
    }

    @Test
    public void testGetPublicKey() throws Exception {
        String publicKey = "Hello, World!";
        EdDSAPublicKey publicKeyData = fromXdrBytes(new ByteString("tag", BytesBuilder().appendString(publicKey).build()));
        String publicKeyStr = publicKeyData.publicKey();
        assertEquals(publicKey, publicKeyStr);
    }

    @Test
    public void testGetPrivateKey() throws Exception {
        String privateKey1 = "Hello, World!";
        EdDSASignatureSpec signatureSpec1 = new EdDSASignatureSpec("tag", BytesBuilder().appendString(privateKey1).build());
        EdDSASignatureSpec signatureSpec2 = new EdDSASignatureSpec("tag", BytesBuilder().appendString(privateKey2).build());
        ByteString privateKeyData1 = fromXdrBytes(new ByteString("tag", BytesBuilder().appendString(privateKey1).build()));
        ByteString privateKeyData2 = fromXdrBytes(new ByteString("tag", BytesBuilder().appendString(privateKey2).build()));

        EdDSASignatureSpec signatureSpec1Data = fromXdrSignature(signatureSpec1);
        EdDSASignatureSpec signatureSpec2Data = fromXdrSignature(signatureSpec2);

        assertEquals(privateKeyData1, privateKeyData2);
    }

    private EdDSAPublicKey fromXdrBytes(ByteString xdrData) throws Exception {
        return fromXdrBytes(new BytesBuilder()
                .applyTag("tag")
                .appendString(xdrData.toString())
                .build());
    }

    private EdDSASignatureSpec fromXdrSignature(EdDSASignatureSpec signatureSpec) throws Exception {
        return fromXdrBytes(new BytesBuilder()
                .applyTag("tag")
                .appendString(signatureSpec.publicKey())
                .build());
    }
}
