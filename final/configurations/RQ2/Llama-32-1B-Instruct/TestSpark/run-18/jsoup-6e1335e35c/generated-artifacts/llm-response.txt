import bs4
from bs4 import Comment

def get_html(element):
    """Get HTML of an element"""
    return element.get_text()

def clone_html(html):
    """Clone the html and create a new Document object"""
    soup = bs4.BeautifulSoup(html, 'html.parser')
    doc = soup.prettify()
    return doc

def clear_attributes(element):
    """Clear attributes of an element"""
    for attr in list(element.attrs.keys()):
        del element[attr]
    if not element.has_attr('id'):
        return
    element['id'] = ''
    # Clear other attributes as well
    for subelem in element.find_all(True):
        if not isinstance(subelem, Comment):  # Skip Comments
            clear_attributes(subelem)

def remove_attribute(element, attr):
    """Remove an attribute of an element"""
    if attr in element.attrs:
        del element[attr]
    elif element.has_attr('id'):
        element['id'] = ''

def add_attribute(element, attr, value):
    """Add a new attribute to an element"""
    if attr in element.attrs:
        element[attr] += f' {value}'
    else:
        element[attr] = value

def add_class(element, class_name):
    """Add a new class to an element"""
    for elem in element.find_all(True):
        if isinstance(elem, Comment):  # Skip Comments
            continue
        elif not isinstance(elem, str) or elem.startswith('#'):
            continue
        if 'class' not in elem.attrs:
            elem['class'] = []
        add_attribute(elem, 'class', class_name)

def toggle_class(element, class_name):
    """Toggle a class of an element"""
    for attr in list(element.attrs.keys()):
        del element[attr]
    if class_name in element.get('class', []):
        element['class'].remove(class_name)
    else:
        element['class'] = ['new-class' if c == class_name else c for c in element.get('class', [])]

    """Get HTML of an element by its ID"""
