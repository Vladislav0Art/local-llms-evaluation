            this.getKeyValues = new ArrayList<>();
        }

        public String getHtmlDocument() {
            return htmlDocument;
        }

        public String getExpectedHtml() {
            return expectedHtml;
        }

        public List<String> getExpectedIds() {
            return expectedIds;
        }

        public List<Map<String, Object>> getExpectedValues() {
            if (expectedValues == null) {
                try {
                    Document document = new Document(htmlDocument);
                    Elements elements = document.getAllElements();
                    for (Element element : elements) {
                        String id = element.attr("id");
                        if (!id.equals("")) {
                            expectedIds.add(id);
                        }
                    }

                    List<Map<String, Object>> values = new ArrayList<>();
                    for (String id : expectedIds) {
                        Map<String, Object> map = new HashMap<>();
                        map.put("value", document.getElementById(id).attr("value"));
                        map.put("id", id);
                        values.add(map);
                    }

                    this.expectedValues = values;
                } catch (Exception e) {
                    throw new RuntimeException(e);
                }
            }
            return expectedValues;
        }

        public void put(Element element) {
            if (!element.attr("name").equals("")) {
                String id = element.attr("id");
                Map<String, Object> map = new HashMap<>();
                map.put("value", element.attr("value"));
                map.put("id", id);
                this.getKeyValues.add(map);
            }

                map.put("value", element.attr("value"));

            }
