public class Main {

    /**
     * Appends a key-value pair to a string.
     *
     * @param str  The input string.
     * @param key  The key of the value to append.
     * @param value The value to be appended.
     * @return The modified string with the key-value pair appended.
     */
    public String appendKeyVal(String str, String key, Object value) {
        return str + " " + key + "=" + value;
    }

    /**
     * Formats a string by inserting an integer at each space between two numbers.
     *
     * @param num1 The first number to insert at the beginning of the string.
     * @param num2 The second number to be inserted after num1.
     * @return The modified string with an integer inserted at each space before the next number.
     */
    public String stringFormat(int num1, int num2) {
        StringBuilder sb = new StringBuilder();
        if (num1 > 0) {
            sb.append(num1);
        }
        sb.append(" ");
        if (num2 > 0) {
            sb.append(num2);
        }

        while (sb.length() < 10 && num1 != 0) {
            sb.append(num1);
            num1--;
        }

        if (sb.length() < 9) {
            sb.append("...");
        } else {
            sb.deleteCharAt(sb.length() - 9);
        }
        return sb.toString();
    }

    public static void testAppendKeyVal() {
        Main obj = new Main();

        String str1 = obj.appendKeyVal("name", "John Doe", "age=30");
        System.out.println(str1); // Output: name John Doe age=30

        int num2 = 5;
        int num3 = 7;

        String result = obj.stringFormat(num2, num3);
        System.out.println(result); // Output: 5 7...

    }

    public static void main(String[] args) {
        Main obj = new Main();

        testAppendKeyVal();
    }
}

    /**
     * Appends a key-value pair to a string.
     *
     * @param str  The input string.
     * @param key  The key of the value to append.
     * @param value The value to be appended.
     * @return The modified string with the key-value pair appended.
     */
    public String appendKeyVal(String str, String key, Object value) {
        // Append the key-value pair
        return str + " " + key + "=" + value;
    }

    /**
     * Formats a string by inserting an integer at each space between two numbers.
     *
     * @param num1 The first number to insert at the beginning of the string.
     * @param num2 The second number to be inserted after num1.
     * @return The modified string with an integer inserted at each space before the next number.
     */
    public String stringFormat(int num1, int num2) {
        StringBuilder sb = new StringBuilder();
        
        // Insert num1
        if (num1 > 0) {
            sb.append(num1);
        }
        sb.append(" ");
        
        // Insert num2
        if (num2 > 0) {
            sb.append(num2);
        }

        while (sb.length() < 10 && num1 != 0) {
            sb.append(num1);
            num1--;

        if (sb.length() < 9) {
            sb.append("...");
