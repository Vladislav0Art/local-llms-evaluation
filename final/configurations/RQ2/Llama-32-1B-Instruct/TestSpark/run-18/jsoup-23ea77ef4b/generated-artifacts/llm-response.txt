```java
public class HtmlProcessor {
    public static List<String> processHtml(String htmlString) {
        List<String> list = new ArrayList<>();
        if (htmlString.startsWith("<")) {
            String startTag = htmlString.substring(1);
            int endTagIndex = startTag.indexOf(">");
            if (endTagIndex != -1) {
                String tag = startTag.substring(0, endTagIndex).trim();
                List<String> attributes = new ArrayList<>();
                for (int i = 0; i < endTagIndex; i++) {
                    if (!tag.isEmpty()) {
                        attributes.add(tag.charAt(i));
                    }
                }

                for (String attribute : attributes) {
                    list.add(attribute);
                }

                String content = htmlString.substring(endTagIndex + 1).trim();

                while (!content.isEmpty() && !content.startsWith("/>")) {
                    if (!content.contains("<")) {
                        break;
                    } else {
                        int index = content.indexOf(">");
                        if (index != -1) {
                            StringBuilder sb = new StringBuilder();
                            for (int i = 0; i < index; i++) {
                                sb.append(content.charAt(i));
                            }
                            String tag = sb.toString();
                            list.add(tag);
                            list.add(content.substring(index + 1).trim());
                        } else {
                            break;
                        }
                    }
                }

                while (!content.isEmpty() && !content.startsWith("/>")) {
                    if (!content.contains(">")) {
                        content += ">";
                    } else {
                        int index = content.indexOf(">");
                        String tag = content.substring(0, index).trim();
                        list.add(tag);
                        content = content.substring(index + 1).trim();
                    }
                }

                return list;
            } else {
                return new ArrayList<>();
            }
        } else {
            return new ArrayList<>();
        }
    }
```

```java
public class HtmlParser {
    public static List<String> parseHtml(String htmlString) {
        List<String> list = new ArrayList<>();
        String[] lines = htmlString.split("<[^>]*>");
        
        for (String line : lines) {
            if (!line.isEmpty()) { // Skip empty lines
                String tagStart = line.substring(0, 1).trim();
                if (tagStart.equals("</")) {
                    break;
                }
                
                int openingIndex = line.indexOf(">");
                String tag = line.substring(1, openingIndex).trim();
                List<String> attributes = new ArrayList<>();
                for (int i = 0; i < openingIndex; i++) {
                    if (!tag.isEmpty()) {
                        attributes.add(tag.charAt(i));
                    }
                }

                while (!attributes.isEmpty() && !attributes.get(attributes.size() - 1).equals(">")) {
                    String attribute = attributes.remove(attributes.size() - 1);
                    list.add(attribute + "=" + attribute.substring(0, attribute.length() - 1) + ">");
                }

                list.add(tag);

                int closingIndex = line.indexOf("</");
                if (closingIndex != -1 && closingIndex < line.length()) {
                    String content = line.substring(closingIndex + 2).trim();
                    while (!content.isEmpty() && !content.startsWith("/>")) {
                        if (!content.contains(">")) {
                            break;
                        } else {
                            int index = content.indexOf(">");
                            StringBuilder sb = new StringBuilder();
                            for (int i = 0; i < index; i++) {
                                sb.append(content.charAt(i));
                            }
                            String tag = sb.toString();
                            list.add(tag);
                            list.add(content.substring(index + 1).trim());
                        }
                    }

                    while (!content.isEmpty() && !content.startsWith("/>")) {
                        if (!content.contains(">")) {
                            content += ">";
                        } else {
                            int index = content.indexOf(">");
                            String tag = content.substring(0, index).trim();
                            list.add(tag);
                            content = content.substring(index + 1).trim();
                        }
                    }
                }

                list.add("</" + tag + ">"); // Close the tag
            }
        }

        return list;
    }

    public static void main(String[] args) {
        String htmlString = "<html><body>Hello World!</body></html>";
        System.out.println(parseHtml(htmlString));
    }
}
```

In this solution, we're using regular expressions to extract the tags and their attributes from the HTML string. We're also handling the case when a tag is closed by checking if it ends with "/>" instead of ">".

This solution assumes that the input HTML string is well-formed and does not contain any invalid or malformed XML/HTML tags. It may not work correctly for such cases.
