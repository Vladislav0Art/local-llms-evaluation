public class TagUtil {

    public static String removeEnforcedAttribute(String[] args, String expected) {
        ArrayList<String> tags = new ArrayList<>();
        for (String tag : args) {
            if (!tag.equals(expected)) {
                tags.add(tag);
            }
        }
        return joinTags(tags);
    }

    private static String joinTags(ArrayList<String> tags) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < tags.size(); i++) {
            if (i > 0) {
                sb.append(" ");
            }
            sb.append(tags.get(i));
        }
        return sb.toString();
    }

    public static boolean isSafeTag(String tag, ArrayList<String> attributes, String defaultValue) {
        for (String attribute : attributes) {
            if (attribute.equals(tag)) {
                return false;
            }
        }
        return true;
    }
}
