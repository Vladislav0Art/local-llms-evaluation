import java.net.URI;
import java.net.URISyntaxException;

public class UrlTest {
    public void test_getUrl() throws URISyntaxException {
        URI url = new URI("http://example.com");
        System.out.println(url);
    }

    public static void main(String[] args) throws URISyntaxException {
        // Create a URL object
        URI uri = new URI("http://example.com");

        // Get the scheme (protocol)
        String scheme = uri.getScheme();

        // Get the authority (domain)
        String authority = uri.getAuthority();

        // Get the host and port
        int port = 0;
        String[] hostParts = uri.getHost().split(":");
        if (hostParts.length > 1) {
            port = Integer.parseInt(hostParts[1]);
        }

        // Print the URL components
        System.out.println("Scheme: " + scheme);
        System.out.println("Authority: " + authority);
        System.out.println("Port: " + port);

        // Get the path and query parameters
        String path = uri.getPath();
        String[] queryParts = uri.getQuery().split("&");
        StringBuilder queryParams = new StringBuilder();

        for (String param : queryParts) {
            if (!param.isEmpty()) {
                queryParams.append(param).append("=");
                break;
            }
        }

        System.out.println("Path: " + path);
        System.out.println("Query Parameters:");
        System.out.println(queryParams.toString());

        // Get the fragment
        String fragment = uri.getFragment();

        // Print the URL components
        System.out.println("Path with query parameters: " + path + "?" + queryParams.toString());
        System.out.println("Fragment: " + fragment);
    }
}

import java.util.logging.Level;
import java.util.logging.Logger;

public class HttpRequestTest {
    private static final String DEFAULT_URL = "http://example.com";

    public void test_getRequest() {
        // Create a URL object
        URL url = new URL(DEFAULT_URL);

        try (Connection connection = (Connection) ((HttpURLConnection) url).openConnection()) {
            // Send a GET request and print the response code
            System.out.println("Response Code: " + connection.getResponseCode());

            // Check if the request was successful
            if (connection.getResponseCode() >= 200 && connection.getResponseCode() < 300) {
                System.out.println("Request was successful");
            } else {
                System.out.println("Request failed with code " + connection.getResponseCode());
            }

            // Read the response content and print it
            String content = new java.net.MimeReader().read(connection.getInputStream());
            System.out.println("Response Content: " + content);
        } catch (IOException e) {
            Logger.getLogger(HttpRequestTest.class.getName()).log(Level.SEVERE, null, e);
        }
    }

    public static void main(String[] args) {
        HttpRequestTest test = new HttpRequestTest();

        // Test a GET request
        test.test_getRequest();
    }
}
