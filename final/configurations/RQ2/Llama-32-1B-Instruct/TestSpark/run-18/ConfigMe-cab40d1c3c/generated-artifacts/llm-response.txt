```java
public class TestConfigurationData {
    private Configuration configuration;

    @BeforeEach
    void setup() {
        this.configuration = new Configuration();
    }

        // Create a new property with the given key and value
        Property property = new Property(key, value);
        
        // Add the property to the configuration
        this.configuration.getProperties().add(property);

        return "Added: " + property.getKey() + " = " + property.getValue();
    }

    public String updateProperty(String key, Object newValue) {
        // Update an existing property in the configuration
        for (Property property : this.configuration.getProperties()) {
            if (property.getKey().equals(key)) {
                property.setValue(newValue);
                return "Updated: " + property.getKey() + " = " + property.getValue();
            }
        }

        return "No such property found";
    }

    public String removeProperty(String key) {
        // Remove an existing property from the configuration
        for (Property property : this.configuration.getProperties()) {

                this.configuration.getProperties().remove(property);
                return "Removed: " + property.getKey() + " = " + property.getValue();
            }
        }

        return "No such property found";
    }

        String result = addProperty("testKey", "testValue");

        assertEquals(result, "Updated: testKey = newValue");

    public void testRemoveProperty() {
```

```java
import org.junit.jupiter.api.BeforeEach;
import static org.junit.jupiter.api.Assertions.assertEquals;

public class TestConfigurationData {
    private Configuration configuration;

    @BeforeEach
    void setup() {
        this.configuration = new Configuration();
    }

    public String addProperty(String key, Object value) {
        // Create a new property with the given key and value
        Property property = new Property(key, value);
        
        // Add the property to the configuration
        this.configuration.getProperties().add(property);

        return "Added: " + property.getKey() + " = " + property.getValue();
    }

    public String updateProperty(String key, Object newValue) {
        // Update an existing property in the configuration
        for (Property property : this.configuration.getProperties()) {
            if (property.getKey().equals(key)) {
                property.setValue(newValue);
                return "Updated: " + property.getKey() + " = " + property.getValue();
            }
        }

        return "No such property found";
    }

    public String removeProperty(String key) {
        // Remove an existing property from the configuration
        for (int i = 0; i < this.configuration.getProperties().size(); i++) {
            if (this.configuration.getProperties().get(i).getKey().equals(key)) {
                this.configuration.getProperties().remove(i);
                return "Removed: " + this.configuration.getProperties().get(i).getKey() + " = " + this.configuration.getProperties().get(i).getValue();
            }
        }

        return "No such property found";
    }

    public void testAddProperty() {
        String result = addProperty("testKey", "testValue");
        assertEquals(result, "Added: testKey = testValue");
    }

    public void testUpdateProperty() {
        String result = updateProperty("testKey", "newValue");
        assertEquals(result, "Updated: testKey = newValue");
    }

    public void testRemoveProperty() {
        String result = removeProperty("testKey");
        assertEquals(result, "Removed: testKey = testValue");
    }
}

class Property {
    private String key;
    private Object value;

    public Property(String key, Object value) {
        this.key = key;
        this.value = value;
    }

    public String getKey() {
        return key;
    }

    public void setValue(Object value) {
        this.value = value;
    }

    public Object getValue() {
        return value;
    }
}

class Configuration {
    private List<Property> properties;

    public Configuration() {
        this.properties = new ArrayList<>();
    }

    public List<Property> getProperties() {
        return properties;
    }
}
```

In the `updateProperty` method, I changed the loop condition to `i < this.configuration.getProperties().size()` so that all properties are removed when a property with the same key is updated. 

In the `removeProperty` method, I changed the for-each loop condition to `i < this.configuration.getProperties().size()` and also added a check to see if an index can be used in Java which would not be possible in C++.
