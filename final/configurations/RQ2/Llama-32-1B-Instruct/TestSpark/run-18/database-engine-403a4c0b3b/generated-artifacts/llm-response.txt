using System;
using System.Data;

class Database {
    public void CreateTable() {
        string sql = "CREATE TABLE IF NOT EXISTS Table1 (Id INT PRIMARY KEY, Name VARCHAR(255))";
        DataTable dt = new DataTable();
        dt.Columns.Add("Id", typeof(int));
        dt.Columns.Add("Name", typeof(string));
        foreach (DataRow row in dt.Rows) {
            row["Id"] = 0;
            row["Name"] = "";
        }
        sql += "\nINSERT INTO Table1 (Id, Name) VALUES (0, 'John Doe');";
        Console.WriteLine(sql);
    }

    public void SelectTable() {
        string sql = "SELECT * FROM Table1";
        DataTable dt = new DataTable();
        foreach (DataRow row in dt.Rows) {
            Console.Write(row["Id"].ToString() + " " + row["Name"].ToString());
        }
        Console.WriteLine();
    }

    public void DeleteTable() {
        string sql = "DROP TABLE IF EXISTS Table1";
        foreach (DataRow row in Console.ReadLine().Split(' ')) {
            sql += "\nINSERT INTO Table1 (Id, Name) VALUES (" + row.Split(", ", 2)[0].Trim() + ", '" + row.Split(", ", 2)[1].Trim() + ");\n";
        }
    }

    public void InsertRow() {
        string id = "1";
        string name = "John Doe";
        string sql = "INSERT INTO Table1 (Id, Name) VALUES (" + id + ", '" + name + "');";
        Console.WriteLine(sql);
    }

    public void SelectAll() {
        string sql = "SELECT * FROM Table1";
        foreach (DataRow row in Console.ReadLine().Split(' ')) {
            Console.Write(row["Id"].ToString() + " " + row["Name"].ToString());
        }
        Console.WriteLine();
    }

    public void DeleteMultipleRows() {
        string[] rows = Console.ReadLine().Split(' ');
        DataTable dt = new DataTable();
        foreach (string row in rows) {
            DataRow newRow = dt.NewRow();
            newRow["Id"] = 0;
            newRow["Name"] = "";
            dt.Rows.Add(newRow);
        }
        foreach (DataRow row in dt.Rows) {
            Console.WriteLine(row["Id"].ToString() + " " + row["Name"].ToString());
        }
    }

    public void InsertMultipleRows() {
        string[] rows = Console.ReadLine().Split(' ');
        DataTable dt = new DataTable();
        for (int i = 0; i < rows.Length; i++) {
            DataRow newRow = dt.NewRow();
            newRow["Id"] = Convert.ToInt32(rows[i]);
            newRow["Name"] = rows[i + 1];
            dt.Rows.Add(newRow);
        }
    }

    static void Main(string[] args) {
        Database db = new Database();
        Console.WriteLine("Create Table:");
        db.CreateTable();
        Console.WriteLine("\nSelect Table:");
        db.SelectTable();

        Console.WriteLine("\nDelete Table:");
        db.DeleteTable();

        Console.WriteLine("\nInsert Row:");
        db.InsertRow();

        Console.WriteLine("\nSelect All:");
        db.selectAll();

        Console.WriteLine("\nDelete Multiple Rows:");
        db.DeleteMultipleRows();
    }

using System;
using System.Data;

class BankAccount {
    public int Id { get; set; }
    public string AccountNumber { get; set; }
    public decimal Balance { get; set; }

    public BankAccount() {}

    public void CreateAccount(int id, string accountNumber, decimal balance) {
        this.Id = id;
        this.AccountNumber = accountNumber;
        this.Balance = balance;
    }

    public void DepositMoney(decimal amount) {
        if (amount > 0) {
            Balance += amount;
            Console.WriteLine("Deposit successful!");
        } else {
            Console.WriteLine("Invalid deposit amount.");
        }
    }

    public void WithdrawMoney(decimal amount) {
        if (amount > 0 && Amount() > 0) {
            Balance -= amount;
            Console.WriteLine("Withdrawal successful!");
        } else {
            Console.WriteLine("Invalid withdrawal amount or insufficient balance.");
        }
    }

    public decimal GetBalance() { return Balance; }

    private decimal Amount() => Balance;

    public override string ToString() => $"Account ID: {Id}, Account Number: {AccountNumber}, Balance: {Balance}";
}

class Program {
    static void Main(string[] args) {
        BankAccount account = new BankAccount();

        Console.WriteLine("Create New Account:");
        int id = Convert.ToInt32(Console.ReadLine());
        string accountNumber = Console.ReadLine();
        decimal balance = Convert.ToDecimal(Console.ReadLine());

        account.CreateAccount(id, accountNumber, balance);

                    return;
                default:

            }
        }
