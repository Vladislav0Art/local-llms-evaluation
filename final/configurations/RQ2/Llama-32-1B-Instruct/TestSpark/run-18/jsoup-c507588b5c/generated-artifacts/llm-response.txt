public class TestNode {
    public static String testToString(TextNode node) {
        return node.text();
    }

    public static boolean isBlank(TextNode node) {
        return node.isBlank();
    }

    public static void main(String[] args) {
        // Create a Document object
        Document document = new Document();

        // Create TextNodes for each character in the string "Hello World!"
        Elements elements = document.select("h1, p");
        for (Element element : elements) {
            String text = element.text();
            System.out.println(testToString(element));
            System.out.println(isBlank(TextNode(text)));
        }

        // Test toString()
        String expected = "Hello World!";
        Document actual = new Document();
        actual.appendChild(document.createText("Hello", "h1"));
        actual.appendChild(document.createText("World!", "p"));

        assertEquals(expected, testToString(actual));
    }

            System.out.println("No " + expectedTag + " element found.");
