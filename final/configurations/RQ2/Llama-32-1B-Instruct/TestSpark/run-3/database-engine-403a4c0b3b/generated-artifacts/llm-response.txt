import java.sql.*;
import java.util.ArrayList;

public class DatabaseTest {

    public ArrayList<SQLTerm> selectFromTable(String dbUrl, String dbName, String query) {
        Connection connection = null;
        ResultSet resultSet = null;
        ArrayList<SQLTerm> result = new ArrayList<>();

        try {
            // Connect to the database
            Class.forName("com.mysql.cj.jdbc.Driver");
            connection = DriverManager.getConnection(dbUrl);

            // Create a statement object
            Statement statement = connection.createStatement();

            // Execute the query
            ResultSet resultSet = statement.executeQuery(query);
            while (resultSet.next()) {
                // Get the table name and column names from the result set
                String tableName = resultSet.getMetaData().getTableName(1);
                String[] columnNameList = resultSet.getString(2).split(",");

                // Create a new SQLTerm object with the table name, column names, and row data
                SQLTerm sqlTerm = new SQLTerm(tableName, columnNameList, resultSet.getString(3));

                result.add(sqlTerm);
            }
        } catch (SQLException e) {
            System.out.println("Error: " + e.getMessage());
        }

        return result;
    }

    public void testSelectFromTable() {
        String dbUrl = "jdbc:mysql://localhost:3306/test";

import java.sql.*;
import java.util.ArrayList;

public class DatabaseTest {

    public ArrayList<SQLTerm> selectFromTable(String dbUrl, String dbName, String query) {
        Connection connection = null;
        PreparedStatement statement = null;
        ResultSet resultSet = null;
        ArrayList<SQLTerm> result = new ArrayList<>();

        try {
            // Connect to the database
            Class.forName("com.mysql.cj.jdbc.Driver");
            connection = DriverManager.getConnection(dbUrl);

            // Create a statement object for the query
            statement = connection.prepareStatement(query);
            resultSet = statement.executeQuery();

            // Add rows from the result set to the array of SQLTerms
            while (resultSet.next()) {
                ArrayList<SQLTerm> row = new ArrayList<>();
                for (int i = 1; i <= resultSet.getMetaData().getColumnCount(); i++) {
                    row.add(new SQLTerm(resultSet.getString(i), resultSet.getObject(i)));
                }
                result.add(row);
            }
        } catch (SQLException e) {
            System.out.println("Error: " + e.getMessage());
        }

        return result;
    }

    public ArrayList<SQLTerm> selectFromTable(String dbUrl, String dbName, String query1, String tableName1, String columnName1, String query2, String tableName2, String columnName2) {
        Connection connection = null;
        PreparedStatement statement1 = null;
        PreparedStatement statement2 = null;
        ResultSet resultSet1 = null;
        ResultSet resultSet2 = null;

        try {
            // Connect to the database
            Class.forName("com.mysql.cj.jdbc.Driver");
            connection = DriverManager.getConnection(dbUrl);

            // Create a statement object for query 1
            statement1 = connection.prepareStatement(query1);
            resultSet1 = statement1.executeQuery();

            // Get the table name and column names from the result set for query 1
            String tableName1 = resultSet1.getMetaData().getTableName(1);
            String[] columnNameList1 = resultSet1.getString(2).split(",");

            // Create a new SQLTerm object with the table name, column names, and row data for query 1
            ArrayList<SQLTerm> sqlTerms1 = new ArrayList<>();
            for (int i = 0; i < columnNameList1.length; i++) {
                sqlTerms1.add(new SQLTerm(tableName1, columnNameList1[i], resultSet1.getString(i)));
            }

            // Create a statement object for query 2
            statement2 = connection.prepareStatement(query2);
            resultSet2 = statement2.executeQuery();

            // Get the table name and column names from the result set for query 2
            String tableName2 = resultSet2.getMetaData().getTableName(1);
            String[] columnNameList2 = resultSet2.getString(2).split(",");

            // Create a new SQLTerm object with the table name, column names, and row data for query 2
            ArrayList<SQLTerm> sqlTerms2 = new ArrayList<>();
            for (int i = 0; i < columnNameList2.length; i++) {
                sqlTerms2.add(new SQLTerm(tableName2, columnNameList2[i], resultSet2.getString(i)));
            }

            // Add rows from the result sets of query 1 and query 2 to the array of SQLTerms
            for (int i = 0; i < sqlTerms1.size(); i++) {
                ArrayList<SQLTerm> row = new ArrayList<>();
                for (int j = 0; j < sqlTerms2.size(); j++) {
                    if (i == j) {
                        row.add(sqlTerms1.get(i));
                    } else {
                        row.add(sqlTerms2.get(j));
                    }
                }
                result.add(row);
            }
        } catch (SQLException e) {
            System.out.println("Error: " + e.getMessage());
        }

        return result;
    }
