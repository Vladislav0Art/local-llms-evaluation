public class ProtocolTimestampRequest {
    private long timestamp;

    public void addTimestamp(long timestamp) {
        this.timestamp = timestamp;
    }

    @Override
    public String toString() {
        return "ProtocolTimestampRequest{" +
                "timestamp=" + timestamp +
                '}';
    }

    public String getTimestampAsString() {
        return timestamp == 0 ? "No timestamp available" : "Timestamp: " + timestamp;
    }
}

class PublicClass {
    public static void main(String[] args) {
        ProtocolTimestampRequest request = new ProtocolTimestampRequest();
        
        // Add a timestamp
        request.addTimestamp(1000000L);
        
        // Get the timestamp as a string
        System.out.println("Timestamp (as number): " + request);
        System.out.println("Timestamp (as string): " + request.getTimestampAsString());
    }
}
