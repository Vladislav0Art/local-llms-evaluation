    public int testGetResponse(int numIterations) {
        for (int i = 0; i < numIterations; i++) {
            String response = server.getRespond("Test");
            assertEquals("Test", response);
        }
        return numIterations;
    }

    public void testGetRequest(String request, int numRequests) {
        for (int i = 0; i < numRequests; i++) {
            String response = server.get(req);
            assertEquals(request, response);
        }
    }

    private class Server {
        @Override
        public String sayHello(String name) {
            return "Hello, " + name;
        }

        @Override
        public int getRespond(String request) {
            if (request.equals("Test")) {
                return 1000;
            } else {
                return -1;
            }
        }
    }
}
