import java.util.*;

public class Grammar {
    private Map<String, List<Production>> rules;

    public Grammar() {
        this.rules = new HashMap<>();
        this.productionRules = new HashSet<>();
    }

    public void addRule(String input, String output) {
        if (rules.containsKey(input)) {
            System.out.println("Input '" + input + "' already has a rule.");
            return;
        }
        rules.put(input, new ArrayList<>());
        productionRules.add(input);
    }

    public boolean match(String input) {
        for (List<Production> productions : rules.values()) {
            for (Production production : productions) {
                if (production.matches(input)) {
                    return true;
                }
            }
        }
        return false;
    }

    public void printProductionRules() {
        System.out.println("Production Rules:");
        for (Map.Entry<String, List<Production>> entry : rules.entrySet()) {
            System.out.println(entry.getKey() + ": " + entry.getValue());
        }
    }
}

class Production {
    private String input;
    private String output;

    public Production(String input, String output) {
        this.input = input;
        this.output = output;
    }

    @Override
    public boolean equals(Object obj) {
        if (!(obj instanceof Production)) {
            return false;
        }
        Production other = (Production) obj;
        return this.input.equals(other.input) && this.output.equals(other.output);
    }

    @Override
    public int hashCode() {
        return input.hashCode() + output.hashCode();
    }

    @Override
    public String toString() {
        return "Input: " + input + ", Output: " + output;
    }
}
