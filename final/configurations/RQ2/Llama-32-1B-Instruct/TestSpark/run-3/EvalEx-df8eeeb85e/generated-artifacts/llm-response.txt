import org.junit.Before;
import org.junit.Test;

public class TokenizerTest {

    private Expression parseExpression(String expression) {
        return parse(expression);
    }

    @Before
    public void setup() {
        this.expressionString = "1 + 2 * (3 - 4)";
    }

    @Test
    public void testParseEmptyExpression() {
        when(parseExpression(this.expressionString)).thenAnswer(a -> {
            List<Token> result = new ArrayList<>();
            result.add(new Token(Tokenizer.DOT, null));
            result.add(new Token(Tokenizer.OPEN_PAREN, null));
            result.add(new Token(Tokenizer.EXpression, this.expressionString));
            result.add(new Token(Tokenizer.CLOSE_PAREN, null));
            return result;
        });
    }

    @Test
    public void testParseEmptyExpressionNoTokenType() {
        when(parseExpression(this.expressionString)).thenAnswer(a -> {
            List<Token> result = new ArrayList<>();
            result.add(new Token(Tokenizer.DOT, null));
            result.add(new Token(Tokenizer.OPEN_PAREN, null));
            result.add(new Token(Tokenizer.EXpression, this.expressionString));
            result.add(new Token(Tokenizer.CLOSE_PAREN, null));
            return result;
        });
