import java.io.BufferedReader;
import java.io.FileReader;
import java.io.IOException;

public class ConfigMe {

    private Map<String, Object> rootEntries;

    public ConfigMe() {
        rootEntries = new HashMap<>();
    }

    // Method to add properties to the root entries
    public void testAddProperty() {
        try (BufferedReader reader = new BufferedReader(new FileReader("config.properties"))) {
            String line;
            while ((line = reader.readLine()) != null) {
                if (!line.equals("#")) {  // Ignore comments
                    String[] parts = line.split("\\s+");
                    String key = parts[0];
                    int value = Integer.parseInt(parts[1]);
                    rootEntries.put(key, value);
                }
            }
        } catch (IOException e) {
            System.out.println("Error reading config file");
        }
    }

    // Method to get the list of properties from the configuration
    public List<Property> getRootEntries() {
        return new ArrayList<>(rootEntries.values());
    }

    public static void main(String[] args) {
        ConfigMe config = new ConfigMe();
        config.testAddProperty();

        PropertyList propertyList = build(config.getRootEntries());

        System.out.println("Properties: " + propertyList);
        for (Property prop : propertyList) {
            System.out.println(prop.getKey() + ": " + prop.getValue());
        }
    }

    // Method to convert a list of properties into a new configuration
    public static PropertyList build(List<Property> properties) {
        return new PropertyList(properties);
    }
}

class ConfigParser {

    private Map<String, Object> configMap;

    public ConfigParser(Map<String, Object> configMap) {
        this.configMap = configMap;
    }

    // Method to parse the configuration file
    public void parse() {
        for (String key : configMap.keySet()) {
            String[] parts = getParts(key);
            if (parts.length == 2) {  // If it's a key-value pair
                try {
                    int value = Integer.parseInt(parts[1]);
                    configMap.put(key, value);
                } catch (NumberFormatException e) {
                    System.out.println("Invalid value for " + key);
                }
            } else if (parts.length == 3) {  // If it's a key-value string
                try {
                    String[] parts2 = getParts(parts[0]);
                    configMap.put(key, parts2[1].equals("") ? null : Integer.parseInt(parts2[1]));
                } catch (Exception e) {
                    System.out.println("Invalid value for " + key);
                }
            }
        }
    }

    // Helper method to split the string into parts
    private String[] getParts(String str) {
        return str.split("\\s+");
    }
