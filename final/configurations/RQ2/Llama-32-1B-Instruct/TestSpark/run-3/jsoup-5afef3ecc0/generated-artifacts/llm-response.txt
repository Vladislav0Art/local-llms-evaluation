import java.net.URI;
import java.util.Arrays;

public class UrlBuilder {

    public static String build(String url) {
        URI uri = URI.create(url);
        return uri.toString();
    }

    public static String buildString(String str, int length) {
        StringBuilder sb = new StringBuilder(str);
        while (sb.length() < length) {
            sb.append(' ');
        }
        return sb.toString();
    }

    public static boolean isUrl(String s) {
        try {
            URI uri = URI.create(s);
            return true;
        } catch (Exception e) {
            return false;
        }
    }

    public static String encodeURL(String url) {
        return buildString(url, url.length() * 2);
    }

    public static boolean isEncodedUrl(String s) {
        try {
            URI uri = new URI(s);
            return true;
        } catch (Exception e) {
            return false;
        }
    }
}
