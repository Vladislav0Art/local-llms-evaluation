import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class DocumentTest {

    private static final String BASE_URI = "https://example.com";
    private static final String BASE_PATH = "/html";

    public static void main(String[] args) {
        Document doc1 = getDocument(BASE_URI, BASE_PATH);
        System.out.println("doc1: " + getDocumentTitle(doc1));

        Element element1 = doc1.selectFirst("#header");
        if (element1 != null) {
            System.out.println("Element with id 'header': " + element1);
        }

        Document doc2 = getDocument(BASE_URI, BASE_PATH);
        System.out.println("doc2: " + getDocumentTitle(doc2));

        Element element2 = doc2.selectFirst("#footer");
        if (element2 != null) {
            System.out.println("Element with id 'footer': " + element2);
        }

        Document doc3 = getDocument(BASE_URI, BASE_PATH);
        System.out.println("doc3: " + getDocumentTitle(doc3));

        Element element3 = doc3.selectFirst("#content");
        if (element3 != null) {
            System.out.println("Element with id 'content': " + element3);
        }

        Document doc4 = getDocument(BASE_URI, BASE_PATH);
        System.out.println("doc4: " + getDocumentTitle(doc4));

        Element element4 = doc4.selectFirst("#header");
        if (element4 != null) {
            System.out.println("Element with id 'header': " + element4);
        }

        String documentTitle = getDocumentTitle(doc1);
        System.out.println("Document title: " + documentTitle);

        assertTrue(document1.selectFirst("#header").exists());
        assertFalse(element3 == null);
    }

    public static Document getDocument(String baseUri, String basePath) {
        try {
            return DocumentBuilderFactory.newInstance().newDocumentBuilder()
                    .parse(new java.io.BufferedReader(new java.io.InputStreamReader(getBaseuriAsStream(baseUri)), java.nio.charset.StandardCharsets.UTF_8));
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }

    public static String getDocumentTitle(Document doc) {
        Elements headings = doc.selectFirst("h1, h2, h3, h4, h5, h6");
        if (headings != null) {
            return headings.get(0).text();
        } else {
            return "";
        }
    }

    public static String getBaseuri() {
        // Replace with actual base URL
        return "https://example.com";
    }

    public static String getBasepath() {
        // Replace with actual base path
        return "/html";
    }

    public static void main(String[] args) throws Exception {
        System.out.println("Running tests...");
    }
}
