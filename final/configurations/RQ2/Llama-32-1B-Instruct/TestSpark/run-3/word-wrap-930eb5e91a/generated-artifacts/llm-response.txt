        Preconditions.checkArgument(s != null);

        }
        return true;

    }

    public static String toStringWithLineBreak(Builder builder) {
        StringBuilder builder = new StringBuilder();
        for (char c : builder.extraWordChars) {
            builder.append(c).append("\n");

            builder.append(builder.newLine);
        }
        return builder.toString();
    }

    public static String toStringWithoutLineBreak(Builder builder) {
        StringBuilder builder = new StringBuilder();
        for (char c : builder.extraWordChars) {
            builder.append(c);

            builder.append("\n");
        }
        return builder.toString();
    }

        while (builder.length() > 0 && builder.charAt(builder.length() - 1) == ' ') {

        }
        if (!insertHyphens || !breakWords) {
            return builder.toString();
        }

        List<String> words = new ArrayList<>();
        while (builder.length() > 0 && builder.charAt(builder.length() - 1) != ' ') {
            words.add(builder.deleteCharAt(builder.length() - 1).toString());
            if (!insertHyphens || !breakWords) {
                break;
            }

        return String.join(" ", words);
    }

    public static boolean needsLineBreak(Builder builder, int maxWordLength) {

        for (char c : builder.extraWordChars) {
            totalChars += c;

        if (totalChars > maxWordLength * 2 + 1) {
            return true;
        }

        totalChars = 0;
        for (int i = 0; i < builder.stringWidth(); i++) {
            int charCount = 0;
            boolean lastWasSpace = false;
            while (i <= builder.maxWidth() && !lastWasSpace) {
                if (builder.extraWordChars[i] != ' ') {
                    lastWasSpace = true;
                    charCount++;
                } else {
                    lastWasSpace = false;
                    charCount--;
                }
                i++;
            }

            if (charCount > maxWordLength) {
                return true;
            }
        }

        return false;
    }

    public static String trimWhitespace(Builder builder) {
        while (builder.stringWidth() > 0 && !builder.extraWordChars[builder.maxWidth()] == ' ') {
            builder.removeMaxWidth();
        }

        if (builder.newLine != null && !builder.newLine.equals("")) {
            return builder.newLine;
        }

        return builder.toString();
    }
