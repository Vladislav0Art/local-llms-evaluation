import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Main {

    public static void main(String[] args) {
        // Create a new document object
        Document doc = Jsoup.create("html", "body");

        // Add some text to the page
        Element element1 = doc.createElement("p");
        element1.text = "This is a paragraph of text.";
        doc.appendChild(element1);

        // Add another paragraph of text
        Element element2 = doc.createElement("p");
        element2.text = "This is another paragraph of text.";
        doc.appendChild(element2);

        // Print the HTML content of the document
        System.out.println(doc.outerHtml());

        // Find all paragraphs on the page
        Element[] paragraphs = doc.select("p");

        for (Element paragraph : paragraphs) {
            System.out.println(paragraph.text());
        }

        // Test if a paragraph is blank or not
        boolean isBlankParagraph = testIsBlank(paragraph);
        System.out.println(isBlankParagraph);

        // Split the text of a paragraph into substrings separated by one space
        String[] splitText1 = testTextSplitText1(paragraph);
        System.out.println(splitText1[0]);

        // Join all paragraphs on the page into a single string separated by commas
        String joinText = testTextJoinText(paragraphs);
        System.out.println(joinText);

        // Find all text elements that are not blank or null
        Node[] nodes = doc.select("p, span");
        for (Node node : nodes) {
            if (!node.isBlank() && !node.isNull()) {
                System.out.println(node.text());
            }
        }

        testNodeToString();
    }

    public static boolean testIsBlank(Node node) {
        return node == null || node.trim().isEmpty();
    }

    public static String[] testTextSplitText1(Node node) {
        String text = node.toString();
        if (!text.isEmpty()) {
            text = text.substring(0, 2); // Remove the first two characters
        }
        return new String[]{text};
    }

    public static String testTextJoinText() {
        Node[] nodes = {doc.select("p").first(), doc.select("span").first()};
        StringBuilder builder = new StringBuilder();
        for (Node node : nodes) {
            if (!node.isBlank()) {
                builder.append(node.text()).append(" ");
            }
        }
        return builder.toString().trim();
    }

    public static String testNodeToString() {
        Node textNode = doc.createTextNode("Hello World");
        System.out.println(textNode.toString());
        return textNode.toString();
    }
}
