public class TextNodeTest {

    private static final String NORMALIZEDWhitespace = "  Hello, World! ";
    private static final String STRIPPED_LEADING_WHITESPACE = "   Hello, World!   ";

    public void testTextToNode() {
        TextNode textNode = new TextNode("Hello, World!");
        assertEquals("Hello, World!", textNode.text());
        assertNotNull(textNode.nodeName());
    }

    public void testBlankNodes() {
        TextNode textNode = new TextNode("");
        assertEquals("", textNode.text());
        assertNotNull(textNode nodeName());
    }

    public void testTextToNodeTrimmingWhitespace() {
        TextNode textNode = new TextNode("Hello  World!");
        String result = textNode.toString();
        assertEquals("Hello World", result);
        assertNotNull(textNode nodeName());
    }

    public void testClone() {
        TextNode clone = textNode.clone();
        assertNotNull(clone);
        assertEquals(textNode, clone);
    }

    public void testTextToNodeSubstringSelection() {
        TextNode textNode = new TextNode("Hello, World! 123");
        String result1 = textNode.substring(0, 5);
        String result2 = textNode.substring(10, 15);
        assertEquals("Hello", result1);
        assertEquals("", result2);
    }

    public void testNodeToString() {
        String[] expectedStrings = {"Hello, World!", "   Hello, World!   ", "", ""};
        for (String str : expectedStrings) {
            TextNode textNode = new TextNode(str);
            assertEquals(expectedStrings[0], str, "");
        }
    }

    public void testTextNodeOutputSettings() throws IOException {
        Document document = DocumentBuilderFactory.newInstance().newDocumentBuilder().newDocument();
        Appendable output = document.open();

        TextNode textNode = new TextNode("Hello, World!");
        textNode.outerHtmlHead(output, 0, null);
        assertEquals(5, output.length());

        String result1 = "";
        textNode.outerHtmlTail(output, 4, Document.OutputSettings.INDENT);
        assertEquals(3, output.length());
        assertEquals("   ", output.toString());

        textNode.outerHtmlHead(output, 2, null);

        textNode.splitText(10);
        assertEquals(7, output.length());
    }

    public void testNormaliseWhitespace() {
        TextNode textNode = new TextNode(NORMALIZEDWhitespace);
        assertEquals(NORMALIZED_whitespace, textNode.text());
        assertNotNull(textNode nodeName());
    }

    public void testStripLeadingWhitespace() {
        String[] expectedStrings = {"Hello, World!", "  Hello, World!  ", "", ""};
        for (String str : expectedStrings) {
            TextNode textNode = new TextNode(STRIPPED_LEADING_WHITESPACE);
            assertEquals(str, textNode.text());
        }
    }

    public void testLastCharIsWhitespace() {
        StringBuilder sb = new StringBuilder();
        assertTrue(TextNode.lastCharIsWhitespace(sb));
    }
}

public class TestString {
    private String str;

    @Test
    public void testGetLength() {
        str = "Hello, World!";
        assertEquals(13, str.length());
    }

    @Test
    public void testNormalizeWhitespace() {
        char[] chars = {" ", "\t", "\r"};
        StringBuilder sb = new StringBuilder();
        for (char c : chars) {
            sb.append(c);
        }
        String normalized = sb.toString().trim();
        assertEquals("Hello World", normalized);
    }

    @Test
    public void testGetLastCharIsWhitespace() {
        char[] chars = {'a', 'b'};
        StringBuilder sb = new StringBuilder();
        for (char c : chars) {
            sb.append(c);
        }
        String str = sb.toString();
        assertTrue(str.startsWith("abc"));
    }

    @Test
    public void testGetNodeName() {
        String str = "Hello, World!";
        assertEquals("Hello", str.getNodeName());
    }

    @Test
    public void testGetNormalizedWhitespace() {
        String str = "  Hello World! ";
        assertEquals("Hello World", str.normalizeWhitespace());
    }
}
