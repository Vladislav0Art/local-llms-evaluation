```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class HttpConnectionTests {

    private HttpConnection httpConnection;

    @Before
    public void setup() {
        httpConnection = new HttpConnection();
    }

    public static String jsoup(String url) throws MalformedURLException, URISyntaxException {
        Document doc = Jsoup.connect(url).get();
        return doc.text();
    }

    @Test
    public void testGetResponseWithCharset() throws IOException {
        String response = jsoup("https://www.example.com");
        assertEquals(200, response.code());
        assertEquals("Hello World", response.body().trim());
        String charset = "UTF-8";
        response.postDataCharset(charset);
        assertEquals(200, response.code());
        assertEquals("Hello World", response.body().trim());
    }

    @Test
    public void testGetResponseWithHeader() throws IOException {
        String response = jsoup("https://www.example.com");
        assertEquals(200, response.code());
        Element headerElement = response.headers();
        String headerName = headerElement.attr("Content-Type").split(";")[0];
        String headerValue = headerElement.attr(headerName);
        assertEquals("text/html; charset=utf-8", headerValue);
    }

    @Test
    public void testGetResponseWithMultipleHeaders() throws IOException {
        String response = jsoup("https://www.example.com");
        assertEquals(200, response.code());
        Element headerElement1 = response.headers();
        String headerName1 = headerElement1.attr("Content-Type").split(";")[0];
        String headerValue1 = headerElement1.attr(headerName1);
        Element headerElement2 = response.headers().header("X-Angular-Header");
        String headerName2 = headerElement2.attr("value");
        String headerValue2 = headerElement2.attr("value");

        assertEquals(headerName1, headerName2);
        assertEquals(headerValue1, headerValue2);
    }

}
```

```java
public class Main {
    public static int statusCode(String body) {
        if (body.equals("Hello World")) {
            return 200;
        } else if (body.equals("Not Found")) {
            return 404;
        } else if (body.equals("OK")) {
            return 200;
        } else {
            return 500;
        }
    }

    public static void main(String[] args) {
        assertEquals(200, Main.statusCode("Hello World"));
        assertEquals(404, Main.statusCode("Not Found"));
        assertEquals(200, Main.statusCode("OK"));
        assertEquals(500, Main.statusCode("Internal Server Error"));
    }
}
```
