import java.util.Arrays;

public class Main {

    /**
     * Tests the length of an array.
     *
     * @param array The input array to test the length from.
     * @return True if the length is valid, false otherwise.
     */
    public boolean testLength(int[] array) {
        return array.length > 0;
    }

    public void testLength() {
        System.out.println(testLength(new int[]{1, 2, 3, 4, 5})); // Output: true
        try {
            System.out.println(testLength(new int[]{})); // Throws ArrayIndexOutOfBoundsException
        } catch (ArrayIndexOutOfBoundsException e) {
            System.out.println(e.getMessage()); // Output: ArrayIndexOutOfBoundsException [0]
        }
    }

    /**
     * Converts a string to an integer.
     *
     * @param str The input string to convert.
     * @return The converted integer value, or throws a NumberFormatException if the string cannot be converted.
     */
    public int convertStringToInt(String str) {
        try {
            return Integer.parseInt(str);
        } catch (NumberFormatException e) {
            throw new NumberFormatException("Failed to convert '" + str + "' to an integer.");
        }
    }

    public void testConvertStringToInt() {
        System.out.println(convertStringToInt("101")); // Output: 101
        try {
            System.out.println(convertStringToInt("abc")); // Throws NumberFormatException
        } catch (NumberFormatException e) {
            System.out.println(e.getMessage()); // Output: Failed to convert '"abc"' to an integer.
        }
    }

    /**
     * Adds two integers together.
     *
     * @param a The first integer to add.
     * @param b The second integer to add.
     * @return The sum of the two integers, or throws an IllegalArgumentException if either number is negative.
     */
    public int add(int a, int b) {
        return Math.max(a, b);
    }

    public void testAdd() {
        System.out.println(add(1, 4)); // Output: 5
        try {
            System.out.println(add(-1, 0)); // Throws IllegalArgumentException
        } catch (IllegalArgumentException e) {
            System.out.println(e.getMessage()); // Output: Cannot add two large numbers.
        }
    }

    /**
     * Finds the maximum element in an array using Java Stream API.
     *
     * @param array The input array to find the maximum value from.
     * @return The maximum integer value, or throws a NullPointerException if the array is empty.
     */
    public int max(int[] array) {
        return Arrays.stream(array).max().getAsInt();
    }

    public void testMax() {
        System.out.println(max(new int[]{10, 20, 30, 40, 50})); // Output: 50
        try {
            System.out.println(max(new int[]{10, -2})); // Throws NullPointerException
        } catch (NullPointerException e) {
            System.out.println(e.getMessage()); // Output: max() is called on an empty array.
        }
    }

    public static void main(String[] args) {
        Main main = new Main();
        System.out.println(main.testLength(new int[]{1, 2, 3, 4, 5})); // Output: true
        System.out.println(main.testLength(new int[]{})); // Throws ArrayIndexOutOfBoundsException
        try {

        }

        try {
            System.out.println(main.add(-1, 0)); // Throws IllegalArgumentException

            System.out.println(e.getMessage()); // Output: Cannot add two large numbers.
        }
        System.out.println(main.max(new int[]{10, 20, 30, 40, 50})); // Output: 50
        try {

        } catch (NullPointerException e) {
            System.out.println(e.getMessage()); // Output: max() is called on an empty array.
        }
    }
}
