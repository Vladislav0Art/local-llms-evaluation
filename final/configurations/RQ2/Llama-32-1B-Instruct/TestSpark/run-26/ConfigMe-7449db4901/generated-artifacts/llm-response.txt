        if (!directory.isAbsolute()) {
            System.out.println("Path is not absolute.");
            return;
        }
        File directoryFile = new File(directory.toURI());
        if (directoryFile.isDirectory() && !extension.equals(null)) {
            try (Scanner scanner = new Scanner(directoryFile)) {
                while (scanner.hasNextLine()) {
                    String line = scanner.nextLine();
                    System.out.println(line);
                }
            } catch (Exception e) {
                System.out.println(e.getMessage());
            }
        } else if (!directoryFile.exists()) {
            System.out.println(directoryFile.getAbsolutePath() + " does not exist or is a directory.");
