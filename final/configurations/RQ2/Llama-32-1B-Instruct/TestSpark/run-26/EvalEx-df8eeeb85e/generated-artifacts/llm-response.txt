```java
import org.junit.Test;
import static org.junit.Assert.assertEquals;

public class TokenizerTest {

    private Tokenizer tokenizer;

    @Test
    public void testGetToken() {
        tokenizer = new Tokenizer();
        String input = "1 + 2";
        String expectedOutput = "Number, +, Number, 2, Number";
        assertEquals(expectedOutput, tokenizer.nextToken(input));
    }

    @Test
    public void testGetTokenWithNoOperation() {
        tokenizer = new Tokenizer();
        String input = "";
        String expectedOutput = "Empty string";
        assertEquals(expectedOutput, tokenizer.nextToken(input));
    }

    @Test
    public void testGetNextToken() {
        tokenizer = new Tokenizer();
        String input = "1 + 2";
        String[] output = tokenizer.nextToken(input);
        assertEquals("Number", output[0]);
        assertEquals("+", output[1]);
        assertEquals("Number", output[2]);
        assertEquals(2, output[3].getLength());
    }

    @Test
    public void testGetNextTokenWithNoOperation() {
        tokenizer = new Tokenizer();
        String input = "1 + 2";
        tokenizer.nextToken(input);
        assertEquals("", input);
    }

    @Test
    public void testGetNumberTokens() {
        tokenizer = new Tokenizer();
        String input = "1 + 2";
        String[] output = tokenizer.nextToken(input);
        assertEquals("Number", output[0]);
        assertEquals(1, output[1].getLength());
        assertEquals("+", output[2]);
        assertEquals("Number", output[3]);
    }

    @Test
    public void testGetNumberTokensWithNoOperation() {
        tokenizer = new Tokenizer();
        String input = "1 + 2";
        tokenizer.nextToken(input);
        assertEquals("", input);
    }
}
```

```java
    public void testGetNextTokenWithNoOperation() {

        assertEquals("", input);
    }

    public void testGetLengthOfToken() {
        String input = "1 + 2";
        int[] output = Tokenizer.nextToken(input).length();
        assertEquals(4, output[0]);
    }

    public void testGetNextTokenWithEmptyInput() {

        }
    }
```
