class Row {
    private int[] data;

    public Row(int[] data) {
        this.data = data;
    }

    public int getID() {
        return data[0];
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        for (int i = 1; i < data.length; i++) {
            sb.append(data[i]).append(",");
        }
        return sb.toString().substring(0, sb.length() - 1);
    }
}

public class CSVWriter {
    private List<String[]> entries;

    public void setEntries(List<String[]> entries) {
        this.entries = entries;
    }

    public static List<String> getCSV(String csv) {
        String[] lines = csv.split("\n");
        List<String> result = new ArrayList<>();
        for (String line : lines) {
            if (!line.isEmpty()) {
                result.add(line);
            }
        }
        return result;
    }

    public static void printCSV(List<String[]> entries, boolean header) {
        if (header) {
            System.out.println("ID\tKey\tValue");
        }
        for (String[] entry : entries) {
            System.out.print(entry[0] + "\t" + entry[1] + "\t" + entry[2]);
            System.out.println();
        }
    }

    public static void main(String[] args) throws IOException, CsvValidationException {
        Validator validator = new Validator();
        CSVReader csvReader = new CSVReader();
        CSVWriter csvWriter = new CSVWriter();

class Row {
    private Student student;

    public Row(Student student) {
        this.student = student;
    }

    public Student getStudent() {
        return student;
    }

    @Override
    public String toString() {
        return "Row [student=" + student + "]";
    }
}

public class CSVWriter {
    private List<Student> entries;

    public void setEntries(List<Student> entries) {
        this.entries = entries;
    }

    public static List<Student> getCSV(String csv) {
        String[] lines = csv.split("\n");
        List<Student> result = new ArrayList<>();
        for (String line : lines) {
            if (!line.isEmpty()) {
                String[] columns = line.split("\\t");
                Student student = new Student(columns[0], Integer.parseInt(columns[1]));
                entries.add(student);
            }
        }
        return result;
    }

    public static void printCSV(List<Student> entries, boolean header) {
        System.out.println("Name\tAge");
        for (Student entry : entries) {
            System.out.print(entry.toString());
            System.out.println();
        }
    }

    public static void main(String[] args) throws IOException, CsvValidationException {
        Validator validator = new Validator();
        CSVReader csvReader = new CSVReader();
        CSVWriter csvWriter = new CSVWriter();
        String csv = csvReader.getCSV("Name\tAge\nJohn\t25\nAlice\t30");
        csvWriter.setHeaders(csv.split("\t"));
        csvWriter.printCSV(csvReader.getEntries(Arrays.asList(new Student[]{new Student("John", 25), new Student("Alice", 30)})), true);
    }
}
