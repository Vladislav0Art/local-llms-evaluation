package org.jsoup.nodes;

import java.util.List;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

public class HtmlDom {
    public static void main(String[] args) {
        String html = "<html><body><p>Hello, World!</p></body></html>";
        System.out.println(parseHtml(html));
    }

    private static Document parseHtml(String html) {
        Document doc = new Document();
        return parse(html, doc);
    }

    private static Document parse(String html, Document document) {
        // Parse HTML
        Element root = new HtmlElement("html", doc);
        Element head = new HtmlElement("head", doc);
        Element title = new HtmlElement("title", doc);

        // Title Tag
        String[] titleValues = {"The HTML5 Book"};
        for (String value : titleValues) {
            element = document.createElement("meta");
            element.setAttribute("name", "author");
            element.setAttribute("content", value);
            head.appendChild(element);
        }

        root.appendChild(head);

        return doc;
    }
}

class HtmlElement extends Element {
    private String name;

    public HtmlElement(String name, Document document) {
        super(name, document);
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        sb.append("<").append(name).append(">");
        for (Attribute attr : attributes()) {
            if (!attr.name.equals("_")) {
                sb.append(attr.name).append("=").append(attr.value()).append(" ");
            }
        }
        sb.append("/>");
        return sb.toString();
    }

    public static class Attributes extends AbstractElementAttributes {
        private List<Attribute> values = new ArrayList<>();

        @Override
        public Iterator getIterator() {
            for (Attribute attr : attributes()) {
                if (!attr.name.equals("_")) {
                    values.add(attr);
                }
            }
            return values.iterator();
        }

        public boolean hasValue(String name) {
            for (Attribute attr : values) {
                if (attr.name.equals(name)) {
                    return true;
                }
            }
            return false;
        }

        public String getAttributeValue(String name) {
            for (Attribute attr : values) {
                if (attr.name.equals(name)) {
                    return attr.value();
                }
            }
            return null;
        }

        public void setValue(String value, boolean set) {
            if (set) {
                for (Attribute attr : values) {
                    if (attr.name.equals(name)) {
                        attr.value = value;
                    }
                }
            } else {
                for (Attribute attr : values) {
                    if (!attr.name.equals(name)) {
                        attr.value = null;
                    }
                }
            }
        }

        public String getName() {
            return name;
        }
    }
}

class Element extends Node {
    private List<Attribute> attributes;
    private Attributes attributes;

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        for (Attribute attr : attributes()) {
            if (!attr.name.equals("_")) {
                sb.append(attr.name).append("=").append(attr.value()).append(" ");
            }
        }
        return super.toString() + sb.toString();
    }

    @Override
    public Element clone() {
        Element clone = new Element(getName(), document());
        for (Attribute attr : attributes()) {
            if (!attr.name.equals("_")) {
                clone.attributes.add(attr);
            }
        }
        return clone;
    }

    @Override
    public Document getDocument() {
        return document;
    }

    @Override

package org.jsoup.nodes;

import java.util.List;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

public class HtmlDom {
    public static void main(String[] args) {
        String html = "<html><head><title>Hello, World!</title></head></html>";
        System.out.println(parseHtml(html));
    }

    private static Document parseHtml(String html) {
        Document doc = new Document();
        return parse(html, doc);
    }

    private static Document parse(String html, Document document) {
        // Parse HTML
        Element root = new HtmlElement("html", doc);
        Element head = new HtmlElement("head", doc);

        // Title Tag
        String titleText = "<title>Hello, World!</title>";
        Matcher matcher = Pattern.compile("<title>(.*?)</title>", Pattern.CASE_INSENSITIVE).matcher(titleText);
        if (matcher.find()) {
            String titleValue = matcher.group(1);
            element = document.createElement("meta");
            element.setAttribute("name", "author");
            element.setAttribute("content", titleValue);
            head.appendChild(element);
        }

        root.appendChild(head);

        return doc;

This program uses regular expressions to parse the `title` tag and extract the text content as a string. The extracted text is then wrapped in an `<meta>` element with the specified properties.

Note that this program assumes that the `title` tag is contained within a `<head>` element, which is also assumed by the CSS selector used for parsing the HTML. If the layout of your HTML document is different from what is depicted here, you may need to adjust the regular expression and CSS selector accordingly.
