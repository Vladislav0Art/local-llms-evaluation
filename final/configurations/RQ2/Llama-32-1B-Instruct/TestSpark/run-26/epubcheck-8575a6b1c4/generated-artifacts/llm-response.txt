class EpubChecker {
    public static String validateFile(String report, String version, String profile) throws Exception {
        try {
            Method method = EpubChecker.class.getMethod("getLocale", Object.class);
            Locale locale = (Locale) method.invoke(null, Object.class);

            return "Validating file '" + report + "' with version " + version + ", profile " + profile + " and locale " + locale.toString();
        } catch (Exception e) {
            throw new Exception("Error validating file: " + e.getMessage());
        }
    }

    public static String getLocale(Locale locale) {
        return locale.getCountry() + "-" + locale.getRegion();
    }
}
