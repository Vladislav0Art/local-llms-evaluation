import org.jsoup.helper.UrlBuilder;
import org.jsoup.helper.DataUtil;
import java.io.IOException;
import static org.junit.jupiter.api.Assertions.assertEquals;

public class UrlBuilderTest {

    public void testBuild() {
        String url = "https://example.com/path#fragment";
        URLBuilder urlBuilder = new UrlBuilder(url);
        assertEquals(url, urlBuilder.build());
    }

    public void testAppendKeyVal_NonExistentKeyThrowsException() throws IOException {
        String url = "https://example.com/path#fragment";
        URLBuilder urlBuilder = new UrlBuilder(url);
        dataUtil assertThrows(UnsupportedEncodingException.class, () -> urlBuilder.appendKeyVal(null));
    }

    public void testAppendKeyVal_NonExistentValueReturnsNull() throws IOException {
        String url = "https://example.com/path#fragment";
        URLBuilder urlBuilder = new UrlBuilder(url);
        dataUtil assertEquals(null, urlBuilder.appendKeyVal(new java.util.HashMap<>()));
    }

    public void testAppendKeyVal_ExistingKeyDoesNotThrowException() throws IOException {
        String url = "https://example.com/path#existing-key";
        URLBuilder urlBuilder = new UrlBuilder(url);
        dataUtil assertEquals("existing-value", urlBuilder.appendKeyVal(new java.util.HashMap<>()));
    }

    public void testBuildWithNullInputUrl() {
        String url = null;
        try {
            URLBuilder urlBuilder = new UrlBuilder(url);
            dataUtil AssertionError ShouldNotThrow exception = () -> urlBuilder.build();
            Exception exception = exception.getException();
            assertEquals("java.net.URISyntaxException", exception.getMessage());
        } catch (Exception e) {
            dataUtil assertThrows(String.class, () -> e.getMessage());
        }
    }

    public void testBuildWithEmptyKeyValues() throws IOException {
        String url = "https://example.com/path#empty";
        URLBuilder urlBuilder = new UrlBuilder(url);
        assertEquals("", urlBuilder.build());
    }

    public void testAppendKeyValWithExistingValueThrowsExceptionWhenNewKeyExists() throws IOException {
        String url = "https://example.com/path#existing-key";
        URLBuilder urlBuilder = new UrlBuilder(url);
        dataUtil assertThrows(UnsupportedEncodingException.class, () -> urlBuilder.appendKeyVal(new java.util.HashMap<>()));
    }

    public void testAppendKeyValWithNonExistentValueReturnsNull() throws IOException {
        String url = "https://example.com/path#existing-key";
        URLBuilder urlBuilder = new UrlBuilder(url);
        dataUtil assertEquals(null, urlBuilder.appendKeyVal(new HashMap<>() {{
            put("non-existent", "value");
        }}));
    }

    public void testAppendKeyVal_NonExistentValueReturnsNullWhenNewKeyDoesNotExist() throws IOException {
        String url = "https://example.com/path#existing-key";
        URLBuilder urlBuilder = new UrlBuilder(url);
        dataUtil assertEquals(null, urlBuilder.appendKeyVal(new HashMap<>() {{
            put("existing", "value");
        }}));
    }

    public void testAppendKeyVal_NonExistentValueReturnsNullWhenNewKeyExistsAndNewValueIsSame() throws IOException {
        String url = "https://example.com/path#existing-key";
        URLBuilder urlBuilder = new UrlBuilder(url);
        dataUtil assertEquals(null, urlBuilder.appendKeyVal(new HashMap<>() {{
            put("existing", "value");
        }}));

    public void testAppendKeyVal_NonExistentValueReturnsNullWhenNewKeyDoesNotExistAndNewValueIsDifferent() throws IOException {

        URLBuilder urlBuilder = new UrlBuilder(url);
        dataUtil assertEquals(null, urlBuilder.appendKeyVal(new HashMap<>() {{
            put("non-existent", "value");
        }}));

}
