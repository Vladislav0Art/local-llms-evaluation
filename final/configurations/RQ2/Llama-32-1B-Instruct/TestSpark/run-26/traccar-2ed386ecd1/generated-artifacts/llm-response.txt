import java.util.HashMap;
import java.util.Map;

public class TestGt06ProtocolDecoder {
    public static String decode(String protocolType, Map<String, Object> channel) {
        if (channel != null && ((Map<String, Object>) channel).containsKey("type") 
                && ((Map<String, Object>) channel).get("type").equals(protocolType)) {
            return protocolType + "(" + channel.get("address") + ":" + channel.get("port") + ")";
        } else {
            return null;
        }
    }

    public static void main(String[] args) {
        // Creating Protocol instance
        Map<String, Object> protocol = new HashMap<>();
        protocol.put("type", "GT06");
        protocol.put("address", "192.168.1.100");
        protocol.put("port", 1234);

        // Using the decode method of GT06ProtocolDecoder with type and address
        String result1 = decode(protocol.get("type"), null);
        
        System.out.println(result1);

        // Using the decode method of GT06ProtocolDecoder with type, address and port
        String result2 = decode(protocolType, protocol);
        
        if (result2 != null) {
            System.out.println(result2);
        } else {
            System.out.println("No channel provided");
        }
    }

    public static class Channel {
        private String type;
        private String address;
        private int port;

        public Channel(String type, String address, int port) {
            this.type = type;
            this.address = address;
            this.port = port;
        }

        public String getType() {
            return type;
        }

        public String getAddress() {
            return address;
        }

        public int getPort() {
            return port;
        }
    }
}

This program creates a `GT06ProtocolDecoder` class with two methods: `decode(String protocolType, Map<String, Object> channel)` and `main(String[] args)`. The `decode()` method takes three parameters (protocol type, channel map, or null), checks the protocol type in the channel map, and returns a string in the format "protocol_type(address:port)" if found. It also handles cases where the channel is not provided.

In the `main()` method, we create a protocol instance with specific details. We then use the `decode()` method to decode the protocol and print the result. Additionally, we test the `decode()` method without providing a channel by checking if it returns null.
