```java
public class Calculator {
    public static void main(String[] args) throws Exception {
        int num1 = 10;
        int num2 = 3;

        System.out.println("Addition: " + add(num1, num2)); // Expected: 13
        System.out.println("Subtraction: " + subtract(num1, num2)); // Expected: 7

        num1 = 15;
        num2 = 5;

        System.out.println("Multiplication: " + multiply(num1, num2)); // Expected: 75

        try {
            System.out.println("Division: " + divide(num1, num2)); // Expected: result is not an integer
        } catch (Exception e) {
            System.out.println(e.getMessage()); // Expected: Cannot divide by zero
        }

        int[] numbers = {10, 20, 30};
        double sum = add(numbers);
        double average = calculateAverage(numbers);

        System.out.println("Sum of array elements: " + sum); // Expected: 60
        System.out.println("Average of array elements: " + average); // Expected: 15.0
    }

    public static int add(int num1, int num2) {
        return num1 + num2;
    }

    public static String subtract(int num1, int num2) throws Exception {
        if (num2 > num1) {
            throw new Exception("Number with second argument cannot be greater than the first");
        }
        return Integer.toString(num1 - num2);
    }

    public static String multiply(int num1, int num2) {
        return Integer.toString(num1 * num2);
    }

    public static double calculateAverage(double[] numbers) throws Exception {
        if (numbers.length == 0) {
            throw new Exception("Cannot calculate average of an empty array");
        }
        double sum = 0;
        for (double number : numbers) {
            sum += number;
        }
        return sum / numbers.length;
    }

    public static String divide(double num1, double num2) throws Exception {
        if (num2 == 0) {
            throw new Exception("Cannot divide by zero");
        }
        return Double.toString(num1 / num2);
    }
}
```
