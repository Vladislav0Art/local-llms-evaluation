import com.netflix.frigga.NameConstants;
import org.junit.Test;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

public class AppVersionTest {

    private static final Pattern APP_VERSION_PATTERN = Pattern.compile("([a-zA-Z0-9_]+)\\.(?:(?:[a-z]{2}\\.|(?:(?:\\d{4}|\\d{2}\\d))\\d{2}))");

    @Test
    public void testParseName() {
        String amiName = "aws-november-2017.1";
        AppVersion appVersion = AppVersion.parseName(amiName);
        assertEquals("aws", appVersion.getPackageName());
        assertEquals("november", appVersion.getVersion());
    }

    @Test
    public void testGetAppVersionPattern() {
        String pattern = APP_VERSION_PATTERN.toString();
        String expected = "[APP_VERSION]";
        Assert.assertTrue(pattern.equals(expected));
    }

    @Test
    public void testGetPackageName() {
        AppVersion appVersion = new AppVersion("aws-november-2017.1");
        assertEquals("aws", appVersion.getPackageName());
    }

    @Test
    public void testGetVersion() {
        AppVersion appVersion = new AppVersion("aws-november-2017.1");
        assertEquals("november-2017.1", appVersion.getVersion());
    }

    @Test
    public void testGetBuildJobName() {
        AppVersion appVersion = new AppVersion("aws-november-2017.1");
        assertEquals("build-job-name-2017-01", appVersion.getBuildJobName());
    }

    @Test
    public void testGetBuildNumber() {
        AppVersion appVersion = new AppVersion("aws-november-2017.1");
        assertEquals("12345", appVersion.getBuildNumber());
    }

    @Test
    public void testGetCommit() {
        AppVersion appVersion = new AppVersion("aws-november-2017.1");
        assertEquals("commit-hash", appVersion.getCommit());
    }

    @Test
    public void testGetChangelist() {
        AppVersion appVersion = new AppVersion("aws-november-2017.1");
        assertNotNull(appVersion.getChangelist());
    }

    @Test
    public void testToString() {
        AppVersion appVersion = new AppVersion("aws-november-2017.1");
        assertEquals("[APP_VERSION][aws-november-2017.1]", appVersion.toString());
    }

    @Test
    public void testEquals() {
        AppVersion appVersion1 = new AppVersion("aws-november-2017.1");
        AppVersion appVersion2 = new AppVersion("aws-november-2017.1");
        assertTrue(appVersion1.equals(appVersion2));
        assertFalse(appVersion1.equals(new String(0)));
    }
}
