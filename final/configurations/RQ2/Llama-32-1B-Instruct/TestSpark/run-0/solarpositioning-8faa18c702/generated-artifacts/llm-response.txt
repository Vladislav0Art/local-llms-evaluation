public class SolarPositionCalculator {

    public double getHour(double[] date, int time) {
        // Convert date to year, month, day
        int year = date[0];
        int month = date[1];
        int day = date[2];

        // Calculate hour from 24 hours at the given time
        double hour = (time + (date[3] - 25) / 15) % 24;

        return hour;
    }

    public static void main(String[] args) {
        SolarPositionCalculator calculator = new SolarPositionCalculator();
        
        double[] date = {2022, 1, 1, 0};
        int time = 9; // in hours
        
        SolarPosition solarPosition = calculator.getSolarPosition(date, time);
        System.out.println(solarPosition.getHour());
    }
}

public class SolarPositionCalculator {

    public double getSolarPosition(double[] date) {
        return new SolarPosition(date, 0).getHour();
    }

    @Test
    void testSolarPosition() {
        double[] date = {2022, 1, 1};
        double hour = calculator.getSolarPosition(date);
        assertEquals(20.0, hour, 0.01), 
                "Incorrect hour";
    }

    public class SolarPosition {

        private double date;
        private int time;

        public SolarPosition(double[] date, int time) {
            this.date = date;
            this.time = time;
        }

        public double getHour() {
            return this.date[2] + (this.time / 15);
        }
    }

    @Test
    void testSolarPosition_12am() {
        SolarPosition solarPosition = new SolarPosition(new double[]{2022, 1, 1, 0}, 18);
        assertEquals(16.0, solarPosition.getHour(), 0.01), 
                "Incorrect hour";
