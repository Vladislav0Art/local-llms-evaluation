def get_attribute_value(attr_key, attributes):
    """
    Returns the value associated with the given attribute key.
    
    Args:
        attr_key (str): The key of the attribute to retrieve.
        attributes (dict): The dictionary containing the attributes.
    
    Returns:
        any: The value associated with the given attribute key. 
             Raises a KeyError if the attribute key is not found.
    """
    return attributes.get(attr_key)

def set_attribute_value(attributes, attr_key, value):
    """
    Sets the value of the given attribute key in the dictionary.
    
    Args:
        attributes (dict): The dictionary containing the attributes.
        attr_key (str): The key to be updated.
        value: The new value for the attribute key.
    """
    if isinstance(attr_key, str):
        attributes[attr_key] = value
    elif len(attr_key) != 1 or not attr_key.isdigit():
        raise TypeError("Invalid attribute key type")
    else:
        try:
            int(attr_key)
        except ValueError:
            raise TypeError("Attribute keys must be integers")

def get_attribute_key(index):
    """
    Returns the index of the given attribute key in the dictionary.
    
    Args:
        index (int): The index to find in the dictionary.
    
    Returns:
        str: The index of the attribute key. Raises a KeyError if the index is out of range.
    """
    return str(range(len(attributes.keys()))).split(',') + [index]

def get_key_dict(keys):
    """
    Returns a dictionary containing the indices and values of the given keys.
    
    Args:
        keys (list): The list of attribute keys.
    
    Returns:
        dict: A dictionary containing the indices and values of the given keys. 
             Raises a ValueError if any key is not found in the dictionary.
    """
    return {get_attribute_key(index): get_attribute_value(key, attributes) for index, key in enumerate(keys)}

def set_attribute_dict(attributes, new_keys):
    """
    Sets the values of the given attribute keys in the dictionary.
    
    Args:
        attributes (dict): The dictionary containing the attributes.
        new_keys (list): The list of new attribute keys to be updated.
    """
    for index, key in enumerate(new_keys):
        set_attribute_value(attributes, key, get_attribute_value(key, attributes))

def print_attributes(attributes):
    """
    Prints the contents of the given dictionary.
    
    Args:
        attributes (dict): The dictionary containing the attributes.
    """
    for key, value in attributes.items():
        print(f"{key}: {value}")

def main():
    # Create an instance of the Attributes class
    attributes = {}
    
    # Set attribute values
    set_attribute_value(attributes, "key1", "value1")
    set_attribute_value(attributes, "key2", "value2")
    
    # Get attribute values
    print_attributes(get_attribute_dict(["key1", "key2"]))
    
    # Check if a key exists
    try:
        get_attribute_key("0")
    except KeyError as e:
        print(e)
    
    # Update attribute value
    set_attribute_value(attributes, "key3", "new value")
    
    # Get updated attribute values
    print_attributes(get_attribute_dict(["key1", "key2"]))
    
    # Add a new key-value pair
    attributes["key4"] = "value4"
    
    # Print the updated dictionary
    print_attributes(attributes)
