        WebDriver driver = new org.openqa.selenium.ChromeDriver();
        driver.get("http://www.test.com");

        WebElement pageTitle = driver.findElement(By.tagName("title"));
        System.out.println(pageTitle.getText());

        String publicMethod1 = "public class Test { @Override public String myMethod() {} }";
        try {
            Thread.sleep(1000);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }

        String publicMethod2 = "public void testPublicMethods() {{ System.out.println(\"Hello World\"); }}";
        try {
            Thread.sleep(2000);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }

        String[] expectedMethods = {"myMethod", "testPublicMethods"};
        for (String method : expectedMethods) {
            if (!method.equals(publicMethod1)) {
                throw new RuntimeException("Expected " + method + ", got " + publicMethod2);
            }
        }
    }

    public void testNestedClasses() throws Exception {

import org.openqa.selenium.By;
import org.testng.annotations.Test;

public class TestBrowser {
    @Test
    public void testClick() {
        // Define the browser method
        public void openPage(String url) {
            System.out.println("Opening page: " + url);
        }

        // Open a new tab with the specified URL
        public void navigateToTab(String url) {
            System.out.println("Navigating to tab: " + url);
        }

        // Click on an element with the specified tag name and id
        public void clickElement(By.byId, String text) {
            System.out.println("Clicking on element: " + text);
        }

        // Get a reference to the current page object
        public PageObject getPage() {
            return new PageObject();
        }
    }
}

class PageObject {}

public class BrowserTest {
    @Test
    public void testClick() {
        TestBrowser browser = new TestBrowser();
        browser.openPage("https://www.google.com");
        browser.navigateToTab("https://www.bing.com");

        // Call the method on the page object
        browser.getPage().clickElement(By.byId, "search_box");
    }
}
