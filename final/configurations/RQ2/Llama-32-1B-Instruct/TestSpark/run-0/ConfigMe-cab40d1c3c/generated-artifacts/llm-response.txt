import ch.jalu.configme.configurationdata.PropertyListBuilder;
import ch.jalu.configme.exception.ConfigMeException;
import ch.jalu.configme.properties.Property;
import org.jetbrains.annotations.NotNull;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

public class PropertyListBuilderTest {

    public void testAddProperty() {
        PropertyListBuilder builder = new PropertyListBuilder();
        Property property1 = new Property("key1", "value1");
        Property property2 = new Property("key2", "value2");

        builder.add(property1);
        builder.add(property2);

        assertEquals(2, builder.create().size());
    }

    public void testCreate() {
        PropertyListBuilder builder = new PropertyListBuilder();
        List<Property> list = builder.create();

        assertEquals(0, list.size());

        Property property = new Property("key", "value");
        list.add(property);

        assertEquals(1, list.size());

        assertEquals(property, list.get(list.size() - 1));
    }

    public void testGetRootEntries() {
        PropertyListBuilder builder = new PropertyListBuilder();
        Map<String, Object> map = new LinkedHashMap<>();
        map.put("key1", "value1");
        map.put("key2", "value2");

        builder.getRootEntries().putAll(map);

        assertEquals(2, builder.getRootEntries().size());

        Property property = new Property("key3", "value3");
        builder.getRootEntries().put(property.getKey(), property.getValue());

        assertEquals(2, builder.getRootEntries().size());
    }

    public void testGetRootEntriesInvalidScenario() {
        PropertyListBuilder builder = new PropertyListBuilder();
        Map<String, Object> map = new LinkedHashMap<>();
        map.put("key", "value");
        builder.getRootEntries().putAll(map);

        try {
            builder.getRootEntries().get("non-existent-key");
        } catch (ConfigMeException e) {
            // expected exception
        }

        assertEquals(2, builder.getRootEntries().size());

        Property property = new Property("key3", "value3");
        builder.getRootEntries().put(property.getKey(), property.getValue());
    }
}
