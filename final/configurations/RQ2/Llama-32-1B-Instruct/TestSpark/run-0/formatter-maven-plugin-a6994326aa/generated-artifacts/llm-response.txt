public class CssFormatterTest {

    public void testCssFormatNoOptions() {
        // Arrange
        String input = "body { color: blue; }";
        LineEnding ending = LineEnding.CRLF;

        // Act
        CssFormatter formatter = new CssFormatter();
        String actual = formatter.doFormat(input, ending);

        // Assert
        assertEquals("body { color: blue; }", actual);
    }

    public void testCssFormatOneRule() {
        // Arrange
        String input = "body { color: blue; }";
        LineEnding ending = LineEnding.CRLF;

        // Act
        CssFormatter formatter = new CssFormatter();
        String actual = formatter.doFormat(input, ending);

        // Assert
        assertEquals("body { color: blue; }", actual);
    }

    public void testCssFormatMultipleRules() {
        // Arrange
        String input = "body { color: blue; } span { font-size: 12px; }";
        LineEnding ending = LineEnding.CRLF;

        // Act
        CssFormatter formatter = new CssFormatter();
        String actual = formatter.doFormat(input, ending);

        // Assert
        assertEquals("body { color: blue; }\nspan { font-size: 12px; }", actual);
    }

    public void testCssFormatEmptyString() {
        // Arrange
        String input = "";
        LineEnding ending = LineEnding.CRLF;

        // Act
        CssFormatter formatter = new CssFormatter();
        String actual = formatter.doFormat(input, ending);

        // Assert
        assertEquals("", actual);
    }

    public void testCssFormatInvalidSyntax() throws Exception {
        // Arrange
        String input = "body { color: blue; } span { font-size: 12px !important; }";
        LineEnding ending = LineEnding.CRLF;

        // Act
        CssFormatter formatter = new CssFormatter();
        try {
            formatter.doFormat(input, ending);
        } catch (Exception e) {
            // Assert
            assertTrue(e.getMessage().contains("Invalid CSS syntax"));
        }
    }

    public void testCssFormatEmptyInput() {
        // Arrange
        String input = "";
        LineEnding ending = LineEnding.CRLF;

        // Act
        CssFormatter formatter = new CssFormatter();
        try {
            formatter.doFormat(input, ending);
        } catch (IOException e) {
            // Assert
            assertTrue(e.getMessage().contains("Invalid CSS code"));
        }
    }

    public void testCssFormatCSSCode() throws Exception {
        // Arrange
        String input = "{body { color: blue; }}\nspan { font-size: 12px; }\n#id { background-color: red; }\n!important";
        LineEnding ending = LineEnding.CRLF;

        // Act
        CssFormatter formatter = new CssFormatter();
        String actual = formatter.doFormat(input, ending);

        // Assert
