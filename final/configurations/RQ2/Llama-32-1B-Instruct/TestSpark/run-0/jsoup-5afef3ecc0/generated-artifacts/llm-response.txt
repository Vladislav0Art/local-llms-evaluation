```java
        URI uri = new URI(url);
        String query = uri.getQuery();

                url += "&" + key + "=" + val;
```

```java
import java.net.URI;
import java.net.URISyntaxException;

public class UrlBuilder {
    private final String url;

    public UrlBuilder(String url) {
        this.url = url;
    }

    public void appendKeyVal(String key, Object value) throws URISyntaxException {
        URI uri = new URI(url);
        String query = uri.getQuery();
        if (query == null || query.isEmpty()) {
            throw new URISyntaxException(uri, "No query found");
        }
        String[] pairs = query.split("&");
        for (int i = 0; i < pairs.length; i++) {
            String pair = pairs[i];
            int index = pair.indexOf("=");
            if (index != -1) {
                String keyVal = pair.substring(0, index);
                String val = pair.substring(index + 1);
                url += "&" + keyVal + "=" + val;
            }
        }
    }

    public void appendQuery(String query) throws URISyntaxException {
        URI uri = new URI(url);
        String existingQuery = uri.getQuery();
        if (existingQuery == null || existingQuery.isEmpty()) {
            uri.setQuery(query);
        } else {
            String[] pairs = existingQuery.split("&");
            for (int i = 0; i < pairs.length; i++) {
                String pair = pairs[i];
                int index = pair.indexOf("=");
                if (index != -1) {
                    String keyVal = pair.substring(0, index);
                    String val = pair.substring(index + 1);
                    url += "&" + keyVal + "=" + val;
                }
            }
        }
    }

    public void appendQueryAndFragment(String query, String fragment) throws URISyntaxException {
        URI uri = new URI(url);
        String existingQuery = uri.getQuery();
        if (existingQuery == null || existingQuery.isEmpty()) {
            uri.setQuery(query);
        } else {
            String[] pairs = existingQuery.split("&");
            for (int i = 0; i < pairs.length; i++) {
                String pair = pairs[i];
                int index = pair.indexOf("=");
                if (index != -1) {
                    String keyVal = pair.substring(0, index);
                    String val = pair.substring(index + 1);
                    url += "&" + keyVal + "=" + val;
                }
            }
        }
    }

    public static void main(String[] args) throws URISyntaxException {
```

```java
import java.net.URI;
import java.util.HashMap;
import java.util.Map;

public class UrlBuilder {
    private final String url;

    public UrlBuilder(String url) {
        this.url = url;
    }

    public void appendKeyVal(String key, Object value) throws URISyntaxException {
        URI uri = new URI(url);
        String query = uri.getQuery();
        if (query == null || query.isEmpty()) {
            throw new URISyntaxException(uri, "No query found");
        }
        Map<String, String> params = parseQuery(query);
        Map<String, String> updatedParams = new HashMap<>();
        for (Map.Entry<String, String> entry : params.entrySet()) {
            if (!entry.getValue().startsWith("http")) {
                updatedParams.put(entry.getKey(), entry.getValue());
            } else {
                updatedParams.putAll(entry);
            }
        }
        query = updateQuery(query, updatedParams);
        url += "?" + query;
    }

    private Map<String, String> parseQuery(String query) throws URISyntaxException {
        URI uri = new URI(url);
        String[] pairs = uri.getQuery().split("&");
        Map<String, String> params = new HashMap<>();
        for (String pair : pairs) {
            int index = pair.indexOf("=");
            if (index != -1) {
                params.put(pair.substring(0, index), pair.substring(index + 1));
            }
        }
        return params;
    }

    private String updateQuery(String query, Map<String, String> params) throws URISyntaxException {
        URI uri = new URI(url);
        String existingQuery = uri.getQuery();
        if (existingQuery == null || existingQuery.isEmpty()) {
            uri.setQuery(query);
        } else {
            String[] pairs = existingQuery.split("&");
            for (int i = 0; i < pairs.length; i++) {
                String pair = pairs[i];
                int index = pair.indexOf("=");
                if (index != -1) {
                    String keyVal = pair.substring(0, index);
                    String val = pair.substring(index + 1);
                    uri.setQuery(uri.getQuery() + "&" + keyVal + "=" + val);
                }
            }
        }
        return uri.getQuery();
    }

    public void appendFragment(String fragment) throws URISyntaxException {
        URI uri = new URI(url);
        String existingFragment = uri.getFragment();
        if (existingFragment == null || existingFragment.isEmpty()) {
            uri.setFragment(fragment);
        } else {
            String[] pairs = existingFragment.split("/");
            for (int i = 0; i < pairs.length; i++) {
                String pair = pairs[i];
                int index = pair.indexOf("=");
                if (index != -1) {
                    String keyVal = pair.substring(0, index);
                    String val = pair.substring(index + 1);
                    uri.setFragment(uri.getFragment() + "/" + keyVal + "=" + val);
                }
            }
        }
    }

    public static void main(String[] args) throws URISyntaxException {
        UrlBuilder builder = new UrlBuilder("https://example.com?a=1&b=2");
        System.out.println(builder.appendKeyVal("a", 3));
        System.out.println(builder.appendQuery("q=3"));
        System.out.println(builder.appendFragment("def"));
    }
}
```
