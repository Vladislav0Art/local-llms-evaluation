public class UrlUtils {
    public static void appendKeyVal(StringBuilder sb, String key, String value) throws NullPointerException {
        if (sb != null && !sb.isEmpty()) {
            sb.append("=").append(key).append("=");
            sb.append(value);
        } else {
            throw new NullPointerException("StringBuilder is empty");
        }
    }

    public static void appendKeyVal(StringBuilder sb, int key, String value) throws NullPointerException {
        if (sb != null && !sb.isEmpty()) {
            sb.append("=").append(key).append("=").append(value);
        } else {
            throw new NullPointerException("StringBuilder is empty");
        }
    }

    public static void appendKeyVal(StringBuilder sb, Object key, Object value) throws NullPointerException {

            sb.append("=").append(key.toString()).append("=").append(value.toString());

            throw new NullPointerException("StringBuilder is empty");
        }
    }

    public static String removeChar(String str, char c) {
        StringBuilder sb = new StringBuilder();
        for (char ch : str.toCharArray()) {
            if (ch != c) {
                sb.append(ch);
            }
        }
        return sb.toString();
    }

    public static int[] swap(int[] arr) {
        int temp = arr[0];
        arr[0] = arr[1];
        arr[1] = temp;
        return arr;
    }

        StringBuilder sb = new StringBuilder(str);
        return sb.reverse().toString();
