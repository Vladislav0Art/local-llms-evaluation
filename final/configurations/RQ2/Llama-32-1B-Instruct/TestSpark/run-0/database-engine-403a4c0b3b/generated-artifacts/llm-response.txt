class CSVWriter {
    private List<String[]> data;

    public CSVWriter() {
        this.data = new ArrayList<>();
    }

    public void writeAll(String[] fields) {
        for (String field : fields) {
            if (!field.isEmpty()) {
                data.add(new String[]{null, field});
            } else {
                data.add(null);
            }
        }
    }

    public Iterator<String[]> iterator() {
        return new Iterator<String[]>() {
            private int index = 0;

            @Override
            public boolean hasNext() {
                return index < data.size();
            }

            @Override
            public String[] next() {
                if (!data.isEmpty()) {
                    return data.get(index++);
                } else {
                    return null;
                }
            }
        };
    }

    public void close() {
        // implementation
    }
}

public class Database {
    private String[][] data;

    public Database() {
        this.data = new String[0][0];
    }

    public void addRow(String[] values) {
        if (data.length == 0 || values.length != data[0].length + 1) {
            System.out.println("Invalid row size. Expected " + (data.length * data[0].length));
            return;
        }
        String[] newValues = new String[data.length];
        for (int i = 0; i < data.length; i++) {
            newValues[i] = values[i];
        }
        System.arraycopy(data, 0, newValues, 0, data[0].length);
        newValues[newValues.length - 1] = values[values.length - 1];
        data = new String[data.length + 1][data[0].length + 1];
        for (int i = 0; i < data.length; i++) {
            System.arraycopy(newValues, 0, data[i], 0, data[i].length);
        }
    }

    public void printTable() {
        if (data.length == 0) {
            return;
        }
        String[] columns = {"Name", "Age"};
        for (int i = 0; i < data[0].length; i++) {
            System.out.print(columns[i] + ": ");
        }
        System.out.println();
        for (String[] row : data) {
            for (int j = 0; j < row.length - 1; j++) {
                System.out.print(row[j] + " | ");
            }
            System.out.println(row[row.length - 1]);
        }
    }

    public static void main(String[] args) throws Exception {
        Database db = new Database();
