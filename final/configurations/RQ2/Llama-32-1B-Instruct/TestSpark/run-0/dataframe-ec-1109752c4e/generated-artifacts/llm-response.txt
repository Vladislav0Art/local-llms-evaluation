```java
public abstract class PrettyPrintVisitor implements Visitor {
    @Override
    public void visit(Annot annot, Value value) {
        // Do nothing for now, but can be used to customize the print behavior
    }

    @Override
    public Object visit(String string, Value value) {
        // Convert the string to a new type (e.g. int)
        return Integer.parseInt(string);
    }
}

public class Visitor implements PrettyPrintVisitor {
    private String result;

    @Override
    public void visit(Annot annot, Value value) {
        this.result = value.toString();
    }

    @Override
    public Object visit(String string, Value value) {
        System.out.println(result + " " + string);
        return null;
    }
}
```

```java
public class PrettyPrinter extends Visitor {
    private String result;

    @Override
    public void print(Object object) {
        if (object instanceof String) {
            result = ((String)object).replace(" ", "");
        } else if (object instanceof Number) {
            result = ((Number)object).doubleValue().toString();
        }
    }

    @Override
    public Object visit(Annot annot, Value value) {
        return print(value);
    }

    @Override
    public void print(String string) {
        System.out.println(string);
    }
}
```

public class PrettyPrintVisitorTest {
    private PrettyPrintVisitor visitor;

    @Before
    public void setup() {
        visitor = new PrettyPrintVisitor();
    }

    public static Value visitConstExpr(Value value) {
        if (value instanceof ConstExpr) {
            return ((ConstExpr)value).getConstantValue();
        } else {
            throw new RuntimeException("Unsupported type: " + value.getClass());
        }
    }

    public static Value visitUnaryExpr(UnaryExpr.unary) {
        if (unary.getOperator().equals("+") || unary.getOperator().equals("-")) {
            return visitor.visit(unary);
        } else if (unary.getOperator().equals("*") || unary.getOperator().equals("/")) {
            Value left = visitor.visit(unary.getLeft());
            Value right = visitor.visit(unary.getRight());
            return left == null ? null : visitor.visit(new UnaryExpr(left, operator, right));
        } else {
            throw new RuntimeException("Unsupported operator: " + unary.getOperator());
        }
    }

    public static Value visitAssignExpr(AssingExpr.assig) {
        if (assingExpr.getLeft() instanceof AssignExpr) {
            return visitor.visit(assingExpr);
        } else {
            throw new RuntimeException("Left side must be an assignment expression");
        }
    }

    public static Value visitBinaryExpr(BinaryExpr.binary) {
        if (binaryExpr.getOperator().equals("+")) {
            Value left = visitor.visit(binaryExpr.getLeft());
            Value right = visitor.visit(binaryExpr.getRight());
            return visitor.visit(new BinaryExpr(left, operator, right));
        } else if (binaryExpr.getOperator().equals("-")) {
            Value left = visitor.visit(binaryExpr.getLeft());
            Value right = visitor.visit(binaryExpr.getRight());
            return visitor.visit(new BinaryExpr(left, operator, right));
        } else if (binaryExpr.getOperator().equals("*") || binaryExpr.getOperator().equals("/")) {
            Value left = visitor.visit(binaryExpr.getLeft());
            Value right = visitor.visit(binaryExpr.getRight());
            return visitor.visit(new BinaryExpr(left, operator, right));
        } else {
            throw new RuntimeException("Unsupported operator: " + binaryExpr.getOperator());
        }
    }

    public static Value visitFunctionCallExpr(FunctionCallExpr.function) {
        if (functionCallExpr.getLeft() instanceof FunctionCallExpr) {
            return visitor.visit(functionCallExpr);
        } else {
            throw new RuntimeException("Left side must be a function call expression");
        }
    }

    public static Value visitPropertyPathExpr(PropertyPathExpr.property) {
        if (property.getPath().length() != 2 || !property.getPath()[0].getOperator().equals(".") || !property.getPath()[1].getOperator().equals(".")) {
            throw new RuntimeException("Invalid property path: " + property);
        }
        return visitor.visit(property);
    }

    public static Value visitAnonymousScriptExpr(AnonymousScript.annots) {
        if (anonymousScriptExpr.getLeft() instanceof AnonymousScript.annots) {
            return visitor.visit(anonymousScriptExpr);
        } else {
            throw new RuntimeException("Left side must be an anonymous function expression");
        }
    }

    public static Value visitFunctionScriptExpr(FunctionScript.function) {
        if (functionScriptExpr.getLeft() instanceof FunctionScript.function) {
            return visitor.visit(functionScriptExpr);
        } else {
            throw new RuntimeException("Left side must be a function expression");
        }
    }

    public static Value visitStatementSequenceScript(StatementSequenceScript.stream) {
        List<Statement> statements = new ArrayList<>();
        while (stream.getNext() != null) {
            statements.add(stream.getNext());
        }
        return visitor.visit(statements);
    }

    public static Value visitVarExpr(VarExpr.var) {
        if (!varExpr.getValue() instanceof ConstExpr || !varExpr.getValue().getClass().isPrimitive()) {
            throw new RuntimeException("Value must be a primitive type or constant");
        }
        return visitor.visit(varExpr);
    }

    public static Value visitProjectionExpr(ProjectionExpr.projection) {
        List<Constant> constants = new ArrayList<>();
        while (projectionExpr.getLeft() instanceof ProjectionExpr || projectionExpr.getLeft() instanceof ConstExpr) {
            if (projectionExpr.getLeft() != null && !projectionExpr.getLeft().getClass().isPrimitive()) {
                throw new RuntimeException("Left side must be a primitive type or constant");
            }
            constants.add(visitor.visit(projectionExpr.getLeft()));
        }
        return visitor.visit(constants);
    }

    public static Value visitAliasExpr(AliasExpr.aliases) {
        if (aliasExpr.getLeft() instanceof AliasExpr || aliasExpr.getLeft() instanceof ConstExpr) {
            List<Constant> constants = new ArrayList<>();
            while (aliasExpr.getLeft() != null && !aliasExpr.getLeft().getClass().isPrimitive()) {
                constants.add(visitor.visit(aliasExpr.getLeft()));
            }
            return visitor.visit(aliasExpr);
        } else {
            throw new RuntimeException("Left side must be an alias expression or constant");
        }
    }

    public static Value visitVectorExpr(VectorExpr.variable) {
        if (!variable instanceof ConstExpr || !variable.getClass().isPrimitive()) {
            throw new RuntimeException("Value must be a primitive type or constant");
        }
        return visitor.visit(variable);
    }

    public static Value visitIndexExpr(IndexExpr.index) {
        if (indexExpr.getLeft() instanceof ConstExpr || indexExpr.getLeft() instanceof Constant) {
            List<Constant> constants = new ArrayList<>();
            while (indexExpr.getLeft() != null && !indexExpr.getLeft().getClass().isPrimitive()) {
                constants.add(visitor.visit(indexExpr.getLeft()));
            }
            return visitor.visit(indexExpr);
        } else if (!indexExpr.getRight() instanceof ConstExpr || indexExpr.getRight() instanceof Constant) {
            throw new RuntimeException("Right side must be a constant or primitive type");
        } else {
            throw new RuntimeException("Index expression has multiple operators: " + indexExpr.getOperator());
        }
    }

    public static Value visitDecimalExpr(DecimalExpr.decimal) {
        if (!decimalExpr.getValue() instanceof ConstExpr || !decimalExpr.getValue().getClass().isPrimitive()) {
            throw new RuntimeException("Value must be a primitive type or constant");
        }
        return visitor.visit(decimalExpr);
    }

    public static void testVisitIfElseExpr(IFElseExpr.else) {
        Value ifResult = visitIfElseExpr(ifElseExpr);
        System.out.println("If result: " + ifResult);
        List<Statement> statements = new ArrayList<>();
        while (ifElseExpr.getNext() != null) {
            statements.add(ifElseExpr.getNext());
        }
        for (Statement statement : statements) {
            System.out.println(statement.toString());
        }
    }

    public static Value visitFunctionCallExpr(FunctionCallExpr.function) {
        List<Constant> constants = new ArrayList<>();
        while (functionCallExpr.getLeft() != null && !functionCallExpr.getLeft().getClass().isPrimitive()) {
            constants.add(visitor.visit(functionCallExpr.getLeft()));
        }
        return visitor.visit(functionCallExpr);
    }

    public static Value visitPropertyPathExpr(PropertyPathExpr.property) {
        List<Constant> constants = new ArrayList<>();
        while (property.getPath().length() != 2 || property.getPath()[0].getOperator().equals(".") || property.getPath()[1].getOperator().equals(".")) {
            throw new RuntimeException("Invalid property path: " + property);
        }
        return visitor.visit(property);
    }

    public static Value visitAnonymousScriptExpr(AnonymousScript.annots) {
        List<Statement> statements = new ArrayList<>();
        while (anonymousScriptExpr.getLeft() != null && !anonymousScriptExpr.getLeft().getClass().isPrimitive()) {
            statements.add(anonymousScriptExpr.getLeft());
        }
        return visitor.visit(statements);
    }

    public static Value visitFunctionScriptExpr(FunctionScript.function) {
        List<Statement> statements = new ArrayList<>();
        while (functionScriptExpr.getLeft() != null && !functionScriptExpr.getLeft().getClass().isPrimitive()) {
            statements.add(functionScriptExpr.getLeft());
        }
        return visitor.visit(statements);
    }

    public static Value visitStatementSequenceScript(StatementSequenceScript.stream) {
        List<Statement> statements = new ArrayList<>();
        while (stream.getNext() != null) {
            statements.add(stream.getNext());
        }
        return visitor.visit(statements);
    }
