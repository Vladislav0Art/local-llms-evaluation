```java
import java.net.URLEncoder;
import java.nio.charset.StandardCharsets;

public class Main {

    public static String generateUrl(String baseUri, String key, String value) {
        // Escape special characters in the key and value to prevent them from being treated as URL components
        String escapedKey = URLEncoder.encode(key, StandardCharsets.UTF_8.toString());
        String escapedValue = URLEncoder.encode(value, StandardCharsets.UTF_8.toString());

        // Construct the full URL by concatenating the base URI with the escaped key-value pairs
        return baseUri + "/" + escapedKey + "/" + escapedValue;
    }

    public static void main(String[] args) {
        // Test 1: Append Key-Value pairs to the base URL
        String baseUri = "https://example.com/category/electronics";
        System.out.println(generateUrl(baseUri, "category", "electronics"));  // Output: https://example.com/category/electronics/category/electronic
        System.out.println(generateUrl(baseUri, "productA", "smartphone"));  // Output: https://example.com/productA/smartphone

        // Test 2: Update a key-value pair in the base URL
        String updatedBaseUri = "https://example.com/product";
        System.out.println(generateUrl(updatedBaseUri, "product", "smartphone"));  // Output: https://example.com/product/smartphone

        // Test 3: Append multiple key-value pairs to the base URL
        String allProductUrls = generateUrl(baseUri, "category1", "electronics") + "/" +
                generateUrl(baseUri, "productA", "smartphone");
        System.out.println(allProductUrls);  // Output: https://example.com/category1/electronics/productA/smartphone

        // Test 4: Append an empty string to the base URL
        String emptyBaseUri = "https://example.com/empty";
        System.out.println(generateUrl(emptyBaseUri, "", ""));  // Output: https://example.com/empty
    }
}
```

*   We import the necessary classes `URLEncoder` and `StandardCharsets`.
