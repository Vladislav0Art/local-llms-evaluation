class Request {
    private String key;
    private String value;

    public static Request copy(Request request) {
        return new Request(request.key, request.value);
    }

    public String getKey() {
        return key;
    }

    public void setKey(String key) {
        this.key = key;
    }

    public String getValue() {
        return value;
    }

    public void setValue(String value) {
        this.value = value;
    }
}

class Document extends DomElement {
    @Override
    public String toString() {
        return "Document{" +
                "body=" + super.toString() +
                '}';
    }
}

package org.jsoup.helper;

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;
import java.io.IOException;
import java.net.URI;
import java.util.List;
import java.util.Optional;

public class Parser {
    public static Parser from(String string) throws IOException {
        return new Parser(string);
    }

    private Parser(String string) {
        Elements elements = Jsoup.parse(string).select("*");
        for (Element element : elements) {
            if (element.tagName().equals("input")) {
                String name = element.attr("name");
                String value = element.attr("value");
                System.out.println(name + " -> " + value);
            }
        }
    }

    public static void main(String[] args) throws IOException {
        Parser parser = new Parser(Request.copy("test_key=test_value"));
        for (String tag : parser.toString().split(">")) {
            if (!tag.trim().startsWith("<") || !tag.trim().endsWith(">")) {
                throw new RuntimeException(tag);
            }
            System.out.println(tag.substring(1, tag.length() - 2));
        }
    }

    public static String toString(Parser parser) throws IOException {
        StringBuilder sb = new StringBuilder();
        for (String tag : parser.toString().split(">")) {
            if (!tag.trim().startsWith("<") || !tag.trim().endsWith(">")) {
                throw new RuntimeException(tag);
            }
            sb.append(tag.substring(1, tag.length() - 2)).append("\n");
        }
        return sb.toString();
    }
}

    public static CookieStore copyCookieStore(CookieStore cookieStore) {
        System.out.println("Copying cookie store:");
        for (String domain : cookieStore.getDomains()) {
            System.out.println(domain);
        }
        return new CookieStore(cookieStore.getDomains());
    }
}

class CookieStore {
    private List<String> domains;

    public CookieStore(List<String> domains) {
        this.domains = domains;
    }

    public String getDomain(String domain) {
        for (String d : domains) {
            if (d.equals(domain)) {
                return domain;
            }
        }
        throw new RuntimeException("Domain not found");
    }
}
