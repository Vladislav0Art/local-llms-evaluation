public class ArithmeticExpressionEvaluator {

    // Method to evaluate an arithmetic expression.
    public static Object evaluateArithmeticExpression(String expression) {
        try {
            // Parse the expression using a JavaScript parser.
            // This is a very basic implementation that only supports addition, subtraction,

            String[] tokens = splitTokens(expression);

            // Initialize variables to store the current number and operator.
            double currentNumber = 0;
            char operator = '+';

            // Iterate over each token in the expression.
            for (int i = 0; i < tokens.length; i++) {
                // Check if the token is a number.
                if (tokens[i].matches("\\d+")) {
                    // If it's a number, store it as the current number.
                    currentNumber = Double.parseDouble(tokens[i]);
                } else if (tokens[i] == '+') {
                    // If it's an operator, add the current number to the result and update 
                    // the operator for the next token.
                    currentNumber += getPreviousNumber(currentNumber);
                    operator = '+';
                } else if (tokens[i] == '-') {
                    // If it's an operator, subtract the current number from the result and 
                    // update the operator for the next token.
                    currentNumber -= getPreviousNumber(currentNumber);
                    operator = '-';
                } else if (tokens[i] == '*') {
                    // If it's an operator, multiply the current number by the result and 
                    // update the operator for the next token.
                    currentNumber *= getPreviousNumber(currentNumber);
                    operator = '*';
                } else if (tokens[i] == '/') {
                    // If it's an operator, divide the current number by the result and 
                    // update the operator for the next token.
                    currentNumber /= getPreviousNumber(currentNumber);
                    operator = '/';
                }
            }

            // Return the final result as a string.
            return Double.toString(getPreviousNumber(currentNumber));

        } catch (Exception e) {
            // If an error occurs, throw it with a meaningful message.
            throw new RuntimeException("Invalid arithmetic expression: " + e.getMessage());
        }
    }

    private static String[] splitTokens(String input) {
        // This is a very basic implementation that only supports simple token 
        // splitting (splitting at whitespace).
