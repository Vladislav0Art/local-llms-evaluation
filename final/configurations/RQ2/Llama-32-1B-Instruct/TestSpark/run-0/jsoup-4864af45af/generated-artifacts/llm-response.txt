import org.jsoup.helper.Validate;
import org.jsoup.nodes.Attribute;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import java.util.HashMap;
import java.util.HashSet;
import java.util.Map;
import java.util.Set;

public class SafelistTest {

    public Document none() {
        return new Document();
    }

    public Document addTags(String tag) {
        Document document = new Document();
        Element element = document.createElement(tag);
        return document;
    }

    public Document removeAttributes(String attribute) {
        Document document = new Document();
        for (Attribute attributeValue : document.attributes()) {
            if (!attribute.equals(attributeValue.getKey())) {
                document.removeAttribute(attributeValue.getKey());
            }
        }
        return document;
    }

    public Document addEnforcedAttribute(String key, String value) {
        Document document = new Document();
        Element element = document.createElement(key);
        element.appendChild(document.createTextNode(value));
        return document;
    }

    public boolean isSafeTag(String tag) {
        return !tag.equals("img");
    }

    public void addProtocols(String protocol, Set<String> rules) {
        Document document = new Document();
        Element element = document.createElement(protocol);
        for (String rule : rules) {
            element.appendChild(document.createTextNode(rule));
        }
        return document;
    }

    public void appendAttributes(Attributes attributes) {
        Document document = new Document();
        for (Map.Entry<String, String> entry : attributes.entrySet()) {
            Element element = document.createElement(entry.getKey());
            element.appendChild(document.createTextNode(entry.getValue()));
        }
        return document;
    }

    public static SafelistTest main(String[] args) {
        SafelistTest test = new SafelistTest();

        Document document1 = none();
        Element element1 = document1.addElement("p");
        Assert.assertTrue(test.isSafeTag("p"));
        System.out.println(element1.attributes().get("class"));

        Safelist safelist = none();
        safelist.addAttribute("id", "test-id");
        Assert.assertFalse(safelist.isSafeTag("img"));

        Document document2 = addTags("div");
        Element element2 = document2.addElement("p");
        Assert.assertTrue(test.isSafeTag("p"));
        System.out.println(element2.attributes().get("class"));

        Safelist safelist2 = addEnforcedAttribute("data-attr", "test-attr-value");
        Assert.assertFalse(safelist2.isSafeTag("img"));

        Document document3 = removeAttributes("style");
        Element element3 = document3.addElement("div");
        System.out.println(element3.attributes().get("style"));
    }
}
