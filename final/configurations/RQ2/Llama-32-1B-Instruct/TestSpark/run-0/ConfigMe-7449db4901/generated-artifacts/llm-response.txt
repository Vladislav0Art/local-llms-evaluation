Here's a simple implementation of the Config class in Java, which includes methods to add and remove configuration settings.

```java
public class Config {

    private static final String CONFIG_FILE = "config.txt";
    private static final int MAX_CONFIG_SIZE = 1000;

    public boolean addConfig(String configName, String value) {
        try (BufferedReader reader = new BufferedReader(new FileReader(CONFIG_FILE))) {
            StringBuilder builder = new StringBuilder();
            String line;
            while ((line = reader.readLine()) != null) {
                if (builder.length() < MAX_CONFIG_SIZE && !builder.toString().equals(line)) {
                    builder.append("\n").append(line);
                }
            }

            // Append the new config to the end of the file
            builder.append("\n").append(value);

            File file = new File(CONFIG_FILE);
            FileWriter writer = new FileWriter(file);
            writer.write(builder.toString());
            writer.close();

            return true;
        } catch (IOException e) {
            System.out.println("Error adding configuration: " + e.getMessage());
            return false;
        }
    }

    public boolean removeConfig(String configName) {
        try (BufferedReader reader = new BufferedReader(new FileReader(CONFIG_FILE))) {
            String line;
            while ((line = reader.readLine()) != null && !line.startsWith(configName + ".")) {
                // Find the config to be removed
                int pos = line.indexOf('.');
                if (pos == -1) {
                    System.out.println("Config not found: " + line);
                    return false;
                }

                String fileName = line.substring(0, pos);
                File file = new File(fileName);

                // Remove the config from the file
                try (BufferedWriter writer = new BufferedWriter(new FileWriter(file))) {
                    writer.write(line.substring(pos + 1));
                } catch (IOException e) {
                    System.out.println("Error removing configuration: " + e.getMessage());
                    return false;
                }
            }

            // If no config is found, delete the file
            if (!file.delete()) {
                System.out.println("File not found or cannot be deleted.");
                return false;
            }

            return true;
        } catch (IOException e) {
            System.out.println("Error removing configuration: " + e.getMessage());
            return false;
        }
    }
}
```

This code creates a simple text file named `config.txt` to store the configurations. The `addConfig()` method appends a new config to the end of the file, and the `removeConfig()` method reads all configurations from the file, deletes each one by name, or deletes the entire file if no config is found.

import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.FileReader;
import java.io.FileWriter;
import java.io.IOException;

/**
 * Configuration manager class.
 */
public class Config {

    /**
     * File name for storing configuration settings.
     */
    private static final String CONFIG_FILE = "config.txt";

    /**
     * Maximum size of the configuration file (in lines).
     */
    private static final int MAX_CONFIG_SIZE = 1000;

    /**
     * Adds a new configuration setting to the file.
     *
     * @param configName Configuration name for this setting.
     * @param value         Configuration value for this setting.
     * @return True if the configuration was added successfully, false otherwise.
     */
    public boolean addConfig(String configName, String value) {
        try (BufferedReader reader = new BufferedReader(new FileReader(CONFIG_FILE))) {
            StringBuilder builder = new StringBuilder();
            String line;
            while ((line = reader.readLine()) != null) {
                // Skip empty lines
                if (builder.length() < MAX_CONFIG_SIZE && !builder.toString().equals(line)) {
                    builder.append("\n").append(line);
                }
            }

            // Append the new config to the end of the file
            builder.append("\n").append(value);

            File file = new File(CONFIG_FILE);
            BufferedWriter writer = new BufferedWriter(new FileWriter(file));
            try {
                writer.write(builder.toString());
            } catch (IOException e) {
                System.out.println("Error writing configuration: " + e.getMessage());
                return false;
            }
            writer.close();

            return true;
        } catch (IOException e) {
            System.out.println("Error reading or writing configuration: " + e.getMessage());
            return false;
        }
    }

    /**
     * Removes a configuration setting from the file by name.
     *
     * @param configName Configuration name for this setting to remove.
     * @return True if the configuration was removed successfully, false otherwise.
     */
    public boolean removeConfig(String configName) {
        try (BufferedReader reader = new BufferedReader(new FileReader(CONFIG_FILE))) {
            String line;
            while ((line = reader.readLine()) != null && !line.startsWith(configName + ".")) {
                // Find the config to be removed
                int pos = line.indexOf('.');
                if (pos == -1) {
                    System.out.println("Config not found: " + line);
                    return false;
                }

                String fileName = line.substring(0, pos);
                File file = new File(fileName);

                // Remove the config from the file
                try (BufferedWriter writer = new BufferedWriter(new FileWriter(file))) {
                    writer.write(line.substring(pos + 1));
                } catch (IOException e) {
                    System.out.println("Error removing configuration: " + e.getMessage());
                    return false;
                }
            }

            // If no config is found, delete the file
            if (!file.delete()) {
                System.out.println("File not found or cannot be deleted.");
                return false;
            }

            return true;
        } catch (IOException e) {
            System.out.println("Error removing configuration: " + e.getMessage());
            return false;
        }
    }
}
