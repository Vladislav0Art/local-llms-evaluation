```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Main {
    public static Document generateXml(String xmlString) {
        return Jsoup.parse(xmlString, "", "jsoup");
    }

    public static void main(String[] args) {
        String xmlString = "<root><name>John</name><age>30</age></root>";
        Document document = generateXml(xmlString);

        Element root = document.body();
        Element nameElement = root.children().first();
        Element ageElement = root.children().last();

        System.out.println("Name: " + nameElement.text());
        System.out.println("Age: " + ageElement.text());

        String name = nameElement.text();
        String age = ageElement.text();

        if (!name.isEmpty() && !age.isEmpty()) {
            System.out.println(name + " is " + age + " years old.");
        } else {
            System.out.println("Name and age are not empty.");
        }
    }
}
```

```java
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        System.out.print("Enter your name: ");
        String name = scanner.nextLine();
        System.out.print("Enter your age: ");
        int age = scanner.nextInt();
        scanner.close();

        Document document = generateXml(name + " <" + age + ">");
        System.out.println(document.outerHtml());
    }

    public static Document generateXml(String xmlString) {
        return Jsoup.parse(xmlString, "", "jsoup");
    }
}
```

In this code:

- We first import the `Scanner` class to get user input.
- We then ask the user to enter their name and age.
- After getting the input from the user, we create a new XML document using the `generateXml` method.
- The XML string is created by concatenating the name and age with the `<` character (representing an element) followed by the age.
- Finally, we print out the generated XML.
