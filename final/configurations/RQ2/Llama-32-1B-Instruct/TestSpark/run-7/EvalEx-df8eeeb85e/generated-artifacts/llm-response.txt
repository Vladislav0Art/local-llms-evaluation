                lineNumber++;
                return true;
            }
            return true;

import java.util.*;

public class Main {

    // Define a custom Token type
    public static enum Token {
        VARIABLE,  // variable name
        NUMBER,     // number
        OPERATOR,   // operator (+, -, *, /)
        LPAREN,    // opening parenthesis
        RPAREN,    // closing parenthesis
        EOF         // end of file
    }

    private String token;
    private int lineNumber;

    public Token(String token) {
        this.token = token;
        this.lineNumber = 0; // Assume the first line is at index 1
    }

    // Define a custom method to tokenize input string
    public static List<Token> tokenizeInput(String inputStr) {
        String[] tokens = inputStr.split("[+*/-]"); // split by operators
        for (String token : tokens) {
            if (!token.isEmpty()) {  // Ignore empty strings
                return Arrays.asList(new Token(token), new Token("EOF"));
            }
        }
        return Arrays.asList(new Token(inputStr.substring(0, 1)), new Token(' ' + inputStr.length()));
    }

    // Define a custom method to get the next token
    private static boolean hasNextToken() {
        if (lineNumber == 1) {  // Assume the first line is at index 1
            return true;
        } else if (!tokenizerMock.hasNextToken()) {
            lineNumber--;  // If there are no more tokens, move back one
            return false;
        } else {
            Token token = tokenizerMock.nextToken();
            if (token == null) {  // If the end of a line is reached, stop looking
                lineNumber++;
                return true;
            }
            return true;
        }
    }

    // Define a custom method to parse tokens into an array of Token objects
    private static List<Token> parseTokens() {
        String inputStr = tokenizeInput(inputStr); // Parse the input string
        while (hasNextToken()) {
            if (!inputStr.substring(0, 1).equals("EOF")) { // If we are not at the end of file
                inputStr = inputStr.substring(0, 1) + " " + inputStr.substring(2); // Add a space before the token
            }
            List<Token> tokensList = parseTokens(); // Parse each token in the string
            String[] currentTokens = new String[tokensList.size()];
            int i = 0;
            for (Token token : tokensList) {
                if (!token.lineNumber.equals(inputStr.length())) {  // Don't add empty strings
                    currentTokens[i++] = token.toString();
                }
            }
            inputStr = Arrays.copyOfRange(inputStr, inputStr.lastIndexOf(" ") + 1, Math.min(inputStr.length(), i)); // Remove the tokens from the string
        }
        return currentTokens;
    }

    public static void main(String[] args) {
        String inputStr = "x+y*z";
        List<Token> result = parseTokens();

        System.out.println("VARIABLES:");  // Print the names of variables (only one in this case)
        for (Token token : result) {
            if (token instanceof Variable) {  // Check if a variable was encountered
                System.out.println(((Variable) token).getName());
            }
        }

        System.out.println("OPERATORS:");
        for (Token token : result) {
            if (token instanceof Operator) {  // Check if an operator was encountered
                System.out.println(((Operator) token));
            }
        }

        System.out.println("numbers:");   // Print the numbers in the expression
        for (int i = 0; i < result.length; i++) {
            Token token = result.get(i);
            if (token instanceof Number) {  // Check if a number was encountered
                System.out.print(token.toString());
            } else if (i == result.size() - 1 && ((Number) token).getValue() > 0) {  // Print the last number and operator
                System.out.println(token);
            }
        }

        System.out.println("OPERATORS after numbers:");  // Print all operators that follow a number in the expression
        for (Token token : result) {
            if (token instanceof Operator && i < result.length - 1 && ((Operator) token).getSymbol() != '+') {  // Check if an operator was encountered and it's not the last one
                System.out.print(token.toString());
            }
        }

        System.out.println("END OF FILE:");  // Print the end of file indicator
    }

    static class TokenizerMock {
        private int lineNumber;
        private String currentLine;

        public boolean hasNextToken() {
            return !currentLine.isEmpty();
        }

        public String nextToken() {
            if (hasNextToken()) {
                String token = currentLine.substring(0, 1);
                currentLine = currentLine.substring(1).trim(); // Remove the newline character
                return token;
            } else {
                return null;
            }
        }

        public void setLineNumber(int lineNumber) {
            this.lineNumber = lineNumber;
            this.currentLine = "";
        }
    }

    static class Variable {
        private String name;

        public Variable(String name) {
            this.name = name;
        }

        @Override
        public String toString() {
            return "VARIABLE" + System.identityHashCode(this);
        }

        public String getName() {
            return name;
        }

        public boolean equals(Object obj) {
            if (this == obj) {
                return true;
            }
            if (!(obj instanceof Variable)) {
                return false;
            }
            Variable other = (Variable) obj;
            return this.name.equals(other.name);
        }

        public int hashCode() {
            return System.identityHashCode(this);
        }
    }

        for (Token token : result) {
