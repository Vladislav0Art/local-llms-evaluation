public class TestUtils {
    public static void testWordWrap_FromReader() throws IOException {
        String input = "Hello World";
        String[] lines = {"This is a sample text", "Another line of text"};
        
        for (String line : lines) {
            System.out.println(wordWrap(line));
        }
    }

    private static String wordWrap(String text) {
        if (text.length() < 50) {
            return text;
        } else {
            int length = Math.min(50, text.length());
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < lines.length - 1; i++) {
                String line1 = lines[i].trim();
                if (!line1.isEmpty()) {
                    String line2 = lines[i + 1];
                    System.out.println(line1);
                    for (int j = 0; j <= Math.max(50, length - line1.length()); j++) {
                        sb.append(line1.substring(j));
                        if (j < line1.length()) {
                            sb.append(" ");
                        }
                    }
                    System.out.println(sb.toString());
                } else {
                    String trimmedLine2 = line2.trim();
                    for (int j = 0; j <= Math.max(50, length - text.length()); j++) {
                        sb.append(trimmedLine2.substring(j));
                        if (j < trimmedLine2.length()) {
                            sb.append(" ");
                        }
                    }
                    System.out.println(sb.toString());
                }
            }
            return text;
        }
    }
