import org.junit.Test;
import static org.mockito.Mockito.*;

public class TestConnection {
    private Connection connection = mock(Connection.class);
    private Request request = new Request();
    private Response response;

    @Test
    public void testGetRequest() {
        doNothing().when(connection).execute();

        verifyNoMoreInteractions(connection);

        when(request.execute()).thenReturn(response);
        when(connection.getResponse()).thenReturn(response);
        when(connection.getMessage()).thenReturn("ok");

        connection.setRequest(request);
        connection.execute();
        verify(response, never());
    }

    @Test
    public void testGetRequestWithException() {
        doThrow(new RuntimeException()).when(connection).execute();

        verifyNoMoreInteractions(connection);

        when(request.execute()).thenReturn(new Response("error", "message"));

        connection.setRequest(request);
        connection.execute();
        verify(response, never());
    }
}

import org.junit.Test;
import static org.mockito.Mockito.*;

public class TestConnection {
    private Connection connection = mock(Connection.class);
    private Request request;

    @Test
    public void testGetRequest() throws Exception {
        when(request.execute()).thenReturn(new Response("ok", "response"));
        
        when(connection.getResponse()).thenReturn(response);

        try {
            connection.setRequest(request);
            connection.execute();
            verifyNoMoreInteractions(connection);
        } catch (Exception e) {
            fail("testGetRequest failed");
        }
    }

    @Test
    public void testGetRequestWithException() throws Exception {
        doThrow(new RuntimeException()).when(connection).execute();

        when(request.execute()).thenReturn(response);

        try {
            connection.setRequest(request);
            connection.execute();
            verifyNoMoreInteractions(connection);
        } catch (Exception e) {
            fail("testGetRequestWithException failed");
        }
    }

    @Test
    public void testGetRequestWithResponse() throws Exception {
        Response response = new Response("ok", "response");

        doReturn(response).when(connection).execute();

        when(request.execute()).thenReturn(response);

        try {
            connection.setRequest(request);
            connection.execute();
            verifyNoMoreInteractions(connection);
        } catch (Exception e) {
            fail("testGetRequestWithResponse failed");
        }
    }

    @Test
    public void testGetRequestWithMultipleResponses() throws Exception {
        Response response1 = new Response("ok", "response1");
        Response response2 = new Response("ok", "response2");

        doReturn(response1).when(connection).execute();
        doReturn(response2).when(connection).execute();

        when(request.execute()).thenReturn(response1);
        try {
            connection.setRequest(request);
            connection.execute();
            verifyNoMoreInteractions(connection);
        } catch (Exception e) {
            fail("testGetRequestWithMultipleResponses failed");
        }
    }

    @Test
    public void testGetResponse() throws Exception {
        String expected = "ok, response1, response2";

        try {
            when(request.execute()).thenReturn(response);

            connection.setRequest(request);
            Connection.Response actual = (Connection.Response) request;
            assert(actual.getBody().equals(expected));
        } catch (Exception e) {
            fail("testGetResponse failed");
        }
    }

    @Test
    public void testGetMultipleResponses() throws Exception {
        String expected = "ok, response1, response2";

        try {
            when(request.execute()).thenReturn(response);

            connection.setRequest(request);
            Connection.Response actual = (Connection.Response) request;
            assert(actual.getBody().equals(expected));
        } catch (Exception e) {
            fail("testGetMultipleResponses failed");
        }
    }

    @Test
    public void testExecute() throws Exception {
        try {
            when(request.execute()).thenReturn(null);

            connection.setRequest(request);
            connection.execute();
            assert(connection.isClosed());
        } catch (Exception e) {
            fail("testExecute failed");
        }
    }
}
