import org.jsoup.Connection;
import org.jsoup.Javascript;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class URLBuilderTest {

    private Document doc;

    @Before
    public void setup() {
        Connection connection = ConnectionBuilder.newConnection();
        connection.connect("https://example.com");
        doc = connection.build().document();
    }

    @Test
    public String testBuild() throws Exception {
        return urlBuilder.toString();
    }

    @Test
    public void testAppendKeyVal() throws Exception {
        URLBuilder urlBuilder = new URLBuilder(doc);
        KeyVal keyVal = new KeyVal(doc.selectFirst("input[name='key1'][value='value1']"), "value2");
        System.out.println(urlBuilder.appendKeyVal(keyVal));
    }

    @Test
    public void testBefore() {
        System.out.println("Before called.");
    }

    @Test
    public void testAfter() {
        System.out.println("After called.");
    }
}

class KeyVal {
    private Element inputElement;
    private String value;

    public KeyVal(Element inputElement, String value) {
        this.inputElement = inputElement;
        this.value = value;
    }

    @Override
    public String toString() {
        return "KeyVal{" +
                "inputElement=" + inputElement.toString() +
                ", value='" + value + '\'' +
                '}';
    }
}

class URLBuilder {
    private Document doc;

    public URLBuilder(Document doc) {
        this.doc = doc;
    }

    @Override
    public String toString() {
        return "URLBuilder{" +
                "doc=" + doc +
                '}';
    }

    public void appendKeyVal(KeyVal keyVal) throws Exception {
        JavaScript script = doc.createElement("script");
        document.write("<" + script.tagName() + " id=\"key1\" value=\"" + keyVal.value + "\">");
        inputElement = doc.selectFirst("input[name='key1']");
        inputElement.value = keyVal.value;
        System.out.println(doc.body().text());
    }
}
