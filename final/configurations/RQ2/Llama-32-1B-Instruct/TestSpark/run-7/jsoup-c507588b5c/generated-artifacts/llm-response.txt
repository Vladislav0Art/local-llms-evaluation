```java
public class Main {
    public static String normalizeWhitespace(String text) {
        // Split the input string into lines and strip leading/trailing whitespaces from each line
        String[] lines = text.split("\n");
        StringBuilder normalizedText = new StringBuilder();
        
        for (String line : lines) {
            if (!line.trim().isEmpty()) {
                normalizedText.append(line).append("\n");
            }
        }
        
        // Remove the trailing newline character
        if (normalizedText.length() > 0) {
            normalizedText.deleteCharAt(normalizedText.length() - 1);
        }
        
        return normalizedText.toString();
    }

    public static void main(String[] args) {
        testNormalizeWhitespace();
        testNormaliseWhitespace();
        testStripLeadingWhitespace();
        testLastCharIsWhitespace();
        testExpectedNormalizedText();
        testExpectedStrippedText();
    }

    private static void testNormalizeWhitespace() {
        String text = "   Hello World!   ";
        System.out.println(normalizeWhitespace(text));
    }

    private static void testNormaliseWhitespace() {
        String text = "\n  Hello\nWorld!";

    }

    private static void testStripLeadingWhitespace() {
```
