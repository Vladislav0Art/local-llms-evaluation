import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class GeneratedTest {

    private static final String ATTR_NAME = "classAttr";
    private static final String CLASS_VALUE = "new-class";

    public void testGetUpdatedClassAttribute() {
        Document doc = new Document();
        Elements nodes = doc.getAllElements();

        Element attrElement = nodes.first()
                .attr(ATTR_NAME)
                .replaceAll("\"\\*.*?\"", "");

        String updatedClassAttribute = getUpdatedClassAttribute(attrElement);

        assertEquals(CLASS_VALUE, updatedClassAttribute);
    }

    public void testRemoveAttr() {
        Document doc = new Document();
        Elements nodes = doc.getAllElements();

        Element attrElement = nodes.first()
                .attr(ATTR_NAME)
                .replaceAll("\"\\*.*?\"", "");

        String updatedAttrValue = removeAttr(attrElement);

        assertEquals("", updatedAttrValue);
    }

    public void testGetNewClassAttribute() {
        Document doc = new Document();
        Elements nodes = doc.getAllElements();

        Element attrElement = nodes.first()
                .attr(ATTR_NAME)
                .replaceAll("\"\\*.*?\"", "");

        String newValueValue = getNewClassAttribute(attrElement, null);

        assertEquals(CLASS_VALUE, newValueValue);
    }

    public void testRemoveClass() {
        Document doc = new Document();
        Elements nodes = doc.getAllElements();

        Element attrElement = nodes.first()
                .attr(ATTR_NAME)
                .replaceAll("\"\\*.*?\"", "");

        String classAttr = attrElement.attr(ATTR_NAME);

        updatedClassAttr = removeClass(classAttr, null);
        assertEquals("node1", updatedClassAttr);

        // test shallow clone
        Element node1 = newElement().text(CLASS_VALUE).attr(ATTR_NAME);
        Elements nodes1 = doc.getAllElements();
        Element attrElement1 = nodes1.first()
                .attr(ATTR_NAME)
                .replaceAll("\"\\*.*?\"", "");

        String updatedClassAttr2 = removeClass(classAttr, null);
        assertEquals("node1", updatedClassAttr2);
    }

    public void testGetShallowClone() {
        Document doc = new Document();
        Elements nodes = doc.getAllElements();

        Element attrElement = nodes.first()
                .attr(ATTR_NAME)
                .replaceAll("\"\\*.*?\"", "");

        String classAttr = attrElement.attr(ATTR_NAME);

        Element node1 = newElement().text(CLASS_VALUE).attr(ATTR_NAME);
        Element node2 = node1.shallowClone(node1);

        assertEquals("node1", node2.text());
    }

    public static void main(String[] args) {
        GeneratedTest test = new GeneratedTest();
        test.testGetUpdatedClassAttribute();
        test.testRemoveAttr();
        test.testGetNewClassAttribute();
        test.testRemoveClass();
        test.testGetShallowClone();
    }
}
