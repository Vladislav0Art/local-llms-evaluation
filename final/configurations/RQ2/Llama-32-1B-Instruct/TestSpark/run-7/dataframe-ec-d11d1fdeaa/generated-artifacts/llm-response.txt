class Column {
    private List<String> columns;

        this.columns = columns;
    }

    public List<String> getColumns() {
        return columns;
    }

    @Override
    public String toString() {
        StringBuilder builder = new StringBuilder();
        for (String column : columns) {
            builder.append(column).append(",");
        }
        return builder.toString().trim();
    }

    public boolean contains(String value) {
        for (String column : columns) {
            if (column.contains(value)) {
                return true;
            }
        }
        return false;
    }

    public Column renameColumn(String oldColumnName, String newColumnName) {
        List<String> columnNames = new ArrayList<>(columns);
        List<String> newColumns = new ArrayList<>(columns);
        for (int i = 0; i < columns.length; i++) {
            if (columnNames.get(i).contains(oldColumnName)) {
                newColumns.set(i, newColumnName + "_" + oldColumnName);
            }
        }
        return new Column(newColumns);
    }

    public String[] toTable() {
        List<String> columnNames = new ArrayList<>(columns);
        StringBuilder builder = new StringBuilder();
        for (String columnName : columnNames) {
            builder.append(columnName).append(",");
        }
        return builder.toString().trim().split(",");
    }

    public int hashCode() {
        int result = 0;
        for (String column : columns) {
            result = result * 31 + column.hashCode();
        }
        return result;
    }

    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;

        Column column = (Column) o;

    }

class Scanner {
    private String[] tokens;
    private int index;

    public Scanner(String input) {
        this.tokens = input.split(" ");
        this.index = 0;
    }

    public String nextToken() {
        if (this.index >= tokens.length) {
            return null;
        }
        return tokens[this.index++];
    }
}

        // Create a Spark session
        SparkSession spark = SparkSession.builder()
                .appName("Drop Columns from DataFrame")
                .getOrCreate();

        // Create a sample DataFrame

        dropColumnsDF.show();
    }

    public static Dataset<Row> dropColumns(Dataset<Row> df, String[] columnNames) {
        // Check if the column names exist in the DataFrame
        for (String columnName : columnNames) {
            boolean found = false;
            for (String[] columns : df.columns) {
                if (ArrayUtils.contains(columns, columnName)) {
                    found = true;
                    break;
                }
            }

            if (!found) {
                throw new RuntimeException("The specified column(s) '" + columnName + "' does not exist in the DataFrame.");
            }
        }

        // Return a new DataFrame with the specified columns removed
        return df.select(columnNames.length - 1);
    }
