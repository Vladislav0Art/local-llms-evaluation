package org.jsoup.helper;

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Selector;
import org.jsoup.util.Parser;
import org.junit.Test;
import static org.junit.Assert.*;
import java.io.StringWriter;

public class W3CDomTest {

    private Document doc = parseString("html");

    @Test
    public void namespaceAware() {
        W3CDom w3cdom = new W3CDom();
        assertNotNull(w3cdom.namespaceAware());
        assertEquals(true, w3cdom.namespaceAware(true));
        assertEquals(false, w3cdom.namespaceAware(false));
    }

    @Test
    public void convert() {
        Document doc1 = parseString("<html><body>Hello World!</body></html>");
        Document doc2 = new W3CDom().convert(doc1);
        assertNotNull(doc2);
        assertTrue(doc1.equals(doc2));
    }

    @Test
    public void fromJsoup() {
        Element el = doc.select("body").first();
        W3CDom w3cdom = new W3CDom();
        Document result = w3cdom.fromJsoup(el);
        assertNotNull(result);
        assertEquals(el, result.getOwnerDocument());
    }

    @Test
    public void selectXpath() {
        Selector selector = doc.select("body");
        NodeList nodeList = selector.select(":nth-child(1)");
        assertNotNull(nodeList);
        assertTrue(nodeList.getLength() == 1);
        assertTrue((Integer) nodeList.item(0).getNodeName().equals("h1"));
    }

    @Test
    public void sourceNodes() {
        NodeList nodeList = doc.sourceNodes(doc.select("p"), org.jsoup.nodes.Node.class);
        assertNotNull(nodeList);
        assertEquals(4, nodeList.getLength());
        assertEquals(org.jsoup.nodes.Text.class, nodeList.item(0).getClass());
    }

    @Test
    public void asString() {
        Document doc = new W3CDom().fromJsoup(doc.select("body").first());
        String result = doc.asString();
        assertNotNull(result);
        assertTrue(result.contains("Hello World!"));
    }

    private Document parseString(String html) {
        return org.jsoup.parser.Parser.parse(html).getDocument();
    }
}

public class TestNSpace {
    public static String parse(String html) {
        W3CDom doc = parse(html);
        return doc.body().first().tagName();
    }

    private static W3CDom parse(String html) {
        Document doc;
        try {
            doc = new DOMParser().parseFromString(html, "text/html");
        } catch (Exception e) {
            System.out.println("Error parsing HTML: " + e.getMessage());
            return null;
        }
        return doc;
    }

    private static void testParseSpace() {
        String html = "<html><body>Hello &nbsp; world!</body></html>";
        String result = parse(html);
        assertEquals("&nbsp;", result);
    }

    public static void main(String[] args) {
        testParseSpace();
    }
}
