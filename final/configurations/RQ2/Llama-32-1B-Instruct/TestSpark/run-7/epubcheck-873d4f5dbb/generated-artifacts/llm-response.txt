            String[] lines = Files.readAllLines(path);

            for (String line : lines) {
                String[] parts = line.split(",");
                if (parts.length == 2 && parts[0].trim().equals("video") && parts[1].trim().equals("mp4")) {
                    System.out.println(parts[1]);
