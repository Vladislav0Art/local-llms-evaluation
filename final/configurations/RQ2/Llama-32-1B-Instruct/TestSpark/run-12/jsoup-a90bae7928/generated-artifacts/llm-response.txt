        assertThat(actualHtml, is(expectedHtml));

    public void testEmptyString() {

    }

    public void testSpacesInText() {

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertTrue;

public class Main {
    public static void main(String[] args) {
        String html = "<html><body>Hello World!</body></html>";
        assertTrue(isValidHtml(html));
    }

    private static boolean isValidHtml(String htmlString) {
        // Test 1
        String[] tests = {"<html><body>Hello World!</body></html>", "<html><p>Hello <b>World</b></p>", "<html><body><a href='https://www.example.com'>Example Website</a></body></html>"};
        for (String test : tests) {
            try {
                if (!isValidHtml(test)) {
                    System.out.println("Test 1 failed: " + test);
                    return false;
                }
            } catch (Exception e) {
                System.out.println(e.getMessage());
                return false;
            }
        }

        // Test 2
        String[] tests = {"<html><body>Hello <b>World</b></p>", "<html><body><a href='https://www.example.com'>Example Website</a></body></html>"};
        for (String test : tests) {
            try {
                if (!isValidHtml(test)) {
                    System.out.println("Test 2 failed: " + test);
                    return false;
                }
            } catch (Exception e) {
                System.out.println(e.getMessage());
                return false;
            }
        }

        // Test 3
        String[] tests = {"<html><body>Hello World! <a href='https://www.example.com'>Example Website</a></body></html>", "<html><body><a href='https://www.example.com'>Example Website</a> <b>Hello</b> World!"}</p></body></html>"};
        for (String test : tests) {
            try {
                if (!isValidHtml(test)) {
                    System.out.println("Test 3 failed: " + test);
                    return false;
                }
            } catch (Exception e) {
                System.out.println(e.getMessage());
                return false;
            }
        }

        // Test 4
        String[] tests = {"<html><body>Hello <b>World</b></p>", "<html><body><a href='https://www.example.com'>Example Website</a> <a href='https://www.example2.com'>Another Example Website</a></body></html>"};
        for (String test : tests) {
            try {
                if (!isValidHtml(test)) {
                    System.out.println("Test 4 failed: " + test);
                    return false;
                }
            } catch (Exception e) {
                System.out.println(e.getMessage());
                return false;
            }
        }

        // Test 5
        String[] tests = {"<html><body>Hello <b>World</b></p>", "<html><body><a href='https://www.example.com'>Example Website</a> <a href='https://www.example2.com' class='good-link'>Good Link</a></body></html>"};
        for (String test : tests) {
            try {
                if (!isValidHtml(test)) {
                    System.out.println("Test 5 failed: " + test);
                    return false;
                }
            } catch (Exception e) {
                System.out.println(e.getMessage());
                return false;
            }
        }

        return true;
    }

    private static boolean isInvalidHtml(String htmlString) throws Exception {
        try {
            new Main(htmlString);
            return false;
        } catch (Exception e) {
            return true;
        }
    }
}
