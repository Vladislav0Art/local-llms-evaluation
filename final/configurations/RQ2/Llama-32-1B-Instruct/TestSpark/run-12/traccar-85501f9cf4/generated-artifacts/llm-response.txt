import java.util.Random;

public class CellTower {
    private int numCells;
    private byte[] cellTables;
    private Random rand = new Random();

    public CellTower(int numCells) {
        this.numCells = numCells;
        this.cellTables = new byte[numCells * 2];
        generateCellTable();
    }

    private void generateCellTable() {
        for (int i = 0; i < numCells; i++) {
            int centerCell = rand.nextInt(numCells);
            cellTables[i] = (byte) Math.random();
            for (int j = 0; j < 2; j++) {
                if (j == 0) {
                    byte[] temp = new byte[2];
                    temp[0] = cellTables[i];
                    cellTables[i] = (byte) centerCell;
                    cellTables[i + 1] = temp[0];
                } else {
                    byte[] temp = new byte[2];
                    temp[0] = centerCell;
                    cellTables[i] = (byte) Math.random();
                    cellTables[i + 1] = temp[0];
                }
            }
        }
    }

    public int getNumCells() {
        return numCells;
    }

    public boolean checkChecksum(int start, int end) {
        if (end - start < 16) return false;
        int checksum = 0;
        for (int i = start; i <= end; i++) {
            byte[] buf = new byte[2];
            buf[0] = cellTables[i];
            checksum += buf[0];
            checksum += buf[1];
        }
        return (checksum & 0x8000) == 0x8000;
    }

    public static void main(String[] args) {
        CellTower cellTower = new CellTower(100);

        // simulate sending and receiving packets
        for (int i = 0; i < 10; i++) {
            int start = rand.nextInt(cellTower.getNumCells());
            int end = Math.min(start + 15, cellTower.getNumCells() - 1);
            boolean result = cellTower.checkChecksum(start, end);
            System.out.println("Result " + (i + 1) + ": " + result);
        }
    }
}
