import ch.jalu.configme.SettingsHolder;

import java.util.Arrays;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class CommentsConfigurationTest {

    private CommentsConfiguration commentsConfiguration;
    private Map<String, List<String>> comments;
    private String path;

    @BeforeEach
    public void init() {
        comments = new HashMap<>();
        comments.put("path", Arrays.asList("", "comment1", "", "comment2"));
        commentsConfiguration = new CommentsConfiguration();
        path = "path";
    }

    @Test
    public void setComment_SingleNewLine_NoComment() {
        String[] commentLines = {""};
        commentsConfiguration.setComment(path, commentLines);
        assertEquals(Collections.emptyList(), comments.get(path));
    }

    @Test
    public void setComment_Newline_Comment1() {
        List<String> commentLines = Collections.singletonList("comment1");
        commentsConfiguration.setComment(path, commentLines);
        assertEquals(Arrays.asList("", "comment1", ""), comments.get(path));
    }

    @Test
    public void setComment_Newline_NoComment2() {
        String[] commentLines = {""};
        commentsConfiguration.setComment(path, commentLines);
        assertEquals(Collections.emptyList(), comments.get(path));
    }

    public void getAllComments() {

}
