import org.junit.jupiter.api.Test;
import static org.assertj.core.api.Assertions.assertThat;

public class UrlBuilderTest {

    @ParameterizedTest
    @CsvSource({
            "\"https://example.com/path1\"",
            "\"https%3A//example.com/path2?param=123&anotherParam=456\""
    })
    void testBuildUrl(String inputUrl) {
        try {
            UrlBuilder builder = new UrlBuilder(new URI(inputUrl));
            assertThat(builder.build().toASCIIString()).isEqualTo("https://example.com/path1");
        } catch (Exception e) {
            fail(e.getMessage());
        }
    }

    public class UrlBuilder {
        private URI uri;

        public UrlBuilder(URI uri) {
            this.uri = uri;
        }

        @Test
        void testAppendKeyVal() throws Exception {
            try {
                String key = "param1";
                String value = "123";
                appendKeyVal(key, value);
                assertThat(uri.toString()).isEqualTo("param1=123");
            } catch (Exception e) {
                fail(e.getMessage());
            }
        }

        public void appendKeyVal(String key, String value) throws Exception {
            appendKeyVal(Connection.KeyVal.of(key, value));
        }

        @Test
        void testUrlEncode() throws Exception {
            try {
                UrlBuilder builder = new UrlBuilder("https://example.com/path?param=value");
                assertThat(builder.build().toASCIIString()).isEqualTo("https%3A//example.com/path?param=value");
            } catch (Exception e) {
                fail(e.getMessage());
            }
        }

        @Test
        void testMalformedURLException() throws Exception {
            try {
                UrlBuilder builder = new UrlBuilder(new URI("https://example.com/path"));
                assertThat(builder.build().toASCIIString()).isEqualTo("https%3A//example.com/path");
            } catch (Exception e) {
                fail(e.getMessage());
            }
