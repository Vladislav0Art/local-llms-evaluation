class DirectiveJavaAnnotationUtil {
    private static final Set<GraphQLDirective> DIRECTIVES = new HashSet<>();

    public static void addDirectiveToField(GraphQLArgument arg) {
        if (arg.getArguments().size() < 1) {
            throw new RuntimeException("Directives must have a single argument");
        }

        for (String directive : Directives) {
            GraphQLDirective[] directivesArray = DIRECTIVES.toArray(new GraphQLDirective[0]);
            int index = findIndex(directivesArray, directive);

            if (index == -1) {
               DIRECTIVES.add(directive);
            }
        }

        GraphQLDirective[] directives = DIRECTIVES.toArray(new GraphQLDirective[0]);
        int index = findIndex(directives, arg.getArguments()[0]);

        if (index != -1 && directives[index].getArgs().size() < 2) {
            throw new RuntimeException("Directives must have exactly two arguments");
        }
    }

    public static int findIndex(Object[] array, Object target) {
        for (int i = 0; i < array.length; i++) {
            if (array[i] instanceof GraphQLDirective && ((GraphQLDirective[]) array[i]).length == 1 && ((GraphQLDirective[]) array[i])[0].getArgs().size() == 2) {
                return i;
            }
        }
        return -1;
    }
}
