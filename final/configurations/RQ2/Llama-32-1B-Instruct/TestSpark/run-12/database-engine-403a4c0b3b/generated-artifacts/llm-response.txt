```java
public class Database {
    private String name;
    private String[][] data;

    public Database(String name) {
        this.name = name;
        this.data = new String[0][0];
    }

    public void addData(String[] row) {
        if (row.length != 2) {
            throw new IllegalArgumentException("Invalid input. Expected two columns.");
        }
        String[] column1 = row[0].split(",");
        String[] column2 = row[1].split(",");

        data = append(data, name, column1);
    }

    public void printData() {
        System.out.println(name + " has the following rows:");
        for (String[] row : data) {
            if (row.length < 2) {
                throw new IllegalArgumentException("Invalid input. Expected at least two columns.");
            }
            String[] columns = row;
            System.out.println(columns[0] + ", " + columns[1]);
        }
    }

    private void append(String[][] data, String name, String[] column1) {
        for (String[] row : data) {
            if (!row[0].equals(column1[0]) || !row[1].equals(column1[1])) {
                System.out.println("Invalid input. The first and second columns in the row must match.");
                return;
            }
            String[][] newRow = new String[data.length + 1][2];
            System.arraycopy(row, 0, newRow, 0, column1.length);
            newRow[newRow.length - 1] = new String[]{name, column1[0], column1[1]};
            data = append(data, name, newRow);
        }
    }

    public static void main(String[] args) {
        Database db1 = new Database("db1");
        db1.addData(new String[]{"A", "B"});
        db1.addData(new String[]{"C", "D"});

        Database db2 = new Database("db2");
        db2.addData(new String[]{"X", "Y"});
    }
}
```

```java
import java.util.ArrayList;
import java.util.List;

public class Database {
    private String name;
    private List<List<String[]>> data;

    public Database(String name) {
        this.name = name;
        this.data = new ArrayList<>();
    }

    public void addData(List<List<String>> row) {
        if (row.size() != 2 || row.get(0).size() != 2) {
            throw new IllegalArgumentException("Invalid input. Expected two columns.");
        }
        List<String[]> column1 = row.get(0);
        List<String> column2 = row.get(1);

        data.add(new ArrayList<>(column1));
    }

    public void printData() {
        System.out.println(name + " has the following rows:");
        for (List<List<String[]>> row : data) {
            if (row.size() < 2) {
                throw new IllegalArgumentException("Invalid input. Expected at least two columns.");
            }
            String[][] columns = row.get(0);
            StringBuilder sb = new StringBuilder();
            for (String[] column : columns) {
                sb.append(column[0] + ", ");
                for (String element : column) {
                    sb.append(element).append(",");
                }
                System.out.println(sb.toString().trim());
            }
        }
    }

    public static void main(String[] args) {
        Database db1 = new Database("db1");
        db1.addData(List.of(List.of("A", "B"), List.of("C", "D")));

        Database db2 = new Database("db2");
        db2.addData(List.of(List.of("X", "Y")));
    }
}
```

This implementation creates a `Database` class with methods to add rows and print the data. The database is represented as a list of lists, where each inner list contains two elements (columns) and one element (row).
