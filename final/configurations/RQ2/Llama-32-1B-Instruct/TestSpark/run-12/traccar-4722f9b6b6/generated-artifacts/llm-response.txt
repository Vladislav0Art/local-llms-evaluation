public class HttpProtocol implements Protocol {
    private HttpRequest request;
    private Map<String, String> headers = new HashMap<>();

    public void addHeader(String header, String value) {
        this.headers.put(header.toLowerCase(), value);
    }

    @Override
    public byte[] decode(byte[] data) {
        // For simplicity, this example just adds a 'Content-Type' header
        String headersString = "HTTP/1.1 200 OK\r\n";
        if (data.length > 0 && data[0] == (byte) 0x00) { // 'Content-Type'
            System.arraycopy(headersString.getBytes(), 0, request.headers, 0, headersString.length());
        }
        return data;
    }

    public String getMethod() {
        return request.method;
    }

    public String getUri() {
        return request.uri;
    }

    @Override
    public int getHeaderCount() {
        return request.headers.size();
    }

    @Override
    public String getBody() {
        StringBuilder body = new StringBuilder();
        for (Map.Entry<String, String> entry : headers.entrySet()) {
            body.append(entry.getKey()).append(": ").append(entry.getValue()).append("\n");
        }
        return body.toString().trim();
    }
}

public class HttpRequest {
    private String method;
    private String uri;

    public void setMethod(String method) {
        this.method = method.toUpperCase();
    }

    public void setUri(String uri) {
        this.uri = uri;
    }

    public byte[] getHeaders() {
        return request.headers.stream()
                .mapToObj(Map.Entry::getKey)
                .map(value -> value.getBytes())
                .toArray(bytes -> (byte[]) bytes);
    }
}
