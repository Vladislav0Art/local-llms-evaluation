import static org.junit.jupiter.api.Assertions.*;

public class TextNodeTest {

        assertEquals("TEXT_NODE", new TextNode(null).nodeName());
    }

    @Test
    public void testText() {
        assertEquals("", new TextNode("").text());
        assertEquals("Hello World!", new TextNode("Hello\nWorld!").text());
        assertEquals("This is a line of text.", new TextNode("This is a line of \n\ntext.").text());
    }

    @Test
    public void testGetWholeText() {
        assertEquals("", new TextNode("").getWholeText());
        assertEquals("Hello World!", new TextNode("Hello\nWorld!").getWholeText());
        assertEquals("This is a line of text.", new TextNode("This is a line of \n\ntext.").getWholeText());
    }

    @Test
    public void testIsBlank() {
        assertTrue(new TextNode("").isBlank());
        assertFalse(new TextNode(new String[]{}).isBlank());
        assertFalse(new TextNode(null).isBlank());
    }

    @Test
    public void testSplitText() {
        assertEquals("", new TextNode("").splitText(0));
        assertEquals("Hello World!", new TextNode("Hello\nWorld!").splitText(1));
        assertEquals("", new TextNode("This is a line of text.").splitText(-1));
    }

    @Test
    public void testOuterHtmlHead() {
        String expected = "<head></head>";
        Assert.assertEquals(expected, new TextNode("").outerHtmlHead(new Appendable[]{}, 0, Document.OutputSettings.DEFAULT));

        assertEquals(expected, new TextNode(null).outerHtmlHead(new Appendable[]{}, 0, Document.OutputSettings.DEFAULT));
    }

    @Test
    public void testOuterHtmlTail() {
        String expected = "</head>";
        Assert.assertEquals(expected, new TextNode("").outerHtmlTail(new Appendable[]{}, 0, Document.OutputSettings.DEFAULT));
        assertEquals(expected, new TextNode(new String[]{}).outerHtmlTail(new Appendable[]{}, 0, Document.OutputSettings.DEFAULT));
        assertEquals(expected, new TextNode(null).outerHtmlTail(new Appendable[]{}, 0, Document.OutputSettings.DEFAULT));
    }

    @Test
    public void testToString() {
        assertEquals("<TEXT_NODE>", new TextNode("").toString());
        assertEquals("TEXT_NODE", new TextNode(new String[]{}).toString());
        assertEquals("<TEXT_NODE>", new TextNode(null).toString());
    }

    @Test
    public void testClone() {
