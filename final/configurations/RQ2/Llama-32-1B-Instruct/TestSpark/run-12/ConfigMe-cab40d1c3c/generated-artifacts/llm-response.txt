import ch.jalu.configme.configurationdata.PropertyListBuilder;
import ch.jalu.configme.exception.ConfigMeException;
import ch.jalu.configme.properties.Property;
import org.jetbrains.annotations.NotNull;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

public class PropertyListBuilderImpl {
    private Map<String, Object> propertyList = new LinkedHashMap<>();

    public void add(Property<?> property) throws ConfigMeException {
        if (!propertyList.containsKey(property.getName())) {
            propertyList.put(property.getName(), null);
        }
    }

    public List<Property<?>> create() {
        return new ArrayList<>(propertyList.values());
    }

    protected @NotNull Map<String, Object> getRootEntries() {
        return propertyList;
    }
}

public class PropertyListBuilderTest {
    private PropertyListBuilderImpl propertyListBuilder;

    public void setUp() {
        propertyListBuilder = new PropertyListBuilderImpl();
    }

    public void testAdd() throws ConfigMeException {
        @MockProperty("test1")
        Property<?> mockProperty1 = createMockProperty();

        @MockProperty("test2")
        Property<?> mockProperty2 = createMockProperty();

        @InjectMocks
        PropertyListBuilderImpl mockPropertyListBuilder;

        propertyListBuilder.add(mockProperty1);
        propertyListBuilder.add(mockProperty2);

        List<Property<?>> list = mockPropertyListBuilder.create();
        assertEquals(3, list.size());
        assertEquals(mockProperty1.getName(), list.get(0).getName());
        assertEquals(mockProperty2.getName(), list.get(1).getName());
    }

    public void testCreate() throws ConfigMeException {
        @MockProperty("test1")
        Property<?> mockProperty1 = createMockProperty();

        @InjectMocks

        List<Property<?>> list = mockPropertyListBuilder.create();
        assertEquals(3, list.size());

    @Override
