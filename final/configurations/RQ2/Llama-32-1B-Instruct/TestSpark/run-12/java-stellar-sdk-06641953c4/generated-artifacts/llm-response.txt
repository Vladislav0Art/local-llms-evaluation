import java.util.Random;

public class Bip39Seed {
    private static final String BIP39_SEEDS = "bc1ck5zrptf3qg8v2p9j7d0twkty5ygv8wftu0m9e7x4f6tpa8ynh8psb8d";

    public static String generate(String seed, int length) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < length; i++) {
            char c;
            do {
                c = (char) (Math.random() * 95 + 32); // Random ASCII character
            } while (BIP39_SEEDS.indexOf(seed.substring(i, i + 1)) != -1);
            sb.append(c);
        }
        return sb.toString();
    }

    public static boolean validate(String seed, String input) {
        Random random = new Random();
        StringBuilder sb = new StringBuilder(input);
        for (int i = 0; i < seed.length(); i++) {
            int index = random.nextInt(seed.length());
            char c1 = Character.toUpperCase(sb.charAt(index));
            sb.setCharAt(index, Character.toLowerCase(c1));
        }
        return sb.toString().equals(input);
    }

    public static String[] generateBip39Seeds(int length) {
        StringBuilder[] seeds = new StringBuilder[length];
        for (int i = 0; i < length; i++) {
            seeds[i] = generate(BIP39_SEEDS.substring(i, i + 8), 8);
        }
        return seeds;
    }

    public static void main(String[] args) {
        String seed = "bc1ck5zrptf3qg8v2p9j7d0twkty5ygv8wftu0m9e7x4f6tpa8ynh8psb8d";
        int length = 20;
        String[] seeds = generateBip39Seeds(length);
        for (String seed : seeds) {
            if (!validate(seed, seed)) {
                System.out.println("Invalid seed: " + seed);
                break;
            }
        }
    }
}

    public static void main(String[] args) {
        int length = 32;
        String[] seeds = generateBip39Seeds(length);
        
        for (String seed : seeds) {
            if (!validate(seed, seed)) {
                System.out.println("Invalid seed: " + seed);
                break;
            }
        }

        // Print generated seeds
        for (String seed : seeds) {
            System.out.println(seed);
        }
    }

    private static String generateBip39Seeds(int length) {
        StringBuilder[] seeds = new StringBuilder[length];
        Random random = new Random();

        }
        
        return seeds;
    }

    private static String generate(String seed) {
        StringBuilder sb = new StringBuilder(seed);
        for (int i = 0; i < 32 - seed.length(); i++) {
            char c1 = Character.toUpperCase(sb.charAt(i));
            sb.setCharAt(i, Character.toLowerCase(c1));
        }
        return sb.toString();
    }

    private static boolean validate(String seed, String input) {

        StringBuilder sb = new StringBuilder(input);

            int index = random.nextInt(seed.length());

        }

    }

import java.util.Random;

public class Bip39Seed {
    public static void main(String[] args) {
        // Define the length of the generated seeds
        int length = 32;
        
        // Generate the seeds
        String[] seeds = generateBip39Seeds(length);
        
        System.out.println("Generated Seeds:");

        }
        
        // Print the validated seeds
        System.out.println("\nValidated Seeds:");
        
        // Iterate through each seed and validate it
        for (String seed : seeds) {
            if (!validate(seed, seed)) {
                System.out.println("Invalid seed: " + seed);
                break;
            }
        }

        // Generate a new list of seeds with different length
        seeds = generateBip39Seeds(length * 2);
        
        System.out.println("\nNew List of Seeds with Different Length:");
        for (String seed : seeds) {
            System.out.println(seed);
        }
    }

    private static String[] generateBip39Seeds(int length) {
        // Initialize an empty array to store the generated seeds
        StringBuilder[] seeds = new StringBuilder[length];
        
        // Generate each seed
        for (int i = 0; i < length; i++) {
            // Use a 8-character long binary string as the seed
            String seed = generate("bc1ck5zrptf3qg8v2p9j7d0twkty5ygv8wftu0m9e7x4f6tpa8ynh8psb8d");
            
            // Add it to the array
            seeds[i] = seed;
        }
        
        return seeds;
    }

    private static String generate(String seed) {
        // Generate a new string with 32 characters (1 base-36 character for each digit)
        StringBuilder sb = new StringBuilder();
        Random random = new Random();
        for (int i = 0; i < 32 - seed.length(); i++) {
            char c1 = Character.toUpperCase(sb.charAt(i));
            sb.setCharAt(i, Character.toLowerCase(c1));
        }
        
        return sb.toString();
    }

    private static boolean validate(String seed, String input) {
        // Use a Random to generate two random digits
        int index1 = new Random().nextInt(32);
        int index2 = new Random().nextInt(32);
        
        // Check if the generated string matches the given input string
        return sb.toString().equals(input.substring(index1, index1 + 8)) && sb.toString().equals(input.substring(index2, index2 + 8));
    }
}
