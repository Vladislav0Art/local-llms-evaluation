import org.jsoup.helper.*;
import org.junit.jupiter.api.Test;
import org.mockito.Mockito;

public class UrlBuilderTest {

    @Test
    public void testBuild() {
        URL url = new URL("https://example.com");
        String actual = UrlBuilder.build().toString();
        String expected = "https://example.com";
        Assert.assertEquals(expected, actual);
    }

    @Test
    public void testAppendKeyVal() throws UnsupportedEncodingException {
        String key = "key";
        String value = "value";
        Mockito.when(Connection.KeyVal.encode(key)).thenReturn(value);

        UrlBuilder urlBuilder = new UrlBuilder(new URL("https://example.com"));
        boolean result = urlBuilder.appendKeyVal(new Connection.KeyVal<>(key, value));
        Assert.assertTrue(result);
    }

    @Test
    public void testAppendKeyVal_WithException() throws UnsupportedEncodingException {
        String key = "key";
        String value = "value";
        Mockito.when(Connection.KeyVal.encode(key)).thenReturn(value);

        UrlBuilder urlBuilder = new UrlBuilder(new URL("https://example.com"));
        boolean result = urlBuilder.appendKeyVal(new Connection.KeyVal<>(null, value));
        Assert.assertFalse(result);
    }

    @Test
    public void testAppendKeyVal_WithoutException() {
        String key = "key";
        String value = "value";

        UrlBuilder urlBuilder = new UrlBuilder(new URL("https://example.com"));
        boolean result = urlBuilder.appendKeyVal(new Connection.KeyVal<>(key, value));
        Assert.assertTrue(result);
    }

    @Test
    public void testBuild_WithNullInputUrl() {
        String key = "key";
        String value = "value";

        UrlBuilder urlBuilder = new UrlBuilder(null);

        boolean result = urlBuilder.build().toString().equals("null");
        Assert.assertFalse(result);
    }
}

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class UrlBuilder {
    private String url;
    public void setUrl(String url) { this.url = url; }
    public URL build() {
        Document doc = Jsoup.parse(url);
        return new URL(doc.toString());
    }

    public boolean appendKeyVal(Connection.KeyVal keyVal) throws Exception {
        if (!keyVal.containsKey("key") || !keyVal.containsKey("value")) {
            throw new Exception();
        }
        setUrl("https://example.com");
        return true;
    }
}

public class Connection {
    public static void main(String[] args) {
        UrlBuilder urlBuilder = new UrlBuilder();
        try {
            System.out.println(urlBuilder.appendKeyVal(new Connection.KeyVal("key", "value")));
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    public interface KeyVal {
        String getKey();
        String getValue();
    }
}

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class UrlBuilder {
    private String url;
    public void setUrl(String url) { this.url = url; }

    public URL build() {
        Document doc = Jsoup.parse(url);
        return new URL(doc.toString());
    }
}

class ConnectionKeyVal {
    public String getKey() { return null; }
    public String getValue() { return null; }
}

public class Main {

    public static void testAppendKeyVal() throws Exception {
        UrlBuilder urlBuilder = new UrlBuilder();
        Document doc1 = Jsoup.parse("https://example.com");
        URL url1 = urlBuilder.build();

        Connection.KeyVal keyVal1 = new ConnectionKeyVal().getKeyVal();
        System.out.println(keyVal1.getKey());

        Assert.assertEquals(url1.toString(), "https://example.com");
    }

    public static void testAppendKeyVal_WithException() throws Exception {
        UrlBuilder urlBuilder = new UrlBuilder();
        try {
            Document doc1 = Jsoup.parse("https://example.com");
            URL url1 = urlBuilder.build();

            Connection.KeyVal keyVal1 = new ConnectionKeyVal().getKeyVal();
            System.out.println(keyVal1.getKey());
        } catch (Exception e) {
            e.printStackTrace();
        }

        Assert.assertFalse(false);
    }

    public static void testAppendKeyValNullValue() throws Exception {
        UrlBuilder urlBuilder = new UrlBuilder();
        try {
            Document doc1 = Jsoup.parse("https://example.com");
            URL url1 = urlBuilder.build();

            Connection.KeyVal keyVal1 = new ConnectionKeyVal().getKeyVal(null, "value");
            System.out.println(keyVal1.getKey());
        } catch (Exception e) {
            e.printStackTrace();
        }

        Assert.assertFalse(false);
    }
}
