import org.jsoup.nodes.TextNode;
import org.jsoup.nodes.TextNodeList;

public class TestTextNode {
    public static String nodeName(TextNode node) {
        return node.nodeName();
    }

    public static String getWholeText(TextNode node) {
        return node.text();
    }

    public static boolean isBlank(TextNode node) {
        return node.isBlank();
    }

    public static String splitText(int length, TextNode node) {
        if (node == null || length <= 0) {
            return "";
        }
        int splitIndex = node.childNodes().length - length;
        if (splitIndex < 0) {
            return "";
        }
        Node[] nodes = new Node[splitIndex + 1];
        for (int i = 0; i < splitIndex; i++) {
            nodes[i] = node.childNodes()[i];
        }
        int tailStart = splitIndex;
        while (tailStart > 0 && !nodes[tailStart - 1].isBlank()) {
            tailStart--;
        }
        return String.join(" ", nodes, tailStart + 1);
    }

    public static void main(String[] args) {
        TestTextNodeTest testTextNodeTest = new TestTextNodeTest();

        testTextNodeTest.testnodeName();
        testTextNodeTest.testGetWholeText();
        testTextNodeTest.testIsBlank();
        testTextNodeTest.testSplitText(5, null);

        System.out.println("All tests passed");
    }
}

import java.util.ArrayList;
import java.util.List;

public class TestArrayList {
    public static void main(String[] args) {
        List<String> list = new ArrayList<>();
        list.add("Hello");
        list.add("World");

        System.out.println(list);

        String text1 = testSplitText(5, null);
        System.out.println(text1);

        for (String str : testGetWholeText(null).split(", ")) {
            System.out.println(str);
        }

        List<String> list2 = new ArrayList<>();
        list2.add("Apple");
        list2.add("Banana");

        String text3 = testGetWholeText(list2);
        System.out.println(text3);

        System.out.println(testIsBlank(null));
    }

    public static void testSplitText(int length, Node node) {
        if (length <= 0 || node == null) {
            return;
        }
        List<String> list = new ArrayList<>();
        for (int i = 0; i < length - 1; i++) {
            String text = splitText(i + 1, node);
            list.add(text);
        }
        String text4 = testGetWholeText(node);

        System.out.println("Text is " + (text4.contains(",")) ? "contains comma" : "does not contain comma");
    }

    public static void testGetWholeText(Node node) {
        if (node == null || node.childNodes().length <= 0) {
            return "";
        }
        StringBuilder builder = new StringBuilder();
        for (int i = 0; i < node.childNodes().length - 1; i++) {
            String text = testGetWholeText(node.childNodes()[i]);
            builder.append(text).append(" ");
        }
        return builder.toString().trim();
    }

    public static boolean testIsBlank(Node node) {
        if (node == null || node.isBlank()) {
            return true;
        } else {
            for (int i = 0; i < node.childNodes().length - 1; i++) {
                if (!testIsBlank(node.childNodes()[i])) {
                    return false;
                }
            }
            return !node.isBlank();
        }
    }
}
