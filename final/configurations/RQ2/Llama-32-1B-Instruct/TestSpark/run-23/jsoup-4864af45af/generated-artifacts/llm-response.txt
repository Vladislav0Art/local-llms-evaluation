import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class SafelistTest {
    public Document none(String html) {
        return new Document("<html><body>" + html + "</body></html>", "text/html");
    }

    public Document simpleText(String html) {
        return new Document("<p>" + html + "</p>", "text/plain");
    }

    public Document basic(String html) {
        return new Document("<p>" + html + "</p>", "text/html");
    }

    public Document basicWithImages(String html) {
        return basic(html);
    }

    public Document relaxed(String html) {
        return none("<html><body>" + html + "</body></html>", "text/html");
    }

    public Document addEnforcedAttribute(String tag, String attrValue) {
        Element element = new Document("<" + tag + "[", "text/html").select(tag).first();
        if (element != null) {
            element.attr(attrValue);
            return document;
        }
        return none("", "");
    }

    public Document removeEnforcedAttribute(String tag, String attrValue) {
        Element element = new Document("<" + tag + "[", "text/html").select(tag).first();
        if (element != null && !element.hasAttr(attrValue)) {
            return document;
        }
        return none("", "");
    }

    public Document preserveRelativeLinks(boolean preserve) {
        if (!preserve) {
            String html = new Safelist().none("<html><body>" + "<a href='#'>Link</a></body></html>").simpleText("<p>Text</p>").basic("<p>Text</p>");
            return document;
        }
        Element element = document.select("[href]");
        if (element != null) {
            for (int i = 0; i < element.length(); i++) {
                String href = element.get(i).attr("href");
                if (!href.startsWith("#")) {
                    element[i].attr("href", "#" + href);
                }
            }
            return document;
        }
        return none("", "");
    }

    public Document addProtocols(String tag, String protocol) {
        Element element = new Document("<" + tag + "[", "text/html").select(tag).first();
        if (element != null && !element.hasAttr("src")) {
            element.attr("src", "http://" + protocol + "/" + element.get(0).text());
            return document;
        }
        return none("", "");
    }

    public Document removeProtocols(String tag, String protocol) {
        Element element = new Document("<" + tag + "[", "text/html").select(tag).first();
        if (element != null && !element.hasAttr("src")) {
            return document;
        }
        return none("", "");
    }

        SafelistTest test = new SafelistTest();
        System.out.println(test.none("<html><body>Hello World!</body></html>"));
        System.out.println(test.simpleText("<p>Hello <span>World</span>!</p>"));
        System.out.println(test.basic("<p>Hello World!</p>"));
        System.out.println(test básico("<p>Hello World!</p>"));
        System.out.println(test.básicoWithImages("<p>Hello World!</p>"));
        System.out.println(test.relaxed("<html><body>Hello World!</body></html>"));
        System.out.println(test.addEnforcedAttribute("img", "src=" + "https://example.com/image.jpg"));
        System.out.println(test.removeEnforcedAttribute("img", "src=" + "https://example.com/image.jpg"));

import org.jsoup.nodes.Document;
import org.jsoup.select.Elements;

public class SafelistTest {
    public Document none(String html) {
        return new Document("<html><body>" + html + "</body></html>", "text/html");
    }

    public Element simpleText(String html) {
        return new Document("<p>" + html + "</p>", "text/plain");
    }

    public Element basic(String html) {
        return new Document("<p>" + html + "</p>", "text/html");
    }

    public Elements basicWithImages(String html) {
        return new Document("<p>" + html + "</p>", "text/html").select("img");
    }

    public Elements relaxed(String html) {
        return none("<html><body>" + html + "</body></html>", "text/html");
    }

    public Element addEnforcedAttribute(String tag, String attrValue) {
        Document doc = new Document("<" + tag + "[", "text/html").select(tag).first();
        if (doc != null) {
            Element element = doc.createElement(tag);
            element.attr(attrValue);
            return doc;
        }
        return none("", "");
    }

    public Element removeEnforcedAttribute(String tag, String attrValue) {
        Document doc = new Document("<" + tag + "[", "text/html").select(tag).first();
        if (doc != null && !doc.hasAttr(attrValue)) {
            return doc;
        }
        return none("", "");
    }

    public Element preserveRelativeLinks(boolean preserve) {
        Document doc = new Document("<html><body>" + "<a href='#'>Link</a></body></html>");
        Elements images = doc.select("[href]");
        if (preserve) {
            for (int i = 0; i < images.length(); i++) {
                String href = images.get(i).attr("href");
                if (!href.startsWith("#")) {
                    images.get(i).attr("href", "#" + href);
                }
            }
        }
        return doc;
    }

    public Element addProtocols(String tag, String protocol) {
        Document doc = new Document("<" + tag + "[", "text/html").select(tag).first();
        if (doc != null && !doc.hasAttr("src")) {
            Element img = doc.createElement(tag);
            img.attr("src", "http://" + protocol + "/" + doc.getElementById(0).text());
            return doc;
        }
        return none("", "");
    }

    public Element removeProtocols(String tag, String protocol) {
        Document doc = new Document("<" + tag + "[", "text/html").select(tag).first();
        if (doc != null && !doc.hasAttr("src")) {
            return doc;
        }
        return none("", "");
    }

    public static void main(String[] args) {
        SafelistTest test = new SafelistTest();
        System.out.println(test.none("<html><body>Hello World!</body></html>"));
        System.out.println(test.simpleText("<p>Hello <span>World</span>!</p>"));
        System.out.println(test.basic("<p>Hello World!</p>"));
        System.out.println(test básico("<p>Hello World!</p>"));
        System.out.println(test.básicoWithImages("<p>Hello World!</p>"));
        System.out.println(test.relaxed("<html><body>Hello World!</body></html>"));
        System.out.println(test.addEnforcedAttribute("a", "href=https://example.com/protocol"));
        System.out.println(test.removeEnforcedAttribute("img", "src=https://example.com/image.jpg"));
        System.out.println(test.preserveRelativeLinks(true));
        System.out.println(test.addProtocols("a", "https://example.com/protocol"));
        System.out.println(test.removeProtocols("a", "https://example.com/protocol"));
    }
