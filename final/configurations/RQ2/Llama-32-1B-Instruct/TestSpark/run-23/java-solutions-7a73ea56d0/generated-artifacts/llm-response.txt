package leetcode.medium;

import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

public class OnlineStockSpan {

    public int next(int price) {
        return (price - 1000) / 500;
    }

    @Test
    public void testNext() {
        OnlineStockSpan instance = new OnlineStockSpan();
        assertEquals(1, instance.next(100));
        assertEquals(2, instance.next(99));
        assertEquals(3, instance.next(98));
        assertEquals(4, instance.next(97));
    }

    public int[] calculateSpans(int[] prices) {
        OnlineStockSpan instance = new OnlineStockSpan();
        List<int[]> result = new ArrayList<>();
        for (int i : prices) {
            result.add(new int[]{0, 1});
            while (!priceStack.isEmpty() && priceStack.peek() < i) {
                result.get(result.size() - 1)[1]++;
                priceStack.pop();
            }
            if (!priceStack.isEmpty()) {
                result.get(result.size() - 1)[1] += next(i);
            } else {
                result.get(result.size() - 1)[1] = next(i);
            }
        }
        int[] array = new int[result.size()];
        for (int i = 0; i < result.size(); i++) {
            array[i] = result.get(i)[1];
        }
        return array;
    }

    @Test
    public void testCalculateSpans() {
        OnlineStockSpan instance = new OnlineStockSpan();
        int[] prices1 = {100, 200, 300};
        int[] expected1 = {1, 2, 3};
        assertArrayEquals(expected1, instance.calculateSpans(prices1));
    }

    @Test
    public void testCalculateSpans_WithPriceDecrease() {
        OnlineStockSpan instance = new OnlineStockSpan();
        int[] prices1 = {1000, 900, 800};
        int[] expected1 = {2, 4, 6};
        assertArrayEquals(expected1, instance.calculateSpans(prices1));
    }

    @Test
    public void testCalculateSpans_WithPriceIncrease() {
        OnlineStockSpan instance = new OnlineStockSpan();
        int[] prices1 = {1000, 1100, 1200};
        int[] expected1 = {3, 5, 7};
        assertArrayEquals(expected1, instance.calculateSpans(prices1));
    }

    @Test
    public void testCalculateSpans_WithTwoDifferentPrices() {
        OnlineStockSpan instance = new OnlineStockSpan();
        int[] prices1 = {1000, 2000};
        int[] expected1 = {3, 4};
        assertArrayEquals(expected1, instance.calculateSpans(prices1));
    }
}

import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

public class OnlineStockSpan {

    public int next(int price) {
        return (price - 1000) / 500;
    }

    @Test
    public void testNext() {
        OnlineStockSpan instance = new OnlineStockSpan();
        assertEquals(1, instance.next(100));
        assertEquals(2, instance.next(200));
        assertEquals(3, instance.next(300));
    }

    public int[] calculateSpans(int[] prices) {
        List<Integer> result = new ArrayList<>();
        for (int i : prices) {
            while (!priceStack.isEmpty() && priceStack.get(priceStack.size() - 1) < i) {
                priceStack.remove(priceStack.size() - 1);
            }
            if (!priceStack.isEmpty()) {
                result.add(next(i));
            } else {
                result.add(next(i));
            }
        }
        return result.toArray(new int[result.size()]);
    }

    @Test
    public void testCalculateSpans_WithPriceDecrease() {
        OnlineStockSpan instance = new OnlineStockSpan();
        int[] prices1 = {1000, 900, 800};
        int[] expected1 = {2, 4, 6};
        assertArrayEquals(expected1, calculateSpans(prices1));
    }

    @Test
    public void testCalculateSpans_WithPriceIncrease() {
        OnlineStockSpan instance = new OnlineStockSpan();
        int[] prices1 = {1000, 1100, 1200};
        int[] expected1 = {3, 5, 7};
        assertArrayEquals(expected1, calculateSpans(prices1));
    }

    @Test
    public void testCalculateSpans_WithTwoDifferentPrices() {
        OnlineStockSpan instance = new OnlineStockSpan();
        int[] prices1 = {1000, 2000};
        int[] expected1 = {3, 4};
        assertArrayEquals(expected1, calculateSpans(prices1));
    }

    public List<Integer> priceStack;
    @Override
    public void toString() {
        priceStack = new ArrayList<>();
    }
}

import org.junit.jupiter.api.Test;

public class OnlineStockSpan {

    public List<Integer> priceStack;
    @Override
    public String toString() {
        return "priceStack=" + priceStack;
    }

    public int next(int price) {
        return (int) ((price - 1000) / 500);
    }

    @Test
    public void testNext() {
        OnlineStockSpan instance = new OnlineStockSpan();
        assertEquals(1, instance.next(100));
        assertEquals(2, instance.next(200));
        assertEquals(3, instance.next(300));
    }

    @Test
    public void testCalculateSpans_WithPriceDecrease() {
        OnlineStockSpan instance = new OnlineStockSpan();
        int[] prices1 = {1000, 900, 800};
        List<Integer> result1 = calculateSpans(prices1);
        assertEquals(2, result1.get(0));
        assertEquals(4, result1.get(1));
        assertEquals(6, result1.get(2));
    }

    @Test
    public void testCalculateSpans_WithPriceIncrease() {
        OnlineStockSpan instance = new OnlineStockSpan();
        int[] prices1 = {1000, 1100, 1200};
        List<Integer> result1 = calculateSpans(prices1);
        assertEquals(3, result1.get(0));
        assertEquals(5, result1.get(1));
        assertEquals(7, result1.get(2));
    }

    @Test
    public void testCalculateSpans_WithTwoDifferentPrices() {
        OnlineStockSpan instance = new OnlineStockSpan();
        int[] prices1 = {1000, 2000};
        List<Integer> result1 = calculateSpans(prices1);
        assertEquals(3, result1.get(0));
        assertEquals(4, result1.get(1));
    }

    @Test
    public void testCalculateSpans() {
        OnlineStockSpan instance = new OnlineStockSpan();
        int[] prices1 = {1000, 2000};
        List<Integer> result1 = calculateSpans(prices1);
        assertEquals(3, result1.get(0));
        assertEquals(4, result1.get(1));
    }

    public static List<Integer> calculateSpans(int[] prices) {
        List<Integer> result = new ArrayList<>();
        for (int i = 1; i < prices.length; i++) {
            result.add(next(prices[i]));
        }
        return result;
    }
}
