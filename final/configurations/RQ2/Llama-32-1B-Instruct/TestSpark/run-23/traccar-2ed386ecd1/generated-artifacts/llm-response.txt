import java.util.function.Function;

public class Main {

    public static Function<String, Boolean> checkLength(String s) {
        return length -> {
            if (length.length() == 0) return false;
            for (char c : s.toCharArray()) {
                if (!Character.isLetterOrDigit(c)) return false;
            }
            return true;
        };
    }

    public static Function<String, Boolean> checkSatellites(String s) {
        return function -> {
            int count = 0;
            for (char c : s.toCharArray()) {
                if (c == 'S') count++;
            }
            return count > 5 || count < 1;
        };
    }

    public static Function<String, Boolean> checkSpeed(String s) {
        return function -> {
            int count = 0;
            for (char c : s.toLowerCase().toCharArray()) {
                if (!Character.isLetter(c)) count++;
            }
            return count > 10 || count < 1;
        };
    }

    public static void main(String[] args) {
