        }

    }

    private static void createDirectory(String packageName) throws Exception {
        String dirName = packageName + "/src/main/java/" + packageName;
        java.io.File dir = new java.io.File(dirName);
        if (!dir.exists()) {
            dir.mkdir();
        }
    }

    private static void publicClass(String className, String packageName) throws Exception {
        String fileName = "MyClass.java";
        String classFileContent = readFile(packageName, className + ".java");
        writeToFile(fileName, classFileContent);
    }

    private static void publicMethod(String className, String packageName) throws Exception {
        String methodName = "test" + className.charAt(0).toLowerCase() + "." + className.substring(1);
        String methodFileContent = readFile(packageName, methodName + ".java");
        writeToFile(methodName + ".java", methodFileContent);
    }

    private static void testMethod(Method method, String moduleName) throws Exception {
        class TestClass {
            public String myField;

            @Override
            public Object invoke(Object... args) throws Throwable {
                return method.invoke(this, args);
            }
        }

        Class<?> testClass = Main.class.getPackage().getName() + "/" + moduleName + "/TestClass.java";
        testMethod(testClass, "MyClass");
    }

    private static void testAllTests(Class<?>[] testClasses) throws Exception {
        for (Class<?> clazz : testClasses) {
            Method[] methods = clazz.getMethods();
            for (Method method : methods) {
                if (!method.getName().startsWith("test")) {
                    continue;
                }
                System.out.println(methodName + " is not a test method");
            }
        }
    }

    private static String readFile(String packageName, String fileName) throws Exception {
        Class<?> packageClass = Main.class.getPackage();
        Package pkg = packageClass.getClassLoader().getSystemResourceAsStream(packageName + "/" + fileName).getFileName().toString();

        return new java.io.File(pkg.toString()).readAllBytes();
    }

    private static void writeToFile(String fileName, String content) throws Exception {
        File file = new java.io.File(fileName);
        file.setWritable(true);

        try (BufferedWriter writer = new BufferedWriter(new FileWriter(file))) {
            try (char[] cbuffer = content.toCharArray()) {
                for (int i = 0; i < cbuffer.length; i++) {
                    writer.write(cbuffer[i]);
                }
            }
        }
    }
}
