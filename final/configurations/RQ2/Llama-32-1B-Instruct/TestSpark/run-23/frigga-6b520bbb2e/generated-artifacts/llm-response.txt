public class AppVersion {
    private String name;
    private String major;
    private String minor;
    private String patch;
    private Map<String, Object> props = new HashMap<>();

    public AppVersion(String name) {
        this.name = name;
    }

    public void setMajor(String major) {
        if (major == null || !isValidMajor(major)) {
            throw new IllegalArgumentException("Invalid major version");
        }
        this.major = major;
    }

    public void setMinor(String minor) {
        if (minor == null || !isValidMinor(minor)) {
            throw new IllegalArgumentException("Invalid minor version");
        }
        this.minor = minor;
    }

    public void setPatch(String patch) {
        if (patch == null || !isValidPatch(patch)) {
            throw new IllegalArgumentException("Invalid patch version");
        }
        this.patch = patch;
    }

    private boolean isValidMajor(String major) {
        return Integer.parseInt(major).isInteger() && Integer.parseInt(major) > 0;
    }

    private boolean isValidMinor(String minor) {
        return Integer.parseInt(minor).isDigit();
    }

    private boolean isValidPatch(String patch) {
        if (patch == null || !isValidString(patch)) {
            throw new IllegalArgumentException("Invalid patch version");
        }
        return Integer.parseInt(patch).isInteger() && Integer.parseInt(patch) < 100;
    }

    @Override
    public String toString() {
        return name + major + minor + patch;
    }

    private boolean isValidString(String str) {
        if (str == null || !str.startsWith("\"") || !str.endsWith("\"")) {
            throw new IllegalArgumentException("Invalid string");
        }
        try {
            Integer.parseInt(str.substring(1, str.length() - 1));
            return true;
        } catch (NumberFormatException e) {
            return false;
        }
    }

    public String getProperties() {
        if (props.isEmpty()) {
            for (Map.Entry<String, Object> entry : props.entrySet()) {
