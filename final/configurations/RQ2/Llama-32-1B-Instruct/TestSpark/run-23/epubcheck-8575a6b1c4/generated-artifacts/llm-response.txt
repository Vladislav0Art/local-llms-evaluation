```java
import java.io.File;
import java.util.Scanner;

public class EpubValidator {

    static class EpubChecker {
        public int main(String[] args) throws InvalidVersionException {
            return 0;
        }
    }

    private Scanner scanner;

    public EpubValidator(File file) {
        this.scanner = new Scanner(file);
    }

    public void validateFile() {
        try {
            while (scanner.hasNextLine()) {
                String line = scanner.nextLine();
                if (line.contains("version")) {
                    System.out.println("Version: " + line.split("=")[1].trim());
                } else if (line.contains("layout")) {
                    System.out.println("Layout: " + line.split("=")[1].trim());
                }
            }

            // Check for invalid versions
            try {
                scanner.useDelimiter("\\A");
                String versionFromFile = scanner.next();
                Version parsedVersion = parseVersion(versionFromFile);
                if (!parsedVersion.isValid()) {
                    throw new InvalidVersionException("Invalid EPUB version: " + parsedVersion);
                }
            } catch (InvalidVersionException e) {
                System.out.println(e.getMessage());
            }

        } catch (FileNotFoundException e) {
            System.out.println("File not found: " + e.getMessage());
        } finally {
            scanner.close();
        }
    }

    private Version parseVersion(String versionFromFile) throws InvalidVersionException {
        // Parse EPUB version string
        String[] parts = versionFromFile.split("\\.");
        if (parts.length != 2 || !parts[0].matches("[0-9]+") || !parts[1].matches("[0-9]+")) {
            throw new InvalidVersionException("Invalid EPUB version format: " + versionFromFile);
        }
        int major = Integer.parseInt(parts[0]);
        int minor = Integer.parseInt(parts[1]);

        if (major < 2 || major > 12) {
            throw new InvalidVersionException("Invalid EPUB version major number");
        }

        if (minor < 0 || minor > 5) {
            throw new InvalidVersionException("Invalid EPUB version minor number");
        }
        return new Version(major, minor);
    }

}

class Version {
    private int major;
    private int minor;

    public Version(int major, int minor) {
        this.major = major;
        this.minor = minor;
    }

    public boolean isValid() {
        if (this.major < 2 || this.major > 12) {
            return false;
        }
        if (this.minor < 0 || this.minor > 5) {
            return false;
        }
        return true;
    }

}

class InvalidVersionException extends Exception {

}
```

In the code above, we first parse a version string and validate its format. We then check for any invalid versions by trying to parse them as EPUB versions. If an invalid version is found, it throws an `InvalidVersionException`.
