import java.util.*;

class DataFrame {
    private String name;
    private List<List<String>> columns;
    private Map<String, Object> data;

    public DataFrame(String name) {
        this.name = name;
        this.columns = new ArrayList<>();
        this.data = new HashMap<>();
    }

    public void addColumn(List<String> columnNames, Object value) {
        this.columns.add(columnNames);
        this.data.put(this.columns.size() - 1, value);
    }

    public String getColumnName(int columnIndex) {
        return columns.get(columnIndex).get(0);
    }

    public List<List<String>> getColumns() {
        return columns;
    }

    public void addData(List<List<String>> data) {
        this.data.putAll(data);
    }

    public Object getData(int row, int column) {
        if (column < 0 || column >= columns.size()) {
            throw new IndexOutOfBoundsException();
        }
        return data.get(column).get(row);
    }

    public List<String> toString() {
        StringBuilder sb = new StringBuilder();
        for (List<String> column : columns) {
            sb.append(column).append(",");
        }
        if (!sb.toString().isEmpty()) {
            sb.deleteCharAt(sb.length() - 1);
        }
        return "[" + sb.toString() + "]";
    }

    public void sortColumns(List<String> columnNames) {
        Collections.sort(this.columns, (column1, column2) -> new Comparator<List<String>>() {
            @Override
            public int compare(List<String> o1, List<String> o2) {
                for (String c : column1) {
                    if (!o2.contains(c)) {
                        return -1;
                    }
                }
                for (String c : column2) {
                    if (!c.equals(o2.get(0))) {
                        return 1;
                    }
                }
                return 0;
            }
        });
    }

    public void setColumnValue(int columnIndex, String columnName, Object value) {
        this.columns.get(columnIndex).set(this.columns.size() - 1, value);
    }
}

public class Main {
    public static void main(String[] args) {
        DataFrame df = new DataFrame("myDataFrame");
        df.addColumn(Arrays.asList("colA", "colB"), Arrays.asList(1, 2));
        df.addColumn(Arrays.asList("colC", "colD"), Arrays.asList(3, 4));

        System.out.println(df.toString());
        for (int i = 0; i < df.getColumns().size(); i++) {
            System.out.print(df.getColumns().get(i).get(0) + ": ");
            for (Object value : df.getData(i)) {
                System.out.print(value + " ");
            }
            System.out.println();
        }

        df.sortColumns(Arrays.asList("colA", "colB"));
        System.out.println(df.toString());

        df.setColumnValue(0, "newColA", 5);
        System.out.println(df.toString());
    }
}

class DataFrame {
    private String name;
    private List<List<String>> columns;
    private Map<String, Object> data;

    public DataFrame(String name) {
        this.name = name;
        this.columns = new ArrayList<>();
        this.data = new HashMap<>();
    }

    public void addColumn(List<String> columnNames, Object value) {
        if (this.columns.size() > 0 && !columnNames.contains(this.columns.get(0))) {
            throw new IllegalArgumentException("Cannot add a column with the same name as an existing one.");

        if (!(o instanceof DataFrame)) return false;
        DataFrame df = (DataFrame) o;
        if (!name.equals(df.name)) return false;
        if (columns.size() != df.columns.size()) return false;
        for (List<String> column : columns) {
            if (!column.equals(df.columns.get(0))) return false;
        }
        return Arrays.equals(data.values(), df.data.values());
    }

    @Override
    public int hashCode() {
        return Objects.hash(name, columns);

        StringBuilder sb = new StringBuilder();

            sb.deleteCharAt(sb.length() - 1);
        }
        return "[" + sb.toString() + "]";
    }

    public void setColumnValue(int columnIndex, Object value) {
        this.data.put(columnIndex, value);
    }
}
