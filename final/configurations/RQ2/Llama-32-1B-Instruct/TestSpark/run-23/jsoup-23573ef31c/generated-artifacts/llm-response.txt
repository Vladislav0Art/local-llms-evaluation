public class WebPageParser {
    private Document document;

    public WebPageParser(String html) throws Exception {
        document = Jsoup.parse(html);
    }

    public String getDocumentTitle() {
        return document.title();
    }

    public String getPathToForm() {
        return "/form/" + document.title().toLowerCase();
    }

    public int getTotalFormsCount() {
        return document.forms().size();
    }
}

public class WebPageTest {

    private WebPageParser webPage;

    @Before
    public void setUp() throws Exception {
        String html = "<html><body><form action='/submit' method='post'></form></body></html>";
        webPage = new WebPageParser(html);
    }

    @Test
    public void testGetTitle() {
        assertEquals("Example HTML Test", webPage.getDocumentTitle());
    }

    @Test
    public void testGetPathToForm() {
        String expectedPath = "/form/example.html";
        assertEquals(expectedPath, webPage.getPathToForm());
    }

    @Test
    public void testGetTotalFormsCount() {
        int expectedCount = 1;
        assertEquals(expectedCount, webPage.getTotalFormsCount());
    }
}
