public class Utils {

    public static String generateUrl(int key, Object value) {
        StringBuilder url = new StringBuilder();
        for (int i = 0; i < key; i++) {
            if (i < value.getClass().getDeclaredFields().length) {
                url.append(value.getClass().getSimpleName()).append("/");
            }
        }
        return "https://example.com/" + url.toString();
    }

    public static int generateId(String name) {
        Random random = new Random();
        return random.nextInt(1000000);
    }
}

        System.out.println(Utils.generateUrl(1, 2));
    }

}

    }

        return num1 * num2;
    }

    public int divide(int num1, int num2) {
        if (num2 == 0) {
            throw new ArithmeticException("Cannot divide by zero");
        }
        return num1 / num2;
    }
}

class AnotherCalculator extends Calculator {

    @Override
    public String toString() {
        return "add: " + add(10, 5);
    }

    @Override
    public int calculate(String operation, int num1, int num2) {
        if (!operation.equals("add") && !operation.equals("subtract") && !operation.equals("multiply") && !operation.equals("divide")) {
            throw new IllegalArgumentException("Invalid operation");
        }
        switch (operation) {
            case "add":
                return add(num1, num2);
            case "subtract":
                return subtract(num1, num2);
            case "multiply":
                return multiply(num1, num2);
            default:
                return divide(num1, num2);
        }
    }
}

public class Main {

    public static void main(String[] args) {
        Calculator calculator = new AnotherCalculator();
        System.out.println(calculator.add(10, 5)); // Output: 15
        System.out.println(calculator.subtract(10, 5)); // Output: 5
        System.out.println(calculator.multiply(10, 5)); // Output: 50
        try {
            System.out.println(calculator.divide(10, 0)); // Throws ArithmeticException
        } catch (ArithmeticException e) {
            System.out.println(e.getMessage()); // Output: Cannot divide by zero
        }
    }
}
