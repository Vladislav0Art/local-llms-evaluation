import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class TestParser {
    public static void main(String[] args) {
        // Create a Jsoup Document object from HTML string
        String html = "<html><body><div><p>Hello World!</p></div></body></html>";
        Document doc = Jsoup.parse(html);

        // Get all tags in the document
        Tag[] tags = doc.getAllTags();

        System.out.println("All Tags:");
        for (Tag tag : tags) {
            System.out.println(tag.tagName() + ": " + tag.toString());
        }

        // Get the first paragraph and its contents
        Tag paraTag = doc.selectFirst("p").first();
        String contents = paraTag.data();

        System.out.println("\nFirst Paragraph Contents:");
        System.out.println(paraTag.toString() + ": " + contents);

        // Check if a tag exists
        boolean exists = false;
        for (Tag tag : tags) {
            if (tag.tagName().equals("p")) {
                exists = true;
                break;
            }
        }

        System.out.println("\nDoes Tag Exist?: " + exists);
    }
}

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class TestParser {
    public static void main(String[] args) {
        // Create a Jsoup Document object from HTML string
        String html = "<html><body><div><p>Hello World!</p></div></body></html>";

        // Parse the HTML and get all tags
        Document doc = Jsoup.parse(html);

        // Print the parsed HTML
        System.out.println("Parsed HTML:");
        System.out.println(doc.outerHtml());

        // Get the first paragraph
        String contents = doc.selectFirst("p").first().data();

        // Print the contents of the first paragraph
        System.out.println("\nContents of First Paragraph:");
        System.out.println(contents);

        // Check if a specific tag exists
        boolean exists = false;
        for (Tag tag : doc.getAllTags()) {
            if (tag.tagName().equals("p")) {
                exists = true;
                break;
            }
        }

        // Print whether the specific tag exists
        System.out.println("\nDoes Tag Exist?: " + (exists ? "true" : "false"));
    }
}
