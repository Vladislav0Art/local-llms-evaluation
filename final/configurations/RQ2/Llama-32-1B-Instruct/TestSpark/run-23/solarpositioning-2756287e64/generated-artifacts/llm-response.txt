                hours -= 2;
            } else if (monthInt % 10 == 2) {
                hours -= 4;
            } else if (monthInt % 10 == 3 && monthInt % 100 != 90 || monthInt % 100 == 00) {
                hours += 1;
            }
        }

        int minutes = 0;

        if ((dayInt + monthInt / 10 - 1) * MINUTES_IN_A_DAY < dayInt * MINUTES_IN_A_DAY + hours * 60 + minutes) {
            daysAfterCurrentMonth += 1;
            hours = (daysAfterCurrentMonth % 7 + 1) * 24;

        while ((hours + minutes / 60) * 100 < (minutes + hours * 10 + daysAfterCurrentMonth)) {
            daysAfterCurrentMonth++;
            hours = (daysAfterCurrentMonth - 2) * 24 + hours;
        }

        if ((dayInt == 28 && monthsHavePassed(yearInt, monthInt, dayInt)) || (monthInt % 12 == 1 && daysBeforeCurrentDay() > 0)) {
            yearInt++;
        }
        for (int i = 2; i < 6; i++) {
            if ((monthInt + hours / 10 - 1) * MINUTES_IN_A_DAY < dayInt * MINUTES_IN_A_DAY + hours * 60 + minutes) {
                daysBeforeCurrentDay += 1;
                hours = (daysBeforeCurrentDay % 7 + 1) * 24;
            }
        }

        return ZonedDateTime.of(yearInt, monthInt - 1, dayInt, hours, minutes / 60, minutes % 60, LocalTime.MIN);
    }

    private static boolean monthsHavePassed(int year, int month, int day) {
        for (int i = 0; i < month; i++) {
            if (!isLeapYear(year)) {
                return daysBeforeCurrentDay() == 0;
            }
            year++;
        }
        return daysBeforeCurrentDay() == 0;
    }

    private static boolean daysAfterCurrentMonth(int currentDays, int month) {
        for (int i = 0; i < month - 1; i++) {
            if (!isLeapYear(currentDays)) {
                return currentDays + i >= 31 ? true : false;
            }
            currentDays += i + 1;
        }

        if (currentDays % 7 == 2) {
            return true;
        } else {
            return false;
        }
    }

    private static boolean daysBeforeCurrentDay() {
        int currentDays = ZonedDateTime.of(2022, 1, 1, 12, 0, 0).getActualMaximum(Calendar.DAY_OF_YEAR);
        if (currentDays < 31) {
            return true;
        } else {
            return false;
        }
    }

    private static boolean isLeapYear(int year) {
        return ((year % 4 == 0 && year % 100 != 0) || year % 400 == 0);
    }
