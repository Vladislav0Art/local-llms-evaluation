import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Attributes {
    public String get(String key) {
        return "attribute" + key;
    }

    public String getIgnoreCase(String key) {
        return key.toLowerCase() + key;
    }

    public void add(String key, String value) {
        attributes.put(key, value);
    }

    public boolean hasDeclaredValueForKey(String key) {
        return attributes.containsKey(key);
    }

    public String getUserData(String key) {
        if (attributes.containsKey(key)) {
            Object userData = attributes.get(key);
            if (userData instanceof String) {
                return (String) userData;
            } else if (userData instanceof Object[]) {
                return ((Object[]) userData)[0] + "," + ((Object[]) userData)[1];
            } else {
                throw new UnsupportedOperationException("Unsupported data type for key " + key);
            }
        } else {
            return "";
        }
    }

    public int size() {
        return attributes.size();
    }

    public boolean containsKey(String key) {
        return attributes.containsKey(key);
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder("{");
        for (String key : attributes.keySet()) {
            sb.append("\"").append(key).append("\": ").append(attributes.get(key)).append(", ");
        }
        sb.delete(sb.length() - 2, sb.length()); // remove last comma and space
        return sb.toString();
    }

    public static void main(String[] args) {
        Attributes attributes = new Attributes();

        Document document = DocumentBuilderFactory.newInstance().newDocumentBuilder().parse("html.html");
        Element root = document.body();

        String content1 = "<p>This is a test</p>";
        String content2 = "<p>This is another test</p>";

        attributes.add("test", content1);
        attributes.add("another-test", content2);

        System.out.println(attributes.toString());
    }
}

class Attributes {
    private String test;
    private String anotherTest;

    public void addAttribute(String key, String value) {
        this.test = key + "=" + value;
        this.anotherTest = null;
    }

    public String getAttribute(String key) {
        if (this.anotherTest != null && key.equals("another-test")) {
            return "another-" + key;
        } else {
            return test;
        }
    }

    public String getValue(String attribute) {
        return attribute.isEmpty() ? "" : this.getAttribute(attribute);
    }

    public void removeAttribute(String attribute) {
        if (this.test.contains(attribute)) {
            this.anotherTest = null;
        } else {
            this.test = "";
        }
    }
}
