```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        // Define a sample JSON string
        String json = "{\"name\":\"John\",\"age\":30,\"city\":\"New York\"}";

        System.out.println("Given JSON:");
        System.out.println(json);

        // Parse the JSON string
        JSONParser parser = new JSONParser(json);
        List<Object> list = parser.parseList();
        
        System.out.println("\nParsed JSON Array:");
        for (Object obj : list) {
            if (obj instanceof String) {
                System.out.println("String: " + ((String) obj));
            } else if (obj instanceof Integer) {
                System.out.println("Integer: " + ((Integer) obj));
            }
        }

        // Parse the JSON object
        parser = new JSONParser(json);
        Object obj = parser.parseObject();

        System.out.println("\nParsed JSON Object:");
        printValue(obj);
    }

    public static void printValue(Object value) {
        if (value instanceof String) {
            System.out.println("String: " + ((String) value));
        } else if (value instanceof Integer) {
            System.out.println("Integer: " + ((Integer) value));
        }
    }
}
```

```java
import java.util.*;

public class Main {
    public static void main(String[] args) {
        // Define a sample JSON string
        String json = "{\"name\":\"John\"}";
        
        System.out.println("Given JSON:");
        System.out.println(json);

        // Parse the JSON string
        JSONParser parser = new JSONParser(json);
        List<String> list = new ArrayList<>();
        parser.startObject();
        while (parser.readNext()) {
            switch (parser.getCurrentType()) {
                case NAME:
                    list.add(parser.getCurrentValue());
                    break;
                default:
                    parser.readLast();
            }
        }
        parser.endObject();
        
        // Print out the JSON string
        for (String s : list) {
            System.out.println(s);
        }

        // Parse the JSON object
        JSONParser parser2 = new JSONParser(json);
        Object obj = parser2.parseObject();

        System.out.println("\nParsed JSON Object:");
        printValue(obj);
    }

    public static void printValue(Object value) {
        switch (value.getClass()) {
            case String:
                System.out.println("String: " + (String) value);
                break;
            case Integer:
                System.out.println("Integer: " + (Integer) value);
                break;
            default:
                System.out.println("Unknown type");
        }
    }
}
```
