public class DbAppTest {
    private String[] strarrOperators = {"/home/user/operator1", "/home/user/operator2"};

    public void writeToDbApp() throws IOException {
        CsvWriter writer = new CsvWriter(new Serializer());
        List<Table> tables = createTables();
        for (Table table : tables) {
            table.getRows().forEach(row -> {
                String[] values = row.toString().split(",");
                for (int i = 0; i < strarrOperators.length; i++) {
                    if (i == 0) {
                        writer.write(table.getName(), values[0]);
                    } else {
                        writer.write(table.getName(), values[i]);
                    }
                    if ((i + 1) % strarrOperators.length == 0 && !strarrOperators.length.equals(3)) {
                        writer.write(",");
                    }
                }
            });
        }

        // Close the CSV file
        try (CsvReader reader = new CsvReader(new FileInputStream("dbapp.csv"))) {
            while (reader.read()) {}
        }
    }

    public void writeToDbAppWithParams() throws IOException {
        CsvWriter writer = new CsvWriter(new Serializer());
        List<Table> tables = createTables();
        for (Table table : tables) {
            table.getRows().forEach(row -> {
                String[] values = row.toString().split(",");
                for (int i = 0; i < strarrOperators.length; i++) {
                    if (i == 0) {
                        writer.write(table.getName(), values[0]);
                    } else if (strarrOperators.length == 1 && i == 2) { // Operator
                        writer.write("operator", values[i]);
                    } else {
                        writer.write(values[i]);
                    }
                    if ((i + 1) % strarrOperators.length == 0 && !strarrOperators.length.equals(3)) {
                        writer.write(",");
                    }
                }
            });

        List<Table> tables = createTables();
        for (String[] row : {"user1", "admin"}) {
            Table table = new Table();
            table.setRows(row);
            tables.add(table);
        }
        for (Table table : tables) {
            writeTableToDbAppWithParams(writer, table);
        }
    }

    public void writeTableToDbApp(CsvWriter writer, Table table) throws IOException {
        String tableName = table.getName();
        if (!tableName.isEmpty()) {
            writer.write(tableName, table.getRows().get(0));
            for (int i = 1; i < table.getRows().size(); i++) {
                writer.write(",", table.getRows().get(i));
                if ((i + 1) % 2 == 0 && !table.getName().isEmpty()) {
                    writer.write(",");
                }
            }
        } else {
            // Handle empty table name
        }
    }

    public void writeTableToDbAppWithParams(CsvWriter writer, Table table) throws IOException {
        String tableName = table.getName();
        if (!tableName.isEmpty()) {
            writer.write(tableName, table.getRows().get(0));
            for (int i = 1; i < table.getRows().size(); i++) {
                writer.write(",", table.getRows().get(i));
                if ((i + 1) % 2 == 0 && !table.getName().isEmpty()) {
                    writer.write(",");
                }
            }
        } else {
            // Handle empty table name
        }
    }

        private List<String> rows;
