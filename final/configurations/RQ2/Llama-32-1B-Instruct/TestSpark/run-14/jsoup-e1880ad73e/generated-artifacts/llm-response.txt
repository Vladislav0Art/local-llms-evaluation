package org.jsoup.helper;

import org.jsoup.Connection;
import org.jsoup.html.parser HtmlParser;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class Main {

    public static void main(String[] args) {
        URLBuilder urlBuilder = new URLBuilder();

        // Test 1: Append key-value pair to URL
        String inputUrl = "https://example.com";
        String key = "key";
        String value = "value";
        try {
            urlBuilder.appendKeyVal(key, value);
            System.out.println("URL with key-value pair appended successfully");
        } catch (UnsupportedEncodingException e) {
            System.out.println("Error occurred while building URL: " + e.getMessage());
        }

        // Test 2: Append empty key-value pair to URL
        inputUrl = "https://example.com";
        try {
            urlBuilder.appendKeyVal(null, null);
            System.out.println("URL with empty key-value pair appended successfully");
        } catch (UnsupportedEncodingException e) {
            System.out.println("Error occurred while building URL: " + e.getMessage());
        }

        // Test 3: Append invalid key-value pair to URL
        inputUrl = "https://example.com";
        try {
            urlBuilder.appendKeyVal(key, null);
            System.out.println("URL with invalid key-value pair appended successfully");
        } catch (UnsupportedEncodingException e) {
            System.out.println("Error occurred while building URL: " + e.getMessage());
        }

        // Test 4: Append invalid URL
        inputUrl = "";
        try {
            urlBuilder.appendKeyVal(key, value);
            System.out.println("URL with null string as input failed");
        } catch (UnsupportedEncodingException e) {
            System.out.println("Error occurred while building URL: " + e.getMessage());
        }

        // Test 5: Append invalid URI
        String uri = "";
        try {
            urlBuilder.appendKeyVal(key, value);
            System.out.println("URL with null string as input failed");
        } catch (UnsupportedEncodingException e) {
            System.out.println("Error occurred while building URL: " + e.getMessage());
        }

        // Test 6: Append key-value pair to HTML document
        Document doc = new HtmlParser().parse("https://example.com");
        Elements elements = doc.getElementsByTag("p");
        for (Element element : elements) {
            Element child = element.children();
            if (child instanceof Element) {
                urlBuilder.appendKeyVal(((Element) child).getAttribute("id"), ((Element) child).text());
            }
        }

        // Test 7: Append empty key-value pair to HTML document
        doc = new HtmlParser().parse("https://example.com");
        elements = doc.getElementsByTag("p");
        for (Element element : elements) {
            Element child = element.children();
            if (child instanceof Element) {
                urlBuilder.appendKeyVal(null, ((Element) child).text());
            }
        }

        // Test 8: Append invalid key-value pair to HTML document
        doc = new HtmlParser().parse("https://example.com");
        elements = doc.getElementsByTag("p");
        for (Element element : elements) {
            Element child = element.children();
            if (child instanceof Element && !child.text().isEmpty()) {
                urlBuilder.appendKeyVal(child.attr("id"), child.text());
            }
        }
    }

    public static void printUrl(String input, String expected) {
        try {
            URLBuilder.urlWithKeyVal(input);
            System.out.println("URL: " + input + ", Expected output: " + expected);
        } catch (UnsupportedEncodingException e) {
            System.out.println("Error occurred while building URL: " + e.getMessage());
        }
    }

    public static void printUrl(String input, String expectedString) {
        try {
            URLBuilder.urlWithKeyVal(input);
            System.out.println("URL: " + input + ", Expected output: " + expectedString);
        } catch (UnsupportedEncodingException e) {
            System.out.println("Error occurred while building URL: " + e.getMessage());
        }
    }

    public static void printUrl(String input, String[] expected) {
        try {
            URLBuilder.urlWithKeyVal(input);
            for (int i = 0; i < expected.length; i++) {
                if (i > 0) {
                    System.out.println("Expected output " + (i - 1) + ": " + expected[i]);
                }
                System.out.println("URL: " + input + ", Expected output: " + expected[i]);
            }
        } catch (UnsupportedEncodingException e) {
            System.out.println("Error occurred while building URL: " + e.getMessage());
        }
    }

    public static void printUrl(String url, String expected) {
        try {
            URLBuilder.urlWithKeyVal(url);
            if (url.contains("key")) {
                System.out.println("URL contains key-value pair");
            } else {
                System.out.println("URL does not contain key-value pair");
            }
        } catch (UnsupportedEncodingException e) {
            System.out.println("Error occurred while building URL: " + e.getMessage());
        }
    }

    public static void printUrl(String url, String[] expected) {
        try {
            URLBuilder.urlWithKeyVal(url);
            for (int i = 0; i < expected.length; i++) {
                if (i > 0) {
                    System.out.println("Expected output " + (i - 1) + ": " + expected[i]);
                }
                System.out.println("URL: " + url + ", Expected output: " + expected[i]);
            }
        } catch (UnsupportedEncodingException e) {
            System.out.println("Error occurred while building URL: " + e.getMessage());
        }
    }

    public static String getAttribute(String html, String key) {
        return HtmlParser().parse(html).select(key).attr("id");
    }
