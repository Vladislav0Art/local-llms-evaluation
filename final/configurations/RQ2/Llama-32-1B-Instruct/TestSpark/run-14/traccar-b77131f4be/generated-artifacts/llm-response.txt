import io.netty.buffer.ByteBuf;
import org.junit.Test;
import org.traccar.BaseProtocolDecoder;
import org.traccar.NetworkMessage;
import org.traccar.Protocol;
import org.traccar.helper.BitBuffer;
import org.traccar.helper.BitUtil;
import org.traccar.helper.UnitsConverter;
import org.traccar.model.Position;
import org.traccar.session.DeviceSession;

public class GalileoProtocolDecoderTest {
    public static final String MSG = "Hello, world!";
    public static final Protocol PROTOCOL = new Protocol();

    @Test
    public void testDecode_Galileo() throws Exception {
        BaseProtocolDecoder decoder = new GalileoProtocolDecoder(PROTOCOL);
        Object decoded = decoder.decode("192.168.1.100:1234", SocketAddress.IPv4, MSG);
        assertNotNull(decoded);
        assertEquals("Hello, world!", (String) decoded);
    }

    @Test
    public void testDecode_Galileo_Position() throws Exception {
        BaseProtocolDecoder decoder = new GalileoProtocolDecoder(PROTOCOL);
        Position position = new Position(37.7749, -122.4194);
        Object decoded = decoder.decode("192.168.1.100:1234", SocketAddress.IPv4, position);
        assertNotNull(decoded);
        assertEquals(position, (Position) decoded);
    }

    @Test
    public void testDecode_Galileo_SocketAddress() throws Exception {
        BaseProtocolDecoder decoder = new GalileoProtocolDecoder(PROTOCOL);
        Object decoded = decoder.decode("192.168.1.100:1234", SocketAddress.IPv4, null);
        assertNotNull(decoded);
        assertEquals(192.168.1.100, (String) decoded);
    }

    @Test
    public void testDecode_Galileo_List() throws Exception {
        BaseProtocolDecoder decoder = new GalileoProtocolDecoder(PROTOCOL);
        List<DeviceSession> devices = new LinkedList<>();
        devices.add(new DeviceSession("192.168.1.101:1234", "device1"));
        Object decoded = decoder.decode("192.168.1.100:1234", SocketAddress.IPv4, devices);
        assertNotNull(decoded);
        assertEquals(devices, (List<DeviceSession>) decoded);
    }

    @Test
    public void testDecode_Galileo_Mapping() throws Exception {
        BaseProtocolDecoder decoder = new GalileoProtocolDecoder(PROTOCOL);
        Map<String, Object> mapping = new HashMap<>();
        mapping.put("key1", "value1");
        Object decoded = decoder.decode("192.168.1.100:1234", SocketAddress.IPv4, mapping);
        assertNotNull(decoded);
        assertEquals(mapping, (Map<String, Object>) decoded);
    }

    @Test
    public void testDecode_Galileo_ByteBuf() throws Exception {
        BaseProtocolDecoder decoder = new GalileoProtocolDecoder(PROTOCOL);
        ByteBuf byteBuf = Unpooled.buffer();
        byteBuf.writeShort(1234); // 1 byte, assuming 2 bytes per short
        byteBuf.writeLong(98765L); // 4 bytes, assuming 8 bits per long
        Object decoded = decoder.decode("192.168.1.100:1234", SocketAddress.IPV6, byteBuf);
        assertNotNull(decoded);
    }

    @Test
    public void testDecode_Galileo_Event() throws Exception {
        BaseProtocolDecoder decoder = new GalileoProtocolDecoder(PROTOCOL);

import org.junit.Test;
import org.mockito.Mockito;

public class BaseProtocolDecoderTest {

    @Mock
    private Channel channel;

    @Mock
    private SocketAddress remoteAddress;

    @Mock
    private Protocol protocol;

    public static final String MSG = "Hello, world!";
    public static final String POSITION = "{\"latitude\": 37.7749, \"longitude\": -122.4194}";

    @Test

        assertNotNull(result);
        assertEquals(192.168 .1 .100, result);
    }

    @Test
    public void testDecode_Galileo_Position() {
        BaseProtocolDecoder decoder = new GalileoProtocolDecoder(protocol);
        String result = decoder.decode("192.168.1.100:1234", remoteAddress, POSITION);
        assertNotNull(result);
        assertEquals(POSITION, result);
    }

    @Test
    public void testDecode_Galileo_ListDevices() {
        BaseProtocolDecoder decoder = new GalileoProtocolDecoder(protocol);
        String result = decoder.decode("192.168.1.100:1234", remoteAddress, LIST_DEVICES);
