import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Main {

    private Map<String, Double> operations = new HashMap<>();
    private Scanner scanner;

    public void runTest(String input, String output) {
        try {
            this.scanner = new Scanner(System.in);
            double result = calculate(input);
            if (result.equals(output)) {
                System.out.println("Test passed");
            } else {
                System.out.println("Test failed");
            }
        } catch (Exception e) {
            System.out.println("An error occurred: " + e.getMessage());
        }
    }

    public void add(double num1, double num2) {
        operations.put("addition", num1 + num2);
    }

    public void subtract(double num1, double num2) {
        operations.put("subtraction", num1 - num2);
    }

    public void multiply(double num1, double num2) {
        operations.put("multiplication", num1 * num2);
    }

    public void divide(double num1, double num2) throws Exception {
        if (num2 == 0) {
            throw new Exception("Cannot divide by zero");
        }
        operations.put("division", num1 / num2);
    }

    private double calculate(String input) {
        String[] tokens = input.split("\\s+");
        int length = tokens.length;

        for (int i = 0; i < length - 1; i++) {
            try {
                double value = Double.parseDouble(tokens[i]);
                if (i == length - 2) {
                    return operations.get("addition").doubleValue();
                }

                String op = tokens[i + 1];
                double operand = Double.parseDouble(tokens[i + 2]);

                switch (op) {
                    case "add":
                        return add(value, operand);
                    case "subtract":
                        return subtract(value, operand);
                    case "multiply":
                        return multiply(value, operand);
                    case "divide":
                        return divide(value, operand);
                    default:
                        throw new Exception("Invalid operation");
                }
            } catch (NumberFormatException e) {
                continue;
            }
        }

        throw new Exception("Unexpected token: " + input.split("\\s+").length);
    }

    public static void main(String[] args) throws Exception {
        Main main = new Main();

        String[][] tests = {
                {"1 2 3", "5"},
                {"a b c d e", "fghijk"},
                {"1 2 1 1 1", "5"}
        };

        for (String[] test : tests) {
            main.runTest(test[0], test[1]);
        }
    }
}

            } else {
                System.out.println("Test failed");
            }
        } catch (Exception e) {
            System.out.println("An error occurred: " + e.getMessage());
        }
    }

    public void pieceMove(String input) throws Exception {
        String[] tokens = input.split("\\s+");
        int length = tokens.length;

        for (int i = 0; i < length - 1; i++) {
            if (tokens[i].equals("r")) {
                board.put("R", "R");
                board.replace("R", tokens[i + 1]);
            } else if (tokens[i].equals("n")) {
                board.put("N", "N");
                board.replace("N", tokens[i + 1]);
            } else if (tokens[i].equals("b")) {
                board.put("B", "B");
                board.replace("B", tokens[i + 1]);
            } else if (tokens[i].equals("q")) {
                board.put("Q", "Q");
                board.replace("Q", tokens[i + 1]);
            } else if (tokens[i].equals("k")) {
                board.put("K", "K");
                board.replace("K", tokens[i + 1]);
            }
        }

        String[] opponent = board.split("\\s+");
        for (String piece : pieces.values()) {
            char letter = piece.charAt(0);
            int row = Integer.parseInt(piece.substring(1, 3));
            int col = Integer.parseInt(piece.substring(4, 6));

            if (!pieceMove(board.get(letter), row, col)) {
                throw new Exception("Invalid move");
            }

            pieces.put(letter + piece, board.get(letter) + piece);
        }
    }

    public boolean pieceMoveEquals(String input) throws Exception {
        String[] tokens = input.split("\\s+");
        int length = tokens.length;

        for (int i = 0; i < length - 1; i++) {

                return board.get(tokens[i]) == "N";
            } else if (tokens[i].equals("b")) {
                return board.get(tokens[i]) == "B";
            } else if (tokens[i].equals("q")) {
                return board.get(tokens[i]) == "Q";
            } else if (tokens[i].equals("k")) {
                return board.get(tokens[i]) == "K";
            }
        }

        throw new Exception("Unexpected token: " + input.split("\\s+").length);
    }

        }
