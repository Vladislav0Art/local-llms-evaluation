import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Safelist {
    private Document doc;
    private String[] safes;
    private StringBuilder safeTags = new StringBuilder();

    public Safelist(Document doc) {
        this.doc = doc;
        safes = new String[0];
    }

    public void addSafe(String tag) {
        safes = new String[safes.length + 1];
        safes[0] = tag;
    }

    public boolean contains(String tagName) {
        for (int i = 1; i < safes.length; i++) {
            if (safes[i].equals(tagName)) return true;
        }
        return false;
    }

    public String[] getSafeTags() {
        StringBuilder sb = new StringBuilder();
        for (String safe : safes) {
            sb.append(safe).append(" ");
        }
        return sb.toString().trim().split("\\s+");
    }

    public boolean containsTag(String tagName, String attrName, String value) {
        if (contains(tagName)) {
            if (attrName.equals("class") || attrName.equals("style")) {
                return false;
            } else {
                for (int i = 1; i < safes.length; i++) {
                    if (safes[i].equals(tagName) && safes[i].contains(attrName) && safes[i].contains(value)) return true;
                }
            }
        }
        return false;
    }

    public boolean containsClass(String className) {
        for (String safe : safes) {
            if (safe.contains("class=") && safe.startsWith(className + " ")) return true;
        }
        return false;
    }

    public String[] getAttributes(String tagName, int attributeCount) {
        StringBuilder sb = new StringBuilder();
        for (int i = 1; i < safes.length; i++) {
            if (safes[i].equals(tagName)) {
                for (int j = 0; j < attributeCount && j < safes[i].length(); j++) {
                    sb.append(safes[i].substring(j, j + 2)).append(" ");
                }
                break;
            }
        }
        return sb.toString().trim().split("\\s+");
    }

    public boolean containsAttribute(String tagName, String attrName) {
        for (int i = 1; i < safes.length; i++) {
            if (contains(tagName)) {
                for (int j = 0; j < safes[i].length() && j < safes[i].length() - 2 * attrName.length(); j++) {
                    if (safes[i].substring(j, j + 2).equals(attrName)) return true;
                }
            }
        }
        return false;
    }

                for (int j = 0; j < safes[i].length(); j++) {
                    if (safes[i].substring(j, j + 2).equals(tagName)) return safes[i];

                }
            }
        }
        return "";
    }

This is a simple example of how to use the provided class. It provides an `Option` object which represents a set of options, and it can be used as a collection of strings that are used to build HTML elements.

The `myMethod` method is called when you need to create an HTML element with attributes that match the ones specified in the option string. You can create an instance of the class by passing an option string like this: `Option opt = new Option("div 1 div 2 style 3");`.

You can also call `myMethod` without creating an instance, it will just return a HTML element with all attributes that match the ones specified in the option string.

The method iterates over each attribute of the option string and creates a new input field for each one. The values of these fields are obtained by calling the `getStyle`, `getAttributes`, `getClass` methods on the resulting DOM elements.

You can also get all styles or attributes from an element using the `getStyle` or `getAttributes` methods.
