import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class W3CDomTest {

    public static Document fromJsoup(String html) {
        return neworg.jsoup.helper.W3CDom.Converter.fromJsoup(html);
    }

    @Test
    void namespaceAware() {
        W3CDom w3cdom = new W3CDom();
        assertTrue(w3cdom.namespaceAware(true));
        assertFalse(w3cdom.namespaceAware(false));
    }

    @Test
    public Document selectXpath(String xpath) {
        Document soupDoc = fromJsoup("<html><body>Hello World!</body></html>");
        Document domDoc = new W3CDom().Converter.convert(soupDoc, null);
        return domDoc.selectXpath(xpath);
    }

    @Test
    public void JsoupNodes() {
        Document soupDoc = fromJsoup("<html><body>Hello World!</body></html>");
        Document domDoc = neworg.jsoup.helper.W3CDom.Converter.fromJsoup(soupDoc.toString());
        Element element = (Element) domDoc.select("*").first();
        if (element != null) {
            System.out.println(element.outerHtml());
        } else {
            System.out.println("No elements found");
        }
    }

    @Test
    public void JsoupNodesNoProperties() {
        Document soupDoc = fromJsoup("<html><body>Hello World!</body></html>");
        Document domDoc = new W3CDom().Converter.convert(soupDoc, null);
        Element element = (Element) domDoc.select("*").first();
        if (element != null && element.hasAttr("style")) {
            System.out.println(element.attr("style"));
        } else {
            System.out.println("No attributes found");
        }
    }

    @Test
    public void JsoupNodesNoContent() {
        Document soupDoc = fromJsoup("");
        Document domDoc = new W3CDom().Converter.convert(soupDoc, null);
        Element element = (Element) domDoc.select("*").first();
        if (element != null && !element.hasAttr("content")) {
            System.out.println(element.toString());
        } else {
            System.out.println("No content found");
        }
    }

    public void contextNode() {
        Document soupDoc = fromJsoup("<html><body>Hello World!</body></html>");
        Document domDoc = neworg.jsoup.helper.W3CDom.Converter.fromJsoup(soupDoc.toString());
        Element element = (Element) domDoc.contextNode();
        System.out.println(element.outerHtml());
    }

    @Test
    public void contextNodeNoContent() {
        Document soupDoc = fromJsoup("<html><body>Hello World!</body></html>");
        Document domDoc = neworg.jsoup.helper.W3CDom.Converter.fromJsoup(soupDoc.toString());

        if (element != null && !element.hasAttr("content")) {
            System.out.println(element.toString());

            System.out.println("No content found");

    public void asString() {
