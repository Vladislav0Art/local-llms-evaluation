import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class MockParser {

    public static String parseFragment(String input, String currentElement, Parser parser) {
        if (currentElement.length() == 0) {
            return input;
        }

        Element element = new Element(currentElement);
        document.add(element);

        // Parse the fragment into a Document object
        String[] fragments = input.split("\n");
        String text = "";
        for (String fragment : fragments) {
            if (!fragment.trim().isEmpty()) {
                text += fragment + "\n";
            }
        }

        return parseFragment(text, currentElement.substring(0, currentElement.length() - 1), parser);
    }
}

public class TreeBuilder {

    public static String parseFragment(String input, String currentElement) {
        if (currentElement.contains("Document")) {
            return "Document" + currentElement;
        }

        Element element = new Element(currentElement);

        // Parse the fragment into a Document object
        for (int i = 0; i < Math.min(input.length(), currentElement.length()); i++) {
            String fragment = input.substring(0, i + 1);
            if (!fragment.trim().isEmpty()) {
                element.text(fragment);
            }
            input = input.substring(i + 1);
        }

        return parseFragment(input, currentElement);
    }
}

public class Parser {

    public static Document parse(String input) {
        String[] treeNodes = new String[0];

        // Parse the fragment into a Tree object
        Document document = Jsoup.parse("Document" + Jsoup.parse(input).text());
        treeNodes = document.body().text().split("\n");
        return treeBuilder.parseFragment(treeNodes);
    }
}
