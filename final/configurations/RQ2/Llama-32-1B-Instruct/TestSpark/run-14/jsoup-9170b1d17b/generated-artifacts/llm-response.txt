import java.util.ArrayList;
import java.util.List;

public class Attribute {
    private String name;
    private Object value;

    public Attribute(String name, Object value) {
        this.name = name;
        this.value = value;
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj)
            return true;
        if (obj == null || getClass() != obj.getClass())
            return false;
        Attribute attribute = (Attribute) obj;
        return Objects.equals(name, attribute.name) && Objects.equals(value, attribute.value);
    }

    @Override
    public int hashCode() {
        int result = name.hashCode();
        result = 31 * result + value.hashCode();
        return result;
    }
}

public class MyService {
    private List<Attribute> attributes = new ArrayList<>();

    public void addAttribute(Attribute attribute) {
        attributes.add(attribute);
    }

    public boolean hasAttribute(String name) {
        for (Attribute attr : attributes) {
            if (attr.name.equals(name))
                return true;
        }
        return false;
    }

    public Attribute getAttribute(String name) {
        for (Attribute attr : attributes) {
            if (attr.name.equals(name))
                return attr;
        }
        return null;
    }
}

public class TestMyService {
    private MyService myService;

    public TestMyService(MyService myService) {
        this.myService = myService;
    }

    @Test
    public void testAddAttribute() {
        Attribute attribute = new Attribute("attr1", "value1");
        myService.addAttribute(attribute);
        assertTrue(myService.hasAttribute("attr1"));
        Attribute retrievedAttribute = myService.getAttribute("attr1");
        assertEquals("value1", retrievedAttribute.value);
    }

    @Test
    public void testGetAttribute() {
        MyService service = new MyService();
        service.addAttribute(new Attribute("attr2", "value2"));
        assertTrue(service.hasAttribute("attr2"));
        Attribute retrievedAttribute = service.getAttribute("attr2");
        assertEquals("value2", retrievedAttribute.value);
    }

    @Test
    public void testGetNonExistentAttribute() {
        MyService service = new MyService();
        assertFalse(service.hasAttribute("nonExistingAttr"));
    }
}

        element.text = "My Service Test";

                .appendChild(document.createTextNode("console.log('Hello World!');"))

        return document;
    }

            fail("Expected NullPointerException");

            // Ignore

    public static class Attr {

        private String key;
        private String value;

        public Attr(String key, String value) {
            this.key = key;
            this.value = value;
        }

        public String getKey() {
            return key;
        }

        public void setValue(String value) {
            this.value = value;
        }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (o == null || getClass() != o.getClass()) return false;
            Attr that = (Attr) o;
            return Objects.equals(key, that.key) && Objects.equals(value, that.value);
        }

        @Override
        public int hashCode() {
            return Objects.hash(key, value);
        }
    }

        private Attr attribute;

        public void addAttribute(Attr attribute) {
            this.attribute = attribute;
        }

        public boolean hasAttribute(String key) {
            return attribute.getKey().equals(key);
        }

        public String getAttribute(String key) {
            return attribute.getValue();
        }
    }
