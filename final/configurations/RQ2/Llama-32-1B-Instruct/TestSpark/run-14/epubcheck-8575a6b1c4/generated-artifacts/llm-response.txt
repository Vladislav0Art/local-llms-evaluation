import java.util.Scanner;

public class EpubCheck {

    public static boolean validateFile(String[] args) {
        if (args.length != 2) {
            throw new IllegalArgumentException("Expected two arguments.");
        }
        
        String filePath = args[0];
        String fileVersion = args[1];

        // Check file version
        if (!fileVersion.equals("1.2.3") && !fileVersion.equals("4.5.6")) {
            System.out.println("Invalid file version: " + fileVersion);
            return false;
        }

        // If all checks pass, return true
        return true;
    }

    public static void main(String[] args) {

        Scanner scanner = new Scanner(System.in);

        while (true) {
            try {
                System.out.print("Enter the path to the Epub file: ");
                String filePath = scanner.nextLine();

                // Check if file exists
                File file = new File(filePath);
                if (!file.exists()) {
                    System.out.println("The file does not exist.");
                    continue;
                }

                // Validate the file version
                validateFile(new String[]{filePath, "1.2.3"});

                break;
            } catch (Exception e) {
                System.out.println("An error occurred: " + e.getMessage());
            }
        }

        scanner.close();
    }
}
