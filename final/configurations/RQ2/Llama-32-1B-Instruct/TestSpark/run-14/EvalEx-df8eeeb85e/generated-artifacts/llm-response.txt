public class Main {
    public static void main(String[] args) {
        System.out.println(checkParentheses("()"));
        System.out.println(checkParentheses("("));
        System.out.println(checkParentheses(")"));
        System.out.println(checkParentheses("("));
        System.out.println(checkParentheses(")"));
        System.out.println(checkParentheses("Hello, World!"));
    }

    public static String checkParentheses(String str) {
        int openCount = 0;
        int closeCount = 0;

        for (int i = 0; i < str.length(); i++) {
            if (str.charAt(i) == '(') {
                openCount++;
            } else if (str.charAt(i) == ')') {
                closeCount++;
            }
        }

        return String.valueOf(openCount - closeCount);
    }
}
