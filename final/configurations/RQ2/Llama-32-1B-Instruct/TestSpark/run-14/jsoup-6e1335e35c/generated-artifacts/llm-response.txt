```java
import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

public class TestExample {
    private Document document = new Document();

    @Test
    public void testText() {
        assertEquals("Hello, World!", document.text("Hello, World!"));
    }

    @Test
    public void testWholeText() {
        assertEquals("", document.wholeText());
        assertEquals("<script>alert('Hello')</script>", document.wholeOwnText());
    }

    @Test
    public void testOwnText() {
        assertTrue(document.ownText().contains("World"));
        assertFalse(document.ownText().contains("Hello"));
    }

    @Test
    public void testData() {
        assertEquals("", document.data());
        assertEquals("test", document.data("test"));
    }

    @Test
    public void testClassName() {
        assertTrue(document.classNames().contains("test"));
        assertTrue(document.classNames().contains("other"));
    }

    @Test
    public void testAddClass() {
        Element element = document.addClass("other");
        assertEquals("other", element.className());
        assertEquals("", element.val());
        assertEquals(null, element.html(""));
    }

    @Test
    public void testRemoveClass() {
        Element element = document.removeClass("test");
        assertEquals("test", element.className());
        assertEquals("", element.val());
        assertEquals(null, element.html(""));
    }

    @Test
    public void testToggleClass() {
        Element element = document.toggleClass("other");
        assertEquals("other", element.className());
        assertEquals("test", element.className());
        assertEquals("", element.val());
        assertEquals(null, element.html(""));
    }

    @Test
    public void testShouldIndent() {
        assertTrue(document.shouldIndent(new Document.OutputSettings()));
        assertFalse(document.shouldIndent(new Document.OutputSettings(true)));
    }

    @Test
    public void testOuterHtmlHead() {
        Element head = document.outerHtmlHead(document.root(), 1, new Document.OutputSettings());
        assertEquals("<head>\n", document.html(head));
    }

    @Test
    public void testOuterHtmlTail() {
        Element tail = document.outerHtmlTail(document.root(), 0, new Document.OutputSettings());
        assertEquals("", tail.html());
    }
}
```

```java
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.Node;
import org.jsoup.nodes.NodeVisitor;

public class TestDocument {
    public static String testText() {
        return "Hello, World!";
    }

    public static String testWholeText() {
        return "";
    }

    public static String testOwnText() {
        assertTrue(document.ownText().contains("World"));
        assertFalse(document.ownText().contains("Hello"));
    }

    public static String testData() {
        return "";
    }

    public static void addClass(Element element) {
        element.className += " other";
    }

    public static void removeClass(Element element) {
        element.className = element.className().replace("other", "");
    }

    public static boolean shouldIndent(Node node, int indentLevel) {
        if (node instanceof Element && ((Element) node).tagName().equals("head")) {
            return true;
        } else if (node instanceof Node || node instanceof Appendable) {
            return false;
        }
        return false;
    }

    public static String outerHtmlHead(Node root, int indentLevel, Document.OutputSettings outputSettings) {
        Node head = document.outerHtmlHead(root, 1, outputSettings);
        StringBuilder html = new StringBuilder();
        for (int i = 0; i < indentLevel; i++) {
            html.append("    ");
        }
        return html.toString() + "<head>\n" + head.html();
    }

    public static String outerHtmlTail(Node root, int indentLevel, Document.OutputSettings outputSettings) {
        Node tail = document.outerHtmlTail(root, 0, outputSettings);
        StringBuilder html = new StringBuilder();
        for (int i = 0; i < indentLevel; i++) {
            html.append("    ");
        }
        return html.toString() + "<head>\n" + tail.html();
    }

    public static String data(String key) {
        if (!document.data().contains(key)) {
            document.data(key);
            return key;
        } else {
            return "";
        }
    }

    public static void main(String[] args) {
        Document document = new Document();

        assertEquals("Hello, World!", testText());

        assertEquals("", testWholeText());
        assertTrue(testOwnText().contains("World"));
        assertFalse(testOwnText().contains("Hello"));

        assertEquals("", testData());

        addClass(document.root().append("test"));
        addClass(document.root().append("other"));

        assertNotEquals("", shouldIndent(document.outerHtmlHead(document.root(), 1, new Document.OutputSettings())));

        assertNotEquals("", outerHtmlTail(document.root(), 0, new Document.OutputSettings()));

        assertEquals("Hello, World!", data("test"));
        assertEquals("", testOwnText());
    }
}
```

```java
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.Node;

public class TestNodeVisitor {
    public static void main(String[] args) {
        Document document = new Document();

        Node visitor = new NodeVisitor() {
            @Override
            public String visit(Element element) {
                System.out.println(element.tagName());
                return "";
            }
        };

        assertEquals("head", visitor.visit(document.outerHtmlHead(document.root(), 1, new Document.OutputSettings())));

        assertEquals("", visitor.visit(document.outerHtmlTail(document.root(), 0, new Document.OutputSettings())));

        assertEquals("", visitor.visit(document.data("test")));

        document.append("test");

        assertNotEquals("", visitor.visit(document.root().append("other")));
    }
}
```
