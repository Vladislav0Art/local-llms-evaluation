        this.rows = new ArrayList<>();

        rows.add(row);
    }

    public List<DataFrameRow> getRows() {
        return rows;
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        for (DataFrameRow row : getRows()) {
            sb.append(row.toString()).append("\n");
        }
        return sb.toString();
    }

    public void sortColumns(String[] columns) {
        for (DataFrameColumn column : getColumnNames()) {
            column.sort(columns);
        }
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj)
            return true;
        if (obj == null || getClass() != obj.getClass())
            return false;
        DataFrame other = (DataFrame) obj;
        return size == other.size && columns.equals(other.columns);
    }

    @Override
    public int hashCode() {
        return 31 * size + getColumns().hashCode();
    }

    private List<DataFrameColumn> getColumnNames() {
        List<DataFrameColumn> columnNames = new ArrayList<>();
        for (DataFrameRow row : rows) {
            columnNames.add(row.getColumn());

    public void renameColumns(String[] columnsToRename) {
        for (DataFrameColumn column : getColumnNames()) {
            column.setName(columnsToRename);

        }
        return new ArrayList<>();

            if (row.getColumnName().equals(columnName)) {
                row.setFlag(value);

        if (!rows.isEmpty()) {
            return rows.get(0);
