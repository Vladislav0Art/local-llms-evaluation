import org.junit.Test;
import static org.junit.Assert.assertEquals;

public class Solution {

    @Test
    public void test1() {
        int[] arr = {1, 2};
        int result = solution(arr);
        assertEquals(3, result);
    }

    @Test
    public void test2() {
        int[] arr = {4, 5, 6, 7};
        int result = solution(arr);
        assertEquals(20, result);
    }

    @Test
    public void test3() {
        int[] arr = {9, 8, 7, 6, 5, 4, 3, 2, 1};
        int result = solution(arr);
        assertEquals(63, result);
    }

    @Test
    public void test4() {
        int[] arr = {19, 17, 15, 13, 11, 9, 7, 5, 3, 1};
        int result = solution(arr);
        assertEquals(209, result);
    }

    @Test
    public void test5() {
        int[] arr = {4, 2, 6, 0, -1, -2, -10};
        int result = solution(arr);
        assertEquals(0, result);
    }

    @Test
    public void test6() {
        int[] arr = {-10000, -20000, -30000};
        int result = solution(arr);
        assertEquals(-60000, result);
    }

    @Test
    public void test7() {
        int[] arr = {1.2, 4, 2, 5, 3.6};
        double[] result = new double[arr.length];
        for (int i = 0; i < arr.length; i++) {
            result[i] = solution(arr, i);
        }
        assertEquals(10.8, result[0], 0.00001);
    }

    @Test
    public void test8() {
        int[] arr = {1, -2, 3, -4};
        double[] result = new double[arr.length];
        for (int i = 0; i < arr.length; i++) {
            result[i] = solution(arr, i);
        }
        assertEquals(10.0, result[0], 0.00001);
    }

    @Test
    public void test9() {
        int[] arr = {1, 2};
        int result = solution(arr);
        assertEquals(5, result);
    }

    private int solution(int[] arr) {
        return Arrays.stream(arr).mapToDouble(x -> (int)x).sum();
    }
}
