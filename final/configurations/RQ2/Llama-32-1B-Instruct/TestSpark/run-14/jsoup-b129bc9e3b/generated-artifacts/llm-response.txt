import java.util.*;

class RequestBody {
    private String body;

    public RequestBody(String body) {
        this.body = body;
    }

    public Map<String, String> requestbody(String method, String url, String body) {
        return new HashMap<>();
    }
}

class Response {
    private Map<String, String> headers;
    private String status;
    private String body;

    public Response(Map<String, String> headers, String status, String body) {
        this.headers = headers;
        this.status = status;
        this.body = body;
    }

    public void print() {
        System.out.println(status + ": " + body);
    }
}

class Cookie {
    private Map<String, String> cookies;

    public Cookie(Map<String, String> cookies) {
        this.cookies = cookies;
    }

    public void print() {
        for (Map.Entry<String, String> entry : cookies.entrySet()) {
            System.out.println(entry.getKey() + ": " + entry.getValue());
        }
    }
}

class HttpConnection {
    private Map<String, RequestBody> requestbody;
    private Map<String, Cookie> cookies;

    public HttpConnection(Map<String, RequestBody> requestbody, Map<String, Cookie> cookies) {
        this.requestbody = requestbody;
        this.cookies = cookies;
    }

    public void addRequestBody(String method, String url, String body) {
        if (requestbody.containsKey(method)) {
            System.out.println("Error: Method already exists.");
            return;
        }
        RequestBody requestBody = new RequestBody(body);
        requestbody.put(method, requestBody);
    }

    public void addCookie(Cookie cookie) {
        cookies.put(cookie.method(), cookie);
    }

    public Response getResponse(String method, String url, String body) {
        if (!requestbody.containsKey(method)) {
            System.out.println("Error: Method does not exist.");
            return null;
        }
        RequestBody requestBody = requestbody.get(method);
        Map<String, String> responseHeaders = new HashMap<>();
        for (Map.Entry<String, String> entry : cookies.entrySet()) {
            if (entry.getKey().equals(method)) {
                responseHeaders.put(entry.getKey(), entry.getValue());
            }
        }

        Response response = new Response(responseHeaders, "OK", requestBody.requestbody(method, url, body));
        return response;
    }

    public void print() {
        System.out.println("Status: OK");
        if (cookies != null) {
            for (Map.Entry<String, Cookie> entry : cookies.entrySet()) {
                System.out.println(entry.getKey() + ": " + entry.getValue().get("Content-Type"));
            }
        } else {
            System.out.println();
        }
    }
}

public class Main {
    public static void main(String[] args) {
        HttpConnection connection = new HttpConnection(new RequestBody("Hello, world!".getBytes()), new Cookie(Map.of("name", "John", "age", "30")));
        
        connection.addRequestBody("POST", "", "");
        System.out.println(connection.getResponse("POST", "/example.com", ""));
    }
}
