```java
import org.junit.Test;
import org.jsoup.helper.UrlBuilder;
import org.jsoup.http.HttpUrl;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

import java.net.URLDecoder;
import java.util.HashMap;
import java.util.Map;

public class UrlBuilderTest {

    @Test
    public void testAppendKeyVal_1() {
        HttpUrl url = new HttpUrl("https://example.com/path?key=-value");
        Map<String, String> kv = new HashMap<>();
        kv.put("key", "value");
        UrlBuilder builder = new UrlBuilder(url);
        builder.appendKeyVal(kv);
        assertEquals("https://example.com/path?key=value", builder.build().toURI().toString());
    }

    @Test
    public void testAppendKeyVal_2() {
        URL url = new URL("https://example.com/path?key=value");

        kv.put("key", "value");
        UrlBuilder builder = new UrlBuilder(url);
        builder.appendKeyVal(kv);
        assertEquals("https://example.com/path?key=value", builder.build().toURI().toString());
    }

    @Test
    public void testAppendKeyVal_3() {
        URL url = new URL("https://example.com/path");
        Map<String, String> kv = new HashMap<>();
        kv.put("key", "value");
        UrlBuilder builder = new UrlBuilder(url);
        builder.appendKeyVal(kv);
        assertEquals(null, builder.build().toURI());
    }

    @Test
    public void testAppendKeyVal_4() {
        URL url = null;
        Map<String, String> kv = new HashMap<>();
        kv.put("key", "value");
        UrlBuilder builder = new UrlBuilder(url);
        builder.appendKeyVal(kv);
        assertEquals(null, builder.build().toURI());
    }

    @Test
    public void testAppendKeyVal_5() {
        HttpUrl url = new HttpUrl("https://example.com/path?key=-value");
        URL decodedUrl = decodeUrl(url.toString());
        Map<String, String> kv = new HashMap<>();
        kv.put("key", "value");
        UrlBuilder builder = new UrlBuilder(decodedUrl);
        builder.appendKeyVal(kv);
        assertEquals("https://example.com/path?key=value", builder.build().toURI().toString());
    }

    private static URL decodeUrl(String url) {
        try {
            return new URI(url).getScheme() + "://" + new URI(url)..getHost();
        } catch (MalformedURLException e) {
            throw new RuntimeException(e);
        }
    }
}
```

```java
import java.net.URI;
import java.util.HashMap;
import java.util.Map;

public class URIHelper {
    public static String getScheme(URI uri) {
        return uri.getScheme();
    }

    public static String getAuthority(URI uri) {
        return uri.getAuthority();
    }

    public static String getPath(URI uri) {
        return uri.getPath();
    }

    public static Map<String, String> queryMap(URI uri) {
        Map<String, String> map = new HashMap<>();
        for (String param : uri.getQuery().split("&")) {
            String[] keyValue = param.split("=");
            if (!keyValue[0].isEmpty()) {
                map.put(keyValue[0], keyValue[1]);
            }
        }
        return map;
    }

    public static void main(String[] args) {
        URI uri = new URI("https://example.com/path?a=1&b=2");
        System.out.println("Scheme: " + getScheme(uri));
        System.out.println("Authority: " + getAuthority(uri));
        System.out.println("Path: " + getPath(uri));

        Map<String, String> queryMap = queryMap(uri);
        System.out.println("Query Map:");
        for (Map.Entry<String, String> entry : queryMap.entrySet()) {
            System.out.println(entry.getKey() + "=" + entry.getValue());
        }

        URI decodedUri = new URI(uri.getScheme(), uri.getHost(), uri.getPath(), null, queryMap, null);
        System.out.println("Decoded URI: " + decodedUri);
    }
}
```
