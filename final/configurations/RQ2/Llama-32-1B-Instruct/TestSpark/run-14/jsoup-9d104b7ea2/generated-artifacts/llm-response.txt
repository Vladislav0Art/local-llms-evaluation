import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import java.util.ArrayList;
import java.util.List;

public class DocumentTest {

    public static class DocumentExtensions {
        private String url;
        private List<Node> children;

        public DocumentExtensions(String url) {
            this.url = url;
            this.children = new ArrayList<>();
        }

        public String getUrl() { return url; }
        public void addChild(Node node) { children.add(node); }
        public Node getChild(int index) throws IndexOutOfBoundsException {
            if (index < 0 || index >= children.size()) {
                throw new IndexOutOfBoundsException();
            }
            return children.get(index);
        }

        public String getBody() {
            StringBuilder body = new StringBuilder();
            for (Node node : children) {
                body.append(node.getBody()).append("\n");
            }
            return body.toString().trim();
        }

        public void appendText(String text) {
            for (Node node : children) {
                node.setText(text);
            }
        }
    }

    public static class NodeExtensions {
        private String text;
        private Document document;

        public NodeExtensions(String text, Document document) {
            this.text = text;
            this.document = document;
        }

        public String getText() { return text; }
        public void setText(String text) { this.text = text; }
        public Document getDocument() { return document; }
    }

    public static class ElementTest {

        public static class ElementExtensions {
            private int index;

            public ElementExtensions(int index) {
                this.index = index;
            }

            public boolean isFirstChild() throws IndexOutOfBoundsException {
                if (index == 0) {
                    throw new IndexOutOfBoundsException();
                }
                return true;
            }
        }

        public static void main(String[] args) {
            Document document = new Document("http://example.com");
            Node node1 = new NodeExtensions("Hello, world!", document);
            Node node2 = new NodeExtensions("This is a test.", document);

            List<Node> children = new ArrayList<>();
            children.add(node1);
            children.add(node2);

            Element element = document.createElement("p");

            for (int i = 0; i < children.size(); i++) {
                Node childNode = children.get(i);
                if (childNode.isFirstChild()) {
                    node1.appendChild(childNode);
                } else {
                    node2.appendChild(childNode);
                }
            }

            System.out.println("Body of element: " + document.getBody());
            System.out.println("Text of element: " + element.getText());

            for (int i = 0; i < children.size(); i++) {
                Node childNode = children.get(i);

                if (i == node1.index) {
                    System.out.println("Child text: " + childNode.getText());
                } else {
                    System.out.println("Child text: " + childNode.getText() + "\n");
                }
            }

            element.appendText("This is the second paragraph.");
        }
    }
}
