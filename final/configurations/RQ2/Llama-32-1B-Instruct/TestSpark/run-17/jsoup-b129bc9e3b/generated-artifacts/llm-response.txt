```java
import java.util.HashMap;
import java.util.Map;

public class Main {
    public static void main(String[] args) {
        // Create a map of methods and their corresponding URLs
        Map<String, String> requestMethods = new HashMap<>();
        requestMethods.put("GET", "http://example.com");
        requestMethods.put("POST", "http://example2.com");

        // Test the test method
        assert test(requestMethods);

    }

    public static boolean test(Map<String, String> requestMethods) {
        for (Map.Entry<String, String> entry : requestMethods.entrySet()) {
            Request request = createRequest(entry.getKey(), entry.getValue());
            int responseCode = request.get.getResponseCode();
            System.out.println("Method: " + entry.getKey() + ", URL: " + entry.getValue() + ", Response Code: " + responseCode);
        }
        
        // Test with invalid method
        try {
            Request requestInvalidMethod = createRequest("INVALID_METHOD", "http://example.com");
            int responseCode = requestInvalidMethod.get.getResponseCode();
            System.out.println("Invalid Method: GET, URL: http://example.com, Response Code: " + responseCode);
        } catch (AssertionError e) {
            System.out.println(e.getMessage());
        }

        return true;
    }

    public static Request createRequest(String method, String url) throws Exception {
        Request request = new Request();
        request.setMethod(method);
        request.setUrl(url);
        
        if (!requestMethods.containsKey(method)) {
            throw new AssertionError("Unsupported method: " + method);
        }
        
        return request;
    }

    public static void testMain() throws Exception {
        Main.main(null);
    }
}
```

In this solution, we use a HashMap to store the methods and their corresponding URLs. We then call the `test` method which iterates over the map, creates requests for each method, gets the response code, and prints it out.

We also added a test case where we create a request with an invalid method and catch the `AssertionError` that will be thrown if the method is not found in the map.
