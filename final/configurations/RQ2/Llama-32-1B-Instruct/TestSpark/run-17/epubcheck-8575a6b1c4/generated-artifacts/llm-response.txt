    public int run(String[] args) {
        int result = 0;
        for (String arg : args) {
            if (arg.startsWith("--locale=")) {
                result |= 1 << 5;
            } else if (arg.startsWith("--version")) {
                String[] parts = arg.split(" ");
                int version = Integer.parseInt(parts[parts.length - 2]);
                EpubVersion[] versions = new EpubVersion[version];
                for (int i = 0; i < version; i++) {
                    versions[i] = EpubVersion.valueOf(parts[i + 1], false);
                }
                result |= 1 << (5 * version);
            } else if (arg.startsWith("--profile=")) {
                String[] parts = arg.split("=");
                String profile = parts[parts.length - 1];
                result |= 1 << 6;
                System.setProperty(profile, profile);
            }
        }

        try {
            EpubChecker checker = new EpubChecker();
            return checker.validateFile(null, checker.getEpubVersion(), null, checker.getEpubProfile());
        } catch (InvalidVersionException e) {
            throw new RuntimeException(e);
        } catch (RuntimeException e) {
            System.err.println("Error: " + e.getMessage());
            return -1;
        }
    }
