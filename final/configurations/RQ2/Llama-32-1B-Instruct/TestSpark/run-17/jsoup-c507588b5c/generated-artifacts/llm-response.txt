    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Node)) return false;
        Node node = (Node) o;
        return Objects.equals(value, node.value);
    }

    @Override
    public int hashCode() {
        return Objects.hash(value);
    }
}

public class Main {

    public static String toString(Node root) {
        StringBuilder sb = new StringBuilder();

        dfs(root, sb);

        return sb.toString();
    }

    private static void dfs(Node node, StringBuilder sb) {
        if (node == null) return;

        sb.append(node.value).append(" ");

        if (node.left != null && node.left.isLeaf()) {
            sb.append("\n").append(node.left.toString());
        } else if (node.left != null && !node.left.isLeaf()) {
            dfs(node.left, sb);
        }

        if (node.right != null && node.right.isLeaf()) {
            sb.append("\n").append(node.right.toString());
        } else if (node.right != null && !node.right.isLeaf()) {
            dfs(node.right, sb);
        }
    }

    public static void main(String[] args) {
        Node root = new Node("A");

        root.left = new Node("B");
        root.right = new Node("C");

        root.left.left = new Node("D");
        root.left.right = new Node("E");

        String result = toString(root);
        System.out.println(result);  // Output: "A B C\n" + "A D\n" + "B E"
    }

    public static void printToConsole(Node node, int level) {
        if (node != null) {
            for (int i = 0; i < level; i++) {
                System.out.print("  ");
            }
            System.out.println(node.value);

            if (node.left != null && !node.left.isLeaf()) {
                printToConsole(node.left, level + 1);
            }

            if (node.right != null && !node.right.isLeaf()) {
                printToConsole(node.right, level + 1);
            }
        } else {
            System.out.println();
        }
    }

    public static void leftRotate(Node root) {
        Node newRoot = new Node(root.value);

        newRoot.left = root.left;
        newRoot.right = root;

        return root = newRoot;
    }

    public static void rightRotate(Node root) {
        if (root == null || !root.isLeaf()) return;

        Node newRoot = root.right;

        root.right = newRoot.left;
        newRoot.left = root;

        return root = newRoot;
    }
