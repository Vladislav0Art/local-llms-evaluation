import org.w3c.dom.Document;
import org.w3c.dom.Element;
import org.w3c.dom.Node;

public class NodeHelper {

    public static boolean isNodeText(Node node) {
        return node.getNodeType() == Node.TEXT_NODE || node.getNodeValue() != null && !node.getValue().trim().isEmpty();
    }

    public static void insertAfter(Element parent, Element child, int depth) {
        if (depth > 0) {
            parent.appendChild(child);
        } else {
            for (int i = 0; i < depth - 1; i++) {
                parent.appendChild(parent.lastChild);
            }
            parent.appendChild(child);
        }
    }

    public static String text(String text, Element parent) {
        if (!parent.hasChildNodes()) {
            return text;
        } else {
            insertAfter(parent, text, 0);
            return text;
        }
    }

    public static void htmlAppend(String prefix, Appendable accum, int depth, Document.OutputSettings out, Node node) throws IOException {
        if (out == null || !node.hasChildNodes()) {
            return;
        } else {
            insertAfter(node, prefix + "\n", depth);
            for (Node child : node.childNodes()) {
                htmlAppend(prefix + "  ", accum, depth + 1, out, child);
            }
        }
    }

    public static String html(String html) {
        Document document = new Document();
        Element root = document.createElement("html");
        document.appendChild(root);

        appendToElement(root, html);

        return document.toString();
    }

    private static void appendToElement(Element parent, String text) {
        if (parent.hasChildNodes()) {
            for (int i = 0; i < parent.childNodes.length; i++) {
                appendToElement(parent.childNodes[i], text);
            }
        } else {
            parent.setTextContent(text);
        }
    }

    public static Node clone(Node node, Document document) throws Exception {
        if (!node.hasChildNodes()) {
            return null;
        } else {
            for (int i = 0; i < node.childNodes.length; i++) {
                Node child = clone(node.childNodes[i], document);
                if (child != null) {
                    insertAfter(document.createElement("body"), child, 1);
                    insertAfter(parent, child, 1);
                    document.appendChild(child);
                }
            }

            return parent;
        }
    }

        for (int i = 0; i < element.childNodes.length; i++) {
            Element clone = shallowClone(element.childNodes[i], document);
            if (clone != null) {
                insertAfter(document.createElement("body"), clone, 1);
                insertAfter(parent, clone, 1);
                document.appendChild(clone);
            }
        }

        return parent;
    }

        while (node != null && !node.hasChildNodes()) {
            node = node.getParentNode();
        }
        return node;
    }

        for (int i = 0; i < depth; i++) {
            for (Node child : document.childNodes) {
                visit(child, visitor);
            }
        }

        if (!visitor.contains(document)) {
            throw new RuntimeException("Visitor not found in document.");
        }
    }

            visit(document.getDocumentElement().getChildNode(i), visitor);
        }

        if (!visitor.contains(document)) {
            throw new RuntimeException("Visitor not found in document.");
        }
    }

    public static void forEach(Consumer<? super Element> action, Document document) throws Exception {
        for (int i = 0; i < document.getDocumentElement().getChildNodes().length; i++) {
            visit(document.getDocumentElement().getChildNode(i), action);
        }

            return;
