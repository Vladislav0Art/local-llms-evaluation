import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.parser.XmlTreeBuilder;

public class Parser {
    public static Document parse(String inputString, String baseUri) throws Exception {
        return XmlTreeBuilder.parse(inputReader, baseUri);
    }

    public static void insert(Node rootElement, Token token) {
        Element newNode = new Element(token.getKey());
        if (rootElement != null && !rootElement.isEmpty()) {
            newNode.appendChild(XmlTreeBuilder.remove(rootElement).children().item(0));
        }
        rootElement.appendChild(newNode);
    }
}

class Token {
    private String key;
    private String text;

    public Token(String key, String text) {
        this.key = key;
        this.text = text;
    }

    public String getKey() {
        return key;
    }

    public String getText() {
        return text;
    }
}

class Node {
    private Document document;
    private Element element;
    private List<Node> children;

    public Node(Document document) {
        this.document = document;
    }

    public void appendChild(Node child) {
        if (children == null) {
            children = new ArrayList<>();
        }
        children.add(child);
    }

    public String getTag() {
        return element.tagName();
    }

    public Element getElement() {
        return element;
    }

    public Document getDocument() {
        return document;
    }
}

class Reader implements java.io.Reader {
    private final byte[] inputBuffer;
    private int inputIndex;
    private int inputLength;

    public Reader(String inputString) throws Exception {
        this.inputBuffer = inputString.getBytes();
        this.inputIndex = 0;
        this.inputLength = 0;
    }

    @Override
    public int read() {
        if (inputIndex >= inputLength) {
            throw new java.io.IOException("No more characters in the input string.");
        }
        return inputBuffer[inputIndex++];
    }

    @Override
    public void close() throws Exception {
        // No implementation needed for Reader.
    }
}

public class XmlTreeBuilderTest {

    private static Document document;

    public static void testParseXmlSimple(String inputString, String baseUri) throws Exception {
        Reader reader = new Reader(inputString);
        document = Parser.parse(reader, baseUri);
        assert document.toString().contains("<a/>");
    }

    public static void testInsertTokenWithStartTag() throws Exception {
        Node rootElement = new Node(document);
        Parser.insert(rootElement, new Token("start", "start"));
        String expectedHtml = "<a/>";
        assertEquals(expectedHtml, document.outerHTML());
    }

    public static void testInsertTokenWithComment() throws Exception {
        Node rootElement = new Node(document);
        Parser.insert(rootElement, new Token("comment", "This is a comment."));
        String expectedHtml = "<a/><!-- This is a comment. -->";
        assertEquals(expectedHtml, document.outerHTML());
    }

    public static void testInsertTokenWithCharacter() throws Exception {
        Node rootElement = new Node(document);
        Parser.insert(rootElement, new Token("character", "c"));
        String expectedHtml = "<a/c>";
        assertEquals(expectedHtml, document.outerHTML());
    }
}

public class Main {

    public static void main(String[] args) throws Exception {
        Document document = parse("a <b>start</b> c", "http://example.com/your/base uri");
        
        testInsertToken(document, new Token("start", "start"));
        testInsertComment(document);
        testInsertCharacter(document);

        System.out.println(document.outerHtml());
    }

    private static void testInsertToken(Document document, Token token) {
        Parser.insert(document, token);
        String expectedHtml = "<b>start/start</b>";
        assertEquals(expectedHtml, document.outerHtml());

        Element commentElement = document.getElementById("comment");
        if (commentElement != null) {
            commentElement.setText(token.getText());
        }
