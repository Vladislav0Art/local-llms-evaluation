public class AttributesTest {

    private Attributes testAttributes;

    @Before
    public void setUp() {
        testAttributes = new Attributes();
    }

    public void testGetStringKeyValue() throws Exception {
        String key = "test";
        String value = "Hello, World!";
        String result = testAttributes.get(key);
        Assert.assertEquals(value, result);

        key = "test";
        result = testAttributes.getUserData(key);
        Assert.assertEquals(value, result);

        key = "test";
        testAttributes.remove(key);
        result = testAttributes.get(key);
        Assert.assertNull(result);
    }

    public void testGetStringIgnoreCaseKey() throws Exception {

        result = testAttributes.getUserData(key);
        Assert.assertEquals(value, result);

        key = "test";
        result = testAttributes.removeIgnoreCase(key);
        Assert.assertNull(result);
    }

        Attributes attributes = new Attributes();
        attributes.put("test", "Hello, World!");

        Assert.assertEquals("Hello, World!", result);

        attributes.put("test2", "World!");

        Assert.assertEquals("Hello, World!\nWorld!", result);

public class AttributeTest {

    private Attributes testAttributes;

    @Before
    public void setUp() {
        testAttributes = new Attributes();
    }

    public void testGetAttributeStringKey() throws Exception {
        String key = "test";
        String value = "Hello, World!";
        Attribute attribute = testAttributes.get(key);
        Assert.assertNotNull(attribute);

        key = "test2";

        attribute = testAttributes.getUserData(key);
        Assert.assertNotNull(attribute);

        key = "test3";

        attribute = testAttributes.remove(key);
        Assert.assertNotNull(attribute);
    }

    public void testGetAttributeStringIgnoreCaseKey() throws Exception {
        String key = "test";
        String value = "Hello, World!";
        Attribute attribute = testAttributes.getIgnoreCase(key);
        Assert.assertNotNull(attribute);

        key = "Test";

        attribute = testAttributes.getUserData(key);
        Assert.assertNotNull(attribute);

        key = "test2";

        attribute = testAttributes.removeIgnoreCase(key);
        Assert.assertNotNull(attribute);
    }

    public void testAddAttributeStringKey() throws Exception {

        attributes.put("test", "Hello, World!");
        String result = attributes.toString();

        Attribute attribute = new Attribute("test", "World!", null, null);

        result = attributes.toString();
        Assert.assertEquals(result + "\n" + attribute.toString(), testAttributes.toString());
    }

    public void testAddAttributeStringKeyIgnoreCase() throws Exception {
        Attributes attributes = new Attributes();
        attributes.putIgnoreCase("test", "Hello, World!");
        String result = attributes.toString();

        Attribute attribute = new Attribute("Test", "World!", null, null);

        result = attributes.toString();
        Assert.assertEquals(result + "\n" + attribute.toString(), testAttributes.toString());
    }

    public void testRemoveAttributeStringKey() throws Exception {

        attributes.add("test", "Hello, World!");
        String result = attributes.remove("test");

        Attribute attribute = new Attribute("test", null, null, null);
