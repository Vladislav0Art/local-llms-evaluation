```java
    private class UrlBuilder {
        private final String uri;
        private final Map<String, Object> params;
        private final Map<String, String> queryParams;

        public UrlBuilder(String url) {
            this.uri = url;
            this.params = new HashMap<>();
            this.queryParams = new HashMap<>();
            parseUrl(url);
        }

        public static class Builder {
            private String uri;
            private Map<String, Object> params = new HashMap<>();
            private Map<String, String> queryParams = new HashMap<>();

            public Builder fromUri(String url) {
                builder.uri(url);
                return this;
            }

            public Builder withParams(Map<String, Object> params) {
                builder.params(params);
                return this;
            }

            public Builder withQueryParams(Map<String, String> queryParams) {
                builder.queryParams(queryParams);
                return this;
            }

            public UrlBuilder build() {
                return new UrlBuilder(uri);
            }
        }

        private void parseUrl(String url) {
            // Parse URL into components
            String[] parts = url.split("/");
            uri = parts[0];
            if (parts.length > 2 && parts[1].equals("http")) {
                params.clear();
                queryParams.clear();
            } else {
                params.clear();
                queryParams.clear();
            }
        }

        public String toString() {
            return "UrlBuilder{" +
                    "uri='" + uri + '\'' +
                    ", params=" + params +
                    ", queryParams=" + queryParams +
                    '}';
        }

        private Map<String, Object> queryParams() {
            // Build query parameters map
            Map<String, String> queryParams = new HashMap<>();
            for (Map.Entry<String, String> entry : params.entrySet()) {
                if (!entry.getKey().equals("_")) {
                    queryParams.put(entry.getKey(), entry.getValue());
                }
            }
            return queryParams;
        }

        private Map<String, Object> params() {
            // Build parameters map
            Map<String, Object> params = new HashMap<>();
            for (Map.Entry<String, String> entry : params.entrySet()) {
                if (!entry.getKey().equals("_")) {
                    params.put(entry.getKey(), entry.getValue());
                }
            }
            return params;
        }

        private String encoding() {
            // Get URL-encoded characters
            return uri.substring(uri.lastIndexOf("))/", uri.length()).trim();
        }

        private String language() {
            // Get URL-encoded language
            return uri.substring(uri.indexOf("lang="), uri.indexOf("&")).trim();
        }
    }
}
```

```java
public class UrlBuilder {

    private final Map<String, Object> params = new HashMap<>();

    public void setParams(Map<String, Object> params) {
        this.params = params;
    }

    public void addParam(String key, String value) {
        params.put(key, value);
    }

    public void addQueryParam(String key, String value) {
        queryParams.put(key, value);
    }

    public String buildUrl() {
        StringBuilder urlBuilder = new StringBuilder();
        for (Map.Entry<String, String> entry : queryParams.entrySet()) {
            urlBuilder.append(entry.getKey()).append("=").append(entry.getValue());
        }
        for (Map.Entry<String, Object> entry : params.entrySet()) {
            if (!queryParams.containsKey(entry.getKey())) {
                urlBuilder.append("").append(entry.getValue());
            } else {
                urlBuilder.append("&").append(entry.getKey()).append("=").append(entry.getValue());
            }
        }
        return urlBuilder.toString();
    }
```

Here is a Java class `UrlBuilder` that constructs URLs based on query parameters and optional parameters:

```java
public class UrlBuilder {

    private final Map<String, String> queryParams = new HashMap<>();
    private final Map<String, Object> params = new HashMap<>();

    public void setParams(Map<String, Object> params) {
        this.params = params;
    }

    public void addParam(String key, String value) {
        params.put(key, value);
    }

    public void addQueryParam(String key, String value) {
        queryParams.put(key, value);
    }

    public String buildUrl() {
        StringBuilder urlBuilder = new StringBuilder();
        for (Map.Entry<String, String> entry : queryParams.entrySet()) {
            urlBuilder.append(entry.getKey()).append("=").append(entry.getValue());
        }
        if (!queryParams.isEmpty()) {
            urlBuilder.append("&");
        }
        for (Map.Entry<String, Object> entry : params.entrySet()) {
            if (!queryParams.containsKey(entry.getKey())) {
                urlBuilder.append("").append(entry.getValue());
            } else {
                urlBuilder.append("&").append(entry.getKey()).append("=").append(entry.getValue());
            }
        }
        return urlBuilder.toString();
    }

    public String buildUrl(String baseString) {
        StringBuilder urlBuilder = new StringBuilder(baseString);
        if (params.size() > 0) {
            for (Map.Entry<String, Object> entry : params.entrySet()) {
                if (!queryParams.containsKey(entry.getKey())) {
                    urlBuilder.append("").append(entry.getValue());
                } else {
                    urlBuilder.append("&").append(entry.getKey()).append("=").append(entry.getValue());
                }
            }
        }
        return urlBuilder.toString();
    }

    public static void main(String[] args) {
        UrlBuilder urlBuilder = new UrlBuilder();

        // Test 1
        Map<String, String> queryParamsMap = new HashMap<>();
        queryParamsMap.put("key1", "value1");
        queryParamsMap.put("key2", "value2");

        urlBuilder.setParams(queryParamsMap);

        System.out.println(urlBuilder.buildUrl());

        // Test 2
        Map<String, Object> paramsMap = new HashMap<>();
        paramsMap.put("key3", "value3");
        urlBuilder.addParam("key4", "value4");
        System.out.println(urlBuilder.buildUrl(paramsMap));

        // Test 3
        String baseString = "https://example.com/path?a=1&b=2";
        System.out.println(urlBuilder.buildUrl(baseString));
    }
}
```

In this code, we have two methods to construct the URL: `buildUrl()` and `buildUrl(String)`. The first method iterates through all query parameters and adds them to the string. If there are no query parameters, it appends an empty string before adding any query parameter strings.

The second method is a wrapper around `buildUrl()`, used when you want to add optional parameters without affecting the existing query parameters.

You can also use this class in your main function:

```java
public static void main(String[] args) {
    UrlBuilder urlBuilder = new UrlBuilder();
    String baseString = "https://example.com/path?a=1&b=2";
    
    System.out.println("Base URL: " + baseString);
    System.out.println("Query Parameters:");
    System.out.println(urlBuilder.buildUrl(baseString));
}
```
