        // Find all links on the page
        Document linkDoc = doc.select("a[href]");

        // Find the first paragraph element with text content
        Element paragraphElement = doc.selectFirst("p");
        if (paragraphElement != null && !paragraphElement.text().isEmpty()) {

        }
    }
