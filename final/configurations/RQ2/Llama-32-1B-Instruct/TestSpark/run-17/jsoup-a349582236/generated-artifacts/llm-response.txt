    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Tag)) return false;

        if (!getTagName().equals(tag.getTagName())) return false;
        if (!getAttributes().equals(tag.getAttributes())) return false;
        if (!tag.settings.equals(settings())) return false;

        return true;
    }

    @Override
    public int hashCode() {
        final int prime = 31;
        int result = 1;

        result = prime * result + getTagName().hashCode();
        result = prime * result + getAttributes().hashCode();

        result = prime * result + (selfClosing() ? 1 : 0);

        return result;
    }

    @Override
    public String toString() {
        StringBuilder builder = new StringBuilder();
        for (Map.Entry<String, Object> entry : getAttributes().entrySet()) {
            if (entry.getValue() instanceof TagValueImpl) {
                builder.append(entry.getKey()).append(": ").append(((TagValueImpl) entry.getValue()).getTagName()).append(", ");
            } else {
                builder.append(entry.getKey()).append(": ").append(entry.getValue());
            }
        }
        return builder.toString().trim();
    }

    public static Map<String, Object> getAttributes() {
        Map<String, Object> attributes = new HashMap<>();

        if (selfClosing()) {
            attributes.put("type", "block");
        } else {
            attributes.put("type", "inline");
        }

        for (Map.Entry<String, Object> entry : getTagSettings().entrySet()) {
            if (entry.getValue() instanceof TagValueImpl) {
                attributes.put(entry.getKey(), ((TagValueImpl) entry.getValue()).getTagName());
            }
        }

        return attributes;
    }

    public static Map<String, String> getTagSettings() {
        Map<String, Object> settings = new HashMap<>();
        for (Map.Entry<String, Object> entry : getAttributes().entrySet()) {
            if (entry.getValue() instanceof TagValueImpl) {
                settings.put(entry.getKey(), ((TagValueImpl) entry.getValue()).getTagName());
            }
        }

        return settings;
    }

    public boolean isBlock() { return selfClosing(); }
    public boolean formatAsBlock() { return !isInline(); }
    public boolean isInline() { return false; }
    public boolean isEmpty() { return true; }
    public boolean isSelfClosing() { return getTagSettings().containsKey("type"); }
    public boolean isKnownTag() { return true; }
    public static boolean isKnownTag(String tagName) {
        if (tagName.equals("img") || tagName.equals("input")) {
            return true;
        }

        try {
            ParseSettings settings = new ParseSettings();
            return valueOf(tagName, settings).isKnownTag();
        } catch (Exception e) {
            return false;
        }
    }

public class Tag {
    public static String getTagName(String tagName) {
        if (tagName.equals("img") || tagName.equals("input")) {
            return tagName;
        }

        try {
            Tag tag = new ParseSettings();
            return valueOf(tagName, tag).getTagName();
        } catch (Exception e) {
            return tagName;
        }
    }

    public static String getAttributes(String tagName) {
        if (tagName.equals("img") || tagName.equals("input")) {
            return null;
        }

        try {
            ParseSettings settings = new ParseSettings();
            return valueOf(tagName, settings).getAttributes().toString();
        } catch (Exception e) {
            return null;
        }
    }

    public static TagValueImpl createTagValue(String tagName, String attributes) {
        if (tagName.equals("img") || tagName.equals("input")) {
            return new TagValueImpl(tagName, new ParseSettings());
        }

        try {
            ParseSettings settings = new ParseSettings();
            return valueOf(tagName, settings).createTagValue(attributes);
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }
}
