```java
import java.util.HashMap;
import java.util.Map;

public class CssFormatter {
    public static String cssMap(Map<String, String> map) {
        StringBuilder sb = new StringBuilder();
        for (Map.Entry<String, String> entry : map.entrySet()) {
            if (sb.length() > 0) {
                sb.append(" ");
            }
            sb.append(entry.getKey()).append(": ").append(entry.getValue());
        }
        return sb.toString().trim();
    }

    public static Map<String, String> getCssMap(String cssContent) {
        Map<String, String> map = new HashMap<>();
        String[] lines = cssContent.split("\n");
        for (String line : lines) {
            if (!line.isEmpty()) {
                String[] parts = line.split(": ");
                if (parts.length == 2) {
                    map.put(parts[0], parts[1]);
                }
            }
        }
        return map;
    }

    public static void main(String[] args) {
        Map<String, String> cssMap = getCssMap("color: blue; background-color: red;");
        System.out.println("Original CSS Map:");
        System.out.println(cssMap);

        String cssContent = "color: blue;\nbackground-color: red;";
        Map<String, String> map = new HashMap<>();
        cssMap = getCssMap(cssContent);
        System.out.println("\nCSS Map after splitting into lines:");
        System.out.println(map);

        // Reverse the order of CSS properties
        String[] reversedProperties = { "color", "background-color" };
        StringBuilder sbReversed = new StringBuilder();
        for (String prop : reversedProperties) {
            if (!cssContent.isEmpty()) {
                String[] parts = cssContent.split(": ");
                if (parts.length > 1 && parts[0].trim().equals(prop)) {
                    sbReversed.append(parts[1]).append("\n");
                }
            }
        }

        System.out.println("\nCSS Map after reversing the order of properties:");
        System.out.println(cssMap);
        
        cssContent = "color: blue;\nbackground-color: red;";
        String[] parts = cssContent.split(": ");
        if (parts.length > 1 && parts[0].trim().equals("background-color")) {
            parts = Arrays.copyOfRange(parts, 1, parts.length);

            // Reorder the properties
            Map<String, String> reversedMap = new HashMap<>();
            for (int i = 0; i < parts.length; i++) {
                if (!reversedProperties.contains(parts[i].trim())) {
                    reversedMap.put("background-color", parts[i]);
                }
            }

            map = getCssMap(reversedMap.toString());
            System.out.println("\nCSS Map after reversing the order of properties:");
            System.out.println(map);
        } else {
            System.out.println("No 'background-color' property found.");
        }
    }
}
```

This program creates a CSS map by splitting the input CSS string into individual lines. It then reverses the order of the properties in the CSS map and prints out the resulting maps after each transformation. The original and reversed maps are printed separately for comparison.
