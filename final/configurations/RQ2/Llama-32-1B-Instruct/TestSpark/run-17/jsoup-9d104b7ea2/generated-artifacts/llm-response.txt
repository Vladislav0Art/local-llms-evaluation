```java
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class TextNodeTest {

    public static String testText = "Hello World";
    public static String testSubstring = "World";

    @Before
    public void setup() {
        document = new Document();
    }

    @Test
    public void test nodeName() {
        assertEquals("TextNode", getNodeName(0));
    }

    @Test
    public void test text() {
        assertEquals(testText, getNodeText(0));
    }

    @Test
    public void test textCreation() {
        TextNode createdNode = createTextNode(testSubstring);
        assertEquals(createdNode, getNode(0));
    }

    @Test
    public void test getWholeText() {
        Document document = new Document();
        appendText(document, testText + " " + testSubstring);
        assertEquals(getWholeText(0), document.getText(0, document.getLength()));
    }

    @Test
    public void test isBlank() {
        assertTrue(isBlank(0));
    }

    private static Element getNode(int index) {
        return document.getElementById(index);
    }

    private static String getNodeName(int index) {
        return getNode(index).nodeName();
    }

    private static String getNodeText(int index) {
        return getNode(index).text();
    }

    private static TextNode createTextNode(String text) {
        Element node = document.createElement("text");
        node.appendChild(document.createTextNode(text));
        return new TextNode(node);
    }

    @Before
    public void setupMockDocument() {
        appendText(document, "Hello World\nWorld");
    }

    private static void appendText(Document document, String text) {
        Element parent = document.body();
        while (parent.parentNode != null && !parent.parentNode.contains(document)) {
            parent = parent.parentNode;
        }
        if (parent == null) {
            parent = document.createElement("body");
            parent.appendChild(document);
        }
        parent.appendChild(document.createTextNode(text));
    }

    @Test
    public void test splitText() {
        TextNode node1 = createTextNode(testText + "\n" + testSubstring);
        TextNode node2 = createTextNode(testSubstring);
        assertEquals(node1, getNode(0));
        assertEquals(node2, getNode(1));
    }

    private static String getWholeText(int index) {
        StringBuilder builder = new StringBuilder();
        appendText(document, "");
        while (index < document.body().length()) {
            if (document.body().contains(index)) {
                builder.append(document.body().substring(index).trim());
            }
            index++;
        }
        return builder.toString();
    }

    private static boolean isBlank(int index) {
        return getNodeName(index).equals("br");
    }

    @Test
    public void test clone() {
        TextNode node1 = createTextNode(testText);
        TextNode node2 = clone(node1);
        assertEquals(node1, getNode(0));
        assertEquals(node2, getNode(0));
    }

    private static TextNode clone(TextNode node) {
        Document document = new Document();
        appendText(document, "");
        Element parent = document.createElement("body");
        parent.appendChild(document);
        parent.appendChild(node);
        return createTextNode(document.body().text());
    }
}
```
