import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        
        while (true) {
            System.out.println("Test 1: Test 2: Test 3:");
            String input = scanner.nextLine();
            
            if (input.equalsIgnoreCase("done")) {
                break;
            }
            
            switch (input.toLowerCase()) {
                case "pass":
                    System.out.print("Enter speed value (max 60): ");
                    int speed = Integer.parseInt(scanner.nextLine());
                    
                    if (speed < 60) {
                        System.out.println("Speed is too slow.");
                    } else {
                        break;
                    }
                    
                    switch (input.toLowerCase()) {
                        case "pass":
                            System.out.print("Enter response time value: ");
                            double responseTime = Double.parseDouble(scanner.nextLine());
                            
                            if (responseTime > 10) {
                                System.out.println("Response time is too long.");
                            } else {
                                break;
                            }
                            
                            switch (input.toLowerCase()) {
                                case "pass":
                                    System.out.print("Enter number of questions: ");
                                    int numQuestions = Integer.parseInt(scanner.nextLine());
                                    
                                    if (numQuestions < 5) {
                                        System.out.println("Too few questions.");
                                    } else {
                                        break;
                                    }
                                
                                    switch (input.toLowerCase()) {

                                            System.out.println("Test 1 failed.");
                                            break;
                                        default:
                                            System.out.println("Invalid input. Exiting...");
                                            return;
                                    }
                                }
                            }

                    }
            }
        }
