public class EdDSASignature {
    private byte[] signature;
    private MessageDigest digest;
    private SecretKeySpec secretKeySpec;

    public EdDSASignature(String data) throws NoSuchAlgorithmException {
        this(signature, data);
    }

    public EdDSASignature(byte[] bytes) throws NoSuchAlgorithmException {
        this.signature = bytes;
        digest = MessageDigest.getInstance("Ed25519");
        secretKeySpec = new SecretKeySpec(digest.digest(), "AES");
