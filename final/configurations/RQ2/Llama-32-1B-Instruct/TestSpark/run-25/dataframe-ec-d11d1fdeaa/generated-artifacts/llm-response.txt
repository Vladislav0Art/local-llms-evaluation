```java
        df1.addColumn(new DataColumn("Color", "String"));
        df1.addColumn(new DataColumn("Weight", "Double"));

        System.out.println("\nDataFrame 1 after adding columns:");
        System.out.println(df1);

        // Add a row with sample data
        DataRow newRow = new DataRow();
        newRow.setName("John");
        newRow.addColumn(new DataColumn("Age", 25));
        newRow.addColumn(new DataColumn("Height", 175.0));
        df1.addRow(newRow);

        System.out.println("\nDataFrame 1 after adding a new row:");
        System.out.println(df1);

        // Join the two DataFrames on the 'Name' column
        DataJoin joinDescriptor = new DataJoin("People");

        DataFrame joinedDf = df1.join(joinDescriptor);
        System.out.println("\nDataFrame 1 after joining:");
```

```java
import java.io.File;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;

public class DataFrameExample {
    public static void main(String[] args) throws IOException {
        // Create two empty DataFrames from text files
        createDataFrameFromExcel("People.xlsx");
        createDataFrameFromTextFile("Fruits.txt", "Fruits.csv");

        System.out.println("\nBoth DataFrames:");
        printDataframe(df1);
        printDataframe(df2);

        // Read the second DataFrame into a new DataFrames Java class
        DataFrames df3 = readDataFrameFromTextFile("Fruits.txt");

        System.out.println("\nDataFrame 3 after reading from text file:");
        printDataframe(df3);
    }

    private static void createDataFrameFromExcel(String filename) throws IOException {
        // Read the Excel file into a XSSFWorkbook
        org.apache.poi.xssf.usermodel.XSSFWorkbook workbook = new org.apache.poi.xssf.usermodel.XSSFWorkbook();
        workbook.readExcel(new File(filename));
    }

    private static void createDataFrameFromTextFile(String filename, String outputFilename) throws IOException {
        // Create a new DataFrames Java class to read the text file
        DataFrame df = new DataFrame("Fruits", new ArrayList<>());
        df.addColumn(new DataColumn("Color", "String"));
        df.addColumn(new DataColumn("Name", "String"));

        // Read the text file into the DataFrame
        String[] lines = new File(filename).readTextLines();
        for (int i = 0; i < lines.length; i++) {
            String line = lines[i];
            if (i == 0) {
                // Use the first line as the header
                String[] headers = line.split(",");
                for (String header : headers) {
                    df.addColumn(new DataColumn(header, "String"));
                }
            } else {
                String[] columns = line.split(",");
                for (int j = 1; j < columns.length; j++) {
                    String column = columns[j];
                    if (!column.isEmpty()) {
                        df.addColumn(new DataColumn(column, "String"));
                    }
                }
            }
        }

        // Save the DataFrame to a new file
        org.apache.poi.xssf.usermodel.XSSFWorkbook workbook = new org.apache.poi.xssf.usermodel.XSSFWorkbook();
        workbook.writeExcel(outputFilename);
    }

    private static void readDataFrameFromTextFile(String filename) {
        String[] lines = new File(filename).readTextLines();

        for (int i = 0; i < lines.length - 1; i++) {
            String line = lines[i];
            if (!line.isEmpty()) {
                String[] columns = line.split(",");
                for (int j = 0; j < columns.length; j++) {
                    if (!columns[j].isEmpty()) {
                        DataFrame df = new DataFrame(columns[j], new ArrayList<>());
                        System.out.println(df);
                    }
                }
            }
        }
    }

    private static void printDataframe(DataFrame dataframe) {
        for (Column column : dataframe.columns()) {
            System.out.print(column.getName() + ": ");
            for (String value : column.getValues()) {
                System.out.print(value + " ");
            }
            System.out.println();
        }
    }
}

    public void addColumn(DataColumn column) {
```

Here's a Java code snippet that demonstrates how to use the Apache POI library to read data from text files and write it into another text file.

```java
import org.apache.poi.ss.usermodel.*;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;

public class TextFileToExcel {
    public static void main(String[] args) throws Exception {
        // Read the first text file into a workbook
        Workbook workbook = new XSSFWorkbook(new FileReader("input.txt"));
        Sheet sheet = workbook.getSheetAt(0);

        // Write the data into another text file
        File outputFile = new File("output.xlsx");
        try (BufferedWriter writer = new BufferedWriter(new FileWriter(outputFile))) {
            for (Row row : sheet) {
                int count = 1;
                for (Cell cell : row) {
                    if (cell instanceof String) {
                        writer.write((count++) + "\n");
                    } else if (cell instanceof Number) {
                        writer.write(cell.toString() + "\n");
                    }
                }
            }
        }

        System.out.println("Data has been successfully written into the output file.");
    }
}

class TextFileToExcelHelper {
    public static void printHeaderSheet(String filename, Sheet sheet) {
        for (int i = 0; i < sheet.getLastRowNum(); i++) {
            Row row = sheet.getRow(i);
            System.out.println(row.getCell(0).getCellType().getName() + " " + (i + 1));
        }
    }

    public static void printDataSheet(String filename, Sheet sheet) throws Exception {
        for (int i = 1; i < sheet.getLastRowNum(); i++) {
            Row row = sheet.getRow(i);
            System.out.println(row.getCell(0).getCellType().getName() + " " + (i + 1));
            int count = 2;
            for (Cell cell : row) {
                if (cell instanceof String) {
                    System.out.print(cell.getStringCellValue() + "\t");
                } else if (cell instanceof Number) {
                    System.out.print(cell.getNumericValue() + "\t");
                }
                count++;
                if (count > 10) {
                    break;
                }
            }
            System.out.println();
        }
    }

    public static void printRowSheet(String filename, Sheet sheet) throws Exception {
        for (int i = 0; i < sheet.getLastRowNum(); i++) {
            Row row = sheet.getRow(i);
            int count = 2;
            for (Cell cell : row) {
                if (cell instanceof String) {
                    System.out.print(cell.getStringCellValue() + "\t");
                } else if (cell instanceof Number) {
                    System.out.print(cell.getNumericValue() + "\t");
                }
                count++;
                if (count > 10) {
                    break;
                }
            }
            System.out.println();
        }
    }

    public static void main(String[] args) throws Exception {
        TextFileToExcelHelper.printHeaderSheet("output.xlsx", workbook.getSheetAt(0));
        TextFileToExcelHelper.printDataSheet("input.txt", workbook.getSheetAt(0));
        TextFileToExcelHelper.printRowSheet("input.txt", workbook.getSheetAt(0));

        // Write the data into another text file
        File outputFile = new File("output.txt");
        try (BufferedWriter writer = new BufferedWriter(new FileWriter(outputFile))) {
            for (int i = 1; i < sheet.getLastRowNum(); i++) {
                Row row = sheet.getRow(i);
                int count = 2;
                for (Cell cell : row) {
                    if (cell instanceof String) {
                        writer.write((count++) + "\n");
                    } else if (cell instanceof Number) {
                        writer.write(cell.toString() + "\n");
                    }
                }
            }
        }
    }
```
