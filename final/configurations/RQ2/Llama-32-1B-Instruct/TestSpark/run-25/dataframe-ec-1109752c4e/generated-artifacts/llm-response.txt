public class PrettyPrintVisitorTest {

    public interface StringPrinter {
        void print(String s);
    }

    public interface PrettyPrintVisitor {
        void visitAssignExpr(AnonymousScriptExpr expr);
        void visitBinaryExpr(BinaryExpr expr);
        void visitUnaryExpr(UnaryExpr expr);
        void visitConstExpr(ConstExpr expr);
        void visitFunctionCallExpr(FunctionCallExpr expr);
        void visitPropertyPathExpr(PropertyPathExpr expr);
        void visitAnonymousScriptExpr(AssingExpr expr);
        void visitStatementSequence(StatementSequence expr);
        void visitProjection(Projection expr);
        void visitVector(Vector expr);
        void visitIfElse	IfElse(ifElse) {
            Assert.assertTrue(expr.visit(ifElse));
        }
    }

    public static class AnonymousScriptExpr implements PrettyPrintVisitor, AnonymousScriptExpr {
        private StringPrinter printer;

        public AnonymousScriptExpr(StringPrinter printer) {
            this.printer = printer;
        }

        @Override
        public void visitAssignExpr(AnonymousScriptExpr expr) {
            printer.print("x = ");
            expr.visit(expr.getExpression());
            printer.print(";");
        }

        @Override
        public void visitBinaryExpr(BinaryExpr expr) {
            printer.print("(");
            for (int i = 0; i < expr.getLeft().size(); i++) {
                printer.print(expr.getLeft().get(i).visit(new VarExpr(printer)));
                if (i < expr.getRight().size() - 1) {
                    printer.print(" ");
                }
            }
            printer.print(")");
        }

        @Override
        public void visitUnaryExpr(UnaryExpr expr) {
            printer.print("(");
            for (int i = 0; i < expr.getOperator().size(); i++) {
                printer.print(expr.getOperator().get(i).visit(new VarExpr(printer)));
                if (i < expr.getRight().size() - 1) {
                    printer.print(" ");
                }
            }
            printer.print(")");
        }

        @Override
        public void visitConstExpr(ConstExpr expr) {
            printer.print("(");
            for (int i = 0; i < expr.getValue().size(); i++) {
                printer.print(expr.getValue().get(i).visit(new VarExpr(printer)));
                if (i < expr.getValue().size() - 1) {
                    printer.print(" ");
                }
            }
            printer.print(")");
        }

        @Override
        public void visitFunctionCallExpr(FunctionCallExpr expr) {
            printer.print("(");
            for (int i = 0; i < expr.getArguments().size(); i++) {
                printer.print(expr.getArguments().get(i).visit(new VarExpr(printer)));
                if (i < expr.getRight().size() - 1) {
                    printer.print(" ");
                }
            }
            printer.print(")");
        }

        @Override
        public void visitPropertyPathExpr(PropertyPathExpr expr) {
            printer.print(expr.getVariable().visit(new VarExpr(printer)));
            for (int i = 0; i < expr.getProperty().size(); i++) {
                printer.print(".");
                if (i < expr.getProperty().size() - 1) {
                    printer.print(" ");
                }
                printer.print(expr.getProperty().get(i).visit(new VarExpr(printer)));
            }
        }

        @Override
        public void visitStatementSequence(StatementSequence expr) {
            for (int i = 0; i < expr.getStatements().size(); i++) {
                this.visit(expr.getStatements().get(i));
                if (i < expr.getStatements().size() - 1) {
                    printer.print(";");
                }
            }
        }

        @Override
        public void visitProjection(Projection expr) {
            printer.print(expr.getVariable().visit(new VarExpr(printer)));
            printer.print(": ");
            for (int i = 0; i < expr.getProperty().size(); i++) {
                printer.print(".");
                if (i < expr.getProperty().size() - 1) {
                    printer.print(" ");
                }
                printer.print(expr.getProperty().get(i).visit(new VarExpr(printer)));
            }
        }

        @Override
        public void visitVector(Vector expr) {
            printer.print("[");
            for (int i = 0; i < expr.getSize().size(); i++) {
                if (i > 0) {
                    printer.print(", ");
                }
                printer.print(expr.getSize().get(i).visit(new VarExpr(printer)));
            }
            printer.println("]");
        }

        @Override
        public void visitIfElse(IFElse ifElse) {
            if (ifElse.getCondition().visit(ifElse)) {
                printer.print("if ");
                for (int i = 0; i < ifElse.getBody1().size(); i++) {
                    this.visit(ifElse.getBody1().get(i));
                    if (i < ifElse.getBody2().size() - 1) {
                        printer.print(" && ");
                    }
                }
                printer.println(": ");
                printer.print(ifElse.getBody2());
            } else {
                printer.print("else: ");
                for (int i = 0; i < ifElse.getBody2().size(); i++) {
                    this.visit(ifElse.getBody2().get(i));
                    if (i < ifElse.getBody1().size() - 1) {
                        printer.print(" && ");
                    }
                }
                printer.println(": ");
                printer.print(ifElse.getBody1());
            }
        }

    }

}

public class VectorTest {

    public static String printVector(Vector vector) {
        StringBuilder sb = new StringBuilder("[");
        for (int i = 0; i < vector.getSize().size(); i++) {
            if (i > 0) {
                sb.append(", ");
            }
            sb.append(vector.getSize().get(i).visit(new VarExpr("x")));
        }
        sb.append("]");
        return sb.toString();
    }

    public static void main(String[] args) {
        PrettyPrintVisitor visitor = new PrettyPrintVisitor() {
            @Override
            public void visitFunctionCallExpr(FunctionCallExpr expr) {
                super.visitFunctionCallExpr(expr);
                System.out.println("expr.getArguments().size(): " + getExpressionArgumentsSize());
            }
        };

        AnonymousScriptExpr anonymousExpr = new AnonymousScriptExpr(visitor);

        StatementSequence statementSequence = new StatementSequence();
        statementSequence.add(anonymousExpr);

        var expr = statementSequence.visit(statementSequence);
        System.out.println(expr.print());

    }

}
