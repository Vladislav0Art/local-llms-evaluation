                        }});
                    }
                }
            } catch (Exception e) {
                System.out.println(e.getMessage());
            }
        }
    }

    public void insert(String tableName, Map<String, Object> values) {

        try (Scanner scanner = new Scanner(System.in)) {
            for (Map.Entry<String, Object> entry : values.entrySet()) {

                if (!columns.length == 2 || !entry.getKey().equals(tableName)) {
                    System.out.println("Error in column count or query.");
                    return;
                }
                List<Object> list = new ArrayList<>();
                for (int i = 0; i < columns.length; i++) {
                    try {
                        list.add(Integer.parseInt(columns[i]));
                    } catch (NumberFormatException e) {
                        System.out.println(e.getMessage());
                        return;
                    }
                }
                table.get(tableName).add(list);
            }
        } catch (Exception e) {
            System.out.println(e.getMessage());
        }
    }

            if (!values.size() == 2 && !values.containsKey(tableName)) {
                System.out.println("Error in column count or query.");
            } else {
                for (Map.Entry<String, Object> entry : values.entrySet()) {
                    String[] columns = scanner.nextLine().split(", ");
                    table.get(tableName).remove(new ArrayList<>(Arrays.asList(entry.getKey())));

                    if (columns.length == 2 && !entry.getKey().equals(tableName)) {
                        try {
                            List<Object> list = new ArrayList<>();
                            for (int i = 0; i < columns.length; i++) {

                            System.out.println(e.getMessage());

                    }

                    updateValues(tableName, entry.getKey(), values);
                }
            }
        } catch (Exception e) {
            System.out.println(e.getMessage());
        }
    }

                return;

    private void updateValues(String tableName, String key, Map<String, Object> values) {

                updateValues(tableName, key, values);
            } else {
                System.out.println("Error in column count or query.");
            }
        }
