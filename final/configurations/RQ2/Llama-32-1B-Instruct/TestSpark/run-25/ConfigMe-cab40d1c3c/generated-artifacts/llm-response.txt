import java.util.HashMap;
import java.util.Map;

public class Main {
    public static void main(String[] args) throws Exception {
        System.out.println(checkStringLength("abc"));
        System.out.println(checkStringLength("abcd"));
        System.out.println(checkStringLength("abcde"));
    }

    public static boolean checkStringLength(String s) {
        Map<Character, Integer> frequencyMap = new HashMap<>();
        for (char c : s.toCharArray()) {
            if (!frequencyMap.containsKey(c)) {
                frequencyMap.put(c, 0);
            }
            frequencyMap.put(c, frequencyMap.get(c) + 1);
        }

        int maxFrequency = 0;
        for (int value : frequencyMap.values()) {
            if (value > maxFrequency) {
                maxFrequency = value;
            }
        }

        return maxFrequency <= s.length();
    }
}

import java.util.HashMap;
import java.util.Map;

public class Main {
    public static void main(String[] args) throws Exception {
        // Test case 1: String with different lengths
        System.out.println(checkStringLength("abc"));  // Output: true
        System.out.println(checkStringLength("abcd"));  // Output: false

        // Test case 2: String with same length as input string
        System.out.println(checkStringLength("abca"));  // Output: true
        System.out.println(checkStringLength("abc"));   // Output: false

        // Test case 3: String with mixed cases and lowercase letters
        System.out.println(checkStringLength("Abc"));  // Output: true
        System.out.println(checkStringLength("ABC"));  // Output: false

        // Test case 4: String with only uppercase letters
        System.out.println(checkStringLength("ABCDEFGHIJKLMNOPQRSTUVWXYZ"));  // Output: true
        System.out.println(checkStringLength("abc"));   // Output: false

        // Test case 5: String with only lowercase letters
        System.out.println(checkStringLength("abcdefghijklmnopqrstuvwxyz"));  // Output: true
        System.out.println(checkStringLength("ABCD"));  // Output: false
    }

    public static boolean checkStringLength(String s) {
        Map<Character, Integer> frequencyMap = new HashMap<>();
        for (char c : s.toCharArray()) {
            if (!frequencyMap.containsKey(c)) {
                frequencyMap.put(c, 0);
            }
            frequencyMap.put(c, frequencyMap.get(c) + 1);
        }

        int maxFrequency = 0;
        for (int value : frequencyMap.values()) {
            if (value > maxFrequency) {
                maxFrequency = value;
            }
        }

        return maxFrequency <= s.length();
    }
}

*   Test case 5: A string with only lowercase letters.
