import org.jsoup.parser.XmlParser;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.DocumentType;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.Entity;
import org.jsoup.nodes.Node;
import org.jsoup.nodes.TextNode;

public class XmlTreeBuilderTest {
    @Before
    public void setup() {
        // Arrange
        String input = "<root><person><name>John</name><age>30</age></person></root>";
        Document document = new Document();
        document.parse(input);
        Element element = document.getDocumentElement();

        Mockito.mockStatic(XmlParser.class);

        XmlDeclaration xmlDeclaration = new XmlDeclaration("1.0", "UTF-8");
        Mockito.when(XmlParser.parseString(xmlDeclaration.toString())).thenReturn(document);
    }

    public void testParse() throws Exception {
        // Act and Assert
        String result = xmlTreeBuilder.parse(element, "");
        assertEquals("<root><person><name>John</name><age>30</age></person></root>", result);
    }

    public void testInsert() throws Exception {
        // Arrange
        Element element = new Element(document, "test");
        Token token = new Token("test");

        // Act and Assert
        String result = xmlTreeBuilder.insert(token);
        assertEquals("<!DOCTYPE html><html><head><title>Test</title></head><body><p test='test'>Value</p></body></html>", result);
    }

    public void testParseFragment() throws Exception {
        // Act and Assert
        String result = xmlTreeBuilder.parseFragment(new Comment(document, "test"), "");
        assertEquals("<root><person><name>John</name><age>30</age></person></root>", result);
    }

    public void testInsertNode() throws Exception {
        // Arrange
        String input = "<person><name>John</name><age>30</age></person>";
        Element element = new Element(document, "test");
        Node node = new Entity("new-tag", "<new-value>" + input + "</new-value>");

        // Act and Assert
        String result = xmlTreeBuilder.insert(node, "");
        assertEquals("<!DOCTYPE html><html><head><title>Test</title></head><body><p test='test' content='value'/><span new-tag='new-value'>Value</span></body></html>", result);
    }

    public void testParseEntity() throws Exception {
        // Act and Assert
        String result = xmlTreeBuilder.parse(new Comment(document, "test"), "");
        assertEquals("<new-value>value<entity>value</entity>", result.substring(1));
    }

    public void testInsertEntity() throws Exception {
        // Arrange
        Element element = new Element(document, "test");
        Token token = new Token("entity");

        // Act and Assert
        String result = xmlTreeBuilder.insert(token);
        assertEquals("value<new-value>", result.substring(4));
    }
}
