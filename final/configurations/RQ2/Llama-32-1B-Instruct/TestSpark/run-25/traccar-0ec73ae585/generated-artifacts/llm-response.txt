public class SocketAddress {
    private String host;
    private int port;

    public SocketAddress(String host, int port) {
        this.host = host;
        this.port = port;
    }

    public String getHost() {
        return host;
    }

    public int getPort() {
        return port;
    }
}

public class ByteBuf {
    private byte[] buffer;

    public ByteBuf(int capacity) {
        buffer = new byte[capacity];
    }

    public void set(int index, byte value) {
        System.arraycopy(buffer, index, buffer, index + 1, Math.min(4 - (index & 3), 4));
        buffer[index] = value;
    }

    public byte get(int index) {
        if (0 <= index && index < 4) {
            return buffer[index];
        } else if (index >= 4 && index < 8) {
            int offset = index - 4;
            return ((byte) buffer[offset + 3] << 24) | ((byte) buffer[offset + 2] << 16) |
                   ((byte) buffer[offset + 1] << 8) | (byte) buffer[offset];
        } else if (index >= 12 && index < 16) {
            int offset = index - 12;
            return (byte) ((buffer[offset] & 0xFF0000) >> 24) |
                   ((byte) ((buffer[offset] & 0x00FF00) << 8) | (byte) ((buffer[offset] & 0xFF0000)));
        } else if (index >= 20 && index < 24) {
            int offset = index - 20;
            return ((byte) (buffer[offset] & 0xFF)) >> 8 |
                   ((byte) (buffer[offset + 1] & 0xFF));
        }
    }

    public boolean contains(int index, byte value) {
        if (0 <= index && index < 4) {
            return true;
        } else if (index >= 4 && index < 8) {
            int offset = index - 4;
            return buffer[offset + 3] == value || buffer[offset + 2] == value ||
                   buffer[offset + 1] == value || buffer[offset] == value;
        } else if (index >= 12 && index < 16) {
            int offset = index - 12;
            return ((buffer[offset] & 0xFF0000) >> 24) != 0 &&
                   ((buffer[offset] & 0x00FF00) << 8) == value ||
                   ((buffer[offset] & 0xFF0000)) == value || (buffer[offset + 1] & 0xFF) == value;
        } else if (index >= 20 && index < 24) {
            int offset = index - 20;
            return buffer[offset] != value;
        }
    }

    public byte[] toByteBuf() {
        ByteBuf buf = new ByteBuf(buffer.length);
        for (int i = 0; i < buffer.length; i += 4) {
            if (i + 3 >= buffer.length) {
                buf.set(i, buffer[i]);
            } else {
                buf.set(i, buffer[i] & 0xFF);
                if ((buffer[i + 1] & 0x80) != 0) {
                    buf.set(i + 1, (byte) ((buffer[i] & 0xFF0000) >> 24));
                    buf.set(i + 2, (byte) ((buffer[i] & 0x00FF00) << 8));
                } else if ((buffer[i + 1] & 0xE0) != 0) {
                    buf.set(i + 1, (byte) ((buffer[i] & 0xFF0000) >> 16));
                    buf.set(i + 2, (byte) ((buffer[i] & 0x00FF00) << 8));
                } else if ((buffer[i + 1] & 0xF0) != 0) {
                    buf.set(i + 1, buffer[i]);
                    buf.set(i + 2, buffer[i] & 0xFF);
                    buf.set(i + 3, buffer[i] >> 8);
                }
            }
        }
        return buf.toByteBuf();
    }

    public static SocketAddress createSocketAddress(String host, int port) {
        return new SocketAddress(host, port);
    }
}

public class ByteBufUtil {
    public static void set(int index, byte value, ByteBuf buffer) {
        if (index >= 0 && index < 4) {
            buffer.set(index, value);
        } else if (index >= 4 && index < 8) {
            int offset = index - 4;
            buffer.set(offset + 3, value);
        } else if (index >= 12 && index < 16) {
            int offset = index - 12;
            buffer.set(offset, value);
        }
    }

    public static byte[] toByteBuf(ByteBuf buffer) {
        return buffer.toByteBuf();
    }

    public static String toString(ByteBuf buffer) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < buffer.buffer.length; i += 4) {
            if (i + 3 >= buffer.buffer.length) {
                sb.append(buffer.get(i));
            } else {
                sb.append((char) ((buffer.get(i) & 0xFF0000) >> 24));
                sb.append((char) ((buffer.get(i) & 0x00FF00) << 8));
                sb.append((char) (buffer.get(i + 1)));
                sb.append((char) (buffer.get(i + 2) >> 8));
            }
        }
        return sb.toString();
    }

    public static boolean contains(int index, byte value, ByteBuf buffer) {
        if (index >= 0 && index < 4) {
            return true;
        } else if (index >= 4 && index < 8) {
            int offset = index - 4;
            return ((buffer.get(offset) & 0xFF0000) >> 24) == value ||
                   ((buffer.get(offset) & 0x00FF00) << 8) == value ||
                   buffer.get(offset + 1) == value ||
                   buffer.get(offset + 2) == value;
        } else if (index >= 12 && index < 16) {
            int offset = index - 12;
            return ((buffer.get(offset) & 0xFF0000) >> 24) != 0 &&
                   ((buffer.get(offset) & 0x00FF00) << 8) == value ||
                   ((buffer.get(offset) & 0xFF0000)) == value || (buffer.get(offset + 1) & 0xFF) == value;
        } else if (index >= 20 && index < 24) {
            int offset = index - 20;
            return buffer.get(offset) != value;
        }
    }

    public static void set(int index, byte value, ByteBuf buffer) {
        set(index, value, new ByteBuf(buffer.buffer.length));
    }
}

Here's a complete Java program that demonstrates how to use `SocketAddress` and `ByteBuf` classes:

```java
import java.net.InetAddress;
import java.net.UnknownHostException;

public class Main {
    public static void main(String[] args) throws UnknownHostException, SocketException {
        System.out.println("Using host: localhost");
        String ipAddress = "127.0.0.1";
        int port = 8080;

        SocketAddress address = createSocketAddress(ipAddress, port);

        System.out.println(address.getHost());
        System.out.println(address.getPort());

        byte[] ipBytes = getIp(address.getHost(), address.getPort());
        for (byte b : ipBytes) {
            System.out.print(String.format("%02x ", b));
        }
        System.out.println();

        byte[] portBytes = getPort(ipAddress, port);
        for (byte b : portBytes) {
            System.out.print(String.format("%02x ", b));
        }
        System.out.println();
    }

    public static SocketAddress createSocketAddress(String host, int port) throws UnknownHostException {
        return new SocketAddress(host, port);
    }

    public static byte[] getIp(SocketAddress address) {
        return ByteUtil.toByteBuf(address);
    }

    public static byte[] getPort(String ipAddress, int port) {
        try {
            InetAddress address = InetAddress.getByName(ipAddress);
            return ByteUtil.toByteBuf(new ByteBuf(1024));
        } catch (UnknownHostException e) {
            throw new UnknownHostException("Unknown host: " + ipAddress);
        }
    }
}
```

This program creates a `SocketAddress` object from the local IP address and port, gets the IP address of the socket using `getIp()`, converts the socket's IP address to bytes, prints it out.
