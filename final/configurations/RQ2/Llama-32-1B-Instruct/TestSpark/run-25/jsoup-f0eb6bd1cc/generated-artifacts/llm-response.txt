public class URLBuilder {
    private String url;

    public URLBuilder(String url) {
        this.url = url;
    }

    public String build() {
        return url;
    }
}

public class TestURLBuilder {

    @Test
    public void testBuildStringToURL() {
        URLBuilder builder = new URLBuilder("https://www.example.com");
        assertEquals("https://www.example.com", builder.build());
    }

    @Test
    public void testBuildEmptyStringToNull() {
        URLBuilder builder = new URLBuilder("");
        assertNotNull(builder.build());
    }
}

import org.jsoup.Jsoup;

public class JSoupHelper {

    public static String getJsoupUrl(String url) {
        try {

        }

        System.out.println(builder.build().toString());

    public void testBuildEmptyStringToNull() {

            System.out.println("URLBuilder has null value");
        }
    }

    public static void main(String[] args) {
