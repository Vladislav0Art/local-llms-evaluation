import com.crowdin.client.core.http.impl.json.JsonSerializer;
import com.fasterxml.jackson.core.JsonProcessingException;

public class TestJacksonJsonTransformerImpl {

    private final ObjectMapper mapper = new ObjectMapper();
    private final JsonSerializer<Project> serializer = new JsonSerializer<>();

    @Before
    public void setup() {
        Project project = new Project();
    }

    @Test
    public void testEquality() {
        try {
            assertEquals("Value", mapper.valueToBean(serializer.transform(mapper.valueToValue(project)), Project.class));
        } catch (JsonProcessingException e) {
            fail("Failed to parse JSON: " + e.getMessage());
        }
    }

    @Test
    public void testToString() {
        String json = "{\"key\":\"value\"}";
        try {
            Project project = mapper.convertValue(json, Project.class);
            assertEquals("\"value\"", serializer.transform(mapper.valueToBean(project), String.class).getValue().toString());
        } catch (JsonProcessingException e) {
            fail("Failed to parse JSON: " + e.getMessage());
        }
    }

    public static class Project {
        private final String key;

        public Project() {
            this.key = null;
        }

        public String getKey() {
            return key;
        }

        public void setValue(String json) {
            try {
                this.key = serializer.transform(mapper.valueToBean(json, String.class), String.class).getValue();
            } catch (JsonProcessingException e) {
                fail("Failed to parse JSON: " + e.getMessage());
            }
        }
    }
}
