        outputs.put("output1", "Hello world");
        outputs.put("output2", "123 abc");

        }
    }

    public static String process(String input) {
        if (inputs.containsKey(input)) {
            return outputs.get(input);
        } else {
            return "";
        }
    }

    public static String greet(String name) {
