        }

        Node cloned = clone(node);
        assertNodeIsEqual(cloned, newNode);
    }

    private void assertSplitText(Node node, int offset) {
        Node[] splitNodes = splitText(offset, null);
        assertEquals(1, splitNodes.length);

        TextNode splitTextNode = splitNodes[0];
        assertNodeIsEqual(node, splitTextNode);
    }

    private void assertGetWholeText(Node node) {
        String text = getNodeWholeText(node);
        assertTrue(text.contains("Hello, World!"));
    }

    private static String getNodeName(Node node) {
        return node.nodeName();
    }

    private static String getNodeToString(Node node) {
        StringBuilder sb = new StringBuilder();
        printNode(sb, node);
        return sb.toString();
    }

    private void printNode(StringBuilder sb, Node node) {
        if (node.isText()) {
            sb.append(node.text());
        } else if (node.isElement() && !node.isSelfClosing()) {
            for (int i = 0; i < node.attributes(); i++) {
                String name = node.attributes().get(i).getName();
                sb.append(name);
                sb.append("=");
                String value = node.attributes().get(i).getValue();
                if (!value.isEmpty()) {
                    sb.append(":").append(value);
                }
            }

            for (int i = 0; i < node.children(); i++) {
                printNode(sb, node.children()[i]);
            }
        } else {
            for (int i = 0; i < node.attributes(); i++) {
                String name = node.attributes().get(i).getName();
                sb.append(name);
            }

            if (!node.isSelfClosing()) {
                sb.append(">");
            } else {
                sb.append("</");
            }
        }
    }

    private static void assertNodeIsEqual(Node node1, Node node2) {
        String text = getNodeToString(node1);

        if (text.length() != getNodeToString(node2)) {
            fail("Text strings do not match: " + text + " vs " + getNodeToString(node2));
        }

        printNode(sb, node1);
        printNode(sb, node2);

        if (sb.toString().length() > 1000) {
            fail("Long text strings do not match: " + sb.toString());
        }
    }
}

        this.tag += " " + text;

        return "<" + tag + ">" + text + "</" + tag + "]";
    }
}

    public String toString() {
        return "<" + tag + ">";
    }
}

public class GeneratedTest {

    public Document body = new Document("html");
    public Document head = new Document("head");

    @Test
    public void testText() {
        document.body.setText("Hello, World!");
        assertEquals("<body>Hello, World!</body>", document.body.toString());
    }

    @Test
    public void testHead() {
        body.setTag("title") ;
        String actualHtml = "<html><head title='Home Page'><body>Hello, World!</body></head></html>";
        head.setText(actualHtml);
        assertEquals("<head title='Home Page'><body>Hello, World!</body></head>", head.toString());
    }

    @Test
    public void testAppendChild() {
        document.body.appendHead(body);
        String actualHtml = "<html><head><body>Hello, World!</body></head></html>";
        document.setTag("title") ;
        head.setText(actualHtml);
        assertEquals("<head title='Home Page'><body>Hello, World!</body></head>", head.toString());
    }

}
