    // Method to test public method

        return x;

    // Function to test public method with parameter
    public static void testPublicMethodWithParam(String str, int y) {
        System.out.println(str + " + " + y);
    }

    // Function to test private class
    public class TestPrivateClass {
        // Function to test private function
        public static String testPrivateFunction() {
            return "This is a private function";
        }
    }

    // Method to test private method with parameter
    public static int testPrivateMethod(int z) {
        return z;
    }

    // Public class and methods
    public static class PublicClass {
        public static String testPublicClass1() {
            return "This is a public class";
        }

        public static int testPublicMethod2() {
            return 10;
        }
    }

    public static void main(String[] args) {
        System.out.println(testPublicClass());
        System.out.println(testPublicMethod(20));
        System.out.println(testPublicMethodWithParam("Hello", 10));

        TestPrivateClass testPrivateClass = new TestPrivateClass();
        System.out.println(testPrivateClass.testPrivateFunction());

        public class PrivateClass {
            private int x;

            public PrivateClass(int z) {
                this.x = z;
            }

            public void testPrivateMethod1() {
                return "This is a private method";
            }
        }

        PublicClass.publicClass();
    }
