package org.davidmoten.text.utils;

import java.util.ArrayList;
import java.util.List;

public class WordWrapTests {

    public static void main(String[] args) {
        new WordWrapTests().testWordWrap_Reader();
        new WordWrapTests().testWordWrap_Lowercase();
        new WordWrapTests().testWordWrap_RightTrim();
        new WordWrapTests().testWordWrap_IsWhitespace();
        new WordWrapTests().testWordWrap_LeftTrim();
    }

    public class Matcher {

        private String actual;
        private String expected;

        public Matcher(String actual, String expected) {
            this.actual = actual;
            this.expected = expected;
        }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (!(o instanceof Matcher)) return false;
            Matcher matcher = (Matcher) o;
            return actual.equals(matcher.actual);
        }

        @Override
        public int hashCode() {
            return java.util.Arrays.hashCode(actual, expected);
        }
    }

    public class WordWrap {

        private List<Character> characters;

        public WordWrap(String input) {
            this.characters = new ArrayList<>();
            for (char c : input.toCharArray()) {
                characters.add(c);
            }
        }

        public String wordWrap(List<Character> list, int length, Matcher matcher) {
            StringBuilder2 sb = new StringBuilder2("");
            List<Character> currentList = new ArrayList<>();

            for (int i = 0; i < list.size(); i++) {
                if (!currentList.isEmpty() && !matcher.equals(currentList.get(0), list.get(i))) {
                    sb.append(currentList);
                    currentList.clear();
                }
                currentList.add(list.get(i));
            }

            if (!currentList.isEmpty()) {
                sb.append(currentList);
            } else {
                for (char c : characters) {
                    sb.append(c);
                }
            }

            return sb.toString();
        }

        public Matcher leftTrim(StringBuilder2 word) {
            return new Matcher(word, this.wordWrap(List.of(), length -> 1, matcher -> true));
        }
    }
}

package org.davidmoten.text.utils;

import java.util.ArrayList;
import java.util.List;

public class WordWrapTests {

    public static void main(String[] args) {
        new WordWrapTests().testWordWrap_Reader();
        new WordWrapTests().testWordWrap_Lowercase();
        new WordWrapTests().testWordWrap_RightTrim();
        new WordWrapTests().testWordWrap_IsWhitespace();
        new WordWrapTests().testWordWrap_LeftTrim();
    }

    public class Matcher {

        private String actual;
        private String expected;

        public Matcher(String actual, String expected) {
            this.actual = actual;
            this.expected = expected;
        }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (!(o instanceof Matcher)) return false;
            Matcher matcher = (Matcher) o;
            return actual.equals(matcher.actual);
        }

        @Override
        public int hashCode() {
            return java.util.Arrays.hashCode(actual, expected);
        }
    }

    public class WordWrap {

        private String original;

        public WordWrap(String input) {
            this.original = input;
        }

        public String wordWrap(List<Character> list, int length, Matcher matcher) {
            List<Character> currentList = new ArrayList<>();
            StringBuilder2 sb = new StringBuilder2("");
            StringBuilder2 leftTrimmedSB = new StringBuilder2("");

            for (int i = 0; i < list.size(); i++) {
                if (!currentList.isEmpty() && !matcher.equals(currentList.get(0), list.get(i))) {
                    if (sb.length() + currentList.size() > length) {
                        sb.append(currentList);
                        currentList.clear();
                    }
                    currentList.add(list.get(i));
                } else {
                    if (sb.length() == 0) {
                        sb.append(list.get(i));
                    } else {
                        sb.append(currentList.get(0));
                        currentList.remove(0);
                    }
                }
            }

            if (!currentList.isEmpty()) {
                if (sb.length() > 0 && matcher.equals(sb, list)) {
                    return new Matcher(sb.toString(), original);
                } else {
                    leftTrimmedSB.append(list.get(0));
                    sb = leftTrimmedSB;
                }
            }

            while (sb.length() > 0) {
                currentList.add(sb.charAt(0));
                sb = new StringBuilder2(sb.substring(1));
            }

            return new Matcher(sb.toString(), original);
        }

        public Matcher leftTrim(StringBuilder2 word) {
            return new Matcher(word, this.wordWrap(List.of(), length -> true, matcher -> true));
        }
    }

    public class String {

        private char[] array;

        public String(String string) {
            this.array = string.toCharArray();
        }

        @Override
        public int hashCode() {
            return java.util.Arrays.hashCode(array);
        }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (!(o instanceof String)) return false;
            String s = (String) o;
            return array.equals(s.array);
        }
    }
}
