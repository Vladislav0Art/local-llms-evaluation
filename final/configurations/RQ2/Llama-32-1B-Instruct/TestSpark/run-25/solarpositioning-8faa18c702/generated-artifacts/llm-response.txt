        assertEquals(6.2e-06, estimate(LocalDate.parse("2024-12-31")), 0.01);
    }

    @Test
    public void TestForNoSolarPositioning() {
        assertEquals(7.9e-07, estimate(LocalDate.parse("2024-01-01")), 0.01);
    }

    @Test
    public void TestForHighAltitude() {
        assertEquals(6.5e-06, estimate(LocalDate.parse("2024-03-30")), 0.01);
    }

    @Test
    public void TestForLowSolarPositioning() {
        assertEquals(8.9e-07, estimate(LocalDate.parse("2024-01-01")), 0.01);
    }
}

public class Helper {

    public static double pow(double a, int b) {
        return Math.pow(a, b);
    }

    public static void main(String[] args) {
        DeltaT deltaT = new DeltaT();

        LocalDate forDate1 = LocalDate.parse("2024-03-01");
        assertEquals(8.5e-05, deltaT.estimate(forDate1), 0.01);

        LocalDate forDate2 = LocalDate.parse("2024-12-31");
        assertEquals(6.2e-06, deltaT.estimate(forDate2), 0.01);

        LocalDate forDate3 = LocalDate.parse("2024-01-01");
        assertEquals(7.9e-07, deltaT.estimate(forDate3), 0.01);

        LocalDate forDate4 = LocalDate.parse("2024-03-30");
        assertEquals(6.5e-06, deltaT.estimate(forDate4), 0.01);

        LocalDate forDate5 = LocalDate.parse("2024-01-01");
        assertEquals(8.9e-07, deltaT.estimate(forDate5), 0.01);
    }
}
