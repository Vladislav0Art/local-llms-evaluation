        if (this.mockPropertyList == null) {

            return this.mockPropertyList.getRootEntries().stream()

                    .collect(Collectors.toList());
        }
    }

    interface Pair<K, V> {
        K getFirst();
        V getSecond();
    }
}

    private List<String> rootEntries = Collections.emptyList();

    public List<String> getRootEntries() {
        return this.rootEntries;
    }

    @Override
    public void setRootEntries(List<String> rootEntries) {
        this.rootEntries = rootEntries;
    }

import java.util.ArrayList;
import java.util.List;

public class MockObject {
    public List<String> getRootEntries() {
        return new ArrayList<>();
    }

    public List<MockProperty> create() {
        return new ArrayList<>();
    }
}

class TestHelper {
    private MockObject mockObject = new MockObject();

    public String test1() {
        List<MockProperty> properties = mockObject.getRootEntries();
        assert properties.size() == 2;
        for (int i = 0; i < 2; i++) {
            if (i == 0) {
                assert properties.get(i).getKey() == "test1";
            } else {
                assert properties.get(i).getValue() == "test1-2";
            }
        }

        List<MockProperty> multipleProperties = mockObject.create();
        for (int i = 0; i < 3; i++) {
            if (i == 0) {
                multipleProperties.add(new MockProperty("test1-2", "test3"));
            } else if (i == 1) {
                multipleProperties.add(new MockProperty("test4-5", "test6"));
            } else {
                multipleProperties.add(new MockProperty("test7-8", "test9"));
            }
        }

        assert multipleProperties.size() == 2;
        for (int i = 0; i < 2; i++) {
            if (i == 0) {
                assert multipleProperties.get(i).getKey().equals("test4-5");
            } else {
                assert multipleProperties.get(i).getValue().equals("test7-8");
            }
        }

        return mockObject.getRootEntries();
    }
}

public class Test {
    private TestHelper testHelper = new TestHelper();

    public void testGetRootEntries() {
        List<String> result = testHelper.test1();
        System.out.println(result);
    }

    public static void main(String[] args) {
        Test test = new Test();
        test.testGetRootEntries();
    }
}

class MockProperty {
    private String key;
    private String value;

    public MockProperty(String key, String value) {
        this.key = key;
        this.value = value;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        MockProperty that = (MockProperty) o;
        return Objects.equals(key, that.key) && Objects.equals(value, that.value);
    }

    @Override
    public int hashCode() {
        return Objects.hash(key, value);
    }
}
