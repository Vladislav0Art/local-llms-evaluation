public class CommentTest {

    private Comment comment;

    public void testCommentNodeName() {
        when(comment.nodeName()).thenReturn("comment");
        assertEquals("comment", comment.nodeName());
    }

    public void testCommentData() {
        when(comment.getData()).thenReturn("example data");
        assertEquals("example data", comment.getData());
    }

    public void testCommentSetData() {
        String data = "example data";
        comment = new Comment(data);
        assertNotEquals("", comment.getData());
        assertEquals("example data", comment.getData());
    }

    public void testOuterHtmlHead() throws IOException {
        when(comment.outerHtmlHead(Mockito.any(), Mockito.anyInt(), Mockito.anyDocumentOutputSettings())).thenReturn("");
        String expected = "<!-- comment -->";
        assertEquals(expected, comment.outerHtmlHead(Appendable.EMPTY, 0, Document.OutputSettings.INDENT_NO_SPACE));
    }

    public void testOuterHtmlTail() throws IOException {
        when(comment.outerHtmlTail(Mockito.any(), Mockito.anyInt(), Mockito.anyDocumentOutputSettings())).thenReturn("");
        String expected = "<!-- -->";
        assertEquals(expected, comment.outerHtmlTail(Appendable.EMPTY, 0, Document.OutputSettings.INDENT_NO_SPACE));
    }

    public void testToString() {
        when(comment.toString()).thenReturn("example");
        assertEquals("example", comment.toString());
    }

    public void testCommentClone() {
        comment = new Comment("example data");
        Comment clonedComment = comment.clone();
        assertNotEquals(comment, clonedComment);
        assertSame(comment, clonedComment);
    }

    public void testIsXmlDeclaration() {
        when(comment.isXmlDeclaration()).thenReturn(false);
        assertTrue(comment.isXmlDeclaration());
    }

    public void testAsXmlDeclaration() {
        String xmlDeclaration = "<!-- comment -->";
        Comment asComment = comment.asXmlDeclaration();
        assertSame(xmlDeclaration, asComment.asXmlDeclaration());
    }
}

public class XmlDeclarationTest {

    private String xmlDeclaration;

    public void testAsXmlDeclarationContent() {
        xmlDeclaration = "<!-- comment -->";
        when(comment.asXmlDeclaration()).thenReturn(xmlDeclaration);
        assertEquals("<!-- comment -->", comment.asXmlDeclaration());
    }

    public void testAsXmlDeclarationEmpty() {
        String emptyDeclaration = "";
        when(comment.asXmlDeclaration()).thenReturn(emptyDeclaration);
        assertEquals(emptyDeclaration, comment.asXmlDeclaration());
    }

    public void testIsXmlDeclarationWithXmlDeclaration() {
        when(comment.isXmlDeclaration()).thenAnswer(a -> a.invoke(xmlDeclaration)).thenReturn(true);
        assertTrue(comment.isXmlDeclaration());
    }
}
