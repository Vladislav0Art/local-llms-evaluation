import org.jsoup.helper.UrlBuilderTestUtils;
import org.jsoup.matcher.KeyMatcher;
import org.junit.Test;

public class UrlBuilderTest {

    private final UrlBuilder urlBuilder = new UrlBuilder(new URL("http://example.com"));

    @Test
    public void testAppendKeyVal() {
        KeyMatcher keyMatcher = KeyMatcher.stringToKey("param1=value1");
        String input = "param1=value1";
        String expected = "param1=value1";
        urlBuilder.appendKeyVal(keyMatcher, input);
        assertEquals(expected, DataUtil.decode(input));
    }

    @Test
    public void testAppendKeyValMalformed() {
        KeyMatcher keyMatcher = KeyMatcher.stringToKey("param1=value1");
        String input = "param1=value1";
        urlBuilder.appendKeyVal(keyMatcher, input);
        assertEquals("error", DataUtil.decode(input));
    }

    @Test
    public void testAppendKeyValNull() {
        KeyMatcher keyMatcher = KeyMatcher.stringToKey("param1=value1");
        String input = null;
        urlBuilder.appendKeyVal(keyMatcher, input);
        assertTrue(urlBuilder.build().isEmpty());
    }

    @Test
    public void testAppendKeyValURI() throws UnsupportedEncodingException {
        URL inputUrl = new URL("https://example.com");
        KeyMatcher keyMatcher = KeyMatcher.stringToKey("param1=value1");
        urlBuilder.appendKeyVal(keyMatcher, inputUrl);
        assertEquals(inputUrl.toURI(), DataUtil.encode(urlBuilder.build()));
    }

    @Test
    public void testAppendKeyValURINotMalformed() throws UnsupportedEncodingException {
        URL inputUrl = new URL("https://example.com");
        KeyMatcher keyMatcher = KeyMatcher.stringToKey("param1=value1");
        String expected = "https://example.com?param1=value1";
        urlBuilder.appendKeyVal(keyMatcher, inputUrl);
        assertEquals(expected, DataUtil.encode(urlBuilder.build()));
    }

    @Test
    public void testAppendKeyValException() {
        KeyMatcher keyMatcher = KeyMatcher.stringToKey("param1=value1");
        try {
            urlBuilder.appendKeyVal(keyMatcher, "non-existent-key");
            fail("Expected exception not thrown");
        } catch (UnsupportedEncodingException e) {
            // Expected
        }
    }

}

import java.net.URL;
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class DataUtil {
    public static String encode(String url) throws UnsupportedEncodingException {
        URL parsedUrl = new URL(url);
        StringBuilder sb = new StringBuilder();
        try {
            Document document = Jsoup.connect(parsedUrl).get();
            for (org.jsoup.nodes.Element element : document.getAllElements()) {
                if (!element.tagName().equals("meta")) {
                    String attrName = element.attr("name");
                    String attrValue = element.attr(attrName);
                    sb.append("-").append(attrName).append("=").append(attrValue).append(",");
                }
            }
        } catch (Exception e) {
            return "Error encoding URL: " + e.getMessage();
        }
        return sb.toString().trim() + "/";
    }

    public static void main(String[] args) throws UnsupportedEncodingException {
        System.out.println(encode("https://example.com"));
        System.out.println(encode(new java.net.URL("https://example.com")));
        try {
            new URL("https://example.com").toURI();
        } catch (java.net.URISyntaxException e) {
            // ignore
        }
    }

    public static void test1() throws UnsupportedEncodingException {
        String expected = "https%3A//example.com%2F";
        System.out.println(encode(expected));
    }

    public static void test2() throws UnsupportedEncodingException {

import java.io.File;
import java.net.FileNameMap;
import java.net.URLConnection;

public class URLUtil {
    public static String encode(String url) throws Exception {
        // Set default encoding to UTF-8
        FileNameMap fileNameMap = URLConnection.getFileNameMap();
        String filename = fileNameMap.getNameForPath(url);
        if (filename != null && !filename.isEmpty()) {
            file = new File(filename);
        }

        // Get the file's size and last modification time
        long fileSize = file.length();
        java.io.File fileObject = file;

        // Encode the URL
        String encodedUrl = url;

        return encodedUrl + "-" + fileSize + "/" + fileObject.getName();
    }

    public static void main(String[] args) throws Exception {
        System.out.println(encode("https://example.com"));
        System.out.println(encode(new java.net.URL("https://example.com")));
        test1();
    }

    public static void test1() throws Exception {
        System.out.println(encode("https://example.com"));
    }

        String actual = encode("https://example.com");
        if (!actual.equals("https%3A//example.com%2F")) {
            throw new Exception("Expected exception not thrown");

            URLConnection connection = new java.net.URL("https://example.com").openConnection();
        } catch (java.net.URISyntaxException e) {
            // ignore
        }
    }
}

    }
}
