import java.io.File;
import java.util.ArrayList;

public class EpubChecker {

    public boolean validateFile(String path, String version, Report report, EPUBProfile profile) {
        File file = new File(path);
        if (!file.exists()) {
            return false;
        }

        String fileName = file.getName();
        int extensionIndex = fileName.lastIndexOf(".");
        String filenameWithoutExtension = fileName.substring(0, extensionIndex);

        EpubChecker.EpubVersion versionNumber;

        try {
            versionNumber = new EpubChecker.EpubVersion(version);
        } catch (Exception e) {
            return false;
        }

        if (!versionNumber.equals(new EpubChecker.EpubVersion("1.2"))) {
            return false;
        }

        int versionNumberIndex = filenameWithoutExtension.length();
        String majorVersion = "";
        while (majorVersion == null && versionNumberIndex > 0) {
            char c = filenameWithoutExtension.charAt(versionNumberIndex - 1);
            if (!Character.isDigit(c)) {
                break;
            }
            majorVersion = new String(new char[] {c}).replace("\u0000", "").toLowerCase();
            versionNumberIndex -= 2;
        }

        int minorVersion = 0;

        while (minorVersion == null && versionNumberIndex > 1) {
            char c = filenameWithoutExtension.charAt(versionNumberIndex - 1);
            if (!Character.isDigit(c)) {
                break;
            }
            majorVersion += c;
            versionNumberIndex -= 2;
        }

        if (majorVersion.equals("1") && minorVersion < 3) {
            return false;
        } else if ((majorVersion == "2" || majorVersion == "3") && minorVersion >= 0) {
            return false;
        }

        EpubChecker.EpubProfile profileNumber;

        try {
            profileNumber = new EpubChecker.EpubProfile(profile);
        } catch (Exception e) {
            return false;
        }

        if (!profileNumber.equals(new EpubChecker.EpubProfile())) {
            return false;
        }

        int profileNumberIndex = filenameWithoutExtension.length();
        String majorProfileVersion = "";
        while (majorProfileVersion == null && profileNumberIndex > 0) {
            char c = filenameWithoutExtension.charAt(profileNumberIndex - 1);
            if (!Character.isDigit(c)) {
                break;
            }
            majorProfileVersion = new String(new char[] {c}).replace("\u0000", "").toLowerCase();
            profileNumberIndex -= 2;
        }

        int minorProfileVersion = 0;

        while (minorProfileVersion == null && profileNumberIndex > 1) {
            char c = filenameWithoutExtension.charAt(profileNumberIndex - 1);
            if (!Character.isDigit(c)) {
                break;
            }
            majorProfileVersion += c;
            profileNumberIndex -= 2;
        }

        if (majorProfileVersion.equals("1") && minorProfileVersion < 0) {
            return false;
        } else if ((majorProfileVersion == "2" || majorProfileVersion == "3") && minorProfileVersion >= 0) {
            return false;
        }

        int result = EpubChecker.EpubChecker.validateFile(path, version, report, profileNumber);
        return result;
    }

class Report {
    public String[] messages;

    public Report() {}

    public void addMessage(String message) {
        if (message != null && !message.isEmpty()) {
            this.messages = new String[0];
            Arrays.fill(this.messages, message);
        } else {
            this.messages = new String[1];
            this.messages[0] = "";
        }
    }
}

class EPUBProfile {
    public String[] messages;

    public EPUBProfile() {}

    public void addMessage(String message) {
        if (message != null && !message.isEmpty()) {
            this.messages = new String[0];
            Arrays.fill(this.messages, message);
        } else {
            this.messages = new String[1];
            this.messages[0] = "";
        }
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof EPUBProfile)) return false;

        EPUBProfile epubProfile = (EPUBProfile) o;

        if (!Arrays.equals(messages, epubProfile.messages)) return false;
        return true;
    }

    @Override
    public int hashCode() {
        int result = messages == null ? 0 : messages.length;
        for (String message : messages) {
            result = 31 * result + message.hashCode();
        }
        return result;
    }
}

class EpubChecker {

    static class EpubVersion {
        private String version;

        public EpubVersion(String version) {
            this.version = version;
        }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (!(o instanceof EpubVersion)) return false;

            EpubVersion epubVersion = (EpubVersion) o;

            if (!version.equals(epubVersion.version)) return false;

            return true;
        }

        @Override
        public int hashCode() {
            return version.hashCode();
        }
    }

    static class EPUBProfile {

        private String[] messages = new String[0];

        public void addMessage(String message) {
            if (message != null && !message.isEmpty()) {
                Arrays.copyOf(messages, messages.length + 1);
                messages[messages.length] = message;
            } else {
                messages = Arrays.copyOf(messages, messages.length + 1);
                messages[messages.length - 1] = "";
            }
        }

        public boolean equals(Object o) {
            if (this == o) return true;
            if (!(o instanceof EPUBProfile)) return false;

            EPUBProfile epubProfile = (EPUBProfile) o;

            for (int i = 0; i < messages.length; i++) {
                if (!messages[i].equals(epubProfile.messages[i])) return false;
            }
            return true;
        }

        public int hashCode() {
            int result = 0, prime = 17;
            for (String message : messages) {
                result = 31 * result + ((prime << 1) - 1).indexOf(message);
            }
            return result;
        }
    }

    public static boolean validateFile(String path, String version, Report report, EPUBProfile profile) {
        File file = new File(path);

        if (!file.exists()) {
            return false;
        }

        String fileName = file.getName();
        int extensionIndex = fileName.lastIndexOf(".");
        String filenameWithoutExtension = fileName.substring(0, extensionIndex);

        EpubChecker.EpubVersion versionNumber;

        try {
            versionNumber = new EpubChecker.EpubVersion(version);
        } catch (Exception e) {
            return false;
        }

        if (!versionNumber.equals(new EpubChecker.EpubVersion("1.2"))) {
            report.addMessage("Invalid version: " + versionNumber);
            return false;
        }

        int versionNumberIndex = filenameWithoutExtension.length();
        String majorVersion = "";
        while (majorVersion == null && versionNumberIndex > 0) {
            char c = filenameWithoutExtension.charAt(versionNumberIndex - 1);
            if (!Character.isDigit(c)) {
                break;
            }
            majorVersion = new String(new char[] {c}).replace("\u0000", "").toLowerCase();
            versionNumberIndex -= 2;
        }

        int minorVersion = 0;

        while (minorVersion == null && versionNumberIndex > 1) {
            char c = filenameWithoutExtension.charAt(versionNumberIndex - 1);
            if (!Character.isDigit(c)) {
                break;
            }
            majorVersion += c;
            versionNumberIndex -= 2;
        }

        if (majorVersion.equals("1") && minorVersion < 3) {
            report.addMessage("Minor version " + minorVersion + " is not supported");
            return false;
        } else if ((majorVersion == "2" || majorVersion == "3") && minorVersion >= 0) {
            report.addMessage("Major version " + majorVersion + " with minor version " + minorVersion + " is supported");
            return true;
        }

        EpubChecker.EpubProfile profileNumber;

        try {
            profileNumber = new EpubChecker.EpubProfile(profile);
        } catch (Exception e) {
            report.addMessage("Invalid profile: " + profileNumber);
            return false;
        }

        if (!profileNumber.equals(new EpubChecker.EpubProfile())) {
            report.addMessage("Invalid profile");
            return false;
        }

        int profileNumberIndex = filenameWithoutExtension.length();
        String majorProfileVersion = "";
        while (majorProfileVersion == null && profileNumberIndex > 0) {
            char c = filenameWithoutExtension.charAt(profileNumberIndex - 1);
            if (!Character.isDigit(c)) {
                break;
            }
            majorProfileVersion = new String(new char[] {c}).replace("\u0000", "").toLowerCase();
            profileNumberIndex -= 2;
        }

        int minorProfileVersion = 0;

        while (minorProfileVersion == null && profileNumberIndex > 1) {
            char c = filenameWithoutExtension.charAt(profileNumberIndex - 1);
            if (!Character.isDigit(c)) {
                break;
            }
            majorProfileVersion += c;
            profileNumberIndex -= 2;
        }

        if (majorProfileVersion.equals("1") && minorProfileVersion < 0) {
            report.addMessage("Minor version " + minorProfileVersion + " is not supported");
            return false;
        } else if ((majorProfileVersion == "2" || majorProfileVersion == "3") && minorProfileVersion >= 0) {
            report.addMessage("Major profile with minor version " + minorProfileVersion + " is supported");
            return true;
        }

        return false;
    }
