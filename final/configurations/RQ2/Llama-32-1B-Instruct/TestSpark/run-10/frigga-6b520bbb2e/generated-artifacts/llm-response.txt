        }
        return "";
    }

import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class BuildVersion {
    private String buildJobName;
    private int buildNumber;
    private Map<String, Integer> commitMap = new HashMap<>();

    public void setBuildJobName(String buildJobName) {
        this.buildJobName = buildJobName;
    }

    public void setBuildNumber(int buildNumber) {
        if (commitMap.containsKey(buildJobName)) {
            commitMap.put(buildJobName, buildNumber);
        } else {
            buildNumber = 0;
        }
    }

    public void addChangelist(String changelist) {
        for (String key : commitMap.keySet()) {
            String[] parts = key.split("-");
            if (parts[1].contains("unknown")) {
                System.out.println("Build number " + parts[2] + " is unknown in the current version.");
                buildNumber = 0;
                break;
            }
        }
        for (String part : changelist.split(",")) {
            String[] splitPart = part.split("-");
            if (splitPart[1].contains("unknown")) {
                System.out.println("Commit '" + part + "' is unknown in the current version.");
                buildNumber = 0;
                break;
            }
        }
    }

    public String toString() {
        return "buildJobName=" + buildJobName +
                ", buildNumber=" + buildNumber +
                ", commitMap=" + commitMap;
    }
}

public class VersionChecker {
    private BuildVersion buildVersion;

    public void setBuildVersion(BuildVersion buildVersion) {
        this.buildVersion = buildVersion;
    }

    public String checkBuildJobName(String buildJobName) {
        if (buildVersion != null && !buildVersion.toString().contains(buildJobName)) {
            return "The build job name '" + buildJobName + "' does not match the current version.";
        } else if (buildVersion == null) {
            return buildJobName;
        }
        return "";
    }

    public String checkBuildNumber(int buildNumber) {
        for (Map.Entry<String, Integer> entry : commitMap.entrySet()) {
            if (entry.getValue() == buildNumber) {
                return "The build number " + buildNumber + " is known in the current version.";
            }
        }
        return "";
    }

    public String checkCommit(String commit) {
        for (String key : commitMap.keySet()) {
            if (key.contains(commit)) {
                return "The commit '" + commit + "' is known in the current version.";
            }
        }
        return "";
    }
}

public class Main {
    public static void main(String[] args) {
        BuildVersion buildVersion = new BuildVersion();
        buildVersion.setBuildJobName("test-job");
        buildVersion.setBuildNumber(1);
        buildVersion.addChangelist("change-1");

        VersionChecker versionChecker = new VersionChecker();
        System.out.println(versionChecker.checkBuildJobName(buildVersion.buildJobName));
        System.out.println(versionChecker.checkBuildNumber(buildVersion.buildNumber));
