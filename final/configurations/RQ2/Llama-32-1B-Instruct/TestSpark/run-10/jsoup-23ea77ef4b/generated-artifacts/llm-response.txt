public class DocumentFactory {
    public String createOutputHtml(Document document, InputSource inputSource) {
        Transformer transformer = TransformerFactory.newInstance().newTransformer();
        StringWriter outputString = new StringWriter();
        
        transformer.setOutputProperty(OutputKeys.INDENT, "yes");
        transformer.transform(inputSource, new OutputHTML(outputString));
        
        return outputString.toString().trim();
    }
}

public class InputSource {
    public InputSource(String string) {
        this.string = string;
    }

    private String string;

    public String get(String prefix, String suffix) {
        if (string == null || string.isEmpty()) {
            throw new RuntimeException("InputSource: no arguments");
        }
        
        return string.substring(prefix.length() + 1, suffix.length());
    }
}

public class DocumentBuilderFactory {
    public Document createOutputHtml(InputStream inputStream, OutputStream outputStream) {
        DocumentFactory factory = new DocumentFactory();
        return factory.createOutputHtml(inputStream, outputStream);
    }
}

import org.w3c.dom.Document;
import org.xml.sax.InputSource;
import org.xml.sax.SAXException;

public class HtmlGenerator {

    public static String generateHtml(String xmlString, String filename) {
        // Create the output HTML string with indentation
        StringBuilder html = new StringBuilder();
        
        try (InputSource inputSource = new InputSource(new java.io.StringReader(xmlString))) {
            DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
            Document document = factory.newDocumentBuilder().parse(inputSource);
            
            Transformer transformer = TransformerFactory.newInstance().newTransformer();
            transformer.setOutputProperty(OutputKeys.INDENT, "yes");
            
            // Save the XML to an HTML file
            java.io.PrintWriter writer = new java.io PrintWriter(new java.io.FileWriter(filename));
            transformer.transform(new DOMSource(document), new OutputHTML(writer, true));
            html.append(writer.toString()).append("\n");
        } catch (SAXException e) {
            throw new RuntimeException("Failed to parse XML", e);
        }
        
        return html.toString();
    }

    public static Document createOutputHtml(InputStream inputStream, String filename) throws SAXException, SAXParseException {

        DocumentBuilder builder = factory.newDocumentBuilder();
        InputSource inputSource = new InputSource(new InputStreamReader(inputStream));
        
        return builder.parse(inputSource).getDocumentElement();
    }
}

    private final java.io.PrintWriter writer;

    public OutputHTML(java.io.PrintWriter writer, boolean withComments) {
        this.writer = writer;
    }
