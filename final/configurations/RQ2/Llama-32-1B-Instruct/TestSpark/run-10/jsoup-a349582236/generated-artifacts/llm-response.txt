import org.junit.Test;
import static org.junit.Assert.assertEquals;

public class TagParserTest {

    @Test
    public void testGetTag() {
        String tag = new Tag("org.jsoup.parser.Tag");
        assertEquals(tag.toString(), tag.get());
    }

    @Test
    public void testGetId() {
        String tag = new Tag("org.jsoup.parser.Tag");
        assertEquals(0, tag.getId());
    }

    @Test
    public void testToString() {
        String[] tags = {"org.jsoup.parser.Tag", "org.jsoup.parser.Tag"};
        for (String tag : tags) {
            assertEquals(tag.toString(), tag.toString());
        }
    }

    @Test
    public void testGetType() {
        String tag = new Tag("org.jsoup.parser.Tag");
        assertEquals(0, tag.getType());
    }

    @Test
    public void testParseTag() {
        String[] tags = {"org.jsoup.parser.Tag", "org.jsoup.parser.Tag"};
        for (String tag : tags) {
            TagParser parser = new TagParser();
            Tag parsedTag = parser.parse(tag);
            assertEquals(tag, parsedTag.toString());
        }
    }

    public static class Tag {
        private final String id;
        private int type;

        public Tag(String id) {
            this.id = id;
        }

        public String get() {
            return id + " (" + type + ")";
        }

        public void setId(int id) {
            this.id = id;
        }

        public int getType() {
            return type;
        }

        public void setType(int type) {
            this.type = type;
        }
    }

    public static class TagParser {
        public String parse(String tag) {
            if (tag.startsWith("org.jsoup.parser.") || tag.equals("org.jsoup.parser.")) {
                try {
                    return new Tag(tag).toString();
                } catch (Exception e) {
                    throw new RuntimeException(e);
                }
            } else {
                throw new RuntimeException("Invalid tag: " + tag);
            }
        }
    }

    public static void main(String[] args) {
        TagParserTest test = new TagParserTest();

        test.testGetTag();
        test.testGetId();
        test.testToString();
        test.testGetType();
        test.testParseTag();
    }
}

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class Tag {

    public static String getId(String tag) {
        return tag.substring(9);
    }

    public static String toString(Tag tag) {
        return "org.jsoup.parser." + getTag(tag).getPrefix();
    }

    public static boolean startsWith(String tag, String prefix) {
        return tag.startsWith(prefix);
    }

    public static boolean endsWith(String tag, String suffix) {
        if (tag.endsWith(suffix)) return true;
        else if (!tag.contains(suffix)) return false;
        else throw new AssertionError("Suffix not found in tag");
    }

    public static Tag parse(String tag) {
        Document doc = Jsoup.parse(tag);
        return new Tag(doc.select("*"));
    }

    public static String getType(Tag tag) {
        try {
            return tag.type().toString();
        } catch (Exception e) {
            throw new AssertionError("Invalid type");
        }
    }
}
