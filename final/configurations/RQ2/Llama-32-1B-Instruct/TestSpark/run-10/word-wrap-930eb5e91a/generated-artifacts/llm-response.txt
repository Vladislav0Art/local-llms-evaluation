```java
public class WordWrap {
    public static void wordWrap(BufferedReader in, StringWriter out, String lineSeparator, long numChars, null, HashSet<String> insertHyphens, boolean insertBreakWords, boolean trimLeft) throws IOException {
        for (String s : insertHyphens) {
            if (!trimLeft && !trimRight(s)) {
                continue;
            }
            in.readLine();
        }

        int i = 0;

        while (i < numChars) {
            out.write(lineSeparator);
            String line = in.readLine();

            if (line.trim().isEmpty()) {
                break;
            }

            if (!insertBreakWords && !trimLeft) {
                continue;
            }

            if (i + 1 >= numChars) {
                throw new IOException("Line too long");
            }

            int j = i;

            while (j < line.length() - 1 && Character.isWhitespace(line.charAt(j))) {
                j++;
            }

            String s = line.substring(i, j + 1);

                continue;
            }

            i += s.length();
        }
```

```java
import java.util.*;

public class WordWrap {
    public static void main(String[] args) throws IOException {
        BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder out = new StringBuilder();

        System.out.println("Enter the number of characters you want to wrap:");
        int numChars;
        while (true) {
            try {
                numChars = Integer.parseInt(in.readLine());
                break;
            } catch (NumberFormatException e) {
                System.out.println("Invalid input. Please enter a valid integer.");
                in.readLine(); // Clear the line
            }
        }

        String[] insertHyphens = {"", " ", ".", ","};

        System.out.println("\nEnter words you want to wrap:");
        for (int i = 0; i < numChars; i++) {
            out.append(in.readLine()).append("\n");
        }

        wordWrap(in, out.toString(), lineSeparator, numChars, insertHyphens, false, trimLeft);
    }

    public static void wordWrap(BufferedReader in, String content, String lineSeparator, long numChars, null, boolean trimLeft, boolean insertBreakWords) throws IOException {
        for (String s : insertHyphens) {
            if (!trimLeft && !trimRight(s)) {
                continue;
            }
            in.readLine();
        }

        int i = 0;

        while (i < numChars) {
            out.append(lineSeparator);
            String line = content.trim();

            if (line.length() == 0 || trimLeft && insertBreakWords) {
                break;
            }

            String lastSpace = line.lastIndexOf(' ');

            if (lastSpace != -1) {
                int leadingLength = i + 1 - lastSpace;

                if (leadingLength > 0) {
                    out.append(line.substring(0, lastSpace + 1)).append(lineSeparator);
                    content = content.substring(lastSpace + 2);

                    i += leadingLength;
                }
            }

            if (i + 1 == numChars) {
                throw new IOException("Line too long");
            }

            int j = i;

            while (j < line.length() - 1 && Character.isWhitespace(line.charAt(j))) {
                j++;
            }

            String word = line.substring(i, j + 1);

            if (!trimLeft && trimRight(word)) {
                continue;
            }

            out.append(trimLeft ? word : line.trim()).append(lineSeparator);
            i += word.length();
        }
    }

    public static void leftTrim(BufferedReader in, StringBuilder out) throws IOException {
        int i = 0;

        while (i < numChars && trimLeft) {
            String line = in.readLine();

            if (!line.isEmpty()) {
                out.append(line).append("\n");
                break;
            }

            i++;
        }
    }

    public static void rightTrim(BufferedReader in, StringBuilder out) throws IOException {
        int j = 0;

        while (j < numChars && trimRight) {
            String line = in.readLine();

            if (!line.isEmpty()) {
                j++;

                if (i == j) {
                    break;
                }

                out.append(line).append("\n");
            }
        }
    }

    public static void printLine(String line) {
        System.out.println(line);
    }
}
```
