public class Channel {
    private byte[] buffer = new byte[1024];
    private int index;

    public void write(String message) throws Exception {
        // Write a chunk of the message to the channel
        System.arraycopy(message.getBytes(), 0, buffer, index, message.length());
        index += message.length();
    }

    public String read() throws Exception {
        if (index == 0) throw new java.util.NoSuchElementException();
        byte[] data = new byte[1024];
        int len = Math.min(data.length, index);
        System.arraycopy(buffer, 0, data, 0, len);
        buffer = data;
        return new String(data, 0, len).replaceAll("\\s+", "");
    }
}

class RemoteAddress implements java.net.SocketAddress {
    private InetSocketAddress address;

    public RemoteAddress(InSocketAddress address) {
        this.address = address;
    }

    @Override
    public boolean equals(java.util.Objects.Object o) {
        if (this == o) return true;
        if (!(o instanceof RemoteAddress)) return false;
        RemoteAddress other = (RemoteAddress) o;
        return Objects.equals(address.getAddress(), other.address.getAddress());
    }
}
