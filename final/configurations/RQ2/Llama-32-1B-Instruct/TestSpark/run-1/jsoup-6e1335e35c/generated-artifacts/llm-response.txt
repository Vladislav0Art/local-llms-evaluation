    private String name;
    private Document document;

    public HTMLElement(String name, Document document) {
        super(name);

        elements.add(head);
        elements.add(appendable);

        List<HTMLElement> filteredElements = HTMLTest.HTMLElementFilter.filter(elements.get(0), elements);

        return DocumentBuilderFactory.newInstance().newDocumentBuilder().parse("example.xml");
    }

    public static List<HTMLElement> test(Document doc, Element root) {
        List<HTMLElement> filteredElements = new ArrayList<>();
        for (Node node : root.childNodes()) {
            if (node.getNodeType() == Node.ELEMENT_NODE && !filteredElements.contains(node)) filteredElements.add((HTMLElement) node);
        }
        return filteredElements;
    }

        DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
