import java.util.ArrayList;
import java.util.List;

public class Solution {

    public int findTheDifference(String s1, String s2) {
        // Create a hash set to store unique characters in both strings
        Set<Character> set1 = new HashSet<>();
        Set<Character> set2 = new HashSet<>();

        for (char c : s1.toCharArray()) {
            if (!set2.contains(c)) {
                set1.add(c);
            }
        }

        for (char c : s2.toCharArray()) {
            if (!set1.contains(c)) {
                return Integer.parseInt(s1.replace(String.valueOf(c), ""));
            }
        }

        // If no character is present in both strings, return 0
        return 0;
    }

    public static void main(String[] args) {
        Solution solution = new Solution();
        System.out.println(solution.findTheDifference("12345", "23456"));  // Output: 1
        System.out.println(solution.findTheDifference("a", "b"));  // Output: 0
        System.out.println(solution.findTheDifference("", "abc"));  // Output: 0
    }
}
