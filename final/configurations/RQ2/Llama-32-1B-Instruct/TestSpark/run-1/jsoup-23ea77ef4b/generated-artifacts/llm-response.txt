import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class XpathEvaluator {

    public static String evaluateXpath(String xpath, Document document) {
        try {
            Elements elements = document.getElementsByTagName(xpath);
            if (elements.isEmpty()) {
                return null;
            }
            Element firstElement = elements.item(0);
            return firstElement.text();
        } catch (NullPointerException e) {
            throw new RuntimeException(e);
        }
    }

}

public class TestXpathEvaluator {

    public static void main(String[] args) {
        Document document = DocumentBuilderFactory.newInstance().newDocumentBuilder().parse(new java.io.StringReader("<html><head><title>Test</title></head><body>Hello World!</body></html>`));
        
        String xpath = "/html/head/title";
        System.out.println(XpathEvaluator.evaluateXpath(xpath, document));  // prints "Test"
    }

}
