package com.crowdin.client.core.http.impl.json;

import com.fasterxml.jackson.databind.ObjectMapper;
import org.junit.jupiter.api.Test;

import java.io.InputStream;
import java.util.Arrays;

public class JacksonJsonTransformerTest {

    private ObjectMapper mapper = new ObjectMapper();

    @Test
    public void Test_Without mocking ParseStringJSON() {
        String json = "{\"key\":\"value\"}";
        Project project = parse(json);
        assertEquals("key", project.getKey());
        assertEquals("value", project.getValue());
    }

    @Test
    public void Test_Without mocking ParseStringJSON_Invalid() {
        String json = "{\"key\":\"\"}";
        Project project = parse(json);
        assertEquals(null, project.getKey());
        assertEquals("", project.getValue());
    }

    @Test
    public void Test_Without mocking ConvertJsonToString() {
        String json = "{\"key\":\"value\"}";
        LanguageTranslations translations = convert(json, LanguageTranslations.class);
        assertEquals("value", translations.getKey());
    }

    @Test
    public void Test_Without mocking ConvertStringToJson() {
        String json = "{\"key\":\"value\"}";
        Info info = convert(json, Info.class);
        assertEquals("path/to/file", info.getFile());
    }

    private Project parse(String json) throws Exception {
        return mapper.readValue(json, Project.class);
    }

    private LanguageTranslations convert(String json, Class<?> type) throws Exception {
        ObjectMapper mapper = new ObjectMapper();
        String stringJson = mapper.writeValueAsString(type.getDeclaredConstructor(String.class).newInstance(json));
        return mapper.readValue(stringJson, LanguageTranslations.class);
    }

    @Test
    public void Test_With_jsonTransformer ParseStringJSON() {
        JacksonJsonTransformer jsonTransformer = new JacksonJsonTransformer();
        Project project = jsonTransformer.parse("{\"key\":\"value\"}");
        assertEquals("key", project.getKey());
        assertEquals("value", project.getValue());
    }

    @Test
    public void Test_With_jsonTransformer ConvertStringToJson() {
        JacksonJsonTransformer jsonTransformer = new JacksonJsonTransformer();
        String stringJson = "{\"key\":\"value\"}";
        Info info = jsonTransformer.convert(stringJson, Info.class);
        assertEquals("path/to/file", info.getFile());
    }
}

This updated version includes a `parse` and `convert` method to mimic the behavior of the `JacksonJsonTransformer` class. The test methods have been modified accordingly to call these new methods.
