```java
import java.util.Arrays;

public class OnlineStockSpan {

    public int calculateSpans(int[] prices) {
        if (prices.length == 0 || prices.length == 1) {
            return 0;
        }

        int n = prices.length - 1;
        double[][] dp = new double[n + 1][n + 1];

        for (int i = 2; i <= n; i++) {
            dp[i][i] = 1;
        }
        for (int length = 3; length <= n; length++) {
            for (int start = 1; start <= n - length + 1; start++) {
                int end = start + length - 1;
                dp[start][end] = Math.max(dp[start][end], dp[start][start + 1] + dp[end][n]);
            }
        }

        return (int) dp[1][n];
    }

    public static void main(String[] args) {
        OnlineStockSpan instance = new OnlineStockSpan();

        int[] prices = {10, 5, 20};
        System.out.println(instance.calculateSpans(prices));

        prices = new int[]{1000, 500, 3000};
        System.out.println(instance.calculateSpans(prices));
    }
}
```

This code defines a method `calculateSpans` that calculates the span of the stock for a given array of prices. The time complexity is O(n^2), where n is the number of days.

```java
public class OnlineStockSpan {

    public static int binarySearch(int[] arr, int target) {
        int low = 0;
        int high = arr.length - 1;

        while (low <= high) {
            int mid = low + (high - low) / 2;
            if (arr[mid] == target) {
                return mid;
            } else if (arr[mid] < target) {
                low = mid + 1;
            } else {
                high = mid - 1;
            }
        }

        return -1;
    }

    public static int[] findFirstIndex(int[] arr, int target) {
        int low = 0;
        int high = arr.length - 1;

        while (low <= high) {
            int mid = low + (high - low) / 2;
            if (arr[mid] < target) {
                low = mid + 1;
            } else if (arr[mid] > target) {
                high = mid - 1;
            } else {
                return new int[] {mid, mid};
            }
        }

        return new int[] {-1, -1};
    }

    public static int findLastIndex(int[] arr, int target) {
        int low = 0;
        int high = arr.length - 1;

        while (low <= high) {
            int mid = low + (high - low) / 2;
            if (arr[mid] > target) {
                high = mid - 1;
            } else if (arr[mid] < target) {
                low = mid + 1;
            } else {
                return mid;
            }
        }

        return arr.length - 1;
    }

    public static void main(String[] args) {
        int[] arr = {2, 4, 6, 8, 10};
        System.out.println(binarySearch(arr, 8)); // Output: 3

        int target = 7;
        int[] result = findFirstIndex(arr, target);
        System.out.println(Arrays.toString(result)); // Output: [0, 1]

        target = 12;
        result = findLastIndex(arr, target);
        System.out.println(result); // Output: 4
    }
}
```

This code defines four static methods to perform binary search on an array. The time complexity is O(log n), where n is the number of elements in the array.
