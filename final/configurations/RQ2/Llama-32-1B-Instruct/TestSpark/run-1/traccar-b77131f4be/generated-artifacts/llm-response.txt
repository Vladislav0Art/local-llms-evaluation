import java.util.Scanner;

public class ResponseGenerator {

    private String response;

    @Before
    public void prepare() {
        // Mock data for the response
        this.response = "{\"error\": \"Invalid JSON\", \"message\": \"This is an invalid response\"}";
    }

        return new Response(this.response);
    }

    public String getStringFromNull() throws Exception {
        try {
            Scanner scanner = new Scanner(System.in);
            System.out.print("Enter a response: ");
            this.response = scanner.nextLine();
            return this.response;
        } catch (Exception e) {
            System.out.println(e.getMessage());
            return null;
        }
    }

    public String getStringFromEmptyString() throws Exception {
        try {
            Scanner scanner = new Scanner(System.in);
            System.out.print("Enter a response: ");
            this.response = scanner.nextLine();
            if (!this.response.isEmpty()) {
                return this.response;
            } else {
                throw new Exception("No response provided");
            }
        } catch (Exception e) {
            System.out.println(e.getMessage());
            return null;
        }
    }
