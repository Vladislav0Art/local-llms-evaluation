```java
public class DataFrameFilter {
    public static void main(String[] args) throws Exception {
        String filterColumn = "column_name";
        List<String> columnNames = new ArrayList<>();
        for (int i = 0; i < 5; i++) {
            columnNames.add("value_" + i);
        }
        String filterValues = "value_4";
        
        DataFrame filterDataFrame = new DataFrame(columnNames, filterColumn);
        DataFrame df = filterDataFrame.getFilteredData(filterColumns, filterValue);

        System.out.println(df.toString());
    }

    public static class DataFrame {
        private List<String> columnNames;
        private String columnName;
        private String[] columns;

        public DataFrame(List<String> columnNames, String columnName) throws Exception {
            this.columnNames = columnNames;
            this.columnName = columnName;
            this.columns = new String[columnNames.size()];
            int index = 0;
            for (String name : columnNames) {
                columns[index++] = name;
            }
        }

        public DataFrame getFilteredData(String filterColumns, String filterValue) throws Exception {
            boolean[][] filteredData = new boolean[columnNames.size()][filterColumns.length];
            for (int i = 0; i < columnNames.size(); i++) {
                for (String column : columns) {
                    if (column.equals(filterColumn)) {
                        for (String value : getValues(filterValue)) {
                            if (Arrays.asList(columns).contains(value)) {
                                filteredData[i][filterColumns.indexOf(column)] = true;
                            }
                        }
                    }
                }
            }

            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < columnNames.size(); i++) {
                for (boolean b : filteredData[i]) {
                    if (b) {
                        sb.append(columns[i]).append(",");
                    } else {
                        break;
                    }
                }
                if (i < columnNames.size() - 1) {
                    sb.append("|");
                }
            }

            return new DataFrame(sb.toString().split(","), columnName);
        }

        public String[] getValues(String filterValue) throws Exception {
            String[] values = filterValue.split(",");
            for (String value : values) {
                if (!value.isEmpty()) {
                    return Arrays.asList(columns).stream()
                            .filter(column -> column.equals(value))
                            .findFirst()
                            .map(String::trim)
                            .orElse("");
                }
            }
            return new String[0];
        }

        @Override
        public String toString() {
            StringBuilder sb = new StringBuilder();
            for (String value : columns) {
                sb.append(value).append(",");
            }
            return sb.toString().substring(0, sb.length() - 1);
        }
    }
}
```

In this solution, we first create a DataFrame with the specified column names and a filter column. We then call the `getFilteredData` method to get the filtered data.

The `getFilteredData` method checks if each value in the filter values matches any of the values in the columns. If it does, it sets the corresponding boolean value in the `filteredData` array. Finally, it returns a string representation of the filtered data by joining the column names with commas and adding a pipe character between each pair of values.

Note that this solution assumes that the filter values are strings containing comma-separated values, where each value is one of the values in the columns. If your filter values are different, you will need to modify the `getValues` method accordingly.

public class DataFrameFilter {
    public static void main(String[] args) throws Exception {
        String filterColumn = "column_name";
        List<String> columnNames = new ArrayList<>();
        for (int i = 0; i < 5; i++) {
            columnNames.add("value_" + i);
        }
        
        DataFrame filterDataFrame = new DataFrame(columnNames, filterColumn);
        DataFrame df = filterDataFrame.getFilteredData(filterColumns, filterValue);

        System.out.println(df.toString());
    }

    public static class DataFrame {
        private List<String> columnNames;
        private String columnName;

        public DataFrame(List<String> columnNames, String columnName) throws Exception {
            this.columnNames = columnNames;
            this.columnName = columnName;
            this.columns = new String[columnNames.size()];
            int index = 0;
            for (String name : columnNames) {
                columns[index++] = name;
            }
        }

        public DataFrame getFilteredData(String filterColumns, String filterValue) throws Exception {
            boolean[][] filteredData = new boolean[columnNames.size()][filterColumns.length];
            for (int i = 0; i < columnNames.size(); i++) {
                for (String column : columns) {
                    if (column.equals(filterColumn)) {
                        for (String value : getValues(filterValue)) {
                            if (Arrays.asList(columns).contains(value)) {
                                filteredData[i][filterColumns.indexOf(column)] = true;
                            }
                        }
                    }
                }
            }

            return new DataFrame(filteredData, columnName);
        }

        public String[] getValues(String filterValue) throws Exception {
            String[] values = filterValue.split(",");
            for (String value : values) {
                if (!value.isEmpty()) {
                    int index = 0;
                    for (int i = 0; i < columnNames.size(); i++) {
                        if (index >= values.length || !values[index].equals(columns[i])) {
                            break;
                        }
                        index++;
                    }
                    return Arrays.asList(columns).stream()
                            .filter(column -> column.equals(value))
                            .findFirst()
                            .map(String::trim)
                            .orElse("");
                }
            }
            return new String[0];
        }

        public String toString() {
            StringBuilder sb = new StringBuilder();
            for (String value : columns) {
                sb.append(value).append(",");
            }
            return sb.toString().substring(0, sb.length() - 1);
        }

        private DataFrame(boolean[][] filteredData, String columnName) throws Exception {
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < columnNames.size(); i++) {
                if (filteredData[i][filterColumns.indexOf(columnName)]) {
                    sb.append(columns[i]).append(",");
                } else {
                    break;
                }
            }
            return new DataFrame(sb.toString().trim(), columnName);
        }
    }

    public static void main(String[] args) throws Exception {
        String filterColumns = "column1,column2";
        String filterValue = "value_4, value_5";
        
        DataFrame df = new DataFrameFilter().getFilteredData(filterColumns, filterValue);

        System.out.println(df.toString());
    }
}
