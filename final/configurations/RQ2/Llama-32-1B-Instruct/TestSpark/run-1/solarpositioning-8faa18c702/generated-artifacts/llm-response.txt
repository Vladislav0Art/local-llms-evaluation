package net.e175.klaus.solarpositioning;

import java.time.LocalDate;
import static java.util.function.Supplier Supplier;

public class DeltaT {
    private final double value;

    public DeltaT(double value) {
        this.value = value;
    }

    @FunctionalInterface
    public interface EstimatorSupplier extends Supplier<Double> {
        default double estimate() {
            return this.get();
        }
    }

    public static class TestDeltaT implements EstimatorSupplier {
        private final String name;

        public TestDeltaT(String name) {
            this.name = name;
        }

        @Override
        public double estimate() {
            System.out.println(name + ": " + value());
            return value();
        }

        @Override
        public String toString() {
            return "(" + name + ", " + value() + ")";
        }
    }

    public static class TestEstimatorSupplier extends Supplier<Double> {
        private final String name;

        public TestEstimatorSupplier(String name) {
            this.name = name;
        }

        @Override
        public Double get() {
            System.out.println(name + ": " + value());
            return value();
        }

        @Override
        public String toString() {
            return "(" + name + ", " + value() + ")";
        }
    }

    public static void main(String[] args) {
        DeltaT delta = new DeltaT(10.5);

        TestDeltaT testDeltaT = new TestDeltaT("TestDeltaT");
        System.out.println(testDeltaT.estimate());

        EstimatorSupplier estimator = new TestEstimatorSupplier("TestEstimatorSupplier");
        double result = estimator.get();
        System.out.println(result);
    }
}
