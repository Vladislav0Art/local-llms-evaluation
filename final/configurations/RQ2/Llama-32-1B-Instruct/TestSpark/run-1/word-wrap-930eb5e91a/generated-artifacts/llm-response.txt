import java.util.ArrayList;
import java.util.List;

public class Main {

    // Create an array to store test cases
    private static List<String[]> tests = new ArrayList<>();

    // Method to add test cases
    public static void addTest(String test) {
        tests.add(test);
    }

    // Method to run a single test case
    public static void testCase() {
        for (String[] test : tests) {
            if (test.length == 0) {
                System.out.println("Invalid test");
                continue;
            }
            String[] args = test.clone();
            execute(args[0], args[1]);
        }
    }

    // Method to print the result of each test case
    public static void printResult() {
        for (String[] test : tests) {
            if (test.length == 0) {
                System.out.println("Invalid test");
                continue;
            }
            String[] args = test.clone();
            execute(args[0], args[1]);
        }
    }

    // Method to trim left
    public static void leftTrim(String input) {
        int length = input.length();
        System.out.print(input.substring(0, length - 1));
    }

    // Method to trim right
    public static void rightTrim(String input) {
        int length = input.length();
        System.out.print(input.substring(length - 1));
    }

    // Method to wrap lines of text
    public static void lineWrap(String input) {
        List<String> output = new ArrayList<>();
        String[] words = input.split(" ");
        for (String word : words) {
            if (!output.isEmpty() && !output.get(output.size() - 1).equals(word)) {
                output.add(word);
                output.add("\n");
            } else {
                output.add(word);
            }
        }
        System.out.println(String.join("", output));
    }

    // Method to wrap text
    public static void wordWrap(String input) {
        List<String> output = new ArrayList<>();
        String[] words = input.split(" ");
        for (String word : words) {
            if (!output.isEmpty() && !output.get(output.size() - 1).equals(word)) {
                output.add(word);
                System.out.print("\n");
            } else {
                output.add(word);
            }
        }
        System.out.println(String.join("", output));
    }

    // Method to close file
    public static void closeFile(String input) {
        int length = input.length();
        System.out.print(input.substring(0, length - 1));
    }

    // Method to trim text from start and end
    public static void trimTextFromStartAndEnd(String input) {
        StringBuilder output = new StringBuilder();
        String[] words = input.split(" ");
        for (String word : words) {
            if (!output.isEmpty() && !output.toString().startsWith(word)) {
                output.append(word).append("\n");
            } else {
                output.append(word);
            }
        }
        System.out.println(output.toString());
    }

    // Method to check close
    public static void checkClose(String input) {
        String[] words = input.split(" ");
        for (String word : words) {
            if (!word.equals("close")) {
                System.out.print(word + "\n");
            } else {
                System.out.println("Close method executed correctly");
            }
        }
    }

    // Method to wrap text by line
    public static void wrapTextByLine(String input) {
        List<String> output = new ArrayList<>();
        String[] words = input.split("\n");
        for (String word : words) {
            if (!output.isEmpty() && !output.get(output.size() - 1).equals(word)) {
                output.add("\n" + word);
            } else {
                output.add(word);
            }
        }
        System.out.println(String.join("", output));
    }

    // Method to execute a function
    public static void execute(String input, String output) {
        if (input.equals("trim text from start and end")) {
            trimTextFromStartAndEnd(output);
        } else if (input.equals("trim text from middle")) {
            int middleIndex = input.length() / 2;
            System.out.print(input.substring(0, middleIndex));
            System.out.println(output);
            System.out.print(input.substring(middleIndex));
            System.out.println(output);
        } else if (input.equals("check close")) {
            checkClose(output);
        } else if (input.equals("word wrap text by line")) {
            wordWrap(output);
        } else if (input.equals("wrap text by line") || input.equals("line wrap")) {
            lineWrap(output);
        } else if (input.equals("left trim")) {
            leftTrim(input);
        } else if (input.equals("right trim")) {
            rightTrim(input);
        } else if (input.equals("close file")) {
            closeFile(input);
        } else {
            System.out.println("Invalid input");
        }
    }

    public static void main(String[] args) {
        addTest("trim text from start and end");
        addTest("check close");
        addTest("word wrap text by line");
        addTest("wrap text by line");
        addTest("left trim");
        addTest("right trim");
        addTest("close file");

        testCase();
    }
}

import java.util.ArrayList;
import java.util.List;

public class Main {

    // Method to add test cases
    public static void addTest(String test) {
        tests.add(test);
    }

    // Method to run a single test case
    public static void testCase() {
        for (String[] test : tests.clone()) {
            if (test.length == 0) {
                System.out.println("Invalid test");
                continue;
            }
            execute(test[0], test[1]);
        }
    }

    // Method to print the result of each test case
    public static void printResult() {
        for (String[] test : tests.clone()) {
            if (test.length == 0) {
                System.out.println("Invalid test");
                continue;
            }
            execute(test[0], test[1]);
        }
    }

    // Method to trim text from start and end
    public static void trimTextFromStartAndEnd(String input, String output) {
        int length = input.length();
        StringBuilder outputBuilder = new StringBuilder(input);
        if (length > 1) {
            for (int i = 0; i < length - 1; i++) {
                char c = outputBuilder.charAt(i);
                if (c == '\n') {
                    outputBuilder.deleteCharAt(i);
                    break;
                }
            }
            input = outputBuilder.toString();
        }
        System.out.println(input + "\n" + output);
    }

    // Method to trim text from middle
    public static void trimTextFromMiddle(String input, String output) {
        int length = input.length();
        if (length > 1 && length % 2 == 0) {
            for (int i = 0; i < length / 2; i++) {
                char c = input.charAt(i);
                if (c == '\n') {
                    break;
                }
                output += c + " ";
            }
        } else {
            int middleIndex = length / 2;
            System.out.println(input.substring(0, middleIndex));
            System.out.print(output);
            System.out.println(input.substring(middleIndex));
        }
    }

    // Method to check close file
    public static void checkClose(String input, String output) {
        if (input.equals("trim text from start and end")) {
            trimTextFromStartAndEnd(output, "close");
        } else if (input.equals("check close")) {
            System.out.println("Close method executed correctly");
        }
    }

    // Method to word wrap text
    public static void wordWrap(String input, String output) {
        List<String> lines = new ArrayList<>();
        for (String line : input.split("\n")) {
            if (!output.isEmpty()) {
                if (line.length() > 50) {
                    lines.add(line.substring(0, 47) + "...");
                    line = line.substring(48);
                }
            }
            output += line + "\n";
        }
        System.out.println(String.join("", lines));
    }

    // Method to wrap text by line
    public static void wrapTextByLine(String input, String output) {
        List<String> lines = new ArrayList<>();
        for (String line : input.split("\n")) {
            if (!output.isEmpty()) {
                if (line.length() > 50) {
                    lines.add(line.substring(0, 47) + "...");
                    line = line.substring(48);
                }
            }
            output += "\n" + line;
        }
        System.out.println(output);
    }

    // Method to left trim text
    public static void leftTrim(String input, String output) {
        int length = input.length();
        for (int i = 0; i < length; i++) {
            char c = input.charAt(i);
            if (c != '\n') {
                output += c;
            }
        }
    }

    // Method to right trim text
    public static void rightTrim(String input, String output) {
        int length = input.length();
        for (int i = 0; i < length; i++) {
            char c = input.charAt(length - i - 1);
            if (c != '\n') {
                output += c;
            }
        }
    }

    // Method to close file
    public static void closeFile(String input) {
        int length = input.length();
        System.out.print(input.substring(0, length - 1));
    }

    // Method to trim text from start and end
    public static void trimTextFromStartAndEnd(String input, String output) {
        StringBuilder outputBuilder = new StringBuilder(input);
        if (input.length() > 1) {
            for (int i = 0; i < input.length() - 1; i++) {
                char c = outputBuilder.charAt(i);
                if (c == '\n') {
                    outputBuilder.deleteCharAt(i);
                    break;
                }
            }
            input = outputBuilder.toString();
        }
        System.out.println(input + "\n" + output);
    }

    public static void main(String[] args) {
        addTest("trim text from start and end");
        addTest("check close");
        addTest("word wrap text by line");
        addTest("wrap text by line");
        addTest("left trim");
        addTest("right trim");
        addTest("close file");

        testCase();
    }
}
