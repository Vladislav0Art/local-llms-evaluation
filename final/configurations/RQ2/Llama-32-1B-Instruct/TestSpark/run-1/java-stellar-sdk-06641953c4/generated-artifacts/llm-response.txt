import org.junit.Test;
import static org.junit.Assert.assertEquals;

public class PrivateKeyGenerator {
    public static String generatePrivateKey() {
        return "private key";
    }

    public static String generatePublicKey() {
        return "public key";
    }

    public static byte[] encodePrivateKey(String privateKey) {
        return privateKey.getBytes();
    }

    public static String encodeEncodedPrivateKey(String publicKey, String privateKey) {
        return publicKey + ", private key: " + privateKey;
    }
}

@Test
public void generatePrivateKeyTest() {
    assertEquals("private key", PrivateKeyGenerator.generatePrivateKey());
}

@Test
public void generatePublicKeyTest() {
    assertEquals("public key", PrivateKeyGenerator.generatePublicKey());
}

@Test
public void encodePrivateKeyTest() {
    String privateKey = "private key";
    byte[] encodedPrivateKey = PrivateKeyGenerator.encodePrivateKey(privateKey);
    assertEquals("encoded private key", encodedPrivateKey.toString());
}

@Test
public void encodeEncodedPrivateKeyTest() {
    String publicKey = "public key";
    String privateKey = "private key";
    String encodedPrivateKey = PrivateKeyGenerator.encodeEncodedPrivateKey(publicKey, privateKey);
    assertEquals("encoded public key and private key", encodedPrivateKey);
}
