import org.jsoup.parser.Tag;
import java.util.HashMap;
import java.util.Map;

public class TagImpl implements Cloneable {
    private String name;
    private String normalName;
    private static Map<String, Tag> tags;

    public boolean isBlock() {
        return false;
    }

    public boolean formatAsBlock() {
        return false;
    }

    public boolean isEmpty() {
        return true;
    }

    public boolean isSelfClosing() {
        return false;
    }

    public boolean isKnownTag() {
        return false;
    }

    public static boolean isKnownTag(String tagName) {
        return tags.containsKey(tagName);
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;

        Tag tag = (Tag) o;

        if (!name.equals(tag.name)) return false;
        if (!normalName.equals(tag.normalName())) return false;
        if (tag.getSettings() != null ? !tag.getSettings().equals(this.getSettings()) : this.getSettings() == null) return false;
        if (tag.isBlock() != isBlock()) return false;
        if (formatAsBlock() != formatAsBlock()) return false;
        return true;

    }

    @Override
    public int hashCode() {
        return 31 * (isBlock() ? this.getClass().getDeclaredField("name").get(this) : null)
                + tags.get(this.name).getClass().getDeclaredField("normalName").get(null);
    }

    @Override
    public String toString() {
        return name;
    }

    @Override
    protected Tag clone() {
        if (isSelfClosing()) return null;
        try {
            Tag result = new Tag();
            result.name = this.name;
            result.normalName = this.normalName;
            result.setSettings(this.getSettings());
            result.isBlock = isBlock();
            result.formatAsBlock = formatAsBlock();
            result.isEmpty = isEmpty();
            result.isSelfClosing = isSelfClosing();
            result.isKnownTag = isKnownTag();
            tags.put(result.name, result);
            return result;

        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }

    public String getName() {
        return name;
    }

    public String normalName() {
        if (isSelfClosing()) return Normalizer.normalize(name).toLowerCase();
        return Normalizer.normalize(name).toLowerCase();
    }

    public static Tag valueOf(String tagName, ParseSettings settings) {
        return tags.get(tagName);
    }

    public static Tag valueOf(String tagName) {
        return tags.get(tagName);
    }

    public boolean isBlock() {
        return false;
    }

    public boolean formatAsBlock() {
        return false;
    }

    public void setSelfClosing(boolean selfClosing) {
        isSelfClosing = selfClosing;
    }
}
