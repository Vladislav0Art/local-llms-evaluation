public class DocumentHelperUtils {
    private Document document;

    public void setUp(Document document) {
        this.document = document;
    }

    public String joinPaths(String basePath, java.util.List<String> paths) {
        StringBuilder pathBuilder = new StringBuilder(basePath);
        for (String path : paths) {
            pathBuilder.append("/").append(path);
        }
        return pathBuilder.toString();
    }
}

public static class DocumentHelperTest {

    public DocumentHelperUtils utils;

    public void setUp(DocumentHelperUtils utils) {
        this.utils = utils;
    }

    public void testJoinPaths() {
        assertEquals("path/to/base", utils.joinPaths("", new java.util.ArrayList<>()));
    }
}
