import java.util.Stack;

public class ArithmeticParser {

    private final String[] tokens;
    private int tokenIndex = 0;

    public ArithmeticParser(String[] tokens) {
        this.tokens = tokens;
    }

    public void parse() {
        while (tokenIndex < tokens.length) {
            String token = tokens[tokenIndex++];
            if (isLiteral(token)) {
                parseLiteral();
            } else if (isIdentifier()) {
                parseIdentifier();
            } else if (isBinaryOperation()) {
                parseBinaryOperation();
            } else {
                parseInvalidToken();
            }
        }
    }

    private boolean isLiteral(String token) {
        try {
            Integer.parseInt(token);
            return true;
        } catch (NumberFormatException e) {
            return false;
        }
    }

    private boolean isIdentifier() {
        String identifier = tokens[tokenIndex++];
        if (identifier.startsWith("$") || identifier.startsWith("--")) {
            return false;
        }
        for (char c : identifier.toCharArray()) {
            if (!Character.isLetterOrDigit(c)) {
                return false;
            }
        }
        return true;
    }

    private boolean isBinaryOperation() {
        String binaryOperator = tokens[tokenIndex++];
        while (tokenIndex < tokens.length && !isUnaryOperation(binaryOperator, tokens[tokenIndex++])) {
        }
        return tokenIndex == tokens.length || !isLiteral(tokens[tokenIndex - 1]) || !isLiteral(binaryOperator);
    }

    private boolean isUnaryOperation(String operator, String operand) {
        if (operator.equals("+")) {
            return true;
        } else if (operator.equals("-")) {
            for (int i = 0; i < operand.length(); i++) {
                if (!Character.isDigit(operand.charAt(i))) {
                    return false;
                }
            }
            return true;
        }
        throw new RuntimeException("Invalid unary operator");
    }

    private void parseLiteral() {
        while (tokenIndex < tokens.length && isLiteral(tokens[tokenIndex++])) {
        }
        if (tokenIndex == tokens.length) {
            throw new RuntimeException("Missing literal at end of expression");
        }
    }

    private void parseIdentifier() {
        String identifier = "";
        for (; tokenIndex >= 0 && !isIdentifier(); tokenIndex--) {
            char c = getCharacter();
            if (c == '$') {
                break;
            } else if (c == '--') {
                return;
            } else if (!Character.isLetterOrDigit(c)) {
                throw new RuntimeException("Invalid identifier");
            }
            identifier += c;
        }
        parseLiteralExpression(identifier);
    }

    private void parseBinaryOperation() {
        String binaryOperator = tokens[tokenIndex++];
        while (tokenIndex < tokens.length && !isUnaryOperation(binaryOperator, tokens[tokenIndex++])) {
        }
        ParseTree node = createParseTreeNode(tokens, tokenIndex - 1, tokenIndex);
        if (!node.isBinaryExpression()) {
            throw new RuntimeException("Invalid binary expression");
        }
    }

    private static class ParseTreeNode {

        boolean isLiteral;
        Stack<ParseNode> stack;

        public ParseTreeNode(String[] tokens, int startTokenIndex, int endTokenIndex) {
            this.startTokenIndex = startTokenIndex;
            this.endTokenIndex = endTokenIndex;
            String literalExpression = "";
            for (int i = startTokenIndex; i < endTokenIndex; i++) {
                if (isLiteral(tokens[i])) {
                    literalExpression += tokens[i];
                } else {
                    break;
                }
            }
            stack = new Stack<>();
            stack.push(new ParseTreeNode(literalExpression, startTokenIndex, startTokenIndex + literalExpression.length()));
        }

        public boolean isBinaryExpression() {
            if (stack.isEmpty()) {
                return false;
            }
            return true;
        }

        @Override
        public String toString() {
            return getIdentifier();
        }

    }

    private static class ParseNode {

        final String identifier;
        Stack<ParseNode> stack;

        public ParseNode(String identifier, int startTokenIndex, int endTokenIndex) {
            this.identifier = identifier;
            stack = new Stack<>();
            stack.push(this);
            parseLiteralExpression(identifier, startTokenIndex, endTokenIndex);
        }

    }

    private static String getIdentifier() {
        return tokens[tokenIndex++];
    }

    private static void parseLiteralExpression(String expression, int startTokenIndex, int endTokenIndex) {
        while (tokenIndex < tokens.length && isLiteral(tokens[tokenIndex++])) {
            ParseNode node = new ParseNode(tokens, startTokenIndex, endTokenIndex);
            if (node.isBinaryExpression()) {
                throw new RuntimeException("Invalid binary literal expression");
            }
            stack.push(node);
            endTokenIndex++;
        }
    }

    private static void parseConstantExpression(String token) {
        if (!isLiteral(token)) {
            throw new RuntimeException("Invalid constant expression");
        }
    }

    private String getCharacter() {
        return tokens[tokenIndex++];
    }

    private boolean isUnaryOperation(String operator, String operand) {
        // This implementation assumes that the operand is a single character
        if (operand.length() != 1) {
            throw new RuntimeException("Invalid unary operator");
        }
        for (int i = 0; i < operand.charAt(0) - '0'; i++) {
            if (!Character.isDigit(operand.charAt(i))) {
                return false;
            }
        }
        return true;
    }

    public static void main(String[] args) {
        String[] tokens = {"2", "+", "1", "*"};
        ArithmeticParser parser = new ArithmeticParser(tokens);
        parser.parse();
    }
}
