```java
import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.FileReader;
import java.io.FileWriter;

public class Main {
    public static void main(String[] args) {
        // Define the number of lines to read from the file and write to.
        int numLines = 1000;

        String fileName = "test.txt";
        StringBuilder result = new StringBuilder();

        try {
            // Open a BufferedReader to read from the file.
            BufferedReader br = new BufferedReader(new FileReader(fileName));

            // Open a BufferedWriter to write to the file, appending to the end if it exists.
            BufferedWriter bw = new BufferedWriter(new FileWriter(fileName, true));

            String line;
            while ((line = br.readLine()) != null) {
                // Add each line to the result StringBuilder in reverse order.
                bw.write(result.reverse().toString());
                result.append(line);
            }

            // Close the reader and writer when done.
            br.close();
            bw.close();

        } catch (Exception e) {
            // Print any errors that occur during file operations.
            System.out.println(e.getMessage());
        }
    }
}
```
