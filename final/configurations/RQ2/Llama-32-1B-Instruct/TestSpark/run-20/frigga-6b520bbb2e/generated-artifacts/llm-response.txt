```java
import java.util.*;

public class Solution {
    public int maximumNumber(String[] nums, String target) {
        int suffix = 0;
        for (int i = nums.length - 1; i >= 0; i--) {
            if (!suffix.isEmpty() && !compareStrings(suffix + nums[i], target)) break;
            suffix = compareStrings(suffix + nums[i], target);
        }
        return Integer.parseInt(suffix + target);
    }

    private boolean compareStrings(String a, String b) {
        int maxLengthA = 0;
        for (int i = 1; i <= Math.min(a.length(), b.length()); i++) {
            if (a.charAt(i - 1) == b.charAt(i - 1)) {
                maxLengthA += i;
            } else {
                break;
            }
        }
        return maxLengthA > a.length() + b.length();
    }

    private int compareStrings(String prefix, String string) {
        for (int i = 0; i < string.length(); i++) {
            if (!prefix.isEmpty() && !compareDigits(prefix + string.charAt(i), string)) break;
            prefix = prefix + string.charAt(i);
        }
        return Integer.parseInt(prefix);
    }

    private boolean compareDigits(String a, String b) {
        for (int i = 0; i < Math.min(a.length(), b.length()); i++) {
            if (a.charAt(i) != b.charAt(i)) break;
            i++;
            if (i == a.length() || i == b.length()) return true;
        }
        return false;
    }

    private int getFirstDigit(int value) {
        if (!isDigit(value)) return 0;
        for (int i = 0; i < 10; i++) {
            if (value % 10 != i) return i;
        }
        return 9;
    }

    private String buildJobName(int firstDigit, int suffix) {
        StringBuilder sb = new StringBuilder();
        while (firstDigit > 0) {
            sb.insert(0, firstDigit);
            firstDigit -= 1;
        }
        if (!suffix.isEmpty()) sb.insert(0, suffix);
        return sb.toString();
    }

    private static boolean isDigit(int num) {
        return num >= 0 && num <= 9;
    }
}
```

This solution works by starting with the longest possible suffix and then moving backwards. It checks each string in `nums` to see if it can be concatenated with the current suffix without exceeding the length of either string or resulting in an incorrect prefix. If a valid concatenation is found, it returns the concatenated string; otherwise, it continues to the next potential suffix. The result is then converted back into an integer and returned.

This solution has a time complexity of O(n*m), where n is the number of strings in `nums` and m is the maximum length of a string in `nums`.
