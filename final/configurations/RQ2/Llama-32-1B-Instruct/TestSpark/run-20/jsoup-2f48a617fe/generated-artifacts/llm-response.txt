public class Document {
    public Node root;

    public void setRoot(Node root) {
        this.root = root;
    }
}

class TagGenerator extends Document {
    @Override
    public void addTag(Node node, String tagName) {
        node.setTag(tagName);
    }

    @Override
    public void removeTag(Node node, String tagName) {
        if (node.getTagName().equals(tagName)) {
            node.removeTag(tagName);
        }
    }

    @Override
    public void tagToString(Node node) {
        System.out.println(node.getTagName());
    }
}

class TaggedNode extends Node {
    private TagGenerator tagGenerator;

    public TaggedNode(TagGenerator tagGenerator) {
        this.tagGenerator = tagGenerator;
    }

    @Override
    public String toString() {
        return super.toString() + " - Tag: " + tagGenerator.getTagName();
    }
}
