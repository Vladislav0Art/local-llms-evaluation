public class CssFormatterTest {

    private static final String CODE = "body {\n  background-color: #f2f2f2;\n}\n";

    @Test
    public void test_css_file() {
        AbstractCacheableFormatter formatter = new CssFormatter();
        String formattedCode = formatter.doFormat(CODE, LineEnding.CRLF);
        assertEquals("body {\n  background-color: #f2f2f2;\n}", formattedCode);
    }

    @Test
    public void test_stylesheet_with_css() {
        AbstractCacheableFormatter formatter = new CssFormatter();
        String stylesheet = "/* Test stylesheet */\nbody {\n  background-color: #f2f2f2;\n}\n";
        CSSFormat format = formatter.doFormat(stylesheet, LineEnding.CRLF);
        assertNotNull(format.getCssBody());
        assertEquals("body {\n  background-color: #f2f2f2;\n}", format.getCssBody().getCssText());
    }

    @Test
    public void test_stylesheet_without_css() {
        AbstractCacheableFormatter formatter = new CssFormatter();
        String stylesheet = "/* Test stylesheet */\npixel {\n  color: red;\n}\n";
        CSSFormat format = formatter.doFormat(stylesheet, LineEnding.CRLF);
        assertNotNull(format.getCssBody());
        assertEquals("pixel {\n  color: red;\n}", format.getCssBody().getCssText());
    }

    @Test
    public void test_css_file_without_options() {
        AbstractCacheableFormatter formatter = new CssFormatter();
        String formattedCode = formatter.doFormat(CODE, LineEnding.CRLF);
        assertEquals("body {\n  background-color: #f2f2f2;\n}", formattedCode);
    }

    @Test
    public void test_css_file_without_cfg() {

        }
    }
