        } catch (UnsupportedEncodingException e) {
            // Expected
        }
    }

    public void testAppendKeyValThrowsMalformedURLExceptionIfUrlIsInvalid() throws MalformedURLException {

        try {
            builder.appendKeyVal("key", "value");
            fail("Expected MalformedURLException");
        } catch (MalformedURLException e) {
            // Expected
        }
    }

    @Test
    public void testAppendKeyValReturnsUrlWithAppendKeyVal() throws UnsupportedEncodingException, IOException {

    }

    public void testAppendKeyValWithNestedMapReturnsNestedUrl() throws UnsupportedEncodingException, IOException {

        Map<String, String> map = builder.appendKeyVal("key", "value");
        Map<String, String> nestedMap = builder.appendKeyVal("nested-key", "nested-value");
        assertEquals("https://example.com?key=value&nested-key=nested-value", mapper.toString());
    }

    @Test
    public void testAppendUrlBuilderWithoutIdn() throws MalformedURLException {
        URLBuilder builder = new UrlBuilder("https://example.com");
        assertTrue(builder.build().getIdN() == null);

    public void testAppendUrlBuilderWithIdN() throws MalformedURLException, URISyntaxException {
        String url = "https://example.com";
        URLBuilder builder = new UrlBuilder(new URI(url));
        assertNotNull(builder.build().getIdN());
    }
