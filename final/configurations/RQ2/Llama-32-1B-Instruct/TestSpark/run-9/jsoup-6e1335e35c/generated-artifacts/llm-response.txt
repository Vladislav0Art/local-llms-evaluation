public class CustomElement {

    public static Document outerHtmlTail(Document tail) {
        if (tail == null) {
            return tail;
        }
        if (tail.tagName().equals("HEAD")) {
            for (int i = 0; i < tail.children().size(); i++) {
                Document subHead = new Document();
                subHead.tagName(tail.tagName());
                subHead.attributes.extend(tail.attributes());
                subHead.className.extend(tail.className());
                if (i > 0) {
                    Element prevTag = tail.children()[i - 1];
                    subHead.className.add(prevTag.className());
                    for (int j = i; j < i + 4; j++) {
                        if (!j % 2 == 0 && !tail.children().get(j).tagName().equals("BR")) {
                            subHead.createElement(tail.children()[j].tagName(), tail.children()[j]);
                        }
                    }
                }
                tail.after(subHead);
            }
        } else if (tail.tagName().equals("BODY")) {
            for (int i = 0; i < tail.children().size(); i++) {
                Document subBody = new Document();
                subBody.tagName(tail.tagName());
                subBody.className.extend(tail.className());
                Element prevTag = tail.children()[i];
                if (prevTag.tagName().equals("P")) {
                    for (int j = i; j < i + 3; j++) {

                        }
                    }
                } else {
                    for (int j = i; j < i + 4; j++) {
                        if (!j % 2 == 0 && !tail.children().get(j).tagName().equals("BR")) {
                            subBody.createElement(tail.children()[j].tagName(), tail.children()[j]);
                        }
                    }
                }
                tail.after(subBody);
            }
        } else {
            for (int i = 0; i < tail.children().size(); i++) {
                Element tag = tail.children()[i];
                if (!tag.tagName().equals("TAG")) {
                    Document subTag = new Document();
                    subTag.tagName(tag.tagName());
                    subTag.className.extend(tag.className());
                    element.appendChild(subTag);
                } else {
                    String text = "";
                    for (int j = 0; j < tag.children().size(); j++) {
                        Element child = tag.children()[j];
                        if (!child.tagName().equals("BR")) {
                            text += child.text() + " ";
                        }
                    }
                    element.appendChild(element.createTextNode(text));
                }
            }
        }
        return tail;
    }

        return outerHtmlTail(head);
    }

        if (parent == null || !parent.hasChildNodes()) {
            parent = new Document().createElement(tagName);
            element.appendChild(parent);
        }
        if (!addNewLine && text.length() > 0) {
            parent.appendChild(document.createElement("BR"));
        } else if (!text.isEmpty()) {
            parent.appendChild(document.createTextNode(text));
        }
    }

    public static void removeTag(String tagName, boolean addNewLine) {
        Element parent = document.getElementsByTagName(tagName).getFirst();
        if (parent != null && !addNewLine) {

                Element child = parent.children()[i];
                if (!child.tagName().equals("BR")) {
                    parent.removeChild(child);
                }
            }
        } else if (parent != null) {
            parent.remove();
        }
    }

        addTag(tag, "class", addNewLine);
        removeTag(tag, addNewLine);
    }

        return new CustomElement(doc);

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class HtmlTemplateEngine {

    public static void main(String[] args) {
        // Set up the test environment
        Document doc = new Document();

        // Create an HTML template engine
        TemplateEngine engine = new TemplateEngine();
        
        // Add a new tag to the document
        addTagToDoc("b", "Bold text");
        System.out.println(doc.toString());

        // Modify some attributes and text content of the document
        modifyTagAttributesAndText(doc, "div", "Width: 100%, Height: 200px");

        // Create another template engine instance
        TemplateEngine newEngine = new TemplateEngine();
        
        // Add a different tag to the second document
        addTagToDoc("h1", "Heading level 1");
        System.out.println(newEngine.toString());
    }

    public static void addTagToDoc(String tagName, String text) {
        Element parent = doc.getElementsByTagName(tagName).getFirst();
        if (parent == null || !parent.hasChildNodes()) {
            parent = new Document().createElement(tagName);
            doc.appendChild(parent);
        }
        
        // Add a new tag with the specified name and text
        addTag(parent, text, true);
    }

    public static void modifyTagAttributesAndText(Document doc, String tagName, String text) {
        Element parent = doc.getElementsByTagName(tagName).getFirst();
        if (parent == null || !parent.hasChildNodes()) {
            parent = new Document().createElement(tagName);
            doc.appendChild(parent);
        }
        
        // Add a new attribute and text to the tag
        addTagAttributeAndText(parent, "class", "custom-class");
        addTag(text, parent);
    }

    private static void addTag(Element parent, String text, boolean addNewLine) {
        // Create a new element with the specified name and text
        Element newElement = doc.createElement(tagName);
        
        if (addNewLine && text.length() > 0) {
            newElement.appendChild(document.createTextNode(text));
        } else if (!text.isEmpty()) {
            newElement.appendChild(document.createTextNode(text));
        }
        
        // Add the new element to the parent element
        parent.appendChild(newElement);
    }

    private static void addTagAttributeAndText(Element parent, String attrName, String value) {
        // Get the existing attributes of the tag
        Element tag = doc.getElementsByTagName(tagName).getFirst();
        if (tag != null && !tag.hasAttributes()) {
            // Create a new set of attributes for the tag
            java.util.Set<String> attrs = new java.util.HashSet<>();
            attrs.add(attrName + ": value=" + value);
            
            // Add the new attribute to the existing ones
            for (String attr : tag.attributes().toArray(new String[0])) {
                if (!attr.startsWith(attrName)) {
                    attrs.add(attr);
                }
            }
            
            // Update the attributes of the tag with the new values
            java.util.Map<String, Object> map = new java.util.HashMap<>();
            map.put(attrName + ": value=" + value, true);
            for (String attr : attrs) {
                java.util.Map<String, Object> mapAttr = new java.util.HashMap<>();
                mapAttr.put(attr.substring(0, attr.length() - 1), true);
                map.put(attr.substring(0, attr.length() - 1), mapAttr);
                tag.attributes().putAll(mapAttr);
            }
        }
    }

    private static Document getTemplate(String template) {
        return engine.getTemplate(template);
    }

    private static class TemplateEngine {
        public static Document getTemplate(String template) {
            try {
                java.util.Properties props = new java.util.Properties();
                
                // Load the template file and load its contents into a string
                java.io.File file = new java.io.File("template.txt");
                java.io.FileInputStream fis = new java.io.FileInputStream(file);
                java.io.PrintWriter writer = new java.io.PrintWriter(fis);
                String content = reader.write(writer);
                
                // Create a new Document from the template string and return it
                return newDocument(content);
            } catch (java.io.FileNotFoundException e) {
                System.out.println("Template file not found.");
                return null;
            }
        }

        private static java.io.InputStream reader = new java.io.FileInputStream("template.txt");
        
        private static java.io.PrintWriter writer = new java.io.PrintWriter(reader);
    }

    private static Document newDocument(String content) {
        // Create a new Document from the template string
        Document doc = new Document();
        doc.setContent(content);
        
        return doc;
    }
}
