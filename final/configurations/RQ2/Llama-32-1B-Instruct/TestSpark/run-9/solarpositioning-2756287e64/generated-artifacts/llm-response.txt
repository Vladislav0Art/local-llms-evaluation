```java
public class SolarPosition {
    public static void main(String[] args) {
        System.out.println(calculateSolarPosition(37, -122));
        System.out.println(calculateSolarPosition(-74.0060, 40));
        System.out.println(calculateSolarPosition(24.8982, 77));
        System.out.println(calculateSolarPosition(34.0522, -118.2437));
        System.out.println(calculateSolarPosition(41.8781, -87.6298));
    }

    public static double calculateSolarLatitude(double latitude) {
        return Math.asin(Math.sin(latitude) * Math.sqrt(Math.pow(1 + 0.000005, 2)));
    }

    public static int calculateSolarLongitude(int longitude) {
        return (int) Math.round(Math.atan(Math.tan(calculateSolarLatitude(-longitude)) * Math.sqrt(1 - Math.cos(calculateSolarLatitude(-longitude))));
    }

    public static String calculateSolarPosition(double latitude, double longitude) {
        int solarLongitude = calculateSolarLongitude(longitude);
        double solarLatitude = calculateSolarLatitude(latitude);

        return (latitude > 0 ? "North" : "South") + ", " + (solarLatitude < -90 ? -90 : solarLatitude) + ", " + (longitude > 0 ? "East" : "West");
    }
}

public class SolarPositionService {
    public static String calculateSolarPosition(String longitude, String latitude) {
        if (!longitude.matches("\\d+"))
            return "Invalid longitude";
        if (!latitude.matches("\\d+\\.\\d*"))
            return "Invalid latitude";

        int result = Integer.parseInt(longitude);
        double solarLatitude = Double.parseDouble(latitude);

        if (result == 0 && solarLatitude > 90)
            return "North";
        else if (result == 0 && solarLatitude < -90)
            return "South";
        else if (solarLatitude < -90)
            return "East" + ", " + (-90 + Math.round(solarLatitude) / 10 * 10);
        else
            return "West" + ", " + ((-90 - Math.round(solarLatitude) / 10 * 10) + 90);
    }
}
```
