    }

import java.util.Scanner;

public class Request {
    private final String url;
    private final String method;

    public Request(String url, String method) {
        this.url = url;
        this.method = method;
    }

        return url;

    private final Request request;

    public TestRequest(Request request) {
        this.request = request;
    }

    public void testGetRequest() {
        System.out.println("Test Get Request");
        
        // Check if the method is GET
        if (request.getMethod().equals("GET")) {
            System.out.println("\nURL: " + request.getUrl());
            System.out.println("Method: " + request.getMethod());

            System.out.println("\nUnsupported method");
        }
    }

    public void testPostRequest() {
        System.out.println("\nTest POST Request");
        
        // Set a value for the url attribute
        request.setUrl("http://example.com/post");
        
        // Call the get method to see if it has changed
        System.out.println("\nURL: " + request.getUrl());
        System.out.println("Method: " + request.getMethod());
    }
}
