```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class URLBuilder {
    public static String build() {
        Document doc = Jsoup.connect("https://www.example.com").get();
        Elements links = doc.select("a");
        StringBuilder sb = new StringBuilder();
        for (Element link : links) {
            if (!link.text().isEmpty()) {
                sb.append(link.attr("href"));
            }
        }
        return sb.toString();
    }

    public static String appendKeyVal(Map<String, String> kvMap) {
        Document doc = Jsoup.connect("https://www.example.com").get();
        Elements links = doc.select("a");
        StringBuilder sb = new StringBuilder();
        for (Element link : links) {
            if (!link.text().isEmpty()) {
                Element href = link.attr("href");
                String key = link.text();
                String value = href;
                sb.append(key).append("=").append(value);
                sb.append("&");
            }
        }
        return sb.toString();
    }

    public static void when(String url) {
        try {
            Document doc = Jsoup.connect(url).get();
            if (doc.title().isEmpty()) {
                throw new AssertionError("Expected a title, but got an empty string");
            }
        } catch (Exception e) {
            throw new AssertionError("Failed to retrieve the page", e);
        }
    }

    public static void when(URL url) {
        try {
            Document doc = Jsoup.connect(url).get();
            if (doc.title().isEmpty()) {
                throw new AssertionError("Expected a title, but got an empty string");
            }
        } catch (Exception e) {
            throw new AssertionError("Failed to retrieve the page", e);
        }
    }

    public static void when(String url, String expectedTitle) {
        Document doc = Jsoup.connect(url).get();
        if (!doc.title().equals(expectedTitle)) {
            throw new AssertionError("Expected '" + expectedTitle + "', but got '" + doc.title() + "'");
        }
    }

    public static void assertThrows(UnsupportedEncodingException e) {
        try {
            URLBuilder.appendKeyVal(new HashMap<String, String>());
        } catch (UnsupportedEncodingException ex) {
            assert false : "Expected an UnsupportedEncodingException but got " + ex.getMessage();
        }
    }
}
```

```java
public class Main {
    public static void main(String[] args) {
        URLBuilder.url("https://www.example.com").when(build());
        URLBuilder.url("https://www.example.com").appendKeyVal(new HashMap<String, String>() {{
            put("key", "value");
        }}).when(build());
        URLBuilder.url("https://www.example.com").setHeader("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/58.0.3029.110 Safari/537.3").when(build());
        URLBuilder.url("https://www.example.com").when(new String[]{"key", "value"});

        try {
            Main.url("https://www.example.com").assertThrows(UnsupportedEncodingException.class);
        } catch (UnsupportedEncodingException e) {
            System.out.println(e.getMessage());
        }
    }
```
