import java.net.URI;
import java.util.ArrayList;

public class CookieManager {

    private ArrayList<String> cookieHeaders = new ArrayList<>();

    public void addHeader(String header) {
        cookieHeaders.add(header);
    }

    public ArrayList<String> getHeaders() {
        return cookieHeaders;
    }
}

class Parser {

    private String headerName = "header-name";
    private String headerValue = "";

    public String getValue() {
        return headerValue;
    }

    public void setValue(String value) {
        headerValue = value;
    }

    public boolean hasHeader(String header) {
        return cookieHeaders.contains(header);
    }
}

public class HttpConnection {

    private CookieManager cookieManager;
    private Parser parser;

    public HttpConnection(CookieManager cookieManager, Parser parser) {
        this.cookieManager = cookieManager;
        this.parser = parser;
    }

    public String connect(String url) throws Exception {
        URI uri = new URI(url);
        String cookies = cookieManager.getHeaders().toString();
        String headerValue = parser.getValue();

        if (headerValue.isEmpty()) {
            System.out.println("No header provided.");
            return uri.toString();
        }

        // Parse the header value to get key-value pairs
        String[] headers = headerValue.split("; ");

        ArrayList<String> parsedHeaderValues = new ArrayList<>();
        for (String header : headers) {
            if (!header.isEmpty()) {
                String[] keyValue = header.split("=", 2);
                parsedHeaderValues.add(keyValue[0] + "=" + keyValue[1]);
            }
        }

        // Set the cookies with their values
        cookieManager.addHeader(uri.getFragment() + "; " + uri.getQuery());
        for (String value : parsedHeaderValues) {
            cookieManager.addHeader(value);
        }

        return uri.toString();
    }

    public String disconnect() throws Exception {
        // Remove the last cookie from the header string
        cookieManager.addHeader("");
        return super.disconnect();
    }
}

public class TestHttpConnection {

    public static void main(String[] args) throws Exception {
        CookieManager cookieManager = new CookieManager();
        Parser parser = new Parser();

        HttpConnection connection = new HttpConnection(cookieManager, parser);
        System.out.println(connection.connect("https://example.com"));
        System.out.println(connection.disconnect());

        HttpConnection connection2 = new HttpConnection(null, null);
        System.out.println(connection2.connect("https://example.com"));
        System.out.println(connection2.disconnect());
    }
}
