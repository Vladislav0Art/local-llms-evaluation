        LineEnding ending = LineEnding.LF_ALL;

        assert formatter.doFormat(code, ending) == expected : "doFormat failed";
    }

    @Test
    public void isInitialized_SimpleTest() throws IOException {
        Map<String, String> options = Collections.emptyMap();
        ConfigurationSource cfg = null;
        CssFormatter formatter = new CssFormatter(options, cfg);
        assert !formatter.isInitialized() : "isInitialized failed";
    }

    @Mock
    private InputSource mockInputSource;

    @Test
    public void doFormat_MockTest() throws IOException {
        when(mockInputSource.getCharacterData()).thenReturn("body { background-color: #f2f2f2; }");
        String code = "body { background-color: #f2f2f2; }";
        LineEnding ending = LineEnding.LF_ALL;
        CssFormatter formatter = new CssFormatter();
        assert formatter.doFormat(code, ending) == "body {\n  background-color: #f2f2f2;\n}";
    }
}
