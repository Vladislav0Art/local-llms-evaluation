package org.davidmoten.text.utils;

import com.github.davidmoten.guavamini.Preconditions;
import com.github.davidmoten.guavamini.annotations.VisibleForTesting;
import java.io.IOException;
import java.util.function.Function;

public final class WordWrap {

    public static void wordWrap(Reader in, Writer out, String newLine, Number maxWidth,
            Function<? super CharSequence, ? extends Number> stringWidth,
            Set<Character> extraWordChars, boolean insertHyphens, boolean breakWords)
            throws IOException {
        Preconditions.checkState(maxWidth != null);
        Preconditions.checkState(newLine != null);
        Preconditions.checkState(stringWidth != null);
        Preconditions.checkState(extraWordChars != null);

        wordWrap(in, out, newLine, maxWidth, stringWidth::apply, extraWordChars,
                insertHyphens, breakWords);
    }

    public static void wordWrap(Reader in, LineConsumer out, Number maxWidth,
            Function<? super CharSequence, ? extends Number> stringWidth,
            Set<Character> extraWordChars, boolean insertHyphens, boolean breakWords)
            throws IOException {
        Preconditions.checkState(maxWidth != null);
        Preconditions.checkState(stringWidth != null);

        String line = in.readLine();
        while (line != null) {
            if (!breakWords || !isWhitespace(line)) {
                out.addWord(line, stringWidth.apply(line));
            }
            line = in.readLine();
        }

        if (insertHyphens && !newLine.isEmpty()) {
            out.addHyphen(newLine);
        }
    }

    @VisibleForTesting
    public static void close(Reader reader) throws IOException {
        Preconditions.checkState(reader != null);

        try (final InputStream in = reader.getInputStream()) {
            in.close();
        }
    }

    @VisibleForTesting
    public static void close(Writer writer) throws IOException {
        Preconditions.checkState(writer != null);

        try (final OutputStream out = new OutputStreamWriter(writer)) {
            out.flush();
            out.close();
        }
    }

    public static CharSequence rightTrim(CharSequence s) {
        StringBuilder2 builder = new StringBuilder2();
        while (!s.isEmpty()) {
            if (!s.startsWith(newLine, 0)) {
                break;
            }
            final int index = s.indexOf(newLine, 1);
            if (index != -1) {
                s.delete(0, index + newLine.length());
                builder.append(s);
            } else {
                break;
            }
        }

        return builder.toString();
    }

    public static boolean isWhitespace(CharSequence s) {
        for (int i = 0; i < s.length(); i++) {
            if (!Character.isWhitespace(s.charAt(i))) {
                return false;
            }
        }

        return true;
    }

    @VisibleForTesting
    public static void leftTrim(StringBuilder2 word) {
        while (!word.isEmpty()) {
            final int index = newLine.indexOf(word.substring(0, 1), 0);
            if (index != -1) {
                word.delete(0, index + 1);
            } else {
                break;
            }
        }

        return word;
    }
}

class StringBuilder2 {
    private final StringBuilder builder;

    public StringBuilder2() {
        this.builder = new StringBuilder();
    }

    public void append(CharSequence csq) {
        builder.append(csq);
    }

    public void add(char c) {
        builder.append(c);
    }

    public String toString() {
        return builder.toString();
    }
}

class LineConsumer {
    private final java.io.Writer out;

    public LineConsumer(java.io.Writer out) {
        this.out = out;
    }

    public void addWord(CharSequence word, Number stringWidth) {
        out.addWord(word, stringWidth);
    }

    public void addHyphen(String newLine) {
        out.add(newLine);
    }
}

public class Main {

    public static void main(String[] args) throws IOException {
