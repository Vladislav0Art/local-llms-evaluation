    public String generateByteArrayFromPublicKey(String publicKeyStr) throws Exception {
        // Generate the private key and public key
        byte[] privateKeyPem;
        byte[] publicKeyPem;

        try {
            // Read the public key from the PEM file
            privateKeyPem = java.util.Base64.getDecoder().decode(publicKeyStr);
            publicKeyPem = java.util.Base64.getDecoder().decode("-----BEGIN PUBLIC KEY-----\n" + publicKeyStr + "\n-----END PUBLIC KEY-----");

            // Generate a random private key
            byte[] privateKeyBytes = new byte[256];
            java.security.SecureRandom.getInstanceStrong().generate(privateKeyBytes);

            // Add the private and public keys to the PEM files
            String privateKeyPemContent = "-----BEGIN PRIVATE KEY-----\n" +
                    java.util.Base64.encodeToString(privateKeyBytes, Base64.DEFAULT);
            String publicKeyPemContent = "-----BEGIN PUBLIC KEY-----\n" + java.util.Base64.encodeToString(publicKeyPem);

            // Return the contents of the PEM files
            return privateKeyPemContent + "\n" + publicKeyPemContent;
        } catch (Exception e) {
            throw new Exception(e.getMessage());
        }
    }

    public String sign(byte[] data) throws Exception {
        // Use the Java Cryptography Architecture (JCA) to generate a digital signature
        java.security.MessageDigest md;
        byte[] digest;

        try {
            // Create an instance of the message digest algorithm
            md = java.security.MessageDigest.getInstance("SHA-256");

            // Convert the data to bytes
            byte[] dataBytes = java.util.Base64.getEncoder().encode(data);

            // Update the digest with the data
            digest = md.digest(dataBytes);
        } catch (Exception e) {
            throw new Exception(e.getMessage());
        }

        // Create a secret key for signing and verifying
        String secretKey = "secret-key";
        java.security.KeyPairGenerator kpg;
        kpg = java.security.KeyPairGenerator.getInstance("RSA", "Default");
        kpg.initialize(2048);
        kpg.generateKeyPair();

        // Create a public key from the private key
        PublicKey publicKey = kpg.getPublic().getPublicKey();
        System.out.println(publicKey);

        byte[] signatureBytes = new byte[256];
        java.security.SignatureContext ctx = java.security.Signature.getInstance("SHA-256");

        ctx.update(dataBytes);
