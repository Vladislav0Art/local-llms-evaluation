import java.util.ArrayList;
import java.util.List;

public abstract class Tokenizer {
    
    public static class TokenType {
        
        public static final TokenType T_TOKEN_TYPE_TOKEN = new TokenType(TokenType.TOKEN_TYPE_TOKEN, "token");
        public static final TokenType T_TOKEN_TYPE_SUBSTRING = new TokenType(TokenType.TOKEN_TYPE_SUBSTRING, "substring");
        public static final TokenType T_TOKEN_TYPE_NUMBER = new TokenType(TokenType.TOKEN_TYPE_NUMBER, "number");
        public static final TokenType T_TOKEN_TYPE_OPERATOR = new TokenType(TokenType.TOKEN_TYPE_OPERATOR, "operator");
        
        private final int type;
        private final String text;
        
        private TokenType(int type, String text) {
            this.type = type;
            this.text = text;
        }
        
        @Override
        public String toString() {
            return "{" +
                    "type=" + type +
                    ", text='" + text + '\'' +
                    '}';
        }
    }
    
    private static class TokenTypeImpl implements TokenType {
        
        private final int type;
        private final String text;
        
        public TokenTypeImpl(int type, String text) {
            this.type = type;
            this.text = text;
        }
        
        @Override
        public int getType() {
            return type;
        }
        
        @Override
        public String getText() {
            return text;
        }
    }
    
    private static class Token {
        
        private final TokenType type;
        private final String text;
        
        public Token(TokenType type, String text) {
            this.type = type;
            this.text = text;
        }
        
        @Override
        public String toString() {
            return "{" +
                    "type=" + type +
                    ", text='" + text + '\'' +
                    '}';
        }
    }
    
    public abstract List<Token> tokenize(String expression);
    
    public static class NumberToken extends Token {
        
        private final double value;
        
        public NumberToken(double value) {
            super(TokenType.TOKEN_TYPE_NUMBER, String.valueOf(value));
            this.value = value;
        }
        
        @Override
        public double getValue() {
            return value;
        }
    }
    
    public static class OperatorToken extends Token {
        
        private final String text;
        
        public OperatorToken(String text) {
            super(TokenType.TOKEN_TYPE_OPERATOR, text);
        }
        
        @Override
        public String getText() {
            return text;
        }
    }
}

public class NumberParser extends Tokenizer {
    
    @Override
    public List<Token> tokenize(String expression) {
        List<Token> tokens = new ArrayList<>();
        
        String[] parts = expression.split("[+-*/() ]+");
        
        for (String part : parts) {
            if (!part.isEmpty()) {
                double value = Double.parseDouble(part);
                
                Token token;
                switch (Tokenizer.TokenTypeImpl.getKind(value)) {
                    case TokenType.TOKEN_TYPE_NUMBER:
                        token = new NumberToken(value);
                        break;
                    default:
                        throw new RuntimeException("Unknown type");
                }
                
                tokens.add(token);
            }
        }
        
        return tokens;
    }
}

public class OperatorParser extends Tokenizer {
    
    @Override
    public List<Token> tokenize(String expression) {
        List<Token> tokens = new ArrayList<>();
        
        String[] parts = expression.split("[+-*/() ]+");
        
        for (String part : parts) {
            if (!part.isEmpty()) {
                Token token;
                
                switch (Tokenizer.TokenTypeImpl.getKind(part)) {
                    case TokenType.TOKEN_TYPE_NUMBER:
                        token = new NumberToken(Double.parseDouble(part));
                        break;
                    default:
                        throw new RuntimeException("Unknown type");
                }
                
                tokens.add(token);
            }
        }
        
        return tokens;
    }
}

public class ExpressionParser {
    public String evaluate(String expression) {
        // implement your logic here to parse the expression and return the result
        if (expression.contains("+")) {
            String[] parts = expression.split("\\+");
            int i = 0;
            while (i < parts.length - 1 && Character.isDigit(parts[i++]) || isOperator(parts[i])) {
                throw new Exception("Invalid input");
            }
            int operand2 = Integer.parseInt(parts[i]);
            if (Character.isDigit(parts[i + 1])) {
                int operand1 = Integer.parseInt(parts[i + 1]);
                return (int) evaluate(operand1, parts, i + 3);
            } else {
                return String.valueOf(evaluate(operand2, parts, i));
            }
        } else if (expression.contains("-")) {
            String[] parts = expression.split("\\-");
            int i = 0;
            while (i < parts.length - 1 && Character.isDigit(parts[i++]) || isOperator(parts[i])) {
                throw new Exception("Invalid input");
            }
            int operand2 = Integer.parseInt(parts[i]);
            if (Character.isDigit(parts[i + 1])) {
                int operand1 = Integer.parseInt(parts[i + 1]);
                return evaluate(operand1, parts, i + 3) - operand2;
            } else {
                throw new Exception("Invalid input");
            }
        } else if (expression.contains("*")) {
            String[] parts = expression.split("\\*");
            int i = 0;
            while (i < parts.length - 1 && Character.isDigit(parts[i++]) || isOperator(parts[i])) {
                throw new Exception("Invalid input");
            }
            int operand2 = Integer.parseInt(parts[i]);
            if (Character.isDigit(parts[i + 1])) {
                int operand1 = Integer.parseInt(parts[i + 1]);
                return evaluate(operand1, parts, i + 3) * operand2;
            } else {
                throw new Exception("Invalid input");
            }
        } else if (expression.contains("/")) {
            String[] parts = expression.split("/");
            int i = 0;
            while (i < parts.length - 1 && Character.isDigit(parts[i++]) || isOperator(parts[i])) {
                throw new Exception("Invalid input");
            }
            int operand2 = Integer.parseInt(parts[i]);
            if (Character.isDigit(parts[i + 1])) {
                int operand1 = Integer.parseInt(parts[i + 1]);
                return evaluate(operand1, parts, i + 3) / operand2;
            } else {
                throw new Exception("Invalid input");
            }
        }

        return "0";
    }

    private boolean isOperator(char c) {
        return !Character.isDigit(c);
    }

    public static void main(String[] args) {
        ExpressionParser parser = new ExpressionParser();
        System.out.println(parser.evaluate("+5+2")); // Output: 7
        System.out.println(parser.evaluate("-3*4-1")); // Output: 4
        System.out.println(parser.evaluate("/10/2")); // Output: 0.5
    }
}
