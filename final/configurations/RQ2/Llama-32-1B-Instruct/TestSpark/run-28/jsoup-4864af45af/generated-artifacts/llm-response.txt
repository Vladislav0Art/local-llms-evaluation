import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class GeneratedTest {
    private Document document;

    public boolean isSafeAttribute(String attribute, String value, String tag) {
        if (tag == null || value.isEmpty()) {
            return true; // empty strings can be treated as safe
        }

        try {
            document = Jsoup.parse(tag);
            Element element = document.getElementById(value);
            if (element != null) {
                return true;
            }
        } catch (Exception e) {}

        return false;
    }

    public void test1() {
        System.out.println(isSafeAttribute("div", "", "div")); // returns true
        System.out.println(isSafeAttribute("span", "text", "span")); // returns true
        System.out.println(isSafeAttribute("", "class", "div")); // returns true
        System.out.println(isSafeAttribute("", "", "a")); // returns false

        test2();
    }

    private void test2() {
        System.out.println(isSafeAttribute("p", "text", "p"));
        System.out.println(isSafeAttribute("b", "class", "b"));
        System.out.println(isSafeAttribute("", "id", "b"));
        System.out.println(isSafeAttribute("", "", "img")); // returns false
    }

    public static void main(String[] args) {
        GeneratedTest test = new GeneratedTest();
        test.test1();
    }
}
