import org.junit.jupiter.api.Test;

public class ConfigMeTest {

    public String getFirstChar(String str) {
        if (str.length() > 0) {
            return "" + str.charAt(0);
        } else {
            throw new RuntimeException("String is empty");
        }
    }

    @Test
    void testMethod1() {
        int[] arr = {1, 2, 3};
        assertEquals(1, getMaxValue(arr));
        assertEquals(2, getMinValue(arr));
    }

    public String getMaxValue(int[] arr) {
        if (arr == null || arr.length == 0) {
            throw new RuntimeException("Array is empty or null");
        }
        return Arrays.stream(arr).max().orElse(null);
    }

    public String getMinValue(int[] arr) {
        if (arr == null || arr.length == 0) {
            throw new RuntimeException("Array is empty or null");
        }
        return Arrays.stream(arr).min().orElse(null);
    }

    @Test
    void testMethod2() {
        int[] arr = {1, 2, 3};
        assertEquals("first", getFirstChar(new String(arr)));
    }

}
