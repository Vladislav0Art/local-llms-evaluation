```java
public abstract class Shape {
    public abstract double getArea();
}

class Circle extends Shape {
    private double radius;

    public Circle(double radius) {
        this.radius = radius;
    }

    @Override
    public double getArea() {
        return Math.PI * (radius * radius);
    }
}

class Rectangle extends Shape {
    private double width;
    private double height;

    public Rectangle(double width, double height) {
        this.width = width;
        this.height = height;
    }

    @Override
    public double getArea() {
        return width * height;
    }
}

public class Main {
    public static void main(String[] args) {
        Shape circle = new Circle(5);
        System.out.println("Circle Area: " + circle.getArea());

        Shape rectangle = new Rectangle(4, 6);
        System.out.println("Rectangle Area: " + rectangle.getArea());
    }
}

interface Condition {
    boolean evaluate();
}

class Action implements Action {
    @Override
    public void execute() {}

    public String format() {
        return "";
    }

    public String toString() {
        return "";
    }
}
```

```java
public class Main {
    public static void main(String[] args) {
        int num1 = 5;
        int num2 = 10;

        if (num1 > num2) {
            System.out.println("Number " + num1 + " is greater than number " + num2);
        } else {
            System.out.println("Number " + num1 + " is less than or equal to number " + num2);
        }

        String fruit = "apple";
        if (fruit.equals("banana")) {
            System.out.println(fruit + " is a type of banana");
        } else {
            System.out.println(fruit + " is not a type of banana");
        }
    }

    public static class Shape {
        private int width;
        private int height;

        public Shape(int width, int height) {
            this.width = width;
            this.height = height;
        }

        public double getArea() {
            return Math.max(width * height, 0.0);
        }
    }

    public static void main2(String[] args) {
        Shape shape = new Shape(10, 20);

        if (shape.getArea() > 100) {
            System.out.println(shape.width + "x" + shape.height + " is greater than 100");
        } else {
            System.out.println(shape.width + "x" + shape.height + " is less than or equal to 100");
        }

        String[] colors = {"red", "green", "blue"};
        for (int i = 0; i < colors.length; i++) {
            if (colors[i].equals("banana")) {
                System.out.println(colors[i] + " is a type of banana");
            } else {
                System.out.println(colors[i] + " is not a type of banana");
            }
        }
    }

    public static class Number {
        int value;

        public Number(int value) {
            this.value = value;
        }

        @Override
        public String toString() {
            return "Number " + value;
        }
    }

    public static void main3(String[] args) {
        Number number1 = new Number(10);
        Number number2 = new Number(-5);

        if (number1.value > number2.value) {
            System.out.println(number1.value + " is greater than " + number2.value);
        } else {
            System.out.println(number1.value + " is less than or equal to " + number2.value);
        }
    }

    public static class StringArray {
        private String[] array;

        public StringArray(String[] array) {
            this.array = array;
        }

        @Override
        public int hashCode() {
            return 31 * Arrays.hashCode(array);
        }

        @Override
        public boolean equals(Object obj) {
            if (this == obj)
                return true;
            if (obj == null || getClass() != obj.getClass())
                return false;
            StringArray other = (StringArray) obj;
            return Objects.equals(array, other.array);
        }
    }

    public static void main4(String[] args) {
        StringArray array = new StringArray(new String[]{"apple", "banana", "cherry"});
        System.out.println(array == new StringArray(new String[]{"apple", "banana", "cherry"}));
    }
}
```
