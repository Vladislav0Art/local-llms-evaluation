```java
public class Solution {
    public int maxSubarraySum(int[] nums) {
        int maxSum = Integer.MIN_VALUE;
        int currentSum = 0;

        for (int num : nums) {
            currentSum = Math.max(num, currentSum + num);
            maxSum = Math.max(maxSum, currentSum);
        }

        return maxSum;
    }
}

class Test {
    public static void main(String[] args) {
        Solution solution = new Solution();

        int[] nums1 = {2, 3, -2, 4};
        System.out.println("Max Subarray Sum: " + solution.maxSubarraySum(nums1)); // Expected output: 6

        int[] nums2 = {-2, -3, 4};
        System.out.println("Max Subarray Sum: " + solution.maxSubarraySum(nums2)); // Expected output: 5
    }
}
```
