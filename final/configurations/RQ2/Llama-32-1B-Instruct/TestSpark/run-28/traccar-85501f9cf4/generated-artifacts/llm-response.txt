```java
        int result2 = 20;

        if (result >= result2) {
            System.out.println("The first result is greater or equal to the second result.");
        } else {
            System.out.println("The two results are different.");
        }

        String testResult = "Test Result";
        
        if (testResult.compareTo(testResult) > 0) {
            System.out.println("The first result is greater than the given string.");
        } else {
            System.out.println("The two results are equal.");
        }
    }

    public static String getDaysBetween(int start, int end) {
        int diff = end - start;
        
        if (diff < 0) {
            return "There are not enough days between";
        }
        
        double numDays = Math.floor(diff / 100);
        
        if (numDays == 28 || numDays == 29) {
            return numDays + " days are in the given month";
        } else if ((numDays == 31 && start < end) || (numDays == 30 && start == end)) {
            return numDays + " days is a leap year and ends on the day of week specified by start";
        } else {
            return numDays + " days are in the given month";
        }
    }
}
```

This program will print the current date, calculate the number of days between 1 and 15, compare two test dates or strings, and determine whether a year is a leap year. It also provides a method to get the number of days between two dates.
