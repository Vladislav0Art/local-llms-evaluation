public class SocketAddress {
    public String toString() { return "[0x" + Long.toHexString(0L) + ", " + Long.toHexString(1234L) + ", " + Long.toHexString(5678L)]"; }
}

public class AddressInfo {
    private SocketAddress address;
    private int port;

    public AddressInfo(int port, SocketAddress address) {
        this.port = port;
        this.address = address;
    }

    public String toString() { return "[" + address.toString() + ", " + port + "]"; }
}

public interface Decoder {
    Object decode(SocketChannel channel, SocketAddress remoteAddress, ByteBuf msg) throws Exception;
}

public class DecoderImpl implements Decoder {
    @Override
    public Object decode(SocketChannel channel, SocketAddress remoteAddress, ByteBuf msg) throws Exception {
        return msg.readLong();
    }
}

public class SocketChannel {
    private boolean connected;

    public void connect() throws Exception {
        connected = true;
    }

    public void disconnect() throws Exception {
        connected = false;
    }

    @Override
    public String toString() {
        return "SocketChannel{" +
                "connected=" + connected +
                '}';
    }
}

public class Unpooled {
    private byte[] buffer;

    public Unpooled(byte[] bytes) {
        buffer = bytes;
    }

    public byte[] buffer() {
        return buffer;
    }
}
