    }

    public void testGet() {
        DataFrame df = new DataFrame();
        System.out.println(df.get("A")); // prints "Column A"
        System.out.println(df.get("B")); // prints "Column B"
    }

    @Test
    public void testAdd() {
        DataFrame df = new DataFrame();
        System.out.println(df.add(1, 5)); // prints 6
        System.out.println(df.add(2, 3)); // prints 5
    }
}

class IndexTest {
    @Test
    public void testIndexGet() {
        DataFrame df = new DataFrame();
        System.out.println(df.getIndex("A")); // prints Column A
        System.out.println(df.getIndex("B")); // prints Column B
    }

    @Test
    public void testIndexAdd() {
        DataFrame df = new DataFrame();
        Index index = df.getIndex("A");
        int[] values = {1, 2};
        for (int value : values) {
            System.out.println(index.get(value)); // prints Column A at position 1
            index.add(value, value * 2);
            System.out.println(index.get(0)); // prints Column A at position 1 with new value 4
        }
    }
}

import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

public class DataFrame {
    public String get(String col) {
        return "Column " + col;
    }

    public int add(int row, int value) {
        return row + value;
    }
}

class DataFrameTest {
    @Test
    public void testGet() {
        DataFrame df = new DataFrame();
        assertEquals("Column A", df.get("A"));
        assertEquals("Column B", df.get("B"));
    }

    @Test
    public void testAdd() {
        DataFrame df = new DataFrame();
        int row = 1;
        assertEquals(6, df.add(row, 5));
        assertEquals(5, df.add(row, 3));
    }
}

class IndexTest {
    @Test
    public void testIndexGet() {
        DataFrame df = new DataFrame();
        Index index = df.getIndex("A");
        assertSame(df, index);
        index.add(0, 1);
        assertEquals(2, index.get(0));
    }

    @Test
    public void testIndexAdd() {
        DataFrame df = new DataFrame();
        int[] values = {1, 2};
        for (int value : values) {
            DataFrame row = new DataFrame();
            row.add(0, value);
            Index index = row.getIndex("A");
            assertSame(row, index);
            index.add(value * 2, value);
            assertEquals(4, index.get(0));
        }
    }

    @Test
    public void testIndexGetAndAdd() {
        DataFrame df = new DataFrame();
        Index index1 = df.getIndex("A");
        int[] values1 = {1, 2};
        for (int value : values1) {
            df.add(0, value);
            Index index2 = df.getIndex("A");
            assertSame(df, index2);
            index2.add(value * 2, value);
            assertEquals(4, index2.get(0));
        }
    }

    @Test
    public void testIndexGetAndAddMultipleTimes() {
        DataFrame df = new DataFrame();
        Index index1 = df.getIndex("A");
        int[] values1 = {1, 2};
        for (int value : values1) {
            df.add(0, value);
            Index index2 = df.getIndex("A");
            assertSame(df, index2);
            index2.add(value * 2, value);
            assertEquals(4, index2.get(0));
        }
    }

    @Test
    public void testIndexGetNullValue() {
        DataFrame df = new DataFrame();
        Index index = df.getIndex("A");
        assertEquals(null, index.get(-1));
    }
}
