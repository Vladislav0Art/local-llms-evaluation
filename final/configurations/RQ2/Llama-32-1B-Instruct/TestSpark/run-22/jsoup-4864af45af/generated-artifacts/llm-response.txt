    public boolean removeEnforcedAttribute(String tag) {

        }

        for (String attribute : getEnforcedAttributes().split(",")) {
            enforcedAttributes.add(attribute.trim());
        }
        String[] parts = tag.split("\\+");
        for (int i = 0; i < parts.length; i++) {
            if (!enforcedAttributes.contains(parts[i])) {
                parts[i] = parts[i].replaceFirst("enforce_", "");

            }

        return true; // Tag has no enforced attributes

        if (tag.contains(";")) {

                enforcedAttributes.add(part.trim());
            }
        } else if (!tag.isEmpty()) {
            enforcedAttributes.add(tag);
        }
        return enforcedAttributes;

import java.util.*;

public class Safelist {
    private Map<String, List<String>> enforcedAttributes = new HashMap<>();

    public void addEnforcedAttribute(String tag, String attribute) {
        enforcedAttributes.computeIfAbsent(tag, k -> new ArrayList<>()).add(attribute);
    }

    public boolean removeEnforcedAttribute(String tag, String attribute) {
        if (tag == null || tag.isEmpty()) {
            return false; // Tag is empty or null
        }
        List<String> attributes = getEnforcedAttributes(tag);
        for (String attr : attributes) {
            if (attr.equals(attribute)) {
                attributes.remove(attr);
                enforceAttributeRemoved(tag, attribute);
                return true;
            }
        }
        return false;
    }

    private void enforceAttributeRemoved(String tag, String attribute) {
        Set<String> enforcedAttributes = new HashSet<>(enforcedAttributes);
        enforcedAttributes.remove(attribute);
        Set<String> filteredAttributes = new HashSet<>();
        for (String attr : enforcedAttributes) {
            if (!attr.isEmpty()) {
                filteredAttributes.add(attr);
            }
        }
        enforcedAttributes.clear();
        enforcedAttributes.addAll(filteredAttributes);
    }

    public List<String> getEnforcedAttributes(String tag) {
        return enforcedAttributes.getOrDefault(tag, new ArrayList<>());
    }

    private Map<String, List<String>> getEnforcedAttributes() {
        return enforcedAttributes;
    }
}

public class TagParser {
    public boolean isSafeTag(String tag) {
        if (tag == null || tag.isEmpty()) {
            return false; // Tag is empty or null
        }
        Safelist safelist = new Safelist();
        List<String> attributes = safelist.getEnforcedAttributes(tag);
        for (String attribute : attributes) {
            safelist.removeEnforcedAttribute(attribute);
        }
        return true;
    }

    public void parse(String tag) {
        if (!isSafeTag(tag)) {
            System.out.println("Tag is not safe");
            return;
        }
        List<String> attributes = new ArrayList<>();
        for (String attribute : getEnforcedAttributes(tag).split(",")) {
            attributes.add(attribute.trim());
        }
        System.out.println(attributes);
    }

    public static void main(String[] args) {
        Safelist safelist = new Safelist();
        safelist.addEnforcedAttribute("tag1", "attr1");
        safelist.addEnforcedAttribute("tag2", "attr2");
        safelist.addEnforcedAttribute("tag3", "attr3");

        System.out.println(safelist.removeEnforcedAttribute("tag1", "attr1"));
        System.out.println(safelist.getEnforcedAttributes("tag1"));
        System.out.println(safelist.getEnforcedAttributes());

        safelist.parse("tag1,attr1");
    }
}

The removeEnforcedAttribute method removes an attribute from a tag's list of enforced attributes. It first checks if the tag is null or empty and returns false in that case. Otherwise, it retrieves the list of attributes for the tag using the getEnforcedAttributes method and removes the specified attribute.

The getEnforcedAttributes method simply returns the list of attributes for a given tag.

In the `TagParser` class, we create an instance of Safelist to manage enforced attributes for each tag.
