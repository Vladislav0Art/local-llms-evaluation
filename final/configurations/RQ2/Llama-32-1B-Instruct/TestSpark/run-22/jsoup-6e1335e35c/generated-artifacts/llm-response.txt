import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Main {

    public static String format(String html) {
        return Jsoup.parse(html).text();
    }

    public static Document parse(String url) {
        Document doc = Jsoup.connect(url).get();
        return doc.body().html();
    }

    public static void main(String[] args) {
        // Test 1
        String htmlString = "<div>Hello <span>World!</span></div>";
        System.out.println(format(htmlString));
        Document doc = parse("https://www.example.com");
        System.out.println(doc.html());

        // Test 2

        Document doc1 = parse("https://www.example.com");
        Document doc2 = parse("https://www.example.org");
        System.out.println(doc1.html() == doc2.html());
