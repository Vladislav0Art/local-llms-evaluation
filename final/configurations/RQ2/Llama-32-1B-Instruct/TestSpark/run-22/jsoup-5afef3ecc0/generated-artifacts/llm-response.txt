import org.junit.jupiter.api.Test;
import org.jsoup.helper.UrlBuilder;
import static org.junit.jupiter.api.Assertions.*;
import java.net.URI;

public class UrlBuilderTest {

    @Test
    public void testBuild() {
        String inputUrl = "http://example.com/path?param1=value1&param2=value2";
        URI uri = UrlBuilder.build(inputUrl);
        assertNotNull(uri, "Invalid URL");
        assertEquals("http://example.com/path", uri.getScheme() + "://" + uri.getHost(), "URL scheme and host are correct");
    }

    @Test
    public void testAppendKeyVal_SingleField() {
        String inputUrl = "http://example.com/path?param1=value1";
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        urlBuilder.appendKeyVal(new java.util.HashMap<String, String>() {{
            put("key", "value");
        }}); // Single field key-value pair
        URI uri = urlBuilder.build();
        assertEquals("http://example.com/path?param1=value1", uri.toString(), "URL representation is correct");
    }

    @Test
    public void testAppendKeyVal_MultipleFields() {
        String inputUrl = "http://example.com/path?param1=value1&param2=value2";
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        urlBuilder.appendKeyVal(new java.util.HashMap<String, String>() {{
            put("key", "value");
        }});
        urlBuilder.appendKeyVal(new java.util.HashMap<String, String>() {{
            put("another_key", "another_value");
        }}); // Multiple fields
        URI uri = urlBuilder.build();
        assertEquals("http://example.com/path?param1=value1&param2=value2&another_key=another_value", uri.toString(), "URL representation is correct");
    }

    @Test
    public void testAppendKeyVal_BadFormat() {
        String inputUrl = "invalid/url";
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        try {
            urlBuilder.appendKeyVal(new java.util.HashMap<String, String>() {{
                put("key", "value");
            }}); // Invalid format
        } catch (UnsupportedEncodingException e) {}
    }

    @Test
    public void testAppendKeyVal_BadFormat_Malformed() {
        String inputUrl = "http://example.com/path?param1=value1&bad_field=malformed_value";
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        try {
            urlBuilder.appendKeyVal(new java.util.HashMap<String, String>() {{
                put("key", "value");
            }}); // Malformed key
        } catch (UnsupportedEncodingException e) {}
    }

    @Test
    public void testAppendKeyVal_BadFormat_URLError() {
        String inputUrl = "http://example.com/path?param1=value1&bad/url";
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        try {
            urlBuilder.appendKeyVal(new java.util.HashMap<String, String>() {{
                put("key", "value");
            }}); // Malformed URL
        } catch (UnsupportedEncodingException e) {}
    }
}

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class UrlBuilder {
    private String inputUrl;

    public UrlBuilder(String url) {
        this.inputUrl = url;
    }

    public String build() {
        return Jsoup.parse(inputUrl).url();
    }

    public void appendKeyVal(java.util.HashMap<String, String> map) {
        for (java.util.Map.Entry<String, String> entry : map.entrySet()) {
            UrlBuilder(urlBuilder = new UrlBuilder("http://localhost:5000/" + inputUrl));
            urlBuilder.appendKeyVal(entry);
        }
    }

    public void appendValue(String value) {
        UrlBuilder(urlBuilder = new UrlBuilder(inputUrl));
        urlBuilder.appendValue(value);
    }
}

public class GeneratedTest {

    @Test
    public void testBuild() {
        String uri = "http://localhost:5000/test";
        UrlBuilder urlBuilder = new UrlBuilder(uri);
        System.out.println("Built URL: " + urlBuilder.build());
    }

    @Test
    public void testAppendKeyVal() {
        java.util.HashMap<String, String> map = new java.util.HashMap<>();
        map.put("key", "value");
        urlBuilder = new UrlBuilder("http://localhost:5000/" + new Document().save().toString());
        urlBuilder.appendKeyVal(map);
        System.out.println("URL after appending key value: " + urlBuilder.build());
    }

    @Test
    public void testAppendValue() {
        String value = "Hello, World!";
        urlBuilder = new UrlBuilder("http://localhost:5000/" + new Document().save().toString());
        urlBuilder.appendValue(value);
        System.out.println("URL after appending value: " + urlBuilder.build());
    }

    @Test
    public void testUnsupportedEncodingException() {
        java.util.HashMap<String, String> map = new java.util.HashMap<>();
        map.put("key", "value");
        try {
            urlBuilder = new UrlBuilder("http://localhost:5000/" + new Document().save().toString());
            urlBuilder.appendKeyVal(map);
        } catch (UnsupportedEncodingException e) {
            System.out.println("Caught exception: " + e.getMessage());
        }
    }

    @Test
    public void testInvalidUrl() {
        urlBuilder = new UrlBuilder("invalid-url");
        System.out.println(urlBuilder.build());
    }
}

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class UrlBuilder {
    private String inputUrl;

    public UrlBuilder(String url) {
        this.inputUrl = url;
    }

    public String build() {
        return Jsoup.parse(inputUrl).url();
    }

    public static class Builder {
        private String url;

        public Builder(String url) {
            this.url = url;
        }

        public UrlBuilder appendKeyVal(java.util.HashMap<String, String> map) {
            for (java.util.Map.Entry<String, String> entry : map.entrySet()) {
                UrlBuilder urlBuilder = new UrlBuilder(url);
                urlBuilder.appendKeyVal(entry);
            }
            return urlBuilder;
        }

        public UrlBuilder appendValue(String value) {
            UrlBuilder urlBuilder = new UrlBuilder(url);
            urlBuilder.appendValue(value);
            return urlBuilder;
        }
    }

    public String build() {
        return UrlBuilder.urlBuilder.build();
    }

    public static void main(String[] args) {
        UrlBuilder builder = new UrlBuilder("http://localhost:5000/test");
        java.util.HashMap<String, String> map = new java.util.HashMap<>();
        map.put("key", "value");
        builder.appendKeyVal(map);
        System.out.println(builder.build());
    }
}
