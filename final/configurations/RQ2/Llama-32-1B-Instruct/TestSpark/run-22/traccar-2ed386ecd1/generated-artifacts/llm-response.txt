public class TestProtocolDecoder {
    // Define ProtocolDecoder class
    public static int decode(byte[] gpsBytes) throws Exception {
        if (gpsBytes.length < 16) {
            throw new Exception("Invalid GPS data");
        }

        // Check checksum
        byte checksum = gpsBytes[15];
        for (int i = 0; i < 8; i++) {
            checksum += gpsBytes[i + 16];
        }
        if ((checksum & 0x80) != 0 || (checksum & 0xFF) != 0x3D) {
            throw new Exception("Invalid GPS data");
        }

        // Check length
        int length = gpsBytes[14];
        if (length < 2 || length > 6) {
            throw new Exception("Invalid GPS data");
        }

        return gpsBytes[15];
    }
