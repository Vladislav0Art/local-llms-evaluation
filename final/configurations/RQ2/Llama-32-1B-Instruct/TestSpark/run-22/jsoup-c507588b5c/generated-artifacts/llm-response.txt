import org.junit.Test;
import static org.junit.Assert.assertEquals;
import static org.mockito.Mockito.verify;

public class TextNodeTest {

    @Test
    public void testnodeName() {
        Node node = new TextNode("Hello");
        assertEquals(TextNode.class, node.getClass());
    }

    @Test
    public void testGetWholeText() {
        Node node = new TextNode("Hello World");
        String expected = "<text> Hello World </text>";
        assertEquals(expected, node.getWholeText());
    }

    @Test
    public void testIsBlank() {
        Node node = new TextNode("");
        boolean expected = false;
        assertEquals(expected, node.isBlank());
    }

    @Test
    public void testGetText() {
        Node node = new TextNode("Hello World");
        String text = "<text> Hello World </text>";
        assertEquals(text, node.getText());
    }

    @Test
    public void testToString() {
        Node node = new TextNode("Hello World");
        String expected = "<text> Hello World </text>";
        assertEquals(expected, node.toString());
    }

    @Test
    public void testAppend() {
        Document document = new Document();
        Appendable accum = document.append(new StringReader("Hello World"));
        Node node = new TextNode("Hello World");
        append(node, accum);
        assertEquals("<text> Hello World </text>", node.getText());
    }

    private void append(Node node, Appendable accum) {
        if (node instanceof String) {
            accum.append((String) node);
        } else if (node instanceof Appendable) {
            ((Appendable) node).append(accum);
        }
    }
}

class Document extends java.util.List<java.lang.String> {

    @Override
    public void append(java.io.IOException e, Appendable src) throws IOException {
        // do nothing
    }

    public String toString() {
        return "";
    }

    @Override
    public int hashCode() {
        return 0;
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj)
            return true;
        if (obj == null || getClass() != obj.getClass())
            return false;

        Document doc = (Document) obj;

        // Simple case: equal due to the override of equals and hashCode
        return doc.length == length && !Objects.equals(elements, doc.elements);
    }
}

class Appendable {
    private java.io.Writer writer;

    public Appendable(java.io.Writer writer) {
        this.writer = writer;
    }

    public void append(String s) throws IOException {
        writer.append(s);
    }

    @Override
    public int hashCode() {
        final int prime = 31;
        int result = 1;
        for (Object o : elements)
            result = prime * result + ((java.lang.Object[]) o).length;
        return result;
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj)
            return true;

        if (!(obj instanceof Appendable))
            return false;

        Appendable other = (Appendable) obj;

        for (Object o : elements)
            if (!o.equals(o))
                return false;

        return elements.length == other.elements.length;
    }

    public int length() {
        return count();
    }

    private java.util.List<java.lang.String> elements;

    public String toString() {
        StringBuilder sb = new StringBuilder();
        for (String s : elements)
            sb.append(s);
        return sb.toString();
    }

    private int count() {
        if (count == 0) {
            String[] strings = elements.toArray(new String[0]);
            for (String s : strings)
                count += s.length();
        }
        return count;
    }

    public Object[] elements;
}

class Node extends java.lang.Object {

    public Node(String value) {
        this("Hello World");
        setChildren(elements);
    }

    private Node(String value, java.util.List<java.lang.String> children) {
        super(value);
        this.children = children;
    }
}
