class HtmlDocument {
    private String html;

    public void appendKeyVal(String key, String value) {
        // Parse HTML and append to string
        StringBuilder sb = new StringBuilder();
        for (String text : document.text()) {
            if (text.startsWith("<key>")) {
                sb.append(text.substring(6));

            } else if (text.startsWith("<value>")) {
                sb.append(text.substring(9));
                sb.append(" -> ").append(value).append("\n");
            }
        }
        html += sb.toString().trim();
    }

    public void appendUrl(String value) {
        // Parse HTML and append to string
        StringBuilder sb = new StringBuilder();
        for (String text : document.text()) {
            if (text.startsWith("<url>")) {
                sb.append(text.substring(6));
                sb.append(" -> ").append(value).append("\n");
            } else if (text.startsWith("<key>")) {
                sb.append(text.substring(7));
                sb.append(" -> ").append(value).append("\n");
            } else if (text.startsWith("<value>")) {
                sb.append(text.substring(9));
                sb.append(" -> ").append(value).append("\n");
            }
        }
        html += sb.toString();
    }

    public void build() {
        // Parse HTML and return as string
        String result = html;
        for (String text : document.text()) {
            if (text.startsWith("<url>")) {
                result = text.substring(6);
            } else if (text.startsWith("<key>")) {
                result += " -> ";
            } else if (text.startsWith("<value>")) {

            }
        }

        this.url = url;
        appendKeyVal("key1", "value1");
        appendUrl(url);
        build();
    }

        // Save parsed HTML to file
        try (BufferedWriter writer = new BufferedWriter(new FileWriter(filename))) {
            writer.write(content);
