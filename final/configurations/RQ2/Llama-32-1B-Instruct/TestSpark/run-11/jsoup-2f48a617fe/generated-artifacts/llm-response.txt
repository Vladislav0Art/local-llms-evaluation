import org.jsoup.nodes.Document;

        // Add the head tag to the document

        headTag.appendElement(new Element("title").text(comment));

        return document;
    }

    public Document outerHtmlTail(String comment) throws Exception {
        // Create a new document
        Document document = Jsoup.parse("<html><head></head><body></body></html>");

        // Add the tail tag to the document
        Element tailTag = document.tail();
        tailTag.appendElement(new Element("meta").attribute("name", "viewport").attribute("content", "width=device-width, initial-scale=1.0"));
        tailTag.appendElement(new Element("meta").attribute("name", "description").attribute("content", comment));

        return document;
    }

    public boolean isXmlDeclaration(String comment) {
        // Check if the comment contains an XML declaration
        String xmlDeclaration = "<!DOCTYPE html PUBLIC \"-//W3C//DTD XHTML " + comment.substring(0, 1).toUpperCase() + comment.substring(1) + "\" \"http://www.w3.org/TR/xhtml1/DTD/xhtml1-transitional.dtd\">";
        return xmlDeclaration.equals(comment);
    }

    public String generateHtml(String comment) {
        Document document = Jsoup.parse("<html><head></head><body></body></html>");
        
        // Check if the HTML is valid
        if (!isXmlDeclaration(comment)) {
            throw new Exception("Invalid XML declaration");
        }
        
        return document.outerHtml();
    }

        } else {
            tailTag.appendElement(new Element("meta"));
            tailTag.appendElement(new Element("meta").attribute("name", "viewport"));
