import java.util.ArrayList;
import java.util.List;

public class DataFrame {
    private String id;
    private List<String> columns;
    private int rows;

    public DataFrame(String id, List<String> columns) {
        this.id = id;
        this.columns = new ArrayList<>(columns);
        this.rows = 0;
    }

    public void addRow() {
        this.rows++;
        System.out.println("Added row " + (this.rows - 1));
    }

    @Override
    public String toString() {
        return "DataFrame{" +
                "id='" + id + '\'' +
                ", columns=[" + String.join(", ", columns) + "]" +
                '}';
    }
}

public class Column {
    private String name;
    private DataFrame dataframe;

    public Column(String name, DataFrame dataframe) {
        this.name = name;
        this.dataframe = dataframe;
    }

    public void setRow(DataFrame dataframe) {
        for (String column : dataframe.columns) {
            if (column.equals(name)) {
                dataframe.rows--;
                System.out.println("Updated row " + (dataframe.rows - 1));
                return;
            }
        }
        throw new RuntimeException("Column not found");
    }

    @Override
    public String toString() {
        return "Column{" +
                "name='" + name + '\'' +
                '}';
    }
}

public class DataFrameComparator {
    private Column column;

    public void setColumn(Column column) {
        this.column = column;
    }

    public int compare(DataFrame other) {
        if (other.columns.contains(column.name)) {
            return 0;
        } else if (column.dataframe.rows < other.dataframe.rows) {
            return -1;
        } else if (column.dataframe.rows > other.dataframe.rows) {
            return 1;
        }
        throw new RuntimeException("Column not found");
    }

    @Override
    public String toString() {
        return "DataFrameComparator{" +
                "column=" + column +
                '}';
    }
}
