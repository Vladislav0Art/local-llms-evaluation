package com.adobe.epubcheck.tool;

import java.io.File;
import java.io.FileNotFoundException;
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class EpubChecker {

    public static void main(String[] args) {
        if (args.length != 2 || !args[0].equals("test") || !args[1].equals("file1.txt")) {
            System.out.println("Usage: java EpubChecker <test> <path_to_file>");
            return;
        }

        String path = args[1];

        try {
            File file = new File(path);
            Scanner scanner = new Scanner(file);

            while (scanner.hasNextLine()) {
                String line = scanner.nextLine();

                if (line.trim().startsWith("<?xml")) {
                    // Parse XML metadata
                    System.out.println("<xmp:Metadata xmlns:xmp=\"http://www.w3.org/2004/xmldsig#\" version=\"1.0\"><xmpData>\n" +
                            line + "\n"
                            + "</xmpData></xmp:Metadata>");

                } else if (line.trim().startsWith("<xhtml")) {
                    // Parse XHTML metadata
                    System.out.println(OPFChecker30.parseXhtmlMetadata(line, path));

                }

            }

        } catch (FileNotFoundException e) {
            System.out.println("File not found: " + path);
        }
    }

}
