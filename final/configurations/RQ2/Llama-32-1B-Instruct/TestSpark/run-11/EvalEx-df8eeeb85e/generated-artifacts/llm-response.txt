            double numberValue = Double.parseDouble(value);
            System.out.println("Test: " + variableType + "'s value is " + numberValue);
        } else if (variableType.equals("string")) {
            String stringValue = (String) value;
            System.out.println("Test: " + variableType + "'s value is '" + stringValue + "'");
