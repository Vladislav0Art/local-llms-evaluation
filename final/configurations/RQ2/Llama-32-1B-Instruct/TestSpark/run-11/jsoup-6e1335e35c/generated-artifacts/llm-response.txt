import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class Document {
    public static Element createElement(String tagName) {
        return document.createElement(tagName);
    }

    public static String createElementWithAttributes(String tagName, java.util.Map<String, Object> attributes) {
        Element elem = createElement(tagName);
        for (java.util.Map.Entry<String, Object> entry : attributes.entrySet()) {
            elem.attr(entry.getKey(), entry.getValue().toString());
        }
        return elem.toString();
    }

    public static String addClassName(Element elem, String className) {
        if (elem.hasAttr("class")) {
            String[] classes = elem.className().split("\\s+");
            classes = classes.length > 0 ? classes : new String[]{""};
            classes[classes.length - 1].equals(className) ? "" : classes.push(className);
        } else {
            elem.className(className);
        }
        return elem.toString();
    }

    public static String removeClassName(Element elem, String className) {
        if (elem.hasAttr("class")) {
            String[] classes = elem.className().split("\\s+");
            for (int i = 0; i < classes.length; i++) {
                if (!classes[i].equals(className)) {
                    classes[i] = classes[i].replaceAll(className, "");
                }
            }
        } else {
            return elem.toString();
        }
        return elem.toString();
    }

    public static String addCssSelector(Element elem, String selector) {
        if (elem.hasAttr("class")) {
            String[] classes = elem.className().split("\\s+");
            for (int i = 0; i < classes.length; i++) {
                classes[i] = classes[i].replaceAll("^\\.", "");
            }
        } else {
            elem.className("primary, secondary");
        }
        return elem.toString() + " {\n" + selector + "\n}";
    }

    public static String removeCssSelector(Element elem) {
        if (elem.hasAttr("class")) {
            String[] classes = elem.className().split("\\s+");
            for (int i = 0; i < classes.length; i++) {
                classes[i] = classes[i].replaceAll("^\\.", "");
            }
        } else {
            return "";
        }
        return elem.toString() + " {\n" + removeCssSelector(elem) + "\n}";
    }

    public static String addStyleAttr(Element elem, String styleAttr, Object styleValue) {
        if (elem.hasAttr("style")) {
            String[] styles = elem.style().split("; ");
            for (int i = 0; i < styles.length; i++) {
                styles[i] = styles[i].replaceAll("^\\.", "");
            }
            styles[styles.length - 1].equals(styleAttr) ? "" : styles.push(styleAttr);
        } else {
            elem.attr("style", styleAttr + " " + removeStyleAttribute(elem.style()));
        }
        return elem.toString() + " {\n" + addStyleAttr(elem, styleAttr, styleValue) + "\n}";
    }

    public static String removeStyleAttribute(Element elem) {
        if (elem.hasAttr("style")) {
            String[] styles = elem.style().split("; ");
            for (int i = 0; i < styles.length; i++) {
                styles[i] = styles[i].replaceAll("^\\.", "");
            }
            return elem.attr("style") + removeStyleAttribute(elem.style());
        } else {
            return "";
        }
    }

    public static Document documentWithAttributes(String htmlString, java.util.Map<String, Object> attributes) {
        Document doc = new Document();
        Elements elements = doc.select(htmlString);
        for (java.util.Map.Entry<String, Object> entry : attributes.entrySet()) {
            String tagName = entry.getKey().replaceAll("^\\.", "");
            Element elem = createElement(tagName);
            for (Object value : entry.getValue()) {
                if (value instanceof java.util.List) {
                    String[] values = ((java.util.List<?>) value).toArray();
                    for (String v : values) {
                        elem.attr(tagName + "." + v.replaceAll("^\\.", ""));
                    }
                } else {
                    elem.attr(tagName, removeStyleAttribute((String) value));
                }
            }
        }
        return doc;
    }

    public static Document document() {
        Document doc = new Document();
        Elements elements = doc.select("*");
        for (Element elem : elements) {
            String className = addClassName(elem, "primary", "secondary");
            if (!elem.hasAttr("class")) {
                className = "";
            }
            System.out.println(className);
            System.out.println(removeCssSelector(elem));
            System.out.println(addStyleAttribute(elem, "font-size: 16px;"));
        }
        return doc;
    }

    public static void main(String[] args) {
        Document document = document();
    }
}

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class Main {
    public static Element createElement(String tagName) {
        return document.createElement(tagName);
    }

    public static java.util.Map<String, Object> createMapElements(String[] keys, java.util.Map<String, String> values) {
        java.util.HashMap<String, String> map = new java.util.HashMap<>();
        for (int i = 0; i < keys.length; i++) {
            map.put(keys[i], values.get(i));
        }
        return map;
    }

    public static Document document(String htmlString) {
        Document doc = new Document();
        Elements elements = doc.select("*");
        for (Element elem : elements) {
            System.out.println(elem.className());
            System.out.println(doc.html() + "\n");
            java.util.Map<String, Object> attributesMap = createMapElements(new String[] {"class", "style"}, new java.util.HashMap<>());
            StringBuilder styleBuilder = new StringBuilder();
            if (elem.hasAttr("class")) {
                String[] classes = elem.className().split("\\s+");
                for (String c : classes) {
                    styleBuilder.append(c).append(": ").append(attributesMap.get(c)).append("; ");
                }
            } else {
                styleBuilder.append(elem.style()).append("\n");
            }
            System.out.println(styleBuilder.toString() + "\n\n");
        }
        return doc;
    }

    public static void main(String[] args) {
        java.util.Map<String, Object> attributesMap = createMapElements(new String[] {"class", "style"}, new java.util.HashMap<>());

}
