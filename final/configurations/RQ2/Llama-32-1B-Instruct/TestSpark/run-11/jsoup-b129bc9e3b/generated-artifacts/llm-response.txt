    public String getBody() {
        return body;
    }
}

    public static Response createResponse(int code, String message, String url) {
        return new Response(code, message);
    }

    public static HttpResponse createNonExistentMethod() {
        return null;
    }
}

public class Response {
    private int code;
    private String message;

    public Response(int code, String message) {
        this.code = code;
        this.message = message;
    }

    public int getCode() {
        return code;
    }

    public String getMessage() {
        return message;
    }
}

public class HttpServer {
    public static HttpResponse sendHttpResponse(int code, String message, String url) {
        // Create a new instance of Response
        Response response = new Response(code, message);
        
        return new HttpResponse().setBody(response.getBody());
    }

    public static void main(String[] args) throws Exception {
        HttpServer server = new HttpServer();
        
        // Send an HTTP request
        HttpResponse response = server.sendHttpResponse(200, "OK", "/users");
        
        System.out.println("Request Method: POST");
        System.out.println("Request URL: /users");
        System.out.println("Request Body: " + response.getBody());
    }
}
