package org.jsoup.parser;

import org.junit.Before;
import org.junit.Test;
import java.util.HashMap;
import java.util.Map;

public class Tag implements Cloneable {
    private static final String TAG_NAME = "tag";
    private static final String TAG_NORMAL_NAME = "name";
    private static final String TAG_VALUE_OF = "valueOf";

    public void setName(String name) { this.setName(name); }
    public String getName() { return this.TAG_NAME; }

    public String normalName() { return Normalizer.normalize(this.TAG_NAME); }
    public Map<String, Object> valueOf(Map<String, Object> settings) { return ValueParser.valueOf(settings.get(TAG_VALUE_OF)); }
    public static Tag valueOf(String tagName, ParseSettings settings) {
        Map<String, Object> settingsMap = new HashMap<>();
        settingsMap.put(TAG_VALUE_OF, settings);
        return ValueParser.valueOf(settingsMap);
    }

    public boolean isBlock() { return this.isBlock(); }
    public boolean formatAsBlock() { return this.formatAsBlock(); }
    public boolean isInline() { return this.isInline(); }
    public boolean isEmpty() { return this.isEmpty(); }
    public boolean isSelfClosing() { return this.isSelfClosing(); }
    public boolean isKnownTag() { return this.isKnownTag(); }

    public static boolean isKnownTag(String tagName) {
        Map<String, Object> settingsMap = new HashMap<>();
        settingsMap.put(TAG_VALUE_OF, "tag");
        ValueParser.valueOf(settingsMap);
        return ValueParser.isKnownTag(tagName);
    }
    public void setSelfClosing() { this.setSelfClosing(true); }

    @Override
    public boolean equals(Object o) {
        if (this == o) { return true; }
        if (!(o instanceof Tag)) { return false; }
        Tag tag = (Tag) o;
        return this.getName().equals(tag.getName()) && this.getNormalName().equals(tag.getNormalName());
    }

    @Override
    public int hashCode() {
        String name = Normalizer.normalize(this.TAG_NAME);
        return name.hashCode();
    }

    @Override
    public String toString() { return "Tag{" +
            "name='" + TAG_NAME + '\'' +
            ", normalName='" + Normalizer.normalize(TAG_NAME) + '\'' +
            ", valueOf=" + TAG_VALUE_OF +
            '}';
    }

    protected void setSelfClosing(boolean selfClosing) {
        this.isSelfClosing = selfClosing;
    }
}
