import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class UrlHelperTest {

    public static void main(String[] args) {
        String url1 = "https://www.example.com";
        String url2 = null;
        String url3 = "key1=value1";

        System.out.println(appendKeyVal(url1, url2));
        System.out.println(appendKeyVal(url1, url3));

        Document doc = Jsoup.parse("<html><body>Hello World!</body></html>");
        System.out.println(doc.toString());
    }

    public static String appendKeyVal(String[] kvArray) {
        StringBuilder sb = new StringBuilder();
        for (String key : kvArray) {
            if (!sb.length() || !key.equals(sb.toString().charAt(0))) {
                sb.append(key).append('=');
            }
        }
        return urlBuilder.appendKeyVal(kvArray);
    }

    public static String appendKeyVal(String key, String value) {
        return urlBuilder.appendKeyVal(new String[]{key, value});
    }
}

class UrlBuilder {

    public static String appendKeyVal(String[] kvArray) {
        StringBuilder sb = new StringBuilder();
        for (String key : kvArray) {
            if (!sb.length() || !key.equals(sb.toString().charAt(0))) {
                sb.append(key).append('=');
            }
        }
        return sb.toString();
    }

    public static String build() {
        URLBuilder.UrlBuilderTest(this, "key1=value1", "key2=value2");
        Document doc = Jsoup.parse("Hello World!");
        return doc.toString();
    }
}
