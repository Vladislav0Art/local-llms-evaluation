import java.time.LocalDate;
import java.time.temporal.ChronoUnit;

public class TestUtils {
    public static String getTodayDate() {
        return LocalDate.now().format("yyyy-MM-dd");
    }

    public static String getTomorrowDate() {
        LocalDate tomorrow = LocalDate.now().plusDays(1);
        return tomorrow.format("yyyy-MM-dd");
    }

    public static int[] getNextWeek() {
        LocalDate today = LocalDate.now();
        LocalDate nextWeek = today.plusDays(7).withDayOfWeek(6); // 6 represents Sunday
        int[] weekNumbers = new int[nextWeek.getDayOfMonth()];
        for (int i = 0; i < nextWeek.getDayOfMonth(); i++) {
            weekNumbers[i] = i + 1;
        }
        return weekNumbers;
    }

    public static int[] getPreviousMonth() {
        LocalDate today = LocalDate.now();
        int[] monthYears = new int[12];
        for (int i = 0; i < 12; i++) {
            if (today.getMonthValue() == i + 1) { // Month is zero-based
                monthYears[i] = today.getYear();
            }
        }
        return monthYears;
    }

    public static String getMonthName(int monthNumber) {
        String[] months = {"January", "February", "March", "April", "May", "June",
                "July", "August", "September", "October", "November", "December"};
        return months[monthNumber - 1];
    }

    public static int countDaysInMonth(int month, int year) {
        if (month == 2 && ((year % 4 == 0 && year % 100 != 0) || year % 400 == 0)) {
            return 29;
        } else {
            return 28;
        }
    }

    public static String getMonthDescription(int monthNumber, int year) {
        String[] months = {"January", "February", "March", "April", "May", "June",
                "July", "August", "September", "October", "November", "December"};
        return months[monthNumber - 1] + ", " + year;
    }

    public static int findDayOfWeek(int day, int month, int year) {
        LocalDate date = LocalDate.of(year, month, day);
        return date.getDayOfWeek().getValue();
    }
}

class Main {
    public static void main(String[] args) {
        System.out.println(TestUtils.getTodayDate());
        System.out.println(TestUtils.getTomorrowDate());
        System.out.println(Arrays.toString(TestUtils.getNextWeek()));
        System.out.println(Arrays.toString(TestUtils.getPreviousMonth()));
        System.out.println(TestUtils.getMonthName(3));
        System.out.println(TestUtils.countDaysInMonth(2, 2020));
        System.out.println(TestUtils.getMonthDescription(12, 2024));
        System.out.println(TestUtils.findDayOfWeek(7, 3, 2020)); // Sunday
    }
}
