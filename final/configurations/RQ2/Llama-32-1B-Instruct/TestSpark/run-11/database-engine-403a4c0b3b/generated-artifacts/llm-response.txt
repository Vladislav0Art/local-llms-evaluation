public class DatabaseManager {
    private static final String TABLE_NAME = "Test";
    private static final Object[] FIELD_NAMES = {"Name", "Age"};

    public static Hashtable<String, String> insertIntoTable(String tableName, Object[] fieldNames) {
        // Create a new table if it doesn't exist
        if (!createTable(tableName)) {
            throw new RuntimeException("Failed to create table");
        }

        // Insert data into the table
        return insertIntoTable(tableName, fieldNames);
    }

    public static Hashtable<String, String> updateTable(String tableName, Object key, Object value) {
        // Update existing record in the table
        return updateRecord(tableName, key, value);
    }

    public static Hashtable<String, String> deleteFromTable(String tableName, Object[] keys) {
        // Delete all records from the table
        return deleteAllRecords(tableName, keys);
    }

    private static boolean createTable(String tableName) {
        try {
            // Try to create the table
            return (Boolean) executeQuery("CREATE TABLE IF NOT EXISTS " + tableName + " (id INT PRIMARY KEY, name VARCHAR(255))");
        } catch (SQLException e) {
            // If the table already exists, throw an exception
            throw new RuntimeException(e.getMessage());
        }
    }

    private static Hashtable<String, String> insertIntoTable(String tableName, Object[] fieldNames) throws SQLException {
        // Insert data into the table
        String query = "INSERT INTO " + tableName + " (" + String.join(", ", fieldNames) + ") VALUES (1, ?)";
        PreparedStatement statement = getPreparedStatement(tableName);
        for (Object fieldName : fieldNames) {
            statement.setObject(1, fieldName);
        }
        return executeInsertQuery(query, tableName, Object.values(statement));
    }

    private static Hashtable<String, String> updateRecord(String tableName, Object key, Object value) throws SQLException {
        // Update existing record in the table
        String query = "UPDATE " + tableName + " SET name = ? WHERE id = ?";
        PreparedStatement statement = getPreparedStatement(tableName);
        statement.setObject(1, value);
        statement.setObject(2, key);
        return executeUpdateQuery(query, tableName, Object.values(statement));
    }

    private static Hashtable<String, String> deleteAllRecords(String tableName, Object[] keys) throws SQLException {
        // Delete all records from the table
        String query = "DELETE FROM " + tableName;
        for (Object key : keys) {
            query += " WHERE id = ?";
        }
        PreparedStatement statement = getPreparedStatement(tableName);
        int recordCount = executeUpdateQuery(query, tableName, Object.values(statement));
        if (recordCount > 0) {
            throw new RuntimeException("Removed " + recordCount + " records");
        }
        return null;
    }

    private static void executeQuery(String query) throws SQLException {
        // Execute the query
        try (PreparedStatement statement = getPreparedStatement(query)) {
            statement.execute();
        }
    }

    private static void insertIntoTable(String tableName, Object[] fieldNames) throws SQLException {
        // Insert data into the table
        String query = "INSERT INTO " + tableName + " (" + String.join(", ", fieldNames) + ") VALUES (1, ?)";
        PreparedStatement statement = getPreparedStatement(tableName);
        for (Object fieldName : fieldNames) {
            statement.setObject(1, fieldName);
        }
        executeInsertQuery(query, tableName, Object.values(statement));
    }

    private static void updateRecord(String tableName, Object key, Object value) throws SQLException {
        // Update existing record in the table
        String query = "UPDATE " + tableName + " SET name = ? WHERE id = ?";
        PreparedStatement statement = getPreparedStatement(tableName);
        statement.setObject(1, value);
        statement.setObject(2, key);
        executeUpdateQuery(query, tableName, Object.values(statement));
    }

    private static void deleteAllRecords(String tableName, Object[] keys) throws SQLException {
        // Delete all records from the table
        String query = "DELETE FROM " + tableName;
        for (Object key : keys) {
            query += " WHERE id = ?";
        }
        executeUpdateQuery(query, tableName, Object.values(statement));
    }

    private static void executeInsertQuery(String query, String tableName, Object[] values) throws SQLException {
        // Execute the insert query
        try (PreparedStatement statement = getPreparedStatement(tableName)) {
            for (Object value : values) {
                statement.setObject(1, value);
            }
            statement.execute();
        }
    }

    private static void executeUpdateQuery(String query, String tableName, Object[] values) throws SQLException {
        // Execute the update query
        try (PreparedStatement statement = getPreparedStatement(tableName)) {
            for (Object value : values) {
                statement.setObject(1, value);
            }
            statement.execute();
        }
    }

    private static PreparedStatement getPreparedStatement(String tableName) throws SQLException {
        // Get a new statement for the table
        return connection.prepareStatement(tableName);
    }
}
