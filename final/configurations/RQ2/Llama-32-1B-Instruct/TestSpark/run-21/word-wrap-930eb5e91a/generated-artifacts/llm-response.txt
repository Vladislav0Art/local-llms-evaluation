```java
import java.io.BufferedReader;
import java.io.FileReader;
import java.io.IOException;

public class TestUtils {
    public static String readFromFile(String fileName) throws IOException {
        try (BufferedReader reader = new BufferedReader(new FileReader(fileName))) {
            StringBuilder content = new StringBuilder();
            String line;
            while ((line = reader.readLine()) != null) {
                if (!line.trim().isEmpty()) {
                    content.append(line).append("\n");
                } else {
                    throw new RuntimeException("Invalid file: " + fileName);
                }
            }
            return content.toString().trim();
        } catch (IOException e) {
            throw e;
        } catch (RuntimeException e) {
            throw new RuntimeException(e.getMessage());
        }
    }

    public static int getMaximumValue() throws Exception {
        String value = readFromFile("max_value.txt");
        if (!value.isEmpty()) {
            try {
                return Integer.parseInt(value);
            } catch (NumberFormatException e) {
                throw new Exception("Invalid maximum value: " + value);
            }
        } else {
            return 0;
        }
    }

    public static void main(String[] args) throws Exception {
        System.out.println(getMaximumValue());
        System.out.println(readFromFile("test.txt"));
        System.out.println(readFromFile("max_value.txt"));
    }
}
```

This code adds some basic error checking and handling to the original code. It checks if the file exists before trying to read it, and throws a `RuntimeException` if it doesn't. If an invalid value is found in the file, it throws an exception with a more informative message.
