    public static boolean doEqual(Object o1, Object o2) {
        if (o1 instanceof java.util.List && o2 instanceof java.util.List) {
            List<?> listO1 = (List<?>) o1;
            List<?> listO2 = (List<?>) o2;

            int lenO1 = listO1.size();
            for (int i = 0; i < lenO1; i++) {
                if (!listO1.get(i).equals(listO2.get(i))) {
                    return false;
                }
            }

            return true;
        } else if (o1 instanceof java.util.Map && o2 instanceof java.util.Map) {
            Map<?, ?> mapO1 = (java.util.Map<? extends String, ? extends Object>) o1;
            Map<?, ?> mapO2 = (java.util.Map<? extends String, ? extends Object>) o2;

            int lenO1 = mapO1.size();
            for (Map.Entry<String, Object> entry : mapO1.entrySet()) {
                if (!mapO2.get(entry.getKey()).equals(entry.getValue())) {
                    return false;
                }
            }

            return true;
        } else if (o1 instanceof java.util.Collection && o2 instanceof java.util.Collection) {
            Collection<?> collectionO1 = (java.util.Collection<?>) o1;
            Collection<?> collectionO2 = (java.util.Collection<?>) o2;

            int lenO1 = collectionO1.size();
            for (int i = 0; i < lenO1; i++) {
                if (!collectionO1.get(i).equals(collectionO2.get(i))) {
                    return false;
                }
            }

            return true;
        } else {
            return o1.equals(o2);
        }
    }
