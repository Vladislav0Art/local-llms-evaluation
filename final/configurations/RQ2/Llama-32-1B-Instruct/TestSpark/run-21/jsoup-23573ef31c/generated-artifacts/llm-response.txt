import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class FormParser {
    public static Map<String, Object> parseForm(String htmlString) {
        List<Map<String, Object>> elements = new ArrayList<>();
        String[] options = { "utf-8" };

        DOMParser parser = new DOMParser();
        for (int i = 0; i < htmlString.length(); i++) {
            if (htmlString.charAt(i) == '<') {
                int j = htmlString.indexOf('>', i);
                Map<String, Object> element = parseElement(parser.parse(htmlString.substring(i + 1, j), options));
                elements.add(element);
            }
        }

        Map<String, Object> formInfo = new HashMap<>();
        for (Map<String, Object> element : elements) {
            if (element.get("id") != null && !element.get("id").equals("")) {
                formInfo.put((String) element.get("id"), element);
            } else if (element.containsKey("title")) {
                formInfo.put("title", element.get("title"));
            }

            if (element.containsKey("cssQuery")) {
                System.out.println(element.get("cssQuery"));
                formInfo.put("cssQuery", element.get("cssQuery"));
            }

            if (element.containsKey("charset")) {
                System.out.println(element.get("charset").getName());
                formInfo.put("charset", element.get("charset"));
            }
        }

        return formInfo;
    }

    private static Map<String, Object> parseElement(String htmlString, String[] options) {
        Map<String, Object> element = new HashMap<>();
        int idIndex = -1;

        for (int i = 0; i < htmlString.length(); i++) {
            if (htmlString.charAt(i) == '<') {
                int j = htmlString.indexOf('>', i);
                String tagName = htmlString.substring(i + 1, j).trim();

                switch (tagName) {
                    case "input":
                        for (int k = i; k < j; k++) {
                            if (htmlString.charAt(k) == "=" && k < j - 1) {
                                element.put((String) htmlString.substring(k, k + 2), options);
                            }
                        }
                        break;
                    default:
                        idIndex = i;
                }

                break;
            }
        }

        return element;
    }

    public static void main(String[] args) {
        String htmlString = "<form id='myForm' title='My Form'><input type='text' name='txtName' value='John Doe'/><label>Country:</label><input type='radio' name='country' id='us' value='USA'/></form>";
        Map<String, Object> formInfo = parseForm(htmlString);

        System.out.println(formInfo);
    }
}
