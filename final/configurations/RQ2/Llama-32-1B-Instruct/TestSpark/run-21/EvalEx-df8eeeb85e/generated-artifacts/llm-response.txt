```java
import java.util.ArrayList;
import java.util.List;

public class Token {
    public String type;
    public String value;

    public Token(String type, String value) {
        this.type = type;
        this.value = value;
    }
}

class PunctuationToken extends Token {
    public PunctuationToken(String type, String value) {
        super(type, value);
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        PunctuationToken p = (PunctuationToken) o;
        return Objects.equals(type, p.type) && value.equals(p.value);
    }

    @Override
    public int hashCode() {
        return Objects.hash(type, value);
    }
}

class Punctuation extends Token {
    private static final String[] PUNCTUATIONS = {".", ",", "!", ";", ":", "<", ">", "="};

    public Punctuation(String type) {
        super(type, "");
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        Punctuation p = (Punctuation) o;
        return Objects.equals(type, p.type);
    }

    @Override
    public int hashCode() {
        return Objects.hash(type);
    }
}

public class TokenParser {

    private List<Token> tokens;

    public TokenParser() {
        this.tokens = new ArrayList<>();
    }

    public void parse(String input) {
        String[] tokensArray = input.split("\\s+");
        for (String token : tokensArray) {
            if (!token.isEmpty()) {
                Token tokenObj;
                switch (token.charAt(0)) {
                    case '.':
                        tokenObj = new Punctuation(token);
                        break;
                    case '!':
                        tokenObj = new Punctuation("NOT");
                        break;
                    default:
                        for (String p : PUNCTUATIONS) {
                            if (p.equals(token)) {
                                tokenObj = new Punctuation(p);
                                break;
                            }
                        }
                }
                tokens.add(tokenObj);
            }
        }

        // Check for end of input
        while (!tokens.isEmpty() && !input.startsWith(tokens.get(tokens.size() - 1).type)) {
            tokens.remove(tokens.size() - 1);
        }
    }

    public Token parse(String input) {
        TokenParser parser = new TokenParser();
        parser.parse(input);
        return parser.tokens.get(0);
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        for (Token token : tokens) {
            if (token.type.equals("NOT")) {
                sb.append("NOT ").append(token.value).append("\n");
            } else {
                sb.append("").append(token.value).append("\n");
            }
        }
        return sb.toString();
    }
}
```

**Step 3: Create a TokenParser Class**

Now, let's create a `TokenParser` class that will parse the input string into tokens.

```java
public class TokenParser {

    private List<Token> tokens;

    public TokenParser() {
        this.tokens = new ArrayList<>();
    }

    public void parse(String input) {
        String[] words = input.split("\\s+");
        for (String word : words) {
            Token token;
            switch (word.charAt(0)) {
                case '.':
                    token = new PunctuationToken(TokenType.PUNCTUATION, ".");
                    break;
                case '!':
                    token = new PunctuationToken(TokenType.PUNCTUATION, "NOT");
                    break;
                default:
                    for (String p : word.split("")) {
                        if (p.matches("[a-zA-Z]+")) {
                            String wordValue = word.substring(1);
                            Token wordObj;
                            switch (wordValue.charAt(0)) {
                                case '.':
                                    wordObj = new WordToken(wordValue);
                                    break;
                                default:
                                    for (String w : word.split("")) {
                                        if (w.matches("[a-zA-Z]+")) {
                                            wordObj = new WordToken(w);
                                            break;
                                        }
                                    }
                            }
                            token = wordObj;
                        } else {
                            throw new RuntimeException("Invalid word: " + word);
                        }
                    }
                    tokens.add(token);
            }
        }

        // Check for end of input
        while (!tokens.isEmpty() && !input.startsWith(tokens.get(tokens.size() - 1).type)) {
            tokens.remove(tokens.size() - 1);
        }
    }

    public Token parse(String input) {
        TokenParser parser = new TokenParser();
        parser.parse(input);
        return parser.tokens.get(0);
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        for (Token token : tokens) {
            if (token.type == TokenType.PUNCTUATION) {
                sb.append(token.value).append("\n");
            } else {
                sb.append(token.value).append("\n");
            }
        }
        return sb.toString();
    }
}
```

This code will parse the sentence into tokens and print them out. It first splits the input string into words, then checks each word to see if it's a punctuation mark or a valid word. If it's a punctuation mark, it adds a `PunctuationToken` object to the list of tokens. If it's not a punctuation mark, it tries to split the word into individual characters and creates a `WordToken` object for that. The final token is added to the end of the list of tokens.
