public class DirectivesBuilderTest {

    @Test
    public void [MethodUnderTest][DirectivesBuilder]Test() {
        AnnotatedElement object = new AnnotatedElement("Greeter", "com.example.Greeter");
        ProcessingElementsContainer container = new ProcessingElementsContainer();
        DirectivesBuilder directiviesBuilder = new DirectivesBuilder(object, container);

        GraphQLDirective[] directives = directiviesBuilder.build();

        for (GraphQLDirective directive : directives) {
            GraphQLArgument argument = directive.getArgument();
            GraphQLType type = argument.getType();

            if (!type.isScalar()) {
                MockitoAnnotations.initMocks(this);
                try (Mockito.mock(argument)) {
                    ArgumentMatcher argumentMatcher = ArgumentMatchers.anyOf(ArgumentMatchers.any(GraphQLDirective.class), ArgumentMatchers.any(GraphQLArgument.class));

                    assertEquals(directives.length, 1);

                    GraphQLDirective[] expectedDirectives = new GraphQLDirective[]{new DirectiveJavaAnnotationUtil().createDirective("example.com", "Greeter", "Hello, World!", argumentMatcher)};
                    Mockito.verifyAll(Mockito.mock(argument), Mockito.times(1), ArgumentMatchers.any(GraphQLDirective.class));

                    assertEquals(expectedDirectives.length, 1);

                    Object result = directiviesBuilder.build();
                    assertEquals(expectedDirectives[0], result);

                    return;
                }
            }

            assertEquals(directives.length, 0);
        }

        return;
    }

    @Test
    public void [MethodUnderTest][DirectivesBuilder]NoReconstructable() {
        AnnotatedElement object = new AnnotatedElement("Greeter", "com.example.Greeter");
        ProcessingElementsContainer container = new ProcessingElementsContainer();

        DirectivesBuilder directiviesBuilder = new DirectivesBuilder(object, container);

        GraphQLDirective[] directives = directiviesBuilder.build();

        for (GraphQLDirective directive : directives) {
            assertEquals(directive.getArgument().getType().isScalar(), true);
        }

        return;
    }
}
