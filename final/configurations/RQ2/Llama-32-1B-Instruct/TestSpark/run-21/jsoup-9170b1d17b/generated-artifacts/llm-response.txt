Here is a simple console-based banking system in Java:

```java
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

class Account {
    private String username;
    private double balance;

    public Account(String username, double initialBalance) {
        this.username = username;
        this.balance = initialBalance;
    }

    public void deposit(double amount) {
        if (amount > 0) {
            balance += amount;
            System.out.println("Deposited " + amount + ". New balance is " + balance);
        } else {
            System.out.println("Invalid deposit amount");
        }
    }

    public void withdraw(double amount) {
        if (amount > 0 && amount <= balance) {
            balance -= amount;
            System.out.println("Withdrew " + amount + ". New balance is " + balance);
        } else if (amount <= 0) {
            System.out.println("Invalid withdrawal amount");
        }
    }

    public void displayAccountInfo() {
        System.out.println("Username: " + username + ", Balance: " + balance);
    }

    public static void main(String[] args) {
        Map<String, Account> accounts = new HashMap<>();

        Scanner scanner = new Scanner(System.in);

        while (true) {
            System.out.println("\n1. Create Account");
            System.out.println("2. Log In");
            System.out.println("3. Exit");

            System.out.print("Choose an option: ");
            int choice = Integer.parseInt(scanner.nextLine()) - 1;

            switch (choice) {
                case 0:
                    System.out.print("Enter username: ");
                    String username = scanner.nextLine();

                    System.out.print("Enter initial balance: ");
                    double initialBalance = Double.parseDouble(scanner.nextLine());

                    Account account = new Account(username, initialBalance);
                    accounts.put(username, account);

                    System.out.println("Account created successfully");
                    break;
                case 1:
                    System.out.print("Enter username: ");
                    String existingUsername = scanner.nextLine();

                    System.out.print("Enter password (leave blank to use default): ");
                    String password = scanner.nextLine();
                    if (password.isEmpty()) {
                        System.out.println("Using default password 'default'");
                        password = "default";
                    }

                    Account account = accounts.get(existingUsername);
                    if (account != null) {
                        while (true) {
                            System.out.println("\n1. Deposit");
                            System.out.println("2. Withdraw");
                            System.out.println("3. Display Account Info");
                            System.out.println("4. Log Out");

                            System.out.print("Choose an option: ");
                            int choice = Integer.parseInt(scanner.nextLine()) - 1;

                            switch (choice) {
                                case 0:
                                    System.out.print("Enter deposit amount: ");
                                    double amount = Double.parseDouble(scanner.nextLine());
                                    account.deposit(amount);
                                    break;
                                case 1:
                                    System.out.print("Enter withdrawal amount: ");
                                    amount = Double.parseDouble(scanner.nextLine());
                                    account.withdraw(amount);
                                    break;
                                case 2:
                                    account.displayAccountInfo();
                                    break;
                                case 3:
                                    account.getBalance();
                                    break;
                                case 4:
                                    System.exit(0);
                            }
                        }
                    } else {
                        System.out.println("Username does not exist");
                    }

                    break;
                case 2:
                    System.out.print("Enter username: ");
                    String usernameToLogIn = scanner.nextLine();

                    System.out.print("Enter password: ");
                    String password = scanner.nextLine();
                    if (password.isEmpty()) {
                        System.out.println("Using default password 'default'");
                        password = "default";
                    }

                    Account account = accounts.get(usernameToLogIn);
                    while (true) {
                        if (account != null && account.getUsername().equals(usernameToLogIn) && account.getPassword().equals(password)) {
                            break;
                        } else {
                            System.out.println("Invalid username or password");
                        }
                    }

                    if (account != null) {
                        scanner.nextLine(); // Consume newline left-over
                        while (true) {
                            System.out.println("\n1. Deposit");
                            System.out.println("2. Withdraw");
                            System.out.println("3. Display Account Info");
                            System.out.println("4. Log Out");

                            System.out.print("Choose an option: ");
                            int choice = Integer.parseInt(scanner.nextLine()) - 1;

                            switch (choice) {
                                case 0:
                                    System.out.print("Enter deposit amount: ");
                                    double amount = Double.parseDouble(scanner.nextLine());
                                    account.deposit(amount);
                                    break;
                                case 1:
                                    System.out.print("Enter withdrawal amount: ");
                                    amount = Double.parseDouble(scanner.nextLine());
                                    account.withdraw(amount);
                                    break;
                                case 2:
                                    account.displayAccountInfo();
                                    break;
                                case 3:
                                    account.getBalance();
                                    break;
                                case 4:
                                    System.exit(0);
                            }
                        }
                    } else {
                        System.out.println("Username or password incorrect");
                    }

                    break;

                default:
                    System.out.println("Invalid option");
            }
        }
    }
}
```

This program provides a menu-driven interface for creating accounts, logging in, and exiting. It allows users to deposit, withdraw, display account info, and log out. If the username or password is incorrect, it displays an error message.
