```java
import java.util.*;

public class MyClass {

    public void traverse(Node n) {
        if (n == null || n.left != null && n.right != null) return;
        
        System.out.println(n.value);
        List<Node> list = new ArrayList<>();
        dfs(list, n);
    }

    private void dfs(List<Node> list, Node node) {
        if (node == null) return;

        list.add(node);

        dfs(list, node.left);
        dfs(list, node.right);
    }

    public static class Node {
        int value;
        List<Node> left;
        List<Node> right;

        public Node(int value) {
            this.value = value;
            this.left = new ArrayList<>();
            this.right = new ArrayList<>();
        }
    }

    public static void main(String[] args) {
        MyClass obj = new MyClass();
        obj.root = new MyNode(1);

        // In-order traversal
        List<MyNode> list = new ArrayList<>();
        dfs(list, obj.root);
        System.out.println("In-order Traversal: ");
        for (MyNode node : list) {
            System.out.print(node.value + " ");
        }
        System.out.println("\n\n");

        // Pre-order traversal
        list.clear();
        dfs(list, obj.root);
        System.out.println("Pre-order Traversal: ");
        for (MyNode node : list) {
            System.out.print(node.value + " ");
        }
        System.out.println("\n\n");

        // Post-order traversal
        list.clear();
        dfs(list, obj.root);
        System.out.println("Post-order Traversal: ");
        for (MyNode node : list) {
            System.out.print(node.value + " ");
        }
    }

    public static void dfs(List<Node> list, Node node) {
        if (node == null) return;
        
        // Traverse left subtree
        list.add(0, node.left);
        dfs(list, node.left);

        // Traverse right subtree
        list.add(0, node.right);
        dfs(list, node.right);
    }
}

class MyNode {
    int value;
    List<Node> left;
    List<Node> right;

    public MyNode(int value) {
        this.value = value;
        this.left = new ArrayList<>();
        this.right = new ArrayList<>();
    }

    @Override
    public String toString() {
        return String.valueOf(value);
    }
}
```

This program creates a binary tree and demonstrates three different types of traversals (in-order, pre-order, and post-order) using the above approach. The output will be the values of all nodes in the tree in different orders.

```java
import java.util.*;

public class MyClass {

    public static String traverse(Node n, List<String> list) {
        if (n == null || n.left != null && n.right != null) return null;

        String current = n.value;
        list.add(current);
        List<Node> leftList = new ArrayList<>();
        List<Node> rightList = new ArrayList<>();

        // Traverse left subtree
        leftList.addAll(traverse(n.left, list));
        list.add(current + " ");

        // Traverse right subtree
        rightList.addAll(traverse(n.right, list));
        if (leftList.size() == 0 || rightList.size() == 0) return null;

        System.out.println(current + " ");
        String finalString = "";

        for (Node node : leftList) {
            finalString += node.value + " ";
        }
        for (Node node : rightList) {
            finalString += node.value + " ";
        }

        // Remove trailing space
        while (!finalString.isEmpty() && finalString.trim().isEmpty()) finalString = finalString.trim();

        return finalString;
    }

    public static void main(String[] args) {
        MyClass obj = new MyClass();
        Node root = new MyNode(1);

        // In-order traversal
        List<String> list = new ArrayList<>();
        System.out.println("In-order Traversal: " + traverse(root, list));
        System.out.println("\n\n");

        // Pre-order traversal
        System.out.println("Pre-order Traversal: " + traverse(root, list));
        System.out.println("\n\n");

        // Post-order traversal
        System.out.println("Post-order Traversal: " + traverse(root, list));
    }

    public static class MyNode {
        int value;
        List<MyNode> left;
        List<MyNode> right;

        public MyNode(int value) {
            this.value = value;
            this.left = new ArrayList<>();
            this.right = new ArrayList<>();
        }
    }
}
```
