            // Print the text content of the response

            // Get the cookies from the store
            Map<String, String> responseCookies = new HashMap<>();

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Main {
    public static void main(String[] args) throws Exception {
        // Create an instance of URL
        URL url = new URL("http://example.com");

        // Set the cookies
        Map<String, String> cookies = new HashMap<>();
        cookies.put("name", "value");
        CookieStore cookieStore = new CookieStore();
        cookieStore.addCookie(new Cookie("name", "value"));

        // Make an HTTP connection with cookies
        HttpURLConnection connection = (HttpURLConnection) url.openConnection();
        connection.setRequestMethod("HEAD");

        if (connection.getResponseCode() == 200) {
            // Parse the HTML content of the response
            Document document = Jsoup.connect(connection.getURL()).get();

            // Print the text content of the response
            System.out.println(document.body().text());

            // Get the cookies from the store
            Map<String, String> responseCookies = new HashMap<>();
            for (Element element : document.select("meta[property*=\"cookie\"]")) {
                for (Map.Entry<String, String> entry : cookieStore.getCookie("name").entrySet()) {
                    if (element.attr("content").equals(entry.getKey())) {
                        responseCookies.put(entry.getValue(), element.attr("content"));
                    }
                }
            }

            // Print the cookies
            System.out.println("Response Cookies: " + responseCookies);

        } else {
            System.out.println("Failed to get response. Status code: " + connection.getResponseCode());
        }
    }
}
