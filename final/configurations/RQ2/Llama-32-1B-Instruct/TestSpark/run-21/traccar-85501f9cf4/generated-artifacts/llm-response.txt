import java.util.Calendar;

public class Main {

    public static void main(String[] args) {
        TestHelper.testClass();
    }

    public static Calendar calendar = Calendar.getInstance();

    public static String generateDate() {
        return calendar.getTime().toString();
    }

    public static byte[] encodeData(int data) {
        return Integer.toBinaryString(data).getBytes();
    }

    public static void testCalendar() {
        System.out.println(generateDate());
        for (int i = 0; i < 10; i++) {
            String date = generateDate();
            Calendar calendar = Calendar.getInstance();
            calendar.setTimeInMillis(Integer.parseInt(date, 2) * 1000);
            Calendar newCalendar = Calendar.getInstance();
            newCalendar.setTime(calendar.getTime());
            System.out.println(newCalendar.get(Calendar.YEAR) + "-" +
                    (newCalendar.get(Calendar.MONTH) + 1) + "-" +
                    newCalendar.get(Calendar.DAY_OF_MONTH));
        }
    }

    public static boolean testEncodeData() {
        byte[] data = encodeData(123);
        for (byte b : data) {
            if ((b & 0xFF) != 0x01) return false;
        }
        return true;
    }

    public static void testProtocol() {
        System.out.println(testEncodeData());
        for (int i = 1; i < 10; i++) {
            int data = i;
            byte[] dataBytes = encodeData(data);
            System.out.println("Decode " + data + ": " + new String(dataBytes, "UTF-8"));
        }
    }

    public static void testSocket() {
        System.out.println(testProtocol());
    }

    public static void testSocket2() {
        System.out.println(generateDate());
    }

    public static void testSocket3() {
        for (int i = 0; i < 10; i++) {
            String date = generateDate();
            Calendar calendar = Calendar.getInstance();
            calendar.setTimeInMillis(Integer.parseInt(date, 2) * 1000);
            System.out.println(new Date(calendar.getTime().getTime()));
        }
    }

    public static void testSocket4() {
        byte[] data = encodeData(123);
        for (byte b : data) {
            if ((b & 0xFF) != 0x01) return;
        }
        System.out.println("Decode successful");
    }
}
