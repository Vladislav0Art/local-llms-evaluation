import java.io.File;
import java.io.IOException;

public class MyTask {

    public void doSomething(String s) {
        System.out.println("Running " + s);
    }

    public int calculate(int a, int b) {
        return a * b;
    }
}

public class Main {

    public static void main(String[] args) throws IOException {
        File file = new File("/path/to/your/file");
        
        if (file.exists()) {
            MyTask task = new MyTask();
            
            // Read the content of the file
            String content = new String(Files.readAllBytes(file.toPath()));
            
            // Replace some words with public classes and methods
            int start = 0;
            while (start < content.length()) {
                if (content.startsWith("public") && start + 6 <= content.length() && content.substring(start, start + 7).equals("+public MyTask doSomething(String s) {")) {
                    start += 7;
                } else if (content.startsWith("+public int calculate(int a, int b) {") && start >= content.length() - 9) {
                    start -= 8;
                }
            }

            // Write the modified code to the file
            String newContent = "";
            for (int i = 0; i < content.length(); i++) {
                if (content.charAt(i) == ' ') {
                    continue;
                } else if (i + 1 < content.length() && content.charAt(i + 1) == '+' && start != -1 && start + 7 <= content.length() - 1) {
                    newContent += "public class MyTask {\n";
                    for (int j = i; j <= i + 6; j++) {
                        if (!content.substring(j, j + 7).equals("+public")) {
                            newContent += "\t" + content.charAt(i) + "static void doSomething(" + content.substring(i + 1, i + 9) + ");\n";
                        }
                    }
                    for (int j = start; j < i; j++) {
                        if (!content.charAt(j).equals(' ') && !content.charAt(j).equals('+') && !content.charAt(j).equals(' ')) {
                            newContent += "\t" + content.charAt(j) + ";\n";
                        }
                    }
                    for (int j = i + 7; j <= start - 8; j++) {
                        if (!content.substring(i, i + 9).equals("+public MyTask")) {
                            newContent += "\t" + content.charAt(i + 1) + ";\n";
                        } else {
                            break;
                        }
                    }
                } else if (i == content.length() - 1 && start >= 0 && start < 7) {
                    for (int j = i; j <= i + 6; j++) {
                        newContent += "\t" + content.charAt(i) + ";\n";
                    }
                } else {
                    newContent += content.substring(i, i + 1) + "\n";
                }
            }

            Files.write(file.toPath(), newContent.getBytes());
        }
    }
}
