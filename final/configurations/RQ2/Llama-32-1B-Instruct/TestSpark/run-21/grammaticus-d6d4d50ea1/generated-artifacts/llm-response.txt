import java.io.IOException;
import java.util.HashMap;
import java.util.Map;

public class PublicClass {

    private Map<String, String> obj;

    public void write(String key) {
        if (obj.containsKey(key)) {
            System.out.println(obj.get(key));
        } else {
            obj.put(key, "Key not found");
        }
    }

    @Override
    protected void writeObject(ObjectOutputStream out, List<PrintStream> osList) throws IOException {
        Map<String, String> map = new HashMap<>();
        for (PrintStream stream : osList) {
            if (!map.containsKey(stream)) {
                map.put(stream, "Value");
            } else {
                map.put(stream, map.get(stream));
            }
            write(stream.toString());
        }

        out.writeObject(map);
    }

    public static void main(String[] args) throws IOException, ClassNotFoundException {
        PublicClass obj = new PublicClass();
        obj.obj = new HashMap<>();
        obj.write("key1");
        obj.write("value1");

        // Testing with RenamingProvider
        System.out.println(RenamingProvider.printStream());
        obj.obj.put("key2", "value2");
        obj.write("key2");
    }
}

    @Override
    public void print(String s, boolean on) throws IOException {
        super.print(s);
        write(System.lineSeparator());
    }

    public static String printStream() {
        return System.out.println();
    }
}

import java.net.*;
import java.io.*;

public class NetworkInterfaceExample {

    private static String host = "8.8.8.8";
    private static int port = 80;
    private static String path = "/";

    public void write(Object obj) {
        if (obj instanceof Map) {
            Map<String, Object> map = (Map<String, Object>) obj;
            try (Socket client = new Socket(host, port)) {
                PrintWriter writer = new PrintWriter(client.getOutputStream(), true);
                for (Map.Entry<String, Object> entry : map.entrySet()) {
                    writer.println(entry.getKey() + "=" + entry.getValue());
                }
                client.close();
            } catch (UnknownHostException e) {
                System.out.println("Unknown host: " + e.getMessage());
            } catch (IOException e) {
                System.out.println("Error writing to socket: " + e.getMessage());
            }
        } else if (obj instanceof String[]) {
            try (Socket client = new Socket(host, port)) {
                BufferedReader reader = new BufferedReader(new InputStreamReader(client.getInputStream()));
                for (String s : (String[]) obj) {
                    System.out.println(reader.readLine());
                }
                client.close();
            } catch (UnknownHostException e) {
                System.out.println("Unknown host: " + e.getMessage());
            } catch (IOException e) {
                System.out.println("Error reading from socket: " + e.getMessage());
            }
        } else if (obj instanceof List) {
            try (Socket client = new Socket(host, port)) {
                BufferedReader reader = new BufferedReader(new InputStreamReader(client.getInputStream()));
                for (Object s : (List<?>) obj) {
                    System.out.println(reader.readLine());
                }
                client.close();
            } catch (UnknownHostException e) {
                System.out.println("Unknown host: " + e.getMessage());
            } catch (IOException e) {
                System.out.println("Error reading from socket: " + e.getMessage());
            }
        } else {
            System.out.println(obj.toString());
        }
    }

    public static void main(String[] args) throws UnknownHostException, IOException {
        NetworkInterfaceExample obj = new NetworkInterfaceExample();

        // Create an instance of the class
        Person person = new Person("John Doe", 30);
        obj.obj.put("name", person.getName());
        obj.obj.put("age", person.getAge());

            this.name = name;
            this.age = age;
        }

        public String getName() {
            return name;
        }

        public int getAge() {
            return age;
        }
    }
}
