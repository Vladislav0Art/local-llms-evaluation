import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class TestTools {

    public static String generateHtml(String text) {
        return Jsoup.parse(text).toString();
    }

    public static Document outerHtmlHead(Document document, int index, ParseSettings parseSettings) {
        // Assume a method to build the outer HTML header
        return document.outerHtml().tail(index);
    }

    @FunctionalInterface
    interface ParseSettings {
        String EMPTY = "";
        String toString() = "";
    }

    public static void main(String[] args) throws Exception {

        TestTools testTools = new TestTools();

        // Example 1: Generate HTML
        String html = generateHtml("<p>This is a test</p>");
        Document document = Jsoup.parse(html);
        System.out.println(testTools.outerHtmlHead(document, 0, testTools.ParseSettings.EMPTY));

        // Example 2: Get the outer HTML header
        String outerHtml = generateHtml("<div>Test comment</div>", 10, testTools.ParseSettings.EMPTY);
        Document document = Jsoup.parse(outerHtml);
        System.out.println(testTools.outerHtmlHead(document, 0, testTools.ParseSettings.EMPTY));

        // Example 3: Get the string value of an element
        Element paragraph = document.select("p").first();
        String text = paragraph.text().trim();
        System.out.println(text);

        // Example 4: Append a string to the end of the HTML string
        StringBuilder appendable = new StringBuilder();
        appendable.append("New text");
        String html2 = testTools.outerHtmlHead(document, 0, testTools.ParseSettings.EMPTY).append(appendable.toString());
        Document document2 = Jsoup.parse(html2);
        System.out.println(document2.outerHtml().tail(0, testTools.ParseSettings.EMPTY).toString());

        // Example 5: Compare the string value of an element to another string
        String comparisonText = "This is a test";
        Element paragraph2 = document.select("p").first();

        System.out.println(result);

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class Main {

    public static void main(String[] args) throws Exception {
        String testText = "This is a test with multiple lines. This line will be ignored.";
        
        // Example 1: Generate HTML
        String html = generateHtml(testText);
        System.out.println(generateOuterHtml(testText, html));
        
        // Example 2: Get the outer HTML header
        String outerHtml = getOuterHtml(testText);
        System.out.println(outerHtml);

        // Example 3: Get the string value of an element
        String paragraphText = findText(testText, "p");
        System.out.println(paragraphText);

        // Example 4: Append a string to the end of the HTML string
        String appendable = "New text";
        String html2 = generateOuterHtml(testText, appendable);
        System.out.println(html2);

        // Example 5: Compare the string value of an element to another string
        boolean result = compareStrings(paragraphText, "This is a test.");
        System.out.println(result);

    }

    public static String generateHtml(String text) {
        return Jsoup.parse(text).toString();
    }

    public static String generateOuterHtml(String text, String html) {
        Document document = Jsoup.parse(html);
        return document.outerHtml().tail(0, 100);
    }

    public static String getOuterHtml(String text) {
        return Jsoup.parse(text).outerHtml();
    }

    public static String findText(String text, String tag) {
        Elements elements = Jsoup.parse(text).select(tag);
        if (elements.isEmpty()) {
            return "";
        }
        return elements.text().trim();
    }

    public static boolean compareStrings(String str1, String str2) {
        return str1.equals(str2);
    }
}
