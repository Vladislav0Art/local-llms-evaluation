import java.util.Arrays;

public class Gt06Position {
    private byte[] data;
    private int length;
    private byte[] offset;
    private boolean hasLength;
    private boolean hasSatellites;
    private int speed;
    private String timezone;

    public Gt06Position(byte[] data) {
        this.data = data;
        this.length = 0;
        this.offset = new byte[0];
        this.hasLength = false;
        this.hasSatellites = false;
        this.speed = 0;
        this.timezone = null;
    }

    public void decodePosition() {
        if (hasLength) return;

        if (!hasSatellites) {
            // Read speed
            for (int i = 4; i < data.length; i += 4) {
                int value = 0x00 | ((data[i] & 0xF8) << 4);
                speed = (value << 8) | (value >> 2);
                if (i + 5 <= data.length) {
                    hasSpeed = true;
                }
            }

            // Read satellites
            for (int i = 7; i < data.length; i += 3) {
                int value = 0x00 | ((data[i] & 0xF8) << 4);
                if (i + 1 <= data.length) {
                    hasSatellites = true;
                }
            }

            // Read length
            for (int i = data.length - 1; i >= 0; i -= 3) {
                int value = 0x00 | ((data[i] & 0xF8) << 4);
                if (i + 1 <= data.length) {
                    hasLength = true;
                }
            }

            return;

        } else {
            // Read length
            for (int i = 0; i < data.length; i += 3) {
                int value = 0x00 | ((data[i] & 0xF8) << 4);
                if (i + 1 <= data.length) {
                    hasLength = true;
                }
            }

            return;

        }

    }

    public Gt06Position decodeGpsPosition() {
        decodePosition();
        return this;
    }
}
