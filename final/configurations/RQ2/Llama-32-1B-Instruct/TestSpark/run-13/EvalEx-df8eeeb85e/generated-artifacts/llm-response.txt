import com.ezylang.evalex.functions.FunctionIfc;
import com.ezylang-evalex.operators.OperatorIfc;

public class TokenizerTest {

    public static void testParseInvalidToken() {
        ExpressionConfiguration configuration = new ExpressionConfiguration();
        FunctionDictionaryIfc functionDict = new FunctionDictionaryIfc();
        OperatorDictionaryIfc operatorDict = new OperatorDictionaryIfc();

        String expressionString = "1+";
        Tokenizer tokenizer = new Tokenizer(expressionString, functionDict, operatorDict);
        List<Token> tokens = tokenizer.parse();
        assertEquals(4, tokens.size());
        assertTrue(tokens.get(0).getType() == TokenType.FUNCTION_NAME);
        assertTrue(tokens.get(1).getType() == TokenType.DOT);
        assertTrue(tokens.get(2).getType() == TokenType.NAME);
        assertTrue(tokens.get(3).getType() == TokenType.STRING_LITERAL);
    }

}
