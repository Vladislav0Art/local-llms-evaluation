```java
public class Solution {
    @Test
    public void testGetStockPrice() {
        System.out.println(Stock.getStockPrice(1)); // Should print 5
        System.out.println(Stock.getStockPrice(0)); // Should print -6
    }
}

class Stock {
    public static int getStockPrice(int index) {
        if (index == 1 || index == 0) {
            return index * 2;
        } else {
            return Math.abs(index);
        }
    }
}
```
