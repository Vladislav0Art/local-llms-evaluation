```java
public class GeneratedTest {
    public static void main(String[] args) {
        String inputUrl = "https://www.google.com";
        String expectedResult = "https://www.google.com/search?q=jsoup-e1880ad73e";

        try {
            URLBuilder urlBuilder = new UrlBuilder(new URI(inputUrl));
            urlBuilder.appendKeyVal("c", "3");
            urlBuilder.appendKeyVal("d", "4");
            urlBuilder.appendKeyVal("e", "5");

            String actualResult = urlBuilder.build();

            System.out.println(actualResult);
        } catch (UnsupportedEncodingException e) {
            System.err.println(e.getMessage());
        }
    }

    public static void appendKeyVal(KeyVal keyVal, String defaultValue) {
        if (keyVal != null && !keyVal.isEmpty()) {
            System.out.print(keyVal.getKey() + "=" + keyVal.getValue() + ", ");
        } else {
            System.out.print(defaultValue + ", ");
        }
    }

    public static class UrlBuilder {
        private URI url;
        private String query;

        public UrlBuilder(URI inputUrl) {
            this.url = inputUrl;
        }

        public UrlBuilder appendKeyVal(String key, String value) {
            if (query == null || !query.isEmpty()) {
                query = "";
            }
            query += key + "=" + value + "&";
            return this;
        }

        public void build() throws UnsupportedEncodingException {
            if (!url.getScheme().equals("https") && !url.getScheme().equals("http")) {
                throw new IllegalArgumentException("Only 'https' and 'http' schemes are allowed.");
            }
            String[] queryParts = url.getQuery().split("&");
            StringBuilder sb = new StringBuilder();
            for (String part : queryParts) {
                if (!part.isEmpty()) {
                    String[] keyValue = part.split("=");
                    if (keyValue.length == 2) {
                        appendKeyVal(new KeyVal(keyValue[0].trim(), keyValue[1].trim()), "");
                    }
                    sb.append(keyValue[0]).append("=").append(keyValue[1]).append(",");
                }
            }
            String result = sb.toString();
            if (!result.isEmpty()) {
                result = new URI(result).toString();
            } else {
                result = "";
            }

            System.out.println("Result: " + result);

            return result;
        }
    }
}
```

```java
import java.net.URI;
import java.util.HashMap;
import java.util.Map;

public class GeneratedTest {
    public static void main(String[] args) throws URISyntaxException {
        String inputUrl = "https://www.google.com";
        String expectedResult = "https://www.google.com/search?q=jsoup-e1880ad73e";

        try {
            UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
            urlBuilder.appendKeyVal("c", 3);
            urlBuilder.appendKeyVal("d", 4);
            urlBuilder.appendKeyVal("e", 5);

            Map<String, String> params = new HashMap<>();
            params.put("c", "3");
            params.put("d", "4");
            params.put("e", "5");

            String actualResult = urlBuilder.build(params);

            System.out.println(actualResult);

        } catch (URISyntaxException e) {
            System.err.println(e.getMessage());
        }
    }

    public static void appendKeyVal(Map<String, String> keyVal, String defaultValue) {
        if (!keyVal.isEmpty()) {
            System.out.print(keyVal.get("c") + "=" + keyVal.get("d") + "," + keyVal.get("e") + ", ");
        } else {
            System.out.print(defaultValue + ", ");
        }
    }

    public static class UrlBuilder {
        private URI url;
        private Map<String, String> queryParams;

        public UrlBuilder(String inputUrl) throws URISyntaxException {
            this.url = new URI(inputUrl);
        }

        public UrlBuilder appendKeyVal(String key, String value) throws URISyntaxException {
            if (url.getQuery() != null && !url.getQuery().isEmpty()) {
                return this;
            } else {
                queryParams = new HashMap<>();
                queryParams.put(key, value);
                return this;
            }
        }

        public void build(Map<String, String> params) throws URISyntaxException {
            if (!url.getScheme().equals("https") && !url.getScheme().equals("http")) {
                throw new IllegalArgumentException("Only 'https' and 'http' schemes are allowed.");
            }

            URI uri = url;
            for (Map.Entry<String, String> entry : queryParams.entrySet()) {
                if (!uri.getQuery().contains(entry.getKey())) {
                    uri.setQuery(uri.getQuery() + entry.getKey() + "=" + entry.getValue());
                }
            }

            System.out.println("Result: " + uri.getScheme() + "://" + uri.getHost() + "?" + uri.getQuery());

        }
    }
}
```
