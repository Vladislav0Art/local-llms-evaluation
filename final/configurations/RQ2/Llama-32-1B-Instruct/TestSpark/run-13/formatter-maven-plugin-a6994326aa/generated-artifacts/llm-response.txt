package net.revelc.code.formatter.css;

import com.steadystate.css.dom.CSSStyleSheetImpl;
import com.steadystate.css.format.CSSFormat;
import com.steadystate.css.parser.CSSOMParser;
import com.steadystate.css.parser.SACParserCSS3;
import net.revelc.code.formatter.AbstractCacheableFormatter;
import net.revelc.code.formatter.ConfigurationSource;
import net.revelc.code.formatter.Formatter;
import net.revelc.code.formatter.LineEnding;

public class CssFormatterTest {

    public void testCssFormatterInitialization() {
        CssFormatter formatter = new CssFormatter();
        assert formatter.isInitialized();
    }

    public void testCssFormatterDoFormat() throws IOException {
        InputSource inputSource = new StringReader("body { background-color: #f2f2f2; }");
        CSSOMParser parser = new SACParserCSS3(inputSource);
        CSSStyleSheetImpl stylesheet = (CSSStyleSheetImpl) parser.getCssStyleSheet();
        String formattedCode = formatter.doFormat(stylesheet.getCssText(), LineEnding.LF_ALL);
        assertEquals("body { background-color: #f2f2f2; }", formattedCode);
    }

    public void testCssFormatterDoFormatWithMultipleSelectors() throws IOException {
        InputSource inputSource = new StringReader("body { background-color: #f2f2f2; } .class1 { color: red; } .class2 { color: blue; }");
        CSSOMParser parser = new SACParserCSS3(inputSource);
        CSSStyleSheetImpl stylesheet = (CSSStyleSheetImpl) parser.getCssStyleSheet();
        String formattedCode = formatter.doFormat(stylesheet.getCssText(), LineEnding.LF_ALL);
        assertEquals("body { background-color: #f2f2f2; } .class1 { color: red; } .class2 { color: blue; }", formattedCode);
    }

    public void testCssFormatterIsInitialized() {
        CssFormatter formatter = new CssFormatter();
        assert !formatter.isInitialized();
        formatter.init(new HashMap<>(), null); // no arguments should be passed to the init method
        assert formatter.isInitialized();
    }
}

        final ConfigurationSource configSource = new ConfigurationSource();
        final CssFormatterConfiguration config = new CssFormatterConfiguration(configSource, formatter);
        final CSSDeclarationList declarationList = new CSSDeclarationList();

        // test the formatter
        Object result = formatter.format(declarationList);
        if (result instanceof String) {
            System.out.println(((String) result));
        }
    }

}

class CssFormatterConfiguration implements ConfigurationConfiguration {
    private final ConfigurationSource configSource;
    private final AbstractCacheableFormatter formatter;

    public CssFormatterConfiguration(ConfigurationSource configSource, AbstractCacheableFormatter formatter) {
        this.configSource = configSource;
        this(formatter = formatter);
    }

    @Override
    public String[] getPriority() {
        return new String[0];
    }

    @Override
    public Object format(final CSSDeclarationList declarationList) {
        // implementation of the format method
        StringBuilder sb = new StringBuilder();
        for (CSSDeclaration declaration : declarationList.getDeclarations()) {
            if (declaration instanceof CSSStyleRule) {
                sb.append(((CSSStyleRule) declaration).getSelector().trim());
                sb.append("\n");
            }
        }
        return sb.toString();
    }
}
