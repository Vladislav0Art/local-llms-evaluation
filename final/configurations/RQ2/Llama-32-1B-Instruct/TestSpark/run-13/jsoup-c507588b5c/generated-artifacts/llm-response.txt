        return doc;
    }

    public static void parseTag(Element element, String string) {
        Element parent = element.parent();
        while (parent != null && !element.hasPreviousSibling()) {
            element.parent().after(element); // Move the tag after its parent
            parent = parent.prevParent(); // Change to prevParent instead of prevSibling
        }

        if (parent == null) {
            parent = document.createElement("div");
            element.parent().appendChild(parent);
        } else {
            parent.appendChild(element);
        }
    }

        StringReader reader = new StringReader(string);
        return parse(reader.toString());
    }

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class GeneratedTest {

    public static Document parse(String string) {
        Document doc = new Document();
        Elements tags = doc.getAllElements();

        for (Element tag : tags) {
            if (tag.tagName().equals("text")) {
                String text = tag.text();
                assertEquals(string, text);
            } else if (tag.tagName().equals("div")) {
                parseTag(tag, string);
            }
        }

        return doc;
    }

    public static void parseAppendable(String string) {
        StringReader reader = new StringReader(string);
        Document parsedDoc = parse(reader.toString());
        Element root = parsedDoc.querySelector("body"); // Select the first child element in the document
        if (root == null) {
            System.out.println("No body tag found");
        } else {
            Element bodyTag = root.SelectSingleNode(".//div[@class='text']"); // Find a div with class 'text' inside the body tag
            if (bodyTag != null) {
                assertEquals(string, bodyTag.text());
            }
        }
    }

    public static void parseAppendableWithParser(String string) {
        Document doc = parseAppendable(new DocumentBuilder().parse(string));
    }

    @Test
    public void testText() {
        GeneratedTest.parse("Hello World!");
    }

    @Test
    public void testBodyTag() {
        GeneratedTest.parseAppendableWithParser("This is a test");
    }
}

class DocumentBuilder {

    public static Document parse(String builder) {
        String[] tags = {"text", "div"};
        Document doc = new Document();
        for (String tag : tags) {
            Element element = doc.createElement(tag);
            if (!tag.equals("text")) {
                element.text(builder);
            }
        }
        return doc;
    }

    public static void main(String[] args) {
        ParseTests.ParseTests.parseTests.main(new String[]{"", "Hello World!"});
        ParseTests.ParseTests.parseTests.main(new String[]{"This is a test", "div class='text' text='This is a body tag!'"});
    }
}
