import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class ResponseTest {

    public static CookieStore response(String message) {
        Document document = Jsoup.parse("html");
        Element responseElement = document.createElement("div");
        responseElement.setTextContent(message);
        return new CookieStore(responseElement);
    }

    public static void main(String[] args) {
        ConnectionResponse connectionResponse = response("Hello, World!");
        System.out.println(connectionResponse);

        String response = response("Cookie: test = \"value\";");
        for (Element element : response.getResponseBody().select("[type='text']")) {
            System.out.println(element.text());
        }
    }

    public static class CookieStore {
        private Element element;

        public CookieStore(Element element) {
            this.element = element;
        }

        @Override
        public String toString() {
            return element.toString();
        }

        public String getResponseBody() {
            StringBuilder responseBody = new StringBuilder();
            for (Element element : element.children()) {
                if (element.tagName().equals("div")) {
                    responseBody.append(element.text()).append("\n");
                } else {
                    System.out.println(element);
                }
            }
            return responseBody.toString();
        }

    }

}
