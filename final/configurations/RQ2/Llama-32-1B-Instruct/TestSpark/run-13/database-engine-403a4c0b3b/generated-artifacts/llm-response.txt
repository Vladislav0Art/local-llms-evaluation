        // Generate table schema
        String tableSchema = "CREATE TABLE users (id INT PRIMARY KEY, name VARCHAR(255), age INT);";

        // Insert data into the table
        System.out.println("Inserting data into the 'users' table:");
        for (String row : data) {
            String[] columns = {"name", "age"};
            String[] values = new String[columns.length];
            for (int i = 0; i < columns.length; i++) {
                if (row != null && row.length > i) {
                    values[i] = row[i];
                } else {
                    values[i] = "NULL";
                }
            }
            String insertQuery = "INSERT INTO users (" + String.join(", ", columns) + ") VALUES (" + String.join(", ", values) + ");";
            System.out.println(insertQuery);
        }
    }
