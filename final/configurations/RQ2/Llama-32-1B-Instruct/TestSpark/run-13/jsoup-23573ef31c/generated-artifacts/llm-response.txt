import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class DocumentTest {

    private Document document;

    @Before
    public void setup() {
        String baseUri = "http://example.com";
        try (Jsoup html = Jsoup.parse(baseUri)) {
            document = html;
        } catch (Exception e) {
            fail("Failed to parse HTML: " + e.getMessage());
        }
    }

    public String getHtml() {
        return document.toString();
    }

    public void testGetLinkText() {
        Document document = new Document();
        Element rootElement = document.createElement("a");
        rootElement.setTextContent("Click me!");
        String linkText = document.title().split(":")[1].trim();
        assertEquals(linkText, rootElement.text());
    }

    public void testGetLinkUrl() {
        Document document = new Document();
        Element rootElement = document.createElement("a");
        rootElement.attr("href", "#");
        String linkUrl = document.title().split(":")[1].trim();
        assertEquals(linkUrl, rootElement.attr("href"));
    }

    public void testGetLinkTarget() {
        Document document = new Document();
        Element rootElement = document.createElement("a");
        rootElement.attr("target", "_blank");
        String linkUrl = document.title().split(":")[1].trim();
        assertEquals(linkUrl, rootElement.attr("target"));
    }

    public void testGetLink.rel() {
        Document document = new Document();
        Element rootElement = document.createElement("a");
        rootElement.attr("href", "#");
        String linkRel = document.title().split(":")[1].trim();
        assertEquals(linkRel, rootElement.attr("rel"));
    }

    public void testGetImgSrc() {
        Document document = new Document();
        Element rootElement = document.createElement("img");
        rootElement.attr("src", "image.jpg");
        String imgSrc = document.title().split(":")[1].trim();
        assertEquals(imgSrc, rootElement.attr("src"));
    }

    public void testGetImgAlt() {
        Document document = new Document();
        Element rootElement = document.createElement("img");
        rootElement.attr("alt", "An image");
        String imgAlt = document.title().split(":")[1].trim();
        assertEquals(imgAlt, rootElement.attr("alt"));
    }

    public void testGetImageClass() {
        Document document = new Document();
        Element rootElement = document.createElement("img");
        rootElement.attr("class", "image-class");
        String imageClass = document.title().split(":")[1].trim();
        assertEquals(imageClass, rootElement.attr("class"));
    }

    public void testGetImageSize() {
        Document document = new Document();
        Element rootElement = document.createElement("img");
        rootElement.attr("src", "image.jpg");
        String imageSize = document.title().split(":")[1].trim();
        assertEquals(imageSize, rootElement.attr("size"));
    }

    public void testGetImageAltText() {
        Document document = new Document();
        Element rootElement = document.createElement("img");
        rootElement.attr("src", "image.jpg");
        String imageSize = document.title().split(":")[1].trim();
        assertEquals(imageSize, rootElement.attr("width").split(":")[0]);
    }

    public void testGetLinkAnchor() {
        Document document = new Document();
        Element rootElement = document.createElement("a");
        rootElement.text("Click me!");
        String linkAnchor = document.title().split(":")[1].trim();
        assertEquals(linkAnchor, rootElement.attr("href").split("#")[0]);
    }

    public void testGetLinkHoverText() {
        Document document = new Document();
        Element rootElement = document.createElement("a");
        rootElement.text("Click me!");
        String linkHoverText = document.title().split(":")[1].trim();
        assertEquals(linkHoverText, rootElement.attr("title"));
    }

    public void testGetTitle() {
        Document document = new Document();
        Element rootElement = document.createElement("html");
        rootElement.appendChild(document.createElement("head"));
        Element headElement = document.createElement("head");
        headElement.appendChild(document.createElement("title"));
        String title = document.title().split(":")[1].trim();
        assertEquals(title, headElement.text());
    }

    public void testGetCharset() {
        Document document = new Document();
        Element rootElement = document.createElement("html");
        element.appendChild(rootElement);
        String charset = document.html().split("\\x7f;")[0];
        assertEquals(charset, "ISO-8859-1");
    }
}
