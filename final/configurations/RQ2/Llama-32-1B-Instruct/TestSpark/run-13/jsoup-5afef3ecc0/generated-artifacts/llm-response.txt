```java
import java.net.URLEncoder;
import java.nio.charset.StandardCharsets;

public class Main {
    public static void main(String[] args) {
        String url = "https://example.com";
        Map<String, String> kv = new HashMap<>();

        System.out.println("Public URL: " + url);

        kv.put("key", "value");
        System.out.println(appendKeyVal(kv));

        kv.put("another", "value2");

        System.out.println(url + "?" + appendKeyVal(kv));
    }

    public static String appendKeyVal(Map<String, String> kv) {
        StringBuilder sb = new StringBuilder();
        for (Map.Entry<String, String> entry : kv.entrySet()) {
            if (sb.length() > 0) {
                sb.append("&");
            }
            sb.append(URLEncoder.encode(entry.getKey(), StandardCharsets.UTF_8));
            sb.append("=");
            sb.append(URLEncoder.encode(entry.getValue(), StandardCharsets.UTF_8));
        }
        return url + "?" + sb.toString();
    }
}
```

This program constructs a URL by appending key-value pairs to the base URL. The `appendKeyVal()` method takes in a map of key-value pairs and appends them to the base URL with the corresponding query parameters.

Note that this program uses `URLEncoder.encode()` to encode special characters in the key and value to prevent any encoding errors when they are appended to the URL.

```java
import java.util.HashMap;
import java.util.Map;

public class UrlBuilder {
    private String inputUrl;

    public void setInputUrl(String url) {
        if (url == null || !url.startsWith("http://") && !url.startsWith("https://")) {
            throw new NullPointerException("Input URL must be a valid URL");
        }
        this.inputUrl = url;
    }

    public Map<String, String> appendKeyVal() {
        Map<String, String> kv = new HashMap<>();
        return inputUrl != null ? new HashMap<>(inputUrl.split("&")) : null;
    }

    public void appendKeyVal(String key, String value) {
        if (key == null || value == null) {
            throw new NullPointerException("Both key and value must be provided");
        }
        Map<String, String> kv = appendKeyVal();
        kv.put(key, value);
        setInputUrl(inputUrl + "?" + appendKeyVal().toString());
    }

    public void printInputUrl() {
        if (inputUrl != null) {
            System.out.println("Input URL: " + inputUrl);
        } else {
            System.out.println("No input URL provided");
        }
    }
}
```
