public class DataFrame {
    private List<DataRow> data;

        this.name = name;
        this.data = new ArrayList<>();
    }

    public void addRow(DataRow row) {
        data.add(row);
    }

    @Override
    public int hashCode() {
        final int prime = 31;
        int result = 1;
        for (Object row : data) {
            if (row instanceof String)
                ((String)row).hashCode();
            else if (row instanceof Object[]) {
                Object[] array = (Object[])row;
                for (int i = 0; i < array.length; i++) {
                    result *= prime;
                    result += ((Integer)array[i]).hashCode();
                }
            } else {
                throw new IllegalArgumentException("Can only add arrays of strings or objects");
            }
        }
        return prime * result + data.hashCode();
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj)
            return true;
        if (obj == null || getClass() != obj.getClass())
            return false;

        return Objects.equals(name, other.name);

import java.util.ArrayList;
import java.util.List;

public class DataFrame {
    private List<DataRow> data;
    private String name;

    public DataFrame(String name) {
        this.name = name;
        this.data = new ArrayList<>();
    }

    /**
     * Adds a row to the DataFrame.
     *
     * @param row The row to add.
     */
    public void addRow(DataRow row) {
        if (row != null && !row.toString().isEmpty()) {
            data.add(row);
        } else {

        }
    }

    /**
     * Returns a string representation of the DataFrame.
     *
     * @return A string representing the DataFrame.
     */
    public String toString() {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < data.size(); i++) {
            if (i > 0) {
                sb.append("\n");

            if (data.get(i).get(0) instanceof String)
                sb.append(((String)data.get(i).get(0))).append(" ");
            else
                for (Object column : data.get(i).get(1)) {
                    if (column instanceof String)
                        sb.append(column).append(" ");
                    else
                        sb.append(((String[])column)[0]).append(" ");
                }
            sb.append("\n");
        }

         *
