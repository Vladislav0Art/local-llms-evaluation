import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class TextSplitter {
    public static void main(String[] args) throws Exception {
        String input = "Lorem ipsum dolor sit amet, consectetur adipiscing elit. Sed do eiusmod tempor incididunt ut labore et dolore magna aliqua.";
        
        // Split text into substrings of 20 characters
        StringBuilder result1 = new StringBuilder();
        for (int i = 0; i < input.length() - 19; i++) {
            if (i == input.length() - 19) {
                String lastSubstring = input.substring(i, input.length());
                result1.append(lastSubstring).append("\n");
            } else {
                String substring = input.substring(i, Math.min(i + 20, input.length()));
                result1.append(substring);
                if (i != input.length() - 19) {
                    result1.append(" ");
                }
            }
        }

        // Split text into substrings of 30 characters
        StringBuilder result2 = new StringBuilder();
        for (int i = 0; i < input.length() - 29; i++) {
            if (i == input.length() - 29) {
                String lastSubstring = input.substring(i, input.length());
                result2.append(lastSubstring).append("\n");
            } else {
                String substring = input.substring(i, Math.min(i + 30, input.length()));
                result2.append(substring);
                if (i != input.length() - 29) {
                    result2.append(" ");
                }
            }
        }

        System.out.println("Split text into substrings of 20 characters:");
        System.out.print(result1.toString());
        
        System.out.println("\nSplit text into substrings of 30 characters:");
        System.out.print(result2.toString());

    }
}
```

public class TextSplitter {
    public static void main(String[] args) throws Exception {

        StringBuilder result1 = new StringBuilder();

            if (i == input.length() - 19) {
                String lastSubstring = input.substring(i, input.length());

                String substring = input.substring(i, Math.min(i + 20, input.length()));
                result1.append(substring);
                if (i != input.length() - 19) {

        // Split text into substrings of 30 characters
        StringBuilder result2 = new StringBuilder();
        for (int i = 0; i < input.length() - 29; i++) {
