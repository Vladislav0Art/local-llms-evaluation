import java.util.ArrayList;
import java.util.List;

public class DirectivesBuilderTest {

    public static void main(String[] args) {
        // Create instances of the DirectivesBuilder class under test
        AnnotatedElement annotatedElement1 = createAnnotatedElement();
        AnnotatedElement annotatedElement2 = createAnnotatedElement();

        List<GraphQLDirective> directives1 = new ArrayList<>();
        List<GraphQLDirective> directives2 = new ArrayList<>();

        // Test DirectivesBuilder with a single element
        DirectivesBuilder builder1 = new DirectivesBuilder(annotatedElement1, null);
        List<GraphQLDirective> result1 = builder1.build();
        assertEquals(directives1.size(), result1.size());
        for (int i = 0; i < directives1.size(); i++) {
            if (i == 0) {
                assertEquals(DirectiveJavaAnnotationUtil.directiveType, directives1.get(i).getDirectiveType());
            } else {
                assertEquals(directives2.get(i - 1).getDirectiveType(), directives1.get(i).getDirectiveType());
            }
        }

        DirectivesBuilder builder2 = new DirectivesBuilder(annotatedElement2, null);
        List<GraphQLDirective> result2 = builder2.build();
        assertEquals(directives2.size(), result2.size());

class GraphQLDirective {
    private String type;
    private List<GraphQLValue> defaultValues;

    public GraphQLDirective(String type, List<GraphQLValue> defaultValues) {
        this.type = type;
        this.defaultValues = defaultValues;
    }

    public String getDirectiveType() {
        return type;
    }

    public List<GraphQLValue> getDefaultValues() {
        return defaultValues;
    }
}

class DirectivesBuilder {
    private AnnotatedElement element;
    private List<GraphQLDirective> directives;

    public DirectivesBuilder(AnnotatedElement element, List<GraphQLDirective> directives) {
        this.element = element;
        this.directives = directives;
    }

    public List<GraphQLDirective> build() {
        // Simulate building the list of directives
        return new ArrayList<>(directives);
    }
}

class DirectiveJavaAnnotationUtil {
    public static String directiveType = "directive";
    public static String directiveType2 = "directive2";

    public static String getDirectiveType(GraphQLDirective directive) {
        return directive.getDirectiveType();
    }

    public static List<GraphQLValue> getDefaultValues(GraphQLDirective directive) {
        return directive.getDefaultValues();
    }
}

class GraphQLValueFactory {
    public static GraphQLValue createString(String value) {
        return new GraphQLValue("string", value);
    }

    public static GraphQLValue createInt(int value) {
        return new GraphQLValue("int", value);
    }
}
