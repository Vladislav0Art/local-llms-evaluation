import java.io.BufferedReader;
import java.io.IOException;
import java.net.HttpURLConnection;
import java.net.URL;

public class Main {
    public static void main(String[] args) throws IOException, InterruptedException {
        String url = "https://jsonplaceholder.typicode.com/posts";
        Thread thread = new Thread(() -> {
            try {
                HttpURLConnection connection = (HttpURLConnection) new URL(url).openConnection();
                connection.setRequestMethod("GET");
                int responseCode = connection.getResponseCode();

                if (responseCode == 200) {
                    System.out.println(connection.getInputStream().readAllBytes());
                } else {
                    System.out.println("Failed to retrieve data. Response code: " + responseCode);
                }
            } catch (IOException e) {
                System.out.println("Error: " + e.getMessage());
            }
        });
        thread.start();
        thread.join();
    }

    public static void testGet(String url) throws IOException {
        try {
            Connection connection = new MainConnection();
            connection.addResponse(new Response(connection.response(url)));
        } catch (IOException e) {
            System.out.println("Error: " + e.getMessage());
        }
    }
}

class Response {
    private byte[] data;

    public Response(byte[] data) {
        this.data = data;
    }

    @Override
    public String toString() {
        return "Response{" +
                "data='" + new String(data) + '\'' +
                '}';
    }
}

public class MainConnection {
    public void addResponse(Response response) {
        responses.add(response);
    }

    private Response makeRequest(String url, String method) throws IOException {
        URL url1 = new URL(url);
        HttpURLConnection connection = (HttpURLConnection) url1.openConnection();
        connection.setRequestMethod(method);

        int responseCode = connection.getResponseCode();

        if (responseCode == 200) {
            return new Response(connection.getInputStream().readAllBytes());
        } else {
            System.out.println("Failed to retrieve data. Response code: " + responseCode);
            return null;
        }
    }

    private static class MethodRequest implements Runnable {
        private String method;
        private URL url;

        public MethodRequest(String method, String url) {
            this.method = method;
            this.url = new URL(url);
        }

        @Override
        public void run() {
            try {
                System.out.println(url);
                Thread.sleep(1000); // simulate a delay
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        }
    }
}
