```java
public class GeneratedTest {
    public static WeatherData getWeatherData(String date, String temperature, String humidity) {
        try {
            int temp = Integer.parseInt(temperature);
            int humidityInt = Integer.parseInt(humidity);
            return new WeatherData(date, temp, humidityInt);
        } catch (NumberFormatException e) {
            System.out.println("Invalid temperature or humidity value");
            return null;
        }
    }

    public static void main(String[] args) {
        WeatherData weatherData = getWeatherData("2024-03-16", "25", "60");
        if (weatherData != null) {
            System.out.println(weatherData.getTemp());
            System.out.println(weatherData.getHumidity());
        }
    }
}

public class WeatherData {
    private String date;
    private int temp;
    private int humidity;

    public WeatherData(String date, int temp, int humidity) {
        this.date = date;
        this.temp = temp;
        this.humidity = humidity;
    }

    public void setTemp(int temp) {
        if (temp >= 0 && temp <= 100) {
            this.temp = temp;
        } else {
            throw new IllegalArgumentException("Temperature must be between 0 and 100");
        }
    }

    public int getTemp() {
        return temp;
    }

    public void setHumidity(int humidity) {
        if (humidity >= 0 && humidity <= 100) {
            this.humidity = humidity;
        } else {
            throw new IllegalArgumentException("Humidity must be between 0 and 100");
        }
    }

    public int getHumidity() {
        return humidity;
    }
}
```
