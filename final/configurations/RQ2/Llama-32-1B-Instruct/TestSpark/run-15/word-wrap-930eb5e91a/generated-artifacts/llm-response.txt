package org.davidmoten.text.utils;

import java.util.Arrays;
import java.util.List;

public final class StringUtilities {
    public static <T> List<T> split(CharSequence s, int threshold) {
        if (s == null || threshold <= 0) return new ArrayList<>();
        Preconditions.checkArgument(s != null && threshold > 0);

        List<String> result = new ArrayList<>();
        int i = 0;

        while (i < s.length()) {
            int j = Math.max(i + threshold, s.length() - 1);
            char[] cArray = s.toCharArray();
            for (; i <= j; ) {
                if (cArray[i] != ' ') break;
                result.add(new String(cArray, i, j - i + 1));
                i++;
            }
        }

        return result;
    }

    public static <T> List<T> split(String s, int threshold) {
        if (s == null || threshold <= 0) return new ArrayList<>();
        Preconditions.checkArgument(s != null && threshold > 0);

        List<String> result = Arrays.asList();
        int i = 0;

        while (i < s.length()) {
            int j = Math.max(i + threshold, s.length() - 1);
            char[] cArray = s.toCharArray();
            for (; i <= j; ) {
                if (cArray[i] != ' ') break;
                result.add(new String(cArray, i, j - i + 1));
                i++;
            }
        }

        return result;
    }

    public static <T> List<T> split(String s, int threshold) {
        if (s == null || threshold <= 0) return new ArrayList<>();
        Preconditions.checkArgument(s != null && threshold > 0);

        List<String> result = new ArrayList<>(Arrays.asList(s));
        for (String string : Arrays.asList(s)) {
            result.add(string);
            String[] substrings = split(string, threshold);
            List<T> list = new ArrayList<>(substrings);
            result.addAll(list);
        }

        return result;
    }
}

public class LineSplitter {
    public static <T> void splitLine(T line, int maxLines) {
        Preconditions.checkArgument(line != null && maxLines > 0);

        int numLines = Arrays.stream(line.toString().split("\n")).limit(maxLines).count();
        StringBuilder builder = new StringBuilder();

        for (int i = 0; i < numLines; i++) {
            if (i == maxLines - 1) {
                String[] substrings = split(line, 2);
                list.addAll(Arrays.asList(substrings));
            } else {
                builder.append("\n").append(substrings.get(i + 1));
            }
        }

        System.out.println(builder.toString());
    }

    public static <T> void splitLine(String line, int maxLines) {
        Preconditions.checkArgument(line != null && maxLines > 0);

        List<String> substrings = StringUtilities.split(line, 2);
        list.addAll(substrings.subList(1, maxLines));
    }

package org.davidmoten.text.utils;

import java.nio.charset.StandardCharsets;
import java.util.*;

public class HashCodeCalculator {
    public static int calculateHashCode(Object obj) {
        return Objects.hash(obj);
    }

    public static <T> int calculateHashCode(T obj) {
        Preconditions.checkArgument(obj != null);

        int hashCode = 0;
        for (int i = 0; i < obj.getClass().getDeclaredFields(); i++) {
            try {
                obj.getClass().getField(i.toString()).set(obj, null);
            } catch (NoSuchFieldException | IllegalAccessException e) {}

            hashCode = Math.abs(hashCode ^ Objects.hash(getFieldValue(obj, i)));

        }
        return hashCode;
    }

    public static List<Integer> getHashCodes(Object[] objArray) {
        int size = objArray.length;

        List<Integer> hashCodes = new ArrayList<>();
        for (int i = 0; i < size; i++) {
            hashCodes.add(calculateHashCode(objArray[i]));
        }
        return hashCodes;
    }

    public static List<Integer> getHashCodes(Set<Object> set) {
        List<Integer> hashCodes = new ArrayList<>();

        for (Object obj : set) {
            hashCodes.add(calculateHashCode(obj));
        }
        return hashCodes;
    }
}

public class StringBuilder {
    private StringBuilder builder;

    public StringBuilder build(StringBuilder builder) {
        this.builder = builder;
        return builder;
    }

    public static void main(String[] args) {
        List<String> strings = Arrays.asList("Hello", "World");
        List<StringBuilder> lineBuilders = new ArrayList<>();
        for (String string : strings) {
            lineBuilders.add(new StringBuilder());
        }
        StringStringBuilder lineBuilder = new StringStringBuilder();
        lineBuilder.append("Line 1\n").append("Line 2\n");
        for (StringBuilder builder : lineBuilders) {
            builder.append(lineBuilder);
        }
