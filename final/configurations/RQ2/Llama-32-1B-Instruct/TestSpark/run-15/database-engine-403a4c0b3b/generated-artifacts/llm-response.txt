class SQLTerm {
    private String sql;
    private String[] conditions;
    private String value;

    public SQLTerm(String sql, String[] conditions, String value) {
        this.sql = sql;
        this.conditions = conditions;
        this.value = value;
    }

    @Override
    public String toString() {
        return "(" + sql + ", " + conditions[0] + ", '" + value + "'");
    }
}

public class DatabaseEngine {

    public SQLTerm generateSql(String query) {
        System.out.println("Generating SQL: " + query);
        return new SQLTerm(query, new String[]{"column1"}, "value");
    }

    public static void main(String[] args) {
        GeneratedTest generatedTest = new GeneratedTest();
        
        List<SQLTerm> sqlTerms = new ArrayList<>();
        DatabaseEngine databaseEngine = new DatabaseEngine();
        for (int i = 0; i < 10; i++) {
            SQLTerm term = generatedTest.executeQuery(databaseEngine.executeQuery());
            sqlTerms.add(term);
        }
        System.out.println("Generated Terms:");
        for (SQLTerm term : sqlTerms) {
            System.out.println(term.toString());
        }
    }
