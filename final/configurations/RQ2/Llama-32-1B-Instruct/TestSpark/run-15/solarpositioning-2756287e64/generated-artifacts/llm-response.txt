public class Main {

    public static String calculateSolarPosition(double latitude) {
        if (latitude < 0) {
            return "The solar position is negative.";
        } else if (latitude > 23.45 || latitude < -22.46) {
            return "Invalid input for latitude.";
        } else {
            double hour = getHour(date, latitude);
            String solarPosition = calculateSolarPosition(hour);
            if (!solarPosition.equals("Test Solar Position")) {
                return "Invalid date";
            }
            System.out.println("The current solar position is at: " + solarPosition);

            return "Positive: 3-4 hours before noon" +
                    (latitude < 23.45 ? (latitude - 22) : (latitude + 1)) +
                    ", Negative: between noon and 3/4 of a full moon." +
                    (latitude > -22.46 ? (latitude + 1) : (-22.46));
        }
    }

    public static String calculateSolarPosition(double hour) {
        if (hour < 0 || hour > 24) {
            return "Invalid input for hour.";
        } else {
            double decimalHour = Math.round(hour * 100 / 360);
            System.out.println("The current solar position is at: " + decimalHour);
            return "Positive: between noon and 3/4 of a full moon." +
                    (decimalHour < 12 ? "before" : "after") +
                    (decimalHour > 18 ? "" : "at");
        }
    }

    public static int getHour(ZonedDateTime date, double latitude) {
        return (int) Math.floor(date.getOffset() / (1.0 * 24.0 * 60.0 * 3600.0));
    }

    public static void main(String[] args) {
        double latitude = 37.7749;
        System.out.println(calculateSolarPosition(latitude));
    }
}

import java.time.*;
import java.time.format.DateTimeFormatter;

public class Main {

    public static String getMonth(int year) {
        DateTimeFormatter formatter = DateTimeFormatter.ofPattern("yyyy-MM");
        LocalDateTime date = LocalDateTime.now();
        return date.format(formatter).toString().substring(6, 8);
    }

    public static int getDayOfMonth(int month, int year) {
        if (month == 2 && isLeapYear(year)) {
            return 29;
        } else {
            return 28;
        }
    }

    public static boolean isLeapYear(int year) {
        return year % 4 == 0 && (year % 100 != 0 || year % 400 == 0);
    }

    public static void main(String[] args) {
        int month = 5; // May
        System.out.println(getMonth(month));
        System.out.println(getDayOfMonth(month, getYear()));
    }
}

class LeapYearChecker {

    private boolean isLeapYear(int year) {
        return (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0);
    }

    public static void main(String[] args) {
        int year = 2020;
        if (isLeapYear(year)) {
            System.out.println("The year " + year + " is a leap year.");
        } else {
            System.out.println("The year " + year + " is not a leap year.");
        }
    }
}

import java.util.*;

class WeatherData {

    private String date;
    private double temperature;

    public WeatherData(String date, double temperature) {
        this.date = date;
        this.temperature = temperature;
    }

    @Override
    public String toString() {
        return "Date: " + date + ", Temperature: " + temperature;
    }
}

public class Main {

    public static void main(String[] args) {
        List<WeatherData> weatherData = new ArrayList<>();
        weatherData.add(new WeatherData("2022-01-01", 5));
        weatherData.add(new WeatherData("2022-02-28", 7));
        weatherData.add(new WeatherData("2023-03-31", 10));

        Collections.sort(weatherData, Comparator.comparingDouble(WeatherData::getTemperature));

        for (WeatherData data : weatherData) {
            System.out.println(data);
        }
    }
}
