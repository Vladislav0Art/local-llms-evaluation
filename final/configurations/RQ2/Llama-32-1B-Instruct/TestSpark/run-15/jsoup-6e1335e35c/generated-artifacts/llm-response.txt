        Document document = new Document();
        document.append(text);
        assertEquals("Hello, world! This is a test.", document.wholeText());
    }

    @Test
    public void testWholeOwnText() {
        String text = "Hello, world!";
        Document document = new Document();
        document.append(text);
        assertEquals("Hello, world!", document.wholeOwnText());
    }

    @Test
    public void testOwnText() {
        String text = "This is a test.";
        Document document = new Document();
        document.append(text);
        assertEquals("This is a test.", document.ownText());
    }
}
