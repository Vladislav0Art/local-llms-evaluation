import ch.jalu.configme.util.TestUtils;
import org.junit.BeforeClass;
import org.junit.Test;

public class CommentsConfigurationTest {

    private static CommentsConfiguration configuration;

    @BeforeClass
    public static void setUp() {
        configuration = new CommentsConfiguration();
    }

    @Test
    public void setCommentSetsCorrectly(@NotNull String path, @NotNull String... commentLines) {
        // Arrange
        String[] expectedCommentLines = {"", "", "Hello World!"};

        // Act
        List<String> actualCommentLines = configuration.setComment(path, commentLines);

        // Assert
        TestUtils.assertListEquals(expectedCommentLines, actualCommentLines);
    }

    @Test
    public void getAllCommentsReturnsCorrectly() {
        // Arrange
        Map<String, List<String>> expectedComments = new HashMap<>();

        // Act
        Map<String, List<String>> actualComments = configuration.getAllComments();

        // Assert
        TestUtils.assertMapEquals(expectedComments, actualComments);
    }
