import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class Main {

    public static Document generateHtml() {
        Document doc = Jsoup.parse("<html><body></body></html>");
        return doc;
    }

    public static String getCssClass(String html) {
        Elements elements = Jsoup.parse(html).select("div.class");
        for (Element element : elements) {
            if (element.text().equals("class")) {
                return "public";
            }
        }
        return "";
    }

    public static void main(String[] args) {
        Document doc = generateHtml();
        System.out.println(doc.html());
        
        String cssClass = getCssClass(doc.html());
        System.out.println(cssClass);
        
        Element element = doc.selectFirst("div.class").firstElement();
        System.out.println(element.attr("class"));
    }
}

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class Main {

    public static void main(String[] args) {
        Document doc = generateHtml();
        
        // Print out the generated HTML
        System.out.println(doc.html());
        
        // Retrieve the CSS class name from the input string
        String cssClass = getCssClass(args[0]);
        
        // Print out the retrieved CSS class name
        if (cssClass != null) {
            System.out.println(cssClass);
        } else {
            System.out.println("No CSS classes found.");
        }
    }

    public static Document generateHtml() {
        Document doc = Jsoup.parse("<html><body></body></html>");
        return doc;
    }

    public static String getCssClass(String html) {
        // Use regular expression to find all class attributes in the HTML
        Pattern pattern = Pattern.compile("\\.\\w+\\.");
        Matcher matcher = pattern.matcher(html);
        
        // If any matches are found, join them with a comma and return as string
        if (matcher.find()) {
            StringBuilder sb = new StringBuilder();
            while (matcher.find()) {
                sb.append(matcher.group()).append(",");
            }
            return sb.toString().trim();
        } else {
            return "";
        }
    }
}
