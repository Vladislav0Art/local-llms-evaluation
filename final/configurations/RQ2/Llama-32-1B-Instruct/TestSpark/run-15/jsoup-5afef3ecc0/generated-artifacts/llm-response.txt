public class TestJsoup {

    }

    public void testConnect() throws Exception {
        Document document = Jsoup.connect(mockUrl.resource()).get();
        System.out.println(document.outerHtml());
    }

    @Test
    public void testConnectMockedUrl() {
        Document document = Jsoup.connect(mockUrl.resource()).get();
        System.out.println(document.outerHtml());
    }

    private class MockUrl {
        public String resource() throws Exception {
            return "mock_url";
        }
    }

    private Element jsoupElement;

    @Test
    public void testJsoupElementGettersAndSetters() {
        Document document = Jsoup.connect(mockUrl.resource()).get();
        jsoupElement = document.selectFirst("body");
        assert jsoupElement != null;
        assert jsoupElement.tagName().equals("BODY");
    }

    private class MockDocument extends Document {

        @Override
        public String toString() {

        }
