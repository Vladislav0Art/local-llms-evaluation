import org.junit.Test;
import static org.junit.Assert.*;

public class KeyPairTest {

    @Test
    public void testCanSign() {
        KeyPair keyPair = new KeyPair(new EdDSAPublicKey("eddsa_key"));
        assertEquals(true, keyPair.canSign());
    }

    @Test
    public void testFromSecretSeed_SeedSize() {
        char[] seed = "secret_seed".getBytes();
        KeyPair keyPair = KeyPair.fromSecretSeed(seed);
        assertNull(keyPair.getSecretSeed());
        assertNotNull(keyPair.getPublicKey());
    }

    @Test
    public void testFromSecretSeed_Bip39Seed() {
        byte[] bip39Seed = "bip39_seed".getBytes();
        KeyPair keyPair = KeyPair.fromBip39Seed(bip39Seed, 0);
        assertNotNull(keyPair.getSecretSeed());
        assertNull(keyPair.getPublicKey());
    }

    @Test
    public void testFromSecretSeed_GeneralSeed() {
        char[] generalSeed = "general_seed".getBytes();
        KeyPair keyPair = KeyPair.fromSecretSeed(generalSeed);
        assertNotNull(keyPair.getSecretSeed());
        assertNotNull(keyPair.getPublicKey());
    }

    @Test
    public void testGetAccountId_SmallAccountId() {
        String smallAccountNumber = "small_account_number";
        KeyPair keyPair = KeyPair.fromAccountId(smallAccountNumber);
        assertEquals(smallAccountNumber, keyPair.getAccountId());
    }

    @Test
    public void testGetSecretSeed_SmallSecretSeed() {
        char[] smallSecretSeed = "small_secret_seed".getBytes();
        KeyPair keyPair = new KeyPair(new EdDSAPublicKey("eddsa_key"));
        assertEquals(smallSecretSeed, keyPair.getSecretSeed());
    }

    @Test
    public void testGetPublicKey_SmallPublicKey() {
        byte[] smallPublicKey = "small_public_key".getBytes();
        KeyPair keyPair = new KeyPair(new EdDSAPublicKey("eddsa_key"));
        assertEquals(smallPublicKey, keyPair.getPublicKey());
    }

    @Test
    public void testSigning_SignatureHint() {
        SignatureHint signatureHint = SignatureHint.PALINDROMIC;
        KeyPair keyPair = new KeyPair(new EdDSAPublicKey("eddsa_key"));
        DecoratedSignature decoratedSignature = keyPair.signDecorated(signatureHint.getBytes());
        assertNotNull(decoratedSignature);
    }

    @Test
    public void testSigning_SignerPayload() {
        byte[] signerPayload = "signer_payload".getBytes();
        KeyPair keyPair = new KeyPair(new EdDSAPublicKey("eddsa_key"));
        DecoratedSignature decoratedSignature = keyPair.signPayloadDecorated(signerPayload);
        assertNotNull(decoratedSignature);
    }

    @Test
    public void testVerification_Signature() {
        byte[] data = "data".getBytes();
        byte[] signature = "signature".getBytes();
        KeyPair keyPair = new KeyPair(new EdDSAPublicKey("eddsa_key"));
        assertTrue(keyPair.verify(data, signature));
    }

    @Test
    public void testVerification_NoSignature() {
        byte[] data = "data_no_signature".getBytes();
        byte[] signature = null;
        assertFalse(keyPair.verify(data, signature));
    }
}
