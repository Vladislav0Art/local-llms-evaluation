public class GrammarParser {

            return ((String) o).matches(regex);
        }
        throw new RuntimeException("Invalid input type");
    }
}

        String[] terms = {"noun", "verb", "adjective"};
        pattern = new Pattern(terms, "[a-zA-Z]+");
        matcher = Pattern.compile(pattern);

        while (matcher.find()) {
            tokens.add(matcher.group());

        return tokens;
    }

class Pattern implements Matcher {
    private String[] terms;
    private String regex;

    public Pattern(String[] terms, String regex) {
        this.terms = terms;
        this.regex = regex;
    }

    @Override
    public boolean matches(Scanner scanner) {
        while (scanner.hasNext()) {
            String match = scanner.next();
            if (match.matches("[a-zA-Z]+")) return true;
            else return false;
        }
        return false;
    }

    @Override
    public Matcher[] matches(Scanner scanner, int start) {
        List<Matcher> matches = new ArrayList<>();
        for (int i = 0; i < terms.length; i++) {
            if (terms[i].matches("[a-zA-Z]+")) {
                String match = scanner.next();
                if (match.matches("[a-zA-Z]+")) matches.add(new Matcher(match));
            } else return null;
        }
        return matches.toArray(new Matcher[0]);
    }

    @Override
    public boolean applyGrammar(Scanner scanner) {
        // Ignore tokens that don't match the pattern
        return true;
    }
}

class Matcher {
    private String match;
