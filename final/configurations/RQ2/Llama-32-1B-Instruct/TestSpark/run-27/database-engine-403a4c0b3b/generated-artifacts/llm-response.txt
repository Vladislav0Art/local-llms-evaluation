### Problem Description
Given a database object `db`, create functions for performing common operations such as selecting data from tables, deleting records, updating table columns, and inserting new records.

### Solution

```java
import java.util.Arrays;

public class Database {
    private String tableName;
    private String[] columns;
    private Object[][] data;
    private int rowCount;

    public Database(String tableName) {
        this.tableName = tableName;
        this.columns = new String[0];
        this.data = new Object[0][0];
        this.rowCount = 0;
    }

    // Insert a record into the table
    public void insert(Object[] data) {
        for (Object item : data) {
            System.out.println(item);
            columns = Arrays.copyOf(columns, columns.length + 1);
            columns[columns.length - 1] = String.valueOf(data.length);
            data = Arrays.copyOf(data, data.length + 1);
        }
        rowCount++;
    }

    // Delete a record from the table
    public void delete(int index) {
        if (index < 0 || index >= rowCount) {
            System.out.println("Invalid index");
            return;
        }
        Object[] newData = Arrays.copyOf(data, data.length - 1);
        System.arraycopy(newData, 0, newData, 0, index);
        System.arraycopy(newData, index + 1, newData, index, rowCount - index - 1);
        rowCount--;
    }

    // Update a record in the table
    public void update(int index, Object value) {
        if (index < 0 || index >= rowCount) {
            System.out.println("Invalid index");
            return;
        }
        data[index][columns.length - 1] = String.valueOf(value);
    }

    // Select records from a table based on given conditions
    public void selectFromTable(String condition) {
        if (condition == null || !condition.contains("WHERE") || !condition.contains("AND")) {
            System.out.println("Invalid condition");
            return;
        }
        String[] columnValuePairs = condition.split(" WHERE ");
        for (String pair : columnValuePairs) {
            String[] columnsAndValues = pair.trim().split(" AND ");
            if (columnsAndValues.length != 2) {
                System.out.println("Invalid condition");
                continue;
            }
            String column = columnsAndValues[0].trim();
            String value = columnsAndValues[1].trim();
            update(column, value);
        }
    }

    public static void main(String[] args) {
        Database db = new Database("Test");

        // Insert records
        Object[] record1 = {1, 11, "10"};
        Object[] record2 = {2, 21, "20"};

        db.insert(record1);
        db.insert(record2);

        // Delete a record
        int indexToDelete = 0;
        db.delete(indexToDelete);

        // Update a record
        String updateCondition = "WHERE id = 2";
        db.selectFromTable(updateCondition);

        // Insert new records
        Object[] newRecord1 = {3, 31, "30"};
        Object[] newRecord2 = {4, 41, "40"};

        db.insert(newRecord1);
        db.insert(newRecord2);

        db.update("id", 3);
    }
}
```

```python
class Database:
    def __init__(self, tableName):
        self.tableName = tableName
        self.columns = []
        self.data = []
        self.rowCount = 0

    # Insert a record into the table
    def insert(self, data):
        for item in data:
            if len(item) == 1:
                continue
            new_row = [str(i) + ": " + str(data[i]) for i in range(len(item))]
            self.data.append(new_row)
        self.rowCount += len(data)

    # Delete a record from the table
    def delete(self, index):
        if index < 0 or index >= self.rowCount:
            print("Invalid index")
            return

        new_data = [row[:index] + row[index+1:] for row in self.data]
        self.data = new_data
        self.rowCount -= 1

    # Update a record in the table
    def update(self, index, value):
        if index < 0 or index >= self.rowCount:
            print("Invalid index")
            return

        self.data[index][index + 1] = str(value)
        self.rowCount -= 1

    # Select records from a table based on given conditions
    def select_from_table(self, condition):
        if not condition:
            print("Invalid condition")

        column_value_pairs = condition.split(" WHERE ")
        for pair in column_value_pairs:
            columns_and_values = pair.strip().split(" AND ")
            if len(columns_and_values) != 2:
                print("Invalid condition")
                continue

            column, value = columns_and_values
            try:
                column_index = int(column)
                self.update(column_index, value)
            except ValueError:
                print(f"Error: Invalid column index for {column}")
```

```java
public class Database {

    public Database(String tableName) {
        this.tableName = tableName;
        this.data = new HashMap<>();
    }

    // Insert a record into the table
    public void insert(Object... values) {
        for (Object value : values) {
            addRow((String[]) Arrays.copyOf(value, 1));
        }
    }

    // Delete a record from the table
    public void delete(int index) {
        if (index < 0 || index >= data.size()) {
            System.out.println("Invalid index");
            return;
        }
        remove(index);
    }

    // Update a record in the table
    public void update(int index, Object value) {
        if (index < 0 || index >= data.size()) {
            System.out.println("Invalid index");
            return;
        }
        data.put(index, Arrays.copyOf(value, 1));
    }

    // Select records from a table based on given conditions
    public Map<String, List<Object>> selectFromTable(String condition) {
        if (!condition.isEmpty()) {
            String[] columns = condition.split(" WHERE ");
            for (String column : columns[0].split(", ")) {
                data.put(column, Arrays.asList(new Object[] {column, "unknown"}));
            }
            return new HashMap<>();
        } else {
            return new HashMap<>();
        }
    }

    private void addRow(Object[] row) {
        if (!data.containsKey(tableName)) {
            data.put(tableName, new ArrayList<>());
        }
        data.get(tableName).add(row);
    }

        if (index < 0 || index >= data.size()) {

            return;
```

```python
class Database:

    # Insert a record into the table
    def insert(self, values):
        rows = [tuple(value) for value in values]
        if len(rows) == 1:
            return
        new_row = tuple(0 for _ in range(len(values)))
        new_row.extend(rows)
        self.data[self.tableName] = list(new_row)

    # Delete a record from the table
    def delete(self, index):
        if index < 0 or index >= len(self.data[self.tableName]):
            print("Invalid index")
            return

        del self.data[self.tableName][index]

    # Update a record in the table
    def update(self, index, value):
        if index < 0 or index >= len(self.data[self.tableName]):
            print("Invalid index")
            return

        new_value = [value]
        new_value.extend(self.data[self.tableName].index(index))
        self.data[self.tableName][index] = new_value

    # Select records from a table based on given conditions
    def select_from_table(self, condition):

            return {}

        for column in columns[0].split(", ")[:-1]:
            value = [column]
            self.data[self.tableName].append(value)
        return {self.tableName: self.data[self.tableName]}
```
