import java.util.Map;

public class ConfigurationManager {

    public static List<Property> addMultipleProperties(Property property, Map<String, Object> map) {
        if (map == null || !(map instanceof Map)) {
            throw new ConfigMeException("Map is required");
        }

        List<Property> result = new ArrayList<>();
        for (String key : map.keySet()) {
            Property prop = (Property) map.get(key);
            if (!(prop instanceof Object[])) {
                prop = prop.getClass().cast(new Object[]{});
            }
            result.add(prop);
        }
        return result;
    }

    public static void getMultipleProperties(Property property, Map<String, Object> map) {
        if (map == null || !(map instanceof Map)) {

package ch.jalu.configme.configurationdata;

import ch.jalu.configme.exception.ConfigMeException;
import ch.jalu.configme.properties.Property;
import org.jetbrains.annotations.NotNull;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;

public class PropertyListBuilder {

    private Map<String, Object> rootEntries = new LinkedHashMap<>();

    public void addProperty(Property property) {
        if (!(property instanceof Object[])) {
            throw new ConfigMeException("Property must be an array");
        }
        for (Object obj : (Object[]) property) {
            if (!(obj instanceof Property)) {
                throw new ConfigMeException("All properties in the root entries must be objects of type Property");
            }
            String key = (String) obj;
            addRootEntry(key, ((Property) obj));
        }
    }

    private void addRootEntry(String key, Property property) {
        if (rootEntries.containsKey(key)) {
            throw new ConfigMeException("Key already exists in the root entries");
        }
        rootEntries.put(key, property);
    }

    public Property getRootEntries() {
        return rootEntries;
    }

}
