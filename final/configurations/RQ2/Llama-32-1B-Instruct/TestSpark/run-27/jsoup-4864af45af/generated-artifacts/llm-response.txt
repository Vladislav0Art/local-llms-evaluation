public enum SafelistType {
    NONE,
    SIMPLE_TEXT,
    BASIC,
    RELAXED
}

public class SafeListImpl implements Safelist {
    private Map<String, Set<String>> map;

    public SafeListImpl() {
        this.map = new HashMap<>();
    }

    @Override
    public Safelist none(Safelist copy) {
        Safelist result = new SafeListImpl();
        for (Safelist safeList : copy.getSafeLists()) {
            result.map.putAll(safeList.map);
        }
        return result;
    }

    @Override
    public String simpleText() {
        Map<String, Set<String>> map = getMap();
        Iterator<String> iterator = map.keySet().iterator();
        StringBuilder builder = new StringBuilder();
        while (iterator.hasNext()) {
            builder.append(iterator.next()).append(" ");
        }
        return builder.toString().trim();
    }

    @Override
    public String basic(Safelist copy) {
        Safelist result = none(copy);
        for (Safelist safeList : getSafeLists(result)) {
            safeList.map.forEach((tag, attributes) -> {
                safeList.addAttributes(tag, attributes);
            });
        }
        return simpleText();
    }

    @Override
    public Safelist addTags(String... tags) {
        Safelist result = none(this);
        for (String tag : allTags()) {
            if (!result.map.containsKey(tag)) {
                SafeListImpl safeList = new SafeListImpl();
                safeList.map.putAll(getMap());
                result.map.put(tag, safeList.map);
            }
        }
        return result;
    }

    @Override
    public Safelist removeTags(String... tags) {
        Safelist result = none(this);
        for (Safelist safeList : getSafeLists(result)) {
            for (String tag : allTags()) {
                if (!tag.equals(tags[0])) {
                    safeList.removeAttributes(tag, tags[1]);
                }
            }
        }
        return result;
    }

    @Override
    public Safelist addAttributes(String tag, String... attributes) {
        Safelist result = none(this);
        for (Safelist safeList : getSafeLists(result)) {
            safeList.addEnforcedAttribute(tag, attributes[0], attributes[1]);
        }
        return result;
    }

    @Override
    public Safelist removeAttributes(String tag, String... attributes) {
        Safelist result = none(this);
        for (Safelist safeList : getSafeLists(result)) {
            safeList.removeEnforcedAttribute(tag, attributes[0], attributes[1]);
        }
        return result;
    }

    @Override
    public Safelist addEnforcedAttribute(String tag, String attribute, String value) {
        Safelist result = none(this);
        for (Safelist safeList : getSafeLists(result)) {
            SafeListImpl safeListImpl = new SafeListImpl();
            safeListImpl.map.putAll(safeList.map);
            safeListImpl.addAttributes(tag, attribute, value);
            result.map.put(tag, safeListImpl.map);
        }
        return result;
    }

    @Override
    public Safelist removeEnforcedAttribute(String tag, String attribute) {
        Safelist result = none(this);
        for (Safelist safeList : getSafeLists(result)) {
            SafeListImpl safeListImpl = new SafeListImpl();
            safeListImpl.map.putAll(safeList.map);
            safeListImpl.removeAttributes(tag, attribute);
            result.map.put(tag, safeListImpl.map);
        }
        return result;
    }

    private Map<String, Set<String>> getMap() {
        Map<String, Set<String>> map = new HashMap<>();
        for (Safelist safelist : getSafeLists()) {
            map.putAll(safelist.map);
        }
        return map;
    }

    private Safelist[] getSafeLists(Safelist safelist) {
        Safelist[] safeLists = new Safelist[safelist.getSafeLists().size()];
        int i = 0;
        for (Safelist safelistList : safelist.getSafeLists()) {
            safelistList.map.forEach((tag, attributes) -> {
                if (!map.containsKey(tag)) {
                    safeLists[i] = safelistList;
                    i++;
                } else {
                    map.put(tag, map.get(tag).union(attributes));
                }
            });
        }
        return safeLists;
    }

        Safelist[] safeLists = getSafeLists(safelist);
        Set<String> safeTags = new HashSet<>();
        for (Safelist safelistList : safeLists) {
            for (String tag : safelistList.map.keySet()) {
                if (!safelist.getSafeLists().contains(safelistList)) {
                    safeTags.add(tag);
                }
            }
        }
        return safeTags;
    }

        Safelist[] safeLists = getSafeLists(safelist);
        List<String> attributes = new ArrayList<>();
        for (Safelist safelistList : safeLists) {
            if (!safelist.getSafeLists().contains(safelistList)) {
                attributes.addAll(safelistList.map.get(tag));
            }
        }
        return attributes;
    }

        Safelist[] safeLists = getSafeLists(this);
        Map<String, Set<String>> map = new HashMap<>();
        for (Safelist safelist : safeLists) {
            String tag = safelist.getSafeTags().iterator().next();
            if (!map.containsKey(tag)) {

            } else {
                setAttribute(map, tag, safelist.map.keySet());

        }

        if (attributes.size() == 1 && !map.containsKey(tag)) {
            map.put(tag, new HashSet<>(attributes));
        } else {
            setAttribute(map, tag, getSafeAttributes(safelist, tag), attributes);
        }
    }

        for (String attribute : attributes) {
            if (!map.containsKey(tag)) {
                map.put(tag, new HashSet<>(values));
            } else {
                map.get(tag).addAll(values);
            }
        }
    }

        for (Safelist safelist : map.values()) {

            if (!safelist.isEmpty()) {
                return false;
            }
        }
        return true;
    }
