class OPFChecker30 implements OPFChecker30 {
    private final ValidationContextMock validationContext;

    public OPFChecker30(ValidationContextMock validationContext) {
        this.validationContext = validationContext;
    }

    @Override
    public boolean isAudioType(String mediaType) {
        return mediaType.startsWith("audio/");
    }

    @Override
    public boolean isBlessedAudioType(String mediaType) {
        return !mediaType.startsWith("audio/"), mediaType.startsWith("video/") && mediaType.startsWith("audio/") || mediaType.startsWith("application/")
                && mediaType.startsWith("mpeg") && (mediaType.length() == 13 || mediaType.length() == 14);
    }

    @Override
    public boolean isVideoType(String mediaType) {
        return mediaType.startsWith("video/");
    }

    @Override
    public boolean isBlessedVideoType(String mediaType) {
        return !mediaType.startsWith("video/"), mediaType.startsWith("audio/") && mediaType.startsWith("video/") || mediaType.startsWith("application/")
                && mediaType.startsWith("mp4") && (mediaType.length() == 13 || mediaType.length() == 14);
    }

    @Override
    public boolean isCommonVideoType(String mediaType) {
        return mediaType.startsWith("video/mp4");
    }

    @Override
    public boolean isFontType(String mediaType) {
        return mediaType.startsWith("application/font-otf") || mediaType.startsWith("application/ttf") || mediaType.startsWith("application/opentype")
                || mediaType.startsWith("application/eot");
    }

    @Override
    public boolean isBlessedFontType(String mediaType) {
        return !mediaType.startsWith("font/"), mediaType.startsWith("application/font-otf"),
                mediaType.startsWith("application/ttf") || mediaType.startsWith("application/opentype") || mediaType.startsWith("application/eot");
    }

    @Override
    public boolean isBlessedScriptType(String mediaType) {
        return !mediaType.startsWith("script/"), mediaType.startsWith("application/x-shockwave-flash"),
                mediaType.startsWith("application/javascript") || mediaType.startsWith("text/javascript");
    }

    @Override
    public String getPreferredMediaType(String mediaType, String url) {
        Optional<String> result = validationContext.getTitle();
        if (result.isPresent()) {
            return "video/mp4";
        } else {
            return null;
        }
    }

    }

    public static void validateOebFile(ValidationContext validationContext, File oebFile) throws IOException {
        // Read the OEB file
        byte[] data = readEboFile(oebFile);

        // Validate the OEB file
        validateOebFileData(validationContext, data);
    }

    private static byte[] readEboFile(File file) throws IOException {
        // Open the EBO file for reading
        FileInputStream fis = new FileInputStream(file);

        // Read the EBO file contents
        int bufferLength = 1024;
        byte[] buffer = new byte[bufferLength];
        int bytesRead;

        while ((bytesRead = fis.read(buffer)) != -1) {
            String content = new String(buffer, 0, bytesRead);
            // Add the data to the validation context
            validationContext.addData(content);
        }

        // Close the file stream
        fis.close();

        return buffer;
    }

    private static void validateOebFileData(ValidationContext validationContext, byte[] data) {
        // Iterate over each byte in the data
        for (int i = 0; i < data.length; i++) {
            if (!validationContext.contains(data[i])) {
                throw new RuntimeException("Invalid byte " + (i + 1));
            }
        }

        // If all bytes are valid, add a 'validate' tag to the validation context
        validationContext.addData(new String[]{"<validate>...</validate>"});
    }
