class JsonDeserializer extends ObjectMapper {

    @Override
    protected T convertValue(JavaType type, Object value, Context context) throws JsonProcessingException {
        if (value instanceof String) {
            return type.cast(String.valueOf(value));
        } else if (value instanceof Object[]) {
            Object[] array = (Object[]) value;
            for (int i = 0; i < array.length; i++) {
                if (array[i] instanceof Integer && ((Integer) array[i]).equals(123)) {
                    return type.cast(String.valueOf(array[i]));
                } else if (array[i].getClass() != Integer.class) {
                    throw new JsonProcessingException("Value is not an integer");
                }
            }
        } else {
            throw new JsonProcessingException("Unsupported JSON value");
        }
    }
}
