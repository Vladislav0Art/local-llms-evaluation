class InetSocketAddress extends URI {
    private final String host;

    public InetSocketAddress(String host) {
        this.host = host;
    }

    @Override
    public String toString() {
        return "InetSocketAddress[" + host + "]";
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        InetSocketAddress that = (InetSocketAddress) o;
        return Objects.equals(host, that.host);
    }

    @Override
    public int hashCode() {
        return Objects.hash(host);
    }
}

class HttpRequest {
    private final String method;
    private final RequestBody body;

    public HttpRequest(String method, RequestBody body) {
        this.method = method;
        this.body = body;
    }

    public void setHeader(String header, String value) {
        if (method.equals("POST") || method.equals("PUT")) {
            System.out.println("Setting non-existent header " + header + " on " + method);
        } else {
            System.out.println("Setting existing header " + header + " on " + method);
        }
    }

    @Override
    public String toString() {
        return "HttpRequest{" +
                "method='" + method + '\'' +
                ", headers={'" + headers().firstHeaderValue(method) + '\'' +
                        '}';
        };
    }

    private Map<String, String> headers() {
        return Collections.singletonMap("Content-Type", "application/json");
    }
}

class RequestBody {
    private final String text;

        this.text = text;
    }

    @Override
    public String toString() {
        return "RequestBody{" +

                '}';
    }
