    private Channel channel;
    private SocketAddress remoteAddress;
    private Position position;
    private ByteBuf buf;
    private boolean hasLength;
    private boolean hasSatellites;
    private boolean hasSpeed;
    private TimeZone timezone;

        decoder = new Gt06ProtocolDecoder(protocol);
        channel = new Channel();
        remoteAddress = new SocketAddress("127.0.0.1", 1234);

        hasSpeed = false;

    }

    }

    private void verifyGt06ProtocolDecoder() {
        assertTrue(hasLength);
        assertTrue(hasSatellites);
        assertTrue(hasSpeed);
    }

    public void testDecodeGpsBuffer() throws Exception {
        decoder.decode(channel, remoteAddress, buf);
        verifyGt06ProtocolDecoder();
    }

    private void verifyGt06ProtocolDecoder() {

        assertTrue(buf.readInt8() == 0x23);

        if (hexString.startsWith("0x")) {
            hexString = hexString.substring(2);
        }
        switch(hexString) {
            case "00 01 02 03":
                position = new Position(10, 20);
                break;
            case "04 05 06 07":
