import org.jsoup.helper.DataUtil;
import org.jsoup.helper.UrlBuilder;
import org.junit.jupiter.api.Test;

import java.io.UnsupportedEncodingException;
import java.net.IDN;
import java.net.MalformedURLException;
import java.net.URI;
import java.net.URISyntaxException;
import java.net.URLDecoder;
import java.net.URLEncoder;
import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertThrows;

public class UrlBuilderTest {

    private final UrlBuilder urlBuilder = new UrlBuilder("http://example.com/path?a=1&b=2");

    @Test
    public void testBuild() {
        URL result = urlBuilder.build();
        assertEquals("http://example.com/path?a=1&b=2", result.toString());
    }

    @Test
    public void testAppendKeyVal() throws UnsupportedEncodingException {
        UrlBuilder builder = new UrlBuilder("http://example.com/path");
        builder.appendKeyVal(new UrlKeyValues("key1", "value1"));
        URL result = urlBuilder.build();
        assertEquals("http://example.com/path?key1=value1", result.toString());
    }

    @Test
    public void testAppendKeyValNoValue() {
        try {
            UrlBuilder builder = new UrlBuilder("http://example.com/path");
            builder.appendKeyVal(new UrlKeyValues("", "value"));
            assertThrows(UnsupportedEncodingException.class, () -> urlBuilder.build());
        } catch (UnsupportedEncodingException e) {
            // expected
        }
    }

    @Test
    public void testAppendKeyValEmpty() {
        try {
            UrlBuilder builder = new UrlBuilder("http://example.com/path");
            builder.appendKeyVal(new UrlKeyValues("", ""));
            assertThrows(UnsupportedEncodingException.class, () -> urlBuilder.build());
        } catch (UnsupportedEncodingException e) {
            // expected
        }
    }

    @Test
    public void testAppendKeyValNull() throws UnsupportedEncodingException {
        try {
            UrlBuilder builder = new UrlBuilder("http://example.com/path");
            builder.appendKeyVal(null);
            assertThrows(UnsupportedEncodingException.class, () -> urlBuilder.build());
        } catch (UnsupportedEncodingException e) {
            // expected
        }
    }

    @Test
    public void testAppendKeyValMalformedKey() throws UnsupportedEncodingException {
        UrlBuilder builder = new UrlBuilder("http://example.com/path");
        builder.appendKeyVal(new UrlKeyValues(null, "value"));
        assertThrows(MalformedURLException.class, () -> urlBuilder.build());
    }
}

import org.jsoup.helper.DataUtil;
import org.jsoup.helper.UrlBuilder;
import org.junit.jupiter.api.Test;

import java.io.UnsupportedEncodingException;
import java.net.IDN;
import java.net.MalformedURLException;
import java.net.URI;
import java.net.URISyntaxException;
import java.net.URLDecoder;
import java.net.URLEncoder;
import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertThrows;

public class UrlBuilderTest {

    private final UrlBuilder urlBuilder = new UrlBuilder("http://example.com/path?a=1&b=2");

    @Test
    public void testBuild() {
        String result = urlBuilder.build();
        assertEquals("http://example.com/path?a=1&b=2", result);
    }

    @Test
    public void testAppendKeyValues() {
        String key1 = "key1";
        String value1 = "value1";
        String key2 = "key2";
        String value2 = "value2";

        UrlBuilder builder = new UrlBuilder("http://example.com/path");
        builder.appendKeyValues(key1, value1);
        builder.appendKeyValues(key2, value2);

        String result = builder.build();

        assertEquals("http://example.com/path?key1=value1&key2=value2", result);
    }

    @Test
    public void testNoKeyValues() {
        String result = urlBuilder.build();
        assertEquals("http://example.com/path", result);
    }

    @Test
    public void testNullKeyValues() {
        UrlBuilder builder = new UrlBuilder("http://example.com/path");
        assertThrows(NullPointerException.class, () -> builder.appendKeyValues(null, "value"));
    }
}
