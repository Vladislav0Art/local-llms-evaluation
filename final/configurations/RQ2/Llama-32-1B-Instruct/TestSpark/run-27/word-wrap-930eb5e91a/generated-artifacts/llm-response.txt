public class TestWordWrap {

    @Test
    public void testFromUtf8() {
        Builder builder = WordWrap.fromClasspathUtf8("test.txt");
        String text = "This is a test string";
        int maxWidth = 10;
        Function<CharSequence, Number> stringWidth = CharSequence::length;
        Set<Character> extraWordChars = new HashSet<>();
        boolean insertHyphens = false;
        boolean breakWords = true;

        WordWrap.builder(text).setMaxWidth(maxWidth)
                .addExtraChar(0, ' ', 4) // Insert hyphen at the beginning
                .addChar(1, '\n')
                .addExtraChar(2, '\t')
                .addExtraChar(3, '-')
                .withInsertHyphens(insertHyphens).withBreakWords(breakWords)
                .build();

        assert WordWrap.wordWrap(builder.reader(), builder.out(), "\n", maxWidth, stringWidth,
                extraWordChars, insertHyphens, breakWords) : "Error in word wrap";

        assertEquals(10, builder.maxWidth);
        assertTrue(extraWordChars.contains(' ', 0));
    }

    @Test
    public void testFromClasspath() {
        Builder builder = WordWrap.fromClasspathUtf8("test.txt");
        String text = "This is a test string";
        int maxWidth = 20;
        Function<CharSequence, Number> stringWidth = CharSequence::length;
        Set<Character> extraWordChars = new HashSet<>();
        boolean insertHyphens = false;
        boolean breakWords = true;

        WordWrap.builder(text).setMaxWidth(maxWidth)
                .addExtraChar(0, ' ', 4) // Insert hyphen at the beginning
                .addChar(1, '\n')
                .addExtraChar(2, '\t')
                .addExtraChar(3, '-')
                .withInsertHyphens(insertHyphens).withBreakWords(breakWords)
                .build();

        assert WordWrap.wordWrap(builder.reader(), builder.out(), "\n", maxWidth, stringWidth,
                extraWordChars, insertHyphens, breakWords) : "Error in word wrap";

        assertEquals(20, builder.maxWidth);
    }

    @Test
    public void testFromClasspathString() {
        Builder builder = WordWrap.fromClasspath("test.txt");
        String text = "This is a test string";
        int maxWidth = 10;
        Function<CharSequence, Number> stringWidth = CharSequence::length;

        assert WordWrap.wordWrap(builder.reader(), builder.out(), "\n", maxWidth, stringWidth,
                new HashSet<>(), false, true)
                : "Error in word wrap";

    }

    public void testFromClasspathStringEmpty() {

        String text = "";
        int maxWidth = 10;
        Function<CharSequence, Number> stringWidth = CharSequence::length;

        assert WordWrap.wordWrap(builder.reader(), builder.out(), "\n", maxWidth, stringWidth,
                new HashSet<>(), false, true)
                : "Error in word wrap";

    }

        String text = "";

        assert WordWrap.wordWrap(builder.reader(), builder.out(), "\n", maxWidth, stringWidth,
                new HashSet<>(), false, true)
                : "Error in word wrap";

    }

    public void testFromUtf8LeftTrim() {

public class TestWordWrap {

    @Test
    public void testFromUtf8() {
        Builder builder = new WordWrap.Builder("test.txt");
        String text = "This is a test string";
        int maxWidth = 10;
        Function<CharSequence, Number> stringWidth = CharSequence::length;

        builder.maxWidth(maxWidth).addSpaces().setIndent(0);
        wordWrap(builder, "", "\n", maxWidth, stringWidth, true);

        StringBuilder sb = new StringBuilder();
        for (char c : text.toCharArray()) {
            sb.append(c).append(" ");
        }
        String trimmedText = sb.toString().trim();

        assert trimmedText.equals(text) : "Trimmed text does not match original";

        builder.removeSpaces().setIndent(0);
        wordWrap(builder, "", "\n", maxWidth, stringWidth, false);
    }

    @Test
    public void testFromClasspath() {
        Builder builder = new WordWrap.Builder("test.txt");
        String text = "This is a test string";
        int maxWidth = 20;
        Function<CharSequence, Number> stringWidth = CharSequence::length;

        builder.maxWidth(maxWidth).addSpaces().setIndent(0);
        wordWrap(builder, "", "\n", maxWidth, stringWidth, true);

        builder.setLineBreaks(false).setIndent(0);
        wordWrap(builder, "", "\n", 20, stringWidth, true);

        StringBuilder sb = new StringBuilder();
        for (char c : text.toCharArray()) {
            sb.append(c).append(" ");
        }
        String trimmedText = sb.toString().trim();

        assert trimmedText.equals(text) : "Trimmed text does not match original";

        builder.setLineBreaks(true);
        wordWrap(builder, "", "\n", 20, stringWidth, false);

        builder.removeSpaces();
        builder.setIndent(0);
        wordWrap(builder, "", "\n", 20, stringWidth, true);
    }

    @Test
    public void testFromClasspathString() {
        WordWrap.Builder builder = new WordWrap.Builder("test.txt");
        String text = "This is a test string";

        builder.setLineBreaks(false).setIndent(0);

        StringBuilder sb = new StringBuilder();
        for (char c : text.toCharArray()) {
            sb.append(c).append(" ");
        }
        String trimmedText = sb.toString().trim();

        assert trimmedText.equals(text) : "Trimmed text does not match original";

        builder.setLineBreaks(true);
        wordWrap(builder, "", "\n", 10, stringWidth, false);

        builder.removeSpaces();
        builder.setIndent(0);
        wordWrap(builder, "", "\n", 10, stringWidth, true);
    }
