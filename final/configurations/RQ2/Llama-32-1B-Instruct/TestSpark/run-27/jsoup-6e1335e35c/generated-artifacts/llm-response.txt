import org.w3c.dom.Document;
import org.w3c.dom.Element;

public class TestUtils {

    private static Document doc = null;

    public static Element createElement(String tag, String attrib) {
        Element elem = doc.createElement(tag);
        if (attrib != null) {
            for (String attr : attrib.split("=")) {
                elem.setAttribute(attr, "");
            }
        }
        return elem;
    }

    public static void addNodeToString(Node node, StringBuilder strBuilder) {
        while (node.getNodeType() == Node.ELEMENT_NODE) {
            strBuilder.append("<").append(node.getNodeName()).append(">");
            appendChildToString(node, strBuilder);
            strBuilder.append("</").append(node.getNodeName()).append(">");
        }
    }

    private static void appendChildToString(Node node, StringBuilder strBuilder) {
        if (node.hasChildNodes()) {
            for (Node child : node.getChildNodes()) {
                appendChildToString(child, strBuilder);
            }
        } else {
            strBuilder.append(node.getNodeName());
        }
    }

    public static String text(Element element) {
        return element.getTextContent().toString();
    }

    public static boolean hasText(Node node) {
        if (node.getNodeType() == Node.ELEMENT_NODE && !node.hasChildNodes()) {
            for (Node child : node.getChildNodes()) {
                if (!child.hasChildNodes()) {
                    return false;
                }
            }
        }
        return node.getTextContent() != null;
    }

    public static void addClass(Element element, String className) {
        if (element.hasAttribute("class")) {
            if (!element.getAttribute("class").equals(className)) {
                element.setAttribute("class", className);
            }
        } else {
            element.setAttribute("class", className);
        }
    }

    public static Element removeClass(Element element, String className) {
        if (element.hasAttribute("class")) {
            if (element.getAttribute("class").equals(className)) {
                return null;
            } else {
                element.setAttribute("class", element.getAttribute("class").replace(className, ""));
                return element;
            }
        } else {
            return element;
        }
    }

    public static void clearAttributes(Element element) {
        for (Node attr : element.getAttributes()) {
            if (!attr.getNodeType() || !attr.getNodeName().equals("href") && !attr.getNodeName().equals("title")) {
                element.removeAttribute(attr.getNodeName());
            } else {
                element.setAttribute(attr.getNodeName(), "");
            }
        }
    }

    public static Element cloneElement(Element element) {
        return doc.importNode(element, true);
    }

    public static Element root() {
        if (doc == null) {
            try {
                doc = createDocument();
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        }
        return doc.getDocumentElement();
    }

    private static Document createDocument() throws Exception {
        // Create a new document
        doc = new Document("root");
        return doc;
    }
}
