        @Override
        public void execute(ProcedureParameters parameters) throws Exception {
            int numRows = parameters.getInteger("numRows");
            for (int i = 0; i < numRows; i++) {
                Object[] rowValues = parameters gettingString("rowValues").getArray();
                if (rowValues == null || rowValues.length != data.getColumnNames().size()) {
                    throw new Exception("Invalid number of values for column " + data.getColumnNames().get(i));
                }
                for (int j = 0; j < rowValues.length; j++) {
                    Object value = rowValues[j];
                    if (!(value instanceof String)) {
                        throw new Exception("All values in column '" + data.getColumnNames().get(j) + "' must be strings");
                    }
                    if (data.isString()) {
                        System.out.println(value.toString());
                    } else {
                        for (int k = 0; k < data.getValues(data.getColumnNames().get(j)).size(); k++) {
                            System.out.print(data.getValues(data.getColumnNames().get(j)).get(k) + ", ");
                        }
                        System.out.println();
