```java
import java.net.URI;
import java.net.URISyntaxException;

public class Main {
    public static void main(String[] args) throws URISyntaxException {
        // Print the original URL
        String url = "https://example.com/path?query=param";
        URI uri = new URI(url);
        System.out.println("Original URL: " + uri.toString());

        // Parse the original URL to extract query parameters
        String[] queryParams = parseQueryParams(url);
        if (queryParams != null) {
            for (String param : queryParams) {
                System.out.println("Parameter: " + param);
            }
        } else {
            System.out.println("No query parameters found.");
        }

        // Print the original URL without query parameters
        url = "https://example.com/path";
        URI uriWithoutQueryParams = new URI(url);
        System.out.println("Public URL (without query parameters): " + uriWithoutQueryParams.toString());

        // Parse the original URL to extract query parameters
        queryParams = parseQueryParams(url);
        if (queryParams != null) {
            for (String param : queryParams) {
                System.out.println("Parameter: " + param);
            }
        } else {
            System.out.println("No query parameters found.");
        }

        // Extract the public URL by replacing the original URL with its query parameter version
        String publicUrl = generatePublicUrl(url);
        System.out.println("Public URL: " + publicUrl);

        // Parse the original URL to extract query parameters
        queryParams = parseQueryParams(url);
        if (queryParams != null) {
            for (String param : queryParams) {
                System.out.println("Parameter: " + param);
            }
        } else {
            System.out.println("No query parameters found.");
        }
    }

    private static String generatePublicUrl(String url) throws URISyntaxException {
        URI uri = new URI(url);
        return uri.getScheme() + "://" + uri.getHost() + uri.getPath();
    }

    private static String[] parseQueryParams(String url) {
        try {
            URI uri = new URI(url);
            String[] queryParams = uri.getQuery().split("&");
            StringBuilder sb = new StringBuilder();
            for (String param : queryParams) {
                if (!sb.toString().isEmpty()) {
                    sb.append("&");
                }
                sb.append(param);
            }
            return sb.toString().trim().split("=");
        } catch (URISyntaxException e) {
            return null;
        }
    }
}
```
