class TestSparkImpl extends EpubChecker {
    @Override
    public String GeneralFileUrl() {
        return super.GeneralFileUrl();
    }

    @Override
    public String ProgressiveFileUrl() {
        return super.ProgressiveFileUrl();
    }
}

class EpubCheckerImpl extends EpubChecker {
    private boolean testCases = false;

    public void runTest(String scenario, String[] arguments) throws Exception {
        if (testCases) {
            System.out.println("Running test case: " + scenario);
        } else {
            super.run(args);
        }
    }

    public static void testCase(EpubChecker checker) {
        String[] args = new String[]{checker.main.getClass().getName()};
        
        checkers[checker.main.getClass().getName()] = checker;
        
        System.out.println("Running test case: " + checker.main.getClass().getSimpleName());
        EpubChecker.run(args);
    }

    public static void runSparkTest(String[] arguments) throws Exception {
        for (String className : new String[]{"test1", "test2"}) {
            EpubChecker checker = checkers[className];
            
            System.out.println("Running test case: " + className);
            
            if (checker.run(args)) {
                break;
            }
        }
    }

    public static void testCases() throws Exception {
        TestSpark spark = new TestSpark();
        
        for (String scenario : new String[]{"GeneralFileUrl", "ProgressiveFileUrl"}) {
            EpubChecker checker = spark.testSpark("test1", arguments);
            
            if (!checker.run(args)) {
                break;
            }
        }
    }

    public static void main(String[] args) throws Exception {
        testCases();
    }
}
