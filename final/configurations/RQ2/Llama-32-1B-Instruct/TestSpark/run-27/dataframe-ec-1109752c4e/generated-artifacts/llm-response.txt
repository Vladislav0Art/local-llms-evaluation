public class PrettyPrintVisitor {
    public static String visit(StatementSequenceScript s) {
        StringBuilder sb = new StringBuilder();
        for (VarExpr var : s.getVars()) {
            sb.append("print(\"").append(var.getName()).append("\")\n");
        }
        return sb.toString();
    }

    public static String visit(FunctionScript f) {
        StringBuilder sb = new StringBuilder();
        sb.append(f.getExpression().getName());
        sb.append("(");
        for (VarExpr var : f.getArguments()) {
            sb.append("").append(var.getName()).append("\n");
        }
        sb.append(")");
        return sb.toString();
    }

    public static String visit(AssignExpr a) {
        return "print(\"=").append(a.getExpression().getName()).append("=\n").append(visit(a.getLeftSide())).append(")\n";
    }

    public static String visit(BinaryExpr b) {
        return "print(\"<\", \"=", ").append(visit(b.getLeftSide())).append(", \".append(visit(b.getRightSide())).append("\")\n");
    }

    public static String visit(UnaryExpr u) {
        if (u.getExpression() instanceof VarExpr v) {
            return "print(\"-", ", \".append(v.getName()).append(")\", \".append(visit(u.getLeftSide())).append("\")\n)";
        } else if (u.getExpression() instanceof ConstExpr c) {
            return "print(\"+", \"=", \".append(c.getValue()).append(", \".append(visit(u.getRightSide())).append("\")\n";
        } else {
            throw new UnsupportedOperationException("Unsupported expression type");
        }
    }

    public static String visit(AliasExpr a) {
        return "print(\"[\", \".append('" + a.getName() + ")\", \".append('a').append('" + a.getName() + ")\"]\n";
    }

    public static String visit(VarExpr v) {
        return "print(\"x\")";
    }

    public static String visit(ProjectorExpr p) {
        StringBuilder sb = new StringBuilder();
        for (VarExpr var : p.getVars()) {
            sb.append(" ").append(var.getName()).append(" ");
        }
        return sb.toString().trim() + "\n";
    }

    public static String visit(AssignmentExpr a) {
        return "print(\"=", \".append('" + a.getLeftSide().getName() + "') + \" = \", \".append(visit(a.getRightSide())).append("\")\n";
    }
}

public interface VarExpr {
    String getName();
}

public class Value implements VarExpr {
    private final Object value;

    public Value(Object value) {
        this.value = value;
    }

    @Override
    public String getName() {
        return value.toString();
    }
}

public class Expression {
    public abstract static class E {
        public abstract void accept(PrettyPrintVisitor visitor);
    }

    public static abstract class VarExpr extends E {
        private final Object name;

        public VarExpr(Object name) {
            this.name = name;
        }

        @Override
        public String getName() {
            return name.toString();
        }
    }

    public static abstract class BinaryExpr extends E {
        private final Expression left;
        private final Expression right;

        public BinaryExpr(Expression left, String operator) {
            this.left = left;
            this.operator = operator;
            this.right = null;
        }

        @Override
        public void accept(PrettyPrintVisitor visitor) {
            if (left != null && right != null) {
                visitor.visit(left);
                visitor.visit(right);
            } else {
                visitor.visit(null);
            }
        }

        public abstract String getName();
    }

    public static abstract class UnaryExpr extends E {
        private final Expression expression;

        public UnaryExpr(Expression expression) {
            this.expression = expression;
            this.operator = null;
        }

        @Override
        public String getName() {
            return expression.getName();
        }

        public void accept(PrettyPrintVisitor visitor) {
            if (expression != null) {
                visitor.visit(expression);
            } else {
                visitor.visit(null);
            }
        }

        public abstract String operator;
    }

    public static abstract class ConstExpr extends E {
        private final Object value;

        public ConstExpr(Object value) {
            this.value = value;
        }

        @Override
        public String getName() {
            return value.toString();
        }

        @Override
        public void accept(PrettyPrintVisitor visitor) {
            if (value instanceof Integer || value instanceof Float || value instanceof Boolean) {
                visitor.visit(value);
            } else {
                throw new UnsupportedOperationException("Unsupported expression type");
            }
        }
    }

    public static abstract class FunctionExpr extends E {
        private final Expression[] arguments;
        private final String name;

        public FunctionExpr(String name, Expression[] args) {
            this.name = name;
            this.arguments = args;
        }

        @Override
        public void accept(PrettyPrintVisitor visitor) {
            for (int i = 0; i < arguments.length; i++) {
                visitor.visit(arguments[i]);
            }
        }

        public abstract String getName();
    }

    public static abstract class IfElseExpr extends E {
        private final Expression then;
        private final Expression else_;
        private final String condition;

        public IfElseExpr(String condition, Expression then, Expression else_) {
            this.condition = condition;
            this.then = then;
            this.else_ = else_;
        }

        @Override
        public void accept(PrettyPrintVisitor visitor) {
            if (condition.equals("true")) {
                visitor.visit(then);
                visitor.visit(Else_.getName());
            } else {
                visitor.visit(else_);
            }
        }

        public abstract String condition;
    }

    public static class IfElseExpr_ extends E {
        private final Expression then;
        private final Expression else_;

        public IfElseExpr_(String condition, Expression then, Expression else_) {
            this.condition = condition;
            this.then = then;
            this.else_ = else_;
        }

        @Override
        public void accept(PrettyPrintVisitor visitor) {
            if (condition.equals("true")) {
                visitor.visit(then);
                visitor.visit(Else_.getName());
            } else {
                visitor.visit(else_);
            }
        }
    }
}
