import java.util.ArrayList;
import java.util.HashMap;
import java.util.Map;

public class UserLocationDecoder {
    public static UserLocation decodeUserLocation(User user, String locationString) {
        Map<String, Position> positionMap = new HashMap<>();
        positionMap.put("East", decodePosition(locationString));
        positionMap.put("South", decodePosition(locationString));
        positionMap.put("North", decodePosition(locationString));

        int userId = Integer.parseInt(user.getUserId());
        Direction direction;

        if (positionMap.containsKey(String.valueOf(user.getDirection().getType()))) {
            direction = positionMap.get(String.valueOf(user.getDirection().getType()));
        } else {
            for (int i = 0; i < user.getPositions().size(); i++) {
                Position pos = user.getPositions().get(i);
                String type;
                switch (pos.getType()) {
                    case "North":
                        type = "N";
                        break;
                    case "South":
                        type = "S";
                        break;
                    case "East":
                        type = "E";
                        break;
                    default:
                        type = null;
                        break;
                }

                if (type != null) {
                    direction = new Direction(Integer.parseInt(String.valueOf(i + 1)), type, Double.parseDouble(pos.getLatitude()), Double.parseDouble(pos.getLongitude()));
                    positionMap.put(String.valueOf(user.getDirection().getType()), direction);
                    break;
                }
            }
        }

        return new UserLocation(userId, direction);
    }

    public static Position decodePosition(String positionString) {
        String[] values = positionString.split(",");
        int id = Integer.parseInt(values[0].trim());
        String type = values[1].trim();
        double latitude = Double.parseDouble(values[2].trim());
