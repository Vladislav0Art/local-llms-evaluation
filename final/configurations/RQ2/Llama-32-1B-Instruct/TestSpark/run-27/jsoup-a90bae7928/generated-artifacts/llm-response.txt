```java
import javax.xml.parsers.DocumentBuilderFactory;
import javax.xml.parsers.ParserConfigurationException;

import org.w3c.dom.Document;
import org.w3c.dom.Element;
import org.w3c.dom.NodeList;
import org.junit.Test;

public class ElementParser {

    public static String parseFragment(String name, String url) {
        try {
            Document doc = DocumentBuilderFactory.newInstance().newDocumentBuilder()
                    .parse("data.xml");
            NodeList fragments = doc.getElementsByTagName(name);
            if (fragments.getLength() > 0) {
                return fragments.item(0).getTextContent();
            } else {
                return "";
            }
        } catch (ParserConfigurationException | NullPointerException e) {
            return "";
        }
    }

    public static boolean hasStartTag(Element element) {
        return element != null && element.getNodeType() == Element.START_TAG;
    }

    @Test
    public void testParseFragment() {
        System.out.println(parseFragment("tag1", "http://example.com"));
        System.out.println(parseFragment("tag2", "http://example.com"));
    }
}
```

This code defines a `parseFragment` method that parses an XML document and returns the content of a specified element. The `hasStartTag` method checks if an element is the start tag. You can create your own XML file (`data.xml`) with elements like this:

```xml
<example xmlns="http://example.com">
    <tag1 href="http://example.com/1">Example 1</tag1>
    <tag2 href="http://example.com/2">Example 2</tag2>
</example>
```

Make sure to replace `"http://example.com"` with the base URL if it's different.
