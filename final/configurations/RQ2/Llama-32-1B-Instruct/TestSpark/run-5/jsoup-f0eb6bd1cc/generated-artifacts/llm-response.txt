```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.select.Elements;

public class JSoupHelper {

    public static String buildUrl(String baseUrl, String path) {
        Document document = Jsoup.connect(baseUrl).get();
        if (path != null && !path.isEmpty()) {
            Elements elements = document.select(path);
            StringBuilder urlBuilder = new StringBuilder(baseUrl);
            for (Element element : elements) {
                urlBuilder.append(element.attr("href")).append("/");
            }
            return urlBuilder.toString().substring(0, urlBuilder.length() - 1);
        } else {
            return baseUrl;
        }
    }

    public static void testAppendKeyVal_JoinEmptyStringToKeyAndValue() throws UnsupportedEncodingException {
        String baseUrl = "https://example.com";
        String path = "/api/data?key=123&value=456";
        String expectedUrl = buildUrl(baseUrl, path);
        System.out.println("Expected URL: " + expectedUrl);
        assertEquals(expectedUrl, new java.net.URL(baseUrl).toString());
    }

    public static void testAppendKeyVal_JoinEmptyStringToKeyAndValue_WithQuery() throws UnsupportedEncodingException {
        String baseUrl = "https://example.com";
        String path = "/api/data?key=123&value=456&query=abcdefg";
        String expectedUrl = buildUrl(baseUrl, path);
        System.out.println("Expected URL: " + expectedUrl);
        assertEquals(expectedUrl, new java.net.URL(baseUrl).toString());
    }

    public static void testAppendKeyVal_JoinEmptyStringToKeyAndValue_WithNullPath() throws UnsupportedEncodingException {
        String baseUrl = "https://example.com";
        String path = null;
        String expectedUrl = buildUrl(baseUrl, path);
        System.out.println("Expected URL: " + expectedUrl);

    public static void testAppendKeyVal_JoinEmptyStringToKeyAndValue_WithNullPathAndQuery() throws UnsupportedEncodingException {

        String expectedUrl = buildUrl(baseUrl, path);
        System.out.println("Expected URL: " + expectedUrl);
        assertEquals(expectedUrl, new java.net.URL(baseUrl).toString());
    }

    public static void main(String[] args) throws UnsupportedEncodingException {

        testAppendKeyVal_JoinEmptyStringToKeyAndValue_WithNullPathAndQuery();
    }
}
```

This code will parse the URL and query string based on the provided input. It first connects to the base URL, then selects all the elements at the specified path (if present), appends their href attributes to the beginning of the base URL, and finally returns the resulting URL. The test cases are used to verify the correctness of this function.

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.select.Elements;

public class JSoupHelper {

    public static String buildUrl(String baseUrl, String path) {
        Document document = Jsoup.connect(baseUrl).get();
        if (path != null && !path.isEmpty()) {
            Elements elements = document.select(path);
            StringBuilder urlBuilder = new StringBuilder(baseUrl);
            for (Element element : elements) {
                urlBuilder.append(element.attr("href")).append("/");
            }
            return urlBuilder.toString().substring(0, urlBuilder.length() - 1);
        } else if (!baseUrl.isEmpty()) {
            return baseUrl;
        } else {
            return null;
        }
    }

    public static void testAppendKeyVal_JoinEmptyStringToKeyAndValue() throws UnsupportedEncodingException {
        String baseUrl = "https://example.com";
        String path = "/api/data?key=123&value=456";
        System.out.println("Expected URL: " + buildUrl(baseUrl, path));
        assertEquals(buildUrl(baseUrl, path), new java.net.URL(baseUrl).toString());
    }

    public static void testAppendKeyVal_JoinEmptyStringToKeyAndValue_WithQuery() throws UnsupportedEncodingException {
        String baseUrl = "https://example.com";
        String path = "/api/data?key=123&value=456&query=abcdefg";
        System.out.println("Expected URL: " + buildUrl(baseUrl, path));
        assertEquals(buildUrl(baseUrl, path), new java.net.URL(baseUrl).toString());
    }

    public static void testAppendKeyVal_JoinEmptyStringToKeyAndValue_WithNullPath() throws UnsupportedEncodingException {
