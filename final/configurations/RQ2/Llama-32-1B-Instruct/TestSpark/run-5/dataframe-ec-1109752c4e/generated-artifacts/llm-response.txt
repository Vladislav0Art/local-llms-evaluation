public class PrettyPrintVisitor {

    public static String exprToString(Expression e) {
        CollectingPrinter printer = PrinterFactory.create();
        if (e instanceof AliasExpr aliasExpr) {
            printer.print("alias ");
            printer.print(aliasExpr.getName());
            printer.print(": ");
            printer.print(e.toString());
        } else if (e instanceof VarExpr varExpr) {
            printer.print(varExpr.getName());
            printer.print(": ");
            printer.print(e.toString());
        } else if (e instanceof Value value) {
            printer.print("value ");
            printer.print(value);
            printer.println();
        } else if (e instanceof UnaryExpr unaryExpr) {
            printer.print(unaryExpr.getExpression().toString());
        } else if (e instanceof BinaryExpr binaryExpr) {
            printer.print("(");
            printer.print(binaryExpr.toString());
            printer.println(")");
        }
    }

public class DataFrame {
    private Object[][] data;

    public DataFrame() {
        this.data = new Object[0][];
    }

    public void addColumn(String name, Class<?> columnType) {
        Object[] newRow = new Object[data.length + 1];
        System.arraycopy(data, 0, newRow, 0, data.length);
        newRow[newRow.length - 1] = columnType.newInstance();
        data = newRow;
    }

    public Dataset<Row> filter(Dataset<Row> conditions) {
        Dataset<Row> result = new Dataset<>();
        for (int i = 0; i < data.length; i++) {
            boolean matchesAllConditions = true;
            for (Dataset<Row>.Column column : conditions.columns()) {
                if (!conditions.col(column).evaluate(data[i], column)) {
                    matchesAllConditions = false;
                    break;
                }
            }
            if (matchesAllConditions) {
                result.addColumn(column, data[i]);
            }
        }
        return result;
    }

    public Dataset<Row> select(Dataset<Row>.Column... columns) {
        Dataset<Row> result = new Dataset<>();
        for (Object[] row : data) {
            boolean matchesAllConditions = true;
            for (Dataset<Row>.Column column : columns) {
                if (!row[column].evaluate(null, column)) {
                    matchesAllConditions = false;
                    break;
                }
            }
            if (matchesAllConditions) {
                result.addColumn(columns[0], row);
            }
        }
        return result;
    }
