package ch.jalu.configme.configurationdata;

import ch.jalu.configme.exception.ConfigMeException;
import ch.jalu.configme.properties.Property;
import org.jetbrains.annotations.NotNull;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

public class ConfigurationManager {
    private final Map<String, Property> rootEntries;
    private List<Property> defaultChildren;

    public ConfigurationManager() {
        this.rootEntries = Map.of("root", new Property());
        this.defaultChildren = List.of(new Property());
    }

    public void addRootEntry(String name, @NotNull Property property) {
        rootEntries.put(name, property);
    }

    public void addDefaultChild(String name, @NotNull Property property) {
        defaultChildren.add(property);
    }

    public void setDefaultChildren(List<Property> children) {
        this.defaultChildren = children;
    }

    public Map<String, Object> getRootEntries() {
        return rootEntries;
    }

    public List<Property> getDefaultChildren() {
        return defaultChildren;
    }

    public @NotNull Property getProperty(String name) {
        if (rootEntries.containsKey(name)) {
            return rootEntries.get(name);
        } else if (defaultChildren.containsAll(getRootEntries())) {
            return getRootEntries().stream().map(Property::new).first();
        }
        throw new ConfigMeException("Property not found");
    }

    public @NotNull List<Property> getProperty(String name) {
        if (rootEntries.containsKey(name)) {
            return rootEntries.get(name);
        } else if (!defaultChildren.containsAll(getRootEntries())) {
            throw new ConfigMeException("Property not found");
        }
        return List.of();
    }

    public void setDefaultEntries(List<Property> children) {
        this.defaultChildren = children;
    }
}

package ch.jalu.configme.configurationdata;

import ch.jalu.configme.exception.ConfigMeException;
import org.jetbrains.annotations.NotNull;

public class ConfigMeProperties {
    private final Map<String, Object> allEntries;
    private @NotNull List<Property> defaultChildren;

    public ConfigMeProperties() {
        this.allEntries = Map.of();
        this.defaultChildren = null;
    }

    public void addEntry(String name, @NotNull Property property) {
        if (defaultChildren != null && ((List<Property>) allEntries.get(name)).containsAll(defaultChildren)) {
            throw new ConfigMeException("Default children are defined");
        }
        allEntries.put(name, property);
    }

    public void setDefaultChildren(@NotNull List<Property> children) {
        defaultChildren = children;
    }

    public @NotNull Property getProperty(String name) {
        if (defaultChildren != null && ((List<Property>) allEntries.get(name)).containsAll(defaultChildren)) {
            return (Property) allEntries.get(name);
        } else {
            throw new ConfigMeException("Property not found");
        }
    }

    public @NotNull List<Property> getProperty(String name) {
        if (defaultChildren != null && ((List<Property>) allEntries.get(name)).containsAll(defaultChildren)) {
            return defaultChildren;
        } else {
            throw new ConfigMeException("Property not found");
        }
    }

    public void setDefaultEntries(@NotNull List<Property> children) {
        this.defaultChildren = children;
    }
}

class Property {
    private @NotNull String name;
    private @NotNull Object value;

    public Property(String name, @NotNull Object value) {
        this.name = name;
        this.value = value;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        Property property = (Property) o;
        return Objects.equals(name, property.name) && Objects.equals(value, property.value);
    }

    @Override
    public int hashCode() {
        return Objects.hash(name, value);
    }
}

public class Main {

    public static void main(String[] args) throws ConfigMeException {
        ConfigMeProperties properties = new ConfigMeProperties();

        properties.addEntry("root", new Property("root", "value"));
        System.out.println(properties.getProperty("root")); // prints: root
        System.out.println(properties.getProperty("root").equals(new Property("root", "value"))); // true

        properties.setDefaultChildren(List.of(new Property("child1", "value")));
        System.out.println(properties.getProperty("root")); // prints: value
        System.out.println(properties.getDefaultChildren().stream().map(Property::getName).collect(Collectors.toList())); // [child1]
    }
}
