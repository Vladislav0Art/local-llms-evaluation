import org.apache.spark.sql.Dataset;
import org.apache.spark.sql.Row;

public class Main {

    public static void main(String[] args) throws DBAppException {
        // Create a new DataFrame (table)
        Dataset<Row> df = createTable("test_table", new String[]{"col1", "col2"});

        // Insert data into the table
        df.insertInto("test_table", new String[]{"value1", "value2"});
        System.out.println(df);

        // Update existing data in the table
        df.update("test_table", new String[]{"new_value1", "new_value2"});
        System.out.println(df);

        // Delete rows from the table
        df.delete("test_table");
        System.out.println(df);
    }

    public static Dataset<Row> createTable(String tableName, String[] columns) throws DBAppException {
        // Return a new DataFrame (table)
        return dbApp.getMyTables().create(tableName);
    }
