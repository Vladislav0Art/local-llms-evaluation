import java.util.Arrays;

public abstract class AbstractWordWrapTest {

    protected String word;
    protected int maxWords;

    @Before
    public void setup() {
        word = "test";
        maxWords = 5;
    }

    @Test
    public void testWordWrap_EmptyString() {
        Preconditions.checkNotNull(word);
        Preconditions.checkNotNull(maxWords);

        WordWrap wordWrap = new WordWrap();
        assertEquals(0, wordWrap.word(word, maxWords).length());
    }

    @Test
    public void testWordWrap_MoreThanOneChar() {
        Preconditions.checkNotNull(word);
        Preconditions.checkNotNull(maxWords);

        WordWrap wordWrap = new WordWrap();
        assertEquals(0, wordWrap.word("abc", maxWords).length());
        assertEquals(1, wordWrap.word("abc", 3).length());
    }

    @Test
    public void testWordWrap_MoreThanMaxWords() {
        Preconditions.checkNotNull(word);
        Preconditions.checkNotNull(maxWords);

        WordWrap wordWrap = new WordWrap();
        assertEquals(0, wordWrap.word(word, maxWords).length());
    }

    protected abstract String stringWidth();

    protected String word(String s) {
        return s;
    }

    public int countCharsBeforeWord(String word, int charsPerLine) {
        int words = Arrays.stream(word.split("\\s+"))
                .mapToObj(String::length)
                .max()
                .orElse(0);
        int charCount = 0;
        for (int i = 0; i < Math.min(words, charsPerLine); i++) {
            charCount += word.charAt(i) + " ";
        }
        return charCount;
    }

    public String wordWrap(String inputString, int maxWords) {
        Preconditions.checkNotNull(inputString);
        Preconditions.checkNotNull(maxWords);

        StringBuilder output = new StringBuilder();
        for (int i = 0; i < inputString.length(); i++) {
            if (output.length() == 0 || output.length() + countCharsBeforeWord(inputString.substring(i), maxWords) > maxWords) {
                output.append(word).append(" ");
                output.append(String.join("", Arrays.stream(word.split("\\s+"))
                        .mapToObj(String::length)
                        .filter(j -> i < j)
                        .toArray(String[]::new))).trim());
            } else {
                output.append(inputString.charAt(i));
            }
        }

        return output.toString().trim();
    }
}
