import org.apache.http.HttpEntity;
import org.apache.http.HttpResponse;

import org.apache.http.client.methods.HttpGet;

import org.apache.http.impl.client.HttpClients;

public class Main {

    public static void main(String[] args) {
        String url = "https://example.com";
        String body = "{\"key\":\"value\"}";
        
        System.out.println("Making POST request to " + url);
        makePostRequest(url, body);
        System.out.println("Making PUT request to " + url);

        // Add the same code to post and put methods
    }

    public static void makePostRequest(String url, String body) {
        try (CloseableHttpClient client = HttpClients.createDefault();
             HttpResponse response = client.execute(new HttpGet(url), null)) {

            System.out.println("Response from " + url + ": " + response.getStatusLine().getStatusCode());

            // Process the response
            HttpEntity entity = response.getEntity();
            String responseBody = new String(entity.getContent());
            System.out.println("Body of response: " + responseBody);

        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    public static void makePutRequest(String url, String body) {
        try (CloseableHttpClient client = HttpClients.createDefault();
             HttpResponse response = client.execute(new HttpPost(url), null)) {

            // Process the response
