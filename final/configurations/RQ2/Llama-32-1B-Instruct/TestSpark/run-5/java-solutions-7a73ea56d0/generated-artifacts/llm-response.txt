public class StockPrices {

    public int maxProfit(int[] prices) {
        if (prices.length < 2) {
            return 0;
        }

        int buy = -1000;
        int sell = 0;

        for (int price : prices) {
            int currBuy = buy;
            int currSell = sell;

            while (price > currBuy) {
                currBuy = buy;
            }
            while (price < currSell) {
                currSell = sell;
            }

            sell += Math.max(0, price - currBuy);
            buy = currBuy;
        }

        return sell;
    }

    public int maxProfit(int[] prices) {
        if (prices.length <= 2) {
            return 0;
        }

        int[] dp = new int[prices.length];
        for (int i = 1; i < prices.length; i++) {
            dp[i] = Math.max(dp[i - 1], prices[i] - prices[i - 1]);
        }
        return Math.max(dp[dp.length - 1], 0);
    }

    public int maxProfit(int[] prices) {
        if (prices.length <= 3) {
            return 0;
        }
        int buy = -1000, sell = 0;
        for (int price : prices) {
            buy = Math.max(buy, sell - price);
            sell = Math.max(sell, buy + price);
        }
        return sell;

            return 0;
        }

        int[] hold1 = new int[prices.length];
        int[] hold2 = new int[prices.length];

        for (int i = 1; i < prices.length; i++) {
            if (i > 1 && prices[i] - prices[i-1] > prices[i] - prices[i-2]) {
                Arrays.fill(hold1, prices[i]);
            } else {
                Arrays.fill(hold2, Math.max(prices[i], prices[i] - hold2));
            }
        }

        int sell = 0;
        for (int i : hold1) {
            sell += Math.max(0, i - hold1[0]);
        }
        return sell;
    }

        if (prices.length <= 3) {
            return 0;

        int[] price = new int[prices.length];

        for (int i = 1; i < prices.length; i++) {
            price[i] = Math.max(prices[i], price[i-1]);
        }
        return price[price.length - 1];

```java
import java.util.*;

public class StockPrices {

    public static int maxProfit(int[] prices) {
        if (prices.length < 2) {
            return 0;
        }

        int buy = -1000, sell = 0;

        for (int price : prices) {
            while (buy + price < price) {
                buy = price;
            }
            sell += Math.max(0, price - buy);
        }

        return sell;
    }

    public static int maxProfit(int[] prices) {
        if (prices.length <= 2) {
            return 0;
        }

        List<Integer> list = new ArrayList<>();
        for (int i : prices) {
            if (!list.isEmpty()) {
                list.add(Math.max(list.get(list.size()-1), i - list.get(0)));
            }
            list.add(i);
        }
        Collections.sort(list);

        int sell = 0;
        for (Integer val : list) {
            sell += Math.max(0, val - val);
        }

        return sell;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        System.out.print("Enter the number of days: ");
        int n = scanner.nextInt();
        int[] prices = new int[n];
        for (int i = 0; i < n; i++) {
            System.out.print("Enter price at day " + (i+1) + ": ");
            prices[i] = scanner.nextInt();
        }
        System.out.println(maxProfit(prices));
    }

    public static void testMaxProfit() {
        int[] arr1 = {7, 1, 5, 3, 6, 4};
        System.out.println(maxProfit(arr1)); // Expected: 7

        int[] arr2 = {1, 2, 3, 4, 5};
        System.out.println(maxProfit(arr2)); // Expected: 4

        int[] arr3 = {10, 20, 30, 40, 50};
        System.out.println(maxProfit(arr3)); // Expected: 100
    }

    public static void main(String[] args) {
        testMaxProfit();
    }
}
```

This code creates a simple program that prompts the user to enter the prices for multiple days. It then calls the `maxProfit` method with these prices and prints out the result.

The `maxProfit` method uses dynamic programming to find the maximum possible profit. It maintains two variables, `buy` and `sell`, representing the minimum price and maximum profit so far. It iterates over the prices array, updating `buy` and `sell` whenever it finds a better deal. The final value of `sell` represents the maximum possible profit.

Please note that this solution assumes that the input is valid, i.e., it does not check if the number of days is positive or if the prices are non-negative.
