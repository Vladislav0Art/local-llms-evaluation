import java.io.ByteArrayInputStream;
import java.io.DataOutputStream;

public class TestUtil {

    public static Position encodePosition(double latitude, double longitude) {
        return new Position(latitude, longitude);
    }

    public static String serializePosition(Position position) throws Exception {
        DataOutputStream out = new DataOutputStream(System.out);
        BcdUtil.encodeByte(position.getLatitude() * 1e7 + position.getLongitude());
        out.writeDouble(position.getLatitude());
        out.writeDouble(position.getLongitude());
        return new ByteArrayInputStream((byte[]) System.out.toString()).toString();
    }

    public static Position deserializePosition(byte[] bytes) {
        System.arraycopy(bytes, 0, System.in);
        DataInputStream in = new DataInputStream(System.in);
        double latitude = in.readDouble() / 1e7;
        double longitude = in.readDouble();
        return encodePosition(latitude, longitude);
    }
}

class BcdUtil {

    public static byte[] encodeByte(long value) {
        // implementation of bcd encoding
        int shift = 24; // bit shifting
        int signBitMask = (int) (1 << 31); // sign bit masking
        long shiftedValue = (value & ~signBitMask);
        return BcdUtil.shiftLeft(32 - shift, shiftedValue);
    }

    public static byte[] shiftLeft(int shift, long value) {
        return ((byte[]) value).concat(new byte[]{(byte) 0, (byte) 0, (byte) 0});
    }
}

class Position {

    private double latitude;
    private double longitude;

    public Position(double latitude, double longitude) {
        this.latitude = latitude;
        this.longitude = longitude;
    }

    public double getLatitude() {
        return latitude;
    }

    public void setLatitude(double latitude) {
        this.latitude = latitude;
    }

    public double getLongitude() {
        return longitude;
    }

    public void setLongitude(double longitude) {
        this.longitude = longitude;
    }
}

class DataUtils {

    public static ByteArrayInputStream serialize(DataOutputStream out) throws Exception {
        out.writeDouble(1.0);
        out.writeDouble(2.0);
        byte[] bytes = new byte[1024];
        int bytesRead = out.read(bytes, 0, 1024);
        System.arraycopy(bytes, 0, bytes, 0, bytesRead);
        return new ByteArrayInputStream(bytes);
    }

    public static DataInputStream deserialize(byte[] bytes) throws Exception {
        ByteArrayInputStream bis = new ByteArrayInputStream(bytes);
        double latitude = bis.readDouble();
        double longitude = bis.readDouble();
        return new DataInputStream(bis);
    }
