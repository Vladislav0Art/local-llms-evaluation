public class TestSpark {

    public static class Main {
        public String toStringArray(int[] data) {
            return Arrays.toString(data);
        }

        public byte[] decode(byte[] data) {
            System.out.println("Decoded: " + data);
            return data;
        }
    }

    public static void main(String[] args) {
        Main m = new Main();
        
        String stringArray1 = m.toStringArray(new int[]{1, 2, 3});
        byte[] byteArray1 = m.decode(new byte[]{1, 2, 3});
        System.out.println("stringArray1: " + stringArray1);
        System.out.println("byteArray1: " + byteArray1);

        String stringArray2 = m.toStringArray(new int[]{1.0, 2.0, 3.0});
        boolean b = (boolean) m.decode(new byte[]{(byte) 1, (byte) 2, (byte) 3, (byte) 4});
        System.out.println("stringArray2: " + stringArray2);
        System.out.println("b: " + b);

        int[] array1 = new int[3];
        int[] array2 = {true};
        
        String stringArray3 = m.toStringArray(array1);
        byte[] byteArray3 = m.decode(byteArray1);
        System.out.println("stringArray3: " + stringArray3);
        System.out.println("byteArray3: " + byteArray3);

        stringArray3 = new String(new ByteArrayInputStream(stringArray2), StandardCharsets.UTF_8);
        byteArray3 = new byte[stringArray2.length];
        for (int i = 0; i < stringArray2.length; i++) {
            byteArray3[i] = (byte) stringArray2.charAt(i);
        }
        
        System.out.println("stringArray3: " + stringArray3);
        System.out.println("byteArray3: " + Arrays.toString(byteArray3));
    }
}
