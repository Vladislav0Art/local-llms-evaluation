        assertTrue(node.isBlank());
    }

        assertEquals(1, node.splitText(0));
        assertEquals("Hello", node.splitText(1).text());
    }

    public void outerHtmlHead_test() throws IOException {
        Document doc = new Document();
        Appendable accum = doc.toString().toCharArray().toArray(new char[0]);
        assertTrue(doc.outerHtmlHead(0, 2).trim().length > 0);
        String expectedOutput = "<p>Hello World</p>";
        assertEquals(expectedOutput, String.join("", (String) accum));
    }

    @Test
    public void outerHtmlTail_test() throws IOException {
        Document doc = new Document();
        Appendable accum = doc.toString().toCharArray().toArray(new char[0]);
        assertTrue(doc.outerHtmlTail(0, 2).trim().length > 0);
        String expectedOutput = "<p>Hello World</p>";
        assertEquals(expectedOutput, String.join("", (String) accum));
    }

    public void toString_test() {

        String output = node.toString();
        assertEquals("<text> Hello <span>World</span></text>", output);
    }

    public void clone_test() {

        TextNode clonedNode = (TextNode) node.clone();
        assertEquals(node, clonedNode);
    }

    public void createFromEncoded_test() throws Exception {

        assertEquals("Hello World", node.text());
