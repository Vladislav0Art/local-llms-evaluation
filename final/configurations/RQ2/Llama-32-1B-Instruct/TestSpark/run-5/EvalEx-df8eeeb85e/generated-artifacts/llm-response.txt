                        case "identifier":
                            return token.value.toString();
                        default:
                            throw new RuntimeException("Unknown token: " + token.type);
                    }
                })
                .collect(Collectors.joining());

    public static String parse(List<Token> tokens) {
        StringBuilder sb = new StringBuilder();
        int start = 0;
        while (start < tokens.size()) {
            Token token = tokens.get(start);

                sb.append(token.value).append(" = ");

                sb.append("unknown token at index ").append(start);
                start++;
            }
        }
        return sb.toString();
