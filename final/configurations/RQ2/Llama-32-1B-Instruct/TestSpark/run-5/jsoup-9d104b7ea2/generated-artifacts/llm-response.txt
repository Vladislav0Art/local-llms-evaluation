public class TextNodeTest {

    private static MockHttpServletRequest request;

    @Before
    public void setup() {
        String encodedText = "{\"hello:world\"";
        request = new MockHttpServletRequest();
        request.setMethod("POST");
        request.getParameterMap().put("_t", encodedText);
        request.getContentType().setValue("text/plain");

        request.reset();
    }

    @Test
    public void [MethodUnderTest]Test() throws IOException {
        String text = request.getParameter("text").toString();
        TextNode node = new TextNode(text);

        assertEquals("hello world", node.text());
    }

    @Test
    public void [MethodUnderTest]Blank() {
        String text = "";
        TextNode node = new TextNode(text);
        assertTrue(node.isBlank());

        text = "hello";
        node = new TextNode(text);
        assertFalse(node.isBlank());
    }

    @Test
    public void [MethodUnderTest]SplitText() throws IOException {
        String text = "hello world";
        int offset = 5;
        TextNode node = new TextNode(text);

        assertEquals("world", node.splitText(offset).text());

        offset = -10;
        node = new TextNode(text);
        assertEquals("", node.splitText(offset));
    }

    @Test
    public void [MethodUnderTest]ToString() {
        String text = "hello world";
        TextNode node = new TextNode(text);

        assertEquals("Hello, world!", node.toString());
    }
}

class MockHttpServletRequest extends HttpServletRequest {

    private String _t;

    public String getParameter(String name) throws IOException {
        return _t;
    }

    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
        _t = "";
    }
}

public class TextNodeTest {

    @MockRequest("POST")
    private MockHttpServletRequest request;

    @Before
    public void setup() {
        String encodedText = "{\"hello:world\"";
        request.setMethod("POST");
        request.getParameterMap().put("_t", encodedText);
        request.getContentType().setValue("text/plain");

        request.reset();
    }

    [Test]
    public void TextNodeTest() throws IOException {
        String text = request.getParameter("text").toString();
        TextNode node = new TextNode(text);

        // Test if method is called
        verify(node, times(1)).text();

        // Test if method returns string
        assertEquals("hello world", node.text());
    }

    [Test]
    public void Blank() throws IOException {
        String text = "Hello";
        TextNode node = new TextNode(text);

        // Test if method is called
        verify(node, times(1)).text();

        // Test if method returns string
        assertEquals("Hello", node.text());
    }

    [Test]
    public void SplitText() throws IOException {
        String text = "hello world";
        TextNode node = new TextNode(text);

        // Test if method is called
        verify(node, times(1)).splitText();

        // Test if method returns string
        assertEquals("hello", node.text());
        assertEquals("world", node.text());
    }

    [Test]
    public void ToString() throws IOException {
        String text = "Hello World";
        TextNode node = new TextNode(text);

        // Test if method is called

    }

public class RequestParameterTest {

    @MockRequest("POST")
    private MockHttpServletRequest request;

    public void testSetup() {
        String encodedText = "{\"hello:world\"";
        request.setMethod("POST");
        request.getParameterMap().put("_t", encodedText);
        request.getContentType().setValue("text/plain");

        request.reset();
    }

    [Test]
    public void ParametersTest() throws IOException {
        String text = request.getParameter("text").toString();
        RequestParameter parameter = new RequestParameter(text);

        assertEquals("hello world", parameter.getValue());
    }

    [Test]
    public void NullParametersTest() throws IOException {
        String text = "Hello";
        RequestParameter parameter = new RequestParameter(text);

        assertEquals(null, parameter.getValue());
    }

    [Test]
    public void SplitParametersTest() throws IOException {
        String text = request.getParameter("text").toString();
        RequestParameter parameter = new RequestParameter(text);

        assertEquals("hello world", parameter.splitText());

        String[] result = parameter.splitText();

        assertEquals("hello world", result[0]);
        assertEquals("world", result[1]);
    }

    [Test]
    public void ToStringParametersTest() throws IOException {
        String text = "Hello World";
        RequestParameter parameter = new RequestParameter(text);

        assertEquals("Hello World", parameter.toString());
    }
}
