import org.jsoup.helper.UrlBuilder;
import org.junit.Test;
import java.net.URL;
import static org.junit.Assert.*;

public class UrlBuilderTest {
    @Test
    public void buildUrlWithoutQuery() throws Exception {
        URL inputUrl = new URL("http://example.com/test");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        URL outputUrl = urlBuilder.build();
        assertEquals("http://example.com/test", outputUrl.toString());
    }

    @Test
    public void buildUrlWithQuery() throws Exception {
        URL inputUrl = new URL("http://example.com/test?key=value");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        URL outputUrl = urlBuilder.build();
        assertEquals("http://example.com/test?key=value", outputUrl.toString());
    }

    @Test
    public void buildUrlWithRef() throws Exception {
        URL inputUrl = new URL("http://example.com/test#ref");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        URL outputUrl = urlBuilder.build();
        assertEquals("http://example.com/test#ref", outputUrl.toString());
    }

    @Test
    public void buildUrlWithQueryAndRef() throws Exception {
        URL inputUrl = new URL("http://example.com/test?key=value#ref");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        URL outputUrl = urlBuilder.build();
        assertEquals("http://example.com/test?key=value#ref", outputUrl.toString());
    }

    @Test
    public void appendKeyVal() throws Exception {
        URL inputUrl = new URL("http://example.com/test");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        urlBuilder.appendKeyVal(new Connection.KeyVal("key", "value"));
        URL outputUrl = urlBuilder.build();
        assertEquals("http://example.com/test?key=value", outputUrl.toString());
