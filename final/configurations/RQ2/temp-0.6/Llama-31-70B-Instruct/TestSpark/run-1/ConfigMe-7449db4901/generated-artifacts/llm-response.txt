```java
import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNotNull;
import org.junit.Test;
import org.mockito.Mockito;
import ch.jalu.configme.configurationdata.CommentsConfiguration;

public class CommentsConfigurationTest {

    @Test
    public void setCommentAddsCommentLinesToMap() {
        CommentsConfiguration commentsConfiguration = new CommentsConfiguration();
        String path = "test.path";
        String[] commentLines = new String[]{"Line 1", "Line 2"};

        commentsConfiguration.setComment(path, commentLines);

        assertEquals(commentLines, commentsConfiguration.getAllComments().get(path));
    }

    @Test
    public void getAllCommentsReturnsUnmodifiableMap() {
        CommentsConfiguration commentsConfiguration = new CommentsConfiguration();
        Map<String, List<String>> comments = commentsConfiguration.getAllComments();

        assertNotNull(comments);
        assertEquals(comments, commentsConfiguration.getAllComments());
    }
}
```

This code tests the `setComment` and `getAllComments` methods of the `CommentsConfiguration` class. The `setComment` method is tested by creating a new instance of `CommentsConfiguration`, adding a comment to the map, and then verifying that the comment lines are present in the map. The `getAllComments` method is tested by creating a new instance of `CommentsConfiguration`, retrieving the unmodifiable map of comments, and verifying that the map is not null and that it is the same map that is returned by the method.
