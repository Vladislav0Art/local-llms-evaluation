import org.junit.Test;
import org.jsoup.helper.UrlBuilder;
import org.jsoup.Connection;
import java.net.URL;
import java.net.MalformedURLException;
import java.net.URISyntaxException;
import java.io.UnsupportedEncodingException;

import static org.junit.Assert.*;
import static org.mockito.Mockito.*;

public class UrlBuilderTest {
    @Test
    public void testBuildWithoutQuery() throws MalformedURLException, URISyntaxException, UnsupportedEncodingException {
        URL inputUrl = new URL("http://www.example.com/");
        UrlBuilder builder = new UrlBuilder(inputUrl);
        URL builtUrl = builder.build();
        assertEquals(inputUrl, builtUrl);
    }

    @Test
    public void testBuildWithQuery() throws MalformedURLException, URISyntaxException, UnsupportedEncodingException {
        URL inputUrl = new URL("http://www.example.com/");
        UrlBuilder builder = new UrlBuilder(inputUrl);
        Connection.KeyVal kv = new Connection.KeyVal("key", "value");
        builder.appendKeyVal(kv);
        URL builtUrl = builder.build();
        assertTrue(builtUrl.getQuery().contains(kv.key()));
        assertTrue(builtUrl.getQuery().contains(kv.value()));
    }

    @Test
    public void testBuildWithQueryAndFragment() throws MalformedURLException, URISyntaxException, UnsupportedEncodingException {

        builder.appendKeyVal(kv);
        builder.u.setRef("fragment");
        URL builtUrl = builder.build();
        assertTrue(builtUrl.getQuery().contains(kv.key()));
        assertTrue(builtUrl.getQuery().contains(kv.value()));
        assertEquals("fragment", builtUrl.getRef());
