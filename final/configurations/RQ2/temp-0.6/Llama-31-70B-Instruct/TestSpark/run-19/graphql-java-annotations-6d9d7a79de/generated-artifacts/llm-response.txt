public void testBuildFromJavaAnnotations() {

    AnnotatedElement mockAnnotatedElement = mock(AnnotatedElement.class);
    ProcessingElementsContainer mockContainer = mock(ProcessingElementsContainer.class);
    GraphQLDirective mockDirective = mock(GraphQLDirective.class);
    when(mockContainer.getDirectiveRegistry()).thenReturn(new HashMap<String, GraphQLDirective>() {
        {
            put("testDirective", mockDirective);
        }
    });
    DirectiveJavaAnnotationUtil mockDirectiveUtil = mock(DirectiveJavaAnnotationUtil.class);
    when(mockDirectiveUtil.getDirectiveAnnotations(mockAnnotatedElement)).thenReturn(Arrays.asList(mock(Annotation.class)));
    when(mockDirectiveUtil.getName(any(Annotation.class))).thenReturn("testDirective");
    DirectivesBuilder directivesBuilder = new DirectivesBuilder(mockAnnotatedElement, mockContainer);

    // when
    GraphQLDirective[] directives = directivesBuilder.build();

    // then
    assertNotNull(directives);
    assertEquals(1, directives.length);
    assertSame(mockDirective, directives[0]);
}
