public class UrlBuilderTest {

    private UrlBuilder urlBuilder;

    @Before
    public void setUp() throws Exception {
        URL inputUrl = new URL("https://example.com/");
        urlBuilder = new UrlBuilder(inputUrl);
    }

    @Test
    public void shouldBuildUrl() throws Exception {
        URL expectedUrl = new URL("https://example.com/");

        URL actualUrl = urlBuilder.build();

        assertEquals(expectedUrl, actualUrl);
    }

    @Test
    public void shouldAppendKeyVal() throws Exception {
        Connection.KeyVal keyVal = new Connection.KeyVal("key", "value");

        urlBuilder.appendKeyVal(keyVal);

        assertEquals("key=value", urlBuilder.q.toString());
    }

    @Test
    public void shouldNormalizeQuery() throws Exception {
        String query = "key=value";

        String normalizedQuery = UrlBuilder.normalizeQuery(query);

        assertEquals("key=value", normalizedQuery);
    }

    @Test
    public void shouldNormalizeRef() throws Exception {
        String ref = "key=value";

        String normalizedRef = UrlBuilder.normalizeRef(ref);

        assertEquals("key%3Dvalue", normalizedRef);
    }

    @Test
    public void shouldDecodePart() throws Exception {
        String encoded = "key%3Dvalue";

        String decoded = UrlBuilder.decodePart(encoded);

        assertEquals("key=value", decoded);
    }

}
