	```java
	@Test
	public void testEnsureChildNodes() {
		Element element = new Element("div");
		assertNull(element.childNodes);
		element.ensureChildNodes();
		assertNotNull(element.childNodes);
		assertTrue(element.childNodes.isEmpty());
	}
