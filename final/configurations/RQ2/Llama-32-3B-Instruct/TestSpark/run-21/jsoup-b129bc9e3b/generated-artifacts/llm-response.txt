public class HttpConnection {
    private String url;
    private Connection conn;

    public void setUrl(String url) {
        this.url = url;
    }

    public Response get() throws IOException, SQLException {
        // Simulate a malformed response
        throw new IOException("Malformed response");
    }

    public Response post() throws IOException, SQLException {
        // Simulate a malformed response
        throw new IOException("Malformed response");
    }

    public Connection execute() throws IOException, SQLException {
        // Simulate an acceptable response
        return this.conn;
    }
}

public class Response {
    private String data;

    public Response(String data) {
        this.data = data;
    }

    public String getData() {
        return data;
    }
}

public class TestHttpConnection {

    public Connection createConn() {
        Connection conn = new Connection();
        return conn;
    }

    @Test
    public void testGetMalformedResponse_ThrowsIOException() throws IOException, SQLException {
        HttpConnection httpConn = new HttpConnection();
        httpConn.setUrl("https://example.com");

        assertThrows(IOException.class, () -> (Response)httpConn.get());
    }

    @Test
    public void testPostMalformedResponse_ThrowsIOException() throws IOException, SQLException {
        HttpConnection httpConn = new HttpConnection();
        httpConn.setUrl("https://example.com");

        assertThrows(IOException.class, () -> (Response)httpConn.post());
    }

    @Test
    public void testExecuteMalformedResponse_ThrowsIOException() throws IOException, SQLException {
        HttpConnection httpConn = new HttpConnection();
        httpConn.setUrl("https://example.com");

        assertThrows(IOException.class, () -> (Response)httpConn.execute());
    }

    @Test
    public void testGetAcceptableResponse_ReturnsResponse() throws IOException, SQLException {
        HttpConnection httpConn = new HttpConnection();
        Connection conn = this.createConn();
        when(conn.data).thenReturn("data");
        httpConn.conn = conn;
        when((Response)httpConn.execute()).thenReturn(new Response("data"));

        Response response = (Response)httpConn.execute();

        assertNotNull(response);
    }

    @Test
    public void testPostAcceptableResponse_ReturnsResponse() throws IOException, SQLException {

        httpConn.conn = conn;

        Response response = (Response)httpConn.post();

        assertNotNull(response);
    }

        httpConn.conn = conn;
