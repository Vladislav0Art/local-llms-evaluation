        this.protocol = protocol;
        this.message = message;
    }

    public abstract void send();
}

// PublicNetworkMessage class extends Message
class PublicNetworkMessage extends Message {
    private String address;

    public PublicNetworkMessage(String protocol, String message, String address) {
        super(protocol, message);
        this.address = address;
    }

    @Override
    public void send() {
        System.out.println("Sending public network message to " + address);
    }
}

// PrivateNetworkMessage class extends Message
class PrivateNetworkMessage extends Message {
    private String recipientAddress;

    public PrivateNetworkMessage(String protocol, String message, String recipientAddress) {
        super(protocol, message);
        this.recipientAddress = recipientAddress;
    }

    @Override
    public void send() {
        System.out.println("Sending private network message to " + recipientAddress);
    }
}

// NetworkClient class
class NetworkClient {
    public void receiveMessage(Message message) {
        message.send();
    }
}

public class Main {
    public static void main(String[] args) {
        // Create a network client
        NetworkClient client = new NetworkClient();

        // Create public and private messages
        PublicNetworkMessage publicMessage = new PublicNetworkMessage("HTTP", "Hello, world!", "127.0.0.1");
        PrivateNetworkMessage privateMessage = new PrivateNetworkMessage("FTP", "Top secret information.", "192.168.1.100");

        // Send messages through the network client
        client.receiveMessage(publicMessage);
        client.receiveMessage(privateMessage);
    }

        String protocol = bufferedReader.readLine();
        String message = bufferedReader.readLine();

        switch (protocol) {
            case "HTTP":
                System.out.println("Received HTTP request: " + message);
                break;
            case "FTP":
                System.out.println("Received FTP request: " + message);
                break;
            default:
                System.out.println("Unknown protocol");
        }

        // Send a response back to the client
        printWriter.println("HTTP Response: Hello, world!");
    }
}
