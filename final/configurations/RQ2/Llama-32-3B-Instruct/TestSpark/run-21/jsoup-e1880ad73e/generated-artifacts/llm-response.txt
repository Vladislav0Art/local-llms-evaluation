import org.jsoup.internal.StringUtil;
import org.junit.Test;

import java.net.URLDecoder;
import java.net.URLEncoder;

@RunWith(MockitoJUnitRunner.class)
public class UrlBuilderTest {

    @Mock
    private Connection connection;

    @Test
    public void ofNull_ThrowsNullPointerException() {
        assertThrows(NullPointerException.class, () -> new UrlBuilder(null));
    }

    @Test
    public void build_ReturnsDefaultUrl() {
        // Arrange
        Mockito.when(connection.getScheme()).thenReturn("https");
        Mockito.when(connection.getHost()).thenReturn("example.com");

        // Act
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        URL result = builder.build();

        // Assert
        assert result.getProtocol().equals("https");
        assert result.getHost().equals("example.com");
    }

    @Test
    public void appendKeyVal_WritesToUrl() {
        // Arrange
        Mockito.when(connection.getScheme()).thenReturn("https");
        Mockito.when(connection.getHost()).thenReturn("example.com");

        Connection.KeyVal kv = new Connection.KeyVal("key", "value");
        builder.appendKeyVal(kv);

        // Act
        String url = Mockito.spy(connection.toString());
        builder.build();

        // Assert
        assert url.startsWith("https:");
        assert url.contains("?key=value");
    }

    @Test
    public void appendKeyVal_ThrowsUnsupportedEncodingException() {

    }

import org.jsoup.Connection;
import org.jsoup.internal.StringUtil;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import java.io.UnsupportedEncodingException;
import java.net.IDN;
import java.net.MalformedURLException;
import java.net.URI;
import java.net.URISyntaxException;
import java.net.URL;
import java.net.URLDecoder;
import java.net.URLEncoder;

@RunWith(MockitoJUnitRunner.class)
public class UrlBuilderTest {

    @Mock
    private Connection connection;

    public String buildUrl() {
        Mockito.when(connection.getScheme()).thenReturn("https");
        Mockito.when(connection.getHost()).thenReturn("example.com");

        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        return builder.build().toString();
    }

    @Test
    public void ofNull_ThrowsNullPointerException() {
        assertThrows(NullPointerException.class, () -> new UrlBuilder(null));
    }

    @Test
    public void testBuildUrl() {
        String url = buildUrl();
        Mockito.verifyNoMoreInteractions(connection);
    }

    @Test
    public void testGetScheme() throws URISyntaxException {
        Mockito.when(connection.getScheme()).thenReturn("https");
        assertThrows(URISyntaxException.class, () -> new URL(buildUrl()));
    }

    @Test
    public void testGetHost() throws URISyntaxException {
        Mockito.when(connection.getHost()).thenReturn("example.com");
        assertThrows(URISyntaxException.class, () -> new URL(buildUrl()));
    }

    public String buildUrlWithKeyVal() {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        Connection.KeyVal kv = new Connection.KeyVal("key", "value");

        Mockito.when(connection.getKey()).thenReturn(kv.getKey());
        Mockito.when(connection.getValue()).thenReturn(kv.getValue());

        return builder.appendKeyVal(kv).toString();
    }

    @Test
    public void testAppendKeyValue() throws URISyntaxException {
        String url = buildUrlWithKeyVal();
        Mockito.verifyNoMoreInteractions(connection);
    }

    @Test
    public void testAppendKeyValEncodesValues() {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        Connection.KeyVal kv = new Connection.KeyVal("key", "value");

        Mockito.when(connection.getKey()).thenReturn(kv.getKey());
        Mockito.when(connection.getValue()).thenReturn(kv.getValue());

        String url = builder.appendKeyVal(kv).toString();
        Mockito.verifyNoMoreInteractions(connection);
    }

    @Test
    public void testAppendKeyValueThrowsException() {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        Connection.KeyVal kv = null;

        assertThrows(NullPointerException.class, () -> builder.appendKeyVal(kv));
    }
}
