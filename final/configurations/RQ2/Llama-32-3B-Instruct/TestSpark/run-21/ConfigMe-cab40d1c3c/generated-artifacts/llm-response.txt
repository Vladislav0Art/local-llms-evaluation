import java.util.ArrayList;
import java.util.List;
import java.util.Map;

public class Property {
    private String name;
    private String value;

    public Property(String name, String value) {
        this.name = name;
        this.value = value;
    }

    public String getName() {
        return name;
    }
}

public class PropertyListBuilder {
    public List<Property> create(List<Property> properties) {
        List<Property> result = new ArrayList<>();
        for (Property property : properties) {
            result.add(property);
        }
        return result;
    }

    public List<Property> create(Map<String, String> map) {
        List<Property> result = new ArrayList<>();
        for (Map.Entry<String, String> entry : map.entrySet()) {
            Property property = new Property(entry.getKey(), entry.getValue());
            result.add(property);
        }
        return result;
    }

    public List<Property> create(List<String> list) {
        List<Property> result = new ArrayList<>();
        for (String str : list) {
            String[] parts = str.split(",");
            if (parts.length == 2) {
                Property property = new Property(parts[0], parts[1]);
                result.add(property);
            }
        }
        return result;
    }

    public List<Property> create(List<PropertyListBuilder.Property> properties) {
        List<Property> result = new ArrayList<>();
        for (PropertyListBuilder.Property property : properties) {
            result.add(new Property(property.getName(), property.getValue()));
        }
        return result;
    }
}

import java.util.ArrayList;
import java.util.List;

public class Property {
    private String name;
    private String value;

    public Property(String name, String value) {
        this.name = name;
        this.value = value;
    }

    public String getName() {
        return name;
    }
}

public class PropertyListBuilder {
    // ... existing methods ...

    public static List<Property> create(List<String> properties) {
        List<Property> result = new ArrayList<>();
        for (String property : properties) {
            String[] parts = property.split(",");
            if (parts.length == 2) {
                result.add(new Property(parts[0], parts[1]));
            }
        }
        return result;
    }

    public static List<Property> create(List<Map.Entry<String, String>> map) {
        List<Property> result = new ArrayList<>();
        for (Map.Entry<String, String> entry : map) {
            result.add(new Property(entry.getKey(), entry.getValue()));
        }
        return result;
    }

    public static List<Property> create(List<List<String>>> list) {
        List<Property> result = new ArrayList<>();
        for (List<String> subList : list) {
            if (subList.size() == 2) {
                result.add(new Property(subList.get(0), subList.get(1)));
            }
        }
        return result;
    }

    public static List<Property> create(List<List<Object>> list) {
        List<Property> result = new ArrayList<>();
        for (List<Object> subList : list) {
            if (subList.size() == 2) {
                Object nameObj = subList.get(0);
                Object valueObj = subList.get(1);
                if (nameObj instanceof String && valueObj instanceof String) {
                    result.add(new Property((String) nameObj, (String) valueObj));
                }
            }
        }
        return result;
    }

    public static List<Property> create(List<Object[]> list) {
        List<Property> result = new ArrayList<>();
        for (Object[] subArray : list) {
            if (subArray.length == 2) {
                String name = (String) subArray[0];
                String value = (String) subArray[1];
                result.add(new Property(name, value));
            }
        }
        return result;
    }
}
