public class Token {
    public enum TokenType {
        INVALID_OPERATOR,
        INVALID_TOKEN,
        STRING_LITERAL,
        INVALID_OPERATOR_TOKEN,
        VALID_TOKEN,
    }

    private final TokenType type;
    private final String value;

    public Token(TokenType type, String value) {
        this.type = type;
        this.value = value;
    }

    public static Token create(String value) {
        if (value.matches("[^\\w\s]+")) { // invalid operator
            return new Token(TokenType.INVALID_OPERATOR_TOKEN, value);
        } else if (value.startsWith("\"") && value.endsWith("\"")) { // string literal
            return new Token(TokenType.STRING_LITERAL, value.substring(1, value.length() - 1));
        } else if (!Character.isLetterOrDigit(value.charAt(0))) {

        }
    }

public class Token {
    public enum TokenType {
        INVALID_OPERATOR,
        INVALID_TOKEN,
        STRING_LITERAL,
        INTEGER_LITERAL,
        VALID_TOKEN,
        ADDITION,
        SUBTRACTION,
        MULTIPLICATION,
        DIVISION
    }

    private final TokenType type;
    private final String value;

    public Token(TokenType type, String value) {
        this.type = type;
        this.value = value;
    }

    public static Token create(String value) {
        if (value.matches("[^\\w\\s\\+\\-\\*\\/]")) { // invalid operator
            return new Token(TokenType.INVALID_OPERATOR, value);
        } else if (Character.isLetterOrDigit(value.charAt(0))) {
            String[] parts = value.split("\\.");
            StringBuilder numberValue = new StringBuilder();
            for (String part : parts) {
                numberValue.append(part);
            }
            try {
                int number = Integer.parseInt(numberValue.toString());
                return new Token(TokenType.INTEGER_LITERAL, value);
            } catch (NumberFormatException e) {
                if (value.equals("0")) {
                    return new Token(TokenType.INTEGER_LITERAL, "0");
                } else {
                    String[] parts = value.split("/");
                    StringBuilder decimalValue = new StringBuilder();
                    for (String part : parts) {
                        decimalValue.append(part);
                    }
                    try {
                        double number = Double.parseDouble(decimalValue.toString());
                        return new Token(TokenType.DECIMAL_LITERAL, value);
                    } catch (NumberFormatException ex) {
                        return new Token(TokenType.STRING_LITERAL, value);
                    }
                }
            }
        } else if (value.startsWith("\"") && value.endsWith("\"")) { // string literal
            return new Token(TokenType.STRING_LITERAL, value.substring(1, value.length() - 1));
        } else if ("+-*/".contains(value)) {
            String[] parts = value.split("\\+");
            StringBuilder operatorValue = new StringBuilder();
            for (String part : parts) {
                operatorValue.append(part);
            }
            return new Token(TokenType.ADDITION, operatorValue.toString());
        } else if (value.equals("+")) {
            return new Token(TokenType.ADDITION, "+");
        } else if (value.equals("-")) {
            return new Token(TokenType.SUBTRACTION, "-");

            return new Token(TokenType.MULTIPLICATION, "*");
        } else if (value.equals("/")) {
            return new Token(TokenType.DIVISION, "/");
        }
        // handle other cases
        return new Token(TokenType.VALID_TOKEN, value);
    }

        return type;
    }
