import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import static org.junit.Assert.*;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class WatchProtocolDecoderTest {

    @Mock
    private Protocol protocol;

    @Mock
    private DeviceSession deviceSession;

    @Mock
    private NetworkMessage networkMessage;

    private WatchProtocolDecoder watchProtocolDecoder = new WatchProtocolDecoder(protocol);

    @Test
    public void getHasIndex_EmptyMessage_ReturnsFalse() {
        assertFalse(watchProtocolDecoder.getHasIndex());
    }

    @Test
    public void getManufacturer_UnknownManufacturer_ReturnsDefault() {
        when(protocol.getDefaultManufacturer()).thenReturn("Unknown");
        assertEquals("Unknown", watchProtocolDecoder.getManufacturer());
    }

    @Test
    public void decode_DecodeMessage_ReturnsDecodedData() throws Exception {
        byte[] data = {0x01, 0x02, 0x03};
        ByteBuf buffer = Unpooled.buffer(data.length);
        BufferUtil.copyBuffer(buffer, data);
        Object result = watchProtocolDecoder.decode(null, null, networkMessage);
        assertEquals(deviceSession, result);
    }

    @Test
    public void decode_DecodeMessageWithInvalidData_ThrowsException() throws Exception {
        byte[] data = {0x01, 0x02, 0x03, (byte) -1};
        ByteBuf buffer = Unpooled.buffer(data.length);
        BufferUtil.copyBuffer(buffer, data);
        assertThrows(Exception.class, () -> watchProtocolDecoder.decode(null, null, networkMessage));
    }

    @Test
    public void decode_DecodeMessageWithInvalidLength_ThrowsException() throws Exception {
        byte[] data = {0x01, 0x02};
        ByteBuf buffer = Unpooled.buffer(data.length);
        BufferUtil.copyBuffer(buffer, data);
        assertThrows(Exception.class, () -> watchProtocolDecoder.decode(null, null, networkMessage));
    }

    @Test
    public void decode_DecodeMessageWithNoIndex_ReturnsDefault() throws Exception {
        byte[] data = {0x01, 0x02, 0x03};
        ByteBuf buffer = Unpooled.buffer(data.length);
        BufferUtil.copyBuffer(buffer, data);
        Object result = watchProtocolDecoder.decode(null, null, networkMessage);
        assertNull(result);
    }
}
