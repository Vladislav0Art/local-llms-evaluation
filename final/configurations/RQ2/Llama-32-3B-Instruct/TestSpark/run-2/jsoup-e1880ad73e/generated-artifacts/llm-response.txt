    public static class KeyVal {
        private String key;
        private String value;

        }

        public Builder appendKeyVal(Connection.KeyVal kv) {
            this.key = kv.getKey();
            this.value = kv.getValue();
            return this;
        }

        public Connection.KeyVal build() {
            return new Connection.KeyVal(key, value);
        }
    }

    public static Builder builder(String key, String value) {
        return new Builder().key(key).value(value);
    }

    public static void main(String[] args) {
        Connection.KeyVal kv = builder("key", "value").build();
        System.out.println(kv.getKey());
        System.out.println(kv.getValue());

        kv = builder("newKey", null).build();
        System.out.println(kv.getKey());
        System.out.println(kv.getValue());
    }
}

        }

    public static class Builder {

        public Builder appendKeyVal(Connection.KeyVal kv) {
            this.kv = kv;
            return this;
        }

        public Connection.KeyVal build() {
            return kv;
        }

        public void appendKeyValue(String key, String value) {
            kv.key = key;
            kv.value = value;
        }
    }

    public static Builder builder() {
