package com.crowdin.client.core.http.impl.json;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNotNull;
import static org.junit.Assert.assertTrue;

import com.crowdin.client.core.http.HttpRequest;
import com.crowdin.client.core.http.impl.json.JacksonJsonTransformer;
import com.crowdin.client.projectsgroups.model.Project;
import com.crowdin.client.sourcefiles.model.ExportOptions;
import com.crowdin.client.sourcefiles.model.FileInfo;
import com.crowdin.client.sourcefiles.model.ImportOptions;
import com.crowdin.client.stringtranslations.model.LanguageTranslations;
import com.fasterxml.jackson.databind.ObjectMapper;

public class JacksonJsonTransformerTest {

    private JacksonJsonTransformer transformer = new JacksonJsonTransformer();

    public void testParseProject() {
        String json = "{\"id\":1,\"name\":\"test\"}";
        Project project = transform(json, Project.class);
        assertNotNull(project);
        assertEquals(1, project.getId());
        assertEquals("test", project.getName());
    }

    public void testParseExportOptions() {
        String json = "{\"exportFormat\":\"csv\",\"delimiter\":\";\",\"encoding\":\"UTF-8\"}";
        ExportOptions exportOptions = transformer.transform(json, ExportOptions.class);
        assertNotNull(exportOptions);
        assertEquals("csv", exportOptions.getExportFormat());
        assertEquals(";", exportOptions.getDelimiter());
    }

    public void testParseFileInfo() {
        String json = "{\"fileName\":\"test.txt\",\"path\":\"/test\"}";
        FileInfo fileInfo = transformer.transform(json, FileInfo.class);
        assertNotNull(fileInfo);
        assertEquals("test.txt", fileInfo.getFileName());
        assertEquals("/test", fileInfo.getPath());
    }

    public void testParseImportOptions() {
        String json = "{\"importFormat\":\"csv\",\"delimiter\":\";\",\"encoding\":\"UTF-8\"}";
        ImportOptions importOptions = transformer.transform(json, ImportOptions.class);
        assertNotNull(importOptions);
        assertEquals("csv", importOptions.getImportFormat());
        assertEquals(";", importOptions.getDelimiter());
    }

    public void testParseLanguageTranslations() {
        String json = "{\"en\": {\"key\":\"value\"}}";
        LanguageTranslations languageTranslations = transformer.transform(json, LanguageTranslations.class);
        assertNotNull(languageTranslations);
        assertEquals(1, languageTranslations.getLanguageTranslations().size());
    }

    @org.junit.Test
    @org.junit.SkipIf(os = Os.JDK7, message = "Not supported")
    public void testParseWithDeserializationFeature() {
        ObjectMapper mapper = new ObjectMapper();
        mapper.configure(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES, false);
        String json = "{\"id\":1,\"name\":\"test\"}";
        Project project = transformer.transform(json, Project.class, mapper);
        assertNotNull(project);
        assertEquals(1, project.getId());
        assertEquals("test", project.getName());
    }

    @org.junit.Test
    public void testConvertEmptyObject() {
        LanguageTranslations languageTranslations = new LanguageTranslations();
        String result = transformer.convert(languageTranslations);
        assertTrue(result.isEmpty());
    }

    @org.junit.Test
    public void testConvertNonEmptyObject() {
        LanguageTranslations languageTranslations = new LanguageTranslations();
        languageTranslations.getLanguageTranslations().add("en" + "{ \"key\":\"value\" }");
        String result = transformer.convert(languageTranslations);
        assertNotNull(result);
        assertTrue(result.contains("\"key\":\"value\""));
    }

    @org.junit.Test
    public void testConvertNullObject() {
        null;
        // No exception is thrown, because JacksonJsonTransformer does not throw exception when converting null object.
    }
}
