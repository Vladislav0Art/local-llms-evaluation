import org.jsoup.helper.Validate;
import org.jsoup.internal.StringUtil;
import java.io.IOException;
import org.junit.Test;
import static org.junit.Assert.*;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;
import org.mockito.SpyBean;

import org.junit.runners.JUnit4;

@RunWith(JUnit4.class)
public class TextNodeTest {

    private Appendable accum;

    @Mock
    private Document.OutputSettings out;

    @SpyBean
    private LeafNode leafNode;

    public void nodeName_EmptyText_ReturnsNull() {
        assertTrue(leafNode.nodeName().equals(""));
    }

    @Test
    public void text_EmptyText_ReturnsEmptyString() {
        String result = new TextNode("").text();
        assertTrue(result.equals(""));
    }

    @Test
    public void text_SingleCharacterText_ReturnsSingleCharacterText() {
        assertEquals("a", new TextNode("a").text());
    }

    @Test
    public void text_MultipleCharactersText_ReturnsMultipleCharactersText() {
        assertEquals("hello", new TextNode("hello").text());
    }

    @Test
    public void isBlank_EmptyText_ReturnsTrue() {
        assertTrue(new TextNode("").isBlank());
    }

    @Test
    public void isBlank_NonEmptyText_ReturnsFalse() {
        assertFalse(new TextNode("abc").isBlank());
    }

    @Test
    public void splitText_NoOffset_ReturnsSameTextNode() {
        TextNode result = new TextNode("");
        assertEquals(result, new TextNode("").splitText(0));
    }

    @Test
    public void outerHtmlHead_EmptyText_DoesNotAppendAnything() throws IOException {
        new TextNode("").outerHtmlHead(accum, 0, out);
        assertEquals(0, accum.length());
    }

    @Test
    public void outerHtmlTail_EmptyText_DoesNotAppendAnything() {
        new TextNode("").outerHtmlTail(accum, 0, out);
        assertEquals(0, accum.length());
    }

    @Test
    public void clone_ReturnsSameTextNode() {
        TextNode original = new TextNode("");
        TextNode result = original.clone();
        assertTrue(result.equals(original));
    }

    @Test
    public void createFromEncoded_ValidText_ReturnsTextNodeWithValidText() throws IOException {
        String encodedText = "Hello, World!";
        TextNode result = TextNode.createFromEncoded(encodedText);
        assertEquals("Hello, World!", result.text());
    }
}

    }

    @Test
    public void testNodeName() {
        String nodeName = this.nodeName();
        assertEquals("", nodeName);
    }

    public String nodeName() {
        return TextNode.NodeName;
    }
    
    @Test
    public void testBlank() {
        boolean blank = TextNode.isBlank("");
        assertTrue(blank);
    }
    
    @Test
    public void testIsBlank() {
        boolean blank = TextNode.isBlank(" ");
        assertTrue(blank);
    }

    @Test
    public void testSplitText() throws IOException {
        Appendable appendable = new StringBuilder();
        TextNode.splitText("", appendable, 0);
        assertEquals(0, ((StringBuilder)appendable).length());
    }

    private static class Accum implements Appendable {
        private final StringBuilder builder = new StringBuilder();

        @Override
        public Appendable append(CharSequence csq, int start, int end) {
            builder.append(csq);
            return this;
        }
        
        public String toString() {
            return builder.toString();
        }
    }

    @Test
    public void testOuterHtmlHead() throws IOException {
        Accum accum = new Accum();
        TextNode.outerHtmlHead(accum, 0, null, null);
        assertEquals(0, ((StringBuilder)accum).length());
    }

    private static class OutputSettings {
    }
    
    @Test
    public void testOuterHtmlTail() throws IOException {
        Accum accum = new Accum();
        TextNode.outerHtmlTail(accum, 0, null, null);
        assertEquals(0, ((StringBuilder)accum).length());
    }
}
