package com.netflix.frigga.ami;

public class AppVersion {
    private static final String APP_VERSION = "1.0.0";
    private static final Pattern PATTERN = Pattern.compile("\\d+\\.\\d+\\.\\d+-[a-zA-Z0-9]+");

    public static AppVersion parseName(String amiName) {
        return new AppVersion(PATTERN.matcher(amiName).find() ? amiName : null);
    }

    public static String getAppVersionPattern() {
        return PATTERN.toString();
    }

    public static String getPackageName() {
        return NameConstants.APP_VERSION;
    }

    public static String getVersion() {
        return APP_VERSION;
    }

    public static String getBuildJobName() {
        return "build-job-" + APP_VERSION;
    }

    public static String getBuildNumber() {
        return APP_VERSION.split("-")[0];
    }

    public static String getCommit() {
        return APP_VERSION.split("-")[1].split("\\.")[0];
    }

    @Override
    public int hashCode() {
        return APP_VERSION.hashCode();
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj)
            return true;
        if (!(obj instanceof AppVersion))
            return false;
        String s = ((AppVersion) obj).APP_VERSION;
        return PATTERN.matcher(s).find() && s.equals(APP_VERSION);
    }

    @Override
    public String toString() {
        return APP_VERSION;
    }
}

class NameConstants {
    public static final String APP_VERSION = "1.0.0";
}
