import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mockito;

import static org.hamcrest.MatcherAssert.assertThat;
import static org.hamcrest.Matchers.is;
import static org.junit.Assert.assertEquals;

@RunWith(MockitoJUnitRunner.class)
public class CommentsConfigurationTest {

    @Test
    public void newCommentsConfigurationHasNoComments() {
        // given
        CommentsConfiguration commentsConfig = new CommentsConfiguration();

        // when
        Map<String, List<String>> allComments = commentsConfig.getAllComments();

        // then
        assertThat(allComments.isEmpty(), is(true));
    }

    @Test
    public void newCommentsConfigurationIsUnmodifiable() {
        // given
        CommentsConfiguration commentsConfig = new CommentsConfiguration();

        // when
        Map<String, List<String>> allComments = commentsConfig.getAllComments();

        // then
        assertThat(commentsConfig.isUnmodifiable(), is(true));
    }

    @Test
    public void existingCommentsConfigurationHasComments() {
        // given
        Map<String, List<String>> comments = new HashMap<>();
        CommentsConfiguration commentsConfig = new CommentsConfiguration(comments);

        // when
        Map<String, List<String>> allComments = commentsConfig.getAllComments();

        // then
        assertThat(allComments.size(), is(1));
    }

    @Test
    public void existingCommentsConfigurationIsUnmodifiable() {
        // given
        Map<String, List<String>> comments = new HashMap<>();
        CommentsConfiguration commentsConfig = new CommentsConfiguration(comments);

        // when
        Map<String, List<String>> allComments = commentsConfig.getAllComments();

        // then
        assertThat(commentsConfig.isUnmodifiable(), is(true));
    }

    @Test
    public void setCommentOverwritesExistingComments() {
        // given
        Map<String, List<String>> comments = new HashMap<>();
        CommentsConfiguration commentsConfig = new CommentsConfiguration(comments);

        String path = "path";
        String commentLine = "comment line";

        // when
        commentsConfig.setComment(path, commentLine);

        // then
        Map<String, List<String>> allComments = commentsConfig.getAllComments();
        assertThat(allComments.get(path), is(Arrays.asList(commentLine)));
    }

    @Test
    public void setCommentDoesNotAddNewLinesWithoutMarker() {
        // given
        Map<String, List<String>> comments = new HashMap<>();
        CommentsConfiguration commentsConfig = new CommentsConfiguration(comments);

        String path = "path";
        String commentLine = "";

        // when
        commentsConfig.setComment(path, commentLine);

        // then
        Map<String, List<String>> allComments = commentsConfig.getAllComments();
        assertThat(allComments.get(path), is(Collections.singletonList(commentLine)));
    }

    @Test
    public void setCommentAddsMultipleLinesWithMarker() {
        // given
        Map<String, List<String>> comments = new HashMap<>();
        CommentsConfiguration commentsConfig = new CommentsConfiguration(comments);

        String path = "path";
        String[] commentLines = {"comment line 1", "comment line 2"};

        // when
        commentsConfig.setComment(path, commentLines);

        // then
        Map<String, List<String>> allComments = commentsConfig.getAllComments();
        assertThat(allComments.get(path), is(Arrays.asList(commentLines)));
    }
}

public class CommentComparator {
    private final String comment;

    public CommentComparator(String comment) {
        this.comment = comment;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        CommentComparator that = (CommentComparator) o;
        return comment.equals(that.comment);
    }
}

public class CommentsComparator {
    private final String comments;

    public CommentsComparator(String comments) {
        this.comments = comments;
    }

    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        CommentsComparator that = (CommentsComparator) o;
        return comments.equals(that.comments);
    }
}

    @Test
    public void testCommentComparator() {

        CommentComparator comparator2 = new CommentComparator(comment1);

        assertTrue(comparator1.equals(comparator2));
        assertFalse(comparator1.equals(new CommentComparator(comment2)));

    public void testCommentsComparator() {
