        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        assertNotNull(builder);
    }

    @Test
    public void builderBuildUrlTest() throws URISyntaxException, MalformedURLException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        URI uri = builder.build().toURI();
        assertEquals("http://example.com", uri.toString());
    }

    public void appendKeyValAppendKeyValueTest() throws UnsupportedEncodingException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));

        assertEquals(true, StringUtil.hasNonEmptyValue(builder.build()));
    }

    @Test
    public void appendKeyValAppendKeyWithoutValueTest() throws UnsupportedEncodingException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));

        assertEquals(true, StringUtil.hasNonEmptyValue(builder.build()));
    }

    public void appendKeyValAppendValueWithoutKeyTest() throws UnsupportedEncodingException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        Connection.KeyVal kv = null;
        builder.appendKeyVal(kv);
        assertEquals(true, StringUtil.hasNonEmptyValue(builder.build()));
    }

    @Test
    public void appendKeyValAppendUrlTest() throws UnsupportedEncodingException, URISyntaxException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        String url = "https://www.example.com";
        builder.appendKeyVal(Connection.KeyVal("", url));
        assertEquals(url, builder.build().toString());
    }

    @Test
    public void appendKeyValAppendUrlWithEncodingTest() throws UnsupportedEncodingException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        String url = "https://www.example.com";
        String encodedUrl = URLEncoder.encode(url, DataUtil.UTF_8);
        builder.appendKeyVal(Connection.KeyVal("", encodedUrl));
        assertEquals(encodedUrl, builder.build().toString());
    }

    @Test
    public void appendKeyValAppendQuery() throws UnsupportedEncodingException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        String query = "?key=value";
        builder.appendKeyVal(Connection.KeyVal(null, query));
        assertEquals(query, builder.build().query());
    }
