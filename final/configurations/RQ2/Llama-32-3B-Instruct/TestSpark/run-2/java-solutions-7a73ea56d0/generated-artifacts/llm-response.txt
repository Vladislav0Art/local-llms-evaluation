import org.junit.Test;
import static org.junit.Assert.*;

public class Solution {
    public int[] twoSum(int[] nums, int target) {
        int n = nums.length;
        for (int i = 0; i < n - 1; i++) {
            if (nums[i] + nums[i + 1] == target)
                return new int[]{i, i + 1};
        }
        return null;
    }

}

    }

}
