    }

    @Test
    public void parseStringLiteral_InvalidInput() throws ParseException {

    }
}

public class TokenizerTest {

    public static class ExpressionConfiguration {
        private static final String expression = "expression";

        public static String getConfiguration() {
            return expression;
        }
    }

    public static class OperatorDictionaryIfc implements OperatorDictionaryIfc {
        @Override
        public Token getToken(String symbol) {
            return new Token();
        }
    }

    public static interface OperatorDictionaryIfc {
        Token getToken(String symbol);
    }

    public static class FunctionDictionaryIfc implements FunctionDictionaryIfc {
        @Override
        public Token getToken(String symbol) {
            return new Token();
        }
    }

    public static interface FunctionDictionaryIfc {
        Token getToken(String symbol);
    }

    public static class Token {
        private TokenType type;
        private String value;

        public Token() {}

        public Token(TokenType type, String value) {
            this.type = type;
            this.value = value;
        }

        public TokenType getType() {
            return type;
        }

        public String getValue() {
            return value;
        }
    }

    public static class TokenType {
        private final String TYPE;

        public TokenType(String type) {
            TYPE = type;
        }

        public static final String CHARACTER = "character";
        public static final String OPERATOR = "operator";
        public static final String END_OF_EXPRESSION = "endOfExpression";

        public boolean isCharacter() {
            return TYPE.equals(CHARACTER);
        }

        public boolean isOperator() {
            return TYPE.equals(OPERATOR);
        }

        public boolean isEndOfExpression() {
            return TYPE.equals(END_OF_EXPRESSION);
        }
    }

    public static class Tokenizer {
        private final OperatorDictionaryIfc operatorDictionary;

        public Tokenizer(OperatorDictionaryIfc operatorDictionary) {
            this.operatorDictionary = operatorDictionary;
        }

        public List<Token> parse(String input) {
            // implementation of parsing logic
            return new ArrayList<>();
        }
    }

    public static class ParseException extends RuntimeException {}

    public static void throwExpectException(Runnable action, Class<? extends Exception> exceptionType) throws Exception {
        try {
            action.run();
        } catch (Exception e) {
            if (exceptionType.isInstance(e)) {
                throw e;
            } else {
                throw new ParseException("Expected " + exceptionType.getName() + ", got " + e.getClass().getName());
            }
        }
    }

    public static void main(String[] args) {
        OperatorDictionaryIfc operatorDictionary = new OperatorDictionaryIfc();
        Tokenizer tokenizer = new Tokenizer(operatorDictionary);
        List<Token> tokens = tokenizer.parse("input");

        throwExpectException(() -> System.out.println(tokens), ParseException.class);
    }
}
