import org.junit.Test;
import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertFalse;
import static org.junit.Assert.assertTrue;

public class TextNodeTest {

    @Test
    public void nodeName_AnonymousTextNode_ReturnsTagName() {
        TextNode node = new TextNode("");
        assertEquals("text", node.nodeName());
    }

    @Test
    public void text_AnonymousTextNode_ReturnsOwnText() {
        TextNode node = new TextNode("");
        assertEquals("", node.text());
    }

    @Test
    public void text_StringText_ReturnsStringText() {
        String text = "Hello";
        TextNode node = new TextNode(text);
        assertEquals(text, node.text());
    }

    @Test
    public void getWholeText_AnonymousTextNode_ReturnsOwnText() {
        TextNode node = new TextNode("");
        assertEquals("", node.getWholeText());
    }

    @Test
    public void isBlank_AnonymousTextNode_ReturnsTrue() {
        TextNode node = new TextNode("");
        assertTrue(node.isBlank());
    }

    @Test
    public void splitText_AnonymousTextNode_ReturnsOldLength() {
        TextNode node = new TextNode("");
        assertEquals(0, node.splitText(0));
    }

    @Test
    public void outerHtmlHead_AnonymousTextNode_ThrowsIOException() {
        Appendable accum = mock(Appendable.class);
        Document.OutputSettings out = mock(Document.OutputSettings.class);
        when(out.getEncoding()).thenReturn("UTF-8");
        TextNode node = new TextNode("");
        try {
            node.outerHtmlHead(accum, 0, out);
            fail("Expected IOException");
        } catch (IOException e) {
            // Expected
        }
    }

    @Test
    public void outerHtmlTail_AnonymousTextNode_DoesNotThrowAnything() {
        Appendable accum = mock(Appendable.class);
        Document.OutputSettings out = mock(Document.OutputSettings.class);
        TextNode node = new TextNode("");
        assertDoesNotThrow(() -> node.outerHtmlTail(accum, 0, out));
    }

    @Test
    public void toString_AnonymousTextNode_ReturnsTagName() {
        TextNode node = new TextNode("");
        assertEquals("text", node.toString());
    }

    @Test
    public void clone_AnnotatedText returnsSameInstance() {
        TextNode node = new TextNode("Hello");
        TextNode cloned = node.clone();
        assertSame(cloned, node);
    }

    @Test
    public void clone_AnonAnonymousTextNode_returningClonedText() {
        TextNode node = new TextNode("");
        TextNode cloned = node.clone();
        assertEquals("", cloned.text());
    }
}

class Appendable implements org.jsoup.helper.Appendable {}

class Document extends org.jsoup.Document {}
class OutputSettings extends org.jsoup.Document.OutputSettings {}

    private String tag;

    public TextNode(String text) {
        this.tag = "text";
        this.text = text;
    }

    public String nodeName() {
        return tag;
    }

    public String text() {
        return text;
    }

    public String getWholeText() {
        return text;
    }

    public boolean isBlank() {
        return text.isEmpty();
    }

    public int splitText(int index) {
        if (index >= 0 && index < text.length()) {
            int length = text.length() - index;
            String substring = text.substring(index, text.length());
            return length;
        }
        return 0;
    }

    public void outerHtmlHead(Appendable accum, int index, OutputSettings out) throws IOException {
        //Implementation here
    }

    public void outerHtmlTail(Appendable accum, int index, OutputSettings out) {
        //Implementation here
    }

    public String toString() {
        return nodeName();
    }

    public TextNode clone() {
        return new TextNode(text);
    }
}
