import org.junit.Test;
import static org.hamcrest.MatcherAssert.assertThat;
import static org.hamcrest.Matchers.*;
import org.mockito.Mockito;

public class CommentsConfigurationTest {

    @Test
    public void newCommentsConfigurationReturnsEmptyMap() {
        CommentsConfiguration comments = new CommentsConfiguration();
        assertThat(comments.getAllComments(), emptyMap());
    }

    @Test
    public void newCommentsConfigurationWithExistingDataReturnsCorrectData() {
        Map<String, List<String>> existingComments = new HashMap<>();
        existingComments.put("path", Arrays.asList("line1", "line2"));
        CommentsConfiguration comments = new CommentsConfiguration(existingComments);
        assertThat(comments.getAllComments(), equalMapContainingEntry("path", Arrays.asList("line1", "line2")));
    }

    @Test
    public void setCommentUpdatesAllComments() {
        Map<String, List<String>> existingComments = new HashMap<>();
        existingComments.put("path", Arrays.asList("line1", "line2"));
        CommentsConfiguration comments = new CommentsConfiguration(existingComments);
        List<String> commentLines = new ArrayList<>(Arrays.asList("newLine", ""));
        comments.setComment("path", commentLines.toArray(new String[0]));
        assertThat(comments.getAllComments(), equalMapContainingEntry("path", Arrays.asList("newLine", "")));
    }

    @Test
    public void setCommentDoesNotAddEmptyLines() {
        Map<String, List<String>> existingComments = new HashMap<>();
        existingComments.put("path", Arrays.asList("line1", "", "line2"));
        CommentsConfiguration comments = new CommentsConfiguration(existingComments);
        String[] commentLines = {"newLine"};
        comments.setComment("path", commentLines);
        assertThat(comments.getAllComments(), equalMapContainingEntry("path", Arrays.asList("newLine")));
    }

    @Test
    public void setCommentWithNullPathIsInvalid() {
        String[] commentLines = {"newLine"};
        Assertions.assertThrows(NullPointerException.class, () -> new CommentsConfiguration().setComment(null, commentLines));
    }
}

public class SettingsHolder {
    private static final CommentsConfiguration COMMENTS_CONFIGURATION = new CommentsConfiguration();

public class CommentsConfiguration {
    private Map<String, List<String>> comments;

    public CommentsConfiguration() {}

    public Map<String, List<String>> getAllComments() {
        return comments;
    }

    public void setComment(String path, String[] commentLines) {
        if (path == null) {
            throw new NullPointerException();
        }
        comments = new HashMap<>();
        comments.put(path, Arrays.asList(commentLines));
    }
}
