import org.jsoup.helper.W3CDom;
import org.junit.Test;
import static org.junit.Assert.*;

import java.util.HashMap;
import java.util.Map;

public class W3CDomTest {

    @Test
    public void namespaceAwareDefaultIsFalse() {
        boolean result = W3CDom.namespaceAware();
        assertTrue(result == false);
    }

    @Test
    public void namespaceAwareSetToTrue() {
        boolean result = W3CDom.namespaceAware(true);
        assertTrue(result);
    }

    @Test
    public void convertJsoupDoc() {
        org.jsoup.nodes.Document in = new org.jsoup.nodes.Document();
        Document out = W3CDom.convert(in);
        assertNotNull(out);
    }

    @Test
    public void asStringDocWithProperties() {
        Document doc = W3CDom.convert(new org.jsoup.nodes.Document());
        Map<String, String> properties = new HashMap<>();
        properties.put("syntax", Syntax.toString());
        String result = W3CDom.asString(doc, properties);
        assertTrue(result.contains("syntax=\""));
    }

    @Test
    public void asStringDocWithoutProperties() {
        Document doc = W3CDom.convert(new org.jsoup.nodes.Document());
        String result = W3CDom.asString(doc);
        assertTrue(result.contains("<html"));
    }

    @Test
    public void propertiesFromMapConstructor() {
        Map<String, String> map = new HashMap<>();
        map.put("key", "value");
        Properties properties = W3CDom.propertiesFromMap(map);
        assertNotNull(properties);
        assertEquals("key", properties.getProperty("key"));
    }

    @Test
    public void OutputHtml() {
        Map<String, String> result = W3CDom.OutputHtml();
        assertNotNull(result);
        assertTrue(result.containsKey("method"));
        assertTrue(result.get("method").equals(METHOD));
    }

    @Test
    public void OutputXml() {
        Map<String, String> result = W3CDom.OutputXml();
        assertNotNull(result);
        assertTrue(result.containsKey("method"));
        assertTrue(result.get("method").equals(METHOD));
    }

    @Test
    public void fromJsoupDoc() {
        org.jsoup.nodes.Document in = new org.jsoup.nodes.Document();
        Document out = W3CDom.fromJsoup(in);
        assertNotNull(out);
    }

    @Test
    public void fromJsoupElement() {
        org.jsoup.nodes.Element in = new org.jsoup.nodes.Element();
        Document out = W3CDom.fromJsoup(in);
        assertNotNull(out);
    }

    @Test
    public void convertElementDoc() {
        org.jsoup.nodes.Document in = new org.jsoup.nodes.Document();
        org.jsoup.nodes.Element element = in.body().appendNode(new org.jsoup.nodes.Element());
        Document out = W3CDom.convert(element, in);
        assertTrue(out.contains("<html><body><div></div></body></html>"));
    }

    @Test
    public void convertElementToDocument() {
        org.jsoup.nodes.Document in = new org.jsoup.nodes.Document();
        org.jsoup.nodes.Element element = in.body().appendNode(new org.jsoup.nodes.Element());
        Document out = W3CDom.convert(element, null);
        assertNotNull(out);
    }

    @Test
    public void selectXpathDoc() {
        Document doc = W3CDom.convert(new org.jsoup.nodes.Document());
        NodeList result = W3CDom.selectXpath("//div", doc);
        assertEquals(1, result.getLength());
    }

    @Test
    public void selectXpathElement() {
        Document doc = W3CDom.convert(new org.jsoup.nodes.Document());
        NodeList result = W3CDom.selectXpath("//div", doc.body().appendChild(doc.createElement("div")));
        assertEquals(2, result.getLength());
    }

    @Test
    public void sourceNodesNodeList() {
        NodeList nodeList = new ArrayList<>();
        List<Node> result = W3CDom.sourceNodes(nodeList, org.jsoup.nodes.Node.class);
        assertTrue(result.size() == 0);
    }

    @Test
    public void sourceNodesElement() {
        NodeList nodeList = new ArrayList<>();
        nodeList.add(new org.jsoup.nodes.Element());
        List<org.jsoup.nodes.Node> result = W3CDom.sourceNodes(nodeList, org.jsoup.nodes.Node.class);
        assertEquals(1, result.size());
    }

    @Test
    public void contextNodeDoc() {
        Document wDoc = new Document();
        Node result = W3CDom.contextNode(wDoc);
        assertTrue(result instanceof Comment);
    }
}
