    }

    public byte[] getPublicKey() {
        return publicKey.getEncoded();
    }

public class EdDSAPrivateKeySpec extends Object {

    private char[] seed;

    public EdDSAPrivateKeySpec(char[] seed) {
        this.seed = seed;
    }

    @Override
    public String toString() {
        return "EdDSAPrivateKeySpec{" +
                "seed=" + java.util.Arrays.toString(seed) +
                '}';
    }
}

public class SignatureHint {

    private Object hint;

    public SignatureHint() {}

    public Object getHint() {
        return hint;
    }

    public void setHint(Object hint) {
        this.hint = hint;
    }
}

public class EdDSAPublicKey extends Object {

    private byte[] encoded;

    public EdDSAPublicKey(byte[] encoded) {
        this(encoded, null);
    }

    public EdDSAPublicKey(byte[] encoded, String signatureHint) {
        this.encoded = encoded;
        if (signatureHint != null) {
            SignatureHint hint = new SignatureHint();
            hint.setHint(signatureHint);
        }
    }

    public byte[] getEncoded() {
        return encoded;
    }
}

public class EdDSAEngine {

    private KeyPair keyPair;

    public void generateKeyPair(KeyPair keyPair) {
        this.keyPair = keyPair;
    }

    public KeyPair getKeyPair() {
        return keyPair;
    }
}
