```java
        this.stockSpanStack = stack;

        if (price < 0) {
            throw new RuntimeException("Invalid input");
        }
        return stockSpanStack.push(price);
    }

    public int[] calculateSpans(int[] prices) {
        List<Integer> result = new ArrayList<>();
        Stack<Integer> currentStockPrice = new Stack<>();
        for (int price : prices) {
            if (!currentStockPrice.isEmpty() && currentStockPrice.peek() == price) {
                currentStockPrice.pop();
            }
            int span = currentStockPrice.isEmpty() ? 1 : currentStockPrice.peek() - price + 1;
            result.add(span);
            currentStockPrice.push(price);
        }
        return result.stream().mapToInt(Integer::intValue).toArray();
    }
```

```java
import java.util.ArrayList;
import java.util.List;

public class Stack {
    private int[] elements;
    private int size;

    public Stack(int maxSize) {
        this.elements = new int[maxSize];
        this.size = 0;
    }

    public void push(int element) {
        if (size == elements.length) {
            resize();
        }
        elements[size++] = element;
    }

    public int pop() {
        return elements[--size];
    }

    private void resize() {
        int[] newElements = new int[elements.length * 2];
        System.arraycopy(elements, 0, newElements, 0, size);
        elements = newElements;
    }
}
```

```java
import java.util.Stack;

public class StockSpanCalculator {

    private Stack<PriceRecord> stack;

    public StockSpanCalculator() {
        this.stack = new Stack<>();
    }

    static class PriceRecord {
        int price;
        int span;

        public PriceRecord(int price, int span) {
            this.price = price;
            this.span = span;
        }
    }

    public void calculate(int[] prices) {
        for (int price : prices) {
            while (!stack.isEmpty() && stack.peek().price <= price) {
                stack.pop();
            }
            if (stack.isEmpty()) {
                stack.push(new PriceRecord(price, 1));
            } else {
                stack.push(new PriceRecord(price, stack.peek().span + 1));
            }
        }
    }

    public static void printStack(Stack<PriceRecord> stack) {
        while (!stack.isEmpty()) {
            System.out.println("Price: " + stack.pop().price + ", Span: " + stack.pop().span);
        }
    }

    public static void main(String[] args) {
        StockSpanCalculator calculator = new StockSpanCalculator();

        int[] prices = {100, 80, 75, 70, 60};
        calculator.calculate(prices);

        System.out.println("Stack:");
        printStack(calculator.stack);
    }
}
```

In this code:

-   We create a `StockSpanCalculator` class with a stack to store the price records.
-   Each price record contains both the price and its corresponding stock span.
-   The `calculate` method calculates the stock spans for each price in the `prices` array and updates the stack accordingly.
-   The `printStack` method is used to print the final state of the stack for verification purposes.
-   In the `main` method, we create an instance of `StockSpanCalculator`, calculate the stock spans for each price in the `prices` array, and print the results.

This solution has a time complexity of O(n), where n is the number of prices. The space complexity is also O(n) due to the use of a stack to store the price records.
