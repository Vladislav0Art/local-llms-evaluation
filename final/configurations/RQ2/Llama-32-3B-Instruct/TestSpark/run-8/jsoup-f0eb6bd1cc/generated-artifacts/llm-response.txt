import org.jsoup.Connection;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertThrows;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class UrlBuilderTest {

    @Mock
    private Connection connection;

    @Test
    public void constructor_withUrlDoesNotChangeOriginal() {
        // Arrange
        URL url = new URL("http://example.com");
        when(connection.getScheme()).thenReturn("http");

        // Act
        UrlBuilder builder = new UrlBuilder(url);

        // Assert
        assertEquals(url, connection);
    }

    @Test
    public void build_urlWithFragment() {
        // Arrange
        String url = "http://example.com#fragment";
        when(connection.getFragment()).thenReturn("fragment");
        when(connection.getQuery()).thenReturn("key=value");

        UrlBuilder builder = new UrlBuilder(new URL(url));

        // Act
        URL result = builder.build();

        // Assert
        assertEquals("http://example.com#fragment", result.toString());
    }

    @Test
    public void build_urlWithoutFragment() {
        // Arrange
        String url = "http://example.com";
        when(connection.getScheme()).thenReturn("http");
        when(connection.getFragment()).thenReturn(null);
        when(connection.getQuery()).thenReturn("key=value");

        UrlBuilder builder = new UrlBuilder(new URL(url));

        // Act
        URL result = builder.build();

        // Assert
        assertEquals("http://example.com", result.toString());
    }

    @Test
    public void appendKeyVal_withKeyAndValue() {
        // Arrange
        String url = "http://example.com";
        Connection.KeyVal kv = new org.jsoup.internal.StringUtil.KeyVal("key", "value");
        when(connection.getScheme()).thenReturn("http");

        UrlBuilder builder = new UrlBuilder(new URL(url));

        // Act
        builder.appendKeyVal(kv);

        // Assert
        assertEquals("http://example.com?key=value", connection.toString());
    }

    @Test
    public void appendKeyVal_withoutKeyOrValue() {
        // Arrange
        String url = "http://example.com";
        Connection.KeyVal kv = new org.jsoup.internal.StringUtil.KeyVal(null, null);
        when(connection.getScheme()).thenReturn("http");

        UrlBuilder builder = new UrlBuilder(new URL(url));

        // Act
        builder.appendKeyVal(kv);

        // Assert
        assertEquals("http://example.com", connection.toString());
    }

    @Test(expected = UnsupportedEncodingException.class)
    public void appendKeyVal_withUnsupportedEncoding() {
        // Arrange
        String url = "http://example.com";
        Connection.KeyVal kv = new org.jsoup.internal.StringUtil.KeyVal("key", "non-utf8-value");
        when(connection.getScheme()).thenReturn("http");

        UrlBuilder builder = new UrlBuilder(new URL(url));

        // Act
        builder.appendKeyVal(kv);
    }

    @Test
    public void appendQueryWithSpace() {
        // Arrange
        String url = "http://example.com";
        Connection.KeyVal kv = new org.jsoup.internal.StringUtil.KeyVal("key", "value with space");
        when(connection.getScheme()).thenReturn("http");

        UrlBuilder builder = new UrlBuilder(new URL(url));

        // Act
        builder.appendKeyVal(kv);

        // Assert
        assertEquals("http://example.com?key=value%20with%20space", connection.toString());
    }

    @Test
    public void appendQueryWithEmptyValue() {
        // Arrange
        String url = "http://example.com";
        Connection.KeyVal kv = new org.jsoup.internal.StringUtil.KeyVal("key", "");
        when(connection.getScheme()).thenReturn("http");

        UrlBuilder builder = new UrlBuilder(new URL(url));

        // Act
        builder.appendKeyVal(kv);

        // Assert
        assertEquals("http://example.com?key=", connection.toString());
    }
}

public class ConnectionUtil {
    private static final String TEST_URL = "http://example.com";

    public static String getScheme() {
        return java.net.URL.getProtocol().replace(":", "");
    }

    public static String getValue(String key) {
        if (key == null || !key.isEmpty()) {
            return java.net.URL.parse(TEST_URL).getProtocol().replace(":", "");
        } else {
            System.out.println("Key is empty");
            return "";
        }
    }
}

public class TestConnectionUtil {

    @org.junit.Test
    public void testGetScheme() {
        System.out.println(ConnectionUtil.getScheme());
    }

    @org.junit.Test
    public void testGetValue() {
        ConnectionUtil.getValue(null);
    }

    @org.junit.Test
    public void testGetValueNonEmptyKey() {
        ConnectionUtil.getValue("key");
    }
}
