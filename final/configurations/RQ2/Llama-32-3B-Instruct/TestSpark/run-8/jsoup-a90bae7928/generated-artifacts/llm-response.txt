```java
        // Parse the HTML string
        Document doc = Jsoup.parse(htmlString);

        // Get all tokens from the parsed document
        Token[] tokens = getTokens(doc);

        for (Token token : tokens) {
            System.out.println(token);
        }
    }

    public static Token[] getTokens(Document doc) {
        Element rootElement = doc.selectFirst("body").first();
        int index = 0;

        while (index < rootElement.length()) {
            char c = rootElement.charAt(index);

            if (c == '<') {
                // Doctype
                System.out.println("DOCTYPE");
            } else if (Character.isWhitespace(c)) {
                // White space
                System.out.println("WHITE_SPACE");
            } else if (c == '/') {
                // End tag
                System.out.println("END_TAG");
            } else if (Character.isLetter(c) || Character.isDigit(c)) {
                // Tag name
                String tagName = getTagName(rootElement, index);
                System.out.println("TAG_NAME:" + tagName);

                // Get the attributes of the tag
                Attribute[] attributes = getAttributes(rootElement, index);
                for (Attribute attribute : attributes) {
                    System.out.println(attribute);
                }
            } else if (!Character.isWhitespace(c)) {
                // Data
                String data = getData(rootElement, index);
                System.out.println("DATA:" + data);

                // Get the next token
                index++;
            }

            index++;
        }

        return new Token[tokens.length];
    }

    public static String getTagName(Element element, int index) {
        while (index < element.length() && Character.isWhitespace(element.charAt(index))) {
            index++;
        }

        if (index >= element.length()) {
            return null;
        }

        StringBuilder tagName = new StringBuilder();
        while (index < element.length() && !Character.isWhitespace(element.charAt(index))) {
            tagName.append(element.charAt(index));
            index++;
        }

        return tagName.toString();
```

```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class Main {

    public static void main(String[] args) {
        String htmlString = "<html><body>Hello, world!</body></html>";

        // Parse the HTML string
        Document doc = Jsoup.parse(htmlString);

        // Get all tokens from the parsed document
        Token[] tokens = getTokens(doc);

        for (Token token : tokens) {
            System.out.println(token);
        }
    }

    public static Token[] getTokens(Document doc) {
        Token[] tokens = new Token[doc.toString().split("<").length + 1];
        int i = 0;

        // Get the doctype
        if (i < tokens.length - 1 && doc.toString().startsWith("<!DOCTYPE")) {
            tokens[i] = new Token("DOCTYPE", "");
            i++;
        }

        // Get all HTML tags
        while (i < tokens.length) {
            String tag = "<" + doc.selectFirst("body").toString();
            if (tag.isEmpty()) break;

            int index = 0;
            boolean inTag = false;
            Token currentToken = null;
            StringBuilder tagName = new StringBuilder();

            while (index < tag.length()) {
                char c = tag.charAt(index);

                // Check if we are inside a tag
                if (!inTag && Character.isLetter(c) || Character.isDigit(c)) {
                    inTag = true;
                    tagName.append(c);
                    index++;
                }

                // Check if we are at the end of a tag
                else if (inTag && c == '>') {
                    tokens[i] = new Token("TAG_NAME", tagName.toString());
                    tokens[++i] = new Token("ATTRIBUTE", "");
                    i++;

                    // Get all attributes
                    String attributes = doc.selectFirst("body").toString().split(" ")[0];
                    while (!attributes.isEmpty() && Character.isLetter(attributes.charAt(0)) || Character.isDigit(attributes.charAt(0))) {
                        attributes = attributes.substring(1);
                    }

                    if (!attributes.isEmpty()) {
                        tokens[i] = new Token("ATTRIBUTE", "name=\"" + attributes.substring(0, 1) + "\"");
                        i++;
                    }

                    while (!attributes.isEmpty() && (Character.isLetter(attributes.charAt(0)) || Character.isDigit(attributes.charAt(0)))) {
                        attributes = attributes.substring(1);
                    }

                    if (!attributes.isEmpty()) {
                        tokens[i] = new Token("ATTRIBUTE", "value=\"" + attributes + "\"");
                        i++;
                    }

                    // Set the current token
                    currentToken = new Token("DATA", "");
                }

                index++;
            }

            // Get any remaining data
            while (index < tag.length() && !Character.isWhitespace(tag.charAt(index))) {
                currentToken.value += tag.charAt(index);
                index++;
            }

            if (!currentToken.value.isEmpty()) {
                tokens[i] = currentToken;
            } else {
                i++;
            }
        }

        return tokens;
    }
}

class Token {

    private int type;
    private String value;

    public Token(String type, String value) {
        this.type = type;
        this.value = value;
    }

    @Override
    public String toString() {
        return "Token{" +
                "type='" + type + '\'' +
                ", value='" + value + '\'' +
                '}';
    }
}
```

In this example, we define a `Token` class to represent each token in the HTML string. The `getTokens` method is used to get all tokens from the parsed document.
