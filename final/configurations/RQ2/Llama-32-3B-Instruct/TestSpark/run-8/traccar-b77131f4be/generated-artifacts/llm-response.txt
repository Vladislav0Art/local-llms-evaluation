    public GalileoProtocolDecoder(String protocolVersion, String sourceAddress) {
        this.protocolVersion = protocolVersion;
        this.sourceAddress = sourceAddress;
    }

    public static Object decode(MockChannel channel, Long timestamp, String message) throws Exception {
        System.out.println("Received Message: " + message);

        if (!channel.remoteAddress.contains(":")) {
            throw new RuntimeException("Invalid remote address");
        }

        String[] parts = channel.remoteAddress.split(":");
        int portNumber = Integer.parseInt(parts[1]);

        SocketImpl socket = (SocketImpl) channel.getParentClass();
        System.out.println("Source Address: " + socket.getParentClass());

        if (!socket.getParentClass().getAddr().equals(channel.sourceAddress)) {
            throw new RuntimeException("Invalid source address");
        }

        System.out.println("Port Number: " + portNumber);
        System.out.println("Protocol Version: " + protocolVersion);

        // Implement Galileo protocol decoding logic here
        return message;
    }
}

    }
