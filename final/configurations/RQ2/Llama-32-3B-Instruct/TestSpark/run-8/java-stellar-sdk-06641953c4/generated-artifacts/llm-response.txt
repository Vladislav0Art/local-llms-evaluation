import org.junit.Before;
import org.junit.Test;
import static org.junit.Assert.*;
import java.security.GeneralSecurityException;
import net.i2p.crypto.eddsa.EdDSAPrivateKey;
import net.i2p.crypto.eddsa.EdDSAPublicKey;
import net.i2p.crypto.eddsa.KeyPairGenerator;
import net.i2p.crypto.eddsa.spec.EdDSANamedCurveSpec;
import net.i2p.crypto.eddsa.spec.EdDSAPrivateKeySpec;
import net.i2p.crypto.eddsa.spec.EdDSAPublicKeySpec;
import org.stellar.sdk.KeyPair;
import org.stellar.sdk.PublicKey;

public class KeyPairTest {

    @Test
    public void newKeyPairFromPublicShouldNotThrowException() {
        EdDSAPublicKey publicKey = EdDSAPrivateKey.generate().getPublic();
        KeyPair keyPair = KeyPair.fromPublic(publicKey);
        assertNotNull(keyPair);
    }

    @Test
    public void newKeyPairFromSecretSeedShouldReturnKeyPair() throws GeneralSecurityException {
        char[] seed = "seed".toCharArray();
        KeyPair keyPair = KeyPair.fromSecretSeed(seed);
        assertNotNull(keyPair);
    }

    @Test
    public void fromXdrPublicKeyShouldReturnCorrectPublic() {
        PublicKey publicKey = new PublicKey(new byte[]{1, 2, 3});
        EdDSAPublicKey expectedPublicKey = EdDSAPrivateKey.generate().getPublic();
        KeyPair keyPair = KeyPair.fromXdrPublicKey(publicKey);
        assertTrue(Objects.equals(expectedPublicKey, keyPair.getPublic()));
    }

    @Test
    public void fromXdrSignerKeyShouldReturnCorrectSigner() {
        SignerKey signerKey = new SignerKey(new byte[]{1, 2, 3});
        EdDSAPrivateKey expectedPrivateKey = KeyPair.random().getPrivate();
        KeyPair keyPair = KeyPair.fromXdrSignerKey(signerKey);
        assertTrue(Objects.equals(expectedPrivateKey, keyPair.getPrivate()));
    }

    @Test
    public void signShouldReturnDecoratedSignature() {
        byte[] data = new byte[]{1, 2, 3};
        DecoratedSignature signature = KeyPair.random().sign(data);
        assertNotNull(signature);
    }

    @Test
    public void verifyShouldReturnTrueForCorrectSignature() throws GeneralSecurityException {
        byte[] data = new byte[]{1, 2, 3};
        byte[] signature = KeyPair.random().sign(data);
        assertTrue(KeyPair.random().verify(data, signature));
    }

    @Test
    public void verifyShouldThrowExceptionForIncorrectSignature() throws GeneralSecurityException {
        byte[] data = new byte[]{1, 2, 3};
        byte[] signature = new byte[]{4, 5, 6};
        assertThrows(GeneralSecurityException.class, () -> KeyPair.random().verify(data, signature));
    }

    @Test
    public void getAccountIdShouldReturnNull() {
        KeyPair keyPair = KeyPair.random();
        assertNull(keyPair.getAccountId());
    }

    @Test
    public void getSecretSeedShouldReturnEmptyArray() {
        KeyPair keyPair = KeyPair.random();
        assertArrayEquals(new char[0], keyPair.getSecretSeed());
    }

    @Test
    public void getPublicKeyShouldReturnCorrectPublic() {
        byte[] publicKey = new byte[]{1, 2, 3};
        PublicKey expectedPublicKey = new PublicKey(publicKey);
        KeyPair keyPair = KeyPair.fromPublic(expectedPublicKey);
        assertTrue(Objects.equals(expectedPublicKey, keyPair.getPublic()));
    }

    @Test
    public void getSignatureHintShouldReturnNull() {
        KeyPair keyPair = KeyPair.random();
        assertNull(keyPair.getSignatureHint());
    }

    @Test
    public void getXdrPublicKeyShouldReturnCorrectPublic() {
