public class MockAttribute implements Attribute {
    private String key;

    public MockAttribute(String key) {
        this.key = key;
    }

    @Override
    public String getKey() {
        return key;
    }

    @Override
    public String getValue() {
        return "test";
    }
}

public class Element extends org.jsoup.nodes.Element {
    public static Element element(String html, java.util.function.Supplier<String> contentSupplier) {
        org.jsoup.nodes.Element element = org.jsoup.nodes.DocumentBuilderFactory.newInstance().newDocumentBuilder()
                .parse(new java.io.ByteArrayInputStream(html.getBytes())).getElementsByTag("html").first();
        if (element == null) {
            throw new java.lang.NullPointerException("No element with tag name 'html' found");
        }
        return element;
    }

    public static org.jsoup.nodes.Element appendText(String text) {
        this.append(text);
        return this;
    }

    @Override
    public String toString() {
        return "Element{" +
                "key='" + key + '\'' +
                '}';
    }
}

public class Attributes {
    private java.util.ArrayList<MockAttribute> attributeList;

    public Attributes(java.util.ArrayList<MockAttribute> attributeList) {
        this.attributeList = attributeList;
    }

    public int normalize() {
        return attributeList.size();
    }

    public boolean hasKey(String key) {
        for (MockAttribute mockAttribute : attributeList) {
            if (mockAttribute.getKey().equals(key)) {
                return true;
            }
        }
        return false;
    }
}

public class ParseSettings {}

}

public class AttributesTest {

    public static void main(String[] args) {
        TestSpark test = new TestSpark();
        test.testAttributes();
    }
}

class TestSpark {
    public void testAttributes() {
        MockAttribute mockAttribute1 = new MockAttribute("key");
        MockAttribute mockAttribute2 = new MockAttribute("value");

        Attributes attributes = new Attributes(new java.util.ArrayList<MockAttribute>() {{
            add(mockAttribute1);
            add(mockAttribute2);
        }});
        assertEquals(0, attributes.normalize());
        assertFalse(attributes.hasKey("key"));
        assertFalse(attributes.hasKey("value"));

        Document document = Element.element("<html>", () -> "test");
        ParseSettings parseSettings = new ParseSettings();
        attributes = new Attributes(new java.util.ArrayList<MockAttribute>() {{
            add(mockAttribute1);
            add(mockAttribute2);
        }});
        assertEquals(0, attributes.normalize());
        assertFalse(attributes.hasKey("key"));
        assertFalse(attributes.hasKey("value"));

        document = Element.element("<html>", () -> "test");
        parseSettings = new ParseSettings();
        attributes = new Attributes(new java.util.ArrayList<MockAttribute>() {{
            add(mockAttribute1);
            add(mockAttribute2);
        }});
        assertEquals(0, attributes.normalize());
        assertFalse(attributes.hasKey("key"));
        assertFalse(attributes.hasKey("value"));

        MockAttribute mockAttribute3 = new MockAttribute("test");
        document = Element.element("<html>", () -> "test");
        parseSettings = new ParseSettings();
        attributes = new Attributes(new java.util.ArrayList<MockAttribute>() {{
            add(mockAttribute1);
            add(mockAttribute2);
            add(mockAttribute3);
        }});
        assertEquals(0, attributes.normalize());
        assertFalse(attributes.hasKey("key"));
        assertFalse(attributes.hasKey("value"));

        mockAttribute3.setValue("new test");
        document = Element.element("<html>", () -> "test");
        parseSettings = new ParseSettings();
        attributes = new Attributes(new java.util.ArrayList<MockAttribute>() {{
            add(mockAttribute1);
            add(mockAttribute2);
            add(mockAttribute3);
        }});
        assertEquals(0, attributes.normalize());
        assertFalse(attributes.hasKey("key"));
        assertFalse(attributes.hasKey("value"));

        mockAttribute3.setValue("new test");
        document = Element.element("<html>", () -> "test");
        parseSettings = new ParseSettings();
        attributes = new Attributes(new java.util.ArrayList<MockAttribute>() {{
            add(mockAttribute1);
            add(mockAttribute2);
            add(mockAttribute3);
        }});
        assertEquals(0, attributes.normalize());
        assertFalse(attributes.hasKey("key"));
        assertTrue(attributes.hasKey("value"));

    }
}
