public interface Expression {
    void accept(PrettyPrintVisitor visitor);
}

public abstract class Expr implements Expression {}

public class AssingExpr extends Expr {}
public class BinaryExpr extends Expr {}
public class Value extends Expr {}
public class FunctionCallExpr extends Expr {}
public class PropertyPathExpr extends Expr {}
public class AnonymousScript extends Expr {}
public class FunctionScript extends Expr {}
public class StatementSequenceScript extends Expr {}
public class VarExpr extends Expr {}
public class ProjectionExpr extends Expr {}
public class IndexExpr extends Expr {}
public class DecimalExpr extends Expr {}
public class IfElseExpr extends Expr {}

public abstract class PrettyPrintVisitor {
    public int getCount() { return count; }
    protected CollectingPrinter printer = new CollectingPrinter();
    
    public abstract void visit(Expr expr);
}

public class CollectingPrinter implements Printer {
    private int count = 0;
    
    @Override
    public void print(String str) {
        count++;
    }
}

public interface Printer {
    void print(String str);
}
