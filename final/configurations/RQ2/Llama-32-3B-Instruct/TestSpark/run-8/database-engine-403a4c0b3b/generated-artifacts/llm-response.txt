import java.util.HashMap;
import java.util.Map;

public class Database {
    private Map<String, Object> data;

    public Database() {
        this.data = new HashMap<>();
    }

    // Insert a new row into the database
    public void insert(String id, String name) {
        data.put(id, name);
    }

    // Select rows from the database based on an SQL query
    public Iterator<Map.Entry<String, Object>> selectFromTable(String... params) throws Exception {
        for (Map.Entry<String, Object> entry : data.entrySet()) {
            boolean match = true;
            if (params != null && params.length > 0) {
                for (int i = 0; i < params.length; i++) {
                    if (!entry.getKey().equals(params[i])) {
                        match = false;
                        break;
                    }
                }
            } else if (params == null && !data.containsKey(entry.getKey())) {
                match = false;
            }

            if (match) {
                System.out.println("ID: " + entry.getKey() + ", Name: " + entry.getValue());
            }
        }

        return data.entrySet().iterator();
    }

    // Update a row in the database
    public void update(String id, String name) throws Exception {
        if (data.containsKey(id)) {
            data.put(id, name);
        } else {
            throw new Exception("Row not found");
        }
    }

    // Delete a row from the database
    public void delete(String id) throws Exception {
        if (data.containsKey(id)) {
            data.remove(id);
        } else {
            throw new Exception("Row not found");
        }
    }
}

import java.util.ArrayList;
import java.util.HashMap;
import java.util.Iterator;
import java.util.List;
import java.util.Map;

public class Database {
    private List<Map<String, Object>> data;

    public Database() {
        this.data = new ArrayList<>();
    }

    // Add a new row to the database
    public void insert(Map<String, Object> row) throws Exception {
        this.data.add(row);
    }

    // Select rows from the database based on an SQL query
    public Iterator<Map<String, Object>> selectFromTable(List<SQLTerm> sqlTerms, String... params) throws Exception {
        for (Map<String, Object> row : data) {
            boolean match = true;
            if (sqlTerms.size() > 0 && params != null && params.length == sqlTerms.size()) {
                for (int i = 0; i < sqlTerms.size(); i++) {
                    String term = sqlTerms.get(i).term;
                    Object value = params[i];
                    if (!row.containsKey(term) || !row.get(term).equals(value)) {
                        match = false;
                        break;
                    }
                }
            } else if (sqlTerms.size() > 0 && params == null) {
                for (int i = 0; i < sqlTerms.size(); i++) {
                    String term = sqlTerms.get(i).term;
                    if (!row.containsKey(term)) {
                        match = false;
                        break;
                    }
                }
            }

            if (match) {
                System.out.println(row);
            }
        }

        return data.iterator();
    }

    // Update a row in the database
    public void update(Map<String, Object> id, Map<String, Object> updates) throws Exception {
        for (Map.Entry<String, Object> entry : data.iterator()) {
            if (entry.getKey().equals(id)) {
                entry.putAll(updates);
                break;
            }
        }
    }

    // Delete a row from the database
    public void delete(Map<String, Object> id) throws Exception {
        data.remove((Object) id);
    }
}

class SQLTerm {
    String term;
    int pos;

    public SQLTerm(String term, int pos) {
        this.term = term;
        this.pos = pos;
    }
}
