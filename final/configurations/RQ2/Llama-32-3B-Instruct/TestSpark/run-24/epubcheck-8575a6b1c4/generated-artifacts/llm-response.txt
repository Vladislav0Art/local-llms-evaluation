public class EPUBVersion {
    public static final int version1 = 1;

    public static final String format1 = "Format 1";

    public static final String[] supportedFormats = {"Format 1", "Format 2"};

    public static final String[] invalidFormats = {"Invalid Format"};

    public boolean isValid(String format) {
        for (String s : supportedFormats) {
            if (s.equals(format)) {
                return true;
            }
        }
        return false;
    }

    public boolean isInvalid(String format) {
        for (String s : invalidFormats) {
            if (s.equals(format)) {
                return true;
            }
        }
        return false;
    }
}

public class Report {
    private String status;

    public void setStatus(String status) {
        this.status = status;
    }

    public boolean isSuccessful() {
        if (status == null || !status.equalsIgnoreCase("success")) {
            return false;
        }
        return true;
    }

    public String getStatus() {
        return status;
    }
}

public class EpubChecker {
    public Locale getLocale() throws IllegalArgumentException {
        return new Locale("en", "US");
    }

    public int run(String[] args) throws java.io.IOException {
        // method implementation
    }

    public int validateFile(String path, EPUBVersion version, Report report, Map<String, String> options) throws java.io.IOException {
        if (!version.isValid(options.get("format"))) {
            return 1;
        }
        if (report.isSuccessful()) {
            return 0;
        } else {
            return 2;
        }
    }
}
