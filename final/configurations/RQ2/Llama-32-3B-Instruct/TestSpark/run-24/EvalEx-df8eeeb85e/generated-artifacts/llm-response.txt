```java
import java.util.ArrayList;
import java.util.List;

// Enum for token types
public enum TokenType {
    NUMBER,
    PLUS,
    MULTIPLY,
    STRING
}

// ExpressionConfiguration class to hold configuration properties
public class ExpressionConfiguration {
    private static final String NUMBER_REGEX = "\\d+(\\.\\d+)?";
    private static final String PLUS_REGEX = "[+]";
    private static final String MULTIPLY_REGEX = "[*]";
    private static final String STRING_REGEX = "\"[^\"]*\"";

    // Configuration properties here
}

// Tokenizer class to parse expression string into tokens
public class Tokenizer {
    private String expressionString;
    private ExpressionConfiguration configuration;

    public Tokenizer(String expressionString, ExpressionConfiguration configuration) {
        this.expressionString = expressionString;
        this.configuration = configuration;
    }

    /**
     * Parse the expression string into a list of tokens.
     *
     * @return A list of tokens
     * @throws ParseException If an error occurs during parsing
     */
    public List<Token> parse() throws ParseException {
        // Use regular expressions to match token types
        String numberRegex = configuration.NUMBER_REGEX;
        String plusRegex = configuration.PLUS_REGEX;
        String multiplyRegex = configuration.MULTIPLY_REGEX;
        String stringRegex = configuration.STRING_REGEX;

        List<Token> tokens = new ArrayList<>();

        // Try to match the number regex
        if (numberRegex.matches(expressionString)) {
            tokens.add(new Token(TokenType.NUMBER, numberRegex));
        } else {

            // Check for '+' operator
            if (plusRegex.matches(expressionString)) {
                tokens.add(new Token(TokenType.PLUS));
            }

            // Check for '*' operator
            if (multiplyRegex.matches(expressionString)) {
                tokens.add(new Token(TokenType.MULTIPLY));
            }

            // Try to match the string regex
            if (stringRegex.matches(expressionString)) {
                tokens.add(parseStringLiteral());
            }
        }

        return tokens;
    }

    /**
     * Parse a string literal token from the expression string.
     *
     * @return A Token object representing the string literal
     * @throws ParseException If an error occurs during parsing
     */
    private Token parseStringLiteral() throws ParseException {
        int startIndex = expressionString.indexOf('"');
        int endIndex = expressionString.indexOf('"', startIndex + 1);

        String stringLiteral = expressionString.substring(startIndex + 1, endIndex);
        return new Token(TokenType.STRING, stringLiteral);
    }
}

// Token class to represent a token in the parsed list
public class Token {
    private TokenType type;
    private Object value;

    public Token(TokenType type) {
        this.type = type;
    }

    public Token(TokenType type, Object value) {
        this.type = type;
        this.value = value;
    }

    public TokenType getType() {
        return type;
    }

    public Object getValue() {
        if (value == null) {
            throw new ParseException("Missing value for token");
        }
        return value;
    }

    @Override
    public String toString() {
        return "Token{" +
                "type=" + type +
                ", value=" + value +
                '}';
    }
}

// ParseException class to represent an error in parsing
public class ParseException extends Exception {}

```

This tokenizer supports three types of tokens: numbers, '+', '*' operators and string literals. The `parse` method splits the input expression string into parts based on '+' and '*' operators, while the `parseStringLiteral` method extracts a quoted string from the input expression string.

The regular expressions used in this tokenizer are stored in the `ExpressionConfiguration` class. You can modify these expressions to support additional token types or customize their behavior as needed.

```java
public class ExpressionTokenizer {
    private String expressionString;
    private TokenizerConfiguration configuration;

    public ExpressionTokenizer(String expressionString, TokenizerConfiguration configuration) {
        this.expressionString = expressionString;
        this.configuration = configuration;
    }

    public List<Token> tokenize() throws ParseException {
        // Initialize an empty list to store tokens
        List<Token> tokens = new ArrayList<>();

        // Split the input string into individual tokens based on operator precedence
        String[] tokensArray = expressionString.split("\\s+");

        for (String token : tokensArray) {
            if (token.matches("[\\d\\.]+")) {
                // If the token is a number, add it to the list of tokens
                tokens.add(new Token(TokenType.NUMBER, token));
            } else if (token.matches("\\+|-|\\*|/")) {
                // If the token is an operator, add it to the list of tokens
                tokens.add(new Token(TokenType.OPERATOR, token));
            } else if (token.startsWith("\"") && token.endsWith("\"")) {
                // If the token is a string literal, extract its value and add it to the list of tokens
                String value = token.substring(1, token.length() - 1);
                tokens.add(new Token(TokenType.STRING, value));
            } else {
                // If the token cannot be parsed, throw a ParseException
                throw new ParseException("Invalid token: " + token);
            }
        }

        return tokens;
    }
}
```

```java
public static void main(String[] args) {
    String expressionString = "2 + 3 * \"Hello World\" - 1";
    TokenizerConfiguration configuration = new TokenizerConfiguration("[\\d\\.]+");
    try {
        ExpressionTokenizer tokenizer = new ExpressionTokenizer(expressionString, configuration);
        List<Token> tokens = tokenizer.tokenize();
        for (Token token : tokens) {
            System.out.println(token.toString());
        }
    } catch (ParseException e) {
        System.err.println("Error parsing expression: " + e.getMessage());
    }
}
```
