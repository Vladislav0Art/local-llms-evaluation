```java
import java.net.URLDecoder;
import java.util.HashMap;
import java.util.Map;

public class UrlTest {
    public static void main(String[] args) throws Exception {
        String url = "https://example.com/path?a=1&b=2";
        
        // Decode the URL query string
        Map<String, String> queryParams = decodeQueryParams(url);
        System.out.println("Query Parameters: " + queryParams);
        
        // Print the decoded URL path and query parameters
        System.out.println("URL Path: " + new URL(url).getPath());
        System.out.println("URL Query Parameters: " + queryParams);
    }
    
    public static Map<String, String> decodeQueryParams(String url) throws Exception {
        URLDecoder.decode(url, "UTF-8");
        
        // Extract query parameters from the URL
        int queryStringIndex = url.indexOf('?');
        if (queryStringIndex == -1) {
            return new HashMap<>();
        }
        
        String queryString = url.substring(queryStringIndex + 1);
        
        // Split query parameters into individual key-value pairs
        String[] params = queryString.split("&");
        
        Map<String, String> decodedParams = new HashMap<>();
        for (String param : params) {
            int keyValueIndex = param.indexOf("=");
            if (keyValueIndex != -1) {
                String key = param.substring(0, keyValueIndex);
                String value = param.substring(keyValueIndex + 1);
                
                // Decode URL-encoded values
                decodedParams.put(key, URLDecoder.decode(value, "UTF-8"));
            }
        }
        
        return decodedParams;
    }
}
```

This Java solution demonstrates how to decode a URL's query string parameters. It uses the `URLDecoder` class to correctly handle any special characters in the URL-encoded values.

```java
import java.net.URI;
import java.net.URLDecoder;

public class UrlTest {
    public static void main(String[] args) throws Exception {
        String url = "https://example.com/path?a=1&b=2";

        URI uri = new URI(url);
        System.out.println("URL: " + uri.toString());

        String decodedPath = uri.getPath();
        System.out.println("Decoded URL Path: " + decodeUrl(decodedPath));

        String decodedQuery = uri.getQuery();
        if (decodedQuery != null) {
            System.out.println("Decoded URL Query: " + decodeUrl(decodedQuery));
        }

        String decodedFragment = uri.getFragment();
        System.out.println("Decoded URL Fragment: " + decodeUrl(decodedFragment));
    }

    public static String decodeUrl(String encodedString) throws Exception {
        return new String(encodedString.getBytes("UTF-8"), "ISO-8859-1");
    }
}
```
