package net.revelc.code.formatter.css;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;
import org.powermock.core.classloader.annotations.PrepareForTest;
import org.powermock.modules.junit4.PowerMockRunner;

import java.io.IOException;
import java.io.StringReader;

import net.revelc.code.formatter.css.CssFormatter;
import net.revelc.code.formatter.css.ConfigurationSource;
import net.revelc.code.formatter.lineending.LineEnding;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertFalse;
import static org.junit.Assert.assertTrue;
import static org.mockito.Mockito.when;

@RunWith(PowerMockRunner.class)
@PrepareForTest(CssFormatter.class)
public class CssFormatterTest {

    @Mock
    private ConfigurationSource cfg;

    @Mock
    private LineEnding lineEnding;

    @Mock
    private CSSStyleSheetImpl cssStyleSheetImpl;

    @Mock
    private CSSOMParser cssOMParser;

    private CssFormatter formatter;

    public void setUp() {
        MockitoAnnotations.initMocks(this);
        formatter = new CssFormatter(cfg, cssOMParser, cssStyleSheetImpl);
    }

    public void setLineEnding(LineEnding lineEnding) {
        this.lineEnding = lineEnding;
    }

    public void setCssOMParser(CSSOMParser cssOMParser) {
        this.cssOMParser = cssOMParser;
    }

    public void setCssStyleSheetImpl(CSSStyleSheetImpl cssStyleSheetImpl) {
        this.cssStyleSheetImpl = cssStyleSheetImpl;
    }

    @Test
    public void init(Map<String, String> options, ConfigurationSource cfg) {
        formatter.init(options, cfg);
    }

    @Test
    public void doFormat(String code, LineEnding ending) throws IOException {
        when(lineEnding.getEnding()).thenReturn(ending.getEnding());
        when(cssStyleSheetImpl.toString()).thenReturn(code);
        String formattedCode = formatter.doFormat(code, lineEnding);
        assertEquals(code, formattedCode);
    }

    @Test
    public void isInitialized() {
        assertTrue(formatter.isInitialized());
    }

    @Test
    public void init_WithOptions(Map<String, String> options) {
        formatter.init(options, cfg);
    }

    @Test
    public void doFormat_WithLineEnding(LineEnding ending) throws IOException {
        when(lineEnding.getEnding()).thenReturn(ending.getEnding());
        String formattedCode = formatter.doFormat(code, lineEnding);
        assertEquals(code, formattedCode);
    }

    @Test
    public void doFormat_WithEmptyCode() throws IOException {
        String formattedCode = formatter.doFormat("", lineEnding);
        assertTrue(formattedCode.isEmpty());
    }

    @Test
    public void doFormat_WithDifferentLineEndings() throws IOException {
        LineEnding ending1 = new LineEnding();
        LineEnding ending2 = new LineEnding("\r\n");
        when(lineEnding.getEnding()).thenReturn(ending1.getEnding());
        String formattedCode = formatter.doFormat(code, lineEnding);
        assertEquals(code, formattedCode);

        when(lineEnding.getEnding()).thenReturn(ending2.getEnding());
        formattedCode = formatter.doFormat(code, lineEnding);
        assertEquals(code, formattedCode);
    }

    @Test
    public void doFormat_WithIOException() throws IOException {
        try {
            formatter.doFormat(code, lineEnding);
            assert false;
        } catch (IOException e) {
            assertTrue(true);
        }
    }

    private String code = "body { background-color: #f2f2f2; }";
}
