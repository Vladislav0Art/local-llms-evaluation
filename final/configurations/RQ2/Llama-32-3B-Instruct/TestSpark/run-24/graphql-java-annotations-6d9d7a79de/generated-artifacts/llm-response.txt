package graphql.annotations.processor.retrievers.fieldBuilders;

import graphql.annotations.annotationTypes.directives.activation.GraphQLDirectives;
import graphql.annotations.processor.ProcessingElementsContainer;
import graphql.annotations.processor.exceptions.GraphQLAnnotationsException;
import graphql.annotations.processor.util.DirectiveJavaAnnotationUtil;
import graphql.schema.GraphQLArgument;
import graphql.schema.GraphQLDirective;
import graphql.schema.GraphQLScalarType;
import graphql.schema.GraphQLType;
import java.lang.annotation.Annotation;
import java.lang.reflect.AnnotatedElement;
import java.lang.reflect.Method;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.stream.Collectors;
import static graphql.schema.GraphQLDirective.newDirective;

public class DirectivesBuilderTest {

    public void testDirectivesBuilderConstructor() {
        AnnotatedElement element = mock(AnnotatedElement.class);
        ProcessingElementsContainer container = mock(ProcessingElementsContainer.class);

        DirectivesBuilder builder = new DirectivesBuilder(element, container);

        verify(builder).setObject(element);
        verify(builder).setContainer(container);
    }

    public void testDirectivesBuilderBuild() {
        AnnotatedElement element = mock(AnnotatedElement.class);
        ProcessingElementsContainer container = mock(ProcessingElementsContainer.class);

        DirectivesBuilder builder = new DirectivesBuilder(element, container);

        List<GraphQLDirective> directives = Arrays.asList(
                GraphQLDirectives.ABSTRACT,
                GraphQLDirectives.EXPORTS
        );

        when(container.getAnnotatedTypes()).thenReturn(Arrays.asList(new GraphQLType(), new GraphQLScalarType()));
        when(builder.getObject()).thenReturn(element);
        when(element.getType()).thenReturn(ProcessingElementsContainer.DEFAULT_TYPE);

        List<GraphQLDirective> result = builder.build();

        verify(result).containsExactly(directives);
    }

    public void testDirectivesBuilderBuildEmpty() {
        AnnotatedElement element = mock(AnnotatedElement.class);
        ProcessingElementsContainer container = mock(ProcessingElementsContainer.class);

        DirectivesBuilder builder = new DirectivesBuilder(element, container);

        List<GraphQLDirective> result = builder.build();

        verify(result).isEmpty();
    }

    public void testDirectivesBuilderBuildNoDirectives() {
        AnnotatedElement element = mock(AnnotatedElement.class);
        ProcessingElementsContainer container = mock(ProcessingElementsContainer.class);

        DirectivesBuilder builder = new DirectivesBuilder(element, container);

        List<GraphQLDirective> result = builder.build();

        verify(result).isEmpty();
    }

    public void testDirectivesBuilderBuildException() {
        AnnotatedElement element = mock(AnnotatedElement.class);
        ProcessingElementsContainer container = mock(ProcessingElementsContainer.class);

        when(container.getAnnotatedTypes()).thenReturn(Arrays.asList(new GraphQLType(), new GraphQLScalarType()));
        when(builder.getObject()).thenReturn(element);
        when(element.getType()).thenReturn(null);

        try {
            builder.build();
            fail("Expected exception of type GraphQLAnnotationsException");
        } catch (GraphQLAnnotationsException e) {
            verify(e).getMessage().isEqualTo("Missing annotated type for element of type null");
        }
    }

    public void testDirectivesBuilderBuildTypeNotSupported() {
        AnnotatedElement element = mock(AnnotatedElement.class);
        ProcessingElementsContainer container = mock(ProcessingElementsContainer.class);

        DirectivesBuilder builder = new DirectivesBuilder(element, container);

        when(container.getAnnotatedTypes()).thenReturn(Arrays.asList(new GraphQLType(), new GraphQLScalarType()));
        when(builder.getObject()).thenReturn(element);
        when(element.getType()).thenReturn(new GraphQLType() {
            @Override
            public Class<? extends Annotation> annotationType() { return null; }
        });

        List<GraphQLDirective> result = builder.build();

        verify(result).isEmpty();
    }

    public void testDirectivesBuilderBuildUnsupportedMethod() {
        AnnotatedElement element = mock(AnnotatedElement.class);
        ProcessingElementsContainer container = mock(ProcessingElementsContainer.class);

        DirectivesBuilder builder = new DirectivesBuilder(element, container);

        when(container.getAnnotatedTypes()).thenReturn(Arrays.asList(new GraphQLType(), new GraphQLScalarType()));
        when(builder.getObject()).thenReturn(element);
        when(element.getType()).thenReturn(new GraphQLType() {
            @Override
            public Class<? extends Annotation> annotationType() { return null; }
        });

        try {
            builder.build();
            fail("Expected exception of type UnsupportedOperationException");
        } catch (UnsupportedOperationException e) {
            verify(e).getMessage().contains("Method not supported for type graphql.Type");
        }
    }

}
