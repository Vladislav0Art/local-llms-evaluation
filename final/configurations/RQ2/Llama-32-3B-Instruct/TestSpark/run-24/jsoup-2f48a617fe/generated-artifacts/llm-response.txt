import org.junit.Test;
import org.jsoup.nodes.Comment;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.parser.Parser;
import org.jsoup.parser.ParseSettings;

import java.io.ByteArrayOutputStream;
import java.io.IOException;

public class CommentTest {

    @Test
    public void newNodeComment()[Test] {
        Comment comment = new Comment("");
        assert comment != null;
    }

    @Test
    public void nodeNameIsComment()[Test] {
        Comment comment = new Comment("");
        assertEquals("comment", comment.nodeName());
    }

    @Test
    public void getDataReturnsString()[Test] {
        Comment comment = new Comment("data");
        assertEquals("data", comment.getData());
    }

    @Test
    public void setDataSetsNewData()[Test] {
        Comment comment = new Comment("");
        comment.setData("new_data");
        assertEquals("new_data", comment.getData());
    }

    @Test
    public void outerHtmlHeadHandlesEmptyString()[Test] {
        ByteArrayOutputStream output = new ByteArrayOutputStream();
        Appendable accum = output;
        int depth = 0;
        Document.OutputSettings out = null;
        Comment comment = new Comment("");
        try {
            comment.outerHtmlHead(accum, depth, out);
            assert !output.toString().isEmpty();
        } catch (IOException e) {
            throw new RuntimeException(e);
        }
    }

    @Test
    public void outerHtmlTailHandlesEmptyString()[Test] {
        ByteArrayOutputStream output = new ByteArrayOutputStream();
        Appendable accum = output;
        int depth = 0;
        Document.OutputSettings out = null;
        Comment comment = new Comment("");
        try {
            comment.outerHtmlTail(accum, depth, out);
            assert !output.toString().isEmpty();
        } catch (IOException e) {
            throw new RuntimeException(e);
        }
    }

    @Test
    public void cloneCreatesNewComment()[Test] {
        Comment original = new Comment("data");
        Comment clone = original.clone();
        assertNotNull(clone);
        assertEquals(original.getData(), clone.getData());
    }

    @Test
    public void isXmlDeclarationReturnsFalse()[Test] {
        Comment comment = new Comment("");
        assertFalse(comment.isXmlDeclaration());
    }

    @Test
    public void asXmlDeclarationReturnsNull()[Test] {
        Comment comment = new Comment("");
        assert comment.asXmlDeclaration() == null;
    }
}
