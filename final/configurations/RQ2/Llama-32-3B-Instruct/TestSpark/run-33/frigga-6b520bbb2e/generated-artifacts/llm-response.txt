public class AppVersionTest {
    @Test
    public void testEquals() {
        AppVersion appVersion = new AppVersion("0.9.1-EC2-X86_64-GP2");
        AppVersion anotherAppVersion = new AppVersion("0.9.1-EC2-X86_64-GP2");

        assertTrue(appVersion.equals(anotherAppVersion));
    }

    @Test
    public void testNotEquals() {
        AppVersion appVersion = new AppVersion("0.9.1-EC2-X86_64-GP2");
        AppVersion anotherAppVersion = new AppVersion("0.10.0-EC2-X86_64-GP2");

        assertFalse(appVersion.equals(anotherAppVersion));
    }

    @Test
    public void testNull() {
        AppVersion appVersion = null;

        assertThrows(NullPointerException.class, () -> appVersion.equals(null));
    }
}

public class AppVersion {
    private String version;

    public AppVersion(String version) {
        this.version = version;
    }

    public boolean equals(AppVersion other) {
        return this.version.equals(other.version);
    }

    @Override
    public String toString() {
        return "AppVersion{" +
                "version='" + version + '\'' +
                '}';
    }
}
