package ch.jalu.configme.configurationdata;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;
import org.junit.runners.JUnit4;

import static org.hamcrest.CoreMatchers.is;
import static org.junit.Assert.assertThat;
import static org.mockito.Mockito.when;

@RunWith(JUnit4.class)
public class CommentsConfigurationTest {

    private CommentsConfiguration commentsConfiguration = new CommentsConfiguration();

    @Mock
    private SettingsHolder settingsHolder;

    @Before
    public void setup() {
        MockitoAnnotations.initMocks(this);
    }

    @Test
    public void newCommentsConfigurationIsEmptyMap() {
        assertThat(commentsConfiguration.getAllComments(), is(Collections.emptyMap()));
    }

    @Test
    public void setCommentSingleLine() {
        commentsConfiguration.setComment("path", "line");
        assertThat(commentsConfiguration.getAllComments().get("path"), is(Collections.singletonList("line")));
    }

    @Test
    public void setCommentMultipleLines() {
        commentsConfiguration.setComment("path", "line1", "line2");
        assertThat(commentsConfiguration.getAllComments().get("path"), is(Arrays.asList("line1", "line2")));
    }

    @Test
    public void setCommentEmptyLine() {
        commentsConfiguration.setComment("path", "");
        assertThat(commentsConfiguration.getAllComments().get("path"), is(Collections.singletonList("")));
    }

    @Test
    public void setCommentMultiplePaths() {
        commentsConfiguration.setComment("path1", "line1");
        commentsConfiguration.setComment("path2", "line2");
        assertThat(commentsConfiguration.getAllComments(), is(Collections.singletonMap("path1", Collections.singletonList("line1")), Collections.singletonMap("path2", Collections.singletonList("line2"))));
    }

    @Test
    public void setCommentOverwriteExistingPath() {
        commentsConfiguration.setComment("path", "line1");
        commentsConfiguration.setComment("path", "line2");
        assertThat(commentsConfiguration.getAllComments().get("path"), is(Collections.singletonList("line2")));
    }
}

package ch.jalu.configme.configurationdata;

import org.junit.Test;
import static org.junit.Assert.*;
import java.util.Collections;
import java.util.Arrays;

public class CommentsTest {

    @Test
    public void testGetAllComments() {
        Comments comments = new Comments();
        assertTrue(comments.getAllComments().isEmpty());
    }

    @Test
    public void testAddComment() {
        Comments comments = new Comments();
        comments.add("path1", "line1");
        assertEquals(1, comments.getAllComments().size());
    }

    @Test
    public void testGetComment() {
        Comments comments = new Comments();
        comments.add("path1", "line1");
        assertEquals("line1", comments.get("path1"));
    }
}

class Comments {
    private java.util.Map<String, String> map = new java.util.HashMap<>();

    public void add(String path, String line) {
        map.put(path, line);
    }

    public java.util.Set<String> getAllComments() {
        return map.keySet();
    }

    public String get(String path) {
        return map.get(path);
    }
}
