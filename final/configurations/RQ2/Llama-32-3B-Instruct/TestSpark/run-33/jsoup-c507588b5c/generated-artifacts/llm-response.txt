import org.jsoup.nodes.Document;
import org.jsoup.nodes.TextNode;
import org.jsoup.nodes.Element;
import org.jsoup.helper.Validate;
import org.junit.Test;
import static org.junit.Assert.*;
import org.mockito.Mockito;

public class TextNodeTest {

    @Test
    public void newNodeTextIsInitialisedCorrectly() {
        // Arrange and Act
        TextNode textNode = new TextNode("Hello World");
        
        // Assert
        assertEquals("Hello World", textNode.text());
    }

    @Test
    public void nodeNameReturnsCorrectValue() {
        // Arrange and Act
        TextNode textNode = new TextNode("Example Node");
        String result = textNode.nodeName();
        
        // Assert
        assertEquals("text_node", result);
    }

    @Test
    public void cloneCreatesNewTextNode() {
        // Arrange
        TextNode original = Mockito.mock(TextNode.class);

        // Act and Assert
        TextNode cloned = new TextNode("Hello Clone");
        assertTrue(cloned.equals(original));
        assertNotSame(original, cloned);
    }

    @Test
    public void cloneReturnsCorrectValueWhenNewTextNodeHasDifferentText() {
        // Arrange
        TextNode original = new TextNode("Original Node");
        
        // Act and Assert
        TextNode cloned = new TextNode("Cloned Node");
        assertFalse(cloned.equals(original));
    }

    @Test
    public void getWholeTextReturnsCorrectValue() {
        // Arrange and Act
        TextNode textNode = new TextNode("Hello World");
        String result = textNode.getWholeText();
        
        // Assert
        assertEquals("Hello World", result);
    }

    @Test
    public void isBlankReturnsTrueWhenTextNodeIsEmpty() {
        // Arrange and Act
        TextNode textNode = new TextNode("");
        assertTrue(textNode.isBlank());
    }

    @Test
    public void isBlankReturnsFalseWhenTextNodeIsNotEmpty() {
        // Arrange and Act
        TextNode textNode = new TextNode("Hello");
        assertFalse(textNode.isBlank());
    }
}
