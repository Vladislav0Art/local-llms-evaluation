public class CssFormatterTest {

    public void init_FailsWhenOptionsAreEmpty() {
        Map<String, String> options = new HashMap<>();
        ConfigurationSource cfg = mock(ConfigurationSource.class);
        CssFormatter formatter = new CssFormatter();
        assertFalse(formatter.init(options, cfg));
    }

    @Test
    public void init_SucceedsWhenOptionsExist() {
        Map<String, String> options = new HashMap<>();
        options.put("option", "value");
        ConfigurationSource cfg = mock(ConfigurationSource.class);
        CssFormatter formatter = new CssFormatter();
        assertTrue(formatter.init(options, cfg));
    }

    @Test
    public void doFormat_NullCode() throws IOException {
        assertThrows(IOException.class, () -> new CssFormatter().doFormat(null, LineEnding.LF));
    }

    @Test
    public void doFormat_EmptyString() throws IOException {
        String code = "";
        assertThrows(IOException.class, () -> new CssFormatter().doFormat(code, LineEnding.LF));
    }

    @Test
    public void doFormat_SimpleCode_LF() throws IOException {
        String code = ".class { }";
        String expected = " .class { }\n";
        String actual = new CssFormatter().doFormat(code, LineEnding.LF);
        assertEquals(expected, actual);
    }

    @Test
    public void doFormat_MultiLineCode_LF() throws IOException {
        String code = ".class {\n  color: red;\n}\n";
        String expected = " .class {\n  color: red;\n} \n";
        String actual = new CssFormatter().doFormat(code, LineEnding.LF);
        assertEquals(expected, actual);
    }

    @Test
    public void doFormat_SimpleCode_CRLF() throws IOException {
        String code = ".class { }";
        String expected = " .class { }\r\n";

    }

    public void doFormat_MultiLineCode_CRLF() throws IOException {
        String code = ".class {\n  color: red;\n}\n";
        String expected = " .class {\n  color: red;\n} \r\n";
        String actual = new CssFormatter().doFormat(code, LineEnding.CRLF);
        assertEquals(expected, actual);
    }

    public void isInitialized_IsFalse() {

class LineEnding {
    public static final String LF = "\n";
    public static final String CRLF = "\n\r";

    public enum Value { LF, CRLF };

    public String getValue() {
        switch (this) {
            case LF:
                return LF;
            case CRLF:
                return CRLF;
            default:
                throw new RuntimeException();
        }
    }
}

class CssFormatter {

    private LineEnding lineEnding;

    public void init(Map<String, String> options, ConfigurationSource configurationSource) {
        this.lineEnding = LineEnding.Value.LF;
    }

    public String doFormat(String code, String lineEnding) {
        return code.replace("\r\n", lineEnding);
    }
}

class ConfigurationSource {

    private String css;

    public String getCss() {
        return css;
    }

    public void setCss(String css) {
        this.css = css;
    }
}
