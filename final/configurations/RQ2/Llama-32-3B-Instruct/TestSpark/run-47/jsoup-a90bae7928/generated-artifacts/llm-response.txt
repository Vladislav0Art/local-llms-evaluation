public class Token {
    public enum Type {
        START_TAG,
        END_TAG,
        CHARACTER
    }

    private Type type;
    private Character character;
    private String tagName;

    public Token(Type type, Object value) {
        if (type == Token.Type.START_TAG) {
            if (value instanceof String) {
                this.tagName = (String) value;
            } else if (value instanceof Token.TokenStartTag) {
                this.tagName = ((Token.TokenStartTag) value).tagName;
            }
        } else if (type == Token.Type.END_TAG) {
            if (value instanceof Token.TokenEndTag) {
                Token.TokenEndTag tag = (Token.TokenEndTag) value;
                if (tag.tagName != null) {
                    this.tagName = tag.tagName;
                }
            }
        } else if (type == Token.Type.CHARACTER) {
            if (value instanceof Character) {
                this.character = (Character) value;
            }
        }
    }

    public Type getType() {
        return type;
    }

    public Character getCharacter() {
        return character;
    }

    public String getTagName() {
        return tagName;
    }
}
