import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class JsoupOutput {
    public static String html() {
        return "<html><body>Hello World</body></html>";
    }

    public static void main(String[] args) {
        Document document = Jsoup.parse(html());
        Element element = document.body();
        System.out.println(element.text());

        // Test 1
        assert element != null;
        System.out.println("Test 1: element not null");

        // Test 2
        String htmlOutput = element.html();
        System.out.println("HTML Output:");
        System.out.println(htmlOutput);

        // Test 3
        String textOutput = element.text();
        System.out.println("\nText Output:");
        System.out.println(textOutput);

        // Test 4
        Elements links = document.select("a");
        for (Element link : links) {
            System.out.println(link.attr("href"));
        }
    }

    public static void xml() {
        String xml = "<root><child>Hello World</child></root>";
        return xml;
    }

    public static class OutputHtml {
        private String output;

        public OutputHtml(String output) {
            this.output = output;
        }

        @Override
        public String toString() {
            return "OutputHtml{" +
                    "output='" + output + '\'' +
                    '}';
        }
    }

    public static OutputHtml htmlOutput() {
        OutputHtml outputHtml = new OutputHtml(html());
        return outputHtml;
    }

    public static void main(String[] args) {
        OutputHtml outputHtml = htmlOutput();
        System.out.println(outputHtml);

        // Test 1
        assert outputHtml != null;
        System.out.println("Test 1: output not null");

        // Test 2
        String outputString = outputHtml.toString();
        System.out.println("\nOutput as string:");
        System.out.println(outputString);
    }

    public static class OutputXml {
        private String output;

        public OutputXml(String output) {
            this.output = output;
        }

        @Override
        public String toString() {
            return "OutputXml{" +
                    "output='" + output + '\'' +
                    '}';
        }
    }

    public static OutputXml xmlOutput() {
        OutputXml outputXml = new OutputXml(xml());
        return outputXml;
    }

    public static void main(String[] args) {

        System.out.println("Test 1: output not null");
