import org.jsoup.Jsoup;

public class JsoupTest {

    @Test
    public void testGet() {
        String url = "http://example.com";
        Document document = Jsoup.connect(url).get();
        System.out.println(document.html());
    }

    @Test
    public void testPost() throws Exception {
        String url = "http://example.com";
        StringBuilder data = new StringBuilder("key=value");
        String[] headers = {"Content-Type: application/x-www-form-urlencoded"};
        Request request = new Connection.Request(url, method -> method.post(), data.toString(), "application/x-www-form-urlencoded", headers);
        Document document = (Document) request.execute();
        System.out.println(document.html());
    }
}

public class Connection {

    public static class Request {
        private String url;
        private Method method;
        private StringBuilder data;
        private String headers;

        public Request(String url, Method method, String data, String headers) {
            this.url = url;
            this.method = method;
            this.data = new StringBuilder(data);
            this.headers = headers;
        }

        public Document execute() throws Exception {
            return Jsoup.connect(url).method(method).data(data.toString()).headers(headers).get();
        }
    }

    @FunctionalInterface
    public interface Method {
        String get();

        default String post() {
            return "POST";
        }
    }

    public static class Document {
        private String html;

        public String getHtml() {
            return html;
        }

        public void setHtml(String html) {
            this.html = html;
        }
    }

public class Jsoup {

    public static class Connection {
        private String url;

        public Connection(String url) {
            this.url = url;
        }

        public Jsoup.Request request(Method method, String data, String headers) {
            return new Jsoup.Request(url, (requestBuilder) -> {
                requestBuilder.method(method);
                return requestBuilder;
            }, data, "application/x-www-form-urlencoded", headers);
        }
    }

    @FunctionalInterface
    public interface Method {
        default void get() {}
        default void post() {
            System.out.println("POST");
        }
    }

    public static class Request {
        private String url;
        private Method method;
        private String data;
        private String headers;

        public Request(String url, Method method, String data, String headers) {
            this.url = url;
            this.method = method;
            this.data = data;
            this.headers = headers;
        }

        public Document execute() throws Exception {
            return new Connection(url).request(method, data, headers).get();
        }
    }

    public static class Document {
        private String html;

        public Document(String html) {
            this.html = html;
        }

        public String getHtml() {
            return html;
        }
    }

    public static Document connect(String url) throws Exception {
        return new Document("html");
    }
}

public class Test {

    @Test
    public void testGet() throws Exception {
        String url = "http://example.com";

        System.out.println(document.getHtml());
    }

    @Test
    public void testPost() throws Exception {

        StringBuilder data = new StringBuilder("key=value");
        Method method = Jsoup.Method.post();
        Jsoup.Document document = new Jsoup.Connection(url).request(method, data.toString(), "").execute();
        System.out.println(document.getHtml());
    }
}
