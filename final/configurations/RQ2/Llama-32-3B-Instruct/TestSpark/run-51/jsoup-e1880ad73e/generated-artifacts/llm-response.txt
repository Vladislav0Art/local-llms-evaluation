import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import static org.junit.Assert.*;
import static org.jsoup.helper.DataUtil.UTF_8;
import static org.jsoup.helper.UrlBuilder.*;

@RunWith(MockitoJUnitRunner.class)
public class UrlBuilderTest {

    @Mock
    private Connection connection;

    public UrlBuilder buildUrl() {
        return new UrlBuilder(connection);
    }

    @Test
    public void appendKeyVal() throws UnsupportedEncodingException {
        String expected = "key=value";
        KeyVal kv = new KeyVal("key", "value");
        appendKeyVal(kv, UTF_8);
        assertEquals(expected, encodedKeyVal(kv));
    }

    @Test
    public void buildUrlWithInvalidProtocol() {
        try {
            build();
            fail("Expected MalformedURLException");
        } catch (MalformedURLException e) {
            // expected
        }
    }

    @Test
    public void buildUrlWithInvalidAuthority() {
        String invalidAuthority = "invalid:authority";
        try {
            connection.setAuthority(invalidAuthority);
            build();
            fail("Expected URISyntaxException");
        } catch (URISyntaxException e) {
            // expected
        }
    }

    private static String encodedKeyVal(KeyVal kv) throws UnsupportedEncodingException {
        StringBuilder sb = new StringBuilder(kv.key());
        if (! StringUtil.isEmpty(kv.value())) {
            sb.append("=").append(URLEncoder.encode(kv.value(), UTF_8));
        }
        return sb.toString();
    }
}
