import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;
import org.jsoup.Connection;
import org.jsoup.internal.StringUtil;
import org.jsoup.helper.DataUtil;
import org.junit.runners.JUnit4;
import static org.junit.Assert.assertEquals;

@RunWith(JUnit4.class)
public class UrlBuilderTest {

    @Mock
    private Connection connection;

    @Mock
    private URL url;

    public UrlBuilderTest() {
        MockitoAnnotations.initMocks(this);
    }

    @Test
    public void constructor_NoInputUrl_ReturnsNull() {
        UrlBuilder builder = new UrlBuilder(null);
        assertNull(builder.getUrl());
    }

    @Test
    public void constructor_InputUrl_ReturnsBuilderWithUrl() {
        URL inputUrl = new URL("http://example.com");
        UrlBuilder builder = new UrlBuilder(inputUrl);
        assertNotNull(builder.getUrl());
        assertEquals(inputUrl, builder.getUrl());
    }

    @Test
    public void build_NoInputUrl_ReturnsNull() {
        UrlBuilder builder = new UrlBuilder(null);
        assertNull(builder.build());
    }

    @Test
    public void build_InputUrl_ReturnsConstructedUrl() throws MalformedURLException {
        URL inputUrl = new URL("http://example.com");
        UrlBuilder builder = new UrlBuilder(inputUrl);
        URL constructedUrl = builder.build();
        assertNotSame(constructedUrl, inputUrl);
        assertEquals("http://example.com", constructedUrl.toString());
    }

    @Test
    public void appendKeyVal_AddsKeyValuePairToUrl() throws UnsupportedEncodingException {
        Connection.KeyVal kv = new Connection.KeyVal("key", "value");
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        builder.appendKeyVal(kv);
        assertEquals("http://example.com?key=value", builder.build().toString());
    }

    @Test
    public void appendKeyVal_MultipleKeyValuePairsAddToUrl() throws UnsupportedEncodingException {
        Connection.KeyVal kv1 = new Connection.KeyVal("key1", "value1");
        Connection.KeyVal kv2 = new Connection.KeyVal("key2", "value2");
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        builder.appendKeyVal(kv1);
        builder.appendKeyVal(kv2);
        assertEquals("http://example.com?key1=value1&key2=value2", builder.build().toString());
    }

}

        UrlBuilder builder = new UrlBuilder();
        builder.appendKeyVal("key", "value");
        assertEquals("key=value", builder.getUrl());
    }

    @Test
    public void appendKeyVal_MultipleKeyValuePairsAddToUrl() throws Exception {
        UrlBuilder builder = new UrlBuilder();
        builder.appendKeyVal("key1", "value1");
        builder.appendKeyVal("key2", "value2");
        assertEquals("key1=value1&key2=value2", builder.getUrl());
    }

    @Test
    public void getUrl() throws Exception {
        UrlBuilder builder = new UrlBuilder();
        URL inputUrl = new URL("http://example.com");
        builder.setBaseUrl(inputUrl);
        assertNotNull(builder.getUrl());
        assertEquals("http://example.com", builder.getUrl());
    }
}

public class UrlBuilder {

    private String baseUrl;

    public void appendKeyVal(String key, String value) {
        if (baseUrl == null) {
            baseUrl = "http://example.com";
        }
        if (!baseUrl.endsWith("?")) {
            baseUrl += "?";
        }
        baseUrl += key + "=" + value;
    }

    public URL getUrl() {
        return new URL(baseUrl);
    }

    private void setBaseUrl(URL base) {
        this.baseUrl = base.toString();
    }
}

public class Connection {

    public static abstract class KeyVal<K, V> {
        public K getKey();
        public V getValue();

        public KeyVal(K key, V value) {
            this.getKey = key;
            this.getValue = value;
        }

        public String toString() {
            return getKey() + "=" + getValue();
        }
    }

}
