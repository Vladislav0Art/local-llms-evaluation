    }

    public void isBlock_BlockTags_ReturnsTrue() {
        Tag tag = new Tag("div");
        assertTrue(tag.isBlock());
    }

        Tag tag = new Tag("p");
        assertFalse(tag.isBlock());
    }

        Tag tag = new Tag("div");
        assertTrue(tag.formatAsBlock());

        Tag tag = new Tag("p");
        assertFalse(tag.formatAsBlock());
    }

        Tag tag = new Tag("div");
        assertFalse(tag.isInline());

        assertFalse(tag.isEmpty());

        assertTrue(tag.isEmpty());

        Tag tag = new Tag("img");
        assertTrue(tag.isSelfClosing());
    }

    @Test

        String tagName = "script";
        assertFalse(Tag.isKnownTag(tagName));
    }

    @Test
    public void preserveWhitespace_AttributesWithoutSpaceReturnTrue() {
        Tag tag = new Tag("input");
        assertTrue(tag.preserveWhitespace());
    }

    @Test
    public void preserveWhitespace_AttributeWithSpace_ReturnsFalse() {
        String tagName = "input";
        Tag tag = new Tag(tagName);
        assertFalse(tag.preserveWhitespace());
    }
