public class NodeList {
    private List<Node> elements;

    public NodeList() {
        this.elements = new ArrayList<>();
    }

    public void add(Node element) {
        this.elements.add(element);
    }

    public List<Node> getElements() {
        return elements;
    }

    @FunctionalInterface
    interface Factory<T> {
        T create();
    }

    public static <T> List<T> sourceNodes(NodeList nodeList, Class<T> clazz) {
        List<T> result = new ArrayList<>();
        for (Node node : nodeList.getElements()) {
            if (clazz.isInstance(node)) {
                result.add(clazz.getConstructor(int.class).newInstance(node.getId()));
            }
        }
        return result;
    }

    public static NodeList copy(NodeList original) {
        NodeList copy = new NodeList();
        for (Node node : original.getElements()) {
            copy.addElement(new Node(node.getId()));
        }
        return copy;
    }
}

public class W3CDom {
    public static NodeList selectXpath(String xpath, String html) {
        Element element = new Element(html);
        Document document = element.parse();
        Element targetElement = document.selectFirst(xpath);
        NodeList nodeList = new NodeList();
        if (targetElement != null) {
            nodeList.addElement(new Node(targetElement.id()));
        }
        return nodeList;
    }
}

public class NodeList {

    public static <T> List<T> sourceNodes(NodeList nodeList, Class<T> clazz) {
        List<T> result = new ArrayList<>();
        for (org.jsoup.nodes.Element element : nodeList.getElements()) {
            if (clazz.isInstance(element)) {
                T t = clazz.getConstructor().newInstance();
                org.jsoup.nodes.Element tempElement = null;
                for (org.jsoup.nodes.Element child : element.children()) {
                    if (child.tagName().equals("p")) {
                        tempElement = child;
                        break;
                    }
                }
                if (tempElement != null) {
                    t.getClass().getMethod("setId", int.class).invoke(t, tempElement.attr("id"));
                }
                result.add((T)t);
            }
        }
        return result;
    }

}

public class W3CDom {

    public static NodeList selectXpath(String xpath, String html) {
        Document document = org.jsoup.nodes.Document.parse(html);
        NodeList nodeList = new NodeList();
        for (org.jsoup.nodes.Element element : document.select(xpath)) {
            nodeList.addElement(new Node(element.attr("id")));
        }
        return nodeList;
    }
}
