    }

    public void newNodeName nodeNameIsCorrect() {
        TextNode textNode = new TextNode("text");
        assertEquals("text_node", textNode.nodeName());
    }

    @Test
    public void appendText appendTextIsCorrect() {
        TextNode textNode = new TextNode("text");
        String expected = "text";
        textNode.text(expected);
        assertEquals(expected, textNode.text());
    }

    @Test
    public void appendBlank appendBlankIsCorrect() {
        TextNode textNode = new TextNode("");
        assertTrue(textNode.isBlank());
    }

    @Test
    public void splitTextSplitAtOffset_SplitAtOffsetCorrectly() throws IOException {
        String originalText = "text";
        int offset = 5;
        StringBuilder sb = new StringBuilder(originalText);
        sb.deleteCharAt(offset);
        TextNode textNode = new TextNode(originalText);
        TextNode actual = textNode.splitText(offset);
        assertEquals(sb.toString(), actual.text());
    }

    @Test
    public void normaliseWhitespace_NormalisationIsCorrect() {
        String expected = "text";
        assertEquals(expected, StringUtil.normaliseWhitespace(expected));
    }

    @Test
    public void stripLeadingWhitespace_LeadingWhitespaceRemovedCorrectly() {

    public void lastCharIsWhitespace_LastCharacterIsWhitespaceCorrectly() throws IOException {
