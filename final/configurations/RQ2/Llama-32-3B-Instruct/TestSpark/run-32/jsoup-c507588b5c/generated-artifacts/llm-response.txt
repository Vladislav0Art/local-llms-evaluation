        textNode.outerHtmlHead(appendable, 0, null);
        textNode.outerHtmlTail(appendable, 0, null);
        assertEquals("<p>Hello</p>", html.toString());
    }

    @Test
    public void toString_ReturnsNormalizedText() {
        TextNode textNode = new TextNode("Hello\nWorld");
        assertEquals("Hello\nWorld", textNode.toString());
    }
}

public class TextNode {

    private String name;
    private String text;

    public TextNode(String name, String text) {
        this.name = name;
        this.text = text;
    }

    public String nodeName() {
        return name;
    }

    public String getText() {
        return text;
    }

    public void outerHtmlHead(Appendable appendable, int offset, String parentName) throws IOException {
        if (parentName != null && !parentName.isEmpty()) {
            append.append("<").append(parentName).append(">");
        }
        append.append("<p>").append(text).append("</p>");
        if (parentName != null && !parentName.isEmpty()) {
            append.append("</").append(parentName).append(">");
        }
    }

    public void outerHtmlTail(Appendable appendable, int offset, String parentName) throws IOException {
        if (parentName != null && !parentName.isEmpty()) {
            append.append("</").append(parentName).append(">");
        }
        append.append("</p>");
    }

    public TextNode clone() {
        return new TextNode(name, text);
    }

    public TextNode splitText(int offset) {
        return new TextNode(text.substring(0, offset), text.substring(offset));
    }

    @Override
    public String toString() {
        return text;
    }

    public boolean isBlank() {
        return text.isEmpty();
    }
}
