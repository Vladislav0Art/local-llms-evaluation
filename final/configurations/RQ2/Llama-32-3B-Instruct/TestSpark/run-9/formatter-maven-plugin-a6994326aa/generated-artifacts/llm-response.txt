import java.util.*;

public class CssFormatter {
    private boolean initialized = false;
    private Map<String, String> options;

    public void init(Map<String, String> options, ConfigurationSource cfg) {
        this.options = options;
        this.initialized = true;
    }

    public String doFormat(InputSource inputSource, LineEnding ending) {
        if (!initialized) {
            throw new RuntimeException("Formatter not initialized");
        }

        // Replace all tabs with spaces
        String formattedInput = inputSource.getInput().replace("\t", "  ");

        // Format the CSS selector (e.g., ".class", "#id")
        List<String> selectors = splitSelectors(formattedInput);
        formattedInput = formatSelectors(selectors);

        // Format the CSS properties (e.g., "color: blue", "font-size: 12px")
        List<Property> properties = parseProperties(formattedInput);
        formattedInput = formatProperties(properties, ending);

        return formattedInput;
    }

    private List<String> splitSelectors(String input) {
        List<String> selectors = new ArrayList<>();
        String currentSelector = "";
        for (char c : input.toCharArray()) {
            if (c == '{') {
                if (!currentSelector.isEmpty()) {
                    selectors.add(currentSelector);
                    currentSelector = "";
                }
            } else if (c == '}') {
                if (!currentSelector.isEmpty()) {
                    selectors.add(currentSelector);
                    currentSelector = "";
                }
            } else {
                currentSelector += c;
            }
        }
        if (!currentSelector.isEmpty()) {
            selectors.add(currentSelector);
        }
        return selectors;
    }

    private String formatSelectors(List<String> selectors) {
        StringBuilder formattedInput = new StringBuilder();
        for (String selector : selectors) {
            formattedInput.append(selector).append(" { ");
        }
        return formattedInput.toString();
    }

    private List<Property> parseProperties(String input) {
        List<Property> properties = new ArrayList<>();
        String[] propertyPairs = input.split(";\\s*");
        for (String pair : propertyPairs) {
            if (!pair.isEmpty()) {
                String[] parts = pair.split(":");
                if (parts.length == 2) {
                    String key = parts[0].trim();
                    String value = parts[1].trim().replaceAll("^\\s+|\\s+$", "");
                    properties.add(new Property(key, value));
                }
            }
        }
        return properties;
    }

    private String formatProperties(List<Property> properties, LineEnding ending) {
        StringBuilder formattedInput = new StringBuilder();
        for (int i = 0; i < properties.size(); i++) {
            Property property = properties.get(i);
            if (i > 0) {
                formattedInput.append("; ");
            }
            String value = formatValue(property.getValue(), options, ending);
            formattedInput.append(property.getKey()).append(": ").append(value).append("; ");
        }
        return formattedInput.toString().trim() + ending.toString();
    }

    private String formatValue(String value, Map<String, String> options, LineEnding ending) {
        if (options.containsKey("indent")) {
            return indent(value, Integer.parseInt(options.get("indent")));
        } else {
            return value;
        }
    }

    private int indent(String input, int level) {
        StringBuilder indentation = new StringBuilder();
        for (int i = 0; i < level; i++) {
            indentation.append("  ");
        }
        return indentation.toString() + input;
    }
}

        return key;
    }

        this.key = key;
    }

    private String key;

    public String getValue() {
        return value;
    }
