public class Gt06ProtocolDecoder {

    public static Position decodeGps(Position position, boolean useDate) {
        if (useDate && position.getLatitude() != 0.0) {
            return new Position(position.getLongitude(), position.getLatitude());
        } else {
            return position;
        }
    }

    public static boolean equalsDecodedGps(Position originalPosition, Position decodedPosition) {
        if (!decodedPosition.getLatitude().equals(originalPosition.getLatitude()) ||
                !decodedPosition.getLongitude().equals(originalPosition.getLongitude())) {
            return false;
        }
        return true;
    }

    public static void testDecodeGps() {
        Position position = new Position(1.0, 2.0);
        Date date = null;
        Calendar calendar = null;

        assertTrue(Gt06ProtocolDecoder.decodeGps(position, true));
        assertFalse(EqualsDecodedGps(position, Gt06ProtocolDecoder.decodeGps(position, true)));

        assertTrue(EqualsDecodedGps(position, position));

        assertFalse(Gt06ProtocolDecoder.decodeGps(new Position(-1.0, -2.0), true));
        assertFalse(EqualsDecodedGps(new Position(-1.0, -2.0), Gt06ProtocolDecoder.decodeGps(new Position(-1.0, -2.0), true)));

        assertFalse(Gt06ProtocolDecoder.decodeGps(position, date));
        assertFalse(EqualsDecodedGps(position, Gt06ProtocolDecoder.decodeGps(position, date)));

        assertFalse(Gt06ProtocolDecoder.decodeGps(position, calendar));
        assertFalse(EqualsDecodedGps(position, Gt06ProtocolDecoder.decodeGps(position, calendar)));
    }

    public static boolean EqualsDecodedGps(Position originalPosition, Position decodedPosition) {
        return Gt06ProtocolDecoder.decodeGps(originalPosition, true).getLatitude().equals(decodedPosition.getLatitude()) &&
               Gt06ProtocolDecoder.decodeGps(originalPosition, true).getLongitude().equals(decodedPosition.getLongitude());
    }
}
