```java
// Define the Expression class as the target of our visitor pattern
public abstract class Expression {
    public static String toString() {
        return "Expression";
    }

    public abstract void accept(Visitor visitor);
}

// Concrete expression classes that can be visited by the visitor
public class AssigningExpr extends Expression {
    private AssigningExpr left;
    private AssignmentExpression right;

    public AssigningExpr(AssigningExpr left, AssignmentExpression right) {
        this.left = left;
        this.right = right;
    }

    @Override
    public void accept(Visitor visitor) {
        visitor.visitAssignExpr(this);
    }
}

public class BinaryExpr extends Expression {
    private Value left;
    private Expression right;

    public BinaryExpr(Value left, Expression right) {
        this.left = left;
        this.right = right;
    }

    @Override
    public void accept(Visitor visitor) {
        visitor.visitBinaryExpr(this);
    }
}

public class UnaryExpr extends Expression {
    private Value arg;

    public UnaryExpr(Value arg) {
        this.arg = arg;
    }

    @Override
    public void accept(Visitor visitor) {
        visitor.visitUnaryExpr(this);
    }
}

// Define the AssignmentExpression interface for AssigningExpr
public interface AssignmentExpression {}

public class NameAssignmentExpression implements AssignmentExpression {
    private String name;
    private Expression expr;

    public NameAssignmentExpression(String name, Expression expr) {
        this.name = name;
        this.expr = expr;
    }

    @Override
    public String toString() {
        return name + "." + expr;
    }
}

// Define the Visitor interface that will be implemented by concrete visitors
public abstract class Visitor {
    protected Printer printer;

    public Visitor(Printer printer) {
        this.printer = printer;
    }

    public abstract void visitAssignExpr(AssigningExpr assigningExpr);

    public abstract void visitBinaryExpr(BinaryExpr binaryExpr);

    public abstract void visitUnaryExpr(UnaryExpr unaryExpr);
}

// Concrete visitor that will print the expression in a pretty format
public class PrettyPrintVisitor extends Visitor {
    public PrettyPrintVisitor(Printer printer) {
        super(printer);
    }

    @Override
    public void visitAssignExpr(AssigningExpr assigningExpr) {
        printer.print(assigningExpr.toString());
    }

    @Override
    public void visitBinaryExpr(BinaryExpr binaryExpr) {
        printer.print(binaryExpr.toString());
    }

    @Override
    public void visitUnaryExpr(UnaryExpr unaryExpr) {
        printer.print(unaryExpr.toString());
    }
}

// Define the Printer interface for printing expressions in a pretty format
public interface Printer {}

public class PrinterImpl implements Printer {
    private StringBuilder sb;

    public PrinterImpl() {
        this.sb = new StringBuilder();
    }

    public void print(String s) {
        sb.append(s);
    }

    @Override
    public String toString() {
        return sb.toString();
    }
}

// Example usage of the visitor pattern:
public class Main {
    public static void main(String[] args) {
        Printer printer = new PrinterImpl();

        Expression left = new BinaryExpr(new Value(1), new NameAssignmentExpression("x", new ConstantValue(2)));
        Expression right = new AssigningExpr(left, new NameAssignmentExpression("y", new ConstantValue(3)));

        Visitor visitor = new PrettyPrintVisitor(printer);
        right.accept(visitor);

        System.out.println(printer.toString());
    }
}
```

This example demonstrates how to implement the visitor pattern in Java using abstract classes and interfaces. The `Expression` class is the target of our visitor pattern, while the `AssigningExpr`, `BinaryExpr`, and `UnaryExpr` classes are concrete expression classes that can be visited by the visitor. The `Visitor` interface defines the methods that must be implemented by concrete visitors, such as `visitAssignExpr(AssigningExpr assigningExpr)`. In this example, we define a `PrettyPrintVisitor` class that implements the `Visitor` interface and prints expressions in a pretty format using the `PrinterImpl` class.
