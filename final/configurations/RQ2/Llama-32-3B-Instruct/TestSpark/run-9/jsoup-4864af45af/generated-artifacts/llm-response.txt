import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class Main {

    public static void main(String[] args) {
        System.out.println(isSafeList(Jsoup.connect("http://example.com").parse()));
        System.out.println(isRelativeLinkPreserved(Jsoup.connect("http://example.com").parse()));
        System.out.println(getEnforcedAttributes(Jsoup.connect("http://example.com").parse()));
    }

    public static boolean isSafeList(Element safeList) {
        return safeList.select(".safe-list").size() > 0;
    }

    public static boolean isRelativeLinkPreserved(Element safeList) {
        return safeList.hasAttr("href") && safeList.attr("href").startsWith("#");
    }

    public static Attributes getEnforcedAttributes(Element safeList) {
        Elements tags = safeList.select("[class]");
        Set<String> attributes = new HashSet<>();
        for (Element tag : tags) {
            String[] attributeArray = tag.attr("class").split("\\s+");
            for (String attribute : attributeArray) {
                if (attribute.equals("img") || attribute.equals("br") || attribute.equals("p")) {
                    continue;
                } else if (attribute.equals("strong") || attribute.equals("b")) {
                    attributes.add(attribute);
                }
            }
        }
        return new Attributes(attributes);
    }

    public static class Attributes {
        private Set<String> set;

        public Attributes(Set<String> set) {
            this.set = set;
        }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (o == null || getClass() != o.getClass()) return false;
            Attributes attributes = (Attributes) o;
            return set.equals(attributes.set);
        }
    }
}
