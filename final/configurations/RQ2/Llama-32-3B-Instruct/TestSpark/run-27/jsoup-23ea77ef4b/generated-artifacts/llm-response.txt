public class W3CDom {
    public static NodeList sourceNodes(NodeList nodes, Class<?> nodeClass) {
        NodeList result = new NodeList();
        for (Node node : nodes) {
            if (nodeClass.isInstance(node)) {
                result.add(new Node());
            }
        }
        return result;
    }

    public static Document contextNode(Document document) {
        return document;
    }
}

public class Node {
    private String tagName;

    public Node(String tagName) {
        this.tagName = tagName;
    }

    @Override
    public String toString() {
        return "<" + tagName + ">";
    }
}

public class Document extends org.jsoup.nodes.Document {
    private NodeList elements;

    public NodeList getElements() {
        return elements;
    }

    public void setElements(NodeList elements) {
        this.elements = elements;
    }
}

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;
import org.junit.Test;
import static org.junit.Assert.assertEquals;

public class TestW3CDom {
    @Test
    public void testContextNode() {
        Document document = new Document();
        NodeList expectedNodes = null; 
        NodeList actualNodes = W3CDom.contextNode(document).getElements();
        assertEquals(expectedNodes, actualNodes);
    }

    @Test
    public void testSourceNodes() {
        Document document = new Document();
        Elements elements = document.select("div");
        NodeList nodes = new NodeList();
        for (Element element : elements) {
            Node node = new Node(element.tagName());
            nodes.add(node);
        }
        NodeList expectedNodes = nodes;
        NodeList actualNodes = W3CDom.sourceNodes(nodes, Node.class);
        assertEquals(expectedNodes, actualNodes);
    }
}
