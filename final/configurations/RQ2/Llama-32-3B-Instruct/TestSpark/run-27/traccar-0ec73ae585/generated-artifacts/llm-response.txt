import static org.junit.Assert.assertTrue;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class WatchProtocolDecoderTest {

    @Mock
    private DeviceSession deviceSession;

    @Mock
    private Session session;

    @Mock
    private Channel channel;

    private WatchProtocolDecoderDecoder decoder = new WatchProtocolDecoder();

    @Before
    public void setUp() {
        MockitoAnnotations.initMocks(this);
        when(deviceSession.getProtocol()).thenReturn(WatchProtocolProtocol.class);
    }

    @Test
    public void getHasIndexTrueTest() {
        assertTrue(decoder.getHasIndex());
    }

    @Test
    public void getManufacturerTest() {
        assertEquals("Watch Protocol", decoder.getManufacturer());
    }

    @Test
    public void decodeDecodeValidMessageTest() throws Exception {
        when(channel.pipeline()).thenReturn(new io.netty.channel.ChannelPipeline());
        when(channel.pipeline().writeAndFlush(any())).thenAnswer(i -> i.invoke(null));

        byte[] bytes = "watch protocol message".getBytes(StandardCharsets.UTF_8);
        ByteBuf buf = Unpooled.buffer(bytes.length);
        buf.writeBytes(bytes);

        Network network = new Network();
        Position position = new Position();
        network.setCellTower(new CellTower());
        network.setPosition(position);

        decoder.decode(channel, deviceSession, network);

        Mockito.verify(channel.pipeline()).writeAndFlush(eq(network));
    }

    @Test
    public void decodeDecodeInvalidMessageTest() throws Exception {
        when(channel.pipeline()).thenReturn(new io.netty.channel.ChannelPipeline());
        when(channel.pipeline().writeAndFlush(any())).thenAnswer(i -> i.invoke(null));

        byte[] bytes = "invalid watch protocol message".getBytes(StandardCharsets.UTF_8);
        ByteBuf buf = Unpooled.buffer(bytes.length);
        buf.writeBytes(bytes);

        decoder.decode(channel, deviceSession, new Network());

        Mockito.verify(channel.pipeline(), never()).writeAndFlush(eq(new Network()));
    }

    @Test
    public void decodeDecodeNoMessageFoundTest() throws Exception {
        when(channel.pipeline()).thenReturn(new io.netty.channel.ChannelPipeline());
        when(channel.pipeline().writeAndFlush(any())).thenAnswer(i -> i.invoke(null));

        decoder.decode(channel, deviceSession, null);

        Mockito.verify(channel.pipeline(), never()).writeAndFlush(eq(null));
    }

    @Test
    public void decodeDecodeMessageTooLongTest() throws Exception {
        when(channel.pipeline()).thenReturn(new io.netty.channel.ChannelPipeline());
        when(channel.pipeline().writeAndFlush(any())).thenAnswer(i -> i.invoke(null));

        byte[] bytes = new byte[1024];
        ByteBuf buf = Unpooled.buffer(bytes.length);
        buf.writeBytes(bytes);

        decoder.decode(channel, deviceSession, null);

        Mockito.verify(channel.pipeline(), never()).writeAndFlush(eq(new Network()));
    }

    private static class WatchProtocolDecoderMock extends WatchProtocolDecoder {
        @Override
        public boolean getHasIndex() {
            return true;
        }
    }
}
