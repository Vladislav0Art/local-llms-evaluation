package org.jsoup.nodes;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

@RunWith(MockitoJUnitRunner.class)
public class DocumentTest {

    @Mock
    private Connection connectionMock;

    @Mock
    private Parser parserMock;

    @Mock
    private Element elementMock;

    public static final String BASE_URI = "https://example.com";

    public static final String CSS_QUERY = "#test";

    @Test
    public void createShell_DoesNotThrowExceptionWhenBaseUriIsNotNull() {
        Document document = Document.createShell(BASE_URI);
        Mockito.verifyNoArguments();
    }

    @Test
    public void createShell_ThrowsNullPointerExceptionWhenBaseUriIsNull() {
        Assert.assertEquals("null", Document.createShell(null).getClass());
    }

    @Test
    public void location_IsNotNullWhenDocumentIsCreated() {
        Document document = new Document(BASE_URI);
        Mockito.verify(connectionMock, Mockito.never()).location();
        Mockito.when(elementMock.location()).thenReturn("");
    }

    @Test
    public void connection_ReturnsConnectionObject() {
        Document document = new Document(BASE_URI);
        Assert.assertEquals(connectionMock, document.connection());
    }

    @Test
    public void expectForm_DoesNotThrowExceptionWhenCssQueryIsNotNull() {
        FormElement form = Document.createShell(BASE_URI).expectForm(CSS_QUERY);
        Mockito.verifyNoArguments();
    }

    @Test
    public void expectForm_ThrowsNullPointerExceptionWhenCssQueryIsNull() {
        Assert.assertEquals("null", Document.createShell(BASE_URI).expectForm(null).getClass());
    }

    @Test
    public void title_ReturnsNotNullStringWhenTitleIsSet() {
        String title = "Test Title";
        Document document = new Document(BASE_URI);
        document.title(title);
        Mockito.verify(elementMock, Mockito.never()).setTitle();
        Assert.assertEquals(title, document.title());
    }

    @Test
    public void createElement_DoesNotThrowExceptionWhenTagNameIsNotNull() {
        Element element = new Document(BASE_URI).createElement("div");
        Mockito.verifyNoArguments();
    }

    @Test
    public void text ReturnsNotNullStringWhenTextIsSet() {
        String text = "Hello World";
        Assert.assertEquals(text, new Document(BASE_URI).text(text));
    }

    @Test
    public void charset_ReturnsNotNullCharsetObjectWhenCharsetIsSet() {
        Charset charset = Charset.forName("UTF-8");
        Document document = new Document(BASE_URI);
        document.charset(charset);
        Mockito.verify(elementMock, Mockito.never()).charset();
        Assert.assertEquals(charset, document.charset());
    }

    @Test
    public void updateMetaCharsetElement_ThrowsNullPointerExceptionWhenUpdateIsNull() {
        Assert.assertEquals("null", new Document(BASE_URI).updateMetaCharsetElement(null).getClass());
    }

    @Test
    public void shallowClone_ReturnsShallowCopyOfDocumentObject() {
        Document document = new Document(BASE_URI);
        Document copy = document.clone();
        Mockito.verify(elementMock, Mockito.never()).clone();
        Assert.assertEquals(document, copy);
    }

    @Test
    public void outputSettings_DoesNotThrowExceptionWhenOutputSettingsIsNotNull() {
        OutputSettings outputSettings = new OutputSettings();
        Document document = new Document(BASE_URI).outputSettings(outputSettings);
        Mockito.verifyNoArguments();
    }

    @Test
    public void connection_ReturnsConnectionObject() {
        Document document = new Document(BASE_URI);
        Assert.assertEquals(connectionMock, document.connection());
    }
}
