import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import java.time.LocalDate;

import static org.junit.Assert.assertTrue;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class DeltaTTest {

    @Mock
    private LocalDate forDate;

    @Test
    public void estimate_DoesNotThrowExceptionForValidDate() {
        when(forDate.toInstant()).thenReturn(0);
        try {
            DeltaT.estimate(forDate);
            assertTrue(true);
        } catch (Exception e) {
            fail("Should not throw exception");
        }
    }

    @Test
    public void estimate_ThrowsExceptionForInvalidDate() {
        when(forDate.toInstant()).thenReturn(-1L);
        assertThrows(Exception.class, () -> DeltaT.estimate(forDate));
    }

    @Test
    public void isLeapYear_DoesNotThrowExceptionWhenInputIsValid() {
        try {
            DeltaT.isLeapYear(LocalDate.of(2020, 2, 29));
            assertTrue(true);
        } catch (Exception e) {
            fail("Should not throw exception");
        }
    }

    @Test
    public void isLeapYear_ThrowsExceptionForInvalidInput() {
        assertThrows(Exception.class, () -> DeltaT.isLeapYear(LocalDate.of(2020, 1, 29)));
    }

}
