import java.io.BufferedReader;
import java.io.FileReader;
import java.io.IOException;

public class LineEndingDetector {

    public enum LineEnding {
        LF("\n"),
        CRLF("\r\n");

        private final String value;

        LineEnding(String value) {
            this.value = value;
        }

        public String getValue() {
            return value;
        }
    }

    public static LineEnding detectLineEnding(String content) {
        if (content.contains(LineEnding.LF.getValue())) {
            return LineEnding.LF;
        } else if (content.contains(LineEnding.CRLF.getValue())) {
            return LineEnding.CRLF;
        } else {
            return null;
        }
    }

    public static String normalizeLineEndings(String content, LineEnding lineEnding) {
        StringBuilder normalizedContent = new StringBuilder();

        for (char c : content.toCharArray()) {
            if (c == '\r') {
                normalizedContent.append(lineEnding.getValue().charAt(0));
            } else if (c == '\n') {
                continue;
            } else {
                normalizedContent.append(c);
            }
        }

        return normalizedContent.toString();
    }

    public static String normalizeLineEndings(String content) {
        LineEnding lineEnding = detectLineEnding(content);

        if (lineEnding != null && lineEnding.getValue().equals("\r\n")) {
            return normalizeLineEndings(content, LineEnding.CRLF);
        } else {
            return normalizeLineEndings(content, LineEnding.LF);
        }
    }

    public static String normalize(String content) {
        return normalizeLineEndings(content);
    }

    public static void main(String[] args) {
        String content = "Hello\r\nWorld!\n";
        System.out.println(normalize(content));
    }
}
