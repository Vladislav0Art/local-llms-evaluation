import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;
import org.junit.BeforeClass;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mockito;

import static org.junit.Assert.*;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class DocumentTest {

    @BeforeClass
    public static void setup() {
        Document.document = null;
    }

    private static Document document = new Document("");

    @Test
    public void createShell_documentCreationSuccess() {
        Document createdDoc = Document.createShell("");
        assertNotNull(createdDoc);
    }

    @Test
    public void createShell_documentCreationFailure() {
        when(Document.parser()).thenReturn(new Parser(null));
        Document createdDoc = Document.createShell("");
        assertNull(createdDoc);
    }

    @Test
    public void location_methodReturnsCorrectLocation() {
        document.location().setUrl("http://example.com");
        assertEquals("http://example.com", document.location());
    }

    @Test
    public void location_methodReturnsEmptyStringWhenNotSet() {
        assertNotNull(document.location());
        assertNull(document.location().getUrl());
    }

    @Test
    public void connection_methodReturnsCorrectConnection() {
        Connection conn = new Connection();
        when(Document.connection()).thenReturn(conn);
        assertEquals(conn, document.connection());
    }

    @Test
    public void createElement_methodCreatesNewElement() {
        Element expectedElement = new Element("");
        when(document.createElement("test")).thenReturn(expectedElement);
        Document createdDoc = Document.createShell("");
        Element actualElement = createdDoc.createElement("test");
        assertNotNull(actualElement);
        assertEquals(expectedElement, actualElement);
    }

    @Test
    public void createElement_methodReturnsNullWhenTagNameIsEmpty() {
        assertNull(document.createElement(""));
    }

    @Test
    public void title_methodSetsCorrectTitle() {
        document.title("newTitle");
        assertEquals("newTitle", document.title());
    }

    @Test
    public void title_methodReturnsCurrentTitleIfNotSet() {
        assertNotNull(document.title());
        assertEquals(document.title(), "title");
    }

    @Test
    public void text_methodSetsCorrectText() {
        document.text("newText");
        assertEquals("newText", document.text());
    }

    @Test
    public void text_methodReturnsCurrentTextIfNone() {
        assertNull(document.text());
        assertEquals("", document.text());
    }
}
