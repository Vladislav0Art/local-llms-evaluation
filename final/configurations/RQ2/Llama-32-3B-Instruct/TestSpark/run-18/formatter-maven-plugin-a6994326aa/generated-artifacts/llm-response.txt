        formatMethodCall(formatter.init, options, mockCfg);

        assertTrue(formatter.isInitialized());
    }

    private void formatMethodCall(final Method method, final Object... args) throws Exception {
        final Object[] originalArgs = method.invoke(null);
        method.invoke(method, args[0], args[1]);
        assertNotNull(originalArgs);
    }
}
