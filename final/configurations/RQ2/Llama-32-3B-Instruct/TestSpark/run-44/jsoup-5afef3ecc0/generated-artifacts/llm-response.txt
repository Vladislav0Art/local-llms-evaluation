    }

    @Test
    public void testAppendKeyValWithQuery() {
        String url = "https://example.com/path/to/resource?query=string";
        assertEquals(url, new UrlBuilder(new URL(url)).appendKeyVal(Connection.KeyVal.of(0, url)).build().toString());
    }

    @Test
    public void testAppendKeyValWithNewQuery() {
        String url = "https://example.com/path/to/resource";
        String expectedUrl = "https://example.com/path/to/resource?query=new=value";
        assertEquals(expectedUrl, new UrlBuilder(new URL(url)).appendKeyVal(Connection.KeyVal.of(0, url)).build().toString());
    }

    @Test
    public void testAppendKeyValWithNewValue() {
        String url = "https://example.com/path/to/resource";
        String expectedUrl = "https://example.com/path/to/resource?query=new=value";
        assertEquals(expectedUrl, new UrlBuilder(new URL(url)).appendKeyVal(Connection.KeyVal.of(0, url)).build().toString());
    }

    public static class Connection {
        public interface KeyVal {
            static String of(int key, String value) {
                return "key=" + key + "&value=" + value;
            }
        }
    }

    public static class UrlBuilder {
        private URL url;

        public UrlBuilder(URL url) {
            this.url = url;
        }

        public UrlBuilder appendKeyVal(Connection.KeyVal keyVal) {
            Map<String, String> params = new HashMap<>();
            if (keyVal != null) {
                params.put("key", String.valueOf(keyVal.getKey()));
                params.put("value", keyVal.getValue());
            }
            return new UrlBuilder(new URL(url.toString() + "?" + String.join("&", params.entrySet().stream().map(e -> e.getKey() + "=" + e.getValue()).toArray(String[]::new))));
        }

        public String build() {
            return url.toString();
        }
    }
}

public class Connection {
    public interface KeyVal {
        String of(int key, String value) {
            return "key=" + key + "&value=" + value;
        }
    }

    public static class Builder {
        private String query;

        public Builder appendKeyVal(Connection.KeyVal keyVal) {
            if (query == null) {
                query = Connection.KeyVal.of(keyVal.getKey(), keyVal.getValue());
            } else {
                StringBuilder sb = new StringBuilder(query);
                sb.append("&").append(keyVal.key).append("=").append(keyVal.value);
                query = sb.toString();
            }
            return this;
        }

        public String build() {
            if (query != null && !query.isEmpty()) {
                return url + "?" + query;
            } else {
                return url.toString();
            }
        }
    }

    private URL url;

    public Builder(String url) {
        this.url = new URL(url);
    }

    public UrlBuilder() {}

    public Builder appendKeyVal(Connection.KeyVal keyVal) {
        if (query == null) {
            query = Connection.KeyVal.of(keyVal.getKey(), keyVal.getValue());
        } else {
            StringBuilder sb = new StringBuilder(query);
            sb.append("&").append(keyVal.key).append("=").append(keyVal.value);
            query = sb.toString();
        }
        return this;
    }

    public String build() {
        if (query != null && !query.isEmpty()) {
            return url + "?" + query;
        } else {
            return url.toString();
        }
    }

    @Override
    public boolean equals(Object o) {
        return Objects.equals(url, ((UrlBuilder) o).url);
    }

    public static class UrlBuilder extends Connection.Builder {
    }
}
