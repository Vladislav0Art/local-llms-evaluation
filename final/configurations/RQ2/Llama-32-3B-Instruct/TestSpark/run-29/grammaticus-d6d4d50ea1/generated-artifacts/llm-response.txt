import java.io.IOException;
import java.util.HashMap;
import java.util.Map;

public class Noun {
    private String nounName;
    private Map<String, String> nounInfo = new HashMap<>();

    public Noun(String name) {
        this.nounName = name;
    }

    public void setNounInfo(String info) {
        this.nounInfo.put("info", info);
    }

    public Map<String, String> getNounInfo() {
        return nounInfo;
    }
}

public class TermMap {
    private Map<String, Noun> termMap = new HashMap<>();

    public void put(String key, Noun value) {
        if (!termMap.containsKey(key)) {
            termMap.put(key, value);
        } else {
            System.out.println("Key already exists in the map.");
        }
    }

    public Noun getTerm(String key) {
        return termMap.get(key);
    }

    public Map<String, Noun> getKeySet() {
        return termMap;
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        for (Map.Entry<String, Noun> entry : termMap.entrySet()) {
            sb.append(entry.getKey()).append(": ").append(entry.getValue().getNounName()).append("\n");
        }
        return sb.toString();
    }

    public Map<String, Noun> addTerm(String key, Noun value) {
        if (!termMap.containsKey(key)) {
            termMap.put(key, value);
        } else {
            System.out.println("Key already exists in the map.");
        }
        return termMap;
    }

    public void removeTerm(String key) {
        termMap.remove(key);
    }

    public static void main(String[] args) {
        TermMap termMap = new TermMap();
        Noun noun1 = new Noun("noun1");
        noun1.setNounInfo("info1");
        termMap.put("noun1", noun1);

        System.out.println(termMap.getKeySet());
        for (Noun value : termMap.getKeySet().values()) {
            System.out.println(value.getNounName() + ": " + value.getNounInfo());
        }
    }
}
