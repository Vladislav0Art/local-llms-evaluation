```java
import java.util.Hashtable;
import java.util.Iterator;

public class Main {

    public static void main(String[] args) {
        String tableName = "test_table";
        String columnNameMinType = "string";
        String columnNameMaxType = "integer";

        Hashtable<String, Object> htblColNameMinValue = new Hashtable<>();
        htblColNameMinValue.put("key", "1");
        htblColNameMinValue.put("value", "one");

        Hashtable<String, Object> htblColNameMaxValue = new Hashtable<>();
        htblColNameMaxValue.put("key", "2");
        htblColNameMaxValue.put("value", "two");

        Hashtable<String, String> htblColNameType = createTable(tableName, columnNameMinType, columnNameMaxType, htblColNameMinValue, htblColNameMaxValue);

        // print the table
        System.out.println(htblColNameType);

        // select from table
        selectFromTable(tableName);
    }

    public static Hashtable<String, String> createTable(
            String tableName,
            String columnNameMinType,
            String columnNameMaxType,
            Hashtable<String, Object> htblColNameMinValue,
            Hashtable<String, Object> htblColNameMaxValue) {
        Hashtable<String, String> htblColNameType = new Hashtable<>();

        // Add columns to the table
        if (htblColNameMinValue != null && !htblColNameMinValue.isEmpty()) {
            for (Iterator iterator = htblColNameMinValue.keySet().iterator(); iterator.hasNext();) {
                Object key = iterator.next();
                Object value = htblColNameMinValue.get(key);
                htblColNameType.put((String) key, columnNameMaxType + "(" + value.toString() + ")");
            }
        }

        if (htblColNameMaxValue != null && !htblColNameMaxValue.isEmpty()) {
            for (Iterator iterator = htblColNameMaxValue.keySet().iterator(); iterator.hasNext();) {
                Object key = iterator.next();
                Object value = htblColNameMaxValue.get(key);
                htblColNameType.put((String) key, columnNameMinType + "(" + value.toString() + ")");
            }
        }

        return htblColNameType;
    }

    public static void selectFromTable(String tableName) {
        // Simulate selecting data from the table
        System.out.println("Selecting data from " + tableName);
        for (String key : tableName.split(",")) {
            System.out.println(key + ": value");
        }
    }
}
```

This program creates a table with two columns, one of type `string` and another of type `integer`. It then prints the table schema and simulates selecting data from the table.
