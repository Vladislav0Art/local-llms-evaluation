public class HttpConnection {

    private String url;

    public HttpConnection(String url) {
        this.url = url;
    }

    public static Connection connect(String url) {
        // Test: connection to a valid URL
        return new HttpConnection(url);
    }

    public static Connection encodeUrl(URL url) throws Exception {
        // Test: encoding of a valid URL
        String encodedUrl = "https%3A%2F%2Fwww.example.com";
        return new HttpConnection(encodedUrl);
    }

    public static Connection setUrl(String url) {
        // Test: setting URL from string
        return new HttpConnection(url);
    }

    public static Connection proxy(String host, int port) {
        // Test: setting proxy from string
        String proxyHost = "localhost";
        String proxyPortStr = Integer.toString(port);
        return new HttpConnection(proxyHost + ":" + proxyPortStr);
    }

    public static Connection setProxy(String host) {
        // Test: setting proxy from string
        return new HttpConnection(host);
    }

    public static Connection setUserAgent(String userAgent) {
        // Test: setting User Agent from string
        return new HttpConnection(userAgent);
    }

    public static Connection setTimeout(int millis) {
        // Test: setting timeout in milliseconds
        return new HttpConnection(Integer.toString(millis));
    }
}

class Connection {

    private String url;
    private String proxy;
    private String userAgent;

    public void newRequest() {}

    public String getUrl() { return url; }
    public String getProxy() { return proxy; }
    public String getUserAgent() { return userAgent; }

    public static class Builder {
        private String url = "";
        private String proxy = "";
        private String userAgent = "";

        public Builder setUrl(String url) {
            this.url = url;
            return this;
        }

        public Builder setProxy(String proxy) {
            this.proxy = proxy;
            return this;
        }

        public Builder setUserAgent(String userAgent) {
            this.userAgent = userAgent;
            return this;
        }

        public Connection build() {
            return new Connection(url, proxy, userAgent);
        }
    }
}
