import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import java.io.IOException;
import java.util.Map;

@RunWith(MockitoJUnitRunner.class)
public class CssFormatterTest {

    @Mock
    private ConfigurationSource configurationSource;

    @Mock
    private Map<String, String> options;

    @Test
    public void init_NullOptions_ThrowsIOException() throws IOException {
        Mockito.when(configurationSource.getOptions()).thenReturn(options);
        Mockito.when(options.isEmpty()).thenThrow(IOException.class);
        CssFormatter cssFormatter = new CssFormatter();
        cssFormatter.init(options, configurationSource);
    }

    @Test
    public void init_ValidOptions_DoesNotThrowException() throws IOException {
        Mockito.when(configurationSource.getOptions()).thenReturn(options);
        Mockito.when(options.size()).thenReturn(1);
        options.put("key", "value");
        CssFormatter cssFormatter = new CssFormatter();
        cssFormatter.init(options, configurationSource);
    }

    @Test
    public void doFormat_EmptyString_ReturnsEmptyString() {
        String code = "";
        LineEnding lineEnding = LineEnding.LF;
        String expected = "";
        String actual = new CssFormatter().doFormat(code, lineEnding);
        assert(actual.equals(expected));
    }

    @Test
    public void doFormat_SimpleString_ReturnsFormattedString() {
        String code = "body { background-color: red; }";
        LineEnding lineEnding = LineEnding.LF;
        String expected = "body {\n  background-color: red;\n}";
        String actual = new CssFormatter().doFormat(code, lineEnding);
        assert(actual.equals(expected));
    }

    @Test
    public void doFormat_ComplexString_ReturnsFormattedString() {
        String code = "body { background-color: red; } div { color: blue; }";
        LineEnding lineEnding = LineEnding.LF;
        String expected = "body {\n  background-color: red;\n}\ndiv {\n  color: blue;\n}";
        String actual = new CssFormatter().doFormat(code, lineEnding);
        assert(actual.equals(expected));
    }

    @Test
    public void isInitialized_ReturnsFalse() {
        boolean result = new CssFormatter().isInitialized();
        assert(!result);
    }

    @Test
    public void isInitialized_ReturnsTrue() {
        boolean result = new CssFormatter().init(options, configurationSource);
        assert(result);
    }
}
