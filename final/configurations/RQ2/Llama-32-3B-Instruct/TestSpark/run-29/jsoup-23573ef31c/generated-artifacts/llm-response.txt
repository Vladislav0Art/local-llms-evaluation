import org.jsoup.Connection;
import org.jsoup.Jsoup;
import org.junit.Test;
import static org.junit.Assert.*;
import org.junit.runner.RunWith;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

@RunWith(MockitoJUnitRunner.class)
public class DocumentTest {

    @Mock
    private Connection connection;

    @Mock
    private Parser parser;

    @Mock
    private QuirksMode quirksMode;

    @InjectMocks
    private Document document;

    @Test
    public void createShell_DoesNotThrowException() {
        assertFalse(document.createShell(null));
    }

    @Test
    public void location_IsBlank() {
        document.baseUri = "";
        assertEquals("", document.location());
    }

    @Test
    public void location_HasCorrectBaseUri() {
        document.baseUri = "https://example.com";
        String expectedLocation = Jsoup.connect(document.baseUri).get().location();
        assertEquals(expectedLocation, document.location());
    }

    @Test
    public void connection_ReturnsConnection() {
        Connection actualConnection = document.connection();
        Mockito.when(connection.get()).thenReturn(actualConnection);
        assertNotNull(actualConnection);
    }

    @Test
    public void connectionThrowsExceptionWhenPassedNull() {
        assertThrows(NullPointerException.class, () -> document.connection(null));
    }

    @Test
    public void documentType_IsBlank() {
        document.documentType = null;
        assertNull(document.documentType());
    }

    @Test
    public void head_DoesNotThrowException() {
        Element headElement = new Element("html");
        Document testDocument = new Document("https://example.com");
        testDocument.head().appendChild(headElement);
        assertNotNull(testDocument.head());
    }

    @Test
    public void body_DoesNotThrowException() {
        Element bodyElement = new Element("body");
        Document testDocument = new Document("https://example.com");
        testDocument.body().appendChild(bodyElement);
        assertNotNull(testDocument.body());
    }

    @Test
    public void forms_DoesNotThrowException() {
        List<FormElement> expectedForms = new ArrayList<>();
        Document testDocument = new Document("https://example.com");
        FormElement formElement1 = new FormElement();
        expectedForms.add(formElement1);
        testDocument.forms().addAll(expectedForms);
        assertNotNull(testDocument.forms());
    }

    @Test
    public void expectForm_DoesNotThrowException() {
        Selector selector = "input";
        Element expectedForm = new Element("input");
        Document testDocument = new Document("https://example.com");
        FormElement formElement = testDocument.expectForm(selector);
        assertNotNull(formElement);
    }

    @Test
    public void title_DoesNotThrowException() {
        String title = "Title";
        document.title(title);
        assertEquals(title, document.title());
    }

    @Test
    public void createElement_ReturnsNewElement() {
        Element expectedElement = new Element("p");
        Document testDocument = new Document("https://example.com");
        Element actualElement = testDocument.createElement("p");
        assertNotNull(actualElement);
        assertEquals(expectedElement.getTagName(), actualElement.getTagName());
    }

    @Test
    public void outerHtml_ReturnsCorrectOutput() {
        String expectedOuterHtml = "<html><body>Hello World!</body></html>";
        Document testDocument = new Document("https://example.com");
        testDocument.appendText("Hello World!");
        testDocument.appendChild(new Element("div"));
        assertEquals(expectedOuterHtml, testDocument.outerHtml());
    }

    @Test
    public void text_ReturnsCorrectOutput() {
        String expectedText = "Hello World!";
        Document testDocument = new Document("https://example.com");
        testDocument.appendText(expectedText);
        assertEquals(expectedText, testDocument.text());
    }

    @Test
    public void nodeName_IsBlank() {
        document.nodeName();
        assertNull(document.nodeName());
    }

    @Test
    public void charset_DoesNotThrowException() {
        Charset expectedCharset = Charset.forName("UTF-8");
        Document testDocument = new Document("https://example.com");
        testDocument.charset(expectedCharset);
        assertEquals(expectedCharset, testDocument.charset());
    }

    @Test
    public void updateMetaCharsetElement_DoesNotThrowException() {
        boolean expectedUpdate = true;
        Document testDocument = new Document("https://example.com");
        testDocument.updateMetaCharsetElement(expectedUpdate);
        assertTrue(testDocument.updateMetaCharsetElement());
    }

    @Test
    public void outputSettings_DoesNotThrowException() {
        OutputSettings expectedSettings = OutputSettings.CleanerHtmlMode;
        Document testDocument = new Document("https://example.com");
        testDocument.outputSettings(expectedSettings);
        assertEquals(expectedSettings, testDocument.outputSettings());
    }
}
