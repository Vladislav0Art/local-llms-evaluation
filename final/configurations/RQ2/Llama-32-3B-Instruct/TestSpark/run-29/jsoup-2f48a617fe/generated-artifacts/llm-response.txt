import org.junit.Test;
import static org.junit.Assert.*;
import org.jsoup.nodes.Comment;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class CommentTest {

    @Test
    public void newNodeDataEmpty_ReturnsEmptyString() {
        Comment comment = new Comment("");
        assertEquals("", comment.getData());
    }

    @Test
    public void newNodeDataNull_ReturnsNull() {
        Comment comment = new Comment(null);
        assertNull(comment.getData());
    }

    @Test
    public void setDataSetsNewValue() {
        Comment comment = new Comment("old data");
        comment.setData("new data");
        assertEquals("new data", comment.getData());
    }

    @Test
    public void nodeNameIsAlwaysComment() {
        Comment comment = new Comment("");
        assertEquals("comment", comment.nodeName());
    }

    @Test
    public void asXmlDeclarationReturnsNullWhenDataIsNull() {
        Comment comment = new Comment(null);
        assertNull(comment.asXmlDeclaration());
    }

    @Test
    public void isXmlDeclarationAlwaysFalse() {
        Comment comment = new Comment("");
        assertFalse(comment.isXmlDeclaration());
    }

    @Test
    public void cloneCreatesNewCommentInstance() {
        Comment comment1 = new Comment("data");
        Comment comment2 = comment1.clone();
        assertTrue(comment1 != comment2);
        assertEquals(1, comment2.getData().length());
    }
}

public class MockParser {

    private Appendable accum;

    @org.junit.Before
    public void setUp() {
        this.accum = mock(Appendable.class);

        verify(accum).append(eq("<comment>data</comment>"));
    }

    @Test
    public void outerHtmlTailWriteToAccum() {

        verify(accum).append(eq("</comment>"));
    }
}
