package ch.jalu.configme.configurationdata;

import org.junit.Test;
import static org.junit.Assert.*;

import java.util.Map;

public class CommentsConfigurationTest {

    @Test
    public void newCommentsConfiguration_isEmpty() {
        CommentsConfiguration config = new CommentsConfiguration();
        assertTrue(config.getAllComments().isEmpty());
    }

    @Test
    public void newCommentsConfiguration_hasNoLines() {
        Map<String, List<String>> comments = new HashMap<>();
        CommentsConfiguration config = new CommentsConfiguration(comments);
        assertEquals(Collections.emptyList(), config.getAllComments().get("path"));
    }

    @Test
    public void newCommentsConfiguration_hasSingleLine() {
        Map<String, List<String>> comments = new HashMap<>();
        comments.put("path", Arrays.asList(""));
        CommentsConfiguration config = new CommentsConfiguration(comments);
        assertTrue(config.getAllComments().containsKey("path") && config.getAllComments().get("path").isEmpty());
    }

    @Test
    public void newCommentsConfiguration_hasMultipleLines() {
        Map<String, List<String>> comments = new HashMap<>();
        comments.put("path", Arrays.asList("line1", "line2"));
        CommentsConfiguration config = new CommentsConfiguration(comments);
        assertEquals(Arrays.asList("line1", "line2"), config.getAllComments().get("path"));
    }

    @Test
    public void setComment_overwritesExistingLines() {
        Map<String, List<String>> comments = new HashMap<>();
        comments.put("path", Arrays.asList("line1", "line2"));
        CommentsConfiguration config = new CommentsConfiguration(comments);
        String newLine = "new line";
        config.setComment("path", newLine);
        assertEquals(Collections.singletonList(newLine), config.getAllComments().get("path"));
    }

    @Test
    public void setComment_addsNewLines() {
        Map<String, List<String>> comments = new HashMap<>();
        CommentsConfiguration config = new CommentsConfiguration(comments);
        String newLine1 = "new line 1";
        String newLine2 = "new line 2";
        config.setComment("path", newLine1, newLine2);
        assertEquals(Arrays.asList(newLine1, newLine2), config.getAllComments().get("path"));
    }

    @Test
    public void setComment_addsNewLinesAndOverwritesExisting() {

        String newLine = "new line";
        config.setComment("path", "", newLine);
        assertEquals(Arrays.asList(newLine), config.getAllComments().get("path"));

public class CommentsConfiguration {
    private Map<String, List<String>> comments;

    public CommentsConfiguration() {}

    public CommentsConfiguration(Map<String, List<String>> comments) {
        this.comments = comments;
    }

    public void setComment(String path, String... lines) {
        if (comments == null) {
            comments = new HashMap<>();
        }
        comments.put(path, Arrays.asList(lines));
    }

    public Map<String, List<String>> getAllComments() {
        return comments != null ? comments : Collections.emptyMap();
    }
}
