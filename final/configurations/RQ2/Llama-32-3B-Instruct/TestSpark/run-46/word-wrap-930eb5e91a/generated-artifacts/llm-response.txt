    private void wordWrap(String input, Function<CharSequence, Number> stringWidth, LineConsumer lineConsumer) {
        String[] words = input.split("\\s+");
        int width = (int) stringWidth.apply(words[0]);
        
        // Create a public method to perform the actual word wrap
        String wrappedString = wrapWords(input, width);
        return wrappedString;
    }

    private String wrapWords(String text, int width) {
        StringBuilder result = new StringBuilder();
        List<String> words = Arrays.asList(text.split("\\s+"));
        
        for (int i = 0; i < words.size(); i++) {
            String word = words.get(i);
            if (result.length() + word.length() + 1 > width) {
                // start a new line
                result.append("\n");
            }
            
            // add the rest of the line
            while (word.length() > width) {
                result.append(word.substring(0, width));
                result.append("\n");
                word = word.substring(width);
            }
            
            if (!result.toString().isEmpty()) {  // don't append a newline after the last word.
                result.append(" ");
            }
        }

        return result.toString();
    }
}
