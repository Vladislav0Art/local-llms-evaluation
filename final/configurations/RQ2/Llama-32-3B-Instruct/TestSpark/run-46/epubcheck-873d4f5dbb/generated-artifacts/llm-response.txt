public class OPFChecker {
    private ValidationContext validationContext;

    public OPFChecker(ValidationContextBuilder builder) {
        this.validationContext = builder.build();
    }

    public boolean isAudioType(String mediaType) {
        return validationContext.isMediaTypeAllowed(mediaType, MediaTypes.AUDIO);
    }

    public boolean isVideoType(String mediaType) {
        return validationContext.isMediaTypeAllowed(mediaType, MediaTypes.VIDEO);
    }
}

public class MediaTypes {
    private static final Set<String> AUDIO_TYPES = Set.of("audio/*");
    private static final Set<String> VIDEO_TYPES = Set.of("video/*");

    public static boolean isAllowed(String mediaType) {
        return (AUDIO_TYPES.contains(mediaType) || VIDEO_TYPES.contains(mediaType));
    }
}

public class ValidationContextBuilder {
    private List<MediaRule> rules = new ArrayList<>();

    public ValidationContext build() {
        ValidationContext validationContext = new ValidationContext();
        for (MediaRule rule : rules) {
            validationContext.addRule(rule);
        }
        return validationContext;
    }

    public static ValidationContextBuilder createDefault() {
        return new ValidationContextBuilder()
                .addMediaRule(new MediaRule())
                .build();
    }

    public ValidationContextBuilder addMediaRule(MediaRule rule) {
        this.rules.add(rule);
        return this;
    }
}

public class ValidationContext {
    private List<MediaRule> rules = new ArrayList<>();

    public void addRule(MediaRule rule) {
        this.rules.add(rule);
    }

    public boolean isMediaTypeAllowed(String mediaType, MediaTypes.MediaTypes mediaType) {
        for (MediaRule rule : rules) {
            if (rule.isAllowed(mediaType)) {
                return true;
            }
        }
        return false;
    }
}
