```java
import java.io.BufferedReader;
import java.io.DataOutputStream;
import java.io.IOException;
import java.io.InputStreamReader;
import java.net.HttpURLConnection;
import java.net.URL;

public class HttpClient {

    private URL url;

    public HttpClient setUrl(String url) {
        this.url = new URL(url);
        return this;
    }

    public Response get() throws IOException {
        HttpURLConnection connection = (HttpURLConnection) url.openConnection();
        connection.setRequestMethod("GET");
        int responseCode = connection.getResponseCode();

        if (responseCode >= 200 && responseCode < 300) {
            BufferedReader in = new BufferedReader(new InputStreamReader(connection.getInputStream()));
            String inputLine;
            StringBuffer response = new StringBuffer();

            while ((inputLine = in.readLine()) != null) {
                response.append(inputLine);
            }
            in.close();

            return new Response(response.toString(), true);
        } else {
            return new Response("Error: " + responseCode, false);
        }
    }

    public Response post(Map<String, Object> data) throws IOException {
        HttpURLConnection connection = (HttpURLConnection) url.openConnection();
        connection.setRequestMethod("POST");
        connection.setDoOutput(true);

        if (data != null) {
            DataOutputStream out = new DataOutputStream(connection.getOutputStream());
            out.write((data.toString()).getBytes());
            out.flush();
            out.close();

            int responseCode = connection.getResponseCode();

            if (responseCode >= 200 && responseCode < 300) {
                BufferedReader in = new BufferedReader(new InputStreamReader(connection.getInputStream()));
                String inputLine;
                StringBuffer response = new StringBuffer();

                while ((inputLine = in.readLine()) != null) {
                    response.append(inputLine);
                }
                in.close();
                return new Response(response.toString(), true);
            } else {
                return new Response("Error: " + responseCode, false);
            }
        }

        return new Response("No data sent", false);
    }

    public void setHeaders(Map<String, String> headers) throws IOException {
        if (headers != null) {
            for (Map.Entry<String, String> entry : headers.entrySet()) {
                connection.setRequestProperty(entry.getKey(), entry.getValue());
            }
        }
    }

    public class Response {
        private String response;
        private boolean postSuccessful;

        public Response(String response, boolean postSuccessful) {
            this.response = response;
            this.postSuccessful = postSuccessful;
        }

        public String getResponse() {
            return response;
        }

        public void setResponse(String response) {
            this.response = response;
        }

        public boolean isPostSuccessful() {
            return postSuccessful;
        }
    }

    public static class Main {
        public static void main(String[] args) throws IOException {
            HttpClient client = new HttpClient();
            client.setUrl("http://example.com");

            Response response = client.get();

            System.out.println(response.getResponse());
            System.out.println("Is Post Successful? " + response.isPostSuccessful());

            Map<String, Object> data = new HashMap<>();
            data.put("key", "value");
            response = client.post(data);

            System.out.println(response.getResponse());
            System.out.println("Is Post Successful? " + response.isPostSuccessful());

            // Set headers
            Map<String, String> headers = new HashMap<>();
            headers.put("Content-Type", "UTF_8");

            client.setHeaders(headers);
            response = client.post();

            System.out.println(response.getResponse());
            System.out.println("Is Post Successful? " + response.isPostSuccessful());
        }
    }
}
```

```java
import okhttp3.*;
import okhttp3.logging.HttpLoggingInterceptor;
import java.io.IOException;

public class HttpClient {

    private final OkHttpClient client;

    public HttpClient() {
        // Create a new instance of OkHttpClient
        client = new OkHttpClient.Builder()
                .build();
    }

    public Response get(String url) throws IOException {
        // Set the URL for the request
        Request request = new Request.Builder()
                .url(url)
                .get()
                .build();

        // Send the GET request and return the response
        return client.newCall(request).execute();
    }

    public Response post(Map<String, Object> data, String url) throws IOException {
        // Set the URL for the request
        RequestBody body = RequestBody.create(data != null ? json(data) : null, "application/json");
        Request request = new Request.Builder()
                .url(url)
                .post(body)
                .build();

        // Send the POST request and return the response
        return client.newCall(request).execute();
    }

    public Response post(String url, Map<String, Object> data) throws IOException {
        // Set the URL for the request
        RequestBody body = RequestBody.create(data != null ? json(data) : null, "application/json");
        Request request = new Request.Builder()
                .url(url)
                .post(body)
                .build();

        // Send the POST request and return the response
        return client.newCall(request).execute();
    }

    public Response post(String url, String data) throws IOException {
        // Set the URL for the request
        RequestBody body = RequestBody.create(data, "application/json");
        Request request = new Request.Builder()
                .url(url)
                .post(body)
                .build();

        // Send the POST request and return the response
        return client.newCall(request).execute();
    }

    private MediaType JSON = MediaType.get("application/json; charset=utf-8");

    private String json(Map<String, Object> data) {
        StringBuilder sb = new StringBuilder();
        sb.append("{");
        for (Map.Entry<String, Object> entry : data.entrySet()) {
            sb.append("\"").append(entry.getKey()).append("\":\"").append(entry.getValue().toString()).append("\"");
            if (entry.getValue() instanceof Map) {
                sb.append(",");
            }
        }
        sb.append("}");
        return sb.toString();
    }

    public void setHeaders(Header[] headers) throws IOException {
        // Set the headers for the request
        Request request = new Request.Builder()
                .headers(headers)
                .build();

        // Send the request and return the response
        Response response = client.newCall(request).execute();
        System.out.println(response.headers());
    }

    public class Response {
        private final MediaType mediaType;
        private final String content;

        public Response(MediaType mediaType, String content) {
            this.mediaType = mediaType;
            this.content = content;
        }

        public String getContent() {
            return content;
        }
    }

    public static void main(String[] args) throws IOException {
        // Create a new instance of HttpClient
        HttpClient client = new HttpClient();

        // Set the URL for the request
        String url = "http://example.com";

        // Send the GET request and print the response
        Response response = client.get(url);
        System.out.println(response.getContent());

        // Send the POST request with a JSON body and print the response
        Map<String, Object> data = new HashMap<>();
        data.put("key", "value");
        response = client.post(url, data);
        System.out.println(response.getContent());
    }
}
```

*   The `HttpClient` class provides methods for making GET and POST requests.
*   The `get` method sends a GET request to the specified URL and returns a `Response` object containing the response content.
*   The `post` methods send POST requests with JSON bodies to the specified URLs and return `Response` objects containing the response content.
*   The `setHeaders` method sets headers for the request and prints them after sending the request.

This example demonstrates how to use the `HttpClient` class to make GET and POST requests with JSON bodies.
