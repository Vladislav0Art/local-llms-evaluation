public class Safelist {

    private static final Set<String> none = new HashSet<>();

    public static Safelist basic() {
        return new Safelist();
    }

    @Override
    protected boolean isSafeTag(String tag) {
        return true; // Always safe
    }

    public Safelist addTags(String tag) {
        if (isSafeTag(tag)) {
            none.add(tag);
        }
        return this;
    }

    public Set<String> getAddedTags() {
        return none;
    }

    public Set<String> getRemovedTags() {
        return new HashSet<>();
    }

    public Attributes getEnforcedAttributes(String tag) {
        return new Attributes();
    }
}
