import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNull;
import static org.mockito.ArgumentMatchers.anyString;
import static org.mockito.Mockito.verify;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class DocumentTest {

    @Mock
    private Element element;

    @Mock
    private Tag tag;

    @Mock
    private String content;

    @Test
    public void createShell_NullBaseUri_ReturnsNull() {
        when(element.constructor(anyString())).thenReturn(null);
        Document document = Document.createShell(null);
        assertNull(document);
    }

    @Test
    public void location_EmptyDocument_LocationReturnsEmptyString() {
        Document document = new Document("");
        assertEquals("", document.location());
    }

    @Test
    public void location_HtmlDocument_LocationReturnsHtmlLocation() {
        String html = "<html><body>Hello</body></html>";
        Document document = new Document(html);
        assertEquals(html, document.location());
    }

    @Test
    public void connection_EmptyDocument_ConnectionReturnsEmptyConnection() {
        Document document = new Document("");
        assertEquals(null, document.connection());
    }

    @Test
    public void title_NullTitle_ReturnsNull() {
        Document document = new Document();
        assertNull(document.title());
    }

    @Test
    public void title_SetTitle_ReturnsPreviousValue() {
        String previousTitle = "Hello World";
        when(element.title(anyString())).thenReturn(previousTitle);
        Document document = new Document();
        document.title("New Title");
        assertEquals(previousTitle, document.title());
    }

    @Test
    public void createElement_EmptyDocument_CreateElementReturnsEmptyTag() {
        Document document = new Document("");
        Element tag = document.createElement("");
        assertNotNull(tag);
        assertEquals("", tag.tagName());
    }

    @Test
    public void outerHtml_NullNode_ReturnsNull() {
        Document document = new Document();
        assertNull(document.outerHtml());
    }

    @Test
    public void text_NullNode_ReturnsNull() {
        Document document = new Document();
        assertNull(document.text(""));
    }

    @Test
    public void nodeName_EmptyDocument_NodeNameReturnsEmptyString() {
        Document document = new Document("");
        assertEquals("", document.nodeName());
    }

    @Test
    public void charset NullCharset ReturnsDefaultCharset() {
        when(element.charset(any())).thenReturn(DataUtil.DEFAULT.charset());
        Document document = new Document();
        assertEquals(DataUtil.DEFAULT.charset(), document.charset());
    }

    @Test
    public void updateMetaCharsetElement_DefaultValueIsFalse() {
        Document document = new Document();
        assertEquals(false, document.updateMetaCharsetElement());
    }

    @Test
    public void clone_NullDocument_ReturnsNull() {
        when(element.clone()).thenReturn(null);
        Document document = new Document();
        assertNull(document.clone());
    }

    @Test
    public void shallowClone_NullNode_ReturnsNull() {
        when(element.shallowClone()).thenReturn(null);
        Document document = new Document();
        assertNull(document.shallowClone());
    }
}

public class Document {
    private String content;
    private String baseUrl;

    // Getters and setters for content and baseUrl
    public String getContent() {
        return content;
    }

    public void setContent(String content) {
        this.content = content;
    }

    public String getBaseUrl() {
        return baseUrl;
    }

    public void setBaseUrl(String baseUrl) {
        this.baseUrl = baseUrl;
    }

    // Public method to check if a tag exists
    public boolean hasTag(Element tag) {
        // Implementation for hasTag method
        return false;
    }

    // Public method to check charset
    public String getCharset() {
        return "utf-8";
    }
}

public class DocumentGenerator {
    public static Document generateDocument(String content, String baseUrl) {
        // Implementation for generateDocument method
        return new Document();
    }
}

public class Element {
    // Getters and setters for element attributes
    private String tagName;
    private List<String> attributes;

    public String getTagName() {
        return tagName;
    }

    public void setTagName(String tagName) {
        this.tagName = tagName;
    }

    public List<String> getAttributes() {
        return attributes;
    }

    public void setAttributes(List<String> attributes) {
        this.attributes = attributes;
    }
}

public class DataUtil {
    public static String DEFAULT_CHARSET;
    public static String DEFAULT_CHARSET_VALUE;

    static {
        DEFAULT_CHARSET = "utf-8";
        DEFAULT_CHARSET_VALUE = "default charset value";
    }

    // Public method to get charset
    public static String getDefaultCharset() {
        return DEFAULT_CHARSET;
    }
}

public class GeneratedTest {
    
    @Test
    public void testGeneratedDocument() throws Exception {
        Document document = new Document();
        String content = "Sample HTML Content";
        String baseUrl = "https://example.com";
        
        DocumentGenerator.generateDocument(content, baseUrl).setContent(content);
        DocumentGenerator.generateDocument(content, baseUrl).setBaseUrl(baseUrl);

        Element element = new Element();
        element.setTagName("html");
        element.getAttributes().add("lang");

        when(element.constructor(anyString())).thenReturn(element);
    }

    @Test
    public void testDocumentHasTitle() throws Exception {
        Document document = new Document();
        String content = "Sample HTML Content";
        String baseUrl = "https://example.com";

        DocumentGenerator.generateDocument(content, baseUrl).setContent(content);
        DocumentGenerator.generateDocument(content, baseUrl).setBaseUrl(baseUrl);

        Element element = new Element();
        element.setTagName("html");
        element.getAttributes().add("lang");

        document.setBaseUrl(baseUrl);
        
        when(element.title(anyString())).thenReturn("Sample HTML Title");

        assertEquals(document.getTitle(), "Sample HTML Title");
    }

    @Test
    public void testDocumentHasCharset() throws Exception {
        Document document = new Document();
        String content = "Sample HTML Content";
        String baseUrl = "https://example.com";

        DocumentGenerator.generateDocument(content, baseUrl).setContent(content);
        DocumentGenerator.generateDocument(content, baseUrl).setBaseUrl(baseUrl);

        Element element = new Element();
        element.setTagName("html");
        element.getAttributes().add("lang");

        when(element.charset()).thenReturn(DataUtil.DEFAULT_CHARSET_VALUE);
        
        assertEquals(DataUtil.getDefaultCharset(), document.getCharset());
    }

    @Test
    public void testDocumentHasTag() throws Exception {
        Document document = new Document();
        String content = "Sample HTML Content";
        String baseUrl = "https://example.com";

        DocumentGenerator.generateDocument(content, baseUrl).setContent(content);
        DocumentGenerator.generateDocument(content, baseUrl).setBaseUrl(baseUrl);

        Element element = new Element();
        element.setTagName("html");
        element.getAttributes().add("lang");

        when(element.constructor(anyString())).thenReturn(element);
        
        assertTrue(document.hasTag(element));
    }
}
