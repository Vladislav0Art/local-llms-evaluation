import org.jsoup.nodes.Comment;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.parser.ParseSettings;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.io.StringWriter;

@RunWith(MockitoJUnitRunner.class)
public class CommentTest {

    @Mock
    private Appendable accum;

    @Mock
    private Document.OutputSettings out;

    @InjectMocks
    private Comment comment;

    @Test
    public void givenCommentWhenGetNodeNameThenReturnsExpectedValue() {
        Mockito.when(comment.nodeName()).thenReturn("comment");
        String nodeName = comment.nodeName();
        assert nodeName.equals("comment");
    }

    @Test
    public void givenCommentWithTextWhenGetDataThenReturnsExpectedString() {
        comment.setData("Hello World");
        String data = comment.getData();
        assert data.equals("Hello World");
    }

    @Test
    public void givenCommentWithoutDataWhenGetDataThenReturnsEmptyString() {
        String data = comment.getData();
        assert data.isEmpty();
    }

    @Test
    public void givenCommentWhenGetOuterHtmlHeadWithDepth1AndAccumNotYetWrittenThenWrapsWithExpectedHtml() throws IOException {
        byte[] html = new byte[100];
        ByteArrayOutputStream outWriter = new ByteArrayOutputStream();
        Element element = new Element("div");
        comment.outerHtmlHead(outWriter, 1, out);
        Mockito.when(accum.write(Mockito.any(String.class))).thenReturn(html);
        String htmlWritten = outWriter.toString().trim();
        assert htmlWritten.startsWith("<div>");
    }

    @Test
    public void givenCommentWhenGetOuterHtmlTailWithDepth2AndAccumNotYetWrittenThenWrapsWithExpectedHtml() throws IOException {
        byte[] html = new byte[100];
        ByteArrayOutputStream outWriter = new ByteArrayOutputStream();
        Element element = new Element("div");
        comment.outerHtmlTail(outWriter, 2, out);
        Mockito.when(accum.write(Mockito.any(String.class))).thenReturn(html);
        String htmlWritten = outWriter.toString().trim();
        assert htmlWritten.startsWith("</div>");
    }

    @Test
    public void givenCommentWhenToStringThenReturnsExpectedString() {
        comment.setData("Hello World");
        String toString = comment.toString();
        assert toString.contains("<comment>Hello World</comment>");
    }

    @Test
    public void givenCommentWhenCloneThenReturnsNewCommentWithSameData() {
        Comment newComment = comment.clone();
        String data = newComment.getData();
        String originalData = comment.getData();
        assert data.equals(originalData);
    }

    @Test
    public void givenCommentIsXmlDeclarationWhenGetIsXmlDeclarationThenReturnsExpectedValue() {
        boolean isXmlDeclaration = comment.isXmlDeclaration();
        assert isXmlDeclaration;
    }

    @Test
    public void givenCommentNotWhenGetAsXmlDeclarationThenReturnsExpectedNull() {
        Comment newComment = new Comment("Hello World");
        XmlDeclaration xmlDeclaration = newComment.asXmlDeclaration();
        Mockito.when(out.isXmlDeclaration()).thenReturn(false);
        assert xmlDeclaration == null;
    }
}
