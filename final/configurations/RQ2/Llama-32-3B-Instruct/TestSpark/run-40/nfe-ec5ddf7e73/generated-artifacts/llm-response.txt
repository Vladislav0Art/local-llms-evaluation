public class DocumentFiscal {
    private String codigoInterno;
    private String placa;
    private String renavam;
    private String tara;
    private String capacidadeKG;
    private String capacidadeM3;

    public static final int CAPACIDADE_MAX = 5000;

    public DocumentFiscal() {}

    public void setCodigoInterno(String codigoInterno) {
        if (codigoInterno.length() != 8) {
            throw new IllegalArgumentException("O código interno deve ter exatamente 8 caracteres.");
        }
        this.codigoInterno = codigoInterno;
    }

    public String getCodigoInterno() {
        return codigoInterno;
    }

    public void setPlaca(String placa) {
        if (placa.length() != 3) {
            throw new IllegalArgumentException("A placa deve ter exatamente 3 caracteres.");
        }
        this.placa = placa;
    }

    public String getPlaca() {
        return placa;
    }

    public void setRenavam(String renavam) {
        if (renavam.length() != 6) {
            throw new IllegalArgumentException("O razonamento deve ter exatamente 6 caracteres.");
        }
        this.renavam = renavam;
    }

    public String getRenavam() {
        return renavam;
    }

    public void setTara(String tara) {
        if (tara.length() != 8) {
            throw new IllegalArgumentException("A tarifa deve ter exatamente 8 caracteres.");
        }
        this.tara = tara;
    }

    public String getTara() {
        return tara;
    }

    public void setCapacidadeKG(String capacidadeKG) {
        if (capacidadeKG.length() != 4) {
            throw new IllegalArgumentException("A capacidade de kg deve ter exatamente 4 caracteres.");
        }
        this.capacidadeKG = capacidadeKG;
    }

    public String getCapacidadeKG() {
        return capacidadeKG;
    }

    public void setCapacidadeM3(String capacidadeM3) {

            throw new IllegalArgumentException("A capacidade de m3 deve ter exatamente 5 caracteres.");

        this.capacidadeM3 = capacidadeM3;
    }

    public String getCapacidadeM3() {
        return capacidadeM3;

}
