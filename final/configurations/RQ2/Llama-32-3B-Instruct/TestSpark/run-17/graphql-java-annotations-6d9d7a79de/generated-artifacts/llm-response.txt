import graphql.annotations.processor.retrievers.fieldBuilders.DirectivesBuilder;
import graphql.schema.GraphQLArgument;
import graphql.schema.GraphQLDirective;
import graphql.schema.GraphQLScalarType;
import graphql.schema.GraphQLType;
import java.lang.reflect.AnnotatedElement;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class TestDirectivesBuilder {

    public static void testConstructWithAnnotatedElementAndContainer() {
        AnnotatedElement object = Mockito.mock(AnnotatedElement.class);
        ProcessingElementsContainer container = new ProcessingElementsContainer();
        DirectivesBuilder builder = new DirectivesBuilder(object, container);
        Assert.assertNotNull(builder);
    }

    public static void testBuildEmptyArrayWhenNoDirectivesAreFound() {
        AnnotatedElement object = Mockito.mock(AnnotatedElement.class);
        ProcessingElementsContainer container = new ProcessingElementsContainer();
        DirectivesBuilder builder = new DirectivesBuilder(object, container);
        List<GraphQLDirective> directives = new ArrayList<>();
        Assert.assertEquals(new ArrayList<>(), builder.build());
    }

    public static void testBuildArrayWithOneDirectiveFound() {
        AnnotatedElement object = Mockito.mock(AnnotatedElement.class);
        ProcessingElementsContainer container = new ProcessingElementsContainer();
        object = Mockito.when(object, () -> Arrays.asList(
                new Annotation()
                        .annotationType(GraphQLDirectives.class)
                        .name("DIRECTIVE_NAME")
                    )
                ));
        DirectivesBuilder builder = new DirectivesBuilder(object, container);
        GraphQLDirective directive = newDirective().withName("DIRECTIVE_NAME").build();
        List<GraphQLDirective> directives = Arrays.asList(directive);
        Assert.assertEquals(directories, builder.build());
    }

    public static void testBuildArrayWithMultipleDirectivesFound() {
        AnnotatedElement object = Mockito.mock(AnnotatedElement.class);
        ProcessingElementsContainer container = new ProcessingElementsContainer();
        object = Mockito.when(object, () -> Arrays.asList(
                new Annotation()

                        .name("DIRECTIVE_NAME_1")
                    ),
                new Annotation()
                        .annotationType(GraphQLDirectives.class)
                        .name("DIRECTIVE_NAME_2")
                    )
                ));
        DirectivesBuilder builder = new DirectivesBuilder(object, container);
        GraphQLDirective directive1 = newDirective().withName("DIRECTIVE_NAME_1").build();
        GraphQLDirective directive2 = newDirective().withName("DIRECTIVE_NAME_2").build();
        List<GraphQLDirective> directives = Arrays.asList(directive1, directive2);
        Assert.assertEquals(directories, builder.build());
    }

    public static void testBuildArrayWithNoDirectivesFoundWhenNoArgumentExists() {
        AnnotatedElement object = Mockito.mock(AnnotatedElement.class);
        ProcessingElementsContainer container = new ProcessingElementsContainer();
        GraphQLArgument argument = Mockito.mock(GraphQLArgument.class);
        DirectivesBuilder builder = new DirectivesBuilder(object, container);
        List<GraphQLDirective> directives = new ArrayList<>();
        Assert.assertEquals(directories, builder.build());
    }
