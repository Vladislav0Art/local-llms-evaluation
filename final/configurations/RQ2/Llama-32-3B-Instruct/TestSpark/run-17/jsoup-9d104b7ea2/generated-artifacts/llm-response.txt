public class StringUtil {

    public static String normalizeWhitespace(String input) {
        return input.replaceAll("\\s+", " ");
    }

        if (builder.length() == 0) {
            return true;
        }
        char lastChar = builder.charAt(builder.length() - 1);
        return Character.isWhitespace(lastChar);
    }

public class StringUtil {

    public static String normalizeWhitespace(String input) {
        return input.replaceAll("\\s+", " ");
    }

    public static boolean lastCharIsWhitespace(StringBuilder builder) {
        if (builder.length() == 0) {
            return true;
        }
        char lastChar = builder.charAt(builder.length() - 1);
        return Character.isWhitespace(lastChar);
    }

    public static String getTextContent(Element element) {
        return element.text();
    }

    public static Document stripLeadingWhitespace(Document document) {
        Element element = document.body().child(0);
        StringBuilder sb = new StringBuilder(element.text());
        while (sb.length() > 0 && Character.isWhitespace(sb.charAt(0))) {
            sb.deleteCharAt(0);
        }
        return Jsoup.parse(sb.toString());
    }

    public static Document removeEmptyElements(Document document) {
        Element element = document.body().child(0);
        if (element != null && element.children().size() == 0) {
            return document;
        } else {
            return Jsoup.parse(element.toString());
        }
    }
}

    public static void main(String[] args) {
