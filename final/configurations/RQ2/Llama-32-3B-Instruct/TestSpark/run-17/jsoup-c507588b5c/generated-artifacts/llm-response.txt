```java
import org.junit.Test;
import static org.junit.Assert.*;
import java.io.ByteArrayOutputStream;
import java.io.IOException;

public class TextNodeTest {

    @Test
    public void createTextNodesFromEncodedStringReturnsNonNull() {
        assertTrue(TextNode.createFromEncoded("hello").isPresent());
    }

    @Test
    public void createTextNodesFromEncodedStringThrowsIOExceptionOnInvalidInput() {
        assertThrows(IOException.class, () -> TextNode.createFromEncoded(null));
    }

    @Test
    public void textOfEmptyTextIsBlank() {
        assertTrue(TextNode.createFromEncoded("").text().isBlank());
    }

    @Test
    public void textOfNonBlankStringIsNotBlank() {
        assertTrue(TextNode.createFromEncoded("hello").text().isPresent());
    }

    @Test
    public void cloneMethodReturnsSameTextNode() {
        TextNode original = TextNode.createFromEncoded("hello");
        TextNode cloned = original.clone();
        assertEquals(original, cloned);
    }

    @Test
    public void splitTextMethodChangesTextNodeContent() {
        TextNode original = TextNode.createFromEncoded("hello");
        int offset = 1;
        TextNode split = original.splitText(offset);
        assertEquals("hello".substring(0, offset), original.text());
        assertTrue(split.text().isPresent());
    }

    @Test
    public void outerHtmlHeadAndTailDoNotAlterTextNode() {
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        Appendable accum = out;
        int depth = 1;
        Document.OutputSettings outSettings = new Document.OutputSettings();
        TextNode node = TextNode.createFromEncoded("hello");
        node.outerHtmlHead(accum, depth, outSettings);
        assertNotEquals("", accum.toString());
        assertEquals("hello", node.text());

        node.outerHtmlTail(accum, depth, outSettings);
        assertNotEquals("", out.toString());
    }

    @Test
    public void toStringMethodReturnsCorrectString() {
        TextNode node = TextNode.createFromEncoded("hello");
        assertEquals("<text>Hello</text>", node.toString());
    }

    @Test
    public void isBlankMethodReturnsTrueForEmptyString() {
        assertTrue(TextNode.createFromEncoded("").isBlank());
    }

    @Test
    public void isBlankMethodReturnsFalseForNonBlankString() {
        assertFalse(TextNode.createFromEncoded("hello").isBlank());
    }
}
```
