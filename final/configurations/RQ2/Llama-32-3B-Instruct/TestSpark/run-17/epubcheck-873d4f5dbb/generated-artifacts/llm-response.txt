import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import java.util.Set;

import static org.junit.Assert.*;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class OPFChecker30Tests {

    @Mock
    private ValidationContext context;

    @Mock
    private OPFHandler opfHandler;

    @Test
    public void initHandler_ShouldInitializeHandler() {
        when(context.getOPFChecker()).thenReturn(new OPFChecker30(context));
        new OPFChecker30(context).initHandler();
    }

    @Test
    public void checkPackage_WhenIsEPUB3_ShouldReturnTrue() {
        Set<String> packageNames = Set.of("application/oebps-package", "application/vnd.oebps-package");
        when(context.getOPFName()).thenReturn(packageNames);
        assertTrue(new OPFChecker30(context).checkPackage());
    }

    @Test
    public void checkPackage_WhenIsEPUB2_ShouldReturnFalse() {
        Set<String> packageNames = Set.of("application/oebps-package", "package");
        when(context.getOPFName()).thenReturn(packageNames);
        assertFalse(new OPFChecker30(context).checkPackage());
    }

    @Test
    public void checkContent_WhenIsEPUB3_ShouldReturnTrue() {
        Set<String> contentTypes = Set.of("application/x-dtd+xml", "text/xml");
        when(context.getContentType()).thenReturn(contentTypes);
        assertTrue(new OPFChecker30(context).checkContent());
    }

    @Test
    public void checkContent_WhenIsEPUB2_ShouldReturnFalse() {
        Set<String> contentTypes = Set.of("application/disk-image", "text/plain");
        when(context.getContentType()).thenReturn(contentTypes);
        assertFalse(new OPFChecker30(context).checkContent());
    }

    @Test
    public void checkItem_WhenIsAudioType_ShouldReturnTrue() {
        String type = "audio/mpeg";
        assertTrue(OPFChecker30.isAudioType(type));
    }

    @Test
    public void checkItem_WhenIsNotAudioType_ShouldReturnFalse() {
        String type = "text/html";
        assertFalse(OPFChecker30.isAudioType(type));
    }

    @Test
    public void isBlessedAudioType_ShouldReturnTrueForMp3() {
        String type = "audio/mpeg-3.1";
        assertTrue(OPFChecker30.isBlessedAudioType(type));
    }

    @Test
    public void isBlessedVideoType_ShouldReturnTrueForMp4() {
        String type = "video/mp4";
        assertTrue(OPFChecker30.isBlessedVideoType(type));
    }

    @Test
    public void getPreferredMediaType_ShouldReturnHtml() {
        String type = "text/html";
        when(opfHandler.getPreferredContentFormat()).thenReturn("html");
        assertEquals("html", OPFChecker30.getPreferredMediaType(type, ""));
    }

    @Test
    public void isFontType_ShouldReturnTrueForWoff2() {
        String type = "application/woff2";
        assertTrue(OPFChecker30.isFontType(type));
    }
}
