package ch.jalu.configme.configurationdata;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import java.util.Arrays;
import java.util.List;
import java.util.Map;

import static org.junit.Assert.*;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class CommentsConfigurationTest {

    @Mock
    private SettingsHolder settingsHolder;

    @InjectMocks
    private CommentsConfiguration commentsConfiguration;

    @Test
    public void defaultConstructor() {
        assertNotNull(commentsConfiguration);
    }

    @Test
    public void constructorWithMapOfComments() {
        Map<String, List<String>> map = new HashMap<>();
        map.put("path1", Arrays.asList("comment1", "comment2"));
        commentsConfiguration = new CommentsConfiguration(map);
        assertNotNull(commentsConfiguration.getComment(map.keySet().iterator().next()));
    }

    @Test
    public void setComment() {
        String path = "path1";
        List<String> commentLines = Arrays.asList("comment1", "comment2");
        commentsConfiguration.setComment(path, commentLines);
        assertEquals(commentLines, commentsConfiguration.getComment(path));
    }

    @Test
    public void setComment_EmptyLineResultingInEmptyLine() {
        String path = "path1";
        String[] commentLines = new String[]{ "", "comment2", "" };
        commentsConfiguration.setComment(path, commentLines);
        List<String> result = commentsConfiguration.getComment(path);
        assertEquals(2, result.size());
    }

    @Test
    public void getAllComments() {
        Map<String, List<String>> expectedMap = new HashMap<>();
        expectedMap.put("path1", Arrays.asList("comment1", "comment2"));
        when(settingsHolder.getSettings()).thenReturn(expectedMap);
        assertNotNull(commentsConfiguration.getAllComments());
        assertEquals(expectedMap, commentsConfiguration.getAllComments());
    }

    @Test
    public void getAllComments_EmptyMap() {
        Map<String, List<String>> expectedMap = Collections.emptyMap();
        when(settingsHolder.getSettings()).thenReturn(expectedMap);
        assertNotNull(commentsConfiguration.getAllComments());
        assertEquals(expectedMap, commentsConfiguration.getAllComments());
    }
}
