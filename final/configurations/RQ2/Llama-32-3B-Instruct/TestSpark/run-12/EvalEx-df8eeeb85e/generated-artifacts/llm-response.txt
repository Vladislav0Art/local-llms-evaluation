public class ExpressionParser {
    private Token token;
    private int position = 0;

    public ExpressionParser(Token token) {
        this.token = token;
    }

    public Token parseExpression() throws ParseException {
        Token leftOperand = parseTerm();
        while (token.type == TokenType.PlusOperatorToken) {
            token = nextToken();
            Token rightOperand = parseTerm();
            if (token.type != TokenType.PlusOperatorToken || !token.value.equals("+")) {
                throw new ParseException("Expected +, but found " + token.value);
            }
        }
        return leftOperand;
    }

    private Token parseTerm() throws ParseException {
        Token leftOperand = parseFactor();
        while (token.type == TokenType.MultiplyOperatorToken) {
            token = nextToken();
            Token rightOperand = parseFactor();
            if (token.type != TokenType.MultiplyOperatorToken || !token.value.equals("*")) {
                throw new ParseException("Expected *, but found " + token.value);
            }
        }
        return leftOperand;
    }

    private Token parseFactor() throws ParseException {
        // Check for parentheses
        if (token.type == TokenType.LeftParenthesisToken) {
            token = nextToken();
            Token result = parseExpression();
            token = nextToken(); // Eat the closing parenthesis
            return result;
        }
        // Check for variable or function call
        else if (token.type == TokenType.VariableToken || token.type == TokenType.FunctionToken) {
            return token;
        } else {
            throw new ParseException("Expected a term, but found " + token.value);
        }
    }

    private Token nextToken() throws ParseException {
        String value = token.value;
        if (position < value.length()) {
            token.value = value.substring(position);
            position += 1;
            return token;
        } else {
            token.type = TokenType.EmptyToken;
            throw new ParseException("Unexpected end of input");
        }
    }

        public ParseException(String message) {
            super(message);
        }
