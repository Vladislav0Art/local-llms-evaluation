import org.json.JSONObject;

public class Gt06ProtocolDecoder {

    private String channel;
    private Address remoteAddress;
    private String message;

    public Gt06ProtocolDecoder(String channel, Address remoteAddress, String message) {
        this.channel = channel;
        this.remoteAddress = remoteAddress;
        this.message = message;
    }

    public Object decode() throws Exception {
        if (channel.equals("test-channel") && remoteAddress != null) {
            try {
                JSONObject jsonObject = new JSONObject(message);
                return jsonObject.toString();
            } catch (Exception e) {
                System.out.println("Error parsing message: " + e.getMessage());
                return null;
            }
        } else {
            System.out.println("Invalid channel or address.");
            return null;
        }
    }

    public static class Address {
        private String host;
        private int port;

        public Address(String host, int port) {
            this.host = host;
            this.port = port;
        }

        @Override
        public boolean equals(Object obj) {
            if (obj == this) {
                return true;
            }
            if (obj == null || getClass() != obj.getClass()) {
                return false;
            }
            Address address = (Address) obj;
            return host.equals(address.host) && port == address.port;
        }

        @Override
        public int hashCode() {
            int result = 17;
            result = 31 * result + ((String) host).hashCode();
            result = 31 * result + port;
            return result;
        }
    }
}
