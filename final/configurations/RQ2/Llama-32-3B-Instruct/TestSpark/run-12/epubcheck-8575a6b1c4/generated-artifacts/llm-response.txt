        // Set locale for EpubChecker
        checker.setLocale(locale);

        // Get actual locale of EpubChecker
        String actualLocale = checker.getLocale().toString();

        // Print header with EPUB version (1.0)
        System.out.println("EPUB Version: 1.0");

        // Load the file you want to validate as an EPUB file
        File epubFile = new File("path_to_your_epub_file.epub");
        FileInputStream fis = new FileInputStream(epubFile);

        try {
            // Check if the file is a valid EPUB file

            if (isValid) {
                System.out.println("The EPUB file is valid.");
            } else {
                System.out.println("The EPUB file is not valid.");

}
