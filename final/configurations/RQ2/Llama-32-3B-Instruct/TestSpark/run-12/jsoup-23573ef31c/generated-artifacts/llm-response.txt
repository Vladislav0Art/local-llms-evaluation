```java
// Document.java
public class Document {
    private String html;

    public Document(String html) {
        this.html = html;
    }

    public void parse() {
        // Simple HTML parser that extracts text from paragraphs
        int start = 0;
        while (start < html.length()) {
            int end = html.indexOf("<p>", start);
            if (end == -1) break;

            int textStart = html.indexOf(">", end) + 1;
            String paragraphText = html.substring(textStart, end);

            System.out.println(paragraphText.trim());
            start = textStart;
        }
    }

    public static void main(String[] args) {
        Document doc = new Document("<html><body><p>This is a paragraph.</p><p>This is another paragraph.</p></body></html>");
        doc.parse();
    }
}
```
