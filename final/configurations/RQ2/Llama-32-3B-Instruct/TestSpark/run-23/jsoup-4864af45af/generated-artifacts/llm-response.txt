import org.junit.Test;
import static org.junit.Assert.*;
import java.util.Arrays;
import org.jsoup.nodes.Attribute;
import org.jsoup.nodes.Element;
import org.jsoup.helper.Validate;
import org.jsoup.safety.Safelist;
import static org.mockito.Mockito.when;

public class SafelistTest {

    @Test
    public void none_SafeList_isCorrectlyCreated() {
        Safelist safeList = Safelist.none();
        assertTrue(safeList == null);
    }

    @Test
    public void simpleText_SimpleTagsAreAllowed() {
        Safelist safeList = Safelist.simpleText();
        assertTrue(safeList.isSafeTag("img"));
        assertFalse(safeList.isSafeTag("script"));
    }

    @Test
    public void basic_BasicTagsAndAttributesAreAllowed() {
        Safelist safeList = Safelist.basic();
        assertTrue(safeList.isSafeTag("a"));
        assertTrue(safeList.isSafeAttribute("href", null, new Attribute()));
        assertFalse(safeList.isSafeTag("script"));
        assertFalse(safeList.isSafeAttribute("src", null, new Attribute()));
    }

    @Test
    public void basicWithImages_ImagesAreAllowed() {
        Safelist safeList = Safelist.basicWithImages();
        assertTrue(safeList.isSafeTag("img"));
        assertFalse(safeList.isSafeTag("script"));
    }

    @Test
    public void relaxed_RelaxedTagsAndAttributesAreAllowed() {
        Safelist safeList = Safelist.relaxed();
        assertTrue(safeList.isSafeTag("div"));
        assertTrue(safeList.isSafeAttribute("id", null, new Attribute()));
        assertFalse(safeList.isSafeTag("script"));
        assertFalse(safeList.isSafeAttribute("src", null, new Attribute()));
    }

    @Test
    public void addTags_AddedTagsAreAllowed() {
        Safelist safeList = Safelist.addTags("div");
        assertTrue(safeList.isSafeTag("div"));
        assertFalse(safeList.isSafeTag("script"));
    }

    @Test
    public void removeTags_RemovedTagsAreNotAllowed() {
        Safelist safeList = Safelist.simpleText();
        when(Safelist.class.isSafeTag("removed")).thenReturn(false);
        Safelist safeList2 = safeList.removeTags("removed");
        assertFalse(safeList2.isSafeTag("removed"));
    }

    @Test
    public void addAttributes_AddedAttributesAreAllowed() {
        Safelist safeList = Safelist.addTags("div");
        when(Safelist.class.isSafeAttribute("id", null, new Attribute())).thenReturn(true);
        Safelist safeList2 = safeList.addAttributes("div", "id", "test");
        assertTrue(safeList2.getEnforcedAttributes("div").size() > 0);
    }

    @Test
    public void removeAttributes_RemovedAttributesAreNotAllowed() {
        Safelist safeList = Safelist.addTags("div");
        when(Safelist.class.isSafeAttribute("removed", null, new Attribute())).thenReturn(false);
        Safelist safeList2 = safeList.removeAttributes("div", "removed");
        assertTrue(!safeList2.getEnforcedAttributes("div").containsKey("removed"));
    }

    @Test
    public void addEnforcedAttribute_AddedEnforcedAttributeIsAllowed() {
        Safelist safeList = Safelist.addTags("div");
        when(Safelist.class.isSafeAttribute("data-test", null, new Attribute())).thenReturn(true);
        Safelist safeList2 = safeList.addEnforcedAttribute("div", "data-test", "test");
        assertEquals(1, safeList2.getEnforcedAttributes("div").size());
    }

    @Test
    public void removeEnforcedAttribute_RemovedEnforcedAttributeIsNotAllowed() {
        Safelist safeList = Safelist.addTags("div");
        when(Safelist.class.isSafeAttribute("data-test", null, new Attribute())).thenReturn(true);
        Safelist safeList2 = safeList.removeEnforcedAttribute("div", "data-test");
        assertEquals(0, safeList2.getEnforcedAttributes("div").size());
    }

    @Test
    public void preserveRelativeLinks_PreservingLinksWorksCorrectly() {
        when(Safelist.class.preserveRelativeLinks(true)).thenReturn(true);
        Safelist safeList = Safelist.relaxed();
        assertTrue(safeList.preserveRelativeLinks(true));
    }

    @Test
    public void addProtocols_AddedProtocolsAreAllowed() {
        Safelist safeList = Safelist.addTags("div");
        when(Safelist.class.isSafeAttribute("data-protocol", null, new Attribute())).thenReturn(true);
        Safelist safeList2 = safeList.addProtocols("div", "data-protocol", "https");
        assertEquals(1, safeList2.getEnforcedAttributes("div").size());
    }

    @Test
    public void removeProtocols_RemovedProtocolsAreNotAllowed() {
        Safelist safeList = Safelist.addTags("div");
        when(Safelist.class.isSafeAttribute("data-protocol", null, new Attribute())).thenReturn(true);
        Safelist safeList2 = safeList.removeProtocols("div", "data-protocol", "https");
        assertEquals(0, safeList2.getEnforcedAttributes("div").size());
    }
}
