import static org.junit.Assert.assertEquals;

public class CssFormatterTests {

        CssFormatter formatter = new CssFormatter();
        formatter.init(options, cfg);
    }

    public void initInvalidOptionsThrows() {

        options.put("invalid", "option");
        ConfigurationSource cfg = null;
        try {
            CssFormatter formatter = new CssFormatter();
            formatter.init(options, cfg);
            assert false;

            // expected
        }
    }

    @Test
    public void doFormatEmptyInputReturnsEmpty() throws Exception {
        String input = "";
        LineEnding ending = LineEnding.LF;
        CssFormatter formatter = new CssFormatter();
        assertEquals("", formatter.format(input, ending));
    }

    @Test
    public void doFormatNonEmptyInputReturnsCorrectOutput() throws Exception {

public class CssFormatter {

    private enum LineEnding {
        LF,
        CRLF
    }

    public String format(String input, LineEnding ending) {
        StringBuilder output = new StringBuilder();
        for (String s : input.split("\n")) {
            switch (ending) {
                case LF:
                    output.append(s).append("\n");
                    break;
                case CRLF:
                    output.append(s).append("\r\n");
                    break;
            }
        }
        return output.toString().trim();
    }
}
