package com.adobe.epubcheck.tool;

import org.junit.Test;
import static org.junit.Assert.*;
import java.io.File;
import java.util.HashMap;

public class EpubCheckerTest {

    @Test
    public void getLocale_DefaultLocale_ReturnsDefaultLocale() {
        EpubChecker checker = new EpubChecker();
        Locale locale = checker.getLocale();
        assertEquals(Locale.ENGLISH, locale);
    }

    @Test
    public void getLocale_CustomLocale_ReturnsCustomLocale() {
        EpubChecker checker = new EpubChecker();
        checker.setLocale(Locale.FRENCH);
        Locale locale = checker.getLocale();
        assertEquals(Locale.FRENCH, locale);
    }

    @Test
    public void run_EmptyArgs_ThrowsInvalidArgumentException() {
        EpubChecker checker = new EpubChecker();
        assertThrows(InvalidArgumentException.class, () -> checker.run(new String[0]));
    }

    @Test
    public void run_ValidArgs_ReturnsSuccessStatus() {
        EpubChecker checker = new EpubChecker();
        int status = checker.run(new String[]{"--epub", "path/to/epub"});
        assertEquals(0, status);
    }

    @Test
    public void validateFile ValidEpubFile_ReturnsZeroStatus() {
        EpubChecker checker = new EpubChecker();
        int status = checker.validateFile("path/to/valid.epub", EPUBVersion.V1_2, new Report(), new EPUBProfile());
        assertEquals(0, status);
    }

    @Test
    public void validateFile InvalidEpubFile_ReturnsNonZeroStatus() {
        EpubChecker checker = new EpubChecker();
        int status = checker.validateFile("path/to/invalid.epub", EPUBVersion.V1_2, new Report(), new EPUBProfile());
        assertTrue(status > 0);
    }
}
