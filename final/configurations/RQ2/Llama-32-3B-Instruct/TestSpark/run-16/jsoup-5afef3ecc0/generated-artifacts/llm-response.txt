public class UrlBuilder {
    private String url;

    public UrlBuilder(String url) {
        this.url = url;
    }

    public URL build() throws URISyntaxException {
        URI uri = new URI(url);
        return new URL(uri);
    }

    public void appendKeyVal(Connection.KeyVal kv) throws UnsupportedEncodingException {
        if (kv == null || kv.getKey() == null || kv.getValue() == null) {
            throw new UnsupportedEncodingException("Either key or value is missing");
        }
        StringBuilder query = new StringBuilder(url).replaceFirst("/$", "?");
        String[] keysAndValues = kv.getValue().split("&");
        for (String kvStr : keysAndValues) {
            if (!kvStr.isEmpty()) {
                String[] keyValue = kvStr.split("=");
                query.append(kv.getKey()).append("=").append(URLEncoder.encode(kvValue, UTF_8)).append("&");
            }
        }
        url = new URI(query.toString().replaceFirst("/$", "/")).toString();
    }

    public static void main(String[] args) throws URISyntaxException {
        UrlBuilder builder = new UrlBuilder("http://example.com");
        System.out.println(builder.build());

        builder.appendKeyVal(new Connection.KeyVal("key", "value"));
        System.out.println(builder.build());
    }
}

public class PublicKeyVal {
    private String key;
    private String val;

    public PublicKeyVal(String key, String val) {
        this.key = key;
        this.val = val;
    }

    @Override
    public boolean keyVal(String k, String v) {
        return k.equals(key);
    }
}

public class PublicKeyValImpl implements KeyVal {
    private String key;
    private String val;

    public PublicKeyValImpl(String key, String val) {
        this.key = key;
        this.val = val;
    }

    @Override
    public boolean keyVal(String k, String v) {
        return k.equals(key);
    }
}

public class PublicKeyValue implements KeyVal {
    private String key;
    private String val;

    public PublicKeyValue(String key, String val) {
        this.key = key;
        this.val = val;
    }

    @Override
    public boolean keyVal(String k, String v) {
        return k.equals(key);
    }
}

public class KeyVal {
    public static boolean keyVal(PublicKeyVal kv, String k, String v) {
        return kv.keyVal(k, v);
    }

    public static void main(String[] args) {
        PublicKeyVal kv1 = new PublicKeyVal("a", "b");
        System.out.println(KeyVal.keyVal(kv1, "a", "b")); // true
        System.out.println(KeyVal.keyVal(kv1, "c", "d")); // false

        KeyValImpl kv2 = new KeyValImpl("x", "y");
        System.out.println(KeyVal.keyVal(kv2, "x", "y")); // true
        System.out.println(KeyVal.keyVal(kv2, "z", "w")); // false

        PublicKeyValue kv3 = new PublicKeyValue("k", "l");
        System.out.println(KeyVal.keyVal(kv3, "k", "l")); // true
        System.out.println(KeyVal.keyVal(kv3, "m", "n")); // false
    }
}
