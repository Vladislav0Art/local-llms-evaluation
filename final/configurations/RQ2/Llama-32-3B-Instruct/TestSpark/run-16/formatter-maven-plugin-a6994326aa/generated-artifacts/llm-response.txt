public class CssFormatterTest {

    private CssFormatter cssFormatter;

    @Before
    public void setUp() {
        cssFormatter = new CssFormatter();
    }

    public static class InitWithValidOptionsAndConfigTest extends AbstractCssFormatterTest {
        @Override
        protected void doTest() throws Exception {
            Map<String, String> options = Collections.singletonMap("option", "value");
            ConfigurationSource cfg = new ConfigurationSource();
            cssFormatter.init(options, cfg);
            assertTrue(cssFormatter.isInitialized());
        }
    }

    public static class InitWithInvalidOptionsAndConfigTest extends AbstractCssFormatterTest {
        @Override
        protected void doTest() throws Exception {
            Map<String, String> options = Collections.emptyMap();
            ConfigurationSource cfg = new ConfigurationSource();
            cssFormatter.init(options, cfg);
            assertFalse(cssFormatter.isInitialized());
        }
    }

    public static class FormatWithValidCodeAndLineEndingTest extends AbstractCssFormatterTest {
        @Override
        protected void doTest() throws Exception {
            String code = "body { color: red; }";
            LineEnding lineEnding = LineEnding.Windows;
            String formattedCode = cssFormatter.doFormat(code, lineEnding);
            assertTrue(formattedCode.contains("color: red;"));
        }
    }

    public static class FormatWithInvalidCodeAndLineEndingTest extends AbstractCssFormatterTest {
        @Override
        protected void doTest() throws Exception {
            String code = " invalid css";
            LineEnding lineEnding = LineEnding.Windows;
            try {
                cssFormatter.doFormat(code, lineEnding);
                fail("IOException not thrown");
            } catch (IOException e) {
                assertEquals("Invalid CSS", e.getMessage());
            }
        }
    }

    public static class FormatWithEmptyCodeAndLineEndingTest extends AbstractCssFormatterTest {
        @Override
        protected void doTest() throws Exception {
            String code = "";
            LineEnding lineEnding = LineEnding.Windows;
            String formattedCode = cssFormatter.doFormat(code, lineEnding);
            assertTrue(formattedCode.isEmpty());
        }
    }

    public static class IsInitializedTest extends AbstractCssFormatterTest {
        @Override
        protected void doTest() throws Exception {
            assertTrue(cssFormatter.isInitialized());
            assertFalse(cssFormatter.isInitialized());
        }
    }

public class CssFormatterTest {

    private CssFormatter cssFormatter;

    @Before
    public void setUp() {
        cssFormatter = new MockedCssFormatter();
    }

    public static class InitWithValidOptionsAndConfigTest extends AbstractCssFormatterTest {
        @Override
        protected void doTest() throws Exception {
            Map<String, String> options = Collections.singletonMap("option", "value");
            ConfigurationSource cfg = new ConfigurationSource();
            cssFormatter.init(options, cfg);
            assertTrue(cssFormatter.options.equals(options));
            assertTrue(cssFormatter.cfg.equals(cfg));
        }
    }

    public static class InitWithInvalidOptionsAndConfigTest extends AbstractCssFormatterTest {
        @Override
        protected void doTest() throws Exception {
            Map<String, String> options = Collections.emptyMap();
            ConfigurationSource cfg = new ConfigurationSource();
            cssFormatter.init(options, cfg);
            assertNull(cssFormatter.options);
            assertNull(cssFormatter.cfg);
        }
    }

    public static class FormatWithValidCodeAndLineEndingTest extends AbstractCssFormatterTest {
        @Override
        protected void doTest() throws Exception {
            String code = "body { color: red; }";
            LineEnding lineEnding = LineEnding.Windows;
            String formattedCode = cssFormatter.doFormat(code, lineEnding);
            Mockito.when(cssFormatter.sacParser.format(cssFormatter.cssStyleSheetImpl, formattedCode)).thenReturn(formattedCode);
            assertTrue(formattedCode.contains("color: red;"));
        }
    }

    public static class FormatWithInvalidCodeAndLineEndingTest extends AbstractCssFormatterTest {
        @Override
        protected void doTest() throws Exception {
            String code = " invalid css";
            LineEnding lineEnding = LineEnding.Windows;
            Mockito.when(cssFormatter.sacParser.format(cssFormatter.cssStyleSheetImpl, code)).thenThrow(new RuntimeException("Invalid CSS"));
            try {
                cssFormatter.doFormat(code, lineEnding);
                fail("Expected IOException");
            } catch (IOException e) {
                assertEquals("Invalid CSS", e.getMessage());
            }
        }
    }

    public static class FormatWithEmptyCodeAndLineEndingTest extends AbstractCssFormatterTest {
        @Override
        protected void doTest() throws Exception {
            String code = "";
            LineEnding lineEnding = LineEnding.Windows;
            Mockito.when(cssFormatter.sacParser.format(cssFormatter.cssStyleSheetImpl, code)).thenReturn(code);
            assertEquals("", cssFormatter.doFormat(code, lineEnding));
        }
    }

    public static class IsInitializedTest extends AbstractCssFormatterTest {
        @Override
        protected void doTest() throws Exception {
            assertTrue(cssFormatter.isInitialized());
            assertFalse(cssFormatter.isInitialized());
        }
    }

}
