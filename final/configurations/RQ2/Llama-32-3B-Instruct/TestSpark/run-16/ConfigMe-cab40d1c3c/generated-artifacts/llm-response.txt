```java
import org.junit.Test;
import static org.junit.Assert.*;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

public class PropertyListBuilderTest {

    @Test
    public void addProperty_NoAdditionalProperties_ReturnsEmptyList() {
        // given:
        PropertyListBuilder builder = new PropertyListBuilder();
        List<Property<?>> properties = new ArrayList<>();

        // when:
        builder.add(properties);

        // then:
        assertEquals(0, builder.create().size());
    }

    @Test
    public void addProperty_AdditionalProperties_ReturnsCorrectlyFormedList() {
        // given:
        PropertyListBuilder builder = new PropertyListBuilder();
        List<Property<?>> properties = new ArrayList<>();
        properties.add(new Property<>("a", "b"));
        properties.add(new Property<>("c", "d"));

        // when:
        builder.add(properties);

        // then:
        assertEquals(2, builder.create().size());
    }

    @Test
    public void addProperty_ThrowsConfigMeException() {
        // given:
        PropertyListBuilder builder = new PropertyListBuilder();
        List<Property<?>> properties = null;

        // when and then:
        assertThrows(ConfigMeException.class, () -> builder.add(properties));
    }

    @Test
    public void create_ReturnsEmptyList() {
        // given:
        PropertyListBuilder builder = new PropertyListBuilder();

        // when:
        List<Property<?>> properties = builder.create();

        // then:
        assertTrue(properties.isEmpty());
    }

    @Test
    public void getRootEntries ReturnsCorrectMap() {
        // given:
        PropertyListBuilder builder = new PropertyListBuilder();
        Map<String, Object> expectedRootEntries = new LinkedHashMap<>();
        expectedRootEntries.put("a", "b");

        // when:
        Map<String, Object> rootEntries = builder.getRootEntries();

        // then:
        assertEquals(expectedRootEntries, rootEntries);
    }
}

public class PropertyListBuilder {
    private List<Property<?>> properties = new ArrayList<>();

    public void add(Property<?> property) {
        if (property == null) {
            throw new ConfigMeException("Property cannot be null");
        }
        this.properties.add(property);
    }

    public List<Property<?>> create() {
        return properties;
    }

    protected final Map<String, Object> getRootEntries() {
        return new LinkedHashMap<>();
    }
}
```

```java
public class Property {
    private String key;
    private String value;

    public Property(String key, String value) {
        this.key = key;
        this.value = value;
    }

    public String getKey() {
        return key;
    }

    public String getValue() {
        return value;
    }
}

public class Properties {
    public List<Property> properties = new ArrayList<>();

    public void addProperty(Property property) {
        properties.add(property);
    }

    public List<Property> getProperties() {
        return properties;
    }
}

public class PropertyListBuilder {
    private Properties properties;

    public PropertyListBuilder() {
        this.properties = new Properties();
    }

    public void addProperty(Property property) throws ConfigMeException {
        if (property == null) {
            throw new ConfigMeException("Property cannot be null");
        }
        properties.addProperty(property);
    }

    public Properties createProperties() {
        return properties;
    }

    public Properties getRootEntries() {
        return properties;
    }
}

public class ConfigMeException extends Exception {}
```
