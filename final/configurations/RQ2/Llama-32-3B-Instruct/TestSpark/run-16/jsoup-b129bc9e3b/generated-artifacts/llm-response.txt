import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.Mock;
import org.mockito.junit.jupiter.MockitoExtension;

import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.net.MalformedURLException;
import java.net.URL;
import java.util.Map;

@ExtendWith(MockitoExtension.class)
public class HttpConnectionTest {

    @Mock
    private SSLSocketFactory sslSocketFactoryMock;

    @Mock
    private HttpURLConnection httpsURLConnectionMock;

    @Test
    public void connectUrl_ReturnsHttpConnection() throws MalformedURLException {
        HttpConnection httpConnection = HttpConnection.connect(new URL("http://example.com"));
        Assertions.assertNotNull(httpConnection);
    }

    @Test
    public void connectString_ReturnsHttpConnection() {
        HttpConnection httpConnection = HttpConnection.connect("http://example.com");
        Assertions.assertNotNull(httpConnection);
    }

    @Test
    public void encodeUrl_ThrowsUnsupportedMimeTypeExceptionForNullUrl() throws MalformedURLException, UnsupportedMimeTypeException {
        Assertions.assertThrows(UnsupportedMimeTypeException.class, () -> HttpConnection.encodeUrl(null));
    }

    @Test
    public void newRequest_ReturnsNewHttpConnection() {
        HttpConnection httpConnection = HttpConnection.connect("http://example.com");
        Assertions.assertNotNull(httpConnection.newRequest());
    }

    @Test
    public void url_ThrowsNullPointerExceptionForNullUrl() {
        Assertions.assertThrows(NullPointerException.class, () -> HttpConnection.url(null));
    }

        Assertions.assertNotNull(httpConnection.proxy());

    public void proxy_ThrowsNullPointerExceptionForNullHostAndPort() {
        Assertions.assertThrows(NullPointerException.class, () -> HttpConnection.proxy(null, 0));
    }

    public void userAgent_ReturnsUserAgentHttpConnection() {
        HttpConnection httpConnection = HttpConnection.userAgent("Mozilla/5.0");
        Assertions.assertNotNull(httpConnection.userAgent());
    }

    public void timeout_ThrowsNullPointerExceptionForNullTimeout() {
        Assertions.assertThrows(NullPointerException.class, () -> HttpConnection.timeout(null));
    }

    public void maxBodySize_ReturnsMaxBodySizeHttpConnection() {

        Assertions.assertNotNull(httpConnection.maxBodySize());
    }

    public void followRedirects_ReturnsFollowRedirectsHttpConnection() {

        Assertions.assertNotNull(httpConnection.followRedirects());
    }

    @Test
    public void referrer_ThrowsNullPointerExceptionForNullReferrer() {
        Assertions.assertThrows(NullPointerException.class, () -> HttpConnection.referrer(null));
    }

    @Test
    public void method_ReturnsMethodHttpConnection() {

        Assertions.assertNotNull(httpConnection.method());
    }

    @Test
    public void ignoreHttpErrors_ThrowsNullPointerExceptionForNullIgnoreHttpErrors() {
        Assertions.assertThrows(NullPointerException.class, () -> HttpConnection.ignoreHttpErrors(null));
    }
