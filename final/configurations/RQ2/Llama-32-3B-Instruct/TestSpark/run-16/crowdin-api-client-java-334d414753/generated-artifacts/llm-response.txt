```java
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;

public class JacksonJsonTransformer {

    private ObjectMapper objectMapper;

    public JacksonJsonTransformer() {
        this.objectMapper = new ObjectMapper();
    }

    // Convert object to JSON string
    public String convertToJSON(Object obj) throws Exception {
        return objectMapper.writeValueAsString(obj);
    }

    // Parse JSON string to object
    public <T> T parseFromString(String jsonString, Class<T> clazz) throws Exception {
        ObjectNode jsonNode = objectMapper.readTree(jsonString);
        return objectMapper.convertValue(jsonNode, clazz);
    }
}
```
