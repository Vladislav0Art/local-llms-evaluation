package org.jsoup.helper;

import org.jsoup.Connection;
import org.jsoup.internal.StringUtil;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import java.io.UnsupportedEncodingException;
import java.net.IDN;
import java.net.MalformedURLException;
import java.net.URI;
import java.net.URISyntaxException;
import java.net.URL;
import java.net.URLDecoder;
import java.net.URLEncoder;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNull;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class UrlBuilderTest {

    @Mock
    private Connection connection;

    @Mock
    private URL url;

    @Mock
    private String key;
    @Mock
    private String value;

    public UrlBuilderTest() {
        new UrlBuilder(url);
    }

    /**
     * Test that the constructor is called with inputUrl.
     */
    @Test
    public void UrlBuilderConstructor_CalledWithInputUrl() throws Exception {
        when(url.toString()).thenReturn("http://test.com");
        assert url != null;
    }

    /**
     * Test that build method returns URL object with scheme and path of inputUrl.
     */
    @Test
    public void build_MethodReturnsCorrectedUrl() throws Exception {
        when(url.toURI()).thenReturn(new URI("http://test.com/path"));
        UrlBuilder builder = new UrlBuilder(url);
        URL result = builder.build();
        assertEquals("http://test.com", result.getScheme());
        assertEquals("path", result.getPath());
    }

    /**
     * Test that appendKeyVal method appends key-value pair correctly.
     */
    @Test
    public void appendKeyVal_MethodAppendsCorrectly() throws Exception {
        when(url.toURI()).thenReturn(new URI("http://test.com/path"));
        UrlBuilder builder = new UrlBuilder(url);
        Connection.KeyVal kv = new Connection.KeyVal(key, value);
        builder.appendKeyVal(kv);
        String expectedUrl = "http://test.com/path?key=value";
        assertEquals(expectedUrl, url.toString());
    }

    /**
     * Test that appendKeyVal method does not throw exception if no key is provided.
     */
    @Test
    public void appendKeyVal_MethodDoesNotThrowWhenNoKeyIsProvided() throws Exception {
        UrlBuilder builder = new UrlBuilder(url);
        Connection.KeyVal kv = null;
        try {
            builder.appendKeyVal(kv);
            assertNull(kv.getKey());
        } catch (UnsupportedEncodingException e) {
            fail();
        }
    }

    /**
     * Test that URLDecoder.decode method is called for key-value pairs.
     */
    @Test
    public void appendKeyVal_MethodCallsUrlDecoderCorrectly() throws Exception {
        when(url.toURI()).thenReturn(new URI("http://test.com/path"));
        UrlBuilder builder = new UrlBuilder(url);
        Connection.KeyVal kv = new Connection.KeyVal(key, value);
        String decodedValue = URLDecoder.decode(value, "UTF-8");
        assertEquals(decodedValue, kv.getValue());
    }

    /**
     * Test that build method returns null if no inputUrl is provided.
     */
    @Test
    public void UrlBuilderConstructor_ReturnsNullWhenNoInputUrlIsProvided() {
        assertNull(new UrlBuilder(null).build());
    }
}
