```java
    public String[] splitIntoTokens(String input) {
        String[] tokens = input.split("\\s+");
        return tokens;
    }
}

public class InputProcessor {
    public String processInput(String input, Tokenizer tokenizer) {
        String[] expectedTokens = tokenizer.splitIntoTokens(input);
        for (int i = 0; i < expectedTokens.length; i++) {
            if (!expectedTokens[i].isEmpty()) {
                return "Invalid input: expected space but found '" + expectedTokens[i] + "'";

        return "Valid input";
```

```java
public class Tokenizer {
    private String input;

    public Tokenizer(String input) {
        this.input = input;
    }

    public String[] splitIntoTokens() {
        return input.split("\\s+");
    }
}

public class InputProcessor {

    public static String processInput(Tokenizer tokenizer) {
        String result = "";
        try {
            String[] expectedTokens = tokenizer.splitIntoTokens();
            for (int i = 0; i < expectedTokens.length; i++) {
                if (!expectedTokens[i].isEmpty()) {
                    return "Invalid input: expected space but found '" + expectedTokens[i] + "'";
                }
            }
            result = "Valid input";
        } catch (Exception e) {
            System.out.println("An error occurred: " + e.getMessage());
        }
        return result;
    }

    public static void testEmptyInput() {
        Tokenizer tokenizer = new Tokenizer("");
        String result = processInput(tokenizer);
        System.out.println(result);
    }

    public static void testInvalidSpace() {
        Tokenizer tokenizer = new Tokenizer("test");
        String result = processInput(tokenizer);
        System.out.println(result);
    }

    public static void testValidInput() {
        Tokenizer tokenizer = new Tokenizer("test ");
        String result = processInput(tokenizer);
        System.out.println(result);
    }
}

public class InputProcessorTest {

    @org.junit.Test
    public void testEmptyInput() {
        InputProcessor.testEmptyInput();
    }

    @org.junit.Test
    public void testInvalidSpace() {
        InputProcessor.testInvalidSpace();
    }

    @org.junit.Test
    public void testValidInput() {
        InputProcessor.testValidInput();
    }
}
```
