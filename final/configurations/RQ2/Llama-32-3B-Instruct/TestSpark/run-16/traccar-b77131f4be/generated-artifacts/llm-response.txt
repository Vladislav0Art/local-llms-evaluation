import org.junit.Test;
import static org.junit.Assert.*;
import io.netty.buffer.ByteBuf;
import io.netty.buffer.Unpooled;
import java.net.SocketAddress;

public class GalileoProtocolDecoderTest {

    @Test
    public void decodeDecodeDecodeTest() throws Exception {
        Channel channel = Mockito.mock(Channel.class);
        SocketAddress remoteAddress = Unpooled.buffer(10).slice();
        Object msg = Unpooled.buffer(20).slice();

        byte[] data = {0x02, 0x01, 0x00, 0x09, 0x02};
        ByteBuf decodedMessage = GalileoProtocolDecoder.decode(channel, remoteAddress, msg);
        assertEquals(data, decodedMessage.array());
    }

    @Test
    public void decodeEmptyDataTest() throws Exception {
        Channel channel = Mockito.mock(Channel.class);
        SocketAddress remoteAddress = Unpooled.buffer(10).slice();
        Object msg = Unpooled.buffer(20).slice();

        ByteBuf emptyData = new ByteBuf(Unpooled.buffer(0));
        byte[] data = {0x02, 0x01, 0x00, 0x09, 0x02};
        GalileoProtocolDecoder.decode(channel, remoteAddress, msg);
        assertEquals(data, emptyData.array());
    }

    @Test
    public void decodeWrongProtocolVersionTest() throws Exception {
        Channel channel = Mockito.mock(Channel.class);
        SocketAddress remoteAddress = Unpooled.buffer(10).slice();
        Object msg = Unpooled.buffer(20).slice();

        byte[] data = {0x03, 0x01, 0x00, 0x09, 0x02};
        assertThrows(Exception.class, () -> GalileoProtocolDecoder.decode(channel, remoteAddress, msg));
    }

    @Test
    public void decodeDecodeDecodeDecodeMessage() throws Exception {
        Channel channel = Mockito.mock(Channel.class);

    public void decodeDecodeDecodeEmptyChannel() throws Exception {
        Channel channel = Mockito.mock(Channel.class);
        SocketAddress remoteAddress = Unpooled.buffer(10).slice();

        GalileoProtocolDecoder.decode(null, remoteAddress, null);
    }

    @Test
    public void decodeDecodeDecodeInvalidMsg() throws Exception {
        Channel channel = Mockito.mock(Channel.class);

        GalileoProtocolDecoder.decode(channel, remoteAddress, "invalid_msg");
