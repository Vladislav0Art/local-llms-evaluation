    private static Tag parseTag(String tagName) {
        Tag parsedTag = new Tag(tagName);

        int start = 0;  // Start index of the tag's content
        while (start < tagName.length() && tagName.charAt(start) == ' ') {  // Skip any whitespace before the first attribute
            start++;
        }

        if (start >= tagName.length()) {  // No attributes found
            return parsedTag;
        }

        int end = tagName.indexOf('>', start);  // Find the end of the tag's content

        if (end == -1) {  // No closing '>' found, assume it's a self-closing tag
            parsedTag.setSelfClosing(true);
        } else {
            String attributesString = tagName.substring(start + 1, end);  // Extract the tag's attributes
