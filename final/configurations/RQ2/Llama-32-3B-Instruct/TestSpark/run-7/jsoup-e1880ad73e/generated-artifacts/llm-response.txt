    private Connection connection;

    public void build_EmptyUrlReturnsNull() {
        Mockito.when(connection.getUrl()).thenReturn(null);
        UrlBuilder urlBuilder = new UrlBuilder(connection);
        URL result = urlBuilder.build();
        assertNull(result);
    }

    @Test
    public void buildValidUrlReturnsURL() throws MalformedURLException, URISyntaxException {
        Mockito.when(connection.getUrl()).thenReturn("https://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(connection);
        URL result = urlBuilder.build();
        assertEquals("https://example.com", result.toString());
    }

    @Test
    public void appendKeyVal_EmptyUrlReturnsSame() {
        UrlBuilder urlBuilder = new UrlBuilder(null);
        urlBuilder.appendKeyVal(Connection.KeyVal.of("key", "value"));
        URL result = urlBuilder.build();
        assertEquals(null, result);
    }

    @Test
    public void appendKeyValValidUrlAppendsQueryParam() throws MalformedURLException, URISyntaxException {
        Mockito.when(connection.getUrl()).thenReturn(new URI("https://example.com"));
        UrlBuilder urlBuilder = new UrlBuilder(connection);
        Connection.KeyVal kv = Connection.KeyVal.of("key", "value");
        urlBuilder.appendKeyVal(kv);
        URL result = urlBuilder.build();
        assertEquals(new URI("https://example.com?key=value"), result);
    }

    @Test
    public void appendKeyValInvalidUrlThrowsUnsupportedEncodingException() {
        UrlBuilder urlBuilder = new UrlBuilder(null);
        try {
            urlBuilder.appendKeyVal(Connection.KeyVal.of("key", "value"));
            fail("Expected UnsupportedEncodingException");
        } catch (UnsupportedEncodingException e) {
            // Expected
        }
    }

    @Test
    public void appendKeyValInvalidUrlParamThrowsURISyntaxException() {
        UrlBuilder urlBuilder = new UrlBuilder(null);
        try {
            Mockito.when(connection.getUrl()).thenReturn(new URI("https://example.com?key=value"));
