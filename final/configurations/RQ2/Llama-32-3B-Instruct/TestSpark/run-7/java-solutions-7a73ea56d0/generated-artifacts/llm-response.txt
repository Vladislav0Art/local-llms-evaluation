import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;

import java.util.Arrays;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertThrows;
import static org.mockito.ArgumentMatchers.anyInt;
import static org.mockito.ArgumentMatchers.eq;
import static org.mockito.Mockito.when;

@RunWith(org.junit.runners.JUnit4.class)
public class OnlineStockSpanTest {

    @Mock
    private Stack<Integer> stockStack;

    public OnlineStockSpanTest() {
        MockitoAnnotations.initMocks(this);
    }

    public void testNextPriceIsZeroForNoPrices() {
        when(stockStack.isEmpty()).thenReturn(true);

        int nextPrice = new OnlineStockSpan().next(0);

        assertEquals(1, nextPrice);
    }

    public void testNextPriceIncreasesWhenPriceIncreases() {
        Stack<Integer> stack = new Stack<>();
        stack.push(5);
        stack.push(6);
        when(stockStack.isEmpty()).thenReturn(true);
        when(stockStack.peek()).thenReturn(5);

        int nextPrice = new OnlineStockSpan().next(7);

        assertEquals(2, nextPrice);
    }

    public void testNextPriceDoesNotIncreaseWhenNoIncrease() {
        Stack<Integer> stack = new Stack<>();
        stack.push(1);
        stack.push(1);
        when(stockStack.isEmpty()).thenReturn(true);
        when(stockStack.peek()).thenReturn(1);

        int nextPrice = new OnlineStockSpan().next(0);

        assertEquals(1, nextPrice);
    }

    public void testCalculateSpansForEmptyArray() {
        int[] prices = {};
        int[] result = new OnlineStockSpan().calculateSpans(prices);

        assertArrayEquals(new int[]{}, result);
    }

    public void testCalculateSpansWithOnePrice() {
        int[] prices = {10};
        int[] result = new OnlineStockSpan().calculateSpans(prices);

        assertArrayEquals(new int[]{1}, result);
    }

    public void testCalculateSpansWithMultiplePrices() {
        int[] prices = {10, 20, 30};
        int[] result = new OnlineStockSpan().calculateSpans(prices);

        assertArrayEquals(new int[]{2, 3, 4}, result);
    }
}

public class OnlineStockSpan {
    
    private int openPositions = 0;
    
    public int[] calculateSpans(int[] prices) {
        
        int[] result = new int[prices.length];
        
        for (int i = 0; i < prices.length; i++) {
            if (i == 0 || prices[i] > prices[i - 1]) {
                openPositions++;
            }
            
            result[i] = openPositions;
        }
        
        return result;
    }

    public int next(int price) throws IllegalArgumentException {
        if (!stockStack.isEmpty()) {
            throw new IllegalArgumentException("Stack is not empty");
        } else {
            return result[0];
        }
    }
}
