import java.util.ArrayList;
import java.util.List;

// Element class represents an HTML element
class Element {
    private final String tag;
    private List<String> text = new ArrayList<>();
    private List<Element> children = new ArrayList<>();

    public Element(String tag) {
        this.tag = tag;
    }

    // Append text content to the element
    public void appendText(String text) {
        this.text.add(text);
    }

    // Append HTML content to the element
    public void appendHtml(String html) {
        if (html.contains("<")) {
            String tagName = "";
            String tagContent = "";
            int startIndex = 0;
            while ((startIndex = html.indexOf('<', startIndex)) != -1) {
                if (startIndex > 0 && !html.substring(startIndex - 1, startIndex).equals(">")) {
                    tagContent += html.substring(startIndex - 1, startIndex);
                }
                tagName = html.substring(startIndex + 1, html.indexOf('>', startIndex));
                Element childElement = new Element(tagName);
                childElement.appendHtml(html.substring(startIndex + 2, html.indexOf('>', startIndex)));
                this.children.add(childElement);
                startIndex += tagContent.length() + tagName.length() + 3;
                tagContent = "";
            }
            if (tagContent != "") {
                Element childElement = new Element(tag);
                childElement.appendText(tagContent);
                this.children.add(childElement);
            }
        } else {
            text.add(html);
        }
    }

    // Append a child element to the current element
    public void appendChild(Element child) {
        children.add(child);
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder("<" + tag + ">");
        if (!text.isEmpty()) {
            for (String textContent : text) {
                sb.append(textContent).append(" ");
            }
        }
        for (Element child : children) {
            sb.append(child.toString());
        }
        sb.append("</" + tag + ">");
        return sb.toString();
    }
}
