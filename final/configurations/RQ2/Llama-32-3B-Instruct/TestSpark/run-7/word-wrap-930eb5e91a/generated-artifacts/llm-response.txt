import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import java.io.BufferedReader;
import java.io.FileReader;
import java.io.IOException;
import java.nio.charset.Charset;
import java.util.Set;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertTrue;
import static org.mockito.ArgumentMatchers.any;
import static org.mockito.Mockito.*;

@RunWith(MockitoJUnitRunner.class)
public class WordWrapTest {

    @Mock
    private Reader reader;

    @Mock
    private Writer writer;

    @Mock
    private LineConsumer lineConsumer;

    @Test
    public void fromReader_ReturnsBuilder() {
        Preconditions.checkState(WordWrap.from(reader).isPresent());
    }

    @Test
    public void fromClasspathUtf8_ReturnsBuilder() {
        Preconditions.checkState(WordWrap.fromClasspathUtf8("resource").isPresent());
    }

    @Test
    public void fromClasspath_ReturnsBuilder() {
        Preconditions.checkState(WordWrap.fromClasspath("resource", StandardCharsets.UTF_8).isPresent());
    }

    @Test
    public void from CharSequence_ReturnsBuilder() {
        Preconditions.checkState(WordWrap.from("text").isPresent());
    }

    @Test
    public void fromUtf8Stream_ReturnsBuilder() {
        Preconditions.checkState(WordWrap.fromUtf8(new FileInputStream("file")).isPresent());
    }

    @Test
    public void wordWrapReaderWriterReturnsCorrectOutput() throws IOException {
        String input = "Hello World";
        WordWrap.wordWrap(reader, writer, "\n", 10L, s -> s.length(), new HashSet<>(), false, true);
        assertEquals(input + "\n", reader.toString());
    }

    @Test
    public void wordWrapReaderLineConsumerReturnsCorrectOutput() throws IOException {
        String input = "Hello World";
        WordWrap.wordWrap(reader, lineConsumer, 10L, s -> s.length(), new HashSet<>(), false, true);
        assertTrue(lineConsumer.isConsumed());
        assertEquals(input + "\n", lineConsumer.getConsumedText());
    }

    @Test
    public void rightTrimReturnCorrectOutput() {
        String input = "   Hello World";
        String expected = "Hello World";
        Preconditions.checkState(WordWrap.rightTrim(input).equals(expected));
    }

    @Test
    public void isWhitespaceReturnsCorrectResult() {
        assertTrue(WordWrap.isWhitespace("   "));
        assertTrue(WordWrap.isWhitespace("\t"));
        assertFalse(WordWrap.isWhitespace("HelloWorld"));
    }

    @Test
    public void leftTrimStringBuilder2ReturnCorrectOutput() {
        StringBuilder word = new StringBuilder();
        WordWrap.leftTrim(word);
        assertEquals(0, word.length());
    }
}
