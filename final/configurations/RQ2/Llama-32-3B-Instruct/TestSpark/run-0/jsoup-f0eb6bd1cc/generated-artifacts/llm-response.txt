package org.jsoup.helper;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;
import org.junit.runners.JUnit4;

public class UrlBuilder {

    private URL url;

    public UrlBuilder(URL url) {
        this.url = url;
    }

    public URL inputUrl() {
        return url;
    }

    public void appendKeyVal(Connection.KeyVal kv) throws UnsupportedEncodingException {
        if (kv.getValue() == null || kv.getValue().isEmpty()) {
            String key = kv.getKey();
            if (url.getProtocol() != null && !url.getProtocol().isEmpty()) {
                url.setProtocol(url.getProtocol());
            }
            url.getHost(kv.getValue());
            url.getPath("/");
        } else {
            String key = kv.getKey();
            url.setFragment(kv.getValue());
        }
    }

    public URL build() throws MalformedURLException, URISyntaxException {
        return url;
    }
}

class Connection {

    public static class KeyVal {
    }

}
