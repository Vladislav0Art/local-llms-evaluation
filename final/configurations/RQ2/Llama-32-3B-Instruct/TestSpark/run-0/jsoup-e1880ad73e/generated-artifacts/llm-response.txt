package org.jsoup.helper;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import javax.annotation.Nullable;
import java.io.UnsupportedEncodingException;
import java.net.IDN;
import java.net.MalformedURLException;
import java.net.URI;
import java.net.URISyntaxException;
import java.net.URL;
import java.net.URLDecoder;
import java.net.URLEncoder;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.fail;
import static org.jsoup.helper.DataUtil.UTF_8;
import static org.jsoup.helper.UrlBuilder.*;

public class UrlBuilderTest {

    @RunWith(MockitoJUnitRunner.class)
    public class UrlBuilderTest {

        @Mock
        private Connection connection;

        @Mock
        private URL url;

        @Test
        public void newUrlBuilderFromEmptyUrl() {
            UrlBuilder builder = new UrlBuilder(url);
            assertEquals(url, builder.build());
        }

        @Test
        public void newUrlBuilderFromNonEmptyUrl() {
            String expected = "http://example.com";
            UrlBuilder builder = new UrlBuilder(url);
            builder.appendKeyVal(connection.KeyVal("key", "value"));
            URI uri = url.toURI();
            assertEquals(expected, uri.toString());
        }

        @Test
        public void appendKeyValNoModification() {
            String expected = "http://example.com";
            UrlBuilder builder = new UrlBuilder(url);
            Mockito.when(connection.toNormalized()).thenReturn(URI.create("http://example.com"));
            URI uri = URLUtil.buildUri(uri, null);
            assertEquals(expected, uri.toString());
        }

        @Test
        public void appendKeyValModified() {
            String expected = "http://example.com?key=value";
            UrlBuilder builder = new UrlBuilder(url);
            Mockito.when(connection.toNormalized()).thenReturn(URI.create("http://example.com"));
            URI uri = URLUtil.buildUri(uri, new Connection.KeyVal("key", "value"));
            assertEquals(expected, uri.toString());
        }

        @Test
        public void appendKeyValNonEmptyValue() {
            String expected = "http://example.com?key=value";
            UrlBuilder builder = new UrlBuilder(url);
            Mockito.when(connection.toNormalized()).thenReturn(URI.create("http://example.com"));
            URI uri = URLUtil.buildUri(uri, new Connection.KeyVal("key", "value"));
            assertEquals(expected, uri.toString());
        }

        @Test
        public void appendKeyValEmptyValue() {
            String expected = "http://example.com";
            UrlBuilder builder = new UrlBuilder(url);
            Mockito.when(connection.toNormalized()).thenReturn(URI.create("http://example.com"));
            URI uri = URLUtil.buildUri(uri, new Connection.KeyVal("key", ""));
            assertEquals(expected, uri.toString());
        }

        @Test
        public void appendKeyValNonEncodedValue() {
            String expected = "http://example.com?key=value";
            UrlBuilder builder = new UrlBuilder(url);
            Mockito.when(connection.toNormalized()).thenReturn(URI.create("http://example.com"));
            URI uri = URLUtil.buildUri(uri, new Connection.KeyVal("key", UTF_8));
            assertEquals(expected, uri.toString());
        }

        @Test
        public void appendKeyValEncodedValue() {
            String expected = "http://example.com?key=%E9%99%8A";
            UrlBuilder builder = new UrlBuilder(url);
            Mockito.when(connection.toNormalized()).thenReturn(URI.create("http://example.com"));
            URI uri = URLUtil.buildUri(uri, new Connection.KeyVal("key", UTF_8));
            assertEquals(expected, uri.toString());
        }

        @Test
        public void appendKeyValUrlEncodedValue() {
            String expected = "http://example.com?key=value";
            UrlBuilder builder = new UrlBuilder(url);
            Mockito.when(connection.toNormalized()).thenReturn(URI.create("http://example.com"));
            URI uri = URLUtil.buildUri(uri, new Connection.KeyVal("key", URLEncoder.encode("value", UTF_8)));
            assertEquals(expected, uri.toString());
        }

        @Test
        public void appendKeyValMalformedUrlEncodedValue() {
            fail("URL encoded value should not be malformed");
        }
    }

    public static class URLUtil {

        public static URI buildUri(URI uri, Connection.KeyVal kv) throws URISyntaxException {
            if (kv != null) {
                String key = kv.getKey();
                String value = kv.getValue();
                if (value == null) {
                    return uri;
                }
                return new URI(uri.getScheme(), uri.getHost(), uri.getPath() + "?" + key + "=" + URLDecoder.decode(value, "UTF-8"), uri.getQuery());
            } else {
                return uri;
            }
        }

    }
}

public class GeneratedTest {

    @Mock

    public static class URLUtil {
        public static URI buildUri(URI uri, Object kv) throws URISyntaxException {
            if (kv != null) {
                String key = ((Connection.KeyVal) kv).getKey();
                String value = ((Connection.KeyVal) kv).getValue();
                if (value == null) {
                    return uri;
                }
                return new URI(uri.getScheme(), uri.getHost(), uri.getPath() + "?" + key + "=" + URLDecoder.decode(value, "UTF-8"), uri.getQuery());
            } else {
                return uri;
            }
        }

    }

    public static class ConnectionKeyVal {
        private String key;
        private String value;

        public ConnectionKeyVal(String key, String value) {
            this.key = key;
            this.value = value;
        }

        public String getKey() {
            return key;
        }

        public String getValue() {
            return value;
        }
    }

}

public class UrlBuilderTest {

    @Test
    public void testBuildUriConnectionKeyValValue() throws URISyntaxException {
        URI uri = new URI("http://localhost:8080", null, "/path/to/resource", null);

        URI result = URLUtil.buildUri(uri, kv);
        assertEquals(result.getScheme(), "http");
        assertEquals(result.getHost(), "localhost");
        assertEquals(result.getPath(), "/path/to/resource?key=value");

        assertEquals(result.getPath(), "/path/to/resource?key=");
