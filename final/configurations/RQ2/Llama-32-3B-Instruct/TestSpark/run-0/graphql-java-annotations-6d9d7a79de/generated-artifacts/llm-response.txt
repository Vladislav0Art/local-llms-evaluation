import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.Mock;
import org.mockito.Mockito;

import static org.junit.jupiter.api.Assertions.assertEquals;

public class GraphQLDirectiveParserTest {
    
    @Mock
    public String graphQLDirective;

    @Test
    void testGenerateClass() {
        // Arrange
        Mockito.when(graphQLDirective).thenReturn("class Test");
        
        // Act
        String generatedClass = GeneratePublicClass(graphQLDirective);
        
        // Assert
        assertEquals("public class Test", generatedClass);
    }
    
    @Test
    void testGenerateMethod() {
        // Arrange
        Mockito.when(graphQLDirective).thenReturn("method add(int param)");
        
        // Act
        String generatedMethod = GeneratePublicMethod(graphQLDirective);
        
        // Assert
        assertEquals("public int add(int param)", generatedMethod);
    }
    
    public static String GeneratePublicClass(String classDefinition) {
        StringBuilder generatedClass = new StringBuilder();
        generatedClass.append("public class ").append(classDefinition.split("\\s+")[1]).append(" {");
        return generatedClass.toString();
    }

    public static String GeneratePublicMethod(String methodDefinition) {
        String[] parts = methodDefinition.split("\\s+");
        StringBuilder generatedMethod = new StringBuilder();
        generatedMethod.append("public ").append(parts[0]);
        for (int i = 1; i < parts.length - 1; i++) {
            if (!parts[i].equals("add")) {
                generatedMethod.append(" ").append(parts[i]);
            }
        }
        generatedMethod.append("(");
        for (int i = 2; i < parts.length - 1; i++) {
            generatedMethod.append(parts[i]).append(",");
        }
        generatedMethod.deleteCharAt(generatedMethod.length() - 1);
        generatedMethod.append(")");
        return generatedMethod.toString();
    }

    @Test
    void testGenerateDirectives() {
        // Arrange
        Mockito.when(graphQLDirective).thenReturn("@directive");

        // Act
        List<String> generatedDirectives = GeneratePublicDirectives(graphQLDirective);

        // Assert
        assertEquals(1, generatedDirectives.size());
        assertEquals("@directive", generatedDirectives.get(0));
    }

    public static List<String> GeneratePublicDirectives(String directiveDefinition) {
        List<String> generatedDirectives = new ArrayList<>();
        generatedDirectives.add(directiveDefinition);
        return generatedDirectives;
    }
}
