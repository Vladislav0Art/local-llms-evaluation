    private java.util.List<String> patterns;
    private java.util.Set<String> enforcedPatterns;
    private Attributes defaultAttributes;

    public void addPattern(String pattern) {
        this.patterns.add(pattern);
    }

    public Safelist() {
        enforcedPatterns = new java.util.HashSet<>();
        defaultAttributes = null;
    }

    public java.util.List<String> getEnforcedAttributes() {
        return new java.util.ArrayList<>(enforcedPatterns);
    }
}

public class Attributes implements java.util.Map<String, String> {

    private final java.util.HashMap<String, String> map = new java.util.HashMap<>();

            "alt", "href", "src", "title"
    ));

    @Override
    public java.util.Map<String, String> getMap() {
        return map;
    }
}

public class Attribute implements java.util.Map.Entry<String, String> {

    private final String key;
    private final String value;

    public Attribute(String key, String value) {
        this.key = key;
        this.value = value;
    }

    @Override
    public String getKey() {
        return key;
    }

    @Override
    public String getValue() {
