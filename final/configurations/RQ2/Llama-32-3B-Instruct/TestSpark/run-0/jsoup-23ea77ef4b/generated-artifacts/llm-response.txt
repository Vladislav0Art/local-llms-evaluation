    private Document mockDocument;

        MockitoAnnotations.initMocks(this);
    }

    public void namespaceAware_ReturnsBoolean() {
        boolean result = W3CDom.namespaceAware();
        assertTrue(result);
    }

    public void namespaceAware_ToggleNamespaceAware_ReturnsBoolean() {

        assertTrue(result);

        result = W3CDom.namespaceAware(false);
        assertFalse(result);
    }

        Document actual = W3CDom.convert(mockDocument);
        assertEquals(expected, actual);
    }

    public void asString_ConvertsHtmlToXhtml() throws TransformerException {
        String expected = "<!-- test -->";
        ByteArrayOutputStream outContent = new ByteArrayOutputStream();

        Document document = builder.parse(new DOMSource("<!-- test -->"));
        Transformer transformer = TransformerFactory.newInstance().newTransformer();
        transformer.setOutputProperty(OutputKeys.INDENT, "yes");
        transformer.setOutputProperty(OutputKeys.METHOD, "xml");
        transformer.transform(new DOMSource(document), new StreamResult(outContent));
        String actual = W3CDom.asString(mockDocument, null);
        assertEquals(expected, actual);
    }
