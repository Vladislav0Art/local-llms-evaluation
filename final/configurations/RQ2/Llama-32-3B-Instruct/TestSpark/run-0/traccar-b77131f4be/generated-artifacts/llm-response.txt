    private byte[] buffer;

    public BitBuffer(int size) {
        buffer = new byte[size];
    }

    public byte getBit(int index) {
        return buffer[index];
    }

    public void setBit(int index, boolean value) {

        } else {
            buffer[index] = 0;
        }
    }
