        Document doc = Jsoup.parse(html);

        Elements headerElements = doc.getElementsByClass("header");

        for (Element element : headerElements) {
            String text = element.text();
            if (!text.isEmpty()) {
                System.out.println(text);
            }
        }

        // Get all links in the HTML document
        Elements links = doc.select("a");
        System.out.println("Number of links: " + links.size());
        
        for (Element link : links) {
            String url = link.attr("href");
            if (!url.isEmpty()) {
                System.out.println(url);
            }
        }

        // Get all comments in the HTML document
        Elements comments = doc.select("comment");
        System.out.println("Number of comments: " + comments.size());
        
        for (Element comment : comments) {

            if (!text.isEmpty()) {
