public class TextNode {
    private String text;

    public TextNode(String text) {
        this.text = text;
    }

    public String text() {
        return text;
    }

    public void setText(String text) {
        this.text = text;
    }

    public boolean isBlank() {
        return StringUtil.normaliseWhitespace(text).isEmpty();
    }

    public String getWholeText() {
        StringBuilder sb = new StringBuilder();
        if (text != null && !text.isEmpty()) {
            sb.append(text);
        }
        return sb.toString();
    }

    public String[] splitText(int offset) throws IOException {
        String[] result = text.split(new String[]{text.substring(offset, offset + 1)}, offset + 1);
        return result;
    }

    public TextNode clone() {
        return new TextNode(text);
    }

    public static TextNode createFromEncoded(String encodedText) {
        return new TextNode(encodedText);
    }

    public static boolean lastCharIsWhitespace(StringBuilder sb) {
        if (sb.length() == 0) {
            return false;
        }
        char c = sb.charAt(sb.length() - 1);
        return Character.isWhitespace(c);
    }

    public static String stripLeadingWhitespace(String text) {
        while (!text.isEmpty() && Character.isWhitespace(text.charAt(0))) {
            text = text.substring(1);
        }
        return text;
    }

    public static String normaliseWhitespace(String text) {
        return StringUtil.stripLeadingWhitespace StringUtil.trimWhitespace StringUtil.replaceWhitespace(text, " "));
    }
}
