public class Gt06ProtocolDecoder {

    /**
     * Decodes the GPS message.
     *
     * @param position The GPS position.
     * @param buf      The buffer containing the encoded data.
     * @param hasLength Whether the length of the data is present in the buffer.
     * @param timezone The timezone to use for the decoded timestamp.
     * @param data     The raw GPS data.
     * @return The decoded GPS message.
     */
    public static Object decodeGps(Position position, ByteBuf buf, boolean hasLength, TimeZone timezone, byte[] data) {
        // assuming gt06 protocol format is:
        // timestamp (4 bytes)
        // length (2 bytes)
        // gps_position (8 bytes)

        if (!hasLength) {
            throw new UnsupportedOperationException("GT06 protocol with no length field");
        }

        long timestamp = buf.getLong();
        short length = buf.getShort();

        byte[] gpsData = new byte[length * 4]; // assuming each gps position is 4 bytes
        buf.readBytes(gpsData);

        return new Gt06Message(timestamp, position, timezone, data, gpsData);
    }

    /**
     * Decodes the GT06 message.
     *
     * @param channel The channel that received the message.
     * @param remoteAddress The address of the remote endpoint.
     * @param msg      The raw message.
     * @return The decoded GT06 message.
     */
    public static Object decode(Channel channel, SocketAddress remoteAddress, Object msg) {
        if (!(msg instanceof Gt06Message)) {
            throw new UnsupportedOperationException("Unsupported message type");

        return ((Gt06Message) msg).decode(channel);
    }
