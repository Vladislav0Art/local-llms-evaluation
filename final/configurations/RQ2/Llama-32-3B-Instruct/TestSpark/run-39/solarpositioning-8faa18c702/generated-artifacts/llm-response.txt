        when(this.atDayOfMonth(this.of(2022, Month.DECEMBER), 1)).thenReturn(of(2022, Month.DECEMBER, 15));

        when(this.atDayOfMonth(this.of(2024, Month.DECEMBER), 31)).thenReturn(of(2024, Month.DECEMBER, 16));
