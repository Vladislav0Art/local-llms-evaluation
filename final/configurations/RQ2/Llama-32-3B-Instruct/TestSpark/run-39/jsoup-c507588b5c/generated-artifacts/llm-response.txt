public class TextNode {

    private String text;

    public TextNode(String text) {
        this.text = text;
    }

    public static TextNode createFromEncoded(String encodedText) {
        return new TextNode(encodedText);
    }

    public String text() {
        return StringUtil.trim(text);
    }

    public void setText(String text) {
        this.text = text;
    }

    public String getWholeText() {
        return text.replaceAll("[^\\w\\s]", "");
    }

    public boolean isBlank() {
        return text.isEmpty();
    }

    public TextNode clone() {
        return new TextNode(text);
    }

    public void outerHtmlHead(Appendable accum, int indentLevel, Document.OutputSettings outputSettings) {
        for (int i = 0; i < indentLevel; i++) {

        if (outputSettings != null) {

        } else {
            accum.append(text);
        }
        accum.append("</p>");
    }

        for (int i = 0; i < indentLevel; i++) {
            accum.append(' ');
        }
        accum.append("<p>");
        if (outputSettings != null) {
            // implement logic for output settings
        } else {
            accum.append(text);
        }
        accum.append("</p>");
    }

    public String toString() {
        return "<p>" + text.replace("\n", "</p><p>") + "</p>";
    }
}

public class StringUtil {

    public static String trim(String str) {
        int start = 0;
        while (start < str.length() && Character.isWhitespace(str.charAt(start))) {
            start++;
        }
        int end = str.length() - 1;
        while (end > start && Character.isWhitespace(str.charAt(end))) {
            end--;
        }
        return str.substring(start, end + 1);
    }
