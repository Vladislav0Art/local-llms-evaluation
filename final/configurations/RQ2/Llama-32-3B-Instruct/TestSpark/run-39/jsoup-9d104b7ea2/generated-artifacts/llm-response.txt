    public void newNodeTextLengthTest() {

        assertEquals(12, node.text().length());

        assertNull(result2.text());

    }

    public void createFromEncodedTextNodeTest() throws IOException {
        String encodedText = "Hello World";

        assertNotNull(node.text());
        assertEquals(encodedText, node.text());
    }

    public void normaliseWhitespaceTest() {
        String text = "\n  Hello\nWorld";
        assertEquals("Hello\nWorld", StringUtil.normaliseWhitespace(text));

    }
}

import static org.junit.Assert.*;

public class TestGeneratedText {

    @Test
    public void testHelloWorld() {
        String expected = "Hello\nWorld";
        assertEquals(expected, StringUtil.normaliseWhitespace("Hello World"));
    }

    @Test
    public void testStripLeadingWhitespace() {
        String expected = "Hello World";
        assertEquals(expected, TextNode.stripLeadingWhitespace("   Hello World"));
    }

    @Test
    public void testLastCharIsWhitespace() {
        StringBuilder sb = new StringBuilder();
        assertTrue(TextNode.lastCharIsWhitespace(sb.append(' ').append('\n')));
        assertFalse(TextNode.lastCharIsWhitespace(sb));
    }
}

public class StringUtil {

    public static String normaliseWhitespace(String text) {
        return text.replaceAll("\\s+", " ");
    }

    // other methods...
}

public class TextNode {

    public static String stripLeadingWhitespace(String text) {
        while (text.startsWith(" ")) {
            text = text.substring(1);
        }
        return text;
    }

    public static boolean lastCharIsWhitespace(char c) {
        return Character.isWhitespace(c);
    }

}
