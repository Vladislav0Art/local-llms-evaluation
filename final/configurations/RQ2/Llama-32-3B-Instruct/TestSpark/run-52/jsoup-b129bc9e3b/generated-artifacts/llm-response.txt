        // Given and When

        // Then

    public void execute_ReturnsResponse() throws IOException {

        Connection connection = HttpConnection.execute(response);
        // Then
        assertNotNull(connection);
    }

    @org.junit.Test
    public void request_ReturnsRequest() {
        // Given and When
        Request request = new Request();
        Connection connection = HttpConnection.request(request);
        // Then
        assertNotNull(connection);
    }

    public void connect_URL_MockingSSLSocketFactory() throws MalformedURLException, URISyntaxException {

        // Then

    }
