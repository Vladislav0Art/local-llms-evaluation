public class WordWrap {

    public static String leftTrim(String input) {
        if (input == null || input.isEmpty()) {
            return "";
        }
        int start = 0;
        while (start < input.length() && Character.isWhitespace(input.charAt(start))) {
            start++;
        }
        return input.substring(start);
    }

    public static String rightTrim(String input) {
        if (input == null || input.isEmpty()) {
            return "";
        }
        int end = input.length() - 1;
        while (end >= 0 && Character.isWhitespace(input.charAt(end))) {
            end--;
        }
        return input.substring(0, end + 1);
    }

    public static String isWhitespace(String input) {
        if (input == null || input.isEmpty()) {
            return "true";
        }
        for (int i = 0; i < input.length(); i++) {
            if (!Character.isWhitespace(input.charAt(i))) {
                return "false";
            }
        }
        return "true";
    }

    public static String wordWrap(String input, int width) {
        StringBuilder output = new StringBuilder();
        String[] words = input.split(" ");
        for (String word : words) {
            if (output.length() + word.length() + 1 > width) {
                output.append("\n");
            }
            output.append(word).append(" ");
        }
        return output.toString().trim();
    }

    public static void main(String[] args) {
        System.out.println(leftTrim(""));
        System.out.println(rightTrim(""));
        System.out.println(isWhitespace(""));
        System.out.println(wordWrap("", 10));
        System.out.println(wordWrap("Hello World", 10));
        System.out.println(wordWrap("This is a test string", 20));
    }
}
