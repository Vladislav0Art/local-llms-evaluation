import org.jsoup.safety.Safelist;
import org.jsoup.nodes.Attribute;
import org.jsoup.nodes.Attributes;
import org.jsoup.nodes.Element;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import static org.hamcrest.MatcherAssert.assertThat;
import static org.hamcrest.Matchers.is;
import static org.jsoup.helper.Validate.validate;
import static org.jsoup.internal.Normalizer.lowerCase;

@RunWith(MockitoJUnitRunner.class)
public class SafelistTest {

    @Mock
    private Element element;

    @Test
    public void none_SafeTag() {
        Safelist safelist = Safelist.none();
        assertThat(safelist.isSafeTag("img"), is(true));
    }

    @Test
    public void simpleText_SimpleText() {
        Safelist safelist = Safelist.simpleText();
        validate(safelist.isSafeTag("p"));
        validate(!safelist.isSafeTag("img"));
    }

    @Test
    public void basic_BasicTags() {
        Safelist safelist = Safelist.basic();
        assertThat(safelist.isSafeTag("h1"), is(true));
        assertThat(safelist.isSafeTag("h2"), is(false));
    }

    @Test
    public void basicWithImages_WithImages() {
        Safelist safelist = Safelist.basicWithImages();
        validate(safelist.isSafeTag("img"));
    }

    @Test
    public void relaxed_RelaxedTags() {
        Safelist safelist = Safelist.relaxed();
        assertThat(safelist.isSafeTag("h1"), is(true));
        assertThat(safelist.isSafeTag("h2"), is(false));
        validate(!safelist.isSafeTag("img"));
    }

    @Test
    public void addTags_AddedTags() {
        Safelist safelist = Safelist.addTags("h3");
        validate(safelist.isSafeTag("h3"));
    }

    @Test
    public void removeTags_RemovedTags() {
        Safelist safelist = Safelist.removeTags("img", "div");
        validate(!safelist.isSafeTag("img"));
        validate(safelist.isSafeTag("div"));
    }

    @Test
    public void addAttributes_AddedAttributes() {
        Safelist safelist = Safelist.addAttributes("img", "alt", "test-alt");
        validate(safelist.getEnforcedAttributes("img").containsKey("alt"));
        validate(!safelist.getEnforcedAttributes("img").containsKey("src"));
    }

    @Test
    public void removeAttributes_RemovedAttributes() {
        Safelist safelist = Safelist.removeAttributes("img", "alt");
        validate(safelist.getEnforcedAttributes("img").isEmpty());
        validate(!safelist.getEnforcedAttributes("img").containsKey("src"));
    }

    @Test
    public void addEnforcedAttribute_AddedEnforcedAttribute() {
        Safelist safelist = Safelist.addEnforcedAttribute("img", "alt", "test-alt");
        validate(safelist.getEnforcedAttributes("img").containsKey("alt"));
        validate(!safelist.getEnforcedAttributes("img").containsKey("src"));
    }

    @Test
    public void removeEnforcedAttribute_RemovedEnforcedAttribute() {
        Safelist safelist = Safelist.removeEnforcedAttribute("img", "alt");
        validate(safelist.getEnforcedAttributes("img").isEmpty());
        validate(!safelist.getEnforcedAttributes("img").containsKey("src"));
    }

    @Test
    public void preserveRelativeLinks_PreserveRelativeLinks() {
        Safelist safelist = Safelist.preserveRelativeLinks(true);
        validate(safelist.isSafeTag("a"));
        validate(!safelist.isSafeTag("div"));
    }

    @Test
    public void addProtocols_AddedProtocols() {
        Safelist safelist = Safelist.addProtocols("img", "src", "https://test.com");
        validate(safelist.getEnforcedAttributes("img").containsKey("src"));
        validate(!safelist.getEnforcedAttributes("img").containsKey("alt"));
    }

    @Test
    public void removeProtocols_RemovedProtocols() {
        Safelist safelist = Safelist.removeProtocols("img", "src");
        validate(safelist.getEnforcedAttributes("img").isEmpty());
        validate(!safelist.getEnforcedAttributes("img").containsKey("alt"));
    }

    @Test
    public void isSafeTag_IsSafeTag() {
        Safelist safelist = Safelist.relaxed();
        assertThat(safelist.isSafeTag("h1"), is(true));
        assertThat(safelist.isSafeTag("div"), is(false));
    }

    @Test
    public void isSafeAttribute_IsSafeAttribute() {
        Safelist safelist = Safelist.basicWithImages();
        validate(Safelist.none().isSafeAttribute("img", element, new Attribute()));
        validate(!Safelist.relaxed().isSafeAttribute("h2", element, new Attribute()));
    }

    @Test
    public void getEnforcedAttributes_GetEnforcedAttributes() {
        Safelist safelist = Safelist.addAttributes("img", "alt");
        Attributes enforcedAttributes = safelist.getEnforcedAttributes("img");
        validate(enforcedAttributes.containsKey("alt"));
        validate(!enforcedAttributes.containsKey("src"));
    }
}
