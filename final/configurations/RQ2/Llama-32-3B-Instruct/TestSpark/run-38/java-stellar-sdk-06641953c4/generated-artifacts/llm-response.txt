```java
import org.spongycastle.asn1.x962.Ed25519PublicKey;
import org.spongycastle.crypto.dsa.DSAKeyPairGenerator;
import org.spongycastle.crypto.dsa Ed25519;

public class Ed25519KeyPair {
    private Ed25519PublicKey publicKey;
    private byte[] privateKeyBytes;

    public Ed25519KeyPair() throws Exception {
        DSAKeyPairGenerator generator = new DSAKeyPairGenerator();
        generator.generateKeyPair(1024);
        Ed25519 ed25519 = new Ed25519();
        publicKey = ed25519.getPublicKey((byte[]) generator.getKey().getEncoded());
    }

    public Ed25519PublicKey getPublicKey() {
        return publicKey;
    }

    public byte[] getPrivateKeyBytes() {
        return privateKeyBytes;
    }
}
```

### Ed25519 Signing
```java
import org.spongycastle.asn1.x962.Ed25519SignatureScheme;
import org.spongycastle.crypto.dsa.DSA;
import org.spongycastle.crypto.dsa.EdDSASignature;

public class Ed25519Signer {
    private DSA dsaclient;

    public Ed25519Signer(Ed25519KeyPair keyPair) throws Exception {
        dsaclient = new DSA(keyPair.getPrivateKeyBytes());
        Ed25519 ed25519 = new Ed25519();
        ed25519.init(dsaclient);
    }

    public void sign(byte[] message, byte[] signature) throws Exception {
        ed25519.sign(message, signature);
    }
}
```

### Example Usage
```java
public class Main {
    public static void main(String[] args) throws Exception {
        Ed25519KeyPair keyPair = new Ed25519KeyPair();
        Ed25519Signer signer = new Ed25519Signer(keyPair);

        byte[] message = "Hello, World!".getBytes();
        byte[] signature = new byte[65]; // 65 bytes for ed25519 signature
        signer.sign(message, signature);
        System.out.println("Signature: " + bytesToHex(signature));

        EdDSAPublicKey publicKey = keyPair.getPublicKey();
        System.out.println("Public Key: " + bytesToHex(publicKey.toAsn1Object()));
    }

    private static String bytesToHex(byte[] bytes) {
        StringBuilder sb = new StringBuilder();
        for (byte b : bytes) {
            sb.append(String.format("%02x", b));
        }
        return sb.toString();
    }
}
```

### Java Implementation for Ed25519 Key Pair Generation and Signing

```java
import java.security.SecureRandom;
import java.util.Base64;

import javax.crypto.Mac;
import javax.crypto.spec.PBEKeySpec;
import java.nio.charset.StandardCharsets;
import java.security.KeyFactory;
import java.security.KeyPair;
import java.security.PrivateKey;
import java.security.PublicKey;
import java.security.spec.X509EncodedKeySpec;
import java.util.HashMap;
import java.util.Map;

public class Ed25519 {

    // Function to generate a random 32-byte salt
    public static byte[] getSalt() {
        SecureRandom secureRandom = new SecureRandom();
        return secureRandom.generateSeed(32);
    }

    // Function to derive the key from the salt and password using PBKDF2HMAC with SHA-256
    public static byte[] deriveKey(byte[] salt, String password) throws Exception {
        Map<String, Object> params = new HashMap<>();
        params.put("salt", salt);
        params.put("password", password.toCharArray());
        params.put("iter", 100000);

        PBEKeySpec spec = new PBEKeySpec(password.toCharArray(), (byte[]) params.get("salt"), (int) params.get("iter"), 256);
        return KeyFactory.getInstance("PBKDF2WithHmacSHA256").generateSecret(spec).getEncoded();
    }

    // Function to generate a random key pair
    public static byte[] generateKeyPair() {
        byte[] salt = getSalt();
        byte[] key = deriveKey(salt, "mysecretpassword");
        return new byte[]{salt[0], salt[1], salt[2], salt[3], salt[4], salt[5],
                (byte) ((key[0] >> 24) & 0xFF), (byte) ((key[0] & 0xFF) >> 8),
                (byte) ((key[0] & 0xFF) << 8), (byte) (key[0] >> 16), (byte) ((key[1] >> 24) & 0xFF),
                (byte) ((key[1] & 0xFF) >> 8), (byte) ((key[1] & 0xFF) << 8), (byte) (key[1] >> 16)};
    }

    // Function to generate a public key from the private key
    public static byte[] getPublicKey(byte[] privateKey) {
        int[] k = new int[]{(int) (privateKey[0] >> 24),
                (int) ((privateKey[0]) & 0xFF),
                (int) ((privateKey[1]) >> 8),
                (int) ((privateKey[1]) & 0xFF)};
        return new byte[]{(byte) ((k[0] << 24) | (k[1] << 16)),
                (byte) ((k[2] << 12) | (k[3] << 8)),
                (byte) ((k[4] << 6) | (k[5])), (byte)(k[6])};
    }

    // Function to generate a signature from the private key and message
    public static byte[] sign(byte[] privateKey, byte[] message) {
        int[] k = new int[]{(int) (privateKey[0] >> 24),
                (int) ((privateKey[0]) & 0xFF),
                (int) ((privateKey[1]) >> 8),
                (int) ((privateKey[1]) & 0xFF)};
        byte[] signature = new byte[]{(byte) k[4],
                (byte) ((k[5] << 2) | ((message[0] >> 6) & 3)),
                (byte) (((message[0] >> 2) & 3) | ((message[1] >> 4) & 3)),
                (byte)((message[1] >> 2) & 3),
                (byte)((message[1]) & 3),
                (byte)(message[2]),
                (byte)(message[3])};

        for (int i = 7; i < message.length; i += 8) {
            int[] m = new int[]{(int) ((message[i] >> 6) & 63),
                    (int) (((message[i]) >> 5) & 31),
                    (int) (((message[i]) >> 4) & 15),
                    (int) (((message[i]) >> 3) & 7),
                    (int) (((message[i]) >> 2) & 3),
                    (int)((message[i]) & 3),
                    (int)(message[i + 1]),
                    (int)(message[i + 2])};

            for (int j = 0; j < m.length; j++) {
                signature[j] ^= m[j];
            }
        }

        return signature;
    }
}
```
