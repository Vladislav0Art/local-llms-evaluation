import org.junit.Test;

public class AppVersionTest {

        assertNotNull(appVersion);
        assertEquals("1.0.2", appVersion.getVersion());
    }

    @Test
    public void parseNameFromInvalidAmiName_ReturnsNull() {

        AppVersion appVersion = AppVersion.parseName(amiName);
        assertNull(appVersion);
    }

    @Test
    public void parseNameWithoutVersion_ReturnsNull() {
        String amiName = "ami-name";
        AppVersion appVersion = AppVersion.parseName(amiName);
        assertNull(appVersion);
    }

    @Test
    public void getPackageNameReturnsExpectedValue() {
        AppVersion appVersion = new AppVersion("com.netflix.frigga", "ami");
        assertEquals("com.netflix.frigga", appVersion.getPackageName());
    }

    @Test
    public void getVersionReturnsExpectedValue() {
        AppVersion appVersion = new AppVersion("1.0.2-20220101", "build-job-name", "commit-hash");
        assertEquals("1.0.2-20220101", appVersion.getVersion());
    }

    @Test
    public void getBuildJobNameReturnsExpectedValue() {
        AppVersion appVersion = new AppVersion("1.0.2-20220101", "build-job-name", "commit-hash");
        assertEquals("build-job-name", appVersion.getBuildJobName());
    }

    @Test
    public void getBuildNumberReturnsExpectedValue() {
        AppVersion appVersion = new AppVersion("1.0.2-20220101", "build-job-name", "commit-hash");
        assertEquals("20220101", appVersion.getBuildNumber());
    }

    @Test
    public void getCommitReturnsExpectedValue() {
        AppVersion appVersion = new AppVersion("1.0.2-20220101", "build-job-name", "commit-hash");
        assertEquals("commit-hash", appVersion.getCommit());
    }

    @Test
    public void getAppVersionPatternReturnsExpectedRegex() {
        AppVersion appVersion = new AppVersion();
        Pattern pattern = appVersion.getAppVersionPattern();
        assertTrue(pattern.matcher("1.0.2-20220101").find());
        assertFalse(pattern.matcher("invalid-ami-name").find());
    }

    @Test
    public void getAppVersionPatternReturnsNullIfRegexIsInvalid() {
        AppVersion appVersion = new AppVersion();
        Pattern pattern = appVersion.getAppVersionPattern();
        assertNull(pattern);
    }
}
