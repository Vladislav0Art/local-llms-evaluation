import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.LeafNode;
import org.jsoup.nodes.TextNode;
import org.jsoup.parser.Parser;
import org.junit.BeforeClass;
import org.junit.Test;

import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertThrows;
import static org.mockito.Mockito.when;

public class TextNodeTest {

    private static Document document;

    @BeforeClass
    public static void beforeClass() {
        Parser parser = new Parser();
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        when(Document.OutputSettings.class).thenReturn(new Document.OutputSettings() {{
            appendable = out;
        }});
        List<LeafNode> nodes = new ArrayList<>();
        nodes.add(new Element("div"));
        nodes.add(new TextNode("Hello, world!"));
        parser.parse(new String[]{""}, nodes);
        document = new Document(out.toString(), Parser.xmlParser());
    }

    @Test
    public void nodeName[]Test() {
        assertEquals("text", document.head().child(0).nodeName());
    }

    @Test
    public void text[]Test() {
        TextNode textNode = (TextNode) document.child(1);
        assertEquals("Hello, world!", textNode.text());
    }

    @Test
    public void text(String text) throws IOException {
        String encodedText = "Hello%20world!";
        TextNode textNode = TextNode.createFromEncoded(encodedText);
        assertEquals(text, textNode.text());
    }

    @Test
    public void getWholeText[]Test() {
        TextNode textNode = document.child(1);
        assertEquals("Hello, world!", textNode.getWholeText());
    }

    @Test
    public void isBlank[]Test() {
        TextNode textNode = document.child(1);
        assertFalse(textNode.isBlank());
    }

    @Test
    public void splitText[]Test() throws IOException {
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        when(Document.OutputSettings.class).thenReturn(new Document.OutputSettings() {{
            appendable = out;
        }});
        List<LeafNode> nodes = new ArrayList<>();
        nodes.add(new Element("div"));
        nodes.add(new TextNode("Hello, world!"));
        nodes.add(new Element("span"));
        Parser parser = new Parser();
        parser.parse(new String[]{""}, nodes);
        Document document2 = new Document(out.toString(), Parser.xmlParser());
        TextNode textNode = (TextNode) document2.child(1);
        assertEquals("", textNode.splitText(0).text());
    }

    @Test
    public void outerHtmlHead[]Test() throws IOException {
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        when(Document.OutputSettings.class).thenReturn(new Document.OutputSettings() {{
            appendable = out;
        }});
        List<LeafNode> nodes = new ArrayList<>();
        nodes.add(new Element("div"));
        nodes.add(new TextNode("Hello, world!"));
        Parser parser = new Parser();
        parser.parse(new String[]{""}, nodes);
        document = new Document(out.toString(), Parser.xmlParser());
        TextNode textNode = (TextNode) document.child(1);
        assertEquals("<p>Hello, world!</p>", textNode.outerHtmlHead(null, 0, null));
    }

    @Test
    public void outerHtmlTail[]Test() {
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        when(Document.OutputSettings.class).thenReturn(new Document.OutputSettings() {{
            appendable = out;
        }});
        List<LeafNode> nodes = new ArrayList<>();
        nodes.add(new Element("div"));
        nodes.add(new TextNode("Hello, world!"));
        Parser parser = new Parser();
        parser.parse(new String[]{""}, nodes);
        document = new Document(out.toString(), Parser.xmlParser());
        TextNode textNode = (TextNode) document.child(1);
        assertEquals("", textNode.outerHtmlTail(null, 0, null));
    }

    @Test
    public void clone[]Test() {
        TextNode textNode = document.child(1);
        TextNode clonedTextNode = textNode.clone();
        assertEquals("Hello, world!", clonedTextNode.text());
    }

    @Test
    public void toString[]Test() {
        assertEquals("<text>Hello, world!</text>", document.head().child(0).toString());
    }
}
