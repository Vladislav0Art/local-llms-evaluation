package org.jsoup.nodes;

import org.junit.Test;
import static org.hamcrest.MatcherAssert.assertThat;
import static org.hamcrest.Matchers.is;
import static org.junit.Assert.fail;

import java.io.IOException;
import java.util.Arrays;

import org.jsoup.helper.Validate;
import org.jsoup.internal.StringUtil;
import org.jsoup.nodes.TextNode;
import org.jsoup.nodes.LeafNode;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Appendable;
import org.jsoup.output.Document.OutputSettings;

public class TextNodeTest {

    @Test
    public void newTextNode_isEmptyText() {
        String text = "hello";
        TextNode node = new TextNode(text);
        assertThat(node.text(), is(""));
    }

    @Test
    public void nodeName_isText() {
        TextNode node = new TextNode("test");
        assertThat(node.nodeName(), is("text"));
    }

    @Test
    public void text_returnsOriginalText() {
        String text = "hello";
        TextNode node = new TextNode(text);
        assertThat(node.text(), is(text));
    }

    @Test
    public void text_setsNewText() {
        TextNode node = new TextNode("test");
        node.text("newText");
        assertThat(node.text(), is("newText"));
    }

    @Test
    public void getWholeText_returnsOriginalText() {
        String text = "hello";
        TextNode node = new TextNode(text);
        assertThat(node.getWholeText(), is(text));
    }

    @Test
    public void isBlank_returnsFalse_ifTextIsNotBlank() {
        String text = "hello";
        TextNode node = new TextNode(text);
        assertThat(node.isBlank(), is(false));
    }

    @Test
    public void isBlank_returnsTrue_ifTextIsEmptyOrWhitespaceOnly() {
        String text = "";
        TextNode node = new TextNode(text);
        assertThat(node.isBlank(), is(true));

        text = "   ";
        node = new TextNode(text);
        assertThat(node.isBlank(), is(true));
    }

    @Test
    public void splitText_splitsAtGivenOffset() {
        String text = "hello world";
        TextNode node = new TextNode(text);
        node.splitText(6);

        assertThat(node.text(), is("world"));
    }

    @Test
    public void outerHtmlHead_appendsToHead() throws IOException {
        Document document = Document.parse("<html><head></head></html>");
        Appendable accum = new StringBuilder();
        int depth = 0;
        TextNode node = new TextNode("test");
        node.outerHtmlHead(accum, depth, document.outputSettings());

        assertThat(accum.toString(), is("<p>test</p>"));
    }

    @Test
    public void outerHtmlTail_appendsToTail() throws IOException {
        Document document = Document.parse("<html><head></head><tail></tail></html>");
        Appendable accum = new StringBuilder();
        int depth = 0;
        TextNode node = new TextNode("test");
        node.outerHtmlTail(accum, depth, document.outputSettings());

        assertThat(accum.toString(), is("</p>"));
    }

    @Test
    public void toString_returnsOriginalText() {
        String text = "hello";
        TextNode node = new TextNode(text);
        assertThat(node.toString(), is("text=\""+text+"\""));
    }

    @Test
    public void clone_returnsNewTextNode() {
        String text = "test";
        TextNode node = new TextNode(text);
        TextNode clonedNode = node.clone();
        assertThat(clonedNode.text(), is(text));
    }

    @Test
    public void createFromEncoded_decodesString() throws IOException {
        String encodedText = "hello";
        TextNode node = TextNode.createFromEncoded(encodedText);
        assertThat(node.text(), is("hello"));
    }

    @Test
    public void normaliseWhitespace_replacesAllSpacesWithHypens() {
        String text = "   ";
        String expected = "_____";
        String result = StringUtil.normaliseWhitespace(text);
        assertThat(result, is(expected));
    }

    @Test
    public void stripLeadingWhitespace_returnsEmptyStringIfInputIsEmptyOrStartsWithWhitespace() {
        String text = "";
        String expected = "";
        String result = StringUtil.stripLeadingWhitespace(text);
        assertThat(result, is(expected));

        text = "   ";
        result = StringUtil.stripLeadingWhitespace(text);
        assertThat(result, is(""));
    }

    @Test
    public void lastCharIsWhitespace_returnsTrueIfLastCharacterIsWhitespace() {
        StringBuilder sb = new StringBuilder("test ");
        boolean result = TextNode.lastCharIsWhitespace(sb);
        assertThat(result, is(true));

        sb.deleteCharAt(sb.length() - 1);
        result = TextNode.lastCharIsWhitespace(sb);
        assertThat(result, is(false));
    }

    @Test
    public void createFromEncoded_throwsExceptionIfInputIsEmptyOrNull() throws IOException {
        assertThrows(IOException.class, () -> TextNode.createFromEncoded(""));
        assertThrows(IOException.class, () -> TextNode.createFromEncoded(null));
    }
}
