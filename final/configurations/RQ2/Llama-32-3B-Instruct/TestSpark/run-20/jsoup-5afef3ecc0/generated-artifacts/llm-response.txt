public class KeyVal {
    public static KeyVal newKeyVal(String key, String value) {
        return new KeyVal(key, value);
    }

    private final String key;
    private final String value;

    public KeyVal(String key, String value) {
        this.key = key;
        this.value = value;
    }
}

public class UrlBuilder {
    private URI baseUri;

    public UrlBuilder(URI baseUri) {
        this.baseUri = baseUri;
    }

    public UrlBuilder appendKeyVal(KeyVal kv) throws UnsupportedEncodingException, URISyntaxException {
        String query = "";
        if (baseUri.getQuery() != null) {
            query = baseUri.getQuery();
        }
        query += "&" + kv.key + "=" + URLEncoder.encode(kv.value, "UTF-8");
        baseUri = new URI(baseUri.toString().replaceFirst("^.*\\?", "") + "?" + query);
        return this;
    }

    public String getQueryString() throws UnsupportedEncodingException {
        return baseUri.getQuery();
    }

    public UrlBuilder appendQuery(String query) throws UnsupportedEncodingException, URISyntaxException {
        if (baseUri.getQuery() != null) {
            throw new RuntimeException("Already have a query.");
        }
        baseUri = new URI(baseUri.toString().replaceFirst("^.*\\?", "") + "?" + query);
        return this;
    }

    public UrlBuilder appendFragment(String fragment) throws URISyntaxException {
        if (baseUri.getFragment() != null) {
            throw new RuntimeException("Already have a fragment.");
        }
        baseUri = new URI(baseUri.toString().replaceFirst("^.*#?", "") + "#" + fragment);
        return this;
    }

    public URI buildUrl() throws URISyntaxException {
        return baseUri;
    }
}

    public KeyVal(String key, String value) {
        this.key = key;
        this.value = value;
    }

public class UrlBuilder {
    private URI uri;

    public UrlBuilder(URI uri) {
        this.uri = uri;
    }

    public UrlBuilder appendKeyVal(KeyVal kv) throws URISyntaxException {
        if (uri.getQuery() == null) {
            uri = new URI(uri.toString().replaceFirst("^.*\\?", "") + "?" + kv.key + "=" + java.net.URLEncoder.encode(kv.value, "UTF-8"));
        } else {
            String query = uri.getQuery();
            query += "&" + kv.key + "=" + java.net.URLEncoder.encode(kv.value, "UTF-8");
            uri = new URI(uri.toString().replaceFirst("^.*\\?", "") + "?" + query);
        }
        return this;
    }

    public void appendQuery(String query) throws URISyntaxException {
        if (uri.getQuery() != null) {
            throw new RuntimeException("Already have a query.");
        }
        uri = new URI(uri.toString().replaceFirst("^.*#?", "") + "?" + query);
    }

    public void appendFragment(String fragment) throws URISyntaxException {
        if (uri.getFragment() != null) {
            throw new RuntimeException("Already have a fragment.");
        }
        uri = new URI(uri.toString().replaceFirst("^.*#?", "") + "#" + fragment);
    }

    public String getQueryString() throws URISyntaxException {
        return uri.getQuery();
    }

    public String toString() {
        return uri.toString();
    }

    public URI buildUrl() throws URISyntaxException {
        return uri;
