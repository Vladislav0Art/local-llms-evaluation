import com.crowdin.client.core.http.impl.json.JacksonJsonTransformer;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import static org.hamcrest.MatcherAssert.assertThat;
import static org.hamcrest.Matchers.*;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class JacksonJsonTransformerTest {

    @Mock
    private ObjectMapper objectMapper;

    @Test
    public void parseEmptyStringAsNullTest() {
        String json = "";
        Class<?> clazz = null;
        Object result = new JacksonJsonTransformer().parse(json, clazz);
        assertThat(result, is(null));
    }

    @Test
    public void parseValidJsonToProjectTest() throws Exception {
        String json = "{\"name\":\"Test\",\"id\":1,\"language\":\"en\"}";
        Class<Project> clazz = Project.class;
        when(objectMapper.findAndBindValue(json, clazz)).thenReturn(new Project("Test", 1, "en"));
        Object result = new JacksonJsonTransformer().parse(json, clazz);
        assertThat(result, instanceOf(Project.class));
    }

    @Test
    public void convertNullToEmptyStringTest() {
        Class<?> clazz = null;
        String result = new JacksonJsonTransformer().convert(null);
        assertThat(result, is(""));
    }

    @Test
    public void convertValidObjectToStringTest() throws Exception {
        Project project = new Project("Test", 1, "en");
        Class<?> clazz = Project.class;
        when(objectMapper.writeValueAsString(project)).thenReturn("{\"name\":\"Test\",\"id\":1,\"language\":\"en\"}");
        String result = new JacksonJsonTransformer().convert(project);
        assertThat(result, is("{\"name\":\"Test\",\"id\":1,\"language\":\"en\"}"));
    }

    @Test
    public void convertValidObjectToStringWithIncludeEmptyValuesTest() throws Exception {
        Project project = new Project("Test", 1, "en");
        Class<?> clazz = Project.class;
        when(objectMapper.writeValueAsString(project)).thenReturn("{\"name\":\"Test\",\"id\":1,\"language\":\"en\"}");
        String result = new JacksonJsonTransformer().convert(project, new IncludeNullValues());
        assertThat(result, is("{\"name\":\"Test\",\"id\":1,\"language\":\"en\"}"));
    }

    @Test
    public void convertValidObjectToStringWithPropertyExcludeTest() throws Exception {
        Project project = new Project("Test", 1, "en");
        Class<?> clazz = Project.class;
        when(objectMapper.writeValueAsString(project)).thenReturn("{\"name\":\"Test\",\"id\":1,\"language\":\"en\"}");
        String result = new JacksonJsonTransformer().convert(project, new ExcludeProperties(new String[]{"id"}));
        assertThat(result, is("{\"name\":\"Test\",\"language\":\"en\"}"));
    }

    @Test
    public void convertValidObjectToStringWithDeserializationFeatureTest() {
        Project project = new Project("Test", 1, "en");
        Class<?> clazz = Project.class;
        when(objectMapper.configure(MapperFeature DesmonderializationFeature, Mockito.anyBoolean())).thenReturn(true);
        String result = new JacksonJsonTransformer().convert(project);
        assertThat(result, is("{\"name\":\"Test\",\"id\":1,\"language\":\"en\"}"));
    }

    public void convertValidObjectToStringWithDeserializationFeaturesAndIncludeEmptyValuesTest() {
        Project project = new Project("Test", 1, "en");

        String result = new JacksonJsonTransformer().convert(project, new IncludeNullValues());
