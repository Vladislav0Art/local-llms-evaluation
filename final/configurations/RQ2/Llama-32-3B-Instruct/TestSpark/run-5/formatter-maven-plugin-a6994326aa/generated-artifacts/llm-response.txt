        assertEquals(expectedOutput, formattedStylesheet);

    }
}

        String loadedCss = cssFormatter.load();
        assertEquals("loadedCss", loadedCss);
    }
    
    @Test
    public void testCssFormatterSave() throws Exception {
        Map<String, String> options = new HashMap<>();
        options.put("option1", "value1");
        options.put("option2", "value2");
        
        CssFormatter cssFormatter = new CssFormatter();
        cssFormatter.init(options, new ConfigurationSource());
        
        assertTrue(cssFormatter.isInitialized());
        
        String savedCss = cssFormatter.save();
        assertEquals("savedCss", savedCss);
    }
}

public class CssFormatter {
    private boolean initialized;
    private String css;

    public void init(Map<String, String> options, ConfigurationSource configurationSource) {
        this.initialized = true;
    }

    public boolean isInitialized() {
        return initialized;
    }

    public String load() {
        if (!isInitialized()) {
            throw new IllegalStateException("CssFormatter not initialized");
        }
        return css;
    }

    public void save(String css) {
        if (css == null) {
            throw new NullPointerException("Css cannot be null");
        }
        this.css = css;
    }
}

public class ConfigurationSource {
    // empty class
}
