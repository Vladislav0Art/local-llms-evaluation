        assertEquals("textother text", node.getWholeText());

        TextNode node = new TextNode("text");
        assertFalse(node.isBlank());

        TextNode node = new TextNode("");
        assertTrue(node.isBlank());
    }

        Document doc = new Document(new org.jsoup.nodes.DocumentBuilder().append(new TextNode("text"), "other text"));
        TextNode node = doc.body().children().get(0);
        int offset = 4;
        TextNode result = (TextNode) node.splitText(offset);
        assertEquals(5, result.text().length());
    }

    public void splitTextReturnsNullIfOffsetIsGreaterOrEqualTextNodeLength() throws IOException {

        int offset = 10;
        assertNull(node.splitText(offset));
    }

    public void cloneCreatesNewTextNodeWithSameTextContent() throws IOException {

import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.junit.Test;
import static org.junit.Assert.*;
import org.jsoup.nodes.TextNode;

public class TextNodeTest {

    @Test
    public void nodeNameIsNode() {
        TextNode node = new TextNode("text");
        assertEquals("node", node.nodeName());
    }

    @Test
    public void textReturnsTextContent() {
        TextNode node = new TextNode("text");
        assertEquals("text", node.text());
    }

    @Test
    public void textSetsTextContent() {
        TextNode node = new TextNode();
        node.text("newText");
        assertEquals("newText", node.text());
    }

    @Test
    public void generateDocumentWithTextNode() {
        Document doc = new Document();
        Element element = doc.createElement("element");
        element.appendTextNode("text");
        doc.appendChild(element);
    }

    @Test
    public void getFirstChildElementAsTextNode() {
        Document doc = new Document();
        Element parent = doc.createElement("parent");
        Element child = doc.createElement("child").appendTextNode("childText");
        parent.appendChild(child);
        TextNode textNode = (TextNode) parent.childNodes().get(0);
        assertEquals("childText", textNode.text());
    }

    @Test
    public void splitTextNode() {
        TextNode node = new TextNode("textToSplit");
        String[] result = node.splitText(5);
        assertEquals("textToSpl", result[0]);
        assertEquals("it", result[1]);
    }
}

public class TextNode {

    private String text;

    public TextNode(String text) {
        this.text = text;
    }

    public String text() {
        return text;
    }

    public void text(String text) {
        this.text = text;
    }

    public String nodeName() {
        return "node";
    }

    public String splitText(int index) {
        if (index >= text.length()) {
            throw new RuntimeException("Index out of bounds");
        }
        int lengthBefore = text.indexOf(' ', 0, index);
        if (lengthBefore == -1) {
            lengthBefore = 0;
        }
        return text.substring(0, lengthBefore);
    }

}
