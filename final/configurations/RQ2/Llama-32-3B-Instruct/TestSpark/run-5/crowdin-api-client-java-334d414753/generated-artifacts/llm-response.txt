public class JsonTransformer {
    private static final String PROJECT_JSON = "{\"name\": \"Project Name\", \"description\": \"This is a test project\"}";
    private static final String EXPORT_OPTIONS_JSON = "{\"type\": \"export\", \"filename\": \"output.csv\"}";
    private static final String FILE_INFO_JSON = "{\"file_name\": \"test_file.txt\"}";

    public static String convert(Project project) {
        return project.getName();
    }

    public static String convert(ExportOptions exportOptions) {
        return exportOptions.getFilename();
    }

    public static String convert(FileInfo fileInfo) {
        return fileInfo.getFile_name();
    }
}

public class Project {
    private String name;
    private String description;

    public Project(String name, String description) {
        this.name = name;
        this.description = description;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }
}

public class ExportOptions {
    private String type;
    private String filename;

    public ExportOptions(String type, String filename) {
        this.type = type;
        this.filename = filename;
    }

    public String getFilename() {
        return filename;
    }

    public void setFilename(String filename) {
        this.filename = filename;
    }
}

public class FileInfo {
    private String file_name;

    public FileInfo(String file_name) {
        this.file_name = file_name;
    }

    public String getFile_name() {
        return file_name;
    }

    public void setFile_name(String file_name) {
        this.file_name = file_name;
    }
}
