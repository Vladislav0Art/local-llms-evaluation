        // ... other self-closing tags ...
    }

        return this.tagName();
    }

    public boolean isSelfClosing() {
        return selfClosingTags.containsKey(tagName());
    }

    @Override

        return tagName().isEmpty();
    }
}

public class TagTest {

    @Test
    public void getName_EmptyTag_ReturnsEmpty() {
