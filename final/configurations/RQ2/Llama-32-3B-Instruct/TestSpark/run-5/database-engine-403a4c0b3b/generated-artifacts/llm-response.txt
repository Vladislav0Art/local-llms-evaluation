public class DatabaseEngine {
    private Database db;

    public DatabaseEngine(Database db) {
        this.db = db;
    }

    public boolean createTable(String tableName, String type, List<String> columns, List<String> constraints, List<String> primaryKeys) {
        db.createTable(tableName, type, new Hashtable<>(), new Hashtable<>(), new Hashtable<>(primaryKeys));
        return true;
    }

    public boolean insertIntoTable(String tableName, List<List<Object>> data) {
        db.insertIntoTable(tableName, new Hashtable<>());
        for (List<Object> row : data) {
            db.insertIntoTable(tableName, new Hashtable<>((Hashtable) row));
        }
        return true;
    }

    public boolean updateTable(String tableName, String type, List<List<Object>> data) {
        db.updateTable(tableName, type, new Hashtable<>());
        for (List<Object> row : data) {
            db.updateTable(tableName, type, new Hashtable<>((Hashtable) row));
        }
        return true;
    }

    public boolean deleteFromTable(String tableName, List<List<Object>> data) {
        db.deleteFromTable(tableName, new Hashtable<>());
        for (List<Object> row : data) {
            db.deleteFromTable(tableName, new Hashtable<>((Hashtable) row));
        }
        return true;
    }

    public Iterator<SQLTerm> selectFromTable(List<SQLTerm> sqlTerms, String[] strarrOperators) {
        List<Iterator<SQLTerm>> iterators = new ArrayList<>();
        for (SQLTerm term : sqlTerms) {
            iterators.add(db.selectFromTable(Arrays.asList(term), strarrOperators));
        }
        return null;
    }
}
