public class Connection {
    public static String encode(String value) throws UnsupportedEncodingException {
        return java.net.URLEncoder.encode(value, "UTF-8");
    }
}

public class UrlBuilder {
    private String baseUrl;

    public UrlBuilder(String baseUrl) {
        this.baseUrl = baseUrl;
    }

    public String getFullUrl() {
        return baseUrl;
    }

    public UrlBuilder appendKeyAndValue(String key, String value) throws UnsupportedEncodingException {
        if (baseUrl.contains("?")) {
            return appendQueryParameter(baseUrl, key, Connection.encode(key), value);
        } else {
            return appendQueryParameter(baseUrl, key, Connection.encode(key), value).append("&");
        }
    }

    private UrlBuilder appendQueryParameter(String baseUrl, String key, String encodedKey, String value) throws UnsupportedEncodingException {
        if (baseUrl.contains("?")) {
            return this.appendQueryParameter(baseUrl.substring(0, baseUrl.indexOf('?')), "query", Connection.encode("query"), "").append(encodedKey + "=").append(Connection.encode(value));
        } else {
            return this.appendQueryParameter(baseUrl, "query", Connection.encode("query"), "").append(key + "=").append(Connection.encode(value));
        }
    }

    public String appendKeyAndValue(String key, String value) throws UnsupportedEncodingException {
        String newUrl = appendKeyAndValue(getFullUrl(), key, value);
        if (newUrl.contains("?")) {
            return newUrl;
        } else {
            return getFullUrl().concat("?").concat(key).concat("=").concat(Connection.encode(value));
        }
    }

    public String appendQueryParameter(String baseUrl, String key, String encodedKey, String value) throws UnsupportedEncodingException {
        if (baseUrl.contains("?")) {
            return baseUrl.substring(0, baseUrl.indexOf('?')).concat("?").concat(encodedKey + "=").concat(Connection.encode(value));
        } else {
            return baseUrl.concat("?").concat(key).concat("=").concat(Connection.encode(value));
        }
    }

    public String appendQueryParameter(String baseUrl, String key, String value) throws UnsupportedEncodingException {
        if (baseUrl.contains("?")) {
            return baseUrl.substring(0, baseUrl.indexOf('?')).concat("?").concat(key).concat("=").concat(Connection.encode(value));
        } else {
            return baseUrl.concat("?").concat(key).concat("=").concat(Connection.encode(value));
        }
    }

    public String append(String key, String value) throws UnsupportedEncodingException {
        if (baseUrl.contains("?")) {
            return baseUrl.concat("&").concat(key).concat("=").concat(Connection.encode(value));
        } else {
            return baseUrl.concat("?").concat(key).concat("=").concat(Connection.encode(value));
        }
    }
}
