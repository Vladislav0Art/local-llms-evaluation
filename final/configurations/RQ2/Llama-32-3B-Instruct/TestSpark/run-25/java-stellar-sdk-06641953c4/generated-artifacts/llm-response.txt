import static org.junit.Assert.*;

        KeyPair keyPair = KeyPair.fromSecretSeed(seed);
        assertNotNull(keyPair);
    }

    public void fromAccountIdReturnsNullWhenAccountIdIsInvalid()[Scenario]Test {
        String accountId = "";
        KeyPair keyPair = KeyPair.fromAccountId(accountId);
        assertNull(keyPair);
    }

    public void fromPublicKeyReturnsNullWhenPublicKeyIsInvalid()[Scenario]Test {
        byte[] publicKey = new byte[0];
        KeyPair keyPair = KeyPair.fromPublicKey(publicKey);
        assertNull(keyPair);
    }

    public void fromBip39SeedReturnsNullWhenAccountNumberIsNegativeOrZero()[Scenario]Test {
        byte[] bip39Seed = "1234567890".getBytes();
        int accountNumber = 0;
        KeyPair keyPair = KeyPair.fromBip39Seed(bip39Seed, accountNumber);
        assertNull(keyPair);
    }

    @Test
    public void fromXdrPublicKeyReturnsNullWhenPublicKeyIsInvalid()[Scenario]Test {
        PublicKey publicKey = null;
        KeyPair keyPair = KeyPair.fromXdrPublicKey(publicKey);
        assertNull(keyPair);
    }

    @Test
    public void signDoesNotThrowExceptionWhenProvidedWithValidDataAndSignature()[Scenario]Test {
        byte[] data = "1234567890".getBytes();
        byte[] signature = new byte[0];
        KeyPair keyPair = new KeyPair(new EdDSAPublicKey());
        try {
            keyPair.sign(data);
        } catch (GeneralSecurityException e) {
            fail("Expected no exception");
        }
    }

    @Test
    public void signThrowsExceptionWhenProvidedWithInvalidDataOrSignature()[Scenario]Test {
        byte[] data = "1234567890".getBytes();
        byte[] signature = new byte[1];
        KeyPair keyPair = new KeyPair(new EdDSAPublicKey());
        assertThrows(GeneralSecurityException.class, () -> keyPair.sign(data, signature));
    }

    @Test
    public void verifyReturnsFalseWhenProvidedWithInvalidSignature()[Scenario]Test {
        byte[] data = "1234567890".getBytes();
        byte[] signature = new byte[1];
        KeyPair keyPair = new KeyPair(new EdDSAPublicKey());
        boolean result = keyPair.verify(data, signature);
        assertFalse(result);
    }

    @Test
    public void hashCodeIsNonZero()[Scenario]Test {
        KeyPair keyPair = new KeyPair(new EdDSAPublicKey());
        assertNotNull(keyPair.hashCode());
    }

    public void equalsReturnsTrueWhenTwoObjectsAreEqual()[Scenario]Test {
        EdDSAPublicKey publicKey = new EdDSAPublicKey();
        KeyPair keyPair1 = new KeyPair(publicKey);
        KeyPair keyPair2 = new KeyPair(publicKey);
        assertTrue(keyPair1.equals(keyPair2));
    }

    @Test
    public void equalsReturnsFalseWhenTwoObjectsAreDifferent()[Scenario]Test {
        EdDSAPublicKey publicKey = new EdDSAPublicKey();
        KeyPair keyPair1 = new KeyPair(publicKey);

        assertFalse(keyPair1.equals(keyPair2));
    }
