    def __str__(self):
        return f"({self.left} + {self.right})"

# Subtracts one expression from another

    def __str__(self):
        return f"({self.left} - {self.right})"

# Multiplies two expressions together
class Mul(BinaryOp):
    def __str__(self):
        return f"({self.left} * {self.right})"

# Divides one expression by another
class Div(BinaryOp):
    def __str__(self):
        return f"({self.left} / {self.right})"

# Represents a variable in an expression
class VarExpr:
    def __init__(self, name):
        self.name = name

    def __str__(self):
        return f"Var{{{self.name}}}"

def evaluate_expr(expr, values):
    if isinstance(expr, Constant):
        return expr.value
    elif isinstance(expr, VarExpr):
        var_name = expr.name
        return values.get(var_name)
    else:
        left_str = evaluate_expr(expr.left, values)
        right_str = evaluate_expr(expr.right, values)

        op_map = {
            Add: lambda x, y: str(eval(f"({x}) + ({y})")),
            Sub: lambda x, y: str(eval(f"({x}) - ({y})")),
            Mul: lambda x, y: str(eval(f"({x}) * ({y})")),
            Div: lambda x, y: str(eval(f"({x}) / ({y})"))
        }

        op = expr.__class__
        if op in op_map:
            return op_map[op](left_str, right_str)
        else:
            raise ValueError("Unsupported operation")

def main():
