package org.jsoup.parser;

import org.jsoup.helper.Validate;
import org.jsoup.nodes.CDataNode;
import org.jsoup.nodes.Comment;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.DocumentType;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.Entities;
import org.jsoup.nodes.Node;
import org.jsoup.nodes.TextNode;
import org.jsoup.nodes.XmlDeclaration;

public class XmlTreeBuilder {
    private Document document = new Document();
    private int stackSize = 0;
    private Element root;

    public Document parser() {
        return document;
    }

    @ParametersAreNonnullByDefault
    public void initialiseParse(Reader reader, String baseUri, String context) throws Exception {
        XmlDeclaration xmlDeclaration = new XmlDeclaration();
        this.xmlDeclaration = xmlDeclaration;
        document.setDocumentType(xmlDeclaration);
    }

    public List<Node> parseFragment(String content, String baseUrl, String context) {
        List<Node> nodes = new ArrayList<>();
        // implement logic to parse fragment and return list of nodes
        return nodes;
    }

    public void insertNode(Node node) {
        if (node instanceof Element) {
            stackSize++;
        }
    }

    public Element root() {
        return this.root;
    }

    @ParametersAreNonnullByDefault
    public boolean children() {
        return stackSize > 0;
    }

    public Token token(String type, String content) {
        Token token = new Token();
        token.type = type;
        token.content = content;
        return token;
    }
}

class Document extends org.jsoup.nodes.Document {
    private XmlDeclaration xmlDeclaration;

    public void setDocumentType(XmlDeclaration xmlDeclaration) {
        this.xmlDeclaration = xmlDeclaration;
    }

    public Element getDocumentElement() {
        root = new Element();
        return root;
    }
}

class Token {
    private final String type;
    private final String content;

    public Token(String type, String content) {
        this.type = type;
        this.content = content;
    }

    @Override
    public String toString() {
        return "{" + "type=" + type + ", content='" + content + '\'' + '}';
    }
}

    // implement logic for element

    public void close() {
        stackSize--;
    }
}
