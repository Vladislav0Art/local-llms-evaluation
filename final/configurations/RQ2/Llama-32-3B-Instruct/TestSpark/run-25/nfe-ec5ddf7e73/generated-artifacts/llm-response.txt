    public void setRenavam(String renavam) {

    }

    public void setTara(String tara) {
        this.tara = tara;
    }

    public void setCapacidadeM3(String capacidadeM3) {
        this.capacidadeM3 = capacidadeM3;
    }

    public void setCapacidadeKG(String capacidadeKG) {
        this.capacidadeKG = capacidadeKG;
    }

    public String getCodigoInterno() {
        return codigoInterno;
    }

    public String getPlaca() {
        return placa;
    }

    public boolean validarRenavam() {
        MDFStringValidador dfStringValidador = new MDFStringValidador();
        return dfStringValidador.validar(this.renavam);
    }
}

public class MDFStringValidador {
    private boolean isValid = true;

    public void setValid(boolean valid) {
        this.isValid = valid;
    }

    public boolean validar(String texto) {
        if (texto.isEmpty()) {
            this.isValid = false;
        }
        return this.isValid;
    }

    public String getIsValid() {
        return Integer.toString(this.isValid);
    }
}
