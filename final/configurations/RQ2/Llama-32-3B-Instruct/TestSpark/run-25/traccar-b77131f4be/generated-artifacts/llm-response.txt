import org.junit.Test;
import static org.junit.Assert.*;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import io.netty.buffer.ByteBuf;
import io.netty.buffer.Unpooled;

import org.traccar.BaseProtocolDecoder;
import org.traccar.DeviceSession;
import org.traccar.NetworkMessage;
import org.traccar.Position;
import org.traccar.Protocol;
import org.traccar.helper.BitBuffer;
import org.traccar.helper.UnitsConverter;

@RunWith(MockitoJUnitRunner.class)
public class GalileoProtocolDecoderTest {

    @Mock
    private Channel channel;

    @Mock
    private DeviceSession deviceSession;

    private GalileoProtocolDecoder decoder = new GalileoProtocolDecoder(Protocol.GALILEO);

    @Test
    public void decode_Galileo_Packet() throws Exception {
        Object msg = Unpooled.buffer().writeByte(1).writeByte(2);
        Object result = decoder.decode(channel, null, msg);
        assertNotNull(result);
    }

    @Test
    public void decode_Galileo_Position() throws Exception {
        BitBuffer bitBuffer = new BitBuffer();
        Position position = new Position(0, 0);
        bitBuffer.packPosition(position);
        byte[] bytes = bitBuffer.getBytes();
        Object msg = Unpooled.buffer().writeByte(3).writeBytes(bytes);
        Object result = decoder.decode(channel, null, msg);
        assertTrue(result instanceof BitBuffer);
    }

    @Test
    public void decode_Galileo_Timestamp() throws Exception {
        Date date = new Date(0);
        Calendar calendar = Calendar.getInstance();
        calendar.setTime(date);
        long timestamp = calendar.getTimeInMillis();
        Object msg = Unpooled.buffer().writeByte(4).writeLong(timestamp);
        Object result = decoder.decode(channel, null, msg);
        assertNotNull(result);
    }

    @Test
    public void decode_Galileo_Data() throws Exception {
        byte[] data = new byte[]{1, 2, 3};
        Object msg = Unpooled.buffer().writeByte(5).writeBytes(data);
        Object result = decoder.decode(channel, null, msg);
        assertTrue(result instanceof Object[]);
    }

    @Test
    public void decode_Galileo_NonPacket() throws Exception {
        Object msg = Unpooled.buffer();
        Object result = decoder.decode(channel, null, msg);
        assertNull(result);
    }
}
