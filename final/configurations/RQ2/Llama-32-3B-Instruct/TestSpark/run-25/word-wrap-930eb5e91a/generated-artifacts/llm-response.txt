import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertTrue;
import static org.junit.jupiter.api.Assertions.assertFalse;

public class WordWrapTest {

    @Test
    public void testRightTrim() {
        WordWrap wordWrap = new WordWrap();
        assertEquals(" text", wordWrap.rightTrim("  text "));
    }

    @Test
    public void testLeftTrim() {
        WordWrap wordWrap = new WordWrap();
        assertEquals("  text ", wordWrap.leftTrim("  text "));
    }

    @Test
    public void testRightTrimEmptyString() {
        WordWrap wordWrap = new WordWrap();
        assertTrue(wordWrap.rightTrim(""));
    }

    @Test
    public void testLeftTrimEmptyString() {
        WordWrap wordWrap = new WordWrap();
        assertEquals("", wordWrap.leftTrim(""));
    }

    @Test
    public void testRightTrimWhitespace() {
        WordWrap wordWrap = new WordWrap();
        assertTrue(wordWrap.rightTrim("   "));
    }

    @Test
    public void testLeftTrimWhitespace() {
        WordWrap wordWrap = new WordWrap();
        assertEquals("   ", wordWrap.leftTrim("   "));
    }

    @Test
    public void testIsWhitespaceEmptyString() {
        WordWrap wordWrap = new WordWrap();
        assertFalse(wordWrap.isWhitespace(""));
    }

    @Test
    public void testIsWhitespaceSingleSpace() {
        WordWrap wordWrap = new WordWrap();
        assertTrue(wordWrap.isWhitespace(" "));
    }

    @Test
    public void testIsWhitespaceMultipleSpaces() {
        WordWrap wordWrap = new WordWrap();
        assertTrue(wordWrap.isWhitespace("   "));
    }

    @Test
    public void testIsWhitespaceNonSpaceCharacter() {
        WordWrap wordWrap = new WordWrap();
        assertFalse(wordWrap.isWhitespace("a"));
    }
}

class WordWrap {

    public String rightTrim(String text) {
        return text.trimEnd();
    }

    public String leftTrim(String text) {
        return text.trimStart();
    }

    public boolean isWhitespace(String text) {
        return text.matches("^\\s*$");
    }

}
