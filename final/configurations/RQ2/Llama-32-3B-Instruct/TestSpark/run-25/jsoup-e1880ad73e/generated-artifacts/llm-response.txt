    }

        assertNotSame(emptyUrl, urlBuilder.build());
    }

    public void appendKeyVal() throws UnsupportedEncodingException {
        Connection.KeyVal kv = Connection.KeyVal.of("key", "val");
        urlBuilder.appendKeyVal(kv);
    }

    public void appendKeyValWithNullValue() throws UnsupportedEncodingException {
        Connection.KeyVal kv = Connection.KeyVal.of("key", null);
        urlBuilder.appendKeyVal(kv);
    }

    @Test
    public void appendKeyValEncoding() throws UnsupportedEncodingException, URLEncoder.encode() {
        String encodedValue = "value with spaces";
        URL url = new URL("http://example.com/path?param=" + encodedValue);
        assertNotSame(url, urlBuilder.build());
    }

    @Test
    public void buildWithQueryParams() throws UnsupportedEncodingException, URLEncoder.encode() {
        String encodedValue = "value with spaces";
        URL url = new URL("http://example.com/path?param=" + encodedValue);
        assertNotSame(url, urlBuilder.build());

    @Test
    public void appendKeyValWithNullKey() throws UnsupportedEncodingException {
