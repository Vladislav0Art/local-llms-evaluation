package org.jsoup.parser;

import org.jsoup.nodes.CDataNode;
import org.jsoup.nodes.Comment;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.DocumentType;
import org.jsoup.nodes.EndTag;
import org.jsoup.nodes.Node;

public class Token {
    public enum Type {
        START_TAG,
        END_TAG,
        DOCTYPE,
        CHARACTER,
        COMMENT
    }

    private String name;
    private Type type;
    private Character character;

    public Token(String name, Type type) {
        this.name = name;
        this.type = type;
    }

    public void setName(String name) {
        this.name = name;
    }
}

public class DocumentType {
    private String systemId;
    private String publicId;

    public DocumentType(String systemId, String publicId) {
        this.systemId = systemId;
        this.publicId = publicId;
    }

    public void setSystemId(String systemId) {
        this.systemId = systemId;
    }
}

public class Node {
    public static void children(Node node) {}
    public static void insert(Node token, Token tokenType) {}
    public static void append(Node node, String content) {}
}

public class XmlTreeBuilder {
    private Stack<Node> stack;

    public XmlTreeBuilder() {
        stack = new Stack<>();
    }

    public void startTag(StartTag tag) {
        Node node = new CDataNode(tag);
        stack.push(node);
    }

    public void character(Character character) {
        if (stack.peek() instanceof CDataNode) {
            ((CDataNode) stack.peek()).append("[" + character.toString() + "]");
        }
    }

    public void endTag(EndTag tag) {
        Node node = stack.pop();
        if (node != null && node instanceof CDataNode) {
            ((CDataNode) node).append("</" + tag.name() + ">");
        }
    }
}

class StartTag {}

class Character {}

class Comment {}
