import org.jsoup.Connection;
import org.jsoup.Document;
import java.net.URL;
import java.util.HashMap;
import java.util.Map;

public class UrlBuilder {
    private Connection conn;
    
    public UrlBuilder(String url) {
        this.conn = new Connection(url);
    }
    
    public void appendQueryParam(String param, String value) {
        Map<String, Object> params = new HashMap<>();
        params.put(param, value);
        this.conn.appendQueryParams(params, null);
    }
    
    public void appendKeyVal(String key, String value) {
        KeyVal kv = new KeyVal(key, StringUtil.quote(value));
        conn.appendkv(kv);
    }
    
    public URL getUrl() {
        return ((URL)conn.url()).toURI().toURL();
    }
}

class KeyVal {
    private String key;
    private String value;

    public KeyVal(String key, String value) {
        this.key = key;
        this.value = value;
    }

    public String getKey() {
        return key;
    }

    public String getValue() {
        return value;
    }
}

public class StringUtil {
    public static String quote(String str) {
        // Implementation of quote method
        return str;
    }
}

class Connection {
    private Document doc;
    
    public Connection(URI uri) {
        try {
            this.doc = org.jsoup.Jsoup.connect(uri).get();
        } catch (Exception e) {}
    }

    public void appendkv(KeyVal kv) {
        String paramString = "key=" + kv.getKey() + "&value=" + kv.getValue();
        ((String)doc.body()).replaceFirst("}", paramString);
    }
}

public class TestUrlBuilder {

    @org.junit.Test
    public void testUrlBuilder() {
        UrlBuilder urlBuilder = new UrlBuilder("https://www.example.com");
        
        urlBuilder.appendKeyVal("key", "value");
        urlBuilder.appendQueryParam("param", "value");
        
        URL url = urlBuilder.getUrl();
        System.out.println(url);
    }
}
```

Note: The `appendkv` method in the `Connection` class replaces the last occurrence of `"}` in the HTML body with the key-value pair. It may not work correctly for all cases and is only provided as a possible solution.
