import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.TextNode;
import org.jsoup.nodes.Element;

public class Main {

    public static String outerHtmlHead(String accum, int depth, StringBuilder out) {
        if (depth == 0) return "";
        Document doc = Jsoup.parse(accum);
        Element head = doc.head();
        if (head != null && !head.isEmpty()) {
            StringBuilder sb = new StringBuilder();
            for (org.jsoup.nodes.Element child : head.children()) {
                sb.append(child.outerHtml());
            }
            out.append("<head>");
            out.append(sb.toString());
            out.append("</head>");
        } else {
            out.append("<head></head>");
        }
        return accum;
    }

    public static String outerHtmlTail(String accum, int depth, StringBuilder out) {
        if (depth == 0) return "";
        Document doc = Jsoup.parse(accum);
        Element tail = doc.body().tail();
        if (tail != null && !tail.isEmpty()) {
            StringBuilder sb = new StringBuilder();
            for (org.jsoup.nodes.Element child : tail.children()) {
                sb.append(child.outerHtml());
            }
            out.append(sb.toString());
        } else {
            out.append("</body></html>");
        }
        return accum;
    }

    public static void main(String[] args) {
        String html = "<html><head><title>Example</title></head><body>Hello, world!</body></html>";
        StringBuilder out = new StringBuilder();
        outerHtmlHead(html, 1, out);
        System.out.println(out.toString());
        outerHtmlTail(out.toString(), 0, out);
        System.out.println(out.toString());
    }
}
