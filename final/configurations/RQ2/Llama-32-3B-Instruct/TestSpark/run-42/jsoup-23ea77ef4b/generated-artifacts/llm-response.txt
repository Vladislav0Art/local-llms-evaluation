public class W3CDom {

    public Node contextNode(Document doc) {

class Node {
    private org.jsoup.nodes.Element element;

    public Node(org.jsoup.nodes.Element element) {
        this.element = element;
    }

    public String outerHtml() {
        return element.outerHtml();
    }
}

public class TestW3CDom {

    @org.junit.Test
    public void testContextNode() {
        W3CDom w3cDom = new W3CDom("id");
        Document doc = w3cDom.wDoc;
        Node node = w3cDom.contextNode(doc);
        String html = node.outerHtml();
        assertTrue(html.contains("<div>"));
    }

    @org.junit.Test
    public void testElementCreation() {
        W3CDom w3cDom = new W3CDom("id");
        Document doc = w3cDom.wDoc;
        Node node = w3cDom.contextNode(doc);
        Element childElement = node.element.createElement("child");

        String html = node.outerHtml();
        assertTrue(html.contains("<div><child>"));
    }
}

class Document {
    public static class Element extends org.jsoup.nodes.Element {
        public Element(String html) {
            super.html(html);
        }

        @Override
        public String outerHtml() {
            return this.toString();
        }
    }
}
