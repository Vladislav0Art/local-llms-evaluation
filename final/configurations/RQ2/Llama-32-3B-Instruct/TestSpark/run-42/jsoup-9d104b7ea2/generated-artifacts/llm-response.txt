        String input = "   Hello, World!   ";
        String expected = "Hello, World!";
        String result = org.jsoup.helper.StringUtil.normaliseWhitespace(input);
        assertEquals(expected, result);
    }

    public static void testLastCharIsWhitespaceEmptyStringBuilder() {
        StringBuilder emptyStringBuilder = new StringBuilder();
        boolean result = org.jsoup.helper.StringUtil.lastCharIsWhitespace(emptyStringBuilder);
        assertTrue(result);
    }

    public static void testLastCharIsWhitespaceNonEmptyStringBuilder() {
        StringBuilder nonEmptyStringBuilder = new StringBuilder("Hello, World!");
        boolean result = org.jsoup.helper.StringUtil.lastCharIsWhitespace(nonEmptyStringBuilder);
        assertFalse(result);
    }

    public static void testNormaliseWhitespaceWithMultipleSpacesAndTabs() {
        String input = "   Hello,  World!  \t\t";
        String expected = "Hello, World!";
        String result = org.jsoup.helper.StringUtil.normaliseWhitespace(input);
        assertEquals(expected, result);
    }
}

class StringUtil {
    public static String normaliseWhitespace(String input) {
        return input.replaceAll("\\s+", "");
    }

    public static boolean lastCharIsWhitespace(charBuilder sb) {
        return sb.length() == 0;
    }

    public static boolean lastCharIsWhitespace(StringBuilder sb) {
        return sb.charAt(sb.length() - 1) == '\u0000';
    }
}
