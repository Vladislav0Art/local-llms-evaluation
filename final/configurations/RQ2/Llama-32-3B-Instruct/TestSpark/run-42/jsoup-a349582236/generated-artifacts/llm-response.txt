import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;
import org.jsoup.helper.Validate;

import static org.junit.Assert.*;
import static org.mockito.Mockito.when;
import static org.mockito.Mockito.mock;

@RunWith(MockitoJUnitRunner.class)
public class TagTest {

    @Mock
    private ParseSettings parseSettings;

    @Test
    public void getName_EmptyTagName ReturnsEmptyString() {
        when(parseSettings.getName()).thenReturn("");
        assertEquals("", Tag.valueOf("").getName());
    }

    @Test
    public void normalName_UndefinedTagName ReturnsUndefinedTagName() {
        when(parseSettings.getName()).thenReturn(null);
        assertNotNull(Tag.valueOf("tag").normalName());
    }

    @Test
    public void valueOf_MultipleArgs_ThrowsException() {
        assertThrows(IllegalArgumentException.class, () -> Tag.valueOf("", parseSettings));
    }

    @Test
    public void isBlock_SelfClosingTagsAreFalse() {
        when(parseSettings.isSelfClosing()).thenReturn(true);
        assertFalse(Tag.valueOf("img").isBlock());
    }

    @Test
    public void isInline_NonSelfClosingTagsAreTrue() {
        when(parseSettings.isSelfClosing()).thenReturn(false);
        assertTrue(Tag.valueOf("div").isInline());
    }

    @Test
    public void isEmpty_EmptyTagsAreFalse() {
        assertFalse(Tag.valueOf("").isEmpty());
    }

    @Test
    public void isFormListed_AllowedTagsAreKnown() {
        when(parseSettings.getAllowedTags()).thenReturn(new String[]{"form", "input"});
        assertTrue(Tag.valueOf("form").isFormListed());
    }

    @Test
    public void isSelfClosing_SelfClosingTagsOnlyReturnTrueIfFormatAsBlockIsFalse() {
        when(parseSettings.formatAsBlock()).thenReturn(false);
        assertTrue(Tag.valueOf("<img>").isSelfClosing());
    }

    @Test
    public void preserveWhitespace_SelfClosingTagsPreserveWhitespaceWhenFormatAsBlock() {
        when(parseSettings.formatAsBlock()).thenReturn(true);
        assertTrue(Tag.valueOf("<input>").preserveWhitespace());
    }
}
