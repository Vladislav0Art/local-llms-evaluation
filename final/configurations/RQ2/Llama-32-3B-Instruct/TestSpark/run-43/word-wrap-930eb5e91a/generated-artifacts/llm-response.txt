import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Optional;

public class WordWrap {

    private int width = 80; // default width

    public WordWrap(int width) {
        this.width = width;
    }

    /**
     * Wraps the given text at the specified word boundary.
     *
     * @param text the input text
     * @return the wrapped text
     */
    public String wrap(String text) {
        StringBuilder result = new StringBuilder();
        int currentLineLength = 0;

        for (String word : text.split("\\s+")) {
            if (currentLineLength + word.length() > width) {
                result.append("\n");
                currentLineLength = 0;
            }
            if (result.length() > 0) {
                result.append(" ");
                currentLineLength++;
            }
            result.append(word);
            currentLineLength += word.length();
        }

        return result.toString().trim();
    }

    /**
     * Reads text from an input stream and wraps it at the specified word boundary.
     *
     * @param input the input stream
     * @return the wrapped text, or empty if the input stream is closed
     */
    public Optional<String> from(InputStream input) {
        try (BufferedReader reader = new BufferedReader(new InputStreamReader(input))) {
            StringBuilder builder = new StringBuilder();
            String line;
            while ((line = reader.readLine()) != null) {
                builder.append(line).append("\n");
            }
            return Optional.of(wrap(builder.toString().trim()));
        } catch (IOException e) {
            return Optional.empty();
        }
    }

    /**
     * Reads text from a file and wraps it at the specified word boundary.
     *
     * @param file the file to read from
     * @return the wrapped text, or empty if the file cannot be read

    public Optional<String> from(File file) {
        try (BufferedReader reader = new BufferedReader(new FileReader(file))) {
            StringBuilder builder = new StringBuilder();
            String line;
            while ((line = reader.readLine()) != null) {
                builder.append(line).append("\n");
            }
            return Optional.of(wrap(builder.toString().trim()));
        } catch (IOException e) {
            return Optional.empty();
        }
    }
