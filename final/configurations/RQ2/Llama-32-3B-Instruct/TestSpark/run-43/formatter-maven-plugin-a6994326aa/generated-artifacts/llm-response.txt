public class CssFormatterTest {

    private CssFormatter formatter = new CssFormatter();

    @Test
    public void initNoOptionsTest() {
        Map<String, String> options = Collections.singletonMap("cssVersion", "3");
        ConfigurationSource cfg = null;
        try {
            formatter.init(options, cfg);
            fail("Expected IllegalArgumentException");
        } catch (IllegalArgumentException e) {
            assertEquals("Either cssVersion or configSource must be provided", e.getMessage());
        }
    }

    @Test
    public void initNoConfigTest() {
        Map<String, String> options = Collections.singletonMap("cssVersion", "3");
        ConfigurationSource cfg = null;
        try {
            formatter.init(options, cfg);
            fail("Expected IllegalArgumentException");
        } catch (IllegalArgumentException e) {
            assertEquals("Either cssVersion or configSource must be provided", e.getMessage());
        }
    }

    @Test
    public void initValidOptionsTest() {
        Map<String, String> options = Collections.singletonMap("cssVersion", "3");
        ConfigurationSource cfg = null;
        try {
            formatter.init(options, cfg);
        } catch (Exception e) {}
        assertNotNull(formatter);
    }

    @Test
    public void initConfigInvalidSourceTest() throws IOException {
        Map<String, String> options = Collections.singletonMap("cssVersion", "3");
        ConfigurationSource cfg = new ConfigurationSource(new StringReader("<invalidconfig/>"));
        try {
            formatter.init(options, cfg);
            fail("Expected IllegalArgumentException");
        } catch (IllegalArgumentException e) {
            assertEquals("Invalid configuration source", e.getMessage());
        }
    }

    @Test
    public void initConfigValidSourceTest() throws IOException {
        Map<String, String> options = Collections.singletonMap("cssVersion", "3");
        ConfigurationSource cfg = new ConfigurationSource(new StringReader("<validconfig/>"));
        try {
            formatter.init(options, cfg);
        } catch (Exception e) {}
        assertNotNull(formatter);
    }

    @Test
    public void doFormatEmptyStringTest() throws IOException {
        assertEquals("", formatter.doFormat("", LineEnding.LF));
    }

    @Test
    public void doFormatSingleLineCommentTest() throws IOException {
        String code = "// comment";
        assertEquals(code, formatter.doFormat(code, LineEnding.LF));
    }

    @Test
    public void doFormatMultilineCommentTest() throws IOException {
        String code = "/* comment */";
        assertEquals(code, formatter.doFormat(code, LineEnding.LF));
    }

    @Test
    public void doFormatCodeBlockTest() throws IOException {
        String code = "/*\n" +
                       "   body {\n" +
                       "     background-color: blue;\n" +
                       "   }\n" +
                       "*/";
        assertEquals(code, formatter.doFormat(code, LineEnding.LF));
    }

    @Test
    public void doFormatWithNewLineTest() throws IOException {
        String code = "body {\n" +
                      "  background-color: blue;\n" +
                      "}";
        assertEquals(code.replace("\n", "\n" + LineEnding.LF), formatter.doFormat(code, LineEnding.LF));
    }

    @Test
    public void isInitializedTest() {
        assertTrue(formatter.isInitialized());
    }
}
