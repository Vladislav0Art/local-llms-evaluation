public class XmlTreeBuilder {
    private boolean initialised = false;
    private StringBuilder currentToken;

    public XmlTreeBuilder() {
        this.currentToken = new StringBuilder();
    }

    public void insert(Token token) {
        if (token.getType() == Token.Type.Character) {
            currentToken.append(token.getValue());
        } else if (token.getType() == Token.Type.StartTag) {
            // handle start tag
            System.out.println("Start Tag: " + token.getValue());
        } else if (token.getType() == Token.Type.EndTag) {
            // handle end tag
            System.out.println("End Tag: " + token.getValue());
        }
    }

    public void popStackToClose(Token endTag) {
        if (endTag.getType() == Token.Type.EndTag && currentToken.toString().equals(endTag.getValue())) {
            initialised = true;
            System.out.println("Stack closed");
        } else {
            System.out.println("Error: unexpected tag");
        }
    }

    public boolean initialised() {
        return initialised;
    }

    public static void main(String[] args) {
        XmlTreeBuilder test = new XmlTreeBuilder();
        test.insert(new Token(XmlTreeBuilder.Token.Type.Character, "&"));
        test.insert(new Token(XmlTreeBuilder.Token.Type.StartTag, "tag"));
        test.popStackToClose(new Token(XmlTreeBuilder.Token.Type.EndTag, "</tag>"));
    }
}
