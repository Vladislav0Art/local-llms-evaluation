import org.junit.Test;
import static org.junit.Assert.*;
import java.net.URI;
import java.net.URISyntaxException;
import java.util.HashMap;
import java.util.Map;

public class UrlBuilderTest {

    @Test
    public void buildUrl_WithInputUrl_ReturnsCorrectly() throws URISyntaxException {
        // Arrange
        URL inputUrl = new URL("http://example.com/path?query=value");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);

        // Act
        URI uri = urlBuilder.build();

        // Assert
        assertEquals("http://example.com/#!/path", uri.toString());
    }

    @Test
    public void appendKeyVal_AddsKeyValueToExistingUrl() throws UnsupportedEncodingException {
        // Arrange
        URL inputUrl = new URL("http://example.com/path");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        Map<String, String> keyValue = new HashMap<>();
        keyValue.put("key", "value");

        // Act
        urlBuilder.appendKeyVal(Connection.KeyVal.KEY_VALUE_MAPPER.convert(keyValue));

        // Assert
        URI uri = urlBuilder.build();
        assertEquals("http://example.com/path?key=value", uri.toString());
    }

    @Test
    public void appendKeyVal_ThrowsMalformedURLException() {
        // Arrange
        URL inputUrl = new URL("http://example.com/path");

        // Act and Assert
        assertThrows(MalformedURLException.class, () -> new UrlBuilder(inputUrl).appendKeyVal(Connection.KeyVal.KEY_VALUE_MAPPER.convert(new HashMap<>())));
    }

    @Test
    public void build_WithoutQueryParameter_ReturnsCorrectly() throws URISyntaxException {
        // Arrange
        URL inputUrl = new URL("http://example.com/path");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);

        // Act
        URI uri = urlBuilder.build();

        // Assert
        assertEquals("http://example.com/#!/path", uri.toString());
    }

    @Test
    public void build_WithQueryParameter_ReturnsCorrectly() throws URISyntaxException {
        // Arrange
        URL inputUrl = new URL("http://example.com/path?query=value");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);

        // Act
        URI uri = urlBuilder.build();

        // Assert
        assertEquals("http://example.com/#!/path?query=value", uri.toString());
    }

    @Test
    public void appendKeyVal_WithNullValue_ThrowsUnsupportedEncodingException() {
        // Arrange
        URL inputUrl = new URL("http://example.com/path");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        Map<String, String> keyValue = new HashMap<>();
        keyValue.put("key", null);

        // Act and Assert
        assertThrows(UnsupportedEncodingException.class, () -> urlBuilder.appendKeyVal(Connection.KeyVal.KEY_VALUE_MAPPER.convert(keyValue)));
    }

}
