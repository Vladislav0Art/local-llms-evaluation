public class HttpConnectionTest {

    @Test
    public void [connect][ReturnsConnection]Test() {
        Connection connection = HttpConnection.connect("http://example.com");
        Assert.assertNotNull(connection);
    }

    @Test
    public void [connect][ReturnsURLConnection]Test() {
        URL url = new URL("http://example.com");
        Connection connection = HttpConnection.connect(url);
        Assert.assertTrue(connection instanceof URLConnection);
    }

    @Test
    public void [newRequest][ReturnsNewRequest]Test() {
        Connection connection = HttpConnection.connect("http://example.com");
        Connection request = connection.newRequest();
        Assert.assertNotNull(request);
    }

    @Test
    public void [url][String][ReturnsUrl]Test() {
        URL url = new URL("http://example.com");
        Connection connection = HttpConnection.url(url);
        Assert.assertEquals(url, connection.url());
    }

    @Test
    public void [url][String][ReturnsHttpConnection]Test() {
        String url = "http://example.com";
        Connection connection = HttpConnection.url(url);
        Assert.assertTrue(connection instanceof HttpURLConnection);
    }

    @Test
    public void [proxy][ReturnsProxyConnection]Test() {
        Proxy proxy = new Proxy(Proxy.Type.HTTP, new InetSocketAddress("proxy.example.com", 8080));
        Connection connection = HttpConnection.proxy(proxy);
        Assert.assertNotNull(connection);
    }

    @Test
    public void [proxy][String][Integer][ReturnsProxyConnection]Test() {
        String host = "proxy.example.com";
        int port = 8080;
        Connection connection = HttpConnection.proxy(host, port);
        Assert.assertNotNull(connection);
    }

    @Test
    public void [method][ReturnsHttpMethod]Test() {
        Method method = Method.GET;
        Connection connection = HttpConnection.method(method);
        Assert.assertEquals(method, connection.method());
    }

    @Test
    public void [ignoreHttpErrors][False][ReturnsTrue]Test() {
        Connection connection = HttpConnection.ignoreHttpErrors(false);
        Assert.assertFalse(connection.ignoreHttpErrors());
    }

    @Test
    public void [ignoreContentType][True][ReturnsTrue]Test() {
        Connection connection = HttpConnection.ignoreContentType(true);
        Assert.assertTrue(connection.ignoreContentType());
    }

    @Test
    public void [data][String][String][Bytes][ReturnsHttpData]Test() {
        String key = "key";
        String value = "value";
        int bytes = 1024;
        Connection connection = HttpConnection.data(key, value, bytes);
        Assert.assertNotNull(connection);
    }

    @Test
    public void [data][Map][ReturnsHttpData]Test() {
        Map<String, String> data = new HashMap<>();
        data.put("key", "value");
        Connection connection = HttpConnection.data(data);
        Assert.assertNotNull(connection);
    }

    @Test
    public void [execute][ReturnsResponse]Test() throws IOException {
        Connection connection = HttpConnection.connect("http://example.com");
        Response response = connection.execute();
        Assert.assertNotNull(response);
    }
}

            this.status = status;
