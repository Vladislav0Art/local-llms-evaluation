```java
import org.bouncycastle.asn1.x500.X500Name;
import org.bouncycastle.jce.provider.BouncyCastleProvider;
import org.bouncycastle.pkcs.PKCS8EncryptedPrivateKeyInfo;
import org.bouncycastle.pkcs.PKCS8EncodedData;
import org.bouncycastle.pkcs.PKCS8Private_key_info;
import org.bouncycastle.pkcs.PKCS10CertificationRequest;
import org.bouncycastle.pkcs.jcajce.JcaPEMEncodedSubjectPublicKeyInfo;

import javax.crypto.spec.PBEParameterSpec;
import java.security.KeyPair;
import java.security.KeyPairGenerator;
import java.security.PrivateKey;
import java.security.PublicKey;
import java.util.Base64;

// Load the Bouncy Castle provider
Security.addProvider(new BouncyCastleProvider());

public class XDRPublicKey {
    public static PublicKey generateXDRPublicKey(String privateKey) throws Exception {
        // Load the private key from PEM format
        PKCS8EncodedData encodedPrivateKey = Base64.getDecoder().decode(privateKey);
        PKCS8Private_key_info privateKeyInfo = new PKCS8Private_key_info(new JcaPEMEncodedSubjectPublicKeyInfo(encodedPrivateKey));

        // Create a key pair generator for RSA with 2048 bits
        KeyPairGenerator kpg = KeyPairGenerator.getInstance("RSA");
        kpg.initialize(2048);
        KeyPair kp = kpg.generateKeyPair();

        // Get the private key from the key pair
        PrivateKey privateKeyToUse = kp.getPrivate();

        // Encrypt the private key with PBKDF2 (1000 iterations, 32 bytes salt, 128 bits cipher)
        PBEParameterSpec pbeParams = new PBEParameterSpec(1024, 3);
        PKCS8EncryptedPrivateKeyInfo encryptedPrivateKey = PKCS8EncryptedPrivateKeyInfo.getInstance(PKCS10CertificationRequest.getEncoded(privateKeyToUse), new JcaPEMEncodedSubjectPublicKeyInfo(kp.getPublic()), Base64.getEncoder(), pbeParams);

        // Return the XDR public key
        return kp.getPublic();
    }
}
```

### Verify XDR Public Key
```java
public class XDRPublicKey {
    public static boolean verifyXDRPublicKey(String publicKey) throws Exception {
        // Load the Bouncy Castle provider
        Security.addProvider(new BouncyCastleProvider());

        // Parse the XDR public key from PEM format
        JcaPEMEncodedSubjectPublicKeyInfo parsedPublicKey = new JcaPEMEncodedSubjectPublicKeyInfo(Base64.getDecoder().decode(publicKey));

        // Create a public key factory for RSA with 2048 bits
        KeyFactory kf = KeyFactory.getInstance("RSA");

        // Get the public key from the parsed XDR public key
        PublicKey publicKeyToVerify = kf.generatePublic(parsedPublicKey);

        // Verify the public key using the original private key (decrypted)
        PrivateKey privateKeyToUse = generateXDRPrivateKey(publicKey);
        return publicKeyToVerify.verify(privateKeyToUse, new byte[0]);
    }

    public static PrivateKey generateXDRPrivateKey(String publicKey) throws Exception {
        // Load the Bouncy Castle provider
        Security.addProvider(new BouncyCastleProvider());

        // Parse the XDR public key from PEM format
        JcaPEMEncodedSubjectPublicKeyInfo parsedPublicKey = new JcaPEMEncodedSubjectPublicKeyInfo(Base64.getDecoder().decode(publicKey));

        // Create a public key factory for RSA with 2048 bits
        KeyFactory kf = KeyFactory.getInstance("RSA");

        // Get the private key from the public key factory (decrypted)
        PrivateKey privateKeyToUse = kf.generatePrivate(parsedPublicKey);

        return privateKeyToUse;
    }
}
```

```java
public class Main {
```
