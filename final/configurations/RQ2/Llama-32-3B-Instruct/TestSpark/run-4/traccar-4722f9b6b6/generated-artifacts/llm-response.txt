package org.traccar.protocol;

import io.netty.buffer.ByteBuf;
import io.netty.buffer.Unpooled;
import io.netty.channel.Channel;
import org.junit.Test;
import org.junit.Assert;
import org.mockito.Mockito;

public class WatchProtocolDecoderTest {

    @Test
    public void decodeNoMessageTest() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(Protocol.WATCH);
        byte[] result = decoder.decode(Mockito.mock(Channel.class), Mockito.anySocketAddress(), null);
        Assert.assertArrayEquals(new byte[0], result);
    }

    @Test
    public void decodeWithHasIndexTest() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(Protocol.WATCH);
        ByteBuf buf = Unpooled.buffer();
        buf.writeByte(1);
        buf.writeShort((short) 2);
        int index = 3;
        buf.readInt();
        WatchProtocolDecoder.DECODING_RESULT result = (WatchProtocolDecoder.DECODING_RESULT) decoder.decode(Mockito.mock(Channel.class), Mockito.anySocketAddress(), null);
        Assert.assertEquals(index, result.index);
    }

    @Test
    public void decodeWithManufacturerTest() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(Protocol.WATCH);
        ByteBuf buf = Unpooled.buffer();
        buf.writeByte(1);
        buf.writeShort((short) 2);
        String manufacturer = "Test";
        buf.writeBytes(manufacturer.getBytes());
        int value = buf.readInt();
        String result = decoder.getManufacturer(buf);
        Assert.assertEquals(manufacturer, result);
    }

    @Test
    public void getHasIndexReturnFalseTest() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(Protocol.WATCH);
        Assert.assertFalse(decoder.getHasIndex());
    }

    @Test
    public void getHasIndexReturnTrueTest() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(Protocol.WATCH);
        decoder.setHasIndex(true);
        Assert.assertTrue(decoder.getHasIndex());
    }

}

public class WatchProtocolDecoder extends BaseProtocolDecoder {

    private boolean hasIndex;

    public WatchProtocolDecoder(Protocol protocol) {
        super(protocol);
    }

    @Override
    protected Object decode(Channel channel, SocketAddress remoteAddress, Object msg) throws Exception {
        if (hasIndex) {
            return getDecodingResult();
        }
        return null;
    }

    public boolean getHasIndex() {
        return hasIndex;
    }

    public void setHasIndex(boolean hasIndex) {
        this.hasIndex = hasIndex;
    }

    protected WatchProtocolDecoder.DECODING_RESULT getDecodingResult() {
        // implementation
    }

}

public class WatchProtocolDecoder.DECODING_RESULT {

    private int index;

    public WatchProtocolDecoder.DECODING_RESULT(int index) {
        this.index = index;
    }

    public int getIndex() {
        return index;
    }
}
