    @Mock

    public void testParseToJsonCorrectlyWhenProvidedValidJsonStringAndClass() {
        when(jsonTransformer.getObjectMapper().findAndBindJavaType(any(), any())).thenReturn(new Object());
        String json = "{\"name\":\"John\"}";
        Class<?> clazz = Object.class;
        jsonTransformer.parse(json, clazz);
        assertEquals("{\"name\":\"John\"}", jsonTransformer.toJsonString());
    }

    public void testConvertToJsonCorrectlyWhenProvidedValidObject() {
        when(jsonTransformer.getObjectMapper().writeValueAsString(any())).thenReturn "{\"name\":\"John\"}");
        Object obj = new Object();
        obj.getClass().getMethod("getName").invoke(obj);
        String result = jsonTransformer.convert(obj);
        assertEquals("{\"name\":\"John\"}", result);
    }
