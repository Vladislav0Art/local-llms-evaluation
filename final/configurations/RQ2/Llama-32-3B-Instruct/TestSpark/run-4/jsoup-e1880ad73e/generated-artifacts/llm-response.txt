    private Connection.KeyVal kv;

    private URL url;

    @Test
    public void newUrlBuilder_fromNull() {
        UrlBuilder builder = new UrlBuilder(null);
        assertTrue(builder == null);
    }

    public void build_urlAlreadySet() {
        UrlBuilder builder = new UrlBuilder(url);
        assertEquals(url, builder.build());
    }

    public void build_newUrl() throws MalformedURLException, URISyntaxException {
        String baseUrl = "http://example.com";
        URI path = uri;
        UrlBuilder builder = new UrlBuilder(baseUrl);
        URL expectedUrl = new URL(baseUrl + path);
        assertEquals(expectedUrl, builder.build());
    }

    public void appendKeyVal_empty() throws UnsupportedEncodingException {
        UrlBuilder builder = new UrlBuilder(url);
        builder.appendKeyVal(kv);
        assertTrue(builder.build().toString().isEmpty());
    }

    @Test
    public void appendKeyVal_addsSinglePair() throws UnsupportedEncodingException, MalformedURLException, URISyntaxException {
        String baseUrl = "http://example.com";
        URI path = uri;

        kv.put("key", "value");
        UrlBuilder builder = new UrlBuilder(baseUrl);
        builder.appendKeyVal(kv);
        assertEquals(baseUrl + "?" + Arrays.toString(kv), builder.build().toString());
    }

    public void appendKeyVal_addsMultiplePairs() throws UnsupportedEncodingException, MalformedURLException, URISyntaxException {

        Map<String, String> kv1 = new HashMap<>();

        Map<String, String> kv2 = new HashMap<>();
        kv2.put("key2", "value2");
        UrlBuilder builder = new UrlBuilder(baseUrl);
        builder.appendKeyVal(kv1);
        builder.appendKeyVal(kv2);
        assertEquals(baseUrl + "?" + Arrays.toString(kv1) + "&" + Arrays.toString(kv2), builder.build().toString());
    }

    @Test
    public void appendKeyVal_multiplePairsWithSameKey() throws UnsupportedEncodingException, MalformedURLException, URISyntaxException {

        URI path = uri;
        Map<String, String> kv1 = new HashMap<>();
        kv1.put("key1", "value1");

        kv2.put("key1", "value2");
        UrlBuilder builder = new UrlBuilder(baseUrl);
        builder.appendKeyVal(kv1);

    public void build_urlDecodedPath() throws MalformedURLException, URISyntaxException {

        URI path = new URI("http%3A%2F%2Fexample.com/path");

public class UrlBuilder {
    private String baseUrl;

    public UrlBuilder(String baseUrl) {
        this.baseUrl = baseUrl;
    }

    public void appendKeyVal(String key, String value) {
        if (baseUrl.isEmpty()) {
            baseUrl += key + "=" + value;
        } else {
            if (!baseUrl.endsWith("&")) {
                baseUrl += "&";
            }
            baseUrl += key + "=" + value;
        }
    }

    public String build() {
        return baseUrl;
    }
}

public class TestUrlBuilder {

    @Test
    public void testUrlBuilder() {
        UrlBuilder urlBuilder = new UrlBuilder("http://example.com");
        urlBuilder.appendKeyVal("key1", "value1");
        urlBuilder.appendKeyVal("key2", "value2");

        String result = urlBuilder.build();

        // Test expected URL
        assertEquals("http://example.com/key1=value1&key2=value2", result);
    }

    @Test
    public void testUrlBuilderEmptyBaseURL() {
        UrlBuilder urlBuilder = new UrlBuilder("");
        urlBuilder.appendKeyVal("key1", "value1");

        String result = urlBuilder.build();

        // Test expected URL
        assertEquals("key1=value1", result);
    }
}

public class UrlBuilder {
    private String baseUrl;

    public UrlBuilder(String baseUrl) {
        this.baseUrl = baseUrl;
    }

    public void appendKeyVal(String key, String value) {
        if (baseUrl.isEmpty()) {
            baseUrl += key + "=" + value;
        } else {
            if (!baseUrl.endsWith("&")) {
                baseUrl += "&";
            }
            baseUrl += key + "=" + value;
        }
    }

    public String build() {
        return baseUrl;
    }

    public static UrlBuilder createUrlBuilder(String baseUrl) {
        return new UrlBuilder(baseUrl);
    }

    public void appendKeyValWithSpaces(String key, String value) {
        if (baseUrl.isEmpty()) {
            baseUrl += key + "=" + value;
        } else {
            if (!baseUrl.endsWith("&")) {
                baseUrl += "&";
            }
            baseUrl += key + "=" + value;
        }
    }

    public static UrlBuilder createUrlBuilderWithSpaces(String baseUrl) {
        return new UrlBuilder(baseUrl);
    }

    public void appendQuery(String query) {
        if (baseUrl.isEmpty()) {
            baseUrl += "?" + query;
        } else {
            if (!baseUrl.endsWith("?")) {
                baseUrl += "&" + query;
            } else {
                baseUrl += "?" + query;
            }
        }
    }

    public static UrlBuilder createUrlBuilderWithQuery(String baseUrl, String query) {
        return new UrlBuilder(baseUrl);
    }
}
