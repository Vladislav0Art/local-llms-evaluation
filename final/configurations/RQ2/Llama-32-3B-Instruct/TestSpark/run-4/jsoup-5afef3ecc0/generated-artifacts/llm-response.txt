import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class Main {

    public static void main(String[] args) {
        testUrlBuilder();
    }

    private static void testUrlBuilder() {
        UrlBuilder urlBuilder = new UrlBuilder();

        System.out.println(urlBuilder.build().toString()); // empty URL

        urlBuilder.appendQuery("key=value");
        System.out.println(urlBuilder.build().toString()); // key=value

        urlBuilder.appendFragment("#fragment");
        System.out.println(urlBuilder.build().toString()); // key=value#fragment

        Document document = Jsoup.parse("<div>Hello World!</div>");
        urlBuilder.buildUrl(document); // should not add fragment if the URL is already present
    }

    public static class UrlBuilder {
        private String base_url;
        private String query;
        private String fragment;

        public UrlBuilder() {
            this.base_url = "https://example.com";
        }

        public void appendQuery(String query) {
            if (this.query.isEmpty()) {
                this.query = query;
            } else {
                this.query += "&" + query;
            }
        }

        public void appendFragment(String fragment) {
            if (!fragment.isEmpty() && !this.fragment.isEmpty()) {
                throw new RuntimeException("Cannot append both query and fragment");
            } else if (this.fragment.isEmpty()) {
                this.fragment = fragment;
            }
        }

        public String buildUrl(String document) {
            Document doc = Jsoup.parse(document);
            Element body = doc.body();
            if (!body.text().equals("Hello World!")) {
                return base_url + "?" + query + "&fragment=" + fragment;
            } else {
                return base_url + "?query=" + query + "&fragment=" + fragment;
            }
        }

        public String build() {
            return buildUrl(null);
        }
    }
}
