import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class GeneratedTest {

    @org.junit.Test
    public void testGeneratedHtml() throws Exception {
        Document document = Jsoup.parse("<html><head><title>page</title></head><body>Hello World!</body></html>");
        Element node = new Element(document);
        String html = node.outerHtmlHead().toString();
        assertEquals("", html);
        assertEquals("", node.outerHtmlTail().toString());
    }

    @org.junit.Test
    public void testGeneratedWhitespace() {
        String input = "   ";
        String expected = "   ";
        String actual = StringUtil.normaliseWhitespace(input);
        assertEquals(expected, actual);

        input = "";
        expected = "";
        actual = StringUtil.normaliseWhitespace(input);
        assertEquals(expected, actual);

        input = "test";
        expected = "test";
        actual = StringUtil.normaliseWhitespace(input);
        assertEquals(expected, actual);
    }

    @org.junit.Test
    public void testLastCharIsWhitespace() {
        StringBuilder sb = new StringBuilder(" ");
        assertTrue(StringBuilder.lastCharIsWhitespace(sb));
        assertFalse(StringBuilder.lastCharIsWhitespace(new StringBuilder("abc")));
    }
}

public class StringUtil {

    public static String normaliseWhitespace(String input) {
        return input.replaceAll("\\s+", " ");
    }

    public static boolean lastCharIsWhitespace(StringBuilder sb) {
        int index = sb.length() - 1;
        char c = sb.charAt(index);
        if (c == ' ') {
            return true;
        } else {
            return false;
        }
    }
}

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class GeneratedTest {

    public static String outerHtmlHead(Element node, int start, OutputSettings settings) {
        return "<head>" + node.outerHtml(start, settings.empty()) + "</head>";
    }

    public static String outerHtmlTail(Element node, int end, OutputSettings settings) {
        return "<body>" + node.outerHtml(end, settings.empty()) + "</body>";
    }
}

public class StringUtil {

    public static String normaliseWhitespace(String input) {
        StringBuilder sb = new StringBuilder();
        for (char c : input.toCharArray()) {
            if (!Character.isWhitespace(c)) {
                sb.append(c);
            }
        }
        return sb.toString();
    }

    public static boolean lastCharIsWhitespace(StringBuilder sb) {
        int index = sb.length() - 1;
        char c = sb.charAt(index);
        return Character.isWhitespace(c);
    }
}

public class OutputSettings {

    public boolean empty() {
        return true;
    }
}
