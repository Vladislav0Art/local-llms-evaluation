public class DeltaT {
    private final Map<Date, Double> estimateMap;

        this.estimateMap = new HashMap<>();
    }

    public void setEstimateMap(Map<Date, Double> map) {
        this.estimateMap.putAll(map);
    }

    public double estimate(Date date) {
        return estimateMap.getOrDefault(date, 0.0);
    }
}

public class When {
    private final DeltaT deltaT = new DeltaT();

    public void whenDeltaTEstimate(Date date) {
        Map<Date, Double> map = new HashMap<>();
        map.put(date, 0.05L);
        deltaT.setEstimateMap(map);
    }

    public void whenMockDeltaTEstimate(double value) {
        Map<Date, Date> mockDateMap = new HashMap<>();
        mockDateMap.put(new Date(), new Date());
        deltaT.setMockDateMap(mockDateMap);
        deltaT.setEstimateMap(Map.of(new Date(), value));
    }
}

public class DeltaTEstimator {

    private final Map<Date, Double> estimateMap;

    public DeltaTEstimator() {

        this.estimateMap.putAll(map);
    }

    public double estimate(Date date) {
        return estimateMap.getOrDefault(date, 0.0);
    }

    public DeltaT mockDate() {
        Map<Date, Date> mockDateMap = new HashMap<>();

    }
}
