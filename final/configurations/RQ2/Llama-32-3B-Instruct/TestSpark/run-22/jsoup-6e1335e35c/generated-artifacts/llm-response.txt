import java.util.*;

public class Element {
    private String tag;
    private List<String> attributes;
    private List<String> children;

    public Element(String tag) {
        this.tag = tag;
        this.attributes = new ArrayList<>();
        this.children = new ArrayList<>();
    }

    public void addAttribute(String key, String value) {
        attributes.add(key + "=" + value);
    }

    public void addChild(Element child) {
        children.add(child);
    }

    public String getTag() {
        return tag;
    }

    public List<String> getAttributes() {
        return attributes;
    }

    public List<Element> getChildren() {
        return children;
    }
}

public class HTMLParser {

    public static Element parse(String html) {
        // Simple parser that only handles basic structure
        if (html.startsWith("<")) {
            int endIndex = html.indexOf(">");
            String tag = html.substring(1, endIndex).trim();
            if (tag.equals("/")) {
                return null;
            }
            Element element = new Element(tag);
            String content = html.substring(endIndex + 1).trim();
            if (!content.isEmpty()) {
                element.addChild(parse(content));
            }
            return element;
        } else if (html.startsWith("<")) {
            // Handle self-closing tags
            int endIndex = html.indexOf("/>");
            String tag = html.substring(1, endIndex).trim();
            Element element = new Element(tag);
            String content = html.substring(endIndex + 2).trim();
            if (!content.isEmpty()) {
                element.addChild(parse(content));
            }
            return element;
        } else {
            return null;
        }
    }

    public static void printElement(Element element) {
        System.out.print("<" + element.getTag());
        for (String attribute : element.getAttributes()) {
            String[] keyValue = attribute.split("=");
            System.out.print(" " + keyValue[0] + "=\"" + keyValue[1].trim() + "\"");
        }
        System.out.println(">");

        if (!element.getChildren().isEmpty()) {
            for (Element child : element.getChildren()) {
                printElement(child);
            }
        }

        System.out.println("</" + element.getTag() + ">");
    }

    public static void main(String[] args) {
        String html = "<html><body><h1>Heading</h1><p>This is a paragraph.</p></body></html>";
        Element root = parse(html);
        printElement(root);
    }
}
