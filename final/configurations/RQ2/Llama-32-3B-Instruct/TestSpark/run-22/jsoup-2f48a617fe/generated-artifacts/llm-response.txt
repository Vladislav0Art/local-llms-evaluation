import org.jsoup.nodes.Comment;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.Node;
import org.jsoup.parser.ParseSettings;
import org.jsoup.parser.Parser;
import org.junit.Before;
import org.junit.Test;
import static org.hamcrest.CoreMatchers.is;
import static org.junit.Assert.assertThat;
import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.util.Map;

public class CommentTest {

    @Test
    public void newNodeDataIsCorrect()[Scenario] {
        Comment comment = new Comment("test");
        assertThat(comment.getData(), is("test"));
    }

    @Test
    public void setNewNodeDataAndVerifyIt()[Scenario] {
        Comment comment = new Comment("");
        comment.setData("new test");
        assertThat(comment.getData(), is("new test"));
    }

    @Test
    public void nodeNameIsCorrect()[Scenario] {
        Comment comment = new Comment("");
        assertThat(comment.nodeName(), is(""));
    }

    @Test
    public void cloneCreatesNewCommentWithSameData()[Scenario] {
        Comment comment = new Comment("test");
        Comment clonedComment = comment.clone();
        assertThat(clonedComment.getData(), is("test"));
    }

    @Test
    public void outerHtmlHeadIsCalledWhenCreated()[Scenario] {
        Document document = new Document();
        Appendable accum = new ByteArrayOutputStream();
        int depth = 0;
        Comment comment = new Comment("");
        comment.outerHtmlHead(accum, depth, new Document.OutputSettings());
        assertThat(document.getOuterHTML(), is("<!-- test -->"));
    }

    @Test
    public void outerHtmlTailIsCalledWhenCreated()[Scenario] {
        Document document = new Document();
        Appendable accum = new ByteArrayOutputStream();
        int depth = 0;
        Comment comment = new Comment("");
        comment.outerHtmlHead(accum, depth, new Document.OutputSettings());
        comment.outerHtmlTail(accum, depth, new Document.OutputSettings());
        assertThat(document.getOuterHTML(), is("<!-- test -->\n"));
    }

    @Test
    public void outerHtmlIsNotCalledWhenCloned()[Scenario] {
        Document document = new Document();
        Appendable accum = new ByteArrayOutputStream();
        int depth = 0;
        Comment comment = new Comment("");
        Comment clonedComment = comment.clone();
        clonedComment.outerHtmlHead(accum, depth, new Document.OutputSettings());
        assertThat(document.getOuterHTML(), is(""));
    }

    @Test
    public void xmlDeclarationIsFalseByDefault()[Scenario] {
        Comment comment = new Comment("");
        assertThat(comment.isXmlDeclaration(), is(false));
    }

    @Test
    public void asXmlDeclarationReturnsNullWhenNotXmlDeclaration()[Scenario] {
        Comment comment = new Comment("");
        assertNull(comment.asXmlDeclaration());
    }

    @Test
    public void asXmlDeclarationIsCorrectForXmlComment()[Scenario] {
        Element element = new Element("xml");
        XmlDeclaration xmlDeclaration = new XmlDeclaration("test", "test");
        element.asXmlDeclaration(xmlDeclaration);
        Comment comment = new Comment("");
        comment.setData("<!-- " + xmlDeclaration.getDeclaration() + " -->");
        assertThat(comment.asXmlDeclaration().getDeclaration(), is(xmlDeclaration.getDeclaration()));
    }

    @Test
    public void cloneCreatesCommentWithCorrectParent()[Scenario] {
        Document document = new Document();
        Appendable accum = new ByteArrayOutputStream();
        int depth = 0;
        Comment comment = new Comment("");
        comment.outerHtmlHead(accum, depth, new Document.OutputSettings());
        Element parent = ((Element) comment.getParent()).asXmlDeclaration().getDeclaration();
        Comment clonedComment = comment.clone();
        assertThat(clonedComment.getData(), is(parent));
    }

    @Test
    public void setAndVerifyParentOfComment()[Scenario] {
        Comment comment = new Comment("");
        Element parent = new Element("test");
        comment.setData("<!-- " + parent.getOuterHTML() + " -->");
        Comment clonedComment = comment.clone();
        assertThat(clonedComment.getData(), is(parent));
    }

}
