import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

import java.io.IOException;
import java.net.URL;
import java.util.Arrays;

public class GeneratedTest {
    public static String build(String url) throws IOException, URISyntaxException {
        Document document = Jsoup.connect(url).get();
        StringBuilder sb = new StringBuilder();
        for (Element element : document.body().getAllElements()) {
            sb.append(element.text());
        }
        return sb.toString();
    }

    public void appendKeyVal(String key, String val, String url) throws IOException, URISyntaxException {
        try {
            Document document = Jsoup.connect(url).get();
            Elements elements = document.select("body");
            StringBuilder sb = new StringBuilder();
            for (Element element : elements) {
                sb.append(element.text());
            }
            if (!sb.toString().contains(key)) {
                sb.append(key);
                sb.append(val);
            }
        } catch (URISyntaxException e) {
            throw new URISyntaxException("Invalid URL", "URL cannot be parsed");
        }
    }

                sb.append(key);
                sb.append(val);
            }
        } catch (URISyntaxException e) {
            throw new URISyntaxException("Invalid URL", "URL cannot be parsed");
        }
    }

                if (!sb.toString().contains(k)) {
                    sb.append(k);
                    sb.append(v);
                }
            }
        } catch (URISyntaxException e) {
            throw new URISyntaxException("Invalid URL", "URL cannot be parsed");
        }
    }
