Here's a complete example of how to parse an Amazon Machine Image (AMI) name and extract the version number:

```java
import org.hamcrest.MatcherAssert;
import org.hamcrest.Matchers;

public class AppVersion {
    private final String version;

    public AppVersion(String version) {
        this.version = version;
    }

    public static AppVersion parseName(String amiName) {
        if (amiName == null || amiName.isEmpty()) {
            return null; // Invalid or empty amiName
        }
        int startIndex = 4; // 'ami-' prefix start index
        String potentialVersion = amiName.substring(startIndex);
        return new AppVersion(potentialVersion);
    }

    public String getVersion() {
        return version;
    }
}

class GeneratedTest {

    @org.junit.Test
    public void testAppVersionParseNameNull() {
        AppVersion expected = null;
        String amiName = null;
        assertThat(AppVersion.parseName(amiName), is(expected));
    }

    @org.junit.Test
    public void testAppVersionParseNameEmpty() {
        AppVersion expected = null;
        String amiName = "";
        assertThat(AppVersion.parseName(amiName), is(expected));
    }

    @org.junit.Test
    public void testAppVersionParseNameMatching() {
        AppVersion expected = new AppVersion("test");
        String amiName = "ami-0c01fa5b2e8d4a2bb";
        assertThat(AppVersion.parseName(amiName), is(expected));
    }

    @org.junit.Test
    public void testAppVersionParseNameNonMatching() {
        AppVersion expected = null;
        String amiName = "ami-bla1ba1b1ba1b1b";
        assertThat(AppVersion.parseName(amiName), is(expected));
    }

    @org.junit.Test
    public void testAppVersionGetVersion() {
        AppVersion expected = new AppVersion("test");
        String version = expected.getVersion();
        assertThat(version, Matchers.is("test"));
    }
}
```

This code defines a class `AppVersion` to hold the parsed version number. The static method `parseName` takes an AMI name and attempts to extract the version number. It first checks for null or empty input, then extracts the potential version number by substringging from index 4 onwards (since 'ami-' is a prefix).

import org.hamcrest.MatcherAssert;
import org.hamcrest.Matchers;

public class AppVersion {
    private final String version;

    public AppVersion(String version) {
        this.version = version;
    }

    public static AppVersion parseName(String amiName) {
        if (amiName == null || amiName.isEmpty()) {
            return null; // Invalid or empty amiName
        }
        int startIndex = 4; // 'ami-' prefix start index
        String potentialVersion = amiName.substring(startIndex);
        
        if (!potentialVersion.startsWith("v")) {
            throw new IllegalArgumentException("Invalid version format. Expected 'vX.X.X'");
        }
        
        return new AppVersion(potentialVersion.substring(1));
    }

    public String getVersion() {
        return version;
    }
}

        String amiName = "";
        assertThat(AppVersion.parseName(amiName), is(expected));
    }

        String amiName = "ami-0c01fa5b2e8d4a2bbv1.2.3";

    }

    }
