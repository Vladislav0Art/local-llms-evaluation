import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import com.crowdin.client.core.http.exceptions.CrowdinApiException;
import com.crowdin.client.core.http.impl.json.JacksonJsonTransformer;
import com.crowdin.client.projectsgroups.model.Project;
import com.fasterxml.jackson.databind.ObjectMapper;

import static org.junit.Assert.assertEquals;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class JacksonJsonTransformerTest {

    @Mock
    private ObjectMapper objectMapper;

    @Test
    public void parseToJson_WhenNoDataReturnsEmptyObject() {
        when(objectMapper.createModule()).thenReturn(new SimpleModule());
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        Project project = transformer.parse("", Project.class);
        assertEquals(null, project);
    }

    @Test
    public void parseToJson_WhenDataReturnsValidProject() {
        String json = "{\"id\": 1, \"name\": \"Crowdin\"}";
        when(objectMapper.createModule()).thenReturn(new SimpleModule());
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        Project project = transformer.parse(json, Project.class);
        assertEquals(1L, project.getId());
        assertEquals("Crowdin", project.getName());
    }

    @Test
    public void parseToJson_WhenDataReturnsInvalidProject_throwsException() {
        String json = "{\"id\": \"abc\", \"name\": \"Crowdin\"}";
        when(objectMapper.createModule()).thenReturn(new SimpleModule());
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        try {
            transformer.parse(json, Project.class);
            fail("Expected CrowdinApiException to be thrown");
        } catch (CrowdinApiException e) {
            // expected
        }
    }

    @Test
    public void convertToJson_WhenNoDataReturnsEmptyString() {
        when(objectMapper.createModule()).thenReturn(new SimpleModule());
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        String json = transformer.convert(null);
        assertEquals("", json);
    }

    @Test
    public void convertToJson_WhenDataReturnsValidProject() {
        Project project = new Project(1L, "Crowdin");
        when(objectMapper.createModule()).thenReturn(new SimpleModule());
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        String json = transformer.convert(project);
        assertEquals("{\"id\": 1, \"name\": \"Crowdin\"}", json);
    }

    @Test
    public void convertToJson_WhenDataReturnsInvalidProject_throwsException() {
        Project project = new Project(1L, " Crowdin");
        when(objectMapper.createModule()).thenReturn(new SimpleModule());
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        try {
            transformer.convert(project);
            fail("Expected CrowdinApiException to be thrown");
        } catch (CrowdinApiException e) {
            // expected
        }
    }

    @Test
    public void convertToJson_WhenDataReturnsUnsupportedType_throwsException() {
        Project project = new Object(1L, "Crowdin"); // not supported type
        when(objectMapper.createModule()).thenReturn(new SimpleModule());
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        try {
            transformer.convert(project);
            fail("Expected CrowdinApiException to be thrown");
        } catch (CrowdinApiException e) {
            // expected
        }
    }

}
