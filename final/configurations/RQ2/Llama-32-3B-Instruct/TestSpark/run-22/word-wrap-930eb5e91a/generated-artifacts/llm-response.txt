    public void fromReader_ReturnsBuilder() {

        assertTrue(WordWrap.from(reader) instanceof Builder);
    }

    @Test
    public void fromClasspathUtf8_ReturnsBuilder() {
        String resource = "test";
        assertSame(WordWrap.fromClasspathUtf8(resource), WordWrap.from());
    }

    @Test
    public void fromUtf8_ReturnsBuilder() {
        byte[] bytes = "Hello World!".getBytes();
        InputStream in = new ByteArrayInputStream(bytes);
        assertTrue(WordWrap.fromUtf8(in) instanceof Builder);
    }

    @Test
    public void wordWrap_NoInput_ReturnsEmptyLine() throws IOException {

                false, false);
        assertTrue(out.toString().equals("\n"));
    }

    @Test
    public void wordWrap_BreakWords_ReturnsWrappedText() throws IOException {
        Reader in = new BufferedReader(new FileReader("test.txt"));
        PrintWriter out = new PrintWriter(new StringWriter());
        WordWrap.wordWrap(in, out, "\n", 10L, Mockito.any(), Mockito.emptySet(),
                true, false);
        assertNotNull(out.toString());
    }

    @Test
    public void isWhitespace_AnySpaceReturnsTrue() {
        assertTrue(WordWrap.isWhitespace("   "));
    }

    @Test
    public void rightTrim_EmptyString_ReturnsEmptyString() {
        assertSame("", WordWrap.rightTrim(""));
    }

    @Test
    public void leftTrim_NothingToTrim_ReturnsOriginal() {

        WordWrap.leftTrim(word);

    }

}

public class WordWrap {
    public static StringBuilder leftTrim(String original) {
        return new StringBuilder(original);
    }

    public static StringBuilder rightTrim(String original) {
        int length = original.length();
        char[] chars = original.toCharArray();
        for (int i = 0; i < length; ++i) {
            if (!Character.isWhitespace(chars[i])) {
                break;
            }
        }
        return new StringBuilder(original.substring(i));
    }

    public static boolean isWhitespace(String input) {
        return input.isEmpty() || input.trim().isEmpty();
    }
}
