public class CommentsConfigurationTest {

    private CommentsConfiguration commentsConfiguration;

    @Before
    public void setUp() {
        commentsConfiguration = new CommentsConfiguration();
    }

    public void testEmptyConstructor() {
        assertTrue(commentsConfiguration instanceof CommentsConfiguration);
    }

    public void testCommentLinesSetForPath() {
        Map<String, List<String>> commentsMap = new HashMap<>();
        commentsMap.put("path", Arrays.asList("line1", "line2"));
        commentsConfiguration = new CommentsConfiguration(commentsMap);
        assertEquals(commentsMap, commentsConfiguration.getAllComments());
    }

    public void testNoCommentsForNewEntry() {
        Map<String, List<String>> commentsMap = new HashMap<>();
        commentsConfiguration.setComment("path", "", "line1");
        assertEquals(HashMap.of("path", Collections.singletonList("line1")), commentsConfiguration.getAllComments());
    }

    public void testOverwriteExistingComments() {
        Map<String, List<String>> commentsMap = new HashMap<>();
        commentsMap.put("path", Arrays.asList("line1", "line2"));
        commentsConfiguration.setComment("path", "", "");
        assertEquals(HashMap.of("path", Collections.singletonList ""), commentsConfiguration.getAllComments());
    }

    public void testNullPathThrowsNullPointerException() {
        assertThrows(NullPointerException.class, () -> commentsConfiguration.setComment(null, "line1", "line2"));
    }

    public void testNullCommentLinesThrowsNullPointerException() {
        Map<String, List<String>> commentsMap = new HashMap<>();
        commentsMap.put("path", Arrays.asList("line1", "line2"));
        assertThrows(NullPointerException.class, () -> commentsConfiguration.setComment("path", null, null));
    }
}
