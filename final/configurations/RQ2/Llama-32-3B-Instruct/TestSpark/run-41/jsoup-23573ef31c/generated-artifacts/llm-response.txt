```java
public class Main {
    // public class to generate the HTML document
    public static String generateHtml(String title, String content) {
        StringBuilder html = new StringBuilder();
        html.append("<!DOCTYPE html>\n");
        html.append("<html lang='en'>\n");
        html.append("<head>\n");
        html.append("<meta charset='UTF-8'>\n");
        html.append("<title>").append(title).append("</title>\n");
        html.append("</head>\n");
        html.append("<body>\n");
        html.append("<h1>").append(title).append("</h1>\n");
        html.append(content);
        html.append("</body>\n");
        html.append("</html>");
        return html.toString();
    }

    // public class to generate the HTML document
    public static Element createElement(String tagName, String content) {
        return new Element(tagName, content);
    }

    public static void main(String[] args) {
        
        // Generate a string that represents an HTML element with a tag name and some content.
        String html = generateHtml("Page Title", "<p>This is the content of the page.</p>");

        // Create a new HTML element using the createElement method.
        Element h1Tag = createElement("h1", "This is the title of the webpage.");

        // Append the h1 tag to the html string.
        html += h1Tag.toString();

        System.out.println(html);
    }
}

// Define an interface for elements
interface Element {
    String toString();
}

// Implement the element class
class ElementImpl implements Element {
    private String tagName;
    private String content;

    public ElementImpl(String tagName, String content) {
        this.tagName = tagName;
        this.content = content;
    }

    @Override
    public String toString() {
        return "<" + tagName + ">" + content + "</" + tagName + ">";
    }
}
```

```python
from xml.dom import minidom

class HtmlDocument:
    def __init__(self, title):
        self.title = title
        self.content = ""

    def add_content(self, content):
        self.content += str(content)

    def generate_html(self):
        html = f"<html><head><title>{self.title}</title></head><body>{self.content}</body></html>"
        return html

class XmlDocument:
    def __init__(self, title):
        self.title = title
        self.content = ""

    def add_content(self, content):
        self.content += str(content)

    def generate_xml(self):
        xml = minidom.parseString(f"<{self.title}>\n{self.content}\n</{self.title}>")
        return xml.toprettyxml()

def main():
    html_doc = HtmlDocument("Page Title")
    html_doc.add_content("<p>This is the content of the page.</p>")
    print("HTML Document:")
    print(html_doc.generate_html())

    xml_doc = XmlDocument("Book")
    xml_doc.add_content("<title>The Great Gatsby</title><author>F. Scott Fitzgerald</author>")
    xml_xml = """
        <?xml version="1.0" encoding="UTF-8"?>
        <catalog>
          <book id="bk101">
            <title>{}</title>
            <author>{}</author>
            <genre> Fiction </genre>
            <price> 19.99 </price>
            <publish_date>2003-10-01</publish_date>
            <description>An entirely fictional tale.</description>
          </book>
        </catalog>
    """.format(xml_doc.title, xml_doc.author)
    print("\nXML Document:")
    print(xml_xml)

if __name__ == "__main__":
    main()
```

In this revised code, we have two classes: `HtmlDocument` and `XmlDocument`. Both classes represent HTML and XML documents respectively. Each class has methods to add content and generate the document in its respective format.

The `generate_html` method generates an HTML document with a title and some content. The `generate_xml` method generates an XML document with a title, author, and book details using string formatting.

In the `main` function, we create instances of these classes, add content to them, and then print out the generated documents in both HTML and XML formats.
