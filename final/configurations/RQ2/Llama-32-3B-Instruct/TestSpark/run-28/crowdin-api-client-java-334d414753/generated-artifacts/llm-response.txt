public class JacksonJsonTransformerTest {

    private ObjectMapper objectMapper;

    @Before
    public void setup() {
        objectMapper = new ObjectMapper();
        SimpleModule module = new SimpleModule();
        module.addDeserializer(Project.class, new ProjectDeserializer());
        module.addDeserializer(ExportOptions.class, new ExportOptionsDeserializer());
        module.addDeserializer(FileInfo.class, new FileInfoDeserializer());
        module.addDeserializer(ImportOptions.class, new ImportOptionsDeserializer());
        module.addDeserializer(LanguageTranslations.class, new LanguageTranslationsDeserializer());
        objectMapper.registerModule(module);
    }

    public void testParseSuccess() {
        Project project = new Project();
        project.setName("TestProject");
        project.setLanguage("en");
        String json = objectMapper.writeValueAsString(project);
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        Object result = transformer.parse(json, Project.class);
        assertNotNull(result);
        assertSame(project, result);
    }

    public void testParseFail() {
        String json = "{\"name\":\"TestProject\"}";
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        try {
            transformer.parse(json, Project.class);
            assert false;
        } catch (CrowdinApiException e) {
            // expected
        }
    }

    @After
    public void tearDown() {
        objectMapper.clear();
    }

    public void testConvertSuccess() {
        Project project = new Project();
        project.setName("TestProject");
        project.setLanguage("en");
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        String result = transformer.convert(project);
        assertNotNull(result);
        String expected = "{\"name\":\"TestProject\",\"language\":\"en\"}";
        assertEquals(expected, result);
    }

    public void testConvertFail() {
        Project project = null;
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        try {
            transformer.convert(project);
            assert false;
        } catch (CrowdinApiException e) {
            // expected
        }
    }

    public class ProjectDeserializer extends JsonDeserializer<Project> {

        @Override
        public Project deserialize(JsonParser jsonParser, DeserializationFeature feature) throws IOException {
            return new Project();
        }
    }

    public class ExportOptionsDeserializer extends JsonDeserializer<ExportOptions> {

        @Override
        public ExportOptions deserialize(JsonParser jsonParser, DeserializationFeature feature) throws IOException {
            return new ExportOptions();
        }
    }

    public class FileInfoDeserializer extends JsonDeserializer<FileInfo> {

        @Override
        public FileInfo deserialize(JsonParser jsonParser, DeserializationFeature feature) throws IOException {
            return new FileInfo();
        }
    }

    public class ImportOptionsDeserializer extends JsonDeserializer<ImportOptions> {

        @Override
        public ImportOptions deserialize(JsonParser jsonParser, DeserializationFeature feature) throws IOException {
            return new ImportOptions();
        }
    }

    public class LanguageTranslationsDeserializer extends JsonDeserializer<LanguageTranslations> {

        @Override
        public LanguageTranslations deserialize(JsonParser jsonParser, DeserializationFeature feature) throws IOException {
            return new LanguageTranslations();
        }
    }
}
