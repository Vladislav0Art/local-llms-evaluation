import java.net.URI;
import java.net.URISyntaxException;

public class UrlBuilder {
    private String url;

    public UrlBuilder(String url) {
        this.url = url;
    }

    public void appendQueryParam(Connection.KeyVal kv, String paramSeparator) throws URISyntaxException {
        if (kv.getKey() != null && kv.getValue() != null) {
            URI uri = new URI(url);
            String queryParam = uri.getQuery();
            String[] params = queryParam.split("&");
            for (int i = 0; i < params.length; i++) {
                String[] keyValue = params[i].split("=");
                if (keyValue[0].equals(kv.getKey())) {
                    uri = new URI(uri.getScheme(), uri.getUserInfo(), uri.getHost(), uri.getPort(), uri.getPath(), uri.getQuery() + paramSeparator + kv.getKey() + "=" + kv.getValue(), uri.getFragment());
                }
            }
            url = uri.toString();
        } else if (kv.getKey() == null && kv.getValue() == null) {
            System.out.println(url);
        }
    }

    public static void main(String[] args) {
        String url = "https://example.com";
        UrlBuilder urlBuilder = new UrlBuilder(url);

        Connection.KeyVal kv1 = new Connection.KeyVal("key1", "value1");
        urlBuilder.appendQueryParam(kv1, "&");

        Connection.KeyVal kv2 = new Connection.KeyVal("key2", null);
        urlBuilder.appendQueryParam(kv2, "&");

        Connection.KeyVal kv3 = new Connection.KeyVal(null, "value3");
        urlBuilder.appendQueryParam(kv3, "&");

        Connection.KeyVal kv4 = new Connection.KeyVal(null, null);
        urlBuilder.appendQueryParam(kv4, "");
    }
}

public class Connection {
    public static class KeyVal {
        private String key;
        private Object value;

        public KeyVal(String key, Object value) {
            this.key = key;
            this.value = value;
        }

        public String getKey() {
            return key;
        }

        public Object getValue() {
            return value;
        }
    }
}
