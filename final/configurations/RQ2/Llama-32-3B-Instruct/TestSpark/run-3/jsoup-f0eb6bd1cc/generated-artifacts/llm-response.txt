        UrlBuilder urlBuilder = new UrlBuilder(new URL("https://www.example.com"));

        assertEquals("https://www.example.com?key=value1&value2", urlString);
    }
}

public class UrlBuilder {
    private URL url;

    public UrlBuilder(URL url) {
        this.url = url;
    }

    public URL build() {
        return new URL(url.toString());
    }

    public String appendKeyVal(Connection.KeyVal kv) {
        StringBuilder sb = new StringBuilder(url.toString());
        if (sb.charAt(sb.length() - 1) == '?') {
            sb.append('&');
        }
        sb.append("key=").append(kv.toString());
        return sb.toString();
    }
}
