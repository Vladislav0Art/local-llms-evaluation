package com.crowdin.client.core.http.impl.json;

import com.crowdin.client.core.http.JsonTransformer;
import com.crowdin.client.core.http.exceptions.CrowdinApiException;
import com.crowdin.client.projectsgroups.model.Project;
import com.crowdin.client.sourcefiles.model.ExportOptions;
import com.crowdin.client.sourcefiles.model.FileInfo;
import com.crowdin.client.sourcefiles.model.ImportOptions;
import com.crowdin.client.stringtranslations.model.LanguageTranslations;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;

import java.io.IOException;
import java.lang.reflect.Type;

import static org.hamcrest.MatcherAssert.assertThat;
import static org.hamcrest.Matchers.is;
import static org.junit.Assert.assertEquals;
import static org.junit.Assert.fail;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class JacksonJsonTransformerTest {

    @Mock
    private ObjectMapper objectMapper;

    private JacksonJsonTransformer jsonTransformer = new JacksonJsonTransformer();

    @Test
    public void parseProject() throws Exception {
        when(objectMapper.findAndRegisterModules()).thenReturn(new SimpleModule());

        Project project = new Project();
        String json = "{\"project_id\": \"123\", \"name\": \"test\"}";
        Project result = (Project) jsonTransformer.parse(json, Project.class);
        assertThat(result.getProjectId(), is("123"));
    }

    @Test
    public void parseExportOptions() throws Exception {
        when(objectMapper.findAndRegisterModules()).thenReturn(new SimpleModule());

        ExportOptions exportOptions = new ExportOptions();
        String json = "{\"file_type\": \"csv\"}";
        ExportOptions result = (ExportOptions) jsonTransformer.parse(json, ExportOptions.class);
        assertThat(result.getFileType(), is("csv"));
    }

    @Test
    public void convertLanguageTranslations() throws Exception {
        LanguageTranslations languageTranslations = new LanguageTranslations();
        String obj = "{\"lang\": \"en\", \"translations\": {\"hello\": \"world\"}}";
        String result = jsonTransformer.convert(languageTranslations);
        assertThat(result, is("{\"lang\":\"en\",\"translations\":{\"hello\":\"world\"}}"));
    }

    @Test
    public void convertFileInfo() throws Exception {
        FileInfo fileInfo = new FileInfo();
        String obj = "{\"file_id\": \"123\", \"name\": \"test.txt\"}";
        String result = jsonTransformer.convert(fileInfo);
        assertThat(result, is("{\"fileId\":\"123\",\"name\":\"test.txt\"}"));
    }

    @Test
    public void parseFile() throws Exception {
        FileInfo fileInfo = new FileInfo();
        when(objectMapper.findAndRegisterModules()).thenReturn(new SimpleModule());
        when(objectMapper.readValue("{\"file_id\": \"123\", \"name\": \"test.txt\"}", FileInfo.class)).thenReturn(fileInfo);
        FileInfo result = (FileInfo) jsonTransformer.parse("{\"file_id\": \"123\", \"name\": \"test.txt\"}", FileInfo.class);
        assertThat(result.getFileId(), is("123"));
    }

    @Test
    public void parseImportOptions() throws Exception {
        ImportOptions importOptions = new ImportOptions();
        String json = "{\"source_type\": \"github\"}";
        ImportOptions result = (ImportOptions) jsonTransformer.parse(json, ImportOptions.class);
        assertThat(result.getSourceType(), is("github"));
    }

    @Test
    public void parseCrowdinApiException() {
        when(objectMapper.findAndRegisterModules()).thenReturn(new SimpleModule());
        CrowdinApiException exception = new CrowdinApiException();
        String result = jsonTransformer.parse(exception, CrowdinApiException.class);
        assertThat(result, is(""));
    }

    @Test
    public void convertCrowdinApiException() {
        CrowdinApiException exception = new CrowdinApiException();
        String result = jsonTransformer.convert(exception);
        assertThat(result, is("CrowdinApiException"));
    }

    @Test
    public void parseHttpBadRequestException() {
        when(objectMapper.findAndRegisterModules()).thenReturn(new SimpleModule());
        HttpBadRequestException exception = new HttpBadRequestException();
        String result = jsonTransformer.parse(exception, HttpBadRequestException.class);
        assertThat(result, is(""));
    }

    @Test
    public void convertHttpBadRequestException() {
        HttpBadRequestException exception = new HttpBadRequestException();
        String result = jsonTransformer.convert(exception);
        assertThat(result, is("HttpBadRequestException"));
    }
