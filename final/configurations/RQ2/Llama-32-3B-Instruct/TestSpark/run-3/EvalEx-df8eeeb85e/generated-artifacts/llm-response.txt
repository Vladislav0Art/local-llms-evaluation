        String expressionString = "x";

    }

    @Test
    public void parseStringLiteralFromOperatorToken() throws ParseException {
        String expressionString = "+";
        List<Token> tokens = new ArrayList<>();
        tokens.add(new Operator(ADD));
        assertEquals(tokens, new Tokenizer(expressionString, new ExpressionConfiguration()).parseStringLiteral());
    }

    @Test
    public void parseWithInvalidExpressionString() {

        try {
            new Tokenizer(expressionString, new ExpressionConfiguration()).parse();

        } catch (ParseException e) {
            assertNotNull(e);
        }
    }

    private static class ExpressionConfiguration {}
    private static class Literal extends Token {}
