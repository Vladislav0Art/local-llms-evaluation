import org.jsoup.parser.XmlTreeBuilder;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;
import org.mockito.junit.MockitoJUnitRunner;

import static org.junit.Assert.*;

import java.io.IOException;
import java.io.Reader;
import java.util.Arrays;
import java.util.List;

@RunWith(MockitoJUnitRunner.class)
public class XmlTreeBuilderTest {

    @Mock
    private TreeBuilder treeBuilderMock;

    public XmlTreeBuilder testInstance() {
        return new XmlTreeBuilder(treeBuilderMock);
    }

    @Test
    public void parseStringBasicTest() throws IOException {
        String xml = "<html><body>Hello World!</body></html>";
        Document document = testInstance().parse(xml, "");
        assertEquals("html", document.documentElement.tagName());
    }

    @Test
    public void parseStringCommentTest() throws IOException {
        String xml = "<!-- This is a comment -->";
        Document document = testInstance().parse(xml, "");
        assertNull(document.body());
    }

    @Test
    public void parseStringDoctypeTest() throws IOException {
        String xml = "<?xml version=\"1.0\"?><!DOCTYPE html>";
        Document document = testInstance().parse(xml, "");
        assertEquals("html", document.documentElement.tagName());
    }

    @Test
    public void parseReaderBasicTest() throws IOException {
        Reader reader = new StringReader("<html><body>Hello World!</body></html>");
        Document document = testInstance().parse(reader, "");
        assertEquals("html", document.documentElement.tagName());
    }

    @Test
    public void parseReaderCommentTest() throws IOException {
        Reader reader = new StringReader("<!-- This is a comment -->");
        Document document = testInstance().parse(reader, "");
        assertNull(document.body());
    }

    @Test
    public void parseReaderDoctypeTest() throws IOException {
        Reader reader = new StringReader("<?xml version=\"1.0\"?><!DOCTYPE html>");
        Document document = testInstance().parse(reader, "");
        assertEquals("html", document.documentElement.tagName());
    }

    @Test
    public void initialiseParseReaderInputBasicTest() {
        Reader reader = new StringReader("<html><body>Hello World!</body></html>");
        Parser parser = Mockito.mock(Parser.class);
        XmlTreeBuilder treeBuilder = testInstance();
        treeBuilder.initialiseParse(reader, "", parser);
        assertFalse(treeBuilder.defaultSettings().useCdata());
    }

    @Test
    public void initialiseParseReaderInputNoUriBasicTest() {
        Reader reader = new StringReader("<html><body>Hello World!</body></html>");
        Parser parser = Mockito.mock(Parser.class);
        XmlTreeBuilder treeBuilder = testInstance();
        treeBuilder.initialiseParse(reader, "", parser);
        assertTrue(treeBuilder.defaultSettings().useCdata());
    }

    @Test
    public void processEndTagBasicTest() {
        Token token = new Token(Token.EndTag, "tag");
        boolean result = testInstance().process(token);
        assertTrue(result);
    }

    @Test
    public void processCharacterBasicTest() {
        Token token = new Token(Token.Character, 'a');
        boolean result = testInstance().process(token);
        assertFalse(result);
    }
}
