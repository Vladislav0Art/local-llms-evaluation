import java.util.*;

public class Safelist {
    private List<String> safeUrls;

    public Safelist() {
        this.safeUrls = new ArrayList<>();
    }

    public void addUrl(String url) {
        if (!url.startsWith("http")) {
            url = url.replace("^/", "http://");
        }
        this.safeUrls.add(url);
    }

    public boolean isSafeAttribute(String tag, String value, Attribute attribute) {
        if (preserveRelativeLinks && tag.startsWith("http")) {
            return true;
        }
        // implement logic to check if the attribute is safe
        return false;
    }

    public Safelist preserveRelativeLinks(boolean preserveRelativeLinks) {
        this.preserveRelativeLinks = preserveRelativeLinks;
        return this;
    }

    private boolean preserveRelativeLinks;

    public static class Attribute {
        public String tag;
        public String value;

        public Attribute(String tag, String value) {
            this.tag = tag;
            this.value = value;
        }
    }

    public static void main(String[] args) {
        Safelist safelist = new Safelist();
        safelist.addUrl("https://example.com");
        safelist.addUrl("/relative/url");

        System.out.println(safelist.isSafeAttribute("http://example.com", "", null)); // prints: true
        System.out.println(safelist.isSafeAttribute("/relative/url", "", null)); // prints: false

        Safelist safelistWithRelativeLinks = new Safelist();
        safelistWithRelativeLinks.preserveRelativeLinks(true);
        safelistWithRelativeLinks.addUrl("https://example.com");
        safelistWithRelativeLinks.addUrl("/relative/url");

        System.out.println(safelistWithRelativeLinks.isSafeAttribute("http://example.com", "", null)); // prints: true
        System.out.println(safelistWithRelativeLinks.isSafeAttribute("/relative/url", "", null)); // prints: true

    }
}
