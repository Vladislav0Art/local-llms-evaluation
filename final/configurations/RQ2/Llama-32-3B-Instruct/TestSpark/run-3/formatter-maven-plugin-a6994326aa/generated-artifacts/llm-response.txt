public class CssFormatter {
    private Map<String, String> options;
    private ConfigurationSource cfg;

    public void init(Map<String, String> options, ConfigurationSource cfg) {
        this.options = options;
        this.cfg = cfg;
    }

    public String doFormat(String css, LineEnding ending) throws IOException {
        String formattedCss = "";
        for (String line : css.split("\n")) {
            if (options.containsKey("line_ending") && options.get("line_ending").equals("both")) {
                formattedCss += line + ending.getSymbol();
            } else {
                formattedCss += line;
            }
        }
        return formattedCss;
    }

    public boolean isInitialized() {
        return true;
    }
}
