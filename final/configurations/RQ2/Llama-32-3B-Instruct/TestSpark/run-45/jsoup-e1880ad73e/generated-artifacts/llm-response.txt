import org.junit.Test;
import static org.junit.Assert.assertEquals;

public class UrlBuilderTest {

    @Test
    public void testDefaultUrl() {
        String result = ((UrlBuilder) new UrlBuilder(new URL("https://www.example.com"))).getUrl();
        assertEquals("https://www.example.com", result);
    }

    @Test
    public void testAddQueryParams() {
        String result = ((UrlBuilder) new UrlBuilder(new URL("https://www.example.com"), "key=value").getUrl());
        assertEquals("https://www.example.com?key=value", result);
    }

    @Test
    public void testAddMultipleQueryParams() {
        String result = ((UrlBuilder) new UrlBuilder(new URL("https://www.example.com"), "key1=value1", "key2=value2").getUrl());
        assertEquals("https://www.example.com?key1=value1&key2=value2", result);
    }
}

class UrlBuilder {

    private URL url;
    private String... queryParams;

    public UrlBuilder(URL url) {
        this.url = url;
    }

    public UrlBuilder(URL url, String... queryParams) {
        this.url = url;
        this.queryParams = queryParams;
    }

    public String getUrl() {
        StringBuilder sb = new StringBuilder(url.toString());
        if (queryParams != null) {
            for (String param : queryParams) {
                sb.append("&").append(param);
            }
        }
        return sb.toString();
    }
}

class Connection {

    public static class KeyVal {
        private final String key;
        private final String value;

        public KeyVal(String key, String value) {
            this.key = key;
            this.value = value;
        }

        public static KeyVal of(String key, String value) {
            return new KeyVal(key, value);
        }
    }
}
