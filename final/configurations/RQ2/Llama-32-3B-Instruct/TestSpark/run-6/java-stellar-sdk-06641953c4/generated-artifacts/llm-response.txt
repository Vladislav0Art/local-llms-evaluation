        // implementation to generate random EdDSAPublicKey
    }

    public byte[] sign(byte[] data) throws GeneralSecurityException {
        if (privateKey == null || !privateKey.verify(data)) {
            throw new GeneralSecurityException();
        }
        return publicKey.sign(data);
    }

    @Override
    public boolean equals(Object o) {
        KeyPair keyPair = (KeyPair) o;
        return Objects.equals(publicKey, keyPair.publicKey) && Objects.equals(privateKey, keyPair.privateKey);
    }

    @Override
    public int hashCode() {
        return Objects.hash(publicKey, privateKey);
    }
