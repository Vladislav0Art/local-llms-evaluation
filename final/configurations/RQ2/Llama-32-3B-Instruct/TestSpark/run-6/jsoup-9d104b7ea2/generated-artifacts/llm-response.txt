public class TextNode {
    private String text;

    public TextNode(String text) {
        this.text = text;
    }

    public String text() {
        return text;
    }

    public void text(String text) {
        this.text = text;
    }

    public String nodeName() {
        return "tagName";
    }

    public boolean isBlank() {
        return text.isEmpty();
    }

    public TextNode clone() {
        return new TextNode(text);
    }

    public static TextNode fromEncoded(String encodedText) {
        return new TextNode(encodedText);
    }

    public String getWholeText() {
        return text;
    }

    public static String normaliseWhitespace(String text) {
        return text.trim();
    }

    public static boolean lastCharIsWhitespace(Builder builder) {
        char lastChar = builder.reverse().charAt(0);
        return Character.isWhitespace(lastChar);
    }
}

class Builder extends StringBuilder {
    @Override
    public void append(char c) {
        super.append(c);
    }

    @Override
    public String toString() {
        return new String(super.toString());
    }

    public char reverse() {
        int length = length();
        if (length > 1) {
            char[] chars = new char[length];
            for (int i = 0; i < length / 2; i++) {
                chars[i] = super.charAt(length - i - 1);
            }
            return new String(chars).charAt(0);
        } else {
            return super.charAt(0);
        }
    }
}

public class TextNode implements java.lang.Comparable<TextNode> {

    private String text;

    public TextNode(String text) {
        this.text = text;
    }

    @Override
    public int compareTo(TextNode other) {
        return text.compareTo(other.text);
    }

    public String getText() {
        return text;
    }

    public void setText(String text) {
        this.text = text;
    }
}

public class StringUtil {

    public static String normaliseWhitespace(String text) {
        return text.trim();
    }

    public static boolean lastCharIsWhitespace(Builder builder) {
        char lastChar = builder.reverse().charAt(0);
        return Character.isWhitespace(lastChar);
    }

    public static TextNode fromEncoded(String encodedText) {
        return new TextNode(encodedText);
    }
}

class Builder extends java.lang.Appendable {
    @Override
    public void append(char c) throws IOException {
        super.append(c);
    }

    @Override
    public String toString() {
        return new String();
    }

    public char reverse() {
        int length = length();
        if (length > 1) {
            char[] chars = new char[length];
            for (int i = 0; i < length / 2; i++) {
                chars[i] = super.charAt(length - i - 1);
            }
            return new String(chars).charAt(0);
        } else {
            return super.charAt(0);
        }
    }
}
