```java
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mockito;

@RunWith(MockitoJUnitRunner.class)
public class UrlBuilderTest {

    @Test
    public void constructorWithUrl_ReturnsInstance() {
        URL url = new URL("http://example.com");
        UrlBuilder builder = new UrlBuilder(url);
        assert builder.build().equals(url);
    }

    @Test
    public void build_WithoutParameters_ReturnsUrl() throws MalformedURLException {
        URL url = new UrlBuilder(null).build();
        assert url.getProtocol().equals("http");
        assert url.getHost().endsWith("/");
        assert url.getPath().isEmpty();
    }

    @Test
    public void build_WithSchemeAndHost_ReturnsUrl() throws MalformedURLException {
        URL url = new UrlBuilder(null);
        url = url.scheme("https").host("example.com").build();
        assert url.getProtocol().equals("https");
        assert url.getHost().endsWith("/");
        assert url.getPath().isEmpty();
    }

    @Test
    public void build_WithPath_ReturnsUrl() throws MalformedURLException {
        URL url = new UrlBuilder(null);
        url = url.path("/path").build();
        assert url.getProtocol().equals("http");
        assert url.getHost().endsWith("/");
        assert url.getPath().equals("/path");
    }

    @Test
    public void appendKeyVal_WithKeyAndValue_ReturnsUrl() throws UnsupportedEncodingException {
        URL url = new UrlBuilder(null);
        url = url.appendKeyVal(Connection.KeyVal.of("key", "value"));
        assert url.getQuery().contains("key=value");
    }

    @Test
    public void appendKeyVal_WithoutKeyOrValue_ThrowsUnsupportedEncodingException() {
        try {
            new UrlBuilder(null).appendKeyVal(Connection.KeyVal.of(1, 2));
            assert false;
        } catch (UnsupportedEncodingException e) {
            // Expected
        }
    }

    @Test
    public void encode_WithQuery_ReturnsUrl() throws UnsupportedEncodingException {
        URL url = new UrlBuilder(null);
        String query = "key=value";
        url = url.appendKeyVal(Connection.KeyVal.of("key", query));
        assert url.getQuery().equals(query);
        assert url.build().getProtocol().equals(url.getScheme());
    }

    @Test
    public void encode_WithoutQuery_ReturnsUrl() throws UnsupportedEncodingException {
        URL url = new UrlBuilder(null);
        String path = "path";
        url = url.path(path).build();
        // No encoding needed for path
    }
}
```

```java
public class Connection {
    private String host;
    private int port;

    public Connection(String host, int port) {
        this.host = host;
        this.port = port;
    }

    public void URL(String url) {
        System.out.println("URL: " + url);
    }

    public void KeyVal(KeyVal keyVal) {
        System.out.println("Key-Val: " + keyVal);
    }

    public static class KeyVal {
        private String key;
        private String value;

        public KeyVal(String key, String value) {
            this.key = key;
            this.value = value;
        }

        @Override
        public String toString() {
            return "{" +
                    "\"key\": \"" + key + "\", " +
                    "\"value\": \"" + value + "\"" +
                    "}";
        }
    }

    public static class URLUtil {
        public static void appendKeyVal(Connection url, KeyVal keyVal) {
            System.out.println("URL: " + url.host + ":" + url.port);
            System.out.println("Key-Val: " + keyVal.key + "=" + keyVal.value);
        }

        public static String encode_WithQuery(String query) {
            return query;
        }
    }

    public static class URL {
        private String host;
        private int port;

        public URL(String host, int port) {
            this.host = host;
            this.port = port;
        }

        @Override
        public String toString() {
            return "URL{" +
                    "host='" + host + '\'' +
                    ", port=" + port +
                    '}';
        }
    }

    public static void main(String[] args) {
        Connection url = new Connection("example.com", 8080);
        KeyVal keyVal = new Connection.KeyVal("key", "value");

        URLUtil.appendKeyVal(url, keyVal);

        System.out.println(URL.encode_WithQuery("query"));
    }
}
```
