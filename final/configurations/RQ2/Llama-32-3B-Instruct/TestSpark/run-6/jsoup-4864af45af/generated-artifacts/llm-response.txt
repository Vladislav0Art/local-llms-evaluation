package org.jsoup.safety;

import org.jsoup.helper.Validate;
import org.jsoup.nodes.Attribute;
import org.jsoup.nodes.Attributes;
import org.jsoup.nodes.Element;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

@RunWith(MockitoJUnitRunner.class)
public class SafelistTest {

    @Mock
    private Set<String> safeTags;

    @Mock
    private Set<String> safeAttributes;

    @Test
    public void noneSafeTagTest() {
        Safelist safelist = Safelist.none();
        org.junit.Assert.assertTrue(safelist.isSafeTag("script"));
    }

    @Test
    public void noneNoSafeTagTest() {
        Safelist safelist = Safelist.none();
        org.junit.Assert.assertFalse(safelist.isSafeTag("test"));
    }

    @Test
    public void simpleTextAddTagsTest() {
        Safelist safelist = Safelist.simpleText().addTags("p", "span");
        Mockito.verify(safeTags).add("p");
        Mockito.verify(safeTags).add("span");
    }

    @Test
    public void simpleTextRemoveTagsTest() {
        Safelist safelist = Safelist.simpleText().addTags("p", "span");
        safelist = safelist.removeTags("p");
        Mockito.verify(safTags, Mockito.never()).add("p");
        Mockito.verify(safTags).add("span");
    }

    @Test
    public void basicAddAttributesTest() {
        Safelist safelist = Safelist.basic().addAttributes("img", "src", "image.jpg");
        org.junit.Assert.assertTrue(safelist.getEnforcedAttributes("img").contains("src"));
    }

    @Test
    public void basicRemoveAttributesTest() {
        Safelist safelist = Safelist.basic().addAttributes("img", "src", "image.jpg");
        safelist = safelist.removeAttributes("img", "src");
        org.junit.Assert.assertFalse(safelist.getEnforcedAttributes("img").contains("src"));
    }

    @Test
    public void basicAddEnforcedAttributeTest() {
        Safelist safelist = Safelist.basic().addEnforcedAttribute("img", "alt", "image.jpg");
        Mockito.verify(safeTags).add("img");
        org.junit.Assert.assertTrue(safelist.getEnforcedAttributes("img").contains("alt"));
    }

    @Test
    public void basicRemoveEnforcedAttributeTest() {
        Safelist safelist = Safelist.basic().addEnforcedAttribute("img", "alt", "image.jpg");
        safelist = safelist.removeEnforcedAttribute("img", "alt");
        org.junit.Assert.assertFalse(safelist.getEnforcedAttributes("img").contains("alt"));
    }

    @Test
    public void relaxedAddProtocolsTest() {
        Safelist safelist = Safelist.relaxed().addProtocols("a", "href", "http://example.com");
        Mockito.verify(safeTags).add("a");
        org.junit.Assert.assertTrue(safelist.getEnforcedAttributes("a").contains("href"));
    }

    @Test
    public void relaxedRemoveProtocolsTest() {
        Safelist safelist = Safelist.relaxed().addProtocols("a", "href", "http://example.com");
        safelist = safelist.removeProtocols("a", "href", "https://example.com");
        org.junit.Assert.assertFalse(safelist.getEnforcedAttributes("a").contains("href"));
    }
}
