import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;

import java.util.ArrayList;
import java.util.List;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertFalse;
import static org.junit.Assert.assertSame;
import static org.junit.Assert.assertTrue;
import static org.jsoup.helper.Validate.notNull;
import static org.jsoup.internal.Normalizer.lowerCase;
import static org.jsoup.nodes.Attributes.*;

@RunWith(MockitoJUnitRunner.class)
public class AttributesTest {

    @Mock
    private Attribute attribute;

    @Mock
    private Document document;

    @Test
    public void getReturnsNullForNonExistentKey() {
        assertEquals(null, get("non-existent-key"));
    }

    @Test
    public void getIgnoreCaseReturnsNullForNonExistentKey() {
        assertEquals(null, getIgnoreCase("non-existent-key"));
    }

    @Test
    public void getUserDataReturnsNullForNonExistentKey() {
        assertEquals(null, getUserData("non-existent-key"));
    }

    @Test
    public void addAddsNewAttribute() {
        Attributes attributes = new Attributes();
        String result = add("new-key", "new-value");
        assertSame(result, "new-value");
        assertEquals(1, attributes.size());
    }

    @Test
    public void putUpdatesExistingAttribute() {
        Attributes attributes = new Attributes();
        attributes.put("existing-key", "existing-value");
        String result = put("existing-key", "updated-value");
        assertSame(result, "updated-value");
        assertEquals(1, attributes.size());
    }

    @Test
    public void removeRemovesKeyFromAttributes() {
        Attributes attributes = new Attributes();
        attributes.put("key", "value");
        attributes.remove("key");
        assertFalse(hasKey("key"));
    }

    @Test
    public void hasKeyReturnsTrueForExistingKey() {
        Attributes attributes = new Attributes();
        attributes.put("key", "value");
        assertTrue(hasKey("key"));
    }

    @Test
    public void hasDeclaredValueForKeyReturnsFalseForNonExistentKey() {
        Attributes attributes = new Attributes();
        assertFalse(hasDeclaredValueForKey("non-existent-key"));
    }

    @Test
    public void sizeReturnsCorrectSize() {
        Attributes attributes = new Attributes();
        assertEquals(0, attributes.size());
        attribute.setKey("key");
        assertEquals(1, attributes.size());
    }

    @Test
    public void isEmptyReturnsFalseForNonEmptyAttributes() {
        Attributes attributes = new Attributes();
        assertFalse(isEmpty());
        attributes.clear();
        assertTrue(isEmpty());
    }

    @Test
    public void addAllAddsIncomingKeysAndValues() {
        Attributes incoming = new Attributes();
        incoming.put("key1", "value1");
        incoming.put("key2", "value2");
        Attributes attributes = new Attributes();
        attributes.addAll(incoming);
        assertEquals(2, attributes.size());
    }

    @Test
    public void iteratorReturnsAttributesAsIterator() {
        Attributes attributes = new Attributes();
        attribute.setKey("key");
        attributes.put("key2", "value2");
        Iterator<Attribute> iterator = attributes.iterator();
        assertSame(attribute, iterator.next());
    }

    @Test
    public void asListReturnsAttributesAsStringList() {
        Attributes attributes = new Attributes();
        attribute.setKey("key");
        attributes.put("key2", "value2");
        List<Attribute> list = attributes.asList();
        assertEquals(1, list.size());
    }

    @Test
    public void datasetReturnsDatasetAsMap() {
        Attributes attributes = new Attributes();
        attribute.setKey("key");
        attributes.put("key2", "value2");
        Map<String, String> map = attributes.dataset();
        assertEquals(1, map.size());
    }
}

    public void testJsoupPut() {
