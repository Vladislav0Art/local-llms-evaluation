public class AppVersion {

    private String version;
    private String packageName;
    private String buildNumber;
    private String commit;

    public static Pattern getAppVersionPattern() {
        return Pattern.compile("([0-9\\.]+)-ami-([A-Za-z0-9]+)");
    }

    public String getPackageName() {
        return packageName;
    }

    public String getVersion() {
        return version;
    }

    public String getBuild() {
        return buildNumber;
    }

    public String getCommit() {
        return commit;
    }

    public static AppVersion parseName(String amiName) {
        Matcher matcher = getAppVersionPattern().matcher(amiName);
        if (matcher.find()) {
            return new AppVersion(matcher.group(1), matcher.group(2));
        } else {
            throw new ParseException("Invalid format", 0, amiName);
        }
    }

    private AppVersion(String version, String packageName) {
        this.version = version;
        this.packageName = packageName;
    }

    public int compareTo(AppVersion other) {
        if (this.version.compareTo(other.version) != 0) {
            return this.version.compareTo(other.version);
        } else if (this.buildNumber.compareTo(other.buildNumber) != 0) {
            return this.buildNumber.compareTo(other.buildNumber);
        } else {
            return this.packageName.compareTo(other.packageName);
        }
    }

    @Override
    public String toString() {
        return "AppVersion{" +
                "version='" + version + '\'' +
                ", packageName='" + packageName + '\'' +
                ", buildNumber='" + buildNumber + '\'' +
                ", commit='" + commit + '\'' +
                '}';
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        AppVersion that = (AppVersion) o;
        return Objects.equals(version, that.version) &&
                Objects.equals(packageName, that.packageName);
    }

    @Override
    public int hashCode() {
        return Objects.hash(version, packageName);
    }
}

public class ParseException extends RuntimeException {

    public ParseException(String message, int line, String sourceCode) {
        super(message + " at line " + line + " in: " + sourceCode);
    }
}

public class AppVersion {

    private String version;
    private String packageName;

    public static Pattern getAppVersionPattern() {
        return Pattern.compile("([0-9\\.]+)-ami-(\\w+)");
    }

    public static boolean checkAppVersion(String amiName) {
        Matcher matcher = getAppVersionPattern().matcher(amiName);
        if (matcher.find()) {
            return true;
        } else {
            throw new ParseException("Invalid format", 0, amiName);
        }
    }

    public AppVersion(String version, String packageName) {
        this.version = version;
        this.packageName = packageName;
    }

    public static boolean compareVersions(AppVersion app1, AppVersion app2) {
        int v1 = Integer.parseInt(app1.getBuild().replace("-", ""));
        int v2 = Integer.parseInt(app2.getBuild().replace("-", ""));

        if (v1 > v2) {
            return true;
        } else if (v1 < v2) {
            return false;
        } else {
            return app1.getPackageName().compareTo(app2.getPackageName()) <= 0;
        }
    }

    public boolean isLessThan(AppVersion other) {
        return compareVersions(this, other);
    }

    @Override
    public String toString() {
        return "AppVersion{" +
                "version='" + version + '\'' +
                ", packageName='" + packageName + '\'' +
                '}';
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        AppVersion that = (AppVersion) o;
        return Objects.equals(version, that.version) &&
                Objects.equals(packageName, that.packageName);
    }

    @Override
    public int hashCode() {
        return Objects.hash(version, packageName);
    }
}

public class ParseException extends Exception {

    public ParseException(String message) {
        super(message);
    }
}
