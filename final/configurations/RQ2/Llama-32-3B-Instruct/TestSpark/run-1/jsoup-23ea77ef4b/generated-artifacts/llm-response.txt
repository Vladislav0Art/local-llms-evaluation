import javax.xml.parsers.DocumentBuilderFactory;
import javax.xml.parsers.ParserConfigurationException;

import org.w3c.dom.Document;
import org.w3c.dom.Element;
import org.w3c.dom.NodeList;
import org.xml.sax.SAXException;

import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;

/**
 * Represents a document builder.
 */
public class DocumentBuilder {

    private List<Element> elements = new ArrayList<>();

    /**
     * Adds an element to the current document.
     *
     * @param tag  The XML tag of the element.
     * @param data The content of the element.
     */
    public void addElement(String tag, String data) {
        Element element = createElement(tag, data);
        elements.add(element);
    }

    /**
     * Creates a new element with the given tag and content.
     *
     * @param tag  The XML tag of the element.
     * @param data The content of the element.
     * @return The created element.
     */
    private Element createElement(String tag, String data) {
        return new Element(tag, data);
    }

    /**
     * Builds and returns the constructed document.
     *
     * @return The constructed document.
     */
    public Document buildDocument() {
        Document document = null;
        try {
            // Create a new document
            document = DocumentBuilderFactory.newInstance().newDocumentBuilder()
                    .newDocument();

            // Add elements to the document
            for (Element element : elements) {
                document.appendChild(element.getDomElement());
            }
        } catch (ParserConfigurationException | SAXException e) {
            System.out.println("Error building document: " + e.getMessage());
        }
        return document;
    }

    /**
     * Represents a single XML element.
     */
    public class Element {

        private String tag;
        private String data;

        /**
         * Creates a new element with the given tag and content.
         *
         * @param tag  The XML tag of the element.
         * @param data The content of the element.
         */
        public Element(String tag, String data) {
            this.tag = tag;
            this.data = data;
        }

        /**
         * Returns the XML tag of the element.
         *
         * @return The XML tag.
         */
        public String getTag() {
            return tag;
        }

        /**
         * Returns the content of the element.
         *
         * @return The content of the element.
         */
        public String getData() {
            return data;
        }

        /**
         * Returns the DOM element for this element.
         *
         * @return The DOM element.
         */
        private Element getDomElement() {
            // Create a new DOM element
            org.w3c.dom.Element domElement = documentBuilderFactory.newDocumentBuilder().createElement(tag);

            // Set the content of the element
            domElement.setTextContent(data);

            return domElement;
        }
    }

    public static void main(String[] args) {
        DocumentBuilder builder = new DocumentBuilder();
        Element root = builder.addElement("root", "This is the root element.");
        Element child1 = builder.addElement("child1", "This is a child element.");
        Element child2 = builder.addElement("child2", "This is another child element.");

}

The `Element` class represents a single XML element and provides methods to get its tag and content.
