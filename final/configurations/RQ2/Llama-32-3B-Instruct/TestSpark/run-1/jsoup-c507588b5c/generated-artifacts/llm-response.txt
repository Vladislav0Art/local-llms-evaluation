import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertFalse;
import static org.junit.Assert.assertTrue;
import static org.jsoup.helper.Validate.isNotBlank;
import static org.jsoup.helper.Validate.notBlank;
import static org.jsoup.helper.Validate.nullsAreEqual;
import static org.jsoup.nodes.TextNode.createFromEncoded;
import static org.jsoup.nodes.TextNode.createText;

@RunWith(MockitoJUnitRunner.class)
public class TextNodeTest {

    @Mock
    private Appendable accum;

    @Test
    public void clone() {
        TextNode textNode = createText("Hello, World!");
        TextNode clonedTextNode = textNode.clone();
        assertEquals(textNode, clonedTextNode);
    }

    @Test
    public void clone_SameTextSameTextNode() {
        TextNode textNode = createText("Hello, World!");
        TextNode clonedTextNode = textNode.clone();
        assertTrue(isNotBlank(clonedTextNode.text()));
        assertTrue(toStringEquals(textNode, clonedTextNode));
    }

    @Test
    public void outerHtmlHead_MultiLineText() throws IOException {
        StringBuilder expected = new StringBuilder("<p>Hello,<br>World!</p>");
        TextNode textNode = createText("Hello,\n World!");
        textNode.outerHtmlHead(accum, 1, Document.OutputSettings.build());
        assertEquals(expected.toString(), accum.toString());
    }

    @Test
    public void outerHtmlTail_MultiLineText() {
        StringBuilder expected = new StringBuilder("<br>");
        TextNode textNode = createText("Hello,\n World!");
        textNode.outerHtmlTail(accum, 1, Document.OutputSettings.build());
        assertEquals(expected.toString(), accum.toString());
    }

    @Test
    public void outerHtmlHead_SingleLineText() throws IOException {
        StringBuilder expected = new StringBuilder("<p>Hello!</p>");
        TextNode textNode = createText("Hello!");
        textNode.outerHtmlHead(accum, 1, Document.OutputSettings.build());
        assertEquals(expected.toString(), accum.toString());
    }

    @Test
    public void outerHtmlTail_SingleLineText() {
        StringBuilder expected = new StringBuilder("");
        TextNode textNode = createText("Hello!");
        textNode.outerHtmlTail(accum, 1, Document.OutputSettings.build());
        assertEquals(expected.toString(), accum.toString());
    }

    @Test
    public void normaliseWhitespace() {
        assertTrue(TextNode.normaliseWhitespace("   Hello World "));
    }

    @Test
    public void stripLeadingWhitespace() {
        assertFalse(TextNode.stripLeadingWhitespace("   Hello World"));
    }

    @Test
    public void lastCharIsWhitespace_WhitespaceCharacter() {
        assertTrue(TextNode.lastCharIsWhitespace(new StringBuilder(" \n")));
    }

    @Test
    public void lastCharIsWhitespace_NoWhitespaceCharacters() {
        assertFalse(TextNode.lastCharIsWhitespace(new StringBuilder("HelloWorld")));
    }
}

import org.jsoup.nodes.Document;
import org.jsoup.nodes.TextNode;
import org.junit.Test;

public class GeneratedTest {
    @Test
    public void testOuterHtmlHead_MultiLineText() throws IOException {
        Document document = new Document();

        TextNode textNode1 = TextNode.createText("Hello,\n World!");
        textNode1.outerHtmlHead(document, 1, Document.OutputSettings.build());
        assertTrue(TextNode.normaliseWhitespace(textNode1.outerHtml()));
    }

    @Test
