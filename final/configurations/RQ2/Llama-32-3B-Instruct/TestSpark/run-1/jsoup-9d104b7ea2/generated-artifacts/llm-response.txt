        assertNotNull(node);
        assertEquals(encodedText, node.text());

    static {
        System.setOut(new PrintStream(new ByteArrayOutputStream()));
        Document document = new Document();

        document.appendChild(TextNode.createFromEncoded("<br>"));
        String output = document.outerHtmlHead(null, 0, null);
        assertEquals("<p>Hello World</p>", output.trim());
    }

    public void splitText(int position) {
