public class W3CDom {
    public static Node contextNode(Document document) {
        return document.getDocumentElement();
    }

    public static String asString(Node node) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < node.getNodeType(); i++) {
            if (node.getNodeType() == Node.ELEMENT_NODE) {
                sb.append("<").append(node.getNodeName()).append(">");
            }
            if (node.getNodeType() == Node.TEXT_NODE) {
                sb.append(node.getNodeValue());
            }
        }
        return sb.toString();
    }

    public static NodeList sourceNodes(Node node) {
        NodeList nodes = new NodeList();
        for (int i = 0; i < node.getNodeType(); i++) {
            if (node.getNodeType() == Node.ELEMENT_NODE) {
                ((Element) node).appendChild(document.createDocumentFragment());
                for (int j = 0; j < ((Element) node).getChildNodes().getLength(); j++) {
                    NodeList nodesList = new NodeList();
                    nodesList.add(((Element) node).getChildNodes().item(j));
                    nodes.addAll(nodesList);
                }
            }
        }
        return nodes;
    }

    public static Document parse(String html) throws Exception {
        DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
        DocumentBuilder builder = factory.newDocumentBuilder();
        ByteArrayInputStream input = new ByteArrayInputStream(html.getBytes());
        return builder.parse(input);
    }
}

public class W3CDomTest {

        Node node = W3CDom.contextNode(document);
        assert node.getNodeName().equals("root");
    }

    @Test
    public void testAsString() {
        Document document = DocumentBuilderFactory.newInstance().newDocumentBuilder().parse("<root><child></child></root>");
        Element element = (Element) W3CDom.contextNode(document);
        String string = W3CDom.asString(element);
        assert string.startsWith("<root><child>");
    }

    @Test
    public void testSourceNodes() {
        Document document = DocumentBuilderFactory.newInstance().newDocumentBuilder().parse("<root><child></child></root>");
        NodeList nodes = W3CDom.sourceNodes(W3CDom.contextNode(document));

        assert nodes.item(0).getNodeName().equals("child");
    }

    @Test
    public void testParse() throws Exception {

        Document document = W3CDom.parse(html);
        assert document.getDocumentElement().getNodeName().equals("root");
