import org.jsoup.nodes.Comment;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.LeafNode;
import org.jsoup.nodes.XmlDeclaration;
import org.jsoup.parser.ParseSettings;
import org.junit.Test;
import static org.junit.Assert.*;

import java.io.ByteArrayOutputStream;
import java.io.IOException;

public class CommentTest {

    @Test
    public void newCommentCreatesNewLeafNode() {
        Comment comment = new Comment("data");
        assertFalse(comment instanceof LeafNode);
    }

    @Test
    public void newNodeNameIsComment() {
        Comment comment = new Comment("data");
        assertEquals("comment", comment.nodeName());
    }

    @Test
    public void dataShouldNotBeEmpty() {
        Comment comment = new Comment("");
        assertTrue(comment.getData().isEmpty());
    }

    @Test
    public void cloneReturnsNewInstance() throws Exception {
        Comment comment = new Comment("data");
        Comment clonedComment = comment.clone();
        assertNotNull(clonedComment);
        assertEquals(comment, clonedComment);
    }

    @Test
    public void isXmlDeclarationAlwaysFalse() {
        Comment comment = new Comment("data");
        assertFalse(comment.isXmlDeclaration());
    }

    @Test
    public void asXmlDeclarationReturnsNotNullForXmlComments() throws Exception {
        ParseSettings settings = new ParseSettings();
        settings.setXmlDeclaration(true);
        Document doc = Document.parse("<comment>data</comment>", settings);
        Comment comment = doc.body().firstChild(Comment.class);
        assertNotNull(comment.asXmlDeclaration());
    }

    @Test
    public void toStringIncludesData() {
        Comment comment = new Comment("data");
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        Appendable accum = out;
        comment.outerHtmlHead(accum, 0, null);
        assertTrue(out.toString().contains("data"));
    }
}

public class MockParser implements Parser {

    }

    }
}

public class MockDocument extends LeafNode {
    private Comment comment;

        return comment;
    }
}
