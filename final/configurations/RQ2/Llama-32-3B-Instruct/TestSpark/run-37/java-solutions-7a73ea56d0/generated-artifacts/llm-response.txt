public class OnlineStockSpan {
    private int[] stack;

    public int[] next(int price) {
        int n = stack.length;
        if (n == 0 || stack[n - 1] <= price) {
            stack = append(stack, price);
        } else {
            removeLast(stack);
            stack = append(stack, price);
        }
        return stack;
    }

    private int[] append(int[] arr, int v) {
        int[] result = new int[arr.length + 1];
        arr.CopyTo(result, 0);
        result[result.length - 1] = v;
        return result;
    }

    private void removeLast(int[] arr) {
        arr = arr.clone();
        arr(arr.length - 1) = Integer.MIN_VALUE;
    }

    public int[][] calculateSpans() {
        Stack<Object> stack = new Stack<>();
        for (int i=0; i<prices.length; i++) {
            while (!stack.isEmpty() && (int)stack.peek() <= prices[i]) {
                stack.pop();
            }
            if (stack.isEmpty()) {
                result[i] = 1;
            } else {
                int span = i - ((Integer)stack.peek());
                result[i] += span;
            }
            stack.push(i);
        }
        return result;
    }

    private Stack<Object> calculateSpans(int[] prices) {
        Stack<Object> stack = new Stack<>();
        for (int i=0; i<prices.length; i++) {
            while (!stack.isEmpty() && (int)stack.peek() <= prices[i]) {
                stack.pop();
            }
            if (stack.isEmpty()) {
                result[i] = 1;
            } else {
                int span = i - ((Integer)stack.peek());
                result[i] += span;
            }
            stack.push(i);
        }
        return stack;
    }

    private Stack<Object> pushMock(Stack<Object> stack, int[] mockStackCount) {
        Mockito.mockStatic(Stack.class);
        Mockito.when(Stack.size()).thenReturn(mockStackCount[0]);
        Mockito.when(Stack.peek()).thenReturn(mockStackCount[1]);
        return stack;
    }
}

public class OnlineStockSpan {
    private Stack<Object> stack;

    public int[] next(int price) {
        int n = stack.size();
        if (n == 0 || (int)stack.peek() <= price) {
            stack.push(price);
        } else {
            stack.pop();
            stack.push(price);
        }
        return new int[]{(int)stack.peek()};
    }

    public static class Stack extends java.util.Stack<Object> {
        private Object topElement;

        public void push(Object element) {
            super.push(element);
            topElement = element;
        }

        public Object peek() {
            return topElement;
        }
    }
}
