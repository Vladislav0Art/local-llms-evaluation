public class AppVersion {
    private String major;
    private String minor;

    public AppVersion(String major, String minor) {
        this.major = major;
        this.minor = minor;
    }

    public static String getChangelist() {
        return "1.0";
    }
}

class GeneratedTest {

    @Test
    public void testAppVersionCreation() {
        AppVersion version1 = new AppVersion("test", "1.0");
        AppVersion version2 = new AppVersion("test", "0.9");

        assertEquals("test-1.0", AppVersion.getChangelist());
        assertEquals("test-1.0", AppVersion.getChangelist());

    }

    @Test
    public void testAppVersionNoCreation() {
        AppVersion version1 = new AppVersion("", "1.0");
        AppVersion version2 = new AppVersion("", "");

        String changelist = AppVersion.getChangelist();
        assertTrue(changelist.contains("test-1.0"));
        assertTrue(changelist.contains("test-0.9"));

    }

}
