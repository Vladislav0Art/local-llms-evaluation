import org.junit.Test;
import static org.junit.Assert.*;

public class Main {

    @Test
    public void testGetScheme() {
        Connection connection = new Connection("http://example.com");
        assertEquals("http", connection.getScheme());
    }

    @Test
    public void testGetHost() {
        Connection connection = new Connection("http://example.com");
        assertEquals("example.com", connection.getHost());
    }

    @Test
    public void testGetPort() {
        Connection connection = new Connection("http://localhost:8080");
        assertEquals(8080, connection.getPort());
    }
}

class Connection {
    private String scheme;
    private String host;
    private int port;

    public Connection(String url) {
        String[] parts = url.split("/");
        this.scheme = parts[0].substring(0, 4);
        this.host = parts[1];
        parts = host.split(":");
        if (parts.length > 1) {
            this.port = Integer.parseInt(parts[1]);
        } else {
            this.port = 80;
        }
    }

    public String getScheme() {
        return scheme;
    }

    public String getHost() {
        return host;
    }

    public int getPort() {
        return port;
    }
}

public class Connection {

    private String host;
    private int port;

    public Connection(String url) {
        if (url.startsWith("http")) {
            this.scheme = url.substring(0, 4);
            String[] parts = url.substring(4).split("/");
            this.host = parts[0];
            parts = host.split(":");
            if (parts.length > 1) {
                this.port = Integer.parseInt(parts[1]);
            } else {
                this.port = 80;
            }
        } else {
            throw new IllegalArgumentException("Invalid URL scheme");
        }
    }

    public String getScheme() {
        return scheme;
    }

    public String getHost() {

        return port;
    }
}

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.junit.Test;

public class Connection {
    private String scheme;
    private String host;
    private int port;

    public Connection(String url) throws Exception {
        Document doc = Jsoup.connect(url).get();
        this.scheme = doc.attr("scheme");
        this.host = doc.attr("host");
        if (doc.attr("port") != null && !doc.attr("port").isEmpty()) {
            this.port = Integer.parseInt(doc.attr("port"));
        } else {
            throw new Exception("Port not specified in URL");
        }
    }

    public String getScheme() {
        return scheme;
    }

    public String getHost() {
        return host;
    }

    public int getPort() {
        return port;
    }
}

public class Main {

    @Test

        Connection connection = new Connection("http://example.com");
        assertEquals("http", connection.getScheme());
    }

    @Test

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;

public class Connection {
    private String scheme;
    private String host;
    private int port;

    public static Connection createConnection(String url) throws Exception {
        Document doc = Jsoup.connect(url).get();
        return new Connection(doc.attr("scheme"), doc.attr("host"));
    }

    public Connection(String scheme, String host) {
        this.scheme = scheme;
        this.host = host;
    }

    public String getScheme() {
        return scheme;
    }

    public String getHost() {
        return host;
    }

    public int getPort() {
        return port;
    }
}

public class Main {

    @Test
    public void testGetScheme() {
        Connection connection = Connection.createConnection("http://example.com");
        assertEquals("http", connection.getScheme());
    }

    @Test
    public void testGetHost() {
        Connection connection = Connection.createConnection("http://example.com");
        assertEquals("example.com", connection.getHost());
    }

    @Test
    public void testGetPort() {
        Connection connection = Connection.createConnection("http://localhost:8080");
        assertEquals(80, connection.getPort());
    }
}
