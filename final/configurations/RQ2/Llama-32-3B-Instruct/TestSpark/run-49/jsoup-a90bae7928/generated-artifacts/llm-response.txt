package org.jsoup.parser;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import java.io.IOException;
import java.io.Reader;
import java.util.List;

import static org.junit.Assert.*;
import static org.mockito.ArgumentMatchers.any;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class XmlTreeBuilderTest {

    @Mock
    private TreeBuilder treeBuilderMock;

    @Test
    public void parseFragment_ValidInput_ReturnsList() throws IOException {
        when(treeBuilderMock.parseFragment(any(), any(), any())).thenReturn(List.of());
        Document result = new XmlTreeBuilder().parseFragment("input", "baseUri", new Parser());
        assertNotNull(result);
    }

    @Test
    public void parseFragment_InvalidInput_ThrowsException() throws IOException {
        try {
            new XmlTreeBuilder().parseFragment(null, "baseUri", new Parser());
            fail("Expected exception not thrown");
        } catch (Exception e) {
            // expected
        }
    }

    @Test
    public void initialiseParse_ValidInput_CalledWithCorrectArguments() throws IOException {
        when(treeBuilderMock.initialiseParse(any(), any(), any())).thenReturn(new XmlTreeBuilder());
        new XmlTreeBuilder().initialiseParse(new StringReader("<html></html>"), "baseUri", new Parser());
        verify(treeBuilderMock).initialiseParse(eq(new StringReader("<html></html>")), eq("baseUri"), eq(new Parser()));
    }

    @Test
    public void initialiseParse_InvalidInput_CalledWithCorrectArguments() throws IOException {
        when(treeBuilderMock.initialiseParse(any(), any(), any())).thenReturn(new XmlTreeBuilder());
        new XmlTreeBuilder().initialiseParse(null, "baseUri", new Parser());
        verify(treeBuilderMock).initialiseParse(eq(null), eq("baseUri"), eq(new Parser()));
    }

    @Test
    public void process_ValidToken_ReturnsTrue() {
        when(treeBuilderMock.process(any())).thenReturn(true);
        boolean result = new XmlTreeBuilder().process(new TokenStartTag());
        assertTrue(result);
    }

    @Test
    public void process_InvalidToken_ReturnsFalse() {
        boolean result = new XmlTreeBuilder().process(new TokenComment());
        assertFalse(result);
    }
}
