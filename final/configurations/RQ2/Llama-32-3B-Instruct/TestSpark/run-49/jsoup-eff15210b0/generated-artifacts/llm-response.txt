import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class Response {

    public static Connection.Response post(String url, Document document) {
        return Jsoup.connect(url)
                .data("username", "user")
                .data("password", "pass")
                .form(document.toString())
                .method("post")
                .execute()
                .response();
    }

    public static Connection execute(Connection response) {
        // Extract the content of the HTML element
        String html = response.body().html();

        // Find all links on the page
        Elements links = Jsoup.parse(html).select("a[href]");

        // Print the URL of each link
        for (Element link : links) {
            System.out.println(link.attr("href"));
        }

        return null;
    }
}

class Connection {

    public static Response post(String url, Document document) {
        return Response.post(url, document);
    }

    public static Object execute(Connection response) {
        return Response.execute(response);
    }
}

class Document {

    private String html;

    public Document(String html) {
        this.html = html;
    }

    @Override
    public String toString() {
        return html;
    }
}
