public class JacksonJsonTransformer {

    public Project parseProject(String json) {
        ObjectMapper mapper = new ObjectMapper();
        try {
            return mapper.readValue(json, Project.class);
        } catch (JsonParseException e) {
            throw new RuntimeException(e);
        }
    }

    public ExportOptions parseExportOptions(String json) {
        ObjectMapper mapper = new ObjectMapper();
        try {
            return mapper.readValue(json, ExportOptions.class);
        } catch (JsonParseException e) {
            throw new RuntimeException(e);
        }
    }

    public FileInfo parseFileInfo(String json) {
        ObjectMapper mapper = new ObjectMapper();
        try {
            return mapper.readValue(json, FileInfo.class);
        } catch (JsonParseException e) {
            throw new RuntimeException(e);
        }
    }

    public String convert(Project project) {
        ObjectMapper mapper = new ObjectMapper();
        return mapper.writeValueAsString(project);
    }

    public String convert(ExportOptions options) {
        ObjectMapper mapper = new ObjectMapper();
        return mapper.writeValueAsString(options);
    }

    public String convert(FileInfo fileInfo) {
        ObjectMapper mapper = new ObjectMapper();
        return mapper.writeValueAsString(fileInfo);
    }
}
