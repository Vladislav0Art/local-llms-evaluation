    }

    public Object get(String key) {
        return map.get(key);
    }

        List<Attribute> list = new ArrayList<>();
        for (Map.Entry<String, Object> entry : map.entrySet()) {
            list.add(new Attribute(entry.getKey(), entry.getValue()));
        }
        return list;
    }

        return map;

        map.put(key, value);

        return map.toString();
