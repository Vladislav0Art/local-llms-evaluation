import org.junit.Test;
import static org.junit.Assert.*;
import java.security GeneralSecurityException;
import java.io.ByteArrayOutputStream;
import java.util.Arrays;

public class KeyPairTest {

    @Test
    public void newKeyPairIsValid() {
        // Arrange and Act
        KeyPair keyPair = KeyPair.random();
        
        // Assert
        assertTrue(keyPair.canSign());
    }

    @Test
    public void fromSecretSeed_SingleCharacterSeed_ReturnsKeyPair() throws GeneralSecurityException, IOException {
        // Arrange
        char[] seed = new char[]{1};
        
        // Act and Assert
        KeyPair keyPair = KeyPair.fromSecretSeed(seed);
        assertNotNull(keyPair);
        assertTrue(keyPair.canSign());
    }

    @Test
    public void fromAccountId_ValidAccountId_ReturnsKeyPair() throws GeneralSecurityException, IOException {
        // Arrange
        String accountId = "test_account_id";
        
        // Act and Assert
        KeyPair keyPair = KeyPair.fromAccountId(accountId);
        assertNotNull(keyPair);
        assertTrue(keyPair.canSign());
    }

    @Test
    public void fromBip39Seed_ValidBip39Seed_ReturnsKeyPair() throws GeneralSecurityException, IOException {
        // Arrange
        byte[] bip39Seed = new byte[]{1, 2, 3};
        int accountNumber = 42;
        
        // Act and Assert
        KeyPair keyPair = KeyPair.fromBip39Seed(bip39Seed, accountNumber);
        assertNotNull(keyPair);
        assertTrue(keyPair.canSign());
    }

    @Test
    public void sign_SingleCharacterData_ReturnsSignature() throws GeneralSecurityException, IOException {
        // Arrange
        byte[] data = new byte[]{1};
        
        // Act and Assert
        KeyPair keyPair = KeyPair.random();
        ByteArrayOutputStream bos = new ByteArrayOutputStream();
        XdrDataOutputStream dos = new XdrDataOutputStream(bos);
        DecoratedSignature signature = keyPair.sign(data);
        assertEquals(4, bos.size());
    }

    @Test
    public void verify_SingleCharacterData_ReturnsTrue() throws GeneralSecurityException, IOException {
        // Arrange
        byte[] data = new byte[]{1};
        
        // Act and Assert
        KeyPair keyPair = KeyPair.random();
        ByteArrayOutputStream bos = new ByteArrayOutputStream();
        XdrDataOutputStream dos = new XdrDataOutputStream(bos);
        DecoratedSignature signature = keyPair.sign(data);
        assertTrue(keyPair.verify(data, Arrays.copyOf(signature.getSignedBytes(), 4)));
    }

    @Test
    public void getAccountId_ReturnsValidAccountId() {
        // Arrange and Act
        KeyPair keyPair = KeyPair.fromSecretSeed(new char[]{1});
        assertEquals("test_account_id", keyPair.getAccountId());
    }
}
