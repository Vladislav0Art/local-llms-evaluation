public class AppVersion {
    private String version;
    private int buildNumber;
    private String commitHash;

    public AppVersion(String version, int buildNumber, String commitHash) throws IllegalArgumentException {
        if (version == null || version.isEmpty()) {
            throw new IllegalArgumentException("Version cannot be empty or null");
        }
        if (buildNumber <= 0) {
            throw new IllegalArgumentException("Build number must be greater than zero");
        }
        if (!commitHash.matches("[a-zA-Z0-9]+")) {
            throw new IllegalArgumentException("Commit hash can only contain letters and numbers");
        }

        this.version = version;
        this.buildNumber = buildNumber;
        this.commitHash = commitHash;
    }

    public String getVersion() {
        return version;
    }

    public void setVersion(String version) {
        if (version == null || version.isEmpty()) {
            throw new IllegalArgumentException("Version cannot be empty or null");
        }
        this.version = version;
    }

    public int getBuildNumber() {
        return buildNumber;
    }

    public void setBuildNumber(int buildNumber) throws IllegalArgumentException {
        if (buildNumber <= 0) {
            throw new IllegalArgumentException("Build number must be greater than zero");
        }
        this.buildNumber = buildNumber;
    }

    public String getCommitHash() {
        return commitHash;
    }

    public void setCommitHash(String commitHash) throws IllegalArgumentException {
        if (!commitHash.matches("[a-zA-Z0-9]+")) {
            throw new IllegalArgumentException("Commit hash can only contain letters and numbers");
        }
        this.commitHash = commitHash;
    }
}

public class AppVersionTest {

    @org.junit.Test
    public void testGetVersion() {
        try {
            AppVersion appVersion1 = new AppVersion(null, 123, "commit-1");

        } catch (IllegalArgumentException e) {}

            AppVersion appVersion2 = new AppVersion("", 123, "commit-1");
            org.junit.Assert.fail("Expected IllegalArgumentException for empty version");
        } catch (IllegalArgumentException e) {}

        AppVersion appVersion3 = new AppVersion("1.9.2", 123, "commit-1");

        org.junit.Assert.assertEquals("1.9.2", appVersion3.getVersion());
    }

    public void testGetBuildNumber() {
        try {

            org.junit.Assert.fail("Expected IllegalArgumentException for build number less than or equal to zero");
        } catch (IllegalArgumentException e) {}

        AppVersion appVersion = new AppVersion("1.9.2", -1, "commit-1");

    }

            AppVersion appVersion = new AppVersion("1.9.2", 123, "a!b!");

        } catch (IllegalArgumentException e) {}

        org.junit.Assert.assertEquals("commit-1", appVersion.getCommitHash());
    }
