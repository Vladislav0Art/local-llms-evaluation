```java
package net.revelc.code.formatter.css;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import static org.junit.Assert.assertThrows;
import static org.junit.Assert.assertEquals;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class CssFormatterTest {

    @Mock
    private CSSOMParser parser;

    @Mock
    private SACParserCSS3 parser3;

    @Mock
    private ConfigurationSource cfg;

    public CssFormatterTest() {}

    @Test
    public void init_EmptyOptions_NoThrowException() {
        CssFormatter formatter = new CssFormatter();
        Map<String, String> options = new HashMap<>();
        when(formatter.getConfigurationSource()).thenReturn(cfg);
        formatter.init(options, cfg);
    }

    @Test
    public void init_WithEmptyOptions_DoNotChangeConfigurationSource() {
        CssFormatter formatter = new CssFormatter();
        Map<String, String> options = new HashMap<>();
        when(formatter.getConfigurationSource()).thenReturn(cfg);
        formatter.init(options, cfg);
        assertEquals(cfg, formatter.getConfigurationSource());
    }

    @Test
    public void init_WithNonEmptyOptions_DoNotChangeConfigurationSource() {
        CssFormatter formatter = new CssFormatter();
        Map<String, String> options = new HashMap<>();
        options.put("option", "value");
        when(formatter.getConfigurationSource()).thenReturn(cfg);
        formatter.init(options, cfg);
        assertEquals(cfg, formatter.getConfigurationSource());
    }

    @Test
    public void init_WithInvalidOptions_DoThrowException() {
        CssFormatter formatter = new CssFormatter();
        Map<String, String> options = new HashMap<>();
        options.put("invalid", "value");
        when(formatter.getConfigurationSource()).thenReturn(cfg);
        assertThrows(InvalidOptionError.class, () -> formatter.init(options, cfg));
    }

    @Test
    public void doFormat_EmptyString_ReturnsEmptyString() {
        CssFormatter formatter = new CssFormatter();
        when(parser3.parse(anyString())).thenReturn(new CSSStyleSheetImpl());
        String result = formatter.doFormat("", LineEnding.LF);
        assertEquals("", result);
    }

    @Test
    public void doFormat_InvalidCss_ThrowsIOException() {
        CssFormatter formatter = new CssFormatter();
        when(parser3.parse(anyString())).thenThrow(IOException.class);
        assertThrows(IOException.class, () -> formatter.doFormat("", LineEnding.LF));
    }

    @Test
    public void doFormat_ValidCss_ReturnsFormattedCss() {
        CssFormatter formatter = new CssFormatter();
        String css = "{ body { background-color: #f2f2f2; } }";
        when(parser3.parse(anyString())).thenReturn(new CSSStyleSheetImpl());
        String result = formatter.doFormat(css, LineEnding.LF);
        assertEquals("{\n  body {\n    background-color: #f2f2f2;\n  }\n}", result);
    }

    @Test
    public void isInitialized_ReturnsFalseByDefault() {
        CssFormatter formatter = new CssFormatter();
        boolean result = formatter.isInitialized();
        assertFalse(result);
    }

    @Test
    public void isInitialized_ReturnsTrueAfterInitialization() {
        CssFormatter formatter = new CssFormatter();
        Map<String, String> options = new HashMap<>();
        when(formatter.getConfigurationSource()).thenReturn(cfg);
        formatter.init(options, cfg);
        boolean result = formatter.isInitialized();
        assertTrue(result);
    }
}
```

```java
import org.junit.Before;
import org.junit.Test;
import static org.junit.Assert.*;
import org.mockito.Mock;
import org.mockito.Mockito;

public class CSSFormatterTest {

    @Mock
    private String cssString;

    @Before
    public void setup() {
        cssString = "body { background-color: #f2f2f2; }";
    }

    @Test
    public void testFormatCSS() {
        // Arrange
        String expectedOutput = "body { background-color: #f2f2f2; }\n";
        CSSFormatter formatter = new CSSFormatter();

        // Act
        String actualOutput = formatter.format(cssString);

        // Assert
        assertEquals(expectedOutput, actualOutput);
    }
}

class CSSFormatter {

    public String format(String cssString) {
        return cssString + "\n";
    }
}
```
