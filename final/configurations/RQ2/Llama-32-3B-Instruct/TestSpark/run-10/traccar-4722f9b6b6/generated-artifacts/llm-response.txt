import org.junit.Test;
import org.mockito.Mock;
import org.mockito.Mockito;

public class Protocol {

    @Test
    public void testGenerateChannel() {
        Channel channel = new Channel();
        when(channel).thenReturn("channel");
        assertEquals("channel", generateChannel());
    }

    @Test
    public void testGenerateAddress() {
        SocketAddress address = new SocketAddress();
        when(address).thenReturn("address");
        assertEquals("address", generateAddress());
    }
}

class Channel {
    public String toString() {
        return "channel";
    }
    
    public String getChannel() {
        return this.toString();
    }
}

class SocketAddress {
    private String value;

    public SocketAddress(String value) {
        this.value = value;
    }

    @Override
    public String toString() {
        return value;
    }

    public String getAddress() {
        return this.toString();
    }
}

public class GeneratedTest {

    @Mock
    private Channel channel;

    @Test
    public void testGenerateChannel() {
        assertEquals("channel", generateChannel());
    }

    @Mock
    private SocketAddress address;

    @Test
    public void testGenerateAddress() {
        assertEquals("address", generateAddress());
    }
    
    public static String generateChannel(Channel channel) {
        return channel.getChannel();
    }
    
    public static String generateAddress(SocketAddress address) {
        return address.getAddress();
    }
}
