import org.jsoup.parser.Tag;
import org.jsoup.helper.Validate;
import org.jsoup.internal.Normalizer;
import org.junit.Test;
import static org.junit.Assert.*;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

@RunWith(MockitoJUnitRunner.class)
public class TagTest {

    @Mock
    private ParseSettings settings;

    @Test
    public void getName_EmptyTag_ReturnsEmptyString() {
        Tag tag = new Tag();
        assertEquals("", tag.getName());
    }

    @Test
    public void normalName_SimpleTag_ReturnsOriginalTagName() {
        Tag tag = new Tag();
        tag.setName("tag");
        assertEquals("tag", tag.normalName());
    }

    @Test
    public void normalName_NonSimpleTag_ReturnsNormalizedTagName() {
        String tagName = "tag!";
        Tag tag = new Tag();
        tag.setName(tagName);
        assertEquals(Normalizer.nfc(tagName), tag.normalName());
    }

    @Test
    public void valueOf_SimpleTag_ReturnsCreatedTag() {
        Tag tag = Tag.valueOf("tag");
        assertNotNull(tag);
        assertEquals("tag", tag.getName());
    }

    @Test
    public void valueOf_KnownSimpleTag_ReturnsKnownTag() {
        Tag tag = Tag.valueOf("tag");
        assertTrue(Tag.isKnownTag(tag));
    }

    @Test
    public void isBlock_SelfClosingTag_ReturnsTrue() {
        Tag tag = new Tag();
        tag.setSelfClosing();
        assertTrue(tag.formatAsBlock());
    }

    @Test
    public void isInline_NonSelfClosingTag_ReturnsFalse() {
        Tag tag = new Tag();
        assertFalse(tag.formatAsBlock());
    }

    @Test
    public void isEmpty_EmptyTag_ReturnsTrue() {
        Tag tag = new Tag();
        assertTrue(tag.isEmpty());
    }

    @Test
    public void isSelfClosing_SelfClosingTag_ReturnsTrue() {
        Tag tag = new Tag();
        tag.setSelfClosing();
        assertTrue(tag.isSelfClosing());
    }

    @Test
    public void isKnownTag_KnownSimpleTag_ReturnsTrue() {
        assertTrue(Tag.isKnownTag("tag"));
    }

    @Test
    public void preserveWhitespace_SelfClosingTag_ReturnsFalse() {
        Tag tag = new Tag();
        assertFalse(tag.preserveWhitespace());
    }

    @Test
    public void formatAsBlock_SelfClosingTag_ReturnsTrue() {
        Tag tag = new Tag();
        assertTrue(tag.formatAsBlock());
    }
}
