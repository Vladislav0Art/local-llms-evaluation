public class WatchProtocolDecoderTest {

    private static final int WATCH_PROTOCOL = 5;

    @Test
    public void decode_WatchProtocol_ReturnsTrue() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(new Protocol(WATCH_PROTOCOL));
        assertTrue(decoder.getHasIndex());
    }

    @Test
    public void getManufacturer_WatchProtocol_MatchesExpectedValue() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(new Protocol(5));
        assertEquals("Watch", decoder.getManufacturer());
    }

    @Test
    public void decode_NoData_ReturnsNull() throws Exception {
        Object msg = null;
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(new Protocol(WATCH_PROTOCOL));
        assertNull(decoder.decode(null, null, msg));
    }

    @Test
    public void decode_ValidData_ReturnsCorrectValues() throws Exception {
        byte[] data = "000102030405060708090a0b0c0d0e".getBytes(StandardCharsets.US_ASCII);
        Object msg = new NetworkMessage(data);
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(new Protocol(WATCH_PROTOCOL));
        SocketAddress remoteAddress = new InetSocketAddress("localhost", 8080);
        Object result = decoder.decode(remoteAddress, null, msg);
        assertTrue(result instanceof Position);
    }

    @Test
    public void decode_InvalidData_ThrowsException() throws Exception {
        byte[] data = "000102030405060708090a".getBytes(StandardCharsets.US_ASCII); // incomplete data
        Object msg = new NetworkMessage(data);
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(new Protocol(WATCH_PROTOCOL));
        assertThrows(Exception.class, () -> decoder.decode(null, null, msg));
    }

    @Test
    public void decode_MultipleValues_ThrowsException() throws Exception {
        byte[] data = "000102030405060708090a0b0c".getBytes(StandardCharsets.US_ASCII); // multiple values
        Object msg = new NetworkMessage(data);
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(new Protocol(WATCH_PROTOCOL));
        assertThrows(Exception.class, () -> decoder.decode(null, null, msg));
    }

    @Test
    public void decode_BadFormat_ThrowsException() throws Exception {
        byte[] data = "InvalidWatchData".getBytes(StandardCharsets.US_ASCII);
        Object msg = new NetworkMessage(data);
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(new Protocol(WATCH_PROTOCOL));
        assertThrows(Exception.class, () -> decoder.decode(null, null, msg));
    }
}
