import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import java.util.Arrays;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

@RunWith(MockitoJUnitRunner.class)
public class CommentsConfigurationTest {

    @Mock
    private SettingsHolder settingsHolder;

    @InjectMocks
    private CommentsConfiguration commentsConfiguration;

    @Test
    public void newCommentsConfigurationIsCreated() {
        Mockito.when(settingsHolder.getComments()).thenReturn(Collections.emptyMap());
        Mockito.verify(commentsConfiguration).setComment(Mockito.anyString(), Mockito.anyString());
    }

    @Test
    public void setComment_setsAllLinesForGivenPath() {
        Map<String, List<String>> expected = new HashMap<>();
        expected.put("path", Arrays.asList("line1", "line2"));
        Mockito.when(settingsHolder.getComments()).thenReturn(expected);
        commentsConfiguration.setComment("path", "line1", "line2");
        Mockito.verify(settingsHolder).setComments(Mockito.anyMap());
    }

    @Test
    public void setComment_setsEmptyLineWithoutMarkerForSingleNewLine() {
        Map<String, List<String>> expected = new HashMap<>();
        expected.put("path", Collections.singletonList(""));
        Mockito.when(settingsHolder.getComments()).thenReturn(expected);
        commentsConfiguration.setComment("path");
        Mockito.verify(settingsHolder).setComments(Mockito.anyMap());
    }

    @Test
    public void getAllComments_returnsEmptyMapForNoConfiguration() {
        Mockito.when(settingsHolder.getComments()).thenReturn(Collections.emptyMap());
        Map<String, List<String>> result = commentsConfiguration.getAllComments();
        Mockito.verify(settingsHolder).getComments();
        assert result.isEmpty();
    }

    @Test
    public void setComment_overridesExistingComments() {
        Map<String, List<String>> expected = new HashMap<>();
        expected.put("path", Arrays.asList("line1", "line2"));
        Mockito.when(settingsHolder.getComments()).thenReturn(expected);
        commentsConfiguration.setComment("path", "newLine1", "newLine2");
        Map<String, List<String>> result = commentsConfiguration.getAllComments();
        assert result.get("path").equals(Arrays.asList("newLine1", "newLine2"));
    }

    @Test
    public void setComment_overridesExistingEmptyLines() {
        Map<String, List<String>> expected = new HashMap<>();
        expected.put("path", Arrays.asList("", "", ""));
        Mockito.when(settingsHolder.getComments()).thenReturn(expected);
        commentsConfiguration.setComment("path");
        Map<String, List<String>> result = commentsConfiguration.getAllComments();
        assert result.get("path").equals(Arrays.asList(""));
    }

    @Test
    public void getAllComments_returnsExpectedMapForExistingConfig() {
        Map<String, List<String>> expected = new HashMap<>();
        expected.put("path", Arrays.asList("line1", "line2"));
        Mockito.when(settingsHolder.getComments()).thenReturn(expected);
        Map<String, List<String>> result = commentsConfiguration.getAllComments();
        assert result.equals(expected);
    }
}

    public static final List<String> COMMENT_LINES = Arrays.asList("line1", "line2");

    @Mock
    private SettingsHolder settingsHolder;

    @InjectMocks
    private CommentsConfiguration commentsConfiguration;

    @Test
    public void testSetComment() {
        commentsConfiguration.setComment(PATH, "newLine1");
        Mockito.verify(settingsHolder).setComments(Mockito.anyMap());
    }

    @Test
    public void testGetAllComments() {
        Map<String, List<String>> expected = new HashMap<>();
        expected.put(PATH, COMMENT_LINES);
        Mockito.when(settingsHolder.getComments()).thenReturn(expected);
        Map<String, List<String>> result = commentsConfiguration.getAllComments();
        assert result.equals(expected);
    }

    @Test
    public void testSetCommentVarargs() {
        commentsConfiguration.setComment(PATH, COMMENT_LINES.toArray(new String[0]));
        Mockito.verify(settingsHolder).setComments(Mockito.anyMap());
    }
}
