public class Url {
    private String url;
    private Map<String, String> kv;

    public Url(String url, Map<String, String> kv) {
        this.url = url;
        this.kv = kv;
    }

    public String appendKeyValue(String key, String value) throws UnsupportedEncodingException {
        if (key == null) {
            throw new NullPointerException();
        }
        return url + "=" + StringUtil.encode(key);
    }

    @Override
    public String toString() {
        return url;
    }
}

public class StringUtil {
    public static String encode(String str) throws UnsupportedEncodingException {
        return java.net.URLEncoder.encode(str, StandardCharsets.UTF_8);
    }
}
