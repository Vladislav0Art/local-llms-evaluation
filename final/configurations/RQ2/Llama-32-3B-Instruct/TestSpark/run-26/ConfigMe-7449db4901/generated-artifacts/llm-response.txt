public class CommentsConfiguration {
    private Map<String, List<String>> allComments = new HashMap<>();

    public void setComment(String path, String... lines) {
        if (!allComments.containsKey(path)) {

import java.util.*;

public class CommentsConfiguration {

    private Map<String, List<String>> allComments = new HashMap<>();

    public void setComment(String path, String... lines) {
        if (!allComments.containsKey(path)) {
            allComments.put(path, new ArrayList<>());
        }
        for (String line : lines) {
            ((List<String>) allComments.get(path)).add(line);
        }
    }

    public Map<String, List<String>> getAllComments() {
        return Collections.unmodifiableMap(allComments);
    }
}

public class GeneratedTest {

    @org.junit.Test
    public void newCommentsConfigurationIsEmpty() {
        CommentsConfiguration config = new CommentsConfiguration();
        assertTrue(config.getAllComments().isEmpty());
    }

    @org.junit.Test
    public void setCommentAddsLines() {
        CommentsConfiguration config = new CommentsConfiguration();
        config.setComment("test", "line1", "line2");
        assertEquals(2, ((List<String>) config.getAllComments().get("test")).size());
    }

    @org.junit.Test
    public void getAllCommentsReturnsUnmodifiableMap() {
        CommentsConfiguration config = new CommentsConfiguration();
        Map<String, List<String>> comments = config.getAllComments();
        assertNot null(comments);
        assertTrue(comments instanceof UnmodifiableMap);
    }
}
