    @Test
    public void initInitializationSuccessTest() {
        // Arrange
        Map<String, String> options = Collections.emptyMap();
        ConfigurationSource cfg = new ConfigurationSource();

        // Act
        CssFormatter formatter = new CssFormatter();
        formatter.init(options, cfg);

        // Assert
        assertTrue(formatter.isInitialized());
    }

    @Test
    public void initInitializationFailureTest() {
        // Arrange
        Map<String, String> options = Collections.emptyMap();
        ConfigurationSource cfg = null;

        // Act and Assert
        assertThrows(NullPointerException.class, () -> new CssFormatter().init(options, cfg));
    }

    @Test
    public void doFormatEmptyStringTest() throws IOException {
        // Arrange
        CssFormatter formatter = new CssFormatter();

        // Act
        String formattedCss = formatter.doFormat("", LineEnding.LF);

        // Assert
        assertEquals("", formattedCss);
    }

    @Test
    public void doFormatNonEmptyStringTest() throws IOException {
        // Arrange
        CssFormatter formatter = new CssFormatter();
        String code = "body { color: red; }";

        // Act
        String formattedCss = formatter.doFormat(code, LineEnding.LF);

        // Assert
        assertFalse(formattedCss.isEmpty());
    }

    @Test
    public void doFormatNonEmptyStringWithLineEndingLFTest() throws IOException {
        // Arrange
        CssFormatter formatter = new CssFormatter();
        String code = "body { color: red; }\ndiv { color: blue; }";

        // Act
        String formattedCss = formatter.doFormat(code, LineEnding.LF);

        // Assert
        assertTrue(formattedCss.contains("\n"));
    }

    @Test
    public void doFormatNonEmptyStringWithLineEndingCRLFTest() throws IOException {
        // Arrange
        CssFormatter formatter = new CssFormatter();
        String code = "body { color: red; }\r\ndiv { color: blue; }";

        // Act
        String formattedCss = formatter.doFormat(code, LineEnding.CRLF);

        // Assert
        assertTrue(formattedCss.contains("\r\n"));
    }

    @Test
    public void isInitializedSuccessTest() {
        // Arrange
        CssFormatter formatter = new CssFormatter();

        // Act and Assert
        assertTrue(formatter.isInitialized());
    }

    @Test
    public void isInitializedFailureTest() {
        // Arrange
        CssFormatter formatter = null;

        // Act and Assert
        assertThrows(NullPointerException.class, () -> formatter.isInitialized());
    }
}
