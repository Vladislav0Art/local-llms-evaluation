import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Comment;
import org.jsoup.select.Elements;

public class HtmlCommentParser {

    @org.junit.Test
    public void parseHtmlComments() throws IOException {
        String htmlString = "<html><head><!-- comment --></head><body></body></html>";

        Document document = Jsoup.parse(htmlString);
        Elements comments = document.getComments();

        for (Element comment : comments) {
            System.out.println(comment.data());
        }
    }

    @org.junit.Test
    public void findHtmlComments() throws IOException {
        String htmlString = "<html><head><!-- comment --></head><body><!-- another comment --></body></html>";

        Document document = Jsoup.parse(htmlString);
        Comment[] comments = document.getComments();

        for (Comment comment : comments) {
            System.out.println(comment.data());
        }
    }

    @org.junit.Test
    public void getHtmlComment() throws IOException {
        String htmlString = "<html><head><!-- comment --></head><body></body></html>";

        Document document = Jsoup.parse(htmlString);
        Comment comment = document.getComment(0);

        System.out.println(comment.data());
    }
}
```

In this example, we use the `Jsoup` library to parse an HTML string and extract comments. We then print out the data of each comment using a `for` loop.

Note: The `getComments()` method returns all comments in the document as an array of `Comment` objects. The `getComment(int)` method returns the first comment in the document.

Please replace the `htmlString` variable with your own HTML string to parse and extract comments.

import org.jsoup.nodes.Comment;
import org.jsoup.nodes.Document;

public class CommentClonerTest {

    @org.junit.Test
    public void cloneComment() {
        Document document = new Document();
        Element headElement = document.head().firstChild();
        Comment comment = new Comment("This is a test comment.");
        headElement.appendChild(comment);
        Comment clonedComment = ((Comment) comment.cloneNode(true)).getData();
        assertEquals("This is a test comment.", clonedComment);
    }

    @org.junit.Test
    public void cloneCommentWithAttributes() {
        Document document = new Document();
        Element headElement = document.head().firstChild();
        Comment comment = new Comment("This is a test comment.");
        headElement.appendChild(comment);
        Comment clonedComment = ((Comment) comment.cloneNode(true)).getData();
        assertEquals("This is a test comment.", clonedComment);

        // Test that attributes are preserved when cloning
        Comment originalComment = (Comment) headElement.children().get(0);
        Element element = originalComment Parent().firstChild();
        String attributeValue = element.attr("data");
        Comment clonedComment = ((Comment) originalComment.cloneNode(true)).getData();
        assertEquals(attributeValue, clonedComment);
    }

    @org.junit.Test
    public void cloneComments() {
        Document document = new Document();
        Element headElement = document.head().firstChild();
        Comment comment1 = new Comment("This is a test comment.");
        Comment comment2 = new Comment("This is another test comment.");
        headElement.appendChild(comment1);
        headElement.appendChild(comment2);

        // Clone all comments
        for (Comment comment : comment1.childNodes()) {
            if (comment instanceof org.jsoup.nodes.Comment) {
                ((org.jsoup.nodes.Comment) comment).cloneNode(true);
            }
        }

        assertEquals(3, document.head().children().size());
    }
}
```

In this example, we create a `Document` object and append two comments to the head element. We then clone these comments using the `cloneNode(true)` method and verify that the cloned comments have the same data as the original comments.
