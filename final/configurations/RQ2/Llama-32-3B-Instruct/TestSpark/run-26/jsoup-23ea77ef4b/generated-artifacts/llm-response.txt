        javax.xml.xpath.XPath xPath = javax.xml.xpath.XPath.newInstance();
        javax.xml.xpath.XPathExpression xPathExpr = xPath.compile(xpath);

        try {
            // Evaluate the XPath expression on the document
            javax.xml.xpath.XPathResult result = xPathExpr.evaluate("/", doc, javax.xml.xpath.XPathConstants.NODESET);
            
            // Return the list of nodes that match the XPath expression
            NodeList nodeList = (NodeList) result.getNodeSet();
            return nodeList;
        } catch (javax.xml.xpath.XPathException e) {
            System.out.println("Error evaluating XPath: " + e.getMessage());
            return null;
        }
    }

    public NodeList sourceNodes(NodeList nodeList, Class<?> typeClass) {
        // Filter the list of nodes based on their class
        NodeList filteredNodeList = new NodeList();
        for (int i = 0; i < nodeList.getLength(); i++) {
            Node node = nodeList.item(i);
            if (typeClass.isInstance(node)) {
                filteredNodeList.add(node);
            }
        }
        return filteredNodeList;
    }
}
