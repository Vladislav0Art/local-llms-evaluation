public class OnlineStockSpan {

    private Stack<Integer> stack;

    public OnlineStockSpan() {
        this.stack = new java.util.Stack<>();
    }

    public void next(int price) {
        while (!stack.isEmpty()) {
            if (price >= stack.peek()) {
                stack.pop();
            } else {
                break;
            }
        }
        stack.push(price);
    }

    public int[] calculateSpans(int[] prices) {
        int n = prices.length;
        int[] result = new int[n];
        for (int i = 0; i < n; i++) {
            if (i == 0) {
                result[i] = 1;
                continue;
            }
            while (!stack.isEmpty() && stack.peek() <= prices[i - 1]) {
                stack.pop();
            }
            if (stack.isEmpty()) {
                result[i] = i + 1;
            } else {
                result[i] = i - stack.peek() + 1;
            }
        }
        return result;
    }

}
