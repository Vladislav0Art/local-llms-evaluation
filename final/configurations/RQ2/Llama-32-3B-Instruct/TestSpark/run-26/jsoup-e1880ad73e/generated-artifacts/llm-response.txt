```java
import org.junit.Test;
import java.net.MalformedURLException;
import java.net.URL;
import static org.junit.Assert.assertEquals;

public class UrlBuilderTest {

    @Test
    public void newUrlBuilder_emptyUrl() {
        UrlBuilder urlBuilder = new UrlBuilder(null);
        assertEquals("https://example.com/path?*", urlBuilder.build());
    }

    @Test
    public void build_urlWithQuery_param() throws MalformedURLException {
        String inputUrl = "https://example.com/path?a=1&b=2";
        URL url = new URL(inputUrl);
        UrlBuilder urlBuilder = new UrlBuilder(url);
        assertEquals("https://example.com/path?a=1&b=2", urlBuilder.build());
    }

    @Test
    public void build_urlWithQuery_param_escapedChar() throws MalformedURLException {
        String inputUrl = "https://example.com/path?a=1&b=%E2%80%A6";
        URL url = new URL(inputUrl);
        UrlBuilder urlBuilder = new UrlBuilder(url);
        assertEquals("https://example.com/path?a=1&b=%C2%A6", urlBuilder.build());
    }

    @Test
    public void build_urlWithQuery_param_multiple() throws MalformedURLException {
        String inputUrl = "https://example.com/path?a=1&b=2&c=3";
        URL url = new URL(inputUrl);
        UrlBuilder urlBuilder = new UrlBuilder(url);
        assertEquals("https://example.com/path?a=1&b=2&c=3", urlBuilder.build());
    }

}

public class UrlBuilder {

    private java.net.URL url;

    public UrlBuilder(java.net.URL url) {
        this.url = url;
    }

    public String build() {
        StringBuilder sb = new StringBuilder(url.toString());
        int queryIndex = sb.indexOf('?');
        if (queryIndex != -1) {
            String query = sb.substring(queryIndex + 1);
            // parse query string and add to sb
            // for simplicity, this example just appends the query to the end
            sb.append("&").append(url.getQuery());
        }
        return sb.toString();
    }

}

```

This implementation includes a `UrlBuilder` class that takes a URL as input and builds a new URL with the query parameters appended. The `build()` method parses the original URL, extracts the query string, and appends it to the end of the URL.

Note: This is a simplified example and does not handle cases such as handling duplicate query parameters or parsing the query string into key-value pairs.
