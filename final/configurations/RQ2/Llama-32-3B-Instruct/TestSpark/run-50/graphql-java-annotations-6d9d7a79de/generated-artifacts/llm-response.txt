    String value();
}

// GraphQLDirective class
@Target(ElementType.METHOD)
@Retention(RetentionPolicy.RUNTIME)
@interface GraphQLDirective {
    String value();
}

    public static List<Method> getMethodsByAnnotation(Class<?> clazz, String annotationName) {
        List<Method> methods = new ArrayList<>();
        for (Method method : clazz.getDeclaredMethods()) {
            if (method.isAnnotationPresent(MyAnnotation.class)) {
                MyAnnotation annotation = method.getAnnotation(MyAnnotation.class);
                if (annotation.value().equals(annotationName)) {
                    methods.add(method);
                }
            }
        }
        return methods;
    }

    public static void main(String[] args) throws Exception {
        Class<?> clazz = MainExample.class;

        // Get methods with MyAnnotation
        List<Method> myMethods = getMethodsByAnnotation(clazz, "myValue");

        for (Method method : myMethods) {
            System.out.println("Method Name: " + method.getName());
            System.out.println("Method Signature: " + method.toString());
            System.out.println();

        // Get methods with GraphQLDirective
