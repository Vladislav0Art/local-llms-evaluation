    }

        Connection request = HttpConnection.newRequest();
        assertNotNull(request);

        assertNotNull(connection);
    }

        Connection connection = HttpConnection.proxy(proxy);
        assertEquals(proxy, connection.proxy());

        String userAgent = "Mozilla/5.0";
        Connection connection = HttpConnection.userAgent(userAgent);
        assertEquals(userAgent, connection.userAgent());

        Connection connection = HttpConnection.timeout(timeout);
        assertEquals(timeout, connection.timeout());

        }});
    }

        byte[] data = {"<html><body>Hello World!</body></html>".getBytes(UTF_8));
        ByteArrayInputStream inputStream = new ByteArrayInputStream(data);
        HttpURLConnection connection = (HttpURLConnection) inputStream.getInputStream();
        Connection request = HttpConnection.request().url("http://example.com");
        request.get().connect();
        assertEquals(document, ((Document) request.execute().response()).get());
    }

    @Test
    public void post_ReturnsDocument() throws IOException {
        Document document = new Document();
        byte[] data = {"<html><body>Hello World!</body></html>".getBytes(UTF_8)};
        ByteArrayInputStream inputStream = new ByteArrayInputStream(data);
        HttpURLConnection connection = (HttpURLConnection) inputStream.getInputStream();
        Connection request = HttpConnection.request().url("http://example.com").requestBody(String.valueOf(document));
        request.post().connect();
        assertEquals(document, ((Document) request.execute().response()).get());
