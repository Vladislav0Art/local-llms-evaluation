public class GalileoProtocolDecoder {
    private static final String GALILEO_PROTOCOL = "Galileo Protocol";

    public static String decode(String message) {
        if (message.startsWith(GALILEO_PROTOCOL)) {
            return message;
        } else {
            return "";
        }
    }

    public static void main(String[] args) {
        System.out.println(decode("Galileo Protocol: Hello World"));
        System.out.println(decode("Invalid Message"));
        System.out.println(decode("Galileo Protocol: Invalid Message"));
        System.out.println(decode("Valid Galileo Protocol"));
    }
}

public class TestGalileoProtocolDecoder {
    @Test
    public void testDecodeValidMessage() {
        String message = "Galileo Protocol: Hello World";
        GalileoProtocolDecoder decoder = new GalileoProtocolDecoder();
        assertEquals(GALILEO_PROTOCOL + ": Hello World", decoder.decode(message));
    }

    @Test
    public void testDecodeInvalidMessage() {
        String message = "Invalid Message";
        GalileoProtocolDecoder decoder = new GalileoProtocolDecoder();
        assertEquals("", decoder.decode(message));
    }

    @Test
    public void testDecodeEmptyMessage() {
        String message = "";
        GalileoProtocolDecoder decoder = new GalileoProtocolDecoder();
        assertEquals("", decoder.decode(message));
    }

    @Test
    public void testDecodeNoProtocol() {
        String message = "Hello World";
        GalileoProtocolDecoder decoder = new GalileoProtocolDecoder();
        assertEquals("", decoder.decode(message));
    }
}

public class GalileoProtocolDecoder {
    private static final String GALILEO_PROTOCOL = "Galileo Protocol";

    public String decode(String message) {
        if (message.startsWith(GALILEO_PROTOCOL)) {
            return message;
        } else {
            return "";
        }
    }

    public boolean assertEquals(String expected, String actual) {
        if (expected.equals(actual)) {
            return true;
        } else {
            return false;
        }
    }

    public static void main(String[] args) {
        GalileoProtocolDecoder decoder = new GalileoProtocolDecoder();
        System.out.println(decoder.decode("Galileo Protocol: Hello World"));
        System.out.println(decoder.decode("Invalid Message"));
        System.out.println(decoder.decode(""));
        System.out.println(decoder.decode("Hello World"));
    }
}

    public void testDecodeInvalidMessage() {
