```java
// Define a Node class to represent a tree node
class Node {
    String data;
    Node left, right;

    // Constructor to initialize the node
    public Node(String data) {
        this.data = data;
        this.left = null;
        this.right = null;
    }
}

// Create a utility function to clone a given tree
Node cloneTree(Node root) {
    if (root == null)
        return null;

    // Create a new node with the same data as the original node
    Node newRoot = new Node(root.data);
    newRoot.left = cloneTree(root.left);
    newRoot.right = cloneTree(root.right);

    return newRoot;
}

// Define a class to represent a cloned tree node
class ClonedNode {
    String data;

    // Constructor to initialize the node with given data
    public ClonedNode(String data) {
        this.data = data;
    }

    @Override
    public String toString() {
        return "ClonedNode{" +
                "data='" + data + '\'' +
                '}';
    }
}

public class Main {

    public static void main(String[] args) {
        // Create the original tree
        Node root = new Node("A");
        root.left = new Node("B");
        root.right = new Node("D");
        root.left.left = new Node("D1");
        root.left.right = new Node("D2");

        // Clone the original tree
        Node clonedRoot = cloneTree(root);

        // Clone the cloned tree with new text
        ClonedNode clonedClonedRoot = new ClonedNode("X");
        System.out.println(clonedClonedRoot);
    }
}
```

```java
public class TextNode {
    private String text;

    // Constructor to initialize the node
    public TextNode(String text) {
        this.text = text;
    }

    @Override
    public String toString() {
        return "TextNode{" +
                "text='" + text + '\'' +
                '}';
    }

    // Method to clone the node with new text
    public TextNode clone(String newText) {
        return new TextNode(newText);
    }
}

public class Main {

    public static void main(String[] args) {
        // Create a cloned tree node
        TextNode clonedTree = new TextNode("Hello World!");
        
        // Clone the cloned tree with new text
        TextNode clonedClonedTree = clonedTree.clone("New Hello World!");

        System.out.println(clonedTree);
        System.out.println(clonedClonedTree.text());
    }
}
```
