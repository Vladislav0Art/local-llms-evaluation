public class Safelist {

    public static java.util.Set<String> simpleText() {
        return new java.util.HashSet<>();
    }

    public static java.util.Set<String> getSafeTags(String html) {
        org.jsoup.safety.Whitelist whitelist = org.jsoup.safety.Whitelist.none();
        Document doc = Jsoup.parse(html);
        for (org.jsoup.nodes.Element element : doc.getAllElements()) {
            for (org.jsoup.nodes.Node node : element.childNodes()) {
                if (node instanceof org.jsoup.nodes.TextNode) {
                    String text = ((org.jsoup.nodes.TextNode)node).text();
                    whitelist.add(text);
                }
            }
        }
        return new java.util.HashSet<>(whitelist);
    }

    public static Attributes removeTags(String... tags) {
        Document doc = Jsoup.parse("<html>");
        for (String tag : tags) {
            doc.select(tag).remove();
        }
        return new Attributes();
    }

    public static java.util.Set<String> getEnforcedAttributes(String tag) {
        String[] words = tag.split(" ");
        java.util.Set<String> set = new java.util.HashSet<>();
        for (String word : words) {
            set.add(word);
        }
        return set;
    }
}

class GeneratedTest {

    public static void main(String[] args) {
        assert Safelist.simpleText().size() == 0;
        Document doc = Jsoup.parse("<html><body><p>test</p></body></html>");
        java.util.Set<String> safeElements = Safelist.getSafeTags(doc.toString());
        assert safeElements.size() == 4;

        Attributes attributes = Safelist.removeTags("img", "a");
        assert attributes.set.isEmpty();

        String tag = "test";
        java.util.Set<String> set = Safelist.getEnforcedAttributes(tag);
        assert set.size() > 0;
    }
}
