package org.jsoup.helper;

import org.jsoup.Connection;
import org.junit.Test;
import static org.junit.Assert.*;
import java.io.UnsupportedEncodingException;
import org.mockito.Mockito;
import org.jsoup.internal.StringUtil;
import javax.annotation.Nullable;
import java.net.IDN;
import java.net.MalformedURLException;
import java.net.URI;
import java.net.URISyntaxException;
import java.net.URL;
import java.net.URLDecoder;
import java.net.URLEncoder;

public class UrlBuilderTest {

    @Test
    public void urlBuilder constructionWithUrl Test() {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        assertNotNull(builder);
    }

    @Test
    public void buildUrlWithEmptyString returnsEmptyString Test() {
        UrlBuilder builder = new UrlBuilder("");
        String result = builder.build();
        assertEquals("", result);
    }

    @Test
    public void buildUrlWithSingleCharacterString returnsSingleCharacterString Test() {
        UrlBuilder builder = new UrlBuilder("a");
        String result = builder.build();
        assertEquals("a", result);
    }

    @Test
    public void appendKeyValToUrlWithEmptyKeyAndValue ReturnsSameUrl Test() throws UnsupportedEncodingException {
        Connection.KeyVal kv = Mockito.mock(Connection.KeyVal.class);
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        builder.appendKeyVal(kv);
        String result = builder.build();
        assertNotNull(result);
    }

    @Test
    public void appendKeyValToUrlWithSingleCharacterKeyAndValue ReturnsSameUrl Test() throws UnsupportedEncodingException {
        Connection.KeyVal kv = Mockito.mock(Connection.KeyVal.class);
        kv.setValue("a");
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        builder.appendKeyVal(kv);
        String result = builder.build();
        assertNotNull(result);
    }

    @Test
    public void appendUrlKeyValWithEmptyStringAndKeyAndValue ReturnsSameUrl Test() throws UnsupportedEncodingException {
        Connection.KeyVal kv = Mockito.mock(Connection.KeyVal.class);
        kv.setValue("");
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        builder.appendUrlKeyVal(kv);
        String result = builder.build();
        assertNotNull(result);
    }

    @Test
    public void appendUrlKeyValWithSingleCharacterStringAndKeyAndValue ReturnsSameUrl Test() throws UnsupportedEncodingException {
        Connection.KeyVal kv = Mockito.mock(Connection.KeyVal.class);
        kv.setValue("a");
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        builder.appendUrlKeyVal(kv);
        String result = builder.build();
        assertNotNull(result);
    }

    @Test
    public void appendUrlKeyValWithMultipleStringAndKeysAndValues ReturnsSameUrl Test() throws UnsupportedEncodingException {
        Connection.KeyVal kv1 = Mockito.mock(Connection.KeyVal.class);
        kv1.setValue("a");
        Connection.KeyVal kv2 = Mockito.mock(Connection.KeyVal.class);
        kv2.setValue("b");
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        builder.appendUrlKeyVal(kv1);
        builder.appendUrlKeyVal(kv2);
        String result = builder.build();
        assertNotNull(result);
    }

    @Test
    public void urlBuilder constructionWithNullUrl throws NullPointerException {
        assertThrows(NullPointerException.class, () -> new UrlBuilder(null));
    }

    @Test
    public void buildUrlWithNullString returnsEmptyString Test() {
        UrlBuilder builder = new UrlBuilder("");
        String result = builder.build();
        assertEquals("", result);
    }
}
