import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;
import org.junit.Test;

import static org.mockito.Mockito.when;
import static org.mockito.Mockito.anyString;

public class DocumentTest {
    @Test
    public void testDocument() {
        Document document = new Document();
        String actualSettings = createDocument().outputSettings();
        when(document.outputSettings()).thenReturn("abc");

        Document expectedDocument = new Document();
        Elements elements = expectedDocument.select("*");
        for (int i = 0; i < elements.size(); i++) {
            Element element = elements.get(i);
            if (i == 0) {
                element.attr("value", "def");
            }
            if (i == 1) {
                element.attr("type", "button");
            }
        }

        when(createDocument()).thenReturn(expectedDocument);

        String actualSetting = createDocument().outputSettings();
        assert actualSetting.equals("abc");

        actualSettings = createDocument().outputSettings();
        assert actualSetting.equals("abc");
    }

    public Document createDocument() {
        return new Document();
    }
}

public class Document {
    private String outputSettings;

    public String outputSettings() {
        return outputSettings;
    }

    public void outputSettings(String settings) {
        this.outputSettings = settings;
    }
}

public class QuirksMode {
    // No implementation
}

public class PublicDocument extends Document {
    @Override
    public String outputSettings() {
        return super.outputSettings();
    }
}
