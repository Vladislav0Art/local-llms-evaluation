package com.example;

public class EvalEx {
    public static void main(String[] args) {
        String expressionString = "sin(x)";
        execute(expressionString);
    }

    private static double execute(String expressionString) {
        return sin(expressionString);
    }

    public static double sin(String x) {
        return Math.sin(Double.parseDouble(x));
    }
}

class Tokenizer {
    private String expressionString;
    private int position;

    public Tokenizer(String expressionString) {
        this.expressionString = expressionString;
        this.position = 0;
    }

    public void parse() {
        while (position < expressionString.length()) {
            if (expressionString.charAt(position) == '(') {
                position++;
                break;
            }
            if (Character.isLetter(expressionString.charAt(position)) || Character.isDigit(expressionString.charAt(position))) {
                buildToken();
            } else if (expressionString.charAt(position) == '+') {
                position++;
                buildToken();
            } else if (expressionString.charAt(position) == '-') {
                position++;
                buildToken();
            }
        }
    }

    private void buildToken() {
        StringBuilder token = new StringBuilder();
        while (position < expressionString.length()) {
            char c = expressionString.charAt(position);
            if (Character.isLetter(c) || Character.isDigit(c)) {
                token.append(c);
                position++;
            } else {
                break;
            }
        }
        returnToken(token.toString());
    }

    private void returnToken(String token) {
        // implement logic to return the token type and variable name
    }
}

class ExpressionConfiguration {
    public static VariableType getVariable() {
        return VariableType.X;
    }

    public static VariableType getVariable1() {
        return VariableType.Y;
    }

    public static FunctionType getFunction() {
        return FunctionType.SIN;
    }
}

enum FunctionType {
    SIN,
    COS,
    TAN
}

enum VariableType {
    X,
    Y
}
