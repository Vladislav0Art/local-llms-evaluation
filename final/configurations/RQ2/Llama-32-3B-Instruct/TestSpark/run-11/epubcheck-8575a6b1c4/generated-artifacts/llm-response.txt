import org.junit.runner.RunWith;

import java.io.File;

import java.util.HashMap;
import java.util.Locale;

import java.util.regex.Matcher;

import static org.junit.Assert.*;
import static org.mockito.Mockito.when;

@RunWith(PowerMockRunner.class)
@PrepareForTest(EPUBProfile.class, EpubCheckFactory.class, Report.class, LocalizableReport.class)
public class EpubChecker {

    @Test
    public void getLocale_EmptyString_ReturnsEmptyLocale() {
        when(EpubCheckFactory.getInstance().getLocale()).thenReturn("");
        Locale locale = new EpubChecker().getLocale();
        assertEquals(new Locale(""), locale);
    }

    @Test
    public void run_NoArgs_ReturnsZero() {
        PowerMockito.mockStatic(EpubCheckFactory.class);
        when(EpubCheckFactory.getInstance().run(null)).thenReturn(0);
        int result = new EpubChecker().run(new String[0]);
        assertEquals(0, result);
    }

    @Test
    public void run_WithArgs_ReturnsZero() {
        PowerMockito.mockStatic(EpubCheckFactory.class);
        when(EpubCheckFactory.getInstance().run(Mockito.anyArray(String[].class))).thenReturn(0);
        int result = new EpubChecker().run(new String[]{"a", "b"});
        assertEquals(0, result);
    }

    @Test
    public void validateFile_WithValidFile_ReturnsZero() throws IOException {
        when(URLUtils.isEpubFile(Mockito.any(File.class))).thenReturn(true);
        PowerMockito.mockStatic(EpubCheckFactory.class);
        when(EpubCheckFactory.getInstance().validateFile(Mockito.anyString(), Mockito.any(EPUBVersion.class), Mockito.any(Report.class), Mockito.any(EPUBProfile.class))).thenReturn(0);
        int result = new EpubChecker().validateFile(new File("a.epub"), EPUBVersion.V3, new Report(), new EPUBProfile());
        assertEquals(0, result);
    }

    @Test
    public void validateFile_WithInvalidFile_ReturnsNonZero() throws IOException {
        when(URLUtils.isEpubFile(Mockito.any(File.class))).thenReturn(false);
        PowerMockito.mockStatic(EpubCheckFactory.class);
        when(EpubCheckFactory.getInstance().validateFile(Mockito.anyString(), Mockito.any(EPUBVersion.class), Mockito.any(Report.class), Mockito.any(EPUBProfile.class))).thenReturn(-1);
        int result = new EpubChecker().validateFile(new File("a.html"), EPUBVersion.V3, new Report(), new EPUBProfile());
        assertEquals(-1, result);
    }

    @Test
    public void getLocale_WithEmptyString_ReturnsNull() {
        when(EpubCheckFactory.getInstance().getLocale()).thenReturn(null);
        Locale locale = new EpubChecker().getLocale();
        assertNull(locale);
    }
}
