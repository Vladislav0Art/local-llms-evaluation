    @Before
    public void setup() {
        this.httpConnection = Mockito.mock(Connection.class);
    }

    @Test
    public void testExecute() {
        // Given
        Response response = Mockito.mock(Response.class);

        when(httpConnection.execute(response)).thenReturn(httpConnection);

        // When
        Connection connection = httpConnection.execute(response);

        // Then
        assertEquals(httpConnection, connection);
    }
}
