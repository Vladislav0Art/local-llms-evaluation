```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

public class HtmlParser {
    public static void main(String[] args) {
        String html = "<html><body><h1>This is a title</h1><p>This is a paragraph.</p></body></html>";

        // Parse the HTML
        Document document = Jsoup.parse(html);

        // Find all headings (h1-h6)
        Elements headings = document.select("h1, h2, h3, h4, h5, h6");

        // Print the text of each heading
        for (Element heading : headings) {
            System.out.println(heading.text());
        }
    }
}
```

This code parses an HTML string and finds all headings (`<h1>`, `<h2>`, etc.) within it. It then prints the text of each heading.
