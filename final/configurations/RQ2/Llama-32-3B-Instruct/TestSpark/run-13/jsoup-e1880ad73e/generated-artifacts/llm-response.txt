import org.junit.Test;
import static org.junit.Assert.*;
import java.io.UnsupportedEncodingException;

import java.net.MalformedURLException;
import java.net.URI;
import java.net.URISyntaxException;
import java.net.URL;
import java.net.URLDecoder;
import java.net.URLEncoder;
import org.mockito.Mockito;
import org.jsoup.Connection;
import org.jsoup.internal.StringUtil;

public class UrlBuilderTest {

    @Test
    public void ofUrlBuilder() {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        assertNotNull(builder);
    }

    @Test
    public void buildUrl() throws MalformedURLException, URISyntaxException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        URI uri = builder.build();
        assertEquals("http://example.com", uri.toString());
    }

    @Test
    public void appendKeyValQuery() throws UnsupportedEncodingException, MalformedURLException, URISyntaxException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com?key=value"));
        Connection.KeyVal kv = Connection.KeyVal.of("key2", "value2");
        builder.appendKeyVal(kv);
        assertEquals("http://example.com?key=value&key2=value2", builder.build().toString());
    }

    @Test
    public void appendKeyValFragment() throws UnsupportedEncodingException, MalformedURLException, URISyntaxException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com#fragment"));
        Connection.KeyVal kv = Connection.KeyVal.of("key", "value");
        builder.appendKeyVal(kv);
        assertEquals("http://example.com#fragment&key=value", builder.build().toString());
    }

    @Test
    public void appendKeyValUrl() throws UnsupportedEncodingException, MalformedURLException, URISyntaxException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        Connection.KeyVal kv = Connection.KeyVal.of("key2", "value2");
        URI uri = new URI("http://other.example.com?key=value2");
        builder.appendKeyVal(kv);
        assertEquals("http://other.example.com?key=value2", builder.build().toString());
    }

    @Test
    public void appendKeyValEmpty() {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        Connection.KeyVal kv = Connection.KeyVal.of("", "");
        assertNotNull(builder.build());
    }
}

import java.net.URISyntaxException;
import java.net.URL;
import java.net.URLDecoder;
import java.net.URLEncoder;
import org.mockito.Mockito;
import org.jsoup.Connection;
import org.jsoup.internal.StringUtil;

public class UrlBuilderTest {

    @Test
    public void ofUrlBuilder() {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        assertNotNull(builder);
    }

    @Test
    public void buildUrl() throws MalformedURLException, URISyntaxException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        String url = builder.build().toString();
        assertEquals("http://example.com", url);
    }

    @Test
    public void appendKeyValQuery() throws UnsupportedEncodingException, MalformedURLException, URISyntaxException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        Connection.KeyVal kv = new Mockito.Object() {
            public String of(String key, String value) { return "key=" + key + "=" + value; }
        };
        String result = builder.appendKeyVal(kv).build().toString();
        assertEquals("http://example.com?key=value", result);
    }

    @Test
    public void appendKeyValFragment() throws UnsupportedEncodingException, MalformedURLException, URISyntaxException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        Connection.KeyVal kv = new Mockito.Object() {
            public String of(String key, String value) { return "#" + value; }
        };
        String result = builder.appendKeyVal(kv).build().toString();
        assertEquals("http://example.com#fragment", result);
    }

    @Test
    public void appendKeyValUrl() throws UnsupportedEncodingException, MalformedURLException, URISyntaxException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        String url = new Mockito.Object() {
            public String of(String key, String value) { return "http://other.example.com?" + value; }
        }.of("key", "value");
        String result = builder.appendKeyVal(new Mockito.Object() {
            public String of(String key, String value) { return "key=" + key + "=" + URLEncoder.encode(value, "UTF-8"); }
        }).appendKeyVal(new Mockito.Object() {
            public String of(String key, String value) { return "http://other.example.com?" + value; }
        }).build().toString();
        assertEquals("http://other.example.com?key=value", result);
    }

    @Test
    public void appendKeyValEmpty() throws UnsupportedEncodingException, MalformedURLException, URISyntaxException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com"));
        String result = builder.appendKeyVal(new Mockito.Object() {
            public String of(String key, String value) { return ""; }
        }).appendKeyVal(new Mockito.Object() {
            public String of(String key, String value) { return "value"; }
        }).build().toString();
        assertNotNull(result);
    }

    public void ofUrlBuilder() {
        UrlBuilder builder = new UrlBuilder();
        assertNotNull(builder);
    }

    @Test
    public void appendKeyVal() {
        UrlBuilder builder = new UrlBuilder();
        Connection connection = new Connection();
        connection.appendKeyVal("key1", "value1");
        assertNotNull(connection.getKeyVals());
    }

    @Test
    public void build() {
        UrlBuilder builder = new UrlBuilder();
        Connection connection = new Connection();
        connection.appendKeyVal("key1", "value1");
        String result = connection.build();
        assertEquals("http://example.com/key1=value1", result);
    }
}

class UrlBuilder {
    private Connection connection;

    public UrlBuilder() {
        this.connection = new Connection();
    }

    public void appendKeyVal(String key, String value) {
        this.connection.appendKeyVal(key, value);
    }

    public String build() {
        return this.connection.build();
    }
}

class Connection {
    private java.util.HashMap<String, String> keyVals;

    public Connection() {
        this.keyVals = new java.util.HashMap<String, String>();
    }

    public void appendKeyVal(String key, String value) {
        this.keyVals.put(key, value);
    }

    public java.util.HashMap<String, String> getKeyVals() {
        return this.keyVals;
    }

    public String build() {
        StringBuilder sb = new StringBuilder("http://example.com/");
        for (java.util.Map.Entry<String, String> entry : this.keyVals.entrySet()) {
            sb.append(entry.getKey()).append("=").append(entry.getValue()).append("/");
        }
        return sb.toString();
    }
}
