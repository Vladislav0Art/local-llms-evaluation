    public void parseName_InvalidFormat_ReturnsNull() {
        AppVersion appVersion = AppVersion.parseName("invalid-format");
        assertNull(appVersion);
    }

    public void parseName_ValidFormat_ReturnsAppVersion() {

        assertEquals(-1, result);
