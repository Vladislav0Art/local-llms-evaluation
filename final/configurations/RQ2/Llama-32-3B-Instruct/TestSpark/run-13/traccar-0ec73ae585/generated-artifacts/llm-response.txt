import org.junit.jupiter.api.Test;
import io.netty.buffer.ByteBuf;

public class WatchProtocolDecoderTest {

    @Test
    public void testWatchProtocolDecoder() {
        // Create a watch protocol decoder
        WatchProtocolDecoder decoder = new WatchProtocolDecoder();

        // Set the protocol version and sequence number
        int version = 1;
        long seqNum = 12345L;

        // Test decoding
        ByteBuf byteBuf = Unpooled.buffer();
        byteBuf.writeInt(version);
        byteBuf.putLong(seqNum);

        decoder.decode(byteBuf);

        // Assert that the decoded bytes are correct
        assertEquals(3, byteBuf.readableBytes());
    }
}

public class WatchProtocolDecoder {

    public void decode(ByteBuf byteBuf) {
        int version = byteBuf.readInt();
        long seqNum = byteBuf.readLong();

        System.out.println("Decoded version: " + version);
        System.out.println("Decoded sequence number: " + seqNum);
    }
}

public class WatchProtocolDecoderFactory {

    public static WatchProtocolDecoder create() {
        return new WatchProtocolDecoder();
    }
}
