    public void init_noOptions_test() {
        Map<String, String> options = new HashMap<>();
        ConfigurationSource cfg = new ConfigurationSource();
        CssFormatter formatter = new CssFormatter();
        assertEquals(false, formatter.isInitialized());
        formatter.init(options, cfg);
        assertEquals(true, formatter.isInitialized());
    }

    @Test
    public void init_withOptions_test() {

        formatter.init(options, cfg);

        assertTrue(result.contains("\n"));
    }

    public void doFormat_codeWithComments_test() throws IOException {
