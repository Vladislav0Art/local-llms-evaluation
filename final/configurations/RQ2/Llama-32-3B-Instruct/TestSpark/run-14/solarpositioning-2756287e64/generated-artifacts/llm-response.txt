import java.time.ZoneId;
import java.time.ZonedDateTime;

public class ZonedDateTimeGenerator {
    public static String generateDate(int year, int month, int dayOfMonth, int hour, int minute, int second) {
        return ZonedDateTime.of(year, month, dayOfMonth, hour, minute, second, 0, ZoneId.of("UTC"))
                .toString();
    }

    public static String generateLatitude() {
        if (Double.isNaN(Double.parseDouble("37.7749"))) {
            Double latitude = 37.7749;
            return Double.toString(latitude);
        } else {
            return "37.7749";
        }
    }

    public static String generateLongitude() {
        if (Double.isNaN(Double.parseDouble("-122.4194"))) {
            Double longitude = -122.4194;
            return Double.toString(longitude);
        } else {
            return "-122.4194";
        }
    }

    public static void main(String[] args) {
        System.out.println(generateDate(2022, 1, 1, 0, 0, 0));
        System.out.println(generateLatitude());
        System.out.println(generateLongitude());
    }
}
