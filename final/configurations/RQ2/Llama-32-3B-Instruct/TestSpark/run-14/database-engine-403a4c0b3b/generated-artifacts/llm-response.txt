### Implementing a SQL Query Generator

Below is an example implementation of a SQL query generator in Java, along with a demonstration of how to use it.

```java
import java.util.ArrayList;
import java.util.List;

class SQLTerm {
    private String column;
    private String value;

    public SQLTerm(String column, String value) {
        this.column = column;
        this.value = value;
    }

    public String getColumn() {
        return column;
    }

    public void setColumn(String column) {
        this.column = column;
    }

    public String getValue() {
        return value;
    }

    public void setValue(String value) {
        this.value = value;
    }
}

class SQLQueryGenerator {
    private List<SQLTerm> terms;

    public SQLQueryGenerator() {
        this.terms = new ArrayList<>();
    }

    public void addTerm(SQLTerm term) {
        this.terms.add(term);
    }

    public String generateQuery() {
        StringBuilder query = new StringBuilder();

        for (int i = 0; i < terms.size(); i++) {
            if (i > 0) {
                query.append(" AND ");
            }
            query.append(terms.get(i).getColumn()).append(" = '").append(terms.get(i).getValue()).append("'");
        }

        return "SELECT * FROM table_name WHERE " + query.toString();
    }
}

public class Main {
    public static void main(String[] args) {
        SQLQueryGenerator generator = new SQLQueryGenerator();

        SQLTerm term1 = new SQLTerm("column1", "value4");
        SQLTerm term2 = new SQLTerm("column2", "value5");

        generator.addTerm(term1);
        generator.addTerm(term2);

        String query = generator.generateQuery();
        System.out.println(query);
    }
}
```

### Example Usage

To use the `SQLQueryGenerator` class, create an instance of it and add terms using the `addTerm` method. Then, call the `generateQuery` method to generate a SQL query string.

In this example, we create two `SQLTerm` objects representing columns "column1" and "column2" with values "value4" and "value5", respectively. We then pass these terms to the `SQLQueryGenerator` instance to generate a SQL query that selects all records from a table where both conditions are met.

The output of this code will be:


```java
import java.util.List;

    private String column;

        this.column = column;

    }

    public String getColumn() {
        return column;
    }

    public void setColumn(String column) {
        this.column = column;

    }

public class SQLQueryGenerator {
    private List<SQLTerm> terms;

    public SQLQueryGenerator() {
        this.terms = new ArrayList<>();
    }

    public void addTerm(SQLTerm term) {
        this.terms.add(term);
    }

    public String generateQuery(String tableName, boolean useWhereClause) {
        StringBuilder query = new StringBuilder();

        if (useWhereClause) {
            query.append("SELECT * FROM ").append(tableName).append(" WHERE ");
            int conditionCount = 0;

            for (int i = 0; i < terms.size(); i++) {
                if (i > 0) {
                    query.append(" AND ");
                }
                String condition;
                switch (terms.get(i).getColumn()) {
                    case "column1":
                        condition = terms.get(i).getValue();
                        break;
                    default:
                        condition = terms.get(i).getValue() + "=" + terms.get(i).getValue();
                        break;
                }

                query.append(terms.get(i).getColumn()).append(" ").append(condition);
                if (i < terms.size() - 1) {
                    query.append(" AND ");
                }
            }

            query.append(";"); // append a semicolon
        } else {
            query.append("SELECT * FROM ").append(tableName); // only select from table name
        }

        return query.toString();
    }
}

        generator.addTerm(term2);
```

### SQL Query Generator Implementation

Below is an example implementation of a SQL query generator in Java, along with a demonstration of how to use it.

```java
import java.util.ArrayList;
import java.util.List;

public class SQLTerm {
    private String column;
    private String value;

    public SQLTerm(String column, String value) {
        this.column = column;
        this.value = value;
    }

    public String getColumn() {
        return column;
    }

    public void setColumn(String column) {
        this.column = column;
    }

    public String getValue() {
        return value;
    }

    public void setValue(String value) {
        this.value = value;
    }
}

public class SQLQueryGenerator {
    private List<SQLTerm> terms;

    public SQLQueryGenerator() {
        this.terms = new ArrayList<>();
    }

    public void addTerm(SQLTerm term) {
        this.terms.add(term);
    }

    public String generateInsertQuery(String tableName, List<String[]> columns, List<Object[]> values) {
        StringBuilder query = new StringBuilder("INSERT INTO " + tableName + " (");
        for (int i = 0; i < columns.size(); i++) {
            if (i > 0) {
                query.append(", ");
            }
            query.append(columns.get(i)[0]);
        }
        query.append(") VALUES (");

        int valueCount = 1;
        for (Object[] value : values) {
            if (value != null) {
                String columnValue = value[0] + " = '" + value[1].toString() + "'";
                if (valueCount > 1) {
                    query.append(", ");
                }
                query.append(columnValue);
                valueCount++;
            } else {
                query.append("null, ");
            }
        }

        query.delete(query.length() - 2, query.length());
        query.append(");");

        return query.toString();
    }
}

public class Main {
    public static void main(String[] args) {
        SQLQueryGenerator generator = new SQLQueryGenerator();

        String tableName = "my_table";
        List<String[]> columns = new ArrayList<>();
        columns.add(new String[]{"column1", "column2"});
        columns.add(new String[]{"column3", "column4"});

        List<Object[]> values = new ArrayList<>();
        values.add(new Object[] {"value1", "value2"});
        values.add(new Object[] {"value3", "value4"});

        String query = generator.generateInsertQuery(tableName, columns, values);
        System.out.println(query);
    }
}
```

### Example Usage

To use the `SQLQueryGenerator` class, create an instance of it and add terms using the `addTerm` method. Then, call the `generateInsertQuery` method to generate a SQL query string.

In this example, we create two columns "column1" and "column2", and two rows with corresponding values. We then pass these terms to the `SQLQueryGenerator` instance to generate an INSERT INTO SQL query that inserts all records from the table.

The output of this code will be:

```
INSERT INTO my_table (column1, column2) VALUES (value1, value2), (value3, value4);
```
