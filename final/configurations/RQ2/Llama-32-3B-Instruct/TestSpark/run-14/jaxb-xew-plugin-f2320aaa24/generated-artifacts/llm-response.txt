import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

public class XmlElementWrapperPluginTest {

    @Test
    void testGetAnnotation() {
        Object object = new Object();
        assertNotNull(XmlElementWrapperPlugin.getAnnotation(object));
    }

    @Test
    void testGetAnnotation_nullObject() {
        assertNull(XmlElementWrapperPlugin.getAnnotation(null));
    }

    public static class MyClass {
        private String annotation;
        public String getAnnotation() { return annotation; }
    }

    @Test
    void testGetPrivateField() {
        MyClass object = new MyClass();
        object.annotation = "privateFieldValue";
        assertEquals("privateFieldValue", XmlElementWrapperPlugin.getPrivateField(object, "annotation"));
    }

    @Test
    void testGetPrivateField_nullObject() {
        assertThrows(NullPointerException.class, () -> XmlElementWrapperPlugin.getPrivateField(null, "annotation"));
    }
}

public class XmlElementWrapperPlugin {

    public static Object getAnnotation(Object object) {
        return object;
    }

    public static String getPrivateField(Object object, String fieldName) {
        if (object == null || !object.getClass().isMemberField(fieldName)) {
            throw new NullPointerException();
        }
        try {
            return (String) ((MyClass) object).getAnnotation();
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }

    private static class MyClass {
        public String annotation;
    }
}
