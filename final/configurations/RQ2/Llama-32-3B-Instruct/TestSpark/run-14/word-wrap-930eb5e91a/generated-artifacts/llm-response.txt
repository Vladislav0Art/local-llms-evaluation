import java.nio.charset.StandardCharsets;
import java.util.*;

public class WordWrapTest {
    @org.junit.Test
    public void testWordWrap() throws Exception {
        String input = "This is a long string that needs to be wrapped.";
        int width = 10;
        char separator = '\n';
        double extraSpaceFactor = 1.0d;

        String expectedOutput = "This is a long string that needs to be wrapped.\n" +
                "              \n" +
                "              \n";

        WordWrap.wordWrap(input, new StringWriter(), separator, width, (String s) -> {
            double spaceMultiplier = extraSpaceFactor * ((double) s.length() / width);
            return String.format("%" + Integer.toString((int) Math.ceil(spaceMultiplier)) + ".0f").trim();
        }, new HashSet<>(), false, true);

        assertEquals(expectedOutput, new String(WordWrap.wordWrap(input, new StringWriter(), separator, width, (String s) -> {
            double spaceMultiplier = extraSpaceFactor * ((double) s.length() / width);
            return String.format("%" + Integer.toString((int) Math.ceil(spaceMultiplier)) + ".0f").trim();
        }, new HashSet<>(), false, true)));
    }
}

public class WordWrap {
    public static void wordWrap(String input, Writer writer, char separator, int width, Function<String, Double> spaceMultiplierFunc, Set<Character> nonSpaceChars, boolean wrapAfterLastChar, boolean leftTrim) throws IOException {
        StringBuilder output = new StringBuilder();

        if (!leftTrim && !wrapAfterLastChar) {
            output.append(input);
        } else {
            for (int i = 0; i < input.length(); i++) {
                char c = input.charAt(i);

                if (nonSpaceChars.contains(c)) {
                    output.append(c);
                } else if (wrapAfterLastChar && i == input.length() - 1) {
                    output.append(separator);
                }

                if ((i + 1) % width == 0 || i == input.length() - 1) {
                    double multiplier = spaceMultiplierFunc.apply(String.valueOf(output).substring(output.length() - width));
                    output.append(multiplier).append(" ");
                }
            }
        }

        writer.write(output.toString());
    }

        R apply(T t);
    }
