```java
import com.fasterxml.jackson.databind.ObjectMapper;
import java.util.Properties;

public class Property {
    private String key;
    private String value;

    public Property(String key, String value) {
        this.key = key;
        this.value = value;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        Property property = (Property) o;
        return key.equals(property.key) &&
                value.equals(property.value);
    }
}

public class ConfigurationBuilder {
    private Properties properties;

    public ConfigurationBuilder() {
        this.properties = new Properties();
    }

    public void add(Property property) {
        properties.setProperty(property.getKey(), property.getValue());
    }

    public String toJson() {
        try {
            ObjectMapper mapper = new ObjectMapper();
            return mapper.writeValueAsString(properties);
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }

    public static ConfigurationBuilder fromJson(String json) {
        try {
            ObjectMapper mapper = new ObjectMapper();
            Properties properties = mapper.readValue(json, Properties.class);
            return new ConfigurationBuilder().properties(properties);
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }
}

public class Main {
    public static void main(String[] args) {
        ConfigurationBuilder configBuilder = new ConfigurationBuilder();

        Property property1 = new Property("key1", "value1");
        Property property2 = new Property("key2", "value2");

        configBuilder.add(property1);
        configBuilder.add(property2);

        String json = configBuilder.toJson();
        System.out.println(json); // prints: key1=value1, key2=value2

        ConfigurationBuilder configBuilderFromJson = ConfigurationBuilder.fromJson(json);
        Property property3 = new Property("key1", "newValue1");
        configBuilderFromJson.add(property3);

        String jsonFromJson = configBuilderFromJson.toJson();
        System.out.println(jsonFromJson); // prints: key1=newValue1, key2=value2
    }
}
```

In this example, the `ConfigurationBuilder` class provides a way to add properties and serialize them to JSON using the `ObjectMapper`. The `fromJson` method is used to deserialize JSON back into a `Properties` object.
