import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import static org.junit.Assert.*;
import static org.jsoup.internal.Normalizer.lowerCase;

@RunWith(MockitoJUnitRunner.class)
public class SafelistTest {

    @Mock
    private Set<String> setOfTags;

    @Mock
    private Iterator<String> iteratorOfTags;

    public Safelist none() {
        return new Safelist();
    }

    public static final String BASIC_TAG = "basic";
    public static final String SIMPLE_TEXT_TAG = "simpleText";

    @Test
    public void isSafeTag_BasicTag_ReturnsTrue() {
        when(setOfTags).contains(BASIC_TAG);
        assertTrue(Safelist.none().isSafeTag(BASIC_TAG));
    }

    @Test
    public void isSafeTag_SimpleTextTag_ReturnsTrue() {
        when(setOfTags).contains(SIMPLE_TEXT_TAG);
        assertTrue(Safelist.none().isSafeTag(SIMPLE_TEXT_TAG));
    }

    @Test
    public void isSafeTag_SafeTag_ReturnsTrue() {
        when(setOfTags).contains("safe");
        assertTrue(Safelist.none().isSafeTag("safe"));
    }

    @Test
    public void isSafeTag_UnsafeTag_ReturnsFalse() {
        when(setOfTags).not().contains("unsafe");
        assertFalse(Safelist.none().isSafeTag("unsafe"));
    }

    @Test
    public void addTags_AddsToSetOfTags() {
        Safelist safelist = new Safelist();
        safelist.addTags("tag1", "tag2");
        assertTrue(setOfTags.contains("tag1") && setOfTags.contains("tag2"));
    }

    @Test
    public void removeTags_RemovesFromSetOfTags() {
        Safelist safelist = new Safelist();
        safelist.addTags("tag1", "tag2");
        safelist.removeTags("tag1", "tag2");
        assertFalse(setOfTags.contains("tag1") && setOfTags.contains("tag2"));
    }

    @Test
    public void addAttributes_AddsToEnforcedAttributes() {
        Safelist safelist = new Safelist();
        Map<String, String> enforcedAttributes = new HashMap<>();
        safelist.addAttributes("img", "src", "url");
        assertEquals(1, enforcedAttributes.size());
    }

    @Test
    public void removeAttributes_RemovesFromEnforcedAttributes() {
        Safelist safelist = new Safelist();
        safelist.addAttributes("img", "src", "url");
        safelist.removeAttributes("img", "src");
        assertEquals(0, safelist.getEnforcedAttributes("img").size());
    }

    @Test
    public void preserveRelativeLinks_PreserveTrue_ReturnsSelf() {
        Safelist safelist = new Safelist();
        assertTrue(safelist.preserveRelativeLinks(true).equals(safelist));
    }

    @Test
    public void addProtocols_AddsToEnforcedAttributes() {
        Safelist safelist = new Safelist();
        Map<String, String> enforcedAttributes = new HashMap<>();
        safelist.addProtocols("img", "src", "http://example.com");
        assertEquals(1, enforcedAttributes.size());
    }

    @Test
    public void removeProtocols_RemovesFromEnforcedAttributes() {
        Safelist safelist = new Safelist();
        Map<String, String> enforcedAttributes = new HashMap<>();
        safelist.addProtocols("img", "src", "http://example.com");
        safelist.removeProtocols("img", "src", "http://example2.com");
        assertEquals(0, safelist.getEnforcedAttributes("img").size());
    }
}
