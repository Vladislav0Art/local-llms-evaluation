```java
import java.net.URL;
import java.util.ArrayList;
import java.util.List;

public class UrlBuilder {
    private URL baseUrl;

    public UrlBuilder(URL baseUrl) {
        this.baseUrl = baseUrl;
    }

    public UrlBuilder appendKey(String key) {
        if (baseUrl.getQuery() == null) {
            baseUrl.setQuery(key);
        } else {
            List<String> params = new ArrayList<>();
            String[] pairs = baseUrl.getQuery().split("&");
            for (String pair : pairs) {
                String[] keyValue = pair.split("=");
                if (keyValue.length > 1 && keyValue[0].equals(key)) {
                    params.remove(keyValue[0]);
                }
                if (!keyValue[0].equals(key)) {
                    params.add(keyValue[0] + "=" + keyValue[1]);
                }
            }
            baseUrl.setQuery(String.join("&", params));
        }
        return this;
    }

    public UrlBuilder appendValue(String value) {
        String query = baseUrl.getQuery();
        if (query != null && !query.isEmpty()) {
            query += "&";
        } else {
            query = "";
        }
        query += value;
        baseUrl.setQuery(query);
        return this;
    }

    @Override
    public String toString() {
        return baseUrl.toString();
    }
}

public class Main {

    public static void main(String[] args) throws Exception {
        URL baseUrl = new URL("http://example.com");
        
        UrlBuilder builder = new UrlBuilder(baseUrl);
        builder.appendKey("key1").appendValue("value1")
                .appendKey("key2").appendValue("value2")
                .appendKey("key3").appendValue("value3");

        System.out.println(builder.toString());

        // Output: http://example.com?key1=value1&key2=value2&key3=value3
    }
}
```

This `UrlBuilder` class allows you to append key-value pairs to the URL. The `appendKey` method appends a key without a value, while the `appendValue` method appends a key with a value. If the same key is appended multiple times, only the last value is kept.
