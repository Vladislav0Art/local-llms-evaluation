        KeyPair keyPair = KeyPair.fromPublicKey(publicKey.getEncoded());
        assertNotNull(keyPair);
    }

    public void canSign_test() throws GeneralSecurityException {

        assertFalse(privateKey.canSign());

        KeyPair keyPair = KeyPair.fromBip39Seed(bip39Seed, 1);
        assertNotNull(keyPair);
    }

    @Test
    public void fromSecretSeed_string_test() throws IOException, GeneralSecurityException {
        String seed = "secret_seed";
        KeyPair keyPair = KeyPair.fromSecretSeed(seed.getBytes());
        assertNotNull(keyPair);
    }

    @Test
    public void fromSecretSeed_charArray_test() throws IOException, GeneralSecurityException {
        char[] seed = {'s', 'e', 'c', 'r', 'e', 't', 's', 'e', 'e', 'd'};
        KeyPair keyPair = KeyPair.fromSecretSeed(seed);
        assertNotNull(keyPair);
    }

    public void fromAccountId_test() throws GeneralSecurityException {
        String accountId = "account_id";
        KeyPair keyPair = KeyPair.fromAccountId(accountId);
        assertNotNull(keyPair);
    }

    @Test
    public void getAccountId_test() {
        KeyPair keyPair = new KeyPair();
        assertNull(keyPair.getAccountId());
    }

    @Test
    public void getSecretSeed_test() {
        KeyPair keyPair = new KeyPair();
        assertArrayEquals(new char[0], keyPair.getSecretSeed());
    }

    @Test
    public void getPublicKey_test() {
        byte[] publicKey = {1, 2, 3};
        KeyPair keyPair = KeyPair.fromPublicKey(publicKey);
        assertNotNull(keyPair.getPublicKey());
    }

    @Test
    public void sign_test() throws GeneralSecurityException, IOException {
        KeyPair keyPair = new KeyPair();
        byte[] data = new byte[4];
        assertNotNull(keyPair.sign(data));
    }

    @Test
    public void verify_test() throws GeneralSecurityException, IOException {
        KeyPair keyPair = new KeyPair();
        byte[] data = new byte[4];
        byte[] signature = new byte[64];
        assertTrue(keyPair.verify(data, signature));
    }
}
