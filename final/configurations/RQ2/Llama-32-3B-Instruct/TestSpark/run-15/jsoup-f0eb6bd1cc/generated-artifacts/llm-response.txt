import org.jsoup.Connection;
import org.jsoup.helper.DataUtil;
import org.junit.Test;
import org.mockito.Mockito;

public class UrlBuilderTest {

    @Test
    public void newUrlBuilder_withValidInputUrl() {
        UrlBuilder urlBuilder = new UrlBuilder(new URL("http://example.com"));
        Mockito.assertDoesNotThrow(() -> urlBuilder.build());
    }

    @Test
    public void appendKeyVal_toEmptyUrl() throws UnsupportedEncodingException {
        UrlBuilder urlBuilder = new UrlBuilder(new URL(""));
        Connection.KeyVal keyVal = new Connection.KeyVal("key", "val");
        urlBuilder.appendKeyVal(keyVal);
        Mockito.verify(urlBuilder).build();
    }

    @Test
    public void appendKeyVal_withValidKeyVal() throws UnsupportedEncodingException {
        UrlBuilder urlBuilder = new UrlBuilder(new URL("http://example.com"));
        Connection.KeyVal keyVal = new Connection.KeyVal("key", "val");
        urlBuilder.appendKeyVal(keyVal);
        Mockito.verify(urlBuilder).build();
    }

    @Test
    public void appendKeyVal_withInvalidCharacterInValue() throws UnsupportedEncodingException {
        UrlBuilder urlBuilder = new UrlBuilder(new URL("http://example.com"));
        Connection.KeyVal keyVal = new Connection.KeyVal("key", "invalid character!");
        try {
            urlBuilder.appendKeyVal(keyVal);
            assert false;
        } catch (UnsupportedEncodingException e) {
            Mockito.verifyNoThrowingExceptions(urlBuilder, () -> urlBuilder.appendKeyVal(keyVal));
        }
    }

    @Test
    public void newUrlBuilder_fromNullInputUrl() {
        UrlBuilder urlBuilder = new UrlBuilder(null);
        Mockito.assertDoesNotThrow(() -> urlBuilder.build());
    }

    @Test
    public void appendKeyVal_fromNullConnection() {
        UrlBuilder urlBuilder = new UrlBuilder(new URL("http://example.com"));
        Connection.KeyVal keyVal = null;
        try {
            urlBuilder.appendKeyVal(keyVal);
            assert false;
        } catch (NullPointerException e) {
            Mockito.verifyNoThrowingExceptions(urlBuilder, () -> urlBuilder.appendKeyVal(keyVal));
        }
    }

}
