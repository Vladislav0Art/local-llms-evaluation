import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import static org.hamcrest.MatcherAssert.assertThat;
import static org.hamcrest.Matchers.is;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class XmlTreeBuilderTest {

    @Mock
    private Reader inputReader;

    @Mock
    private String baseUri;

    @Mock
    private Parser parser;

    @Test
    public void initialiseParse_WithValidInput_CallsInitializeWithReader() {
        when(inputReader.read()).thenReturn(10);
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        xmlTreeBuilder.initialiseParse(inputReader, baseUri, parser);
        Mockito.verify(parser).initialiseWithReader(inputReader);
    }

    @Test
    public void initialiseParse_WithInvalidInput_ThrowsException() {
        when(inputReader.read()).thenReturn(-1);
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        try {
            xmlTreeBuilder.initialiseParse(inputReader, baseUri, parser);
            assert false;
        } catch (Exception e) {
            assertThat(e.getMessage(), is("Error reading input"));
        }
    }

    @Test
    public void parse_WithValidInput_ReturnsDocument() {
        when(parser.parse(inputReader, baseUri)).thenReturn(new Document());
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        Document document = xmlTreeBuilder.parse(inputReader, baseUri);
        assertThat(document, is(notNullValue()));
    }

    @Test
    public void parse_WithInvalidInput_ReturnsEmptyDocument() {
        when(parser.parse(inputReader, baseUri)).thenReturn(new Document());
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        Document document = xmlTreeBuilder.parse(inputReader, baseUri);
        assertThat(document.getHtml(), is(""));
    }

    @Test
    public void newInstance_ReturnsNewInstance() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        XmlTreeBuilder newXmlTreeBuilder = xmlTreeBuilder.newInstance();
        assertThat(newXmlTreeBuilder, is(notSameInstance(xmlTreeBuilder)));
    }
}
