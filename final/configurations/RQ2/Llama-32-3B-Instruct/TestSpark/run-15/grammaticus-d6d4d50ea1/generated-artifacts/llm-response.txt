class Solution {
    public List<String> findAndReplacePattern(String s, String[] wordlist) {
        Set<String> set = new HashSet<>();
        for (String word : wordlist) {
            set.add(replace(word));
        }

        return match(s, set);
    }

    private String replace(String str) {
        StringBuilder res = new StringBuilder();
        char ch;
        int i;

        for (i = 0; i < str.length(); i++) {
            switch (str.charAt(i)) {
                case 'a' -> res.append('1');
                case 'b' -> res.append('2');
                case 'c' -> res.append('3');
                default -> throw new RuntimeException("Bad character");
            }
        }

        return res.toString();
    }

    private List<String> match(String s, Set<String> set) {
        List<String> result = new ArrayList<>();
        for (String word : set) {
            if (match(s, word)) {
                result.add(word);
            }
        }
        return result;
    }

    private boolean match(String s, String w) {
        int[] a = new int[26];
        int i = 0;

        while (i < s.length() && i < w.length()) {
            if (a[s.charAt(i) - 'a'] == w.charAt(i) - '1') {
                ++i;
            } else {
                return false;
            }
        }

        return i == s.length();
    }
}
