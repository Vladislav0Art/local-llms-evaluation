import java.net.URI;

    private String query;

    public UrlBuilder(String url) throws URISyntaxException {
        URI uri = new URI(url);
        this.scheme = uri.getScheme();
        this.netloc = uri.getHost();
        this.path = uri.getPath();
        this.query = uri.getQuery();
    }

    public void setPath(String path) {
        this.path = path;
    }

    public String getScheme() {
        return scheme;
    }

    public String getNetloc() {
        return netloc;
    }

    public String getPath() {
        return path;
    }

    public String getQuery() {
        return query;
    }

    @Override
    public String toString() {
        return scheme + "://" + netloc + path;
    }
}

        } else {
            query = key + "=" + value;
        }
    }

    public UrlBuilder withQuery(String query) {
        this.query = query;
        return this;
    }

    public String getScheme() {
        return scheme;
    }

        return netloc;
    }
