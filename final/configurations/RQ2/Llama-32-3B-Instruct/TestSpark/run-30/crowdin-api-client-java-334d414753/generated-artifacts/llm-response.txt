@RunWith(MockitoJUnitRunner.class)

    private ObjectMapper mapper;

    @Mock
    private SimpleModule module;

    @Test
    public void parse.JsonParsingSuccess() {
        when(mapper.findAndRegisterModules()).thenReturn(module);
        when(mapper.convertValue(any(), any())).thenReturn("JSON DATA");
        String json = "{\"key\":\"value\"}";
        Class<Project> clazz = Project.class;
        Project result = JacksonJsonTransformer.parse(json, clazz);
        assertEquals(result, new Project());
    }

    @Test
    public void parse.JsonParsingException() {
        when(mapper.findAndRegisterModules()).thenReturn(module);
        when(mapper.convertValue(any(), any())).thenThrow(new RuntimeException("EXCEPTION"));
        String json = "{\"key\":\"value\"}";
        Class<Project> clazz = Project.class;
        try {
            JacksonJsonTransformer.parse(json, clazz);
            assert false;
        } catch (RuntimeException e) {
            assertEquals(e.getMessage(), "EXCEPTION");
        }
    }

    @Test
    public void convert.JsonConversionSuccess() {
        when(mapper.findAndRegisterModules()).thenReturn(module);
        Class<Project> clazz = Project.class;
        Project obj = new Project();

        assertTrue(result.startsWith("{\"key\":\"value\"}"));
    }

    public void convert.JsonConversionException() {
        when(mapper.findAndRegisterModules()).thenReturn(module);
