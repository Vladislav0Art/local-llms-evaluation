import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertFalse;
import static org.junit.Assert_NULL;
import static org.junit.Assert.assertTrue;
import static org.jsoup.helper.Validate.isValid;

@RunWith(MockitoJUnitRunner.class)
public class TextNodeTest {

    @Test
    public void textNode_Ctor() {
        TextNode node = new TextNode("Hello World");
        assertEquals("Hello World", node.text());
    }

    @Test
    public void nodeName_TextNode() {
        TextNode node = new TextNode("Hello World");
        assertEquals("text-node", node.nodeName());
    }

    @Test
    public void isBlank_EmptyText() {
        TextNode node = new TextNode("");
        assertTrue(node.isBlank());
    }

    @Test
    public void isBlank_HasText() {
        TextNode node = new TextNode("Hello World");
        assertFalse(node.isBlank());
    }

    @Test
    public void clone_TextNode_clone() {
        TextNode original = new TextNode("Hello World");
        TextNode cloned = original.clone();
        assertEquals(original, cloned);
        assertNotNull(cloned);
    }

    @Test
    public void splitText_SplitTextNoOffset() {
        TextNode node = new TextNode("Hello World");
        TextNode leftChild = node.splitText(0);
        assertEquals(node, leftChild);
    }

    @Test
    public void splitText_SplitTextNonZeroOffset() {
        TextNode node = new TextNode("Hello World");
        TextNode leftChild = node.splitText(5);
        assertNotNull(leftChild);
        assertTrue(leftChild.text().isEmpty());
    }

    @Test
    public void getWholeText_NoChildren() {
        TextNode root = new TextNode("Hello World");
        assertEquals("Hello World", root.getWholeText());
    }

    @Test
    public void outerHtmlHead_NoAppendable() {
        Appendable appendable = Mockito.mock(Appendable.class);
        Document document = new Document();
        TextNode textNode = new TextNode("");
        textNode.outerHtmlHead(appendable, 0, document.outputSettings);
        assertNull(appendable.toString());
    }

    @Test
    public void outerHtmlTail_NoAppendable() {
        Appendable appendable = Mockito.mock(Appendable.class);
        Document document = new Document();
        TextNode textNode = new TextNode("");
        textNode.outerHtmlTail(appendable, 0, document.outputSettings);
        assertNull(appendable.toString());
    }

    @Test
    public void normaliseWhitespace_AngleBrackets() {
        String input = "<Hello> World";
        assertEquals("Hello World", StringUtil.normaliseWhitespace(input));
    }

    @Test
    public void stripLeadingWhitespace_NoLeadingWhitespace() {
        String input = "Hello World";
        assertEquals("Hello World", StringUtil.stripLeadingWhitespace(input));
    }

    @Test
    public void stripLeadingWhitespace_WithLeadingWhitespace() {
        String input = "<Hello>   World";
        assertEquals("Hello>   World", StringUtil.stripLeadingWhitespace(input));
    }

    @Test
    public void lastCharIsWhitespace_EmptyString() {
        StringBuilder builder = Mockito.mock(StringBuilder.class);
        assertFalse(TextNode.lastCharIsWhitespace(builder));
    }

    @Test
    public void lastCharIsWhitespace_NonWhitespaceCharacter() {
        StringBuilder builder = Mockito.mock(StringBuilder.class);
        when(builder.toString()).thenReturn("Hello");
        assertFalse(TextNode.lastCharIsWhitespace(builder));
    }

    @Test
    public void lastCharIsWhitespace_WhitespaceCharacter() {
        StringBuilder builder = Mockito.mock(StringBuilder.class);
        when(builder.toString()).thenReturn("Hello ");
        assertTrue(TextNode.lastCharIsWhitespace(builder));
    }
}
