import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;
import org.jsoup.helper.Validate;
import org.jsoup.nodes.CDataNode;
import org.jsoup.nodes.Comment;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.DocumentType;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.TextNode;
import org.jsoup.nodes.XmlDeclaration;
import org.jsoup.parser.Parser;
import org.jsoup.parser.Token;
import org.jsoup.parser.TreeBuilder;
import org.jsoup.nodes.Node;

import static org.junit.Assert.*;
import static org.mockito.Mockito.*;

@RunWith(MockitoJUnitRunner.class)
public class XmlTreeBuilderTest {

    @Mock
    private Parser parser;

    private XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();

    @Test
    public void initializeParse_WithReader_andBaseUri_andParser_success() {
        xmlTreeBuilder.initialiseParse(new StringReader("<root></root>"), "https://example.com", parser);
        verify(parser).initialiseParse(eq(new StringReader("<root></root>")), eq("https://example.com"));
    }

    @Test
    public void initializeParse_WithEmptyString_andBaseUri_andParser_throwsNullPointerException() {
        assertThrows(NullPointerException.class, () -> xmlTreeBuilder.initialiseParse(new StringReader(""), "https://example.com", parser));
    }

    @Test
    public void parse_WithReader_andBaseUri_success() {
        Document document = xmlTreeBuilder.parse(new StringReader("<root></root>"), "https://example.com");
        assertNotNull(document);
    }

    @Test
    public void parse_WithEmptyString_andBaseUri_throwsNullPointerException() {
        assertThrows(NullPointerException.class, () -> xmlTreeBuilder.parse(new StringReader(""), "https://example.com"));
    }

    @Test
    public void parse_WithInvalidXmlInput_throwsParseException() {
        assertThrows(ParseException.class, () -> xmlTreeBuilder.parse("<root>", "https://example.com"));
    }

    @Test
    public void newInstance_success() {
        XmlTreeBuilder newXmlTreeBuilder = xmlTreeBuilder.newInstance();
        assertNotNull(newXmlTreeBuilder);
    }

    @Test
    public void process_WithStartTag_andParser_success() {
        assertTrue(xmlTreeBuilder.process(new Token.StartTag()));
    }

    @Test
    public void process_WithEndTag_andParser_success() {
        assertTrue(xmlTreeBuilder.process(new Token.EndTag()));
    }

    @Test
    public void insertNode_WithNode_success() {
        xmlTreeBuilder.insertNode(new Element());
        verify(parser).insertNode(eq(Validate.nodes()));
    }

    @Test
    public void insertNode_WithNode_andToken_success() {
        xmlTreeBuilder.insertNode(new Element(), new Token.StartTag());
        verify(parser).insertNode(eq(Validate.nodes()), eq(new Token.StartTag()));
    }

    @Test
    public void insertElement_WithStartTag_success() {
        Element element = xmlTreeBuilder.insert(new Token.StartTag());
        assertNotNull(element);
    }

    @Test
    public void insertCharacter_WithToken_success() {
        xmlTreeBuilder.insert(new Token.Character('a'));
        verify(parser).insert(eq(Validate.chars()), eq('a'));
    }

    @Test
    public void insertComment_WithCommentToken_success() {
        xmlTreeBuilder.insert(new Token.Comment(""));
        verify(parser).insert(eq(Validate.comments()), eq(""));
    }

    @Test
    public void insertDoctype_WithDoctype_success() {
        xmlTreeBuilder.insert(new Token.Doctype());
        verify(parser).insert(eq(Validate.doctypes()), eq(""));
    }

    @Test
    public void popStackToClose_WithEndTag_andParser_success() {
        xmlTreeBuilder.popStackToClose(new Token.EndTag());
        verify(parser).popStackToClose(eq(new Token.EndTag()));
    }

    @Test
    public void parseFragment_WithInputFragment_andBaseUri_andParser_success() {
        List<Node> nodes = xmlTreeBuilder.parseFragment("inputFragment", "baseUri", parser);
        assertNotNull(nodes);
    }

    @Test
    public void parseFragment_WithEmptyInputFragment_andBaseUri_andParser_throwsNullPointerException() {
        assertThrows(NullPointerException.class, () -> xmlTreeBuilder.parseFragment("", "baseUri", parser));
    }
}
