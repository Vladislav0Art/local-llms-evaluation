public class EpubChecker {
    public String getLocale() {
        return Locale.getDefault();
    }

    public int run(String[] args) {
        EpubCheck check = new EpubCheck(new EPUBProfile());
        check.processEpubFile(args);
        return 0;
    }

    @Deprecated
    public int processEpubFile(String[] args) {
        try {
            File file = new File(args[0]);
            if (file.exists()) {
                String command = "mola -e:validate " + args[0];
                Process process = Runtime.getRuntime().exec(command);
                return process.waitFor();
            } else {
                throw new GalimatiasParseException("File does not exist");
            }
        } catch (IOException | InterruptedException e) {
            throw new RuntimeException(e);
        }
    }

    public int validateFile(String path, EPUBVersion version, Report report, EPUBProfile profile) {
        if (!version.equals(EPubVersion.NA)) {
            try (Checker checker = new EpubCheckFactory().createCheck(path, profile)) {
                check = checker;
                return 0;
            } catch (InvalidVersionException e) {
                throw new RuntimeException(e);
            }
        } else {
            return 1; // ValidateEpubFile
        }
    }

}

public class EpubCheckerTest {
    @Test
    public void testGetLocale() {
        EpubChecker checker = new EpubChecker();
        assertEquals(Locale.getDefault(), checker.getLocale());
    }

    @Test
    public void testRun() throws InterruptedException {
        EpubChecker checker = new EpubChecker();
        Thread thread = new Thread(() -> {
            checker.run(new String[] {"-p", "en_US"));
        });
        thread.start();
        thread.join();
        assertEquals(0, checker.getLocale().getCountry());
    }

    @Test
    public void testProcessEpubFile() throws IOException, InterruptedException {
        EpubChecker checker = new EpubChecker();
        File file = new File("test.epub");
        check.check(file);
        assertEquals(1, checker.processEpubFile(new String[] {"-p", "en_US"}));
    }

    @Test
    public void testValidateFile() throws InterruptedException {
        EpubChecker checker = new EpubChecker();
        Thread thread = new Thread(() -> {
            int result = checker.validateFile("test.epub", EPUBVersion.NA, null, null);
            System.out.println(result);
        });
        thread.start();
        thread.join();
    }
}
