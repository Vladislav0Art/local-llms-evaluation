import org.jsoup.Connection;
import org.jsoup.helper.DataUtil;
import org.jsoup.helper.HttpConnection;
import org.jsoup.helper.RequestUtils;
import org.jsoup.helper.ResponseBuilder;

public class HttpConnectionTest {

    public static void testConnect(String url) {
        Connection connection = new HttpConnection();
        if (connection.connect(url).isSuccess()) {
            System.out.println("Connected to " + url);
        } else {
            throw new RuntimeException("Failed to connect to " + url);
        }
    }

    public static void testRequestMethod(Method method) {
        HttpConnection request = new HttpConnection();
        Connection connection = request.newRequest().method(method).execute();
        System.out.println("Executing method " + method);
        if (connection.response().isSuccess()) {
            System.out.println("Received response from server");
        } else {
            throw new RuntimeException("Failed to execute method on server: " + connection.response());
        }
    }

    public static void testUrl(String url) {
        HttpConnection request = new HttpConnection();
        Request requestRequest = new Request(url);
        Connection connection = request.newRequest().url(requestRequest).execute();
        System.out.println("Received response from server using " + url);
    }

    public static void testProxy(String host, int port) {
        HttpConnection request = new HttpConnection();
        Proxy proxy = new Proxy(Proxy.Type.HTTP, new InetSocketAddress(host, port));
        Connection connection = request.proxy(proxy).execute();
        System.out.println("Received response from server using proxy");
    }

    public static void testUserAgent(String userAgent) {
        HttpConnection request = new HttpConnection();
        Request requestRequest = new Request(url());
        ResponseBuilder responseBuilder = new ResponseBuilder();
        Response response = request.request().userAgent(userAgent).build(responseBuilder);
        System.out.println("Received response from server using user agent");
    }

    public static void testTimeout(int millis) {
        HttpConnection request = new HttpConnection();
        Connection connection = request.timeout(millis).execute();
        if (connection.response().isSuccess()) {
            System.out.println("Received response from server within " + millis + "ms");
        } else {
            throw new RuntimeException("Failed to receive response from server within " + millis + "ms");
        }
    }

    public static void testMaxBodySize(int bytes) {
        HttpConnection request = new HttpConnection();
        Request requestRequest = new Request(url());
        Connection connection = request.newRequest().maxBodySize(bytes).execute();
        System.out.println("Received response from server with maximum body size of " + bytes + "bytes");
    }

    public static void testFollowRedirects(boolean followRedirects) {
        HttpConnection request = new HttpConnection();
        Request requestRequest = new Request(url());
        Connection connection = request.newRequest().followRedirects(followRedirects).execute();
        System.out.println("Received response from server with " + (followRedirects ? "redirect" : "") + " redirects");
    }

    public static void testIgnoreHttpErrors(boolean ignoreHttpErrors) {
        HttpConnection request = new HttpConnection();
        Request requestRequest = new Request(url());
        Connection connection = request.ignoreHttpErrors(ignoreHttpErrors).execute();
        if (connection.response().isSuccess()) {
            System.out.println("Received response from server without HTTP errors");
        } else {
            throw new RuntimeException("Failed to receive response from server with " + ignoreHttpErrors ? "HTTP" : "non-HTTP" + " errors");
        }
    }

    public static void testIgnoreContentType(boolean ignoreContentType) {
        HttpConnection request = new HttpConnection();
        Request requestRequest = new Request(url());
        Connection connection = request.ignoreContentType(ignoreContentType).execute();
        System.out.println("Received response from server with " + (ignoreContentType ? "" : "not") + " content type");
    }

    public static void testData(String key, String value) {
        HttpConnection request = new HttpConnection();
        Request requestRequest = new Request(url());
        DataUtil.put(requestRequest, key, value);
        Connection connection = request.data(key, value).execute();
        System.out.println("Received response from server with " + key + "=" + value);
    }

    public static void testData(String... keyvals) {
        HttpConnection request = new HttpConnection();
        Request requestRequest = new Request(url());
        DataUtil.put(requestRequest, keyvals[0], keyvals[1]);
        Connection connection = request.data(keyvals).execute();
        System.out.println("Received response from server with " + keyvals);
    }

    public static void testDataCharset(String charset) {
        HttpConnection request = new HttpConnection();
        Request requestRequest = new Request(url());
        DataUtil.put(requestRequest, "key", "value");
        ResponseBuilder responseBuilder = new ResponseBuilder();
        Response response = request.request().dataCharset(charset).build(responseBuilder);
        System.out.println("Received response from server with charset " + charset);
    }

    public static void testMethod(Method method) {
        HttpConnection request = new HttpConnection();
        Request requestRequest = new Request(url());
        Connection connection = request.method(method).execute();
        System.out.println("Executing method " + method);
        if (connection.response().isSuccess()) {
            System.out.println("Received response from server");
        } else {
            throw new RuntimeException("Failed to execute method on server: " + connection.response());
        }
    }

    public static void testGet(String url) {
        HttpConnection request = new HttpConnection();
        Request requestRequest = new Request(url);
        DataUtil.put(requestRequest, null, null);
        Connection connection = request.get().execute();
        System.out.println("Received response from server using GET");
    }

    public static void testPost(String url) {
        HttpConnection request = new HttpConnection();
        Request requestRequest = new Request(url);
        DataUtil.put(requestRequest, "key", "value");
        Connection connection = request.post().execute();
        System.out.println("Received response from server using POST");
    }

    public static void main(String[] args) {
        testConnect("http://example.com");
        testRequestMethod(Method.GET);
        testUrl("http://example.com");
        testProxy("192.168.1.100", 8080);
        testUserAgent("Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/58.0.3029.110 Safari/537.3");
        testTimeout(30000);
        testMaxBodySize(1000000);
        testFollowRedirects(true);
        testIgnoreHttpErrors(false);
        testIgnoreContentType(false);
        testData("key", "value");
        testData("key1", "value1");
        testDataCharset("utf-8");
    }
}
