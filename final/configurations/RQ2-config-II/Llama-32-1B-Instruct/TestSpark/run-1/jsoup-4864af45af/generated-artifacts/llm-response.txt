import org.jsoup.nodes.Attribute;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.Node;
import org.jsoup.select.Elements;

public class SafelistTest {

    @Test
    public void testNone() {
        Safelist none = new Safelist();
        Assert.assertTrue(none.isSafeTag("a"));
        Assert.assertFalse(none.isSafeTag("script"));
    }

    @Test
    public void testSimpleText() {
        Safelist simpleText = new Safelist().simpleText();
        Assert.assertTrue(simpleText.isSafeAttribute("href", "http://example.com"));
        Assert.assertFalse(simpleText.isSafeAttribute("href", "https://example.com"));
    }

    @Test
    public void testBasic() {
        Safelist basic = new Safelist().basic();
        Assert.assertTrue(basic.isSafeTag("a"));
        Assert.assertTrue(basic.isSafeAttribute("href", "http://example.com"));
        Assert.assertFalse(basic.isSafeAttribute("href", "https://example.com"));
    }

    @Test
    public void testBasicWithImages() {
        Safelist basic = new Safelist().basicWithImages();
        Assert.assertTrue(basic.isSafeTag("img"));
        Assert.assertTrue(basic.isSafeAttribute("src", "http://example.com"));
        Assert.assertFalse(basic.isSafeAttribute("src", "https://example.com"));
    }

    @Test
    public void testRelaxed() {
        Safelist relaxed = new Safelist().relaxed();
        Assert.assertTrue(relaxed.isSafeTag("a"));
        Assert.assertFalse(relaxed.isSafeTag("script"));
        Assert.assertTrue(relaxed.isSafeAttribute("href", "http://example.com"));
        Assert.assertFalse(relaxed.isSafeAttribute("href", "https://example.com"));
    }

    @Test
    public void testPreserveRelativeLinks() {
        Safelist safelist = new Safelist().preserveRelativeLinks(true);
        Element elem = new Element("a");
        elem.attr("href", "http://example.com");
        Assert.assertTrue(safelist.isSafeTag("a"));
        Assert.assertFalse(safelist.isSafeTag("script"));
    }

    @Test
    public void testAddTags() {
        Safelist safelist = new Safelist();
        safelist.addTags("span", "class='new-class'");
        Assert.assertTrue(safelist.isSafeAttribute("href", "http://example.com"));
        Assert.assertFalse(safelist.isSafeAttribute("href", "https://example.com"));
    }

    @Test
    public void testRemoveTags() {
        Safelist safelist = new Safelist();
        safelist.removeTags("span");
        Assert.assertTrue(safelist.isSafeAttribute("href", "http://example.com"));
        Assert.assertFalse(safelist.isSafeAttribute("href", "https://example.com"));
    }

    @Test
    public void testAddAttributes() {
        Safelist safelist = new Safelist();
        safelist.addAttributes("span", "style='color: blue'");
        Assert.assertTrue(safelist.getEnforcedAttributes("a").get(0).getValue());
        Assert.assertFalse(safelist.getEnforcedAttributes("href").get(0).getValue());
    }

    @Test
    public void testRemoveAttributes() {
        Safelist safelist = new Safelist();
        safelist.removeAttributes("span", "style='color: blue'");
        Assert.assertTrue(safelist.getEnforcedAttributes("a").isEmpty());
        Assert.assertFalse(safelist.getEnforcedAttributes("href").isEmpty());
    }

    @Test
    public void testAddEnforcedAttribute() {
        Safelist safelist = new Safelist();
        safelist.addEnforcedAttribute("a", "class='new-class'", "value='old-value'");
        Element elem = new Element("a");
        elem.attr("class", "new-class");
        Assert.assertTrue(safelist.isSafeTag("a"));
        Assert.assertFalse(safelist.isSafeTag("script"));
    }

    @Test
    public void testRemoveEnforcedAttribute() {
        Safelist safelist = new Safelist();
        safelist.removeEnforcedAttribute("href", "http://example.com");
        Element elem = new Element("a");
        elem.attr("href", "https://example.com");
        Assert.assertTrue(safelist.isSafeTag("a"));
        Assert.assertFalse(safelist.isSafeTag("script"));
    }

    @Test
    public void testPreserveRelativeLinksWithProtocols() {
        Safelist safelist = new Safelist().preserveRelativeLinks(true);
        Element elem = new Element("a");
        elem.attr("href", "http://example.com");
        Assert.assertTrue(safelist.isSafeTag("a"));
        Assert.assertFalse(safelist.isSafeTag("script"));
    }

    @Test
    public void testAddProtocols() {
        Safelist safelist = new Safelist();
        safelist.addProtocols("a", "rel='stylesheet'");
        Assert.assertTrue(safelist.isSafeAttribute("href", "http://example.com"));
        Assert.assertFalse(safelist.isSafeAttribute("href", "https://example.com"));
    }

    @Test
    public void testRemoveProtocols() {
        Safelist safelist = new Safelist();
        safelist.removeProtocols("a", "rel='stylesheet'");
        Assert.assertTrue(safelist.getEnforcedAttributes("a").isEmpty());
    }
}
