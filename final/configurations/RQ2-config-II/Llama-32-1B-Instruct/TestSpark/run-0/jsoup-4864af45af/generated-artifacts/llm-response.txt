import org.jsoup.safety.Safelist;
import org.jsoup.helper.Validate;
import org.jsoup.nodes.Attribute;
import org.jsoup.nodes.Attributes;
import org.jsoup.nodes.Element;
import java.util.HashMap;
import java.util.HashSet;
import java.util.Iterator;
import java.util.Map;
import java.util.Set;

public class SafelistTest {

    public static void testNone() {
        Safelist safelist = Safelist.none();
        Assert.assertTrue(safelist == Safelist.none());
    }

    public static void testSimpleText() {
        Safelist safelist = Safelist.simpleText();
        Assert.assertTrue(safelist.isSafeTag("text"));
        Assert.assertTrue(safelist.isSafeAttribute("a", "href", "#"));
        Safelist copiedSafelist = new Safelist(safelist);
        Assert.assertFalse(copiedSafelist.isSafeTag("img"));
    }

    public static void testBasic() {
        Safelist safelist = Safelist.basic();
        Assert.assertTrue(safelist.isSafeTag("a"));
        Assert.assertTrue(safelist.getEnforcedAttributes("img") != null);
        Safelist copiedSafelist = new Safelist(safelist);
        Assert.assertFalse(copiedSafelist.isSafeTag("span"));
    }

    public static void testBasicWithImages() {
        Safelist safelist = Safelist.basicWithImages();
        Assert.assertTrue(safelist.isSafeTag("img"));
        Assert.assertTrue(safelist.getEnforcedAttributes("a") != null);
        Safelist copiedSafelist = new Safelist(safelist);
        Assert.assertFalse(copiedSafelist.isSafeTag("span"));
    }

    public static void testRelaxed() {
        Safelist safelist = Safelist.relaxed();
        Assert.assertTrue(safelist.isSafeTag("img"));
        Assert.assertTrue(safelist.getEnforcedAttributes("a") != null);
        Safelist copiedSafelist = new Safelist(safelist);
        Assert.assertFalse(copiedSafelist.isSafeTag("span"));
    }

    public static void testAddTags() {
        Safelist safelist = Safelist.simpleText();
        safelist.addTags("b", "strong");
        Assert.assertTrue(safelist.getEnforcedAttributes("img") != null);
    }

    public static void testRemoveTags() {
        Safelist safelist = Safelist.basic();
        safelist.removeTags("a");
        Safelist copiedSafelist = new Safelist(safelist);
        Assert.assertFalse(copiedSafelist.isSafeTag("span"));
    }

    public static void testAddAttributes() {
        Safelist safelist = Safelist.relaxed();
        safelist.addAttributes("style", "width:100px;");
        Safelist copiedSafelist = new Safelist(safelist);
        Assert.assertFalse(copiedSafelist.isSafeAttribute("img", "src", "#"));
    }

    public static void testRemoveAttributes() {
        Safelist safelist = Safelist.basic();
        safelist.removeAttributes("style");
        Safelist copiedSafelist = new Safelist(safelist);
        Assert.assertTrue(copiedSafelist.isSafeAttribute("a", "href", "#"));
    }

    public static void testAddEnforcedAttribute() {
        Safelist safelist = Safelist.relaxed();
        safelist.addEnforcedAttribute("a", "href", "#");
        Safelist copiedSafelist = new Safelist(safelist);
        Assert.assertTrue(copiedSafelist.isSafeTag("img"));
    }

    public static void testRemoveEnforcedAttribute() {
        Safelist safelist = Safelist.relaxed();
        safelist.removeEnforcedAttribute("a", "href");
        Safelist copiedSafelist = new Safelist(safelist);
        Assert.assertFalse(copiedSafelist.isSafeTag("img"));
    }

    public static void testPreserveRelativeLinks() {
        Safelist safelist = Safelist.relaxed();
        safelist.preserveRelativeLinks(true);
        Safelist copiedSafelist = new Safelist(safelist);
        Assert.assertTrue(copiedSafelist.isSafeTag("a"));
        Assert.assertTrue(copiedSafelist.getEnforcedAttributes("img") != null);
    }

    public static void testAddProtocols() {
        Safelist safelist = Safelist.basic();
        safelist.addProtocols("a", "href", "#");
        Safelist copiedSafelist = new Safelist(safelist);
        Assert.assertFalse(copiedSafelist.isSafeTag("span"));
    }

    public static void testRemoveProtocols() {
        Safelist safelist = Safelist.basic();
        safelist.removeProtocols("a", "href", "#");
        Safelist copiedSafelist = new Safelist(safelist);
        Assert.assertTrue(copiedSafelist.isSafeTag("img"));
    }
}
