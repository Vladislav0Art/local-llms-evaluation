public class CommentTest {

    private Comment comment;

    @Before
    public void setup() {
        ParseSettings settings = new ParseSettings();
        settings.setAllowCommentTags(true);
        comment = new Comment("<!-- this is a sample comment -->");
    }

    public void testIsXmlDeclaration() {
        assertTrue(comment.isXmlDeclaration());
    }

    public void testAsXmlDeclaration() {
        String xmlDeclaration = comment.asXmlDeclaration();
        assertTrue(xmlDeclaration != null && !xmlDeclaration.isEmpty());
    }

    public void testToString() {
        String expectedString = "<!-- this is a sample comment -->";
        assertEquals(expectedString, comment.toString());
    }

    public void testClone() {
        Comment cloneComment = (Comment) comment.clone();
        assertEquals(comment, cloneComment);
    }
}
