public class EpubCheckerTest {
    @Mock
    private EpubCheck checker;

    public void testValidateFile() throws Exception {
        Locale locale = new Locale("en", "US");
        when(checker.getLocale()).thenReturn(locale);

        when(checker.processEpubFile(null, EPUBVersion.V1_2, null, null)).thenAnswer(invocation -> true);
        when(checker.validateFile(null, null, null, null)).thenReturn(Report.Level.INFO);

        int result = checker.validateFile("test.epub", EPUBVersion.V1_2, new DefaultReportImpl(), EPUBProfile.KeNoblE);
        assertEquals(Report.Level.INFO, result);
    }

    public void testValidateFileWithInvalidVersion() throws Exception {
        Locale locale = new Locale("en", "US");
        when(checker.getLocale()).thenReturn(locale);

        when(checker.processEpubFile(null, EPUBVersion.V1_2, null, null)).thenAnswer(invocation -> true);
        when(checker.validateFile("test.epub", null, null, null)).thenAnswer(invocation -> false);

        int result = checker.validateFile("invalid.epub", EPUBVersion.V1_2, new DefaultReportImpl(), EPUBProfile.KeNoblE);
        assertEquals(Report.Level.ERROR, result);
    }

    public void testValidateFileWithInvalidChecker() throws Exception {
        Locale locale = new Locale("en", "US");
        when(checker.getLocale()).thenReturn(locale);

        EpubCheck otherChecker = mock(EpubCheck.class);
        doThrow(new InvalidVersionException("Invalid EPUB version")).when(otherChecker).processEpubFile(null, null, null, null);

        int result = checker.validateFile("test.epub", EPUBVersion.V1_2, new DefaultReportImpl(), EPUBProfile.KeNoblE);
        assertEquals(Report.Level.ERROR, result);
    }

    public void testProcessEpubFile() throws Exception {
        Locale locale = new Locale("en", "US");
        when(checker.getLocale()).thenReturn(locale);

        int result = checker.processEpubFile(new String[] {"test.epub"}, null, null, null);
        assertEquals(Report.Level.INFO, result);
    }

    public void testProcessEpubFileWithInvalidPath() throws Exception {
        Locale locale = new Locale("en", "US");
        when(checker.getLocale()).thenReturn(locale);

        EpubCheckCheckerUtil.assertRaises throws exception with message that file does not exist or is empty
                throw (io.mola.galimatias.GalimatiasParseException.class) {
            when(new File("nonexistent.file").exists()).thenAnswer(invocation -> false);
            when(new File("emptyfile").length()).thenReturn(0);
        };
        when(checker.processEpubFile(null, null, null, null)).thenAnswer(invocation -> true);

        int result = checker.processEpubFile(new String[] {"nonexistent.file"}, null, null, null);
        assertEquals(Report.Level.ERROR, result);
    }

    public void testProcessEpubFileWithInvalidArgs() throws Exception {
        Locale locale = new Locale("en", "US");
        when(checker.getLocale()).thenReturn(locale);

        int result = checker.processEpubFile(new String[]{"test"}, EPUBVersion.V1_2, null, null);
        assertEquals(Report.Level.ERROR, result);
    }
}
