public class XmlTreeBuilderTest {

        xmlTreeBuilder = new XmlTreeBuilder();
        parser = Parser.xmlParser();
        baseUri = "http://example.com/";
    }

    @Test
    public void initialiseParseTest() {
        StringReader input = new StringReader("<root></root>");
        xmlTreeBuilder.initialiseParse(input, baseUri, parser);

    }

    @Test
    public void parseReaderAndStringTest() {
        StringReader input = new StringReader("<root></root>");
        // test that parse correctly returns a Document
        assertNotNull(xmlTreeBuilder.parse(input, baseUri));
    }

    public void parseTwoStringsTest() {
        String input = "<root></root>";
        // test that parse correctly returns a Document
        assertNotNull(xmlTreeBuilder.parse(input, baseUri));
    }

    public void newInstanceTest() {
        assertNotNull(xmlTreeBuilder.newInstance());
    }

    public void processTokenTest() {
        Token.StartTag token = new Token.StartTag();
