import org.jsoup.Connection.Method;
import org.jsoup.nodes.Document;
import org.junit.Test;
import javax.net.ssl.SSLSocketFactory;
import java.io.IOException;
import java.io.InputStream;
import java.net.Proxy;
import java.net.URL;
import java.util.HashMap;
import java.util.Map;
import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNotNull;

public class HttpConnectionTest {

  @Test
  public void connectUrlTest() {
    Connection connection = HttpConnection.connect("http://example.com");
    assertNotNull(connection);
  }

  @Test(expected = IllegalArgumentException.class)
  public void connectInvalidUrlTest() {
    HttpConnection.connect("invalid Url");
  }

  @Test
  public void connectStringTest() throws MalformedURLException {
    Connection connection = HttpConnection.connect(new URL("http://example.com")); 
    assertNotNull(connection);
  }

  @Test
  public void urlTest() {
    Connection connection = HttpConnection.connect("http://example.com").url("http://changed.com");
    assertEquals("http://changed.com", connection.request().url().toString());
  }

  @Test(expected = IllegalArgumentException.class)
  public void urlInvalidTest() {
    HttpConnection.connect("http://example.com").url("invalid Url");
  }

  @Test
  public void userAgentTest() {
    Connection connection = HttpConnection.connect("http://example.com").userAgent("User-Agent");
    assertEquals("User-Agent", connection.request().userAgent());
  }

  @Test
  public void timeoutTest() {
    Connection connection = HttpConnection.connect("http://example.com").timeout(5000);
    assertEquals(5000, connection.request().timeout());
  }

  @Test
  public void followRedirectsTest() {
    Connection connection = HttpConnection.connect("http://example.com").followRedirects(false);
    assertEquals(false, connection.request().followRedirects());
  }

  @Test
  public void postDataCharsetTest() {
    HttpConnection httpConnection = new HttpConnection();
    httpConnection = httpConnection.postDataCharset("UTF-8");
    assertEquals("UTF-8", httpConnection.request().postDataCharset());
  }

  @Test
  public void newRequestTest() {
    HttpConnection httpConnection = new HttpConnection();
    assertNotNull(httpConnection.newRequest());
  }

  @Test
  public void executeTest() throws IOException {
    HttpConnection httpConnection = new HttpConnection();
    assertNotNull(httpConnection.execute());
  }

  @Test
  public void getTest() throws IOException {
    HttpConnection httpConnection = new HttpConnection();
    Document document = httpConnection.get();
    assertNotNull(document);
    assertEquals("http://example.com", document.location());
  }

  @Test
  public void postTest() throws IOException {
    HttpConnection httpConnection = new HttpConnection();
    Document document = httpConnection.post();
    assertNotNull(document);
    assertEquals("http://example.com", document.location());
  }

  @Test
  public void methodTest() {
    HttpConnection httpConnection = new HttpConnection();
    httpConnection = httpConnection.method(Method.POST);
    assertEquals(Method.POST, httpConnection.request().method());
  }

  @Test
  public void ignoreHttpErrorsTest() {
    HttpConnection httpConnection = new HttpConnection();
    httpConnection = httpConnection.ignoreHttpErrors(true);
    assertEquals(true, httpConnection.request().ignoreHttpErrors());
  }

  @Test
  public void referrerTest() {
    HttpConnection httpConnection = new HttpConnection();
    httpConnection = httpConnection.referrer("http://referrer.com");
    assertEquals("http://referrer.com", httpConnection.request().referrer());
  }

  @Test
  public void headersTest() {
    Map<String, String> headers = new HashMap<>();
    headers.put("Content-Type", "application/json");
    Connection connection = HttpConnection.connect("http://example.com").headers(headers);
    assertEquals("application/json", connection.request().headers().get("Content-Type"));
  }

  @Test
  public void ignoreContentTypeTest() {
    Connection connection = HttpConnection.connect("http://example.com").ignoreContentType(true);
    assertEquals(true, connection.request().ignoreContentType());
  }

  @Test
  public void dataWithInputStreamTest() {
    InputStream inputStream = new ByteArrayInputStream("test input stream".getBytes());
    Connection connection = HttpConnection.connect("http://example.com").data("key","filename", inputStream);
    assertEquals("filename", connection.request().data().get(0).value());
    assertEquals("key", connection.request().data().get(0).key());
  }
  
  @Test
  public void sslSocketFactoryTest() {
    SSLSocketFactory sslSocketFactory = (SSLSocketFactory) SSLSocketFactory.getDefault();
    Connection connection = HttpConnection.connect("http://example.com").sslSocketFactory(sslSocketFactory);
    assertEquals(sslSocketFactory, connection.request().sslSocketFactory());
  }
}
