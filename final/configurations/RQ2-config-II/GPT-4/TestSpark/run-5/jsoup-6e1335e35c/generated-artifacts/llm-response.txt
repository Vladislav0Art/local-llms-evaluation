import org.junit.Assert;
import org.junit.Test;
import org.jsoup.nodes.Element;
import org.jsoup.parser.Tag;
import org.jsoup.select.Elements;

import static org.mockito.Mockito.mock;
import static org.mockito.Mockito.when;

public class ElementTest {
	
    @Test
    public void hasChildNodesTest() {
        Element testElement = new Element(Tag.valueOf("div"), "");
        Assert.assertFalse(testElement.hasChildNodes());
    }

    @Test
    public void hasChildNodesWhenChildExistsTest() {
        Element testElement = new Element(Tag.valueOf("div"), "");
        testElement.appendChild(new Element(Tag.valueOf("p"), ""));
        Assert.assertTrue(testElement.hasChildNodes());
    }

    @Test
    public void attributesTest() {
        Element testElement = new Element(Tag.valueOf("div"), "");
        testElement.attr("class", "testing");
        Assert.assertEquals("testing", testElement.attributes().get("class"));
    }

    @Test(expected = IllegalArgumentException.class)
    public void attrWithoutValueTest() {
    	Element testElement = new Element(Tag.valueOf("div"), "");
    	testElement.attr("class", true);
    }

    @Test
    public void datasetTest() {
        Element testElement = new Element(Tag.valueOf("div"), "");
        testElement.attr("data-test", "123");
        Assert.assertEquals("123", testElement.dataset().get("test"));
    }

    @Test
    public void isTestPositive() {
        String cssQuery = ".testing";
        Element testElement = new Element(Tag.valueOf("div"), "");
        testElement.attr("class", "testing");
        Assert.assertTrue(testElement.is(cssQuery));
    }

    @Test
    public void selectTestPositive() {
        String cssQuery = "p";
        Element testElement = new Element(Tag.valueOf("div"), "");
        testElement.appendChild(new Element(Tag.valueOf("p"), ""));
        Elements elements = testElement.select(cssQuery);
        Assert.assertNotNull(elements);
        Assert.assertEquals(1, elements.size());
    }

    @Test
    public void prependElementTest() {
        String tagName = "p";
        Element testElement = new Element(Tag.valueOf("div"), "");
        testElement.prependElement(tagName);
        Assert.assertEquals(tagName, testElement.child(0).tagName());
    }

    @Test
    public void childTest() {
        String childTagName = "p";
        Element parentElement = new Element(Tag.valueOf("div"), "");
        parentElement.appendChild(new Element(Tag.valueOf(childTagName), ""));
        Assert.assertEquals(childTagName, parentElement.child(0).tagName());
    }

    @Test
    public void getElementsContainingTextTest() {
        Element mockElement = mock(Element.class);
        when(mockElement.text()).thenReturn("Hello World");
        Element parentElement = new Element(Tag.valueOf("div"), "");
        parentElement.appendChild(mockElement);
        Elements elements = parentElement.getElementsContainingText("Hello");
        Assert.assertEquals(1, elements.size());
        Assert.assertEquals("Hello World", elements.get(0).text());
    }

}
