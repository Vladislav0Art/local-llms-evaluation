package org.davidmoten.text.utils;

import static org.junit.Assert.assertNotNull;
import static org.junit.Assert.assertTrue;
import static org.junit.Assert.assertEquals;

import java.io.File;
import java.io.FileNotFoundException;
import java.io.FileReader;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.Reader;
import java.io.StringReader;
import java.nio.charset.StandardCharsets;
import java.util.HashSet;
import org.junit.Rule;
import org.junit.Test;
import org.junit.rules.TemporaryFolder;
import org.junit.runner.RunWith;
import org.mockito.junit.MockitoJUnitRunner;

@RunWith(MockitoJUnitRunner.class)
public class WordWrapTest {

    @Rule
    public TemporaryFolder folder = new TemporaryFolder();

    @Test
    public void fromReaderTest() {
        Reader reader = new StringReader("Testing from Reader");
        WordWrap.Builder result = WordWrap.from(reader);
        assertNotNull(result);
    }
    
    @Test
    public void fromClasspathUtf8Test() throws FileNotFoundException {
        String resource = "/test.txt";
        WordWrap.Builder result = WordWrap.fromClasspathUtf8(resource);
        assertNotNull(result);
    }
  
    @Test
    public void fromClasspathTest() throws FileNotFoundException {
        String resource = "/test.txt";
        WordWrap.Builder result = WordWrap.fromClasspath(resource, StandardCharsets.UTF_8);
        assertNotNull(result);
    }
    
    @Test
    public void fromCharSequenceTest() {
        CharSequence text = "Testing from CharSequence";
        WordWrap.Builder result = WordWrap.from(text);
        assertNotNull(result);
    }
    
    @Test
    public void fromUtf8InputStreamTest() {
        InputStream in = this.getClass().getResourceAsStream("/test.txt");
        WordWrap.Builder result = WordWrap.fromUtf8(in);
        assertNotNull(result);
    }
    
    @Test
    public void fromInputStreamCharsetTest() {
        InputStream in = this.getClass().getResourceAsStream("/test.txt");
        WordWrap.Builder result = WordWrap.from(in, StandardCharsets.UTF_8);
        assertNotNull(result);
    }

    @Test
    public void fromFileCharsetTest() throws Exception {
        File createdFile = folder.newFile("myfile.txt");
        WordWrap.Builder result = WordWrap.from(createdFile, StandardCharsets.UTF_8);
        assertNotNull(result);
    }
  
    @Test
    public void isWhitespaceTest() {
        CharSequence whitespace = " ";
        boolean result = WordWrap.isWhitespace(whitespace);
        assertTrue(result);
    }

    @Test
    public void isWhitespaceTestNonWhitespace() {
        CharSequence nonWhitespace = "Test";
        boolean result = WordWrap.isWhitespace(nonWhitespace);
        assertTrue(!result);
    }

    @Test
    public void rightTrimTest() {
        CharSequence input = " Test ";
        CharSequence result = WordWrap.rightTrim(input);
        assertEquals(" Test", result);
    }

    @Test
    public void leftTrimTest() {
        StringBuilder2 input = new StringBuilder2(" Test ");
        WordWrap.leftTrim(input);
        assertTrue(input.toString().equals("Test "));
    }
    
    @Test
    public void wordWrapTest() throws IOException {
        Reader in = new StringReader("Testing wordWrap");
        StringWriter out = new StringWriter();
        Function<CharSequence, Number> stringWidth = CharSequence::length;
        WordWrap.wordWrap(in, out, "\n", 10, stringWidth, new HashSet<>(), false, false);
        assertEquals("Testing wordWrap", out.toString());
    }
}
