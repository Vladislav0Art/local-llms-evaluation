import com.ezylang.evalex.config.ExpressionConfiguration;
import com.ezylang.evalex.parser.Token;
import com.ezylang.evalex.functions.FunctionIfc;
import com.ezylang.evalex.operators.OperatorIfc;
import com.ezylang.evalex.parser.Tokenizer;
import com.ezylang.evalex.parser.ParseException;
import org.junit.Assert;
import org.junit.Test;
import org.mockito.Mockito;

import java.util.List;

public class TokenizerTest {

    @Test
    public void TokenizerConstructionTest(){
        ExpressionConfiguration configuration = Mockito.mock(ExpressionConfiguration.class);
        Tokenizer token = new Tokenizer("expression", configuration);
        Assert.assertNotNull(token);
    }

    @Test
    public void parseEmptyExpressionTest(){
        ExpressionConfiguration configuration = Mockito.mock(ExpressionConfiguration.class);
        Tokenizer token = new Tokenizer("", configuration);
        List<Token> tokens = token.parse();
        Assert.assertTrue(tokens.isEmpty());
    }

    @Test(expected = ParseException.class)
    public void parseNullExpressionTest(){
        ExpressionConfiguration configuration = Mockito.mock(ExpressionConfiguration.class);
        Tokenizer token = new Tokenizer(null, configuration);
        token.parse();
    }

    @Test
    public void parseSingleOperandExpressionTest(){
        ExpressionConfiguration configuration = Mockito.mock(ExpressionConfiguration.class);
        Tokenizer token = new Tokenizer("10", configuration);
        List<Token> tokens = token.parse();
        Assert.assertEquals(1, tokens.size());
        Assert.assertEquals(Token.TokenType.LITERAL_NUM, tokens.get(0).getType());
    }

    @Test
    public void parseSingleOperatorExpressionTest(){
        ExpressionConfiguration configuration = Mockito.mock(ExpressionConfiguration.class);
        Mockito.when(configuration.getOperator("/")).thenReturn(Mockito.mock(OperatorIfc.class));
        Tokenizer token = new Tokenizer("/", configuration);
        List<Token> tokens = token.parse();
        Assert.assertEquals(1, tokens.size());
        Assert.assertEquals(Token.TokenType.OPERATOR, tokens.get(0).getType());
    }

    @Test
    public void parseStringLiteralTest(){
        ExpressionConfiguration configuration = Mockito.mock(ExpressionConfiguration.class);
        Tokenizer token = new Tokenizer("\"StringLiteral\"", configuration);
        Token stringToken = token.parseStringLiteral();
        Assert.assertEquals(Token.TokenType.STRINGPARAM, stringToken.getType());
    }

    @Test(expected = ParseException.class)
    public void parseStringLiteralWithoutQuotesTest(){
        ExpressionConfiguration configuration = Mockito.mock(ExpressionConfiguration.class);
        Tokenizer token = new Tokenizer("StringLiteral", configuration);
        token.parseStringLiteral();
    }
}
