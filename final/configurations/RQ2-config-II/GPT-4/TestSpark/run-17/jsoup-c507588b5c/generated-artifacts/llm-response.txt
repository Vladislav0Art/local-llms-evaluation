import org.junit.Test;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.TextNode;

import static org.junit.Assert.*;

public class TextNodeTest {

    @Test
    public void textNodeConstructorTest() {
        assertNotNull(new TextNode("some text"));
    }

    @Test
    public void nodeNameTest() {
        assertEquals("#text", new TextNode("some text").nodeName());
    }

    @Test
    public void textTest() {
        assertEquals("some text", new TextNode("some text").text());
    }

    @Test
    public void textSetterTest() {
        TextNode tn = new TextNode("old text");
        tn.text("new text");
        assertEquals("new text", tn.text());
    }

    @Test
    public void getWholeTextTest() {
        assertEquals("some text", new TextNode("some text").getWholeText());
    }

    @Test
    public void isBlankTest() {
        assertTrue(new TextNode("     ").isBlank());
        assertFalse(new TextNode("  a  ").isBlank());
    }

    @Test
    public void splitTextTest() {
        TextNode tn = new TextNode("original text");
        TextNode newTn = tn.splitText(9);
        assertEquals("original ", tn.text());
        assertEquals("text", newTn.text());
    }

    @Test
    public void outerHtmlHeadTest() throws Exception {
        TextNode tn = new TextNode("Some text");
        StringBuilder sb = new StringBuilder();
        Document.OutputSettings outputSettings = new Document("").outputSettings();
        tn.outerHtmlHead(sb, 0, outputSettings);
        assertEquals("Some text", sb.toString());
    }

    @Test
    public void outerHtmlTailTest() throws Exception {
        TextNode tn = new TextNode("Some text");
        StringBuilder sb = new StringBuilder();
        Document.OutputSettings outputSettings = new Document("").outputSettings();
        tn.outerHtmlTail(sb, 0, outputSettings);
        assertEquals("", sb.toString());
    }
    
    @Test
    public void toStringTest() {
        assertEquals("Some text", new TextNode("Some text").toString());
    }

    @Test
    public void cloneTest() {
        TextNode tn = new TextNode("some text");
        TextNode cloned = tn.clone();
        assertNotSame(tn, cloned);
        assertEquals(tn.text(), cloned.text());
    }

    @Test
    public void createFromEncodedTest() {
        assertEquals("some text", TextNode.createFromEncoded("some text").text());
    }

    @Test
    public void normaliseWhitespaceTest() {
        assertEquals("a b c", TextNode.normaliseWhitespace("a    b  c"));
    }

    @Test
    public void stripLeadingWhitespaceTest() {
        assertEquals("abc  ", TextNode.stripLeadingWhitespace("   abc  "));
    }

    @Test
    public void lastCharIsWhitespaceTest() {
        assertTrue(TextNode.lastCharIsWhitespace(new StringBuilder("a ")));
        assertFalse(TextNode.lastCharIsWhitespace(new StringBuilder("a")));
    }
}
