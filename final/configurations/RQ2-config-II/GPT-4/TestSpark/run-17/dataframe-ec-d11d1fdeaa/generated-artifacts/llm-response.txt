import io.github.vmzakharov.ecdataframe.dataframe.*;
import io.github.vmzakharov.ecdataframe.dsl.EvalContext;
import io.github.vmzakharov.ecdataframe.dsl.value.Value;
import org.eclipse.collections.api.list.ListIterable;
import org.eclipse.collections.api.list.primitive.MutableIntList;
import org.junit.Assert;
import org.junit.Test;
import org.mockito.Mockito;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalDateTime;

public class DataFrameTest {

    @Test
    public void addStringColumnTest() {
        DataFrame dataFrame = new DataFrame("DataFrame1");
        dataFrame.addStringColumn("TestColumn");
        Assert.assertEquals(1, dataFrame.columnCount());
        Assert.assertEquals("TestColumn", dataFrame.getColumnAt(0).getName());
    }

    @Test
    public void rowCountTest() {
        DataFrame dataFrame = new DataFrame("DataFrame1");
        Assert.assertEquals(0, dataFrame.rowCount());
        dataFrame.addRow();
        Assert.assertEquals(1, dataFrame.rowCount());
        dataFrame.addRow();
        Assert.assertEquals(2, dataFrame.rowCount());
    }

    @Test
    public void addRowTest() {
        DataFrame dataFrame = new DataFrame("DataFrame1");
        dataFrame.addStringColumn("TestColumn");
        Assert.assertEquals(1, dataFrame.columnCount());
        dataFrame.addRow(Mockito.mock(ListIterable.class));
        Assert.assertEquals(1, dataFrame.rowCount());
    }

    @Test
    public void asCsvStringTest() {
        DataFrame dataFrame = new DataFrame("DataFrame1");
        Assert.assertEquals("", dataFrame.asCsvString());

        dataFrame.addStringColumn("Column");
        dataFrame.addRow();
        Assert.assertEquals("Column\n\n", dataFrame.asCsvString());
    }

    @Test
    public void getNameTest() {
        DataFrame dataFrame = new DataFrame("DataFrame1");
        Assert.assertEquals("DataFrame1", dataFrame.getName());

        DataFrame dataFrame2 = new DataFrame("DataFrame2");
        Assert.assertEquals("DataFrame2", dataFrame2.getName());
    }

    @Test
    public void setValueTest() {
        DataFrame dataFrame = new DataFrame("DataFrame1");
        dataFrame.addStringColumn("ColumnString");
        dataFrame.addRow();
        Assert.assertEquals("", dataFrame.getString("ColumnString", 0));

        dataFrame.setValue(0, 0, new Value("test", java.lang.String.class));
        Assert.assertEquals("test", dataFrame.getString("ColumnString", 0));
    }

    @Test
    public void setFlagTest() {
        DataFrame dataFrame = new DataFrame("DataFrame1");
        assert (!dataFrame.isFlagged(0));
        dataFrame.setFlag(0);
        assert (dataFrame.isFlagged(0));
    }

    @Test
    public void dropColumnTest() {
        DataFrame dataFrame = new DataFrame("DataFrame1");
        dataFrame.addStringColumn("ColumnString");
        dataFrame.addRow();
        dataFrame.addStringColumn("AnotherColumnString");
        Assert.assertEquals(2, dataFrame.columnCount());
        Assert.assertEquals("ColumnString", dataFrame.getColumnAt(0).getName());

        dataFrame = dataFrame.dropColumn("ColumnString");
        Assert.assertEquals(1, dataFrame.columnCount());
        Assert.assertEquals("AnotherColumnString", dataFrame.getColumnAt(0).getName());
    }

    @Test
    public void addDateTimeColumnWithListTest() {
        DataFrame dataFrame = new DataFrame("DataFrame1");
        ListIterable<LocalDateTime> dates = Mockito.mock(ListIterable.class);
        dataFrame.addDateTimeColumn("ColumnDateTime", dates);
    }

    @Test
    public void addDecimalColumnWithExpressionTest() {
        DataFrame dataFrame = new DataFrame("DataFrame1");
        ListIterable<BigDecimal> decimals = Mockito.mock(ListIterable.class);
        dataFrame.addDecimalColumn("ColumnDecimal", decimals);
    }

    @Test
    public void getAggregateIndexTest() {
        DataFrame dataFrame = new DataFrame("DataFrame1");
        MutableIntList result = dataFrame.getAggregateIndex(1);
        Assert.assertNotNull(result);
        Assert.assertTrue(result.isEmpty());
    }

    @Test
    public void setExternalEvalContextTest() {
        DataFrame dataFrame = new DataFrame("DataFrame1");
        EvalContext evalContext = Mockito.mock(EvalContext.class);
        dataFrame.setExternalEvalContext(evalContext);
    }
}
