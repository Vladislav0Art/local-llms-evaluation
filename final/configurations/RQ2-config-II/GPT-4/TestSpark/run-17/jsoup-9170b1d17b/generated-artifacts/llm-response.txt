import java.util.Iterator;

import static org.junit.Assert.*;

public class AttributesTest {

    @Test
    public void getValidKeyTest() {
        Attributes attrs = new Attributes();
        attrs.add("key", "value");
        assertEquals("value", attrs.get("key"));
    }

    public void getInvalidKeyTest() {
        Attributes attrs = new Attributes();
        attrs.get("key");

    public void addKeyWithValueTest() {

        assertTrue(attrs.hasKey("key"));

    public void addKeyWithoutValueTest() {
        Attributes attrs = new Attributes();
        attrs.add("key", null);
        assertTrue(attrs.hasKey("key"));
    }

    public void removeKeyTest() {
        Attributes attrs = new Attributes();
        attrs.add("key", "value");
        attrs.remove("key");
        assertFalse(attrs.hasKey("key"));
    }

    @Test
    public void removeNonExistingKeyTest() {
        Attributes attrs = new Attributes();
        attrs.remove("key");
        assertFalse(attrs.hasKey("key"));
    }

    public void hasKeyTest() {
        Attributes attrs = new Attributes();
        attrs.add("key", "value");
        assertTrue(attrs.hasKey("key"));

        attrs.add("key", "value");

        attrs.add("KEY", "value");
        attrs.normalize();
