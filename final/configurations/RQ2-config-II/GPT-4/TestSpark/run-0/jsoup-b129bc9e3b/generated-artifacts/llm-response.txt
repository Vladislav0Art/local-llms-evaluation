import org.jsoup.Connection;
import org.jsoup.HttpStatusException;
import org.jsoup.helper.HttpConnection;
import org.jsoup.nodes.Document;
import org.junit.Test;

import javax.net.ssl.SSLSocketFactory;
import java.io.IOException;
import java.util.HashMap;
import java.util.Map;

import static org.junit.Assert.assertEquals;

public class HttpConnectionTest {
    @Test
    public void connectByUrlTest() {
        String url = "http://example.com";
        HttpConnection connect = (HttpConnection) HttpConnection.connect(url);
        assertEquals(url, connect.request().url().toString());
    }

    @Test(expected = IllegalArgumentException.class)
    public void connectByUrlWithIllegalArgumentTest() {
        String url = "invalidUrl";
        HttpConnection.connect(url);
    }

    @Test
    public void newRequestTest() {
        HttpConnection conn = new HttpConnection();
        Connection newConn = conn.newRequest();
        assertEquals(null, newConn.request().url());
    }

    @Test
    public void urlByUrlTest() {
        String url = "http://example.com";
        HttpConnection conn = new HttpConnection();
        conn.url(url);
        assertEquals(url, conn.request().url().toString());
    }

    @Test(expected = IllegalArgumentException.class)
    public void urlByUrlWithInvalidUrlTest() {
        String url = "invalidUrl";
        HttpConnection conn = new HttpConnection();
        conn.url(url);
    }

    @Test
    public void userAgentTest() {
        String userAgent = "Mozilla/5.0";
        HttpConnection conn = new HttpConnection();
        conn.userAgent(userAgent);
        assertEquals(userAgent, conn.request().header("User-Agent"));
    }

    @Test
    public void timeoutTest() {
        int timeout = 5000;
        HttpConnection conn = new HttpConnection();
        conn.timeout(timeout);
        assertEquals(timeout, conn.request().timeout());
    }

    @Test
    public void cookieTest() {
        String name = "cookieName";
        String value = "cookieValue";
        HttpConnection conn = new HttpConnection();
        conn.cookie(name, value);
        assertEquals(value, conn.request().cookie(name));
    }

    @Test
    public void cookiesTest() {
        Map<String, String> cookies = new HashMap<>();
        cookies.put("cookieName", "cookieValue");
        HttpConnection conn = new HttpConnection();
        conn.cookies(cookies);
        assertEquals("cookieValue", conn.request().cookie("cookieName"));
    }

    @Test
    public void getExecuteTest() throws IOException {
        String url = "http://example.com";
        HttpConnection conn = (HttpConnection) HttpConnection.connect(url);
        Document document = conn.get();
        assertEquals(url, document.location());
    }

    @Test(expected = HttpStatusException.class)
    public void getExecuteWithHttpStatusExceptionTest() throws IOException {
        String url = "http://nonexistentwebsite222222222.com";
        HttpConnection conn = (HttpConnection) HttpConnection.connect(url);
        conn.get();
    }

    @Test
    public void sslSocketFactoryTest() {
        SSLSocketFactory sslSF = (SSLSocketFactory) SSLSocketFactory.getDefault();
        HttpConnection conn = new HttpConnection();
        conn.sslSocketFactory(sslSF);
        assertEquals(sslSF, conn.request().sslSocketFactory());
    }
}
