import io.netty.buffer.Unpooled;
import io.netty.channel.Channel;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.runners.MockitoJUnitRunner;
import org.traccar.Protocol;
import org.traccar.model.Position;

import java.net.SocketAddress;

import static org.junit.Assert.*;

@RunWith(MockitoJUnitRunner.class)
public class WatchProtocolDecoderTest {

    @Mock
    private Protocol protocol;

    @Mock
    private Channel channel;

    @Mock
    private SocketAddress remoteAddress;

    @Test
    public void getHasIndexTest() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(protocol);
        assertFalse(decoder.getHasIndex());
    }

    @Test
    public void getManufacturerTest() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(protocol);
        assertEquals("Generic", decoder.getManufacturer());
    }

    @Test
    public void decodeValidMessageTest() throws Exception {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(protocol);
        String msg = "[3G*8800000015*0002*LK]";
        Object decoded = decoder.decode(channel, remoteAddress, Unpooled.wrappedBuffer(msg.getBytes()));
        assertTrue(decoded instanceof Position);
    }

    @Test
    public void decodeNullMessageTest() throws Exception {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(protocol);
        assertNull(decoder.decode(channel, remoteAddress, null));
    }

    @Test
    public void decodeInvalidMessageTest() throws Exception {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(protocol);
        String msg = "[3G*invalid*message*format]";
        assertNull(decoder.decode(channel, remoteAddress, Unpooled.wrappedBuffer(msg.getBytes())));
    }

    @Test
    public void decodeResponseTest() throws Exception {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(protocol);
        String msg = "[3G*8800000015*0002*LK]";
        decoder.decode(channel, remoteAddress, Unpooled.wrappedBuffer(msg.getBytes()));
        Mockito.verify(channel).writeAndFlush(new NetworkMessage("[3G*8800000015*0002*LK]", remoteAddress));
    }
}
