import leetcode.medium.OnlineStockSpan;
import org.junit.Test;
import static org.junit.Assert.*;

public class OnlineStockSpanTest {

    @Test
    public void nextPriceIsHigherTest() {
        OnlineStockSpan stockSpan = new OnlineStockSpan();
        stockSpan.next(100);
        int result = stockSpan.next(200);
        assertEquals(3, result);
    }

    @Test
    public void nextPriceIsLowerTest() {
        OnlineStockSpan stockSpan = new OnlineStockSpan();
        stockSpan.next(200);
        int result = stockSpan.next(100);
        assertEquals(1, result);
    }

    @Test
    public void nextPriceIsSameTest() {
        OnlineStockSpan stockSpan = new OnlineStockSpan();
        stockSpan.next(200);
        int result = stockSpan.next(200);
        assertEquals(2, result);
    }

    @Test(expected = IllegalArgumentException.class)
    public void nextPriceIsNegativeTest() {
        OnlineStockSpan stockSpan = new OnlineStockSpan();
        stockSpan.next(-100);
    }

    @Test
    public void calculateSpansAllPricesIncreasingTest() {
        OnlineStockSpan stockSpan = new OnlineStockSpan();
        int[] result = stockSpan.calculateSpans(new int[]{100, 200, 300, 400, 500});
        assertArrayEquals(new int[]{1, 2, 3, 4, 5}, result);
    }

    @Test
    public void calculateSpansAllPricesDecreasingTest() {
        OnlineStockSpan stockSpan = new OnlineStockSpan();
        int[] result = stockSpan.calculateSpans(new int[]{500, 400, 300, 200, 100});
        assertArrayEquals(new int[]{1, 1, 1, 1, 1}, result);
    }

    @Test
    public void calculateSpansSomePricesSameTest() {
        OnlineStockSpan stockSpan = new OnlineStockSpan();
        int[] result = stockSpan.calculateSpans(new int[]{100, 200, 200, 100, 300});
        assertArrayEquals(new int[]{1, 2, 3, 1, 5}, result);
    }

    @Test(expected = IllegalArgumentException.class)
    public void calculateSpansEmptyArrayTest() {
        OnlineStockSpan stockSpan = new OnlineStockSpan();
        stockSpan.calculateSpans(new int[]{});
    }

    @Test(expected = IllegalArgumentException.class)
    public void calculateSpansNullArrayTest() {
        OnlineStockSpan stockSpan = new OnlineStockSpan();
        stockSpan.calculateSpans(null);
    }
}
