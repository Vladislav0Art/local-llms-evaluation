import org.junit.Test;
import org.jsoup.Connection;
import org.jsoup.helper.UrlBuilder;
import java.net.MalformedURLException;
import java.net.URL;
import java.net.UnsupportedEncodingException;

import static org.junit.Assert.*;

public class UrlBuilderTest {

    @Test
    public void UrlBuilderConstructorValidURLTest() throws MalformedURLException {
        URL url = new URL("https://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        assertNotNull(urlBuilder);
    }
    
    @Test(expected = NullPointerException.class)
    public void UrlBuilderConstructorNullURLTest() {
        new UrlBuilder(null);
    }

    @Test
    public void buildTest() throws MalformedURLException {
        URL url = new URL("https://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        URL builtURL = urlBuilder.build();
        assertEquals(url, builtURL);
    }

    @Test
    public void appendKeyValTest() throws MalformedURLException, UnsupportedEncodingException {
        URL url = new URL("https://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        Connection.KeyVal keyVal = Connection.KeyVal.create("key", "value");
        urlBuilder.appendKeyVal(keyVal);
        URL builtURL = urlBuilder.build();
        assertEquals("https://example.com?key=value", builtURL.toString());
    }

    @Test(expected = NullPointerException.class)
    public void appendKeyValNullKeyValTest() throws MalformedURLException, UnsupportedEncodingException {
        URL url = new URL("https://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        urlBuilder.appendKeyVal(null);
    }

    @Test(expected = UnsupportedEncodingException.class)
    public void appendKeyValUnsupportedEncodingTest() throws MalformedURLException, UnsupportedEncodingException {
        URL url = new URL("https://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        Connection.KeyVal keyVal = Connection.KeyVal.create("key", "va\ud83d\ude00lue");
        urlBuilder.appendKeyVal(keyVal);
    }

}
