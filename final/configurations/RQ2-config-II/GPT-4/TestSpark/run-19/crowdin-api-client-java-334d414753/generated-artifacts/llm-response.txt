package com.crowdin.client.core.http.impl.json;

import com.crowdin.client.core.http.exceptions.CrowdinApiException;
import com.crowdin.client.sourcefiles.model.FileInfo;
import org.junit.Assert;
import org.junit.Test;
import org.mockito.Mockito;

public class JacksonJsonTransformerTest {
    private static final String VALID_JSON = "{\"name\":\"testFile.txt\",\"path\":\"testPath\"}";
    private static final String INVALID_JSON = "{\"name\"::\"testFile.txt\",,\"path\"::\"testPath\"}";
    private static final FileInfo FILE_INFO = new FileInfo("testFile.txt", "testPath");

    @Test
    public void parseValidJsonTest() throws Exception {
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        FileInfo fileInfo = transformer.parse(VALID_JSON, FileInfo.class);
        Assert.assertEquals(FILE_INFO, fileInfo);
    }

    @Test(expected = CrowdinApiException.class)
    public void parseInvalidJsonTest() throws Exception {
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        transformer.parse(INVALID_JSON, FileInfo.class);
    }

    @Test
    public void parseNullJsonTest() throws Exception {
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        FileInfo fileInfo = transformer.parse(null, FileInfo.class);
        Assert.assertNull(fileInfo);
    }

    @Test
    public void convertValidObjectTest() throws Exception {
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        String json = transformer.convert(FILE_INFO);
        Assert.assertEquals(VALID_JSON, json);
    }

    @Test(expected = CrowdinApiException.class)
    public void convertInvalidObjectTest() throws Exception {
        JacksonJsonTransformer transformer = Mockito.spy(JacksonJsonTransformer.class);
        Mockito.doThrow(new CrowdinApiException("Exception")).when(transformer).convert(Mockito.any());
        
        transformer.convert(new Object());
    }

    @Test
    public void convertNullObjectTest() throws Exception {
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        String json = transformer.convert(null);
        Assert.assertNull(json);
    }
}
