import org.jsoup.nodes.Element;
import org.jsoup.parser.Tag;
import org.jsoup.select.Elements;
import org.junit.Assert;
import org.junit.Test;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class ElementTest {

    @Test
    public void tagNameWithValidTagTest() {
        Element element = new Element("div");
        Assert.assertEquals("div", element.tagName());
    }

    @Test
    public void tagNameWithEmptyTagTest() {
        Element element = new Element("");
        Assert.assertEquals("", element.tagName());
    }

    @Test
    public void normalNameTest() {
        Element element = new Element(Tag.valueOf("h1"), "");
        Assert.assertEquals("h1", element.normalName());
    }

    @Test
    public void idSetAndGetTest() {
        Element element = new Element("p");
        element.id("unique");
        Assert.assertEquals("unique", element.id());
    }

    @Test
    public void hasChildNodesWithNoChildTest() {
        Element element = new Element("p");
        Assert.assertFalse(element.hasChildNodes());
    }

    @Test
    public void childNodesSizeWithNoChildTest() {
        Element element = new Element("p");
        Assert.assertEquals(0, element.childNodeSize());
    }

    @Test
    public void childNodesSizeWithOneChildTest() {
        Element parent = new Element("div");
        Element child = new Element("p");
        parent.appendChild(child);
        Assert.assertEquals(1, parent.childNodeSize());
    }

    @Test
    public void attributesTest() {
        Element element = new Element("p");
        element.attr("class", "test");
        Assert.assertEquals("test", element.attributes().get("class"));
    }

    @Test
    public void isBlockTest() {
        Element element = new Element("div");
        Assert.assertTrue(element.isBlock());
    }

    @Test
    public void attrTest() {
        Element element = new Element("p");
        element.attr("data-test", "value");
        Assert.assertEquals("value", element.attributes().get("data-test"));
    }

    @Test
    public void childrenTest() {
        Element parent = new Element("div");
        Element child1 = new Element("p");
        Element child2 = new Element("h1");
        parent.appendChild(child1);
        parent.appendChild(child2);

        List<String> expected = new ArrayList<>(Collections.singletonList("p"));
        expected.add("h1");

        Elements children = parent.children();
        List<String> actual = new ArrayList<>();
        for (Object child : children) {
            actual.add(((Element) child).tagName());
        }

        Assert.assertEquals(expected, actual);
    }

    @Test
    public void hasAttributesWithNoAttributesTest() {
        Element element = new Element("div");
        Assert.assertFalse(element.hasAttributes());
    }

    @Test
    public void hasAttributesWithOneAttributeTest() {
        Element element = new Element("div");
        element.attr("id", "test");
        Assert.assertTrue(element.hasAttributes());
    }

    @Test
    public void baseUriTest() {
        Element element = new Element(Tag.valueOf("h1"), "http://localhost");
        Assert.assertEquals("http://localhost", element.baseUri());
    }

    @Test
    public void nodeNameTest() {
        Element element = new Element("div");
        Assert.assertEquals("div", element.nodeName());
    }
}
