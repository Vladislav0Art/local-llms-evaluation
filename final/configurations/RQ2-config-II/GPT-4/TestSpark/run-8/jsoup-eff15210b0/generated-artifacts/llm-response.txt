import org.jsoup.Connection;
import org.jsoup.parser.Parser;
import org.jsoup.helper.HttpConnection;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import java.io.IOException;
import java.io.InputStream;
import java.net.CookieStore;
import java.net.Proxy;
import java.net.SocketException;
import java.net.URL;
import java.util.HashMap;
import java.util.Map;

import static org.junit.Assert.*;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class HttpConnectionTest {

    @Mock
    private URL urlMock;

    @Mock
    private InputStream inputStreamMock;

    @Mock
    private Proxy proxyMock;

    @Mock
    private CookieStore cookieStoreMock;

    @Mock
    private Parser parserMock;

    @Test
    public void connectUrlTest() {
        Connection connection = HttpConnection.connect("https://test.com");
        assertNotNull(connection);
        assertTrue(connection instanceof HttpConnection);
    }

    @Test(expected = IllegalArgumentException.class)
    public void connectNullUrlTest() {
        HttpConnection.connect((String) null);
    }

    @Test
    public void newRequestTest() {
        Connection connection = new HttpConnection();
        assertNotNull(connection.newRequest());
    }

    @Test
    public void urlTest() throws IOException {
        when(urlMock.toString()).thenReturn("https://test.com");

        Connection connection = new HttpConnection();
        assertNotNull(connection.url(urlMock));
    }

    @Test(expected = IllegalArgumentException.class)
    public void urlNullTest() {
        new HttpConnection().url((URL) null);
    }

    @Test
    public void userAgentTest() {
        Connection connection = new HttpConnection();
        assertNotNull(connection.userAgent("test-agent"));
    }

    @Test
    public void dataKeyAndValueTest() {
        Connection connection = new HttpConnection();
        assertNotNull(connection.data("key", "value"));
    }

    @Test
    public void dataMapTest() {
        Connection connection = new HttpConnection();
        Map<String, String> dataMap = new HashMap<>();
        dataMap.put("key", "value");

        assertNotNull(connection.data(dataMap));
    }

    @Test
    public void dataInputStreamTest() {
        Connection connection = new HttpConnection();
        assertNotNull(connection.data("key", "test.txt", inputStreamMock));
    }

    @Test(expected = IllegalArgumentException.class)
    public void dataNullInputStreamTest() {
        new HttpConnection().data("key", "test.txt", null);
    }

    @Test
    public void headersTest() {
        Map<String, String> headers = new HashMap<>();
        headers.put("Content-Type", "application/json");
        Connection connection = new HttpConnection().headers(headers);
        assertEquals("application/json", connection.request().headers().get("Content-Type"));
    }

    @Test
    public void cookiesTest() {
        Map<String,String> cookies = new HashMap<>();
        cookies.put("key", "value");
        Connection connection = new HttpConnection().cookies(cookies);

        assertEquals("value", connection.request().cookies().get("key"));
    }

    @Test
    public void getTest() throws IOException {
        when(urlMock.toString()).thenReturn("https://test.com");

        Connection connection = new HttpConnection().url(urlMock);
        assertNotNull(connection.get());
    }

    @Test(expected = SocketException.class)
    public void getWithIOExceptionTest() throws IOException {
        when(urlMock.toString()).thenReturn("https://invalid-url");

        Connection connection = new HttpConnection().url(urlMock);
        connection.get();
    }

    @Test
    public void executeTest() throws IOException {
        when(urlMock.toString()).thenReturn("https://test.com");

        Connection connection = new HttpConnection().url(urlMock);
        assertNotNull(connection.execute());
    }

    @Test(expected = SocketException.class)
    public void executeWithIOExceptionTest() throws IOException {
        when(urlMock.toString()).thenReturn("https://invalid-url");

        Connection connection = new HttpConnection().url(urlMock);
        connection.execute();
    }

}
