import io.github.vmzakharov.ecdataframe.dataframe.DataFrame;
import io.github.vmzakharov.ecdataframe.dsl.Expression;
import io.github.vmzakharov.ecdataframe.dsl.value.ValueType;
import org.eclipse.collections.impl.factory.Lists;
import org.eclipse.collections.impl.list.primitive.IntInterval;
import org.junit.Assert;
import org.junit.Test;
import static org.mockito.Matchers.any;
import static org.mockito.Mockito.mock;
import static org.mockito.Mockito.when;

public class DataFrameTest {

    @Test
    public void getNameTest() {
        DataFrame dataframe = new DataFrame("test");
        Assert.assertEquals("test", dataframe.getName());
    }

    @Test
    public void addStringColumnTest() {
        DataFrame dataframe = new DataFrame("test");
        dataframe.addStringColumn("Column1");
        Assert.assertTrue(dataframe.hasColumn("Column1"));
    }

    @Test
    public void rowCountTest() {
        DataFrame dataframe = new DataFrame("test");
        dataframe.addStringColumn("Column1", Lists.immutable.of("Value1", "Value2"));
        Assert.assertEquals(2, dataframe.rowCount());
    }

    @Test
    public void addColumn_From_StringTest() {
        DataFrame dataframe = new DataFrame("test");
        Expression expr = mock(Expression.class);
        when(expr.evaluate(any())).thenReturn("Value1");
        dataframe.addColumn("Column1", expr);
        Assert.assertTrue(dataframe.hasColumn("Column1"));
    }
    
    @Test
    public void dropColumnTest() {
        DataFrame dataframe = new DataFrame("test");
        dataframe.addStringColumn("Column1");
        dataframe.dropColumn("Column1");
        Assert.assertFalse(dataframe.hasColumn("Column1"));
    }

    @Test
    public void addRowTest() {
        DataFrame dataframe = new DataFrame("test");
        dataframe.addLongColumn("Column1");
        dataframe.addRow(1);
        Assert.assertEquals(1, dataframe.rowCount());
    }

    @Test
    public void addDoubleColumnTest() {
        DataFrame dataframe = new DataFrame("test");
        dataframe.addDoubleColumn("Column1", IntInterval.fromTo(1, 5).collect(i -> i * 1.0).toList());
        Assert.assertEquals(5, dataframe.getColumnAt(0).getSize());
    }

    @Test
    public void addColumn_ValueType_Test() {
        DataFrame dataframe = new DataFrame("test");
        dataframe.addColumn("Column1", ValueType.STRING);
        Assert.assertTrue(dataframe.hasColumn("Column1"));
    }

    @Test(expected = AssertionError.class)
    public void getDecimal_AssertionError_Test() {
        DataFrame dataframe = new DataFrame("test");
        dataframe.addStringColumn("Column1");
        dataframe.getDecimal("Column1", 0);
    }
}
