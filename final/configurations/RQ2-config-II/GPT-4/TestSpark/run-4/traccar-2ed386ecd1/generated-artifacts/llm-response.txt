package org.traccar.protocol;

import io.netty.buffer.ByteBuf;
import io.netty.buffer.Unpooled;
import io.netty.channel.Channel;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;
import org.traccar.Protocol;
import org.traccar.model.Position;

import java.net.SocketAddress;
import java.nio.charset.StandardCharsets;
import java.util.TimeZone;

import static org.junit.Assert.assertFalse;
import static org.junit.Assert.assertTrue;
import static org.mockito.Mockito.*;

@RunWith(MockitoJUnitRunner.class)
public class Gt06ProtocolDecoderTest {

    @Mock
    private Protocol protocol;

    @Mock
    private Channel channel;

    @Mock
    private SocketAddress remoteAddress;

    private Gt06ProtocolDecoder decoder;

    @Test
    public void decodeGpsWithLengthTest() {
        decoder = new Gt06ProtocolDecoder(protocol);
        ByteBuf buf = Unpooled.copiedBuffer("1234567890", StandardCharsets.UTF_8);
        Position position = new Position();

        boolean result = Gt06ProtocolDecoder.decodeGps(position, buf, true, TimeZone.getDefault());

        assertTrue(result);
    }

    @Test
    public void decodeGpsWithoutLengthTest() {
        decoder = new Gt06ProtocolDecoder(protocol);
        ByteBuf buf = Unpooled.copiedBuffer("1234567890", StandardCharsets.UTF_8);
        Position position = new Position();

        boolean result = Gt06ProtocolDecoder.decodeGps(position, buf, false, TimeZone.getDefault());

        assertFalse(result);
    }

    @Test
    public void decodeGpsWithMultipleSettingsTest() {
        decoder = new Gt06ProtocolDecoder(protocol);
        ByteBuf buf = Unpooled.copiedBuffer("1234567890", StandardCharsets.UTF_8);
        Position position = new Position();

        boolean result = Gt06ProtocolDecoder.decodeGps(position, buf, true, true, true, TimeZone.getDefault());

        assertTrue(result);
    }

    @Test
    public void decodeTest() throws Exception {
        decoder = new Gt06ProtocolDecoder(protocol);
        ByteBuf msg = Unpooled.copiedBuffer("1234567890", StandardCharsets.UTF_8);

        Object result = decoder.decode(channel, remoteAddress, msg);

        verify(channel, atLeastOnce()).writeAndFlush(any());
    }
}
