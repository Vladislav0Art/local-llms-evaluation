import java.math.BigDecimal;
import static org.mockito.Mockito.mock;

public class PrettyPrintVisitorTest {

  public void ConstructorWithPrinterTest() {

  public void VisitAssignExprTest() {
    Printer printer = mock(Printer.class);
    PrettyPrintVisitor visitor = new PrettyPrintVisitor(printer);
    AssingExpr expr = new AssingExpr(new VarExpr("x"), new VarExpr("t"));
    visitor.visitAssignExpr(expr);
    Assert.assertEquals("x = t", visitor.exprToString(expr)); 
  }

  @Test
  public void VisitUnaryExprTest() {
    Printer printer = mock(Printer.class);
    PrettyPrintVisitor visitor = new PrettyPrintVisitor(printer);
    UnaryExpr expr = new UnaryExpr(UnaryOp.NOT, new VarExpr("a"));
    visitor.visitUnaryExpr(expr);
    Assert.assertEquals("!a", visitor.exprToString(expr)); 
  }

  @Test
  public void VisitPropertyPathExprTest() {
    Printer printer = mock(Printer.class);
    PrettyPrintVisitor visitor = new PrettyPrintVisitor(printer);
    PropertyPathExpr expr = new PropertyPathExpr("myProperty");
    visitor.visitPropertyPathExpr(expr);
    Assert.assertEquals("myProperty", visitor.exprToString(expr)); 
  }

  @Test
  public void VisitVarExprTest() {
    Printer printer = mock(Printer.class);

  }

  public void VisitIfElseExprTest() {
    Printer printer = mock(Printer.class);

    visitor.visitIfElseExpr(expr);
