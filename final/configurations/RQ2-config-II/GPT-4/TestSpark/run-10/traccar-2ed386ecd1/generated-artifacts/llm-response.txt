import org.junit.Assert;

import org.traccar.model.Position;

import java.nio.charset.StandardCharsets;
import java.util.TimeZone;

public class Gt06ProtocolDecoderTest {

    private Gt06ProtocolDecoder decoder = new Gt06ProtocolDecoder(new Gt06Protocol());

    @Test
    public void decodeGpsNoLengthTest() {
        ByteBuf buf = Unpooled.copiedBuffer("69dc035bf0".getBytes(StandardCharsets.UTF_8));
        Position position = new Position();
        Assert.assertTrue(Gt06ProtocolDecoder.decodeGps(position, buf, false, TimeZone.getDefault()));
    }

    @Test
    public void decodeGpsWithLengthTest() {
        ByteBuf buf = Unpooled.copiedBuffer("69dc035bf0".getBytes(StandardCharsets.UTF_8));

    }

    public void decodeGpsWithLengthSatellitesTest() {
