public void testCheckPackage() {
    // given
    OPFChecker30 opfChecker = new OPFChecker30(mock(ValidationContext.class));
    PackageVocabs packageVocabs = mock(PackageVocabs.class);
    EPUBProfile epubProfile = EPUBProfile.EDUPUB;
    EPUBVersion epubVersion = EPUBVersion.VERSION_3;

    // when
    when(packageVocabs.getEpubVersion()).thenReturn(epubVersion);
    when(packageVocabs.getValidatingProfile()).thenReturn(epubProfile);
    opfChecker.checkPackage(packageVocabs);

    // then

    assertEquals(epubProfile, opfChecker.getEpubProfile());
}
