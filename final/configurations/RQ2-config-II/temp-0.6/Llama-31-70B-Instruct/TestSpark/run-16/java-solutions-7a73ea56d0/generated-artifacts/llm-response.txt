```
package leetcode.medium;

import org.junit.Test;
import static org.junit.Assert.*;

public class OnlineStockSpanTest {

    @Test
    public void nextTest() {
        OnlineStockSpan obj = new OnlineStockSpan();
        assertEquals(1, obj.next(10));
    }

    @Test
    public void calculateSpansTest() {
        OnlineStockSpan obj = new OnlineStockSpan();
        int[] prices = {10, 20, 30, 40, 50};
        int[] expected = {1, 2, 3, 4, 5};
        assertArrayEquals(expected, obj.calculateSpans(prices));
    }

}
```

This code defines two unit tests for the `next` and `calculateSpans` methods in the `OnlineStockSpan` class. The first test checks that `next` returns 1 when a price of 10 is passed in. The second test checks that `calculateSpans` returns the expected array when an array of prices is passed in.
