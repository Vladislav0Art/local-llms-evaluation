```java
import org.jsoup.helper.UrlBuilder;
import org.junit.Assert;
import org.junit.Test;

import java.io.UnsupportedEncodingException;
import java.net.URI;
import java.net.URISyntaxException;
import java.net.URL;

public class UrlBuilderTest {

    @Test
    public void buildUrlTest() throws URISyntaxException {
        // Arrange
        URL url = new URL("http://www.example.com/");
        UrlBuilder urlBuilder = new UrlBuilder(url);

        // Act
        URL result = urlBuilder.build();

        // Assert
        Assert.assertEquals(url, result);
    }

    @Test(expected = UnsupportedEncodingException.class)
    public void appendKeyValTest() throws UnsupportedEncodingException {
        // Arrange
        URL url = new URL("http://www.example.com/");
        UrlBuilder urlBuilder = new UrlBuilder(url);

        // Act
        urlBuilder.appendKeyVal(null);
    }
}
```

In the above code, I have created two test methods:

1. `buildUrlTest()`: This test method checks that the `build()` method returns the correct URL.
2. `appendKeyValTest()`: This test method checks that the `appendKeyVal()` method throws `UnsupportedEncodingException` when `Connection.KeyVal` is `null`.

Please note that I have not created a test method to cover the `appendKeyVal()` method with a non-null `Connection.KeyVal` as I don't have the complete implementation of the `appendKeyVal()` method. If you provide the complete implementation, I can create a test method for it
