```
package org.jsoup.helper;

import org.jsoup.Connection;
import org.junit.Test;
import java.io.UnsupportedEncodingException;
import static org.junit.Assert.*;

public class UrlBuilderTest {
    @Test
    public void buildTest() throws MalformedURLException, URISyntaxException {
        URL inputUrl = new URL("http://www.example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        URL outputUrl = urlBuilder.build();
        assertEquals(new URI("http://www.example.com"), outputUrl.toURI());
    }

    @Test
    public void appendKeyValTest() throws UnsupportedEncodingException {
        URL inputUrl = new URL("http://www.example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        Connection.KeyVal kv = new Connection.KeyVal("key", "value");
        urlBuilder.appendKeyVal(kv);
        assertEquals(kv.key(), "key");
        assertEquals(kv.value(), "value");
    }
}
```

This code tests the `build` method and `appendKeyVal` method of the `UrlBuilder` class.

The `buildTest` method tests the `build` method by creating a `UrlBuilder` object with a sample URL, building the URL, and then verifying that the output URL is correct.

The `appendKeyValTest` method tests the `appendKeyVal` method by creating a `Connection.KeyVal` object with a sample key and value, appending it to the `UrlBuilder`, and then verifying that the key and value are correct.
