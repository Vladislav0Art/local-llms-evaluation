```java
@Test
public void testBuildWithDirectives() {
    AnnotatedElement annotatedElement = mock(AnnotatedElement.class);
    ProcessingElementsContainer processingElementsContainer = mock(ProcessingElementsContainer.class);

    DirectiveRegistry directiveRegistry = mock(DirectiveRegistry.class);
    when(processingElementsContainer.getDirectiveRegistry()).thenReturn(directiveRegistry);

    GraphQLDirective graphQLDirective = mock(GraphQLDirective.class);
    when(directiveRegistry.get(anyString())).thenReturn(graphQLDirective);

    DirectiveJavaAnnotationUtil.getDirectiveAnnotations(annotatedElement).forEach(annotation -> {
        String name = DirectiveJavaAnnotationUtil.getName(annotation);
        when(directiveRegistry.containsKey(name)).thenReturn(true);
    });

    DirectivesBuilder directivesBuilder = new DirectivesBuilder(annotatedElement, processingElementsContainer);
    GraphQLDirective[] directives = directivesBuilder.build();
    assertNotNull(directives);
}
```
