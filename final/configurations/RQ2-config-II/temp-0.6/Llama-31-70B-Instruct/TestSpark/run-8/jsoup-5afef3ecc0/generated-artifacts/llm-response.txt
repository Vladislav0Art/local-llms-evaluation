```
package org.jsoup.helper;

import org.jsoup.Connection;
import org.jsoup.helper.DataUtil;
import org.junit.Test;

import java.io.UnsupportedEncodingException;
import java.net.MalformedURLException;
import java.net.URL;
import java.net.URLEncoder;

import static org.junit.Assert.*;
import static org.mockito.Mockito.when;

public class UrlBuilderTest {

    @Test
    public void buildUrlTest() throws MalformedURLException {
        URL inputUrl = new URL("https://www.example.com/");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        URL buildUrl = urlBuilder.build();
        assertEquals(inputUrl, buildUrl);
    }

    @Test
    public void appendKeyValTest() throws UnsupportedEncodingException {
        URL inputUrl = new URL("https://www.example.com/");
        Connection.KeyVal kv = new Connection.KeyVal("name", "value");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        urlBuilder.appendKeyVal(kv);
        assertEquals("value", kv.value());
    }

}
```

This code tests the `build` and `appendKeyVal` methods of the `UrlBuilder` class. The `buildUrlTest` method checks if the `build` method returns the same URL as the input URL. The `appendKeyValTest` method checks if the `appendKeyVal` method correctly assigns the value to the `Connection.KeyVal` object.
