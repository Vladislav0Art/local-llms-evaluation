```java
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import java.util.ArrayList;
import java.util.List;

import static org.junit.Assert.assertEquals;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class PropertyListBuilderCreateTest {

    @Mock
    private Property<Object> property;

    @Test
    public void create_validProperty_createsList() {
        PropertyListBuilder propertyListBuilder = new PropertyListBuilder();
        List<Property<?>> properties = new ArrayList<>();
        properties.add(property);
        when(property.getProperties()).thenReturn(properties);
        List<Property<?>> result = propertyListBuilder.create();

        assertEquals(1, result.size());
        assertEquals(property, result.get(0));
    }
}
```
