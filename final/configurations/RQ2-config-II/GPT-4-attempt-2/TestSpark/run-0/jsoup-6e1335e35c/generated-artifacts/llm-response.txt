import org.jsoup.nodes.Element;
import org.jsoup.nodes.Node;
import org.jsoup.parser.Tag;
import org.jsoup.select.Elements;
import org.junit.Test;

import static org.junit.Assert.*;

public class ElementTest {

    @Test
    public void tagNameConstructorTest() {
        Element element = new Element("p");
        assertEquals("p", element.tagName());
    }

    @Test
    public void tagBaseUriAttributesConstructorTest() {
        Tag tag = Tag.valueOf("div");
        Element element = new Element(tag, "http://example.com/", null);
        assertEquals("div", element.tagName());
        assertEquals("http://example.com/", element.baseUri());
    }

    @Test
    public void hasChildNodesTest() {
        Element parent = new Element("div");
        Element child = new Element("p");
        parent.appendChild(child);
        assertTrue(parent.hasChildNodes());
    }

    @Test
    public void hasAttributesTest() {
        Element element = new Element("p");
        element.attr("class", "test");
        assertTrue(element.hasAttributes());
    }

    @Test
    public void idTest() {
        Element element = new Element("p");
        element.id("sampleId");
        assertEquals("sampleId", element.id());
    }

    @Test
    public void attrTest() {
        Element element = new Element("p");
        element.attr("class", "test");
        assertEquals("test", element.attr("class"));
    }

    @Test
    public void appendChildTest() {
        Element parent = new Element("div");
        Element child = new Element("p");
        parent.appendChild(child);
        assertEquals(1, parent.childNodeSize());
    }

    @Test
    public void selectTest() {
        Element parent = new Element("div");
        Element child = new Element("p");
        parent.appendChild(child);
        Elements selected = parent.select("p");
        assertEquals(1, selected.size());
    }

    @Test
    public void addChildTest() {
        Element parent = new Element("div");
        parent.appendElement("p");
        assertEquals(1, parent.childNodeSize());
    }

    @Test
    public void emptyTest() {
        Element parent = new Element("div");
        parent.appendElement("p");
        parent.empty();
        assertEquals(0, parent.childNodeSize());
    }

    @Test
    public void cloneTest() {
        Element parent = new Element("div");
        parent.appendElement("p");
        Element clone = parent.clone();
        assertEquals(parent.tag(), clone.tag());
        assertEquals(parent.childNodeSize(), clone.childNodeSize());
    }

    @Test
    public void textTest() {
        Element element = new Element("p").text("this is a test");
        assertEquals("this is a test", element.text());
    }

    @Test
    public void removeClassTest() {
        Element element = new Element("p").addClass("test").removeClass("test");
        assertFalse(element.hasClass("test"));
    }

    @Test
    public void clearAttributesTest() {
        Element element = new Element("p").attr("class", "test").clearAttributes();
        assertNull(element.attr("class"));
    }

    // Additional tests can be written for other methods in the class
}
