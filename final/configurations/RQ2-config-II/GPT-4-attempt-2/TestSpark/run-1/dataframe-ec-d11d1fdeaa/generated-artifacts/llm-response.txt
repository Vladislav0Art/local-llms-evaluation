import io.github.vmzakharov.ecdataframe.dataframe.DataFrame;
import io.github.vmzakharov.ecdataframe.dataframe.DfColumn;
import io.github.vmzakharov.ecdataframe.dsl.Expr;
import io.github.vmzakharov.ecdataframe.dsl.value.StringValue;
import io.github.vmzakharov.ecdataframe.dsl.value.ValueType;
import org.eclipse.collections.impl.factory.Lists;
import org.junit.Assert;
import org.junit.Test;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.math.BigDecimal;

public class DataFrameTest {

    @Test
    public void addStringColumnTest() {
        DataFrame df = new DataFrame("TestFrame");
        df.addStringColumn("TestColumn");
        Assert.assertEquals("TestColumn", df.getColumnNamed("TestColumn").getName());
    }

    @Test
    public void getColumnAtTest() {
        DataFrame df = new DataFrame("TestFrame");
        df.addStringColumn("TestColumn");
        DfColumn column = df.getColumnAt(0);
        Assert.assertEquals("TestColumn", column.getName());
    }

    @Test
    public void rowCountEmptyTest() {
        DataFrame df = new DataFrame("TestFrame");
        int count = df.rowCount();
        Assert.assertEquals(0, count);
    }

    @Test
    public void addRowTest() {
        DataFrame df = new DataFrame("TestFrame");
        df.addStringColumn("TestColumn");
        df.addRow(Lists.immutable.of(Expr.stringLiteral("TestValue")));
        Assert.assertEquals("TestValue", df.getObject("TestColumn", 0));
    }

    @Test
    public void addRowValuesTest() {
        DataFrame df = new DataFrame("TestFrame");
        df.addStringColumn("TestColumn");
        df.addRow("TestValue");
        Assert.assertEquals("TestValue", df.getObject("TestColumn", 0));
    }

    @Test
    public void columnCountTest() {
        DataFrame df = new DataFrame("TestFrame");
        df.addStringColumn("TestColumn1");
        df.addStringColumn("TestColumn2");
        int count = df.columnCount();
        Assert.assertEquals(2, count);
    }

    @Test
    public void getNameTest() {
        DataFrame df = new DataFrame("TestFrame");
        String name = df.getName();
        Assert.assertEquals("TestFrame", name);
    }

    @Test
    public void addColumnTest() {
        DataFrame df = new DataFrame("TestFrame");
        df.addColumn("TestColumn", ValueType.STRING);
        Assert.assertEquals("TestColumn", df.getColumnNamed("TestColumn").getName());
    }

    @Test
    public void hasColumnTest() {
        DataFrame df = new DataFrame("TestFrame");
        df.addStringColumn("TestColumn");
        Assert.assertTrue(df.hasColumn("TestColumn"));
    }

    @Test
    public void sealTest() {
        DataFrame df = new DataFrame("TestFrame");
        df.seal();
        Assert.assertTrue(df.isSealed());
    }

    @Test
    public void addDoubleColumnWithExpressionTest() {
        DataFrame df = new DataFrame("TestFrame");
        df.addDoubleColumn("TestColumn", "2.0");
        Assert.assertEquals(2.0, df.getDouble("TestColumn", 0), 0);
    }

    @Test
    public void addDateColumnValuesTest() {
        DataFrame df = new DataFrame("TestFrame");
        df.addDateColumn("TestColumn", Lists.immutable.of(LocalDate.now()));
        Assert.assertEquals(LocalDate.now(), df.getDate("TestColumn", 0));
    }

    @Test
    public void addDateTimeColumnExpressionTest() {
        DataFrame df = new DataFrame("TestFrame");
        df.addDateTimeColumn("TestColumn", "now");
        Assert.assertEquals(LocalDateTime.now(), df.getDateTime("TestColumn", 0));
    }

    @Test
    public void addDecimalColumnValuesTest() {
        DataFrame df = new DataFrame("TestFrame");
        df.addDecimalColumn("TestColumn", Lists.immutable.of(new BigDecimal("10.0")));
        Assert.assertEquals(new BigDecimal("10.0"), df.getDecimal("TestColumn", 0));
    }

    @Test
    public void isEmptyTest() {
        DataFrame df = new DataFrame("TestFrame");
        Assert.assertTrue(df.isEmpty());
    }

    @Test
    public void isNotEmptyTest() {
        DataFrame df = new DataFrame("TestFrame");
        df.addStringColumn("TestColumn");
        df.addRow(Lists.immutable.of(Expr.stringLiteral("TestValue")));
        Assert.assertTrue(df.isNotEmpty());
    }

    @Test
    public void createIndexTest() {
        DataFrame df = new DataFrame("TestFrame");
        df.addStringColumn("TestColumn");
        df.addRow(Lists.immutable.of(Expr.stringLiteral("TestValue")));
        df.createIndex("TestIndex", Lists.immutable.of("TestColumn"));
        Assert.assertNotNull(df.index("TestIndex"));
    }

    @Test
    public void dropIndexTest() {
        DataFrame df = new DataFrame("TestFrame");
        df.addStringColumn("TestColumn");
        df.addRow(Lists.immutable.of(Expr.stringLiteral("TestValue")));
        df.createIndex("TestIndex", Lists.immutable.of("TestColumn"));
        df.dropIndex("TestIndex");
        Assert.assertNull(df.index("TestIndex"));
    }

    @Test
    public void unionTest() {
        DataFrame df1 = new DataFrame("TestFrame1");
        df1.addStringColumn("TestColumn");
        df1.addRow(Lists.immutable.of(Expr.stringLiteral("TestValue1")));
        DataFrame df2 = new DataFrame("TestFrame2");
        df2.addStringColumn("TestColumn");
        df2.addRow(Lists.immutable.of(Expr.stringLiteral("TestValue2")));
        DataFrame dfUnion = df1.union(df2);
        Assert.assertEquals(2, dfUnion.rowCount());
    }
}
