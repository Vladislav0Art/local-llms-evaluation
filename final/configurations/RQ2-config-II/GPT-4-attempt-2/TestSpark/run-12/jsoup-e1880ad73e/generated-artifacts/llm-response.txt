import org.junit.Test;
import org.jsoup.Connection;
import org.jsoup.helper.UrlBuilder;

import java.io.UnsupportedEncodingException;
import java.net.MalformedURLException;
import java.net.URL;
import java.nio.charset.StandardCharsets;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNotNull;
import static org.mockito.Mockito.mock;
import static org.mockito.Mockito.when;

public class UrlBuilderTest {

	@Test
	public void UrlBuilder_ConstructedWithUrlTest() throws MalformedURLException {
		URL testUrl = new URL("http://example.com");
		UrlBuilder builder = new UrlBuilder(testUrl);
		assertNotNull(builder);
	}

	@Test
	public void build_ReturnsURLTest() throws MalformedURLException {
		URL testUrl = new URL("http://example.com");
		UrlBuilder builder = new UrlBuilder(testUrl);
		assertNotNull(builder.build());
		assertEquals(testUrl, builder.build());
	}

	@Test(expected = NullPointerException.class)
	public void appendKeyValThrowsNullPointerExceptionWhenNullPassedTest() throws UnsupportedEncodingException {
		URL testUrl = new URL("http://example.com");
		UrlBuilder builder = new UrlBuilder(testUrl);
		builder.appendKeyVal(null);
	}

	@Test
	public void appendKeyVal_AppendsValidKeyValueTest() throws UnsupportedEncodingException, MalformedURLException {
		URL testUrl = new URL("http://example.com");
		UrlBuilder builder = new UrlBuilder(testUrl);	
		Connection.KeyVal kv = mock(Connection.KeyVal.class);
		when(kv.key()).thenReturn("Key");
		when(kv.value()).thenReturn("Value");
		builder.appendKeyVal(kv);
		assertEquals("http://example.com?Key=Value", builder.build().toString());
	}
}
