import org.junit.Assert;
import org.junit.Test;
import org.jsoup.Connection;
import org.jsoup.helper.UrlBuilder;

import java.net.MalformedURLException;
import java.net.URL;
import java.io.UnsupportedEncodingException;

public class UrlBuilderTest {

    @Test
    public void UrlBuilderValidUrlTest() throws MalformedURLException {
        URL inputUrl = new URL("https://www.example.com");
        UrlBuilder builder = new UrlBuilder(inputUrl);
        Assert.assertNotNull(builder);
    }

    @Test(expected = NullPointerException.class)
    public void UrlBuilderNullUrlTest() {
        UrlBuilder builder = new UrlBuilder(null);
    }

    @Test
    public void UrlBuilderBuildValidUrlTest() throws MalformedURLException {
        URL inputUrl = new URL("https://www.example.com");
        UrlBuilder builder = new UrlBuilder(inputUrl);
        URL outputUrl = builder.build();
        Assert.assertNotNull(outputUrl);
        Assert.assertEquals(inputUrl, outputUrl);
    }

    @Test
    public void UrlBuilderBuildInvalidUrlTest() throws MalformedURLException {
        URL inputUrl = new URL("://www.invalid.com");
        UrlBuilder builder = new UrlBuilder(inputUrl);
        URL outputUrl = builder.build();
        Assert.assertNull(outputUrl);
    }

    @Test
    public void appendKeyValValidKeyValTest() throws UnsupportedEncodingException {
        URL inputUrl = new URL("https://www.example.com");
        UrlBuilder builder = new UrlBuilder(inputUrl);
        Connection.KeyVal kv = new Connection.KeyVal("key", "value");
        builder.appendKeyVal(kv);
        URL appendedUrl = builder.build();
        Assert.assertNotNull(appendedUrl);
        Assert.assertEquals("https://www.example.com?key=value", appendedUrl);
    }

    @Test(expected = UnsupportedEncodingException.class)
    public void appendKeyValInvalidKeyValTest() throws UnsupportedEncodingException {
        URL inputUrl = new URL("https://www.example.com");
        UrlBuilder builder = new UrlBuilder(inputUrl);
        Connection.KeyVal kv = new Connection.KeyVal("key", "\\uFFFF");
        builder.appendKeyVal(kv);
    }

}
