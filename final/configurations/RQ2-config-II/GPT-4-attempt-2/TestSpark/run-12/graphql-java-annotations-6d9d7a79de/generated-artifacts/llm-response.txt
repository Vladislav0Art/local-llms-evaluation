```java
import graphql.annotations.processor.ProcessingElementsContainer;
import graphql.annotations.processor.retrievers.fieldBuilders.DirectivesBuilder;
import graphql.annotations.processor.util.DirectiveJavaAnnotationUtil;
import graphql.schema.GraphQLDirective;
import org.junit.Test;
import org.mockito.Mockito;

import java.lang.reflect.Method;

import static org.junit.Assert.*;
import static org.mockito.Mockito.when;

public class DirectivesBuilderTest {

    @Test
    public void buildMethod_Test_WhenDirectivesArePresent() {
        // given
        DirectiveJavaAnnotationUtil utilMock = Mockito.mock(DirectiveJavaAnnotationUtil.class);
        ProcessingElementsContainer containerMock = Mockito.mock(ProcessingElementsContainer.class);
        when(containerMock.getDirectiveJavaAnnotationUtil()).thenReturn(utilMock);
        when(utilMock.getDirectiveInfo(any(), any())).thenReturn(new GraphQLDirective[]{});

        DirectivesBuilder builder = new DirectivesBuilder(SomeClassWithDirectives.class, containerMock);

        // when
        GraphQLDirective[] result = builder.build();

        // then
        assertNotNull(result);
        assertEquals(0, result.length);
    }

    @Test(expected = SomeException.class)
    public void buildMethod_Test_WhenNoDirectivesArePresent() {
        // given
        DirectiveJavaAnnotationUtil utilMock = Mockito.mock(DirectiveJavaAnnotationUtil.class);
        ProcessingElementsContainer containerMock = Mockito.mock(ProcessingElementsContainer.class);
        when(containerMock.getDirectiveJavaAnnotationUtil()).thenReturn(utilMock);
        when(utilMock.getDirectiveInfo(any(), any())).thenThrow(new SomeException());

        DirectivesBuilder builder = new DirectivesBuilder(SomeClassWithoutDirectives.class, containerMock);

        // when
        builder.build();

        // then exception expected
    }
}
```
In both test cases I'm using Mockito to mock out the ProcessingElementsContainer, which is then used in the constructor of the DirectivesBuilder. The mocked method getDirectiveJavaAnnotationUtil will then be called in the build method of DirectivesBuilder. 

In the first test case we are simulating that the class has directives that can be processed. Therefore, we return an empty array of GraphQLDirectives. 

The second test case is simulating a situation where some exception is thrown during the process of retrieving information. For this, we are throwing a SomeException from the mocked method in the utilMock.
 
Please notice that `SomeClassWithDirectives.class` and `SomeClassWithoutDirectives.class` are placeholders for classes with and without directives respectively. Also, SomeException is a placeholder and should be replaced with a real exception class that might be thrown in the function being tested.
