import org.junit.Assert;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mockito;

import java.util.List;

public class TokenizerTest {

    @Test
    public void tokenizerConstructorTest() {
        String expressionString = "2+2";
        ExpressionConfiguration configuration = new ExpressionConfiguration();
        new Tokenizer(expressionString, configuration);
    }

    @Test
    public void parseValidExpressionTest() throws ParseException {
        String expressionString = "2+2";
        ExpressionConfiguration configuration = new ExpressionConfiguration();
        Tokenizer tokenizer = new Tokenizer(expressionString, configuration);

        List<Token> result = tokenizer.parse();

        Assert.assertEquals(result.size(), 3); 
        Assert.assertEquals(result.get(0).getValue(), "2");
        Assert.assertEquals(result.get(1).getValue(), "+");
        Assert.assertEquals(result.get(2).getValue(), "2");
    }

    @Test(expected = ParseException.class)
    public void parseInvalidExpressionTest() throws ParseException {
        String expressionString = "2+";
        ExpressionConfiguration configuration = new ExpressionConfiguration();
        Tokenizer tokenizer = new Tokenizer(expressionString, configuration);

        tokenizer.parse();
    }

    @Test
    public void parseStringLiteralToValidStringTest() throws ParseException {
        String expressionString = "'Hello, World!'";
        ExpressionConfiguration configuration = new ExpressionConfiguration();
        Tokenizer tokenizer = new Tokenizer(expressionString, configuration);

        Token result = tokenizer.parseStringLiteral();

        Assert.assertNotNull(result); 
        Assert.assertEquals(result.getValue(), "'Hello, World!'");
    }

    @Test(expected = ParseException.class)
    public void parseStringLiteralToInvalidStringTest() throws ParseException {
        String expressionString = "'Hello, World";
        ExpressionConfiguration configuration = new ExpressionConfiguration();
        Tokenizer tokenizer = new Tokenizer(expressionString, configuration);

        tokenizer.parseStringLiteral();
    }
