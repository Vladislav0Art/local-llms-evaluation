    }

        Assert.assertTrue(formatter.isInitialized());

    @Test
    public void doFormatEmptyCodeTest() throws IOException {

        Assert.assertEquals("", result);
    }

    @Test
    public void doFormatNonEmptyCodeWithAutoLineEndingTest() throws IOException {
