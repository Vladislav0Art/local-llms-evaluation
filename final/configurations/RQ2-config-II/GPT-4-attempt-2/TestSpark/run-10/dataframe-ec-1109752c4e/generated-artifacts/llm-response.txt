import io.github.vmzakharov.ecdataframe.util.CollectingPrinter;
import org.junit.Assert;
import org.junit.Test;

public class PrettyPrintVisitorTest {

    @Test
    public void exprToStringTest() {
        Expression e = new BinaryExpr(Operation.ADDITION, new ConstExpr(1), new ConstExpr(2));
        String res = PrettyPrintVisitor.exprToString(e);
        
        Assert.assertEquals("1 + 2", res);
    }
    
    @Test
    public void visitAssignExprTest() {
        CollectingPrinter collectingPrinter = new CollectingPrinter();
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(collectingPrinter);
        AssingExpr expr = new AssingExpr(new VarExpr("x"), new ConstExpr(10));
        visitor.visitAssignExpr(expr);
        
        Assert.assertEquals("x := 10", collectingPrinter.getBufferedPrint());
    }
    
    @Test
    public void visitBinaryExprTest() {
        CollectingPrinter collectingPrinter = new CollectingPrinter();
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(collectingPrinter);
        BinaryExpr expr = new BinaryExpr(Operation.ADDITION, new ConstExpr(1), new ConstExpr(2));
        visitor.visitBinaryExpr(expr);
        
        Assert.assertEquals("1 + 2", collectingPrinter.getBufferedPrint());
    }
    
    @Test
    public void visitUnaryExprTest() {
        CollectingPrinter collectingPrinter = new CollectingPrinter();
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(collectingPrinter);
        UnaryExpr expr = new UnaryExpr(Operation.NEGATIVE, new ConstExpr(5));
        visitor.visitUnaryExprTest(expr);
        
        Assert.assertEquals("-5", collectingPrinter.getBufferedPrint());
    }
    
    @Test
    public void visitConstExprTest() {
        CollectingPrinter collectingPrinter = new CollectingPrinter();
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(collectingPrinter);
        ConstExpr expr = new ConstExpr(new Value(77));
        visitor.visitConstExpr(expr);
        
        Assert.assertEquals("77", collectingPrinter.getBufferedPrint());
    }
    
    @Test
    public void visitFunctionCallExprTest() {
        CollectingPrinter collectingPrinter = new CollectingPrinter();
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(collectingPrinter);
        FunctionCallExpr expr = new FunctionCallExpr("foo");
        visitor.visitFunctionCallExpr(expr);
        
        Assert.assertEquals("foo", collectingPrinter.getBufferedPrint());
        
    }
