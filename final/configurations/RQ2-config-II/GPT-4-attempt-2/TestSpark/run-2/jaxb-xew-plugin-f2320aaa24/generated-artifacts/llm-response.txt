import com.sun.tools.xjc.model.CPropertyInfo;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import java.io.IOException;

import static org.junit.Assert.assertEquals;
import static org.mockito.Mockito.mock;
import static org.mockito.Mockito.times;

@RunWith(MockitoJUnitRunner.class)
public class XmlElementWrapperPluginTest {

    @Test(expected = ClassNotFoundException.class)
    public void runInternalWhenClassNotFoundTest() throws Exception {
        Outline outline = Mockito.mock(Outline.class);
        Mockito.when(Ring.get(Options.class)).thenReturn(new Options());
        XmlElementWrapperPlugin xmlElementWrapperPlugin = new XmlElementWrapperPlugin();
        xmlElementWrapperPlugin.runInternal(outline);
    }

    @Test(expected = IOException.class)
    public void runInternalWhenIOExceptionOccursTest() throws Exception {
        Outline outline = Mockito.mock(Outline.class);
        Mockito.when(Ring.get(Options.class)).thenReturn(new Options());
        Mockito.when(Ring.get(CPropertyInfo.class)).thenReturn(mock(CPropertyInfo.class));
        XmlElementWrapperPlugin xmlElementWrapperPlugin = new XmlElementWrapperPlugin();
        xmlElementWrapperPlugin.runInternal(outline);
    }

    @Test
    public void runInternalMethodUnderTestCheckTest() throws ClassNotFoundException, IOException {
        Outline outline = Mockito.mock(Outline.class);
        XmlElementWrapperPlugin plugin = Mockito.spy(XmlElementWrapperPlugin.class);
        plugin.runInternal(outline);
        Mockito.verify(plugin, times(1)).runInternal(outline);
    }

    @Test
    public void testPluginGetterAndSetter() {
        XmlElementWrapperPlugin xmlElementWrapperPlugin = new XmlElementWrapperPlugin();
        String optionName = "testOption";
        xmlElementWrapperPlugin.setOptionName(optionName);
        assertEquals(optionName, xmlElementWrapperPlugin.getOptionName());
    }
}
