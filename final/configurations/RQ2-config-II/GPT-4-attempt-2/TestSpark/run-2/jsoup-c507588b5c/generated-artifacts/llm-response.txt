import org.jsoup.nodes.Document;
import org.jsoup.nodes.TextNode;
import org.junit.Assert;
import org.junit.Test;

import static org.hamcrest.CoreMatchers.*;

public class TextNodeTest {
    
    @Test
    public void TextNodeNonEmptyTextTest() {
        TextNode node = new TextNode("Test");
        Assert.assertThat(node.text(), is("Test"));
    }

    @Test
    public void TextNodeEmptyTextTest() {
        TextNode node = new TextNode("");
        Assert.assertThat(node.text(), is(""));
    }

    @Test
    public void nodeNameTest() {
        TextNode node = new TextNode("Test");
        Assert.assertThat(node.nodeName(), is("#text"));
    }

    @Test
    public void textStringTest() {
        TextNode node = new TextNode("First");
        node.text("Second");
        Assert.assertThat(node.text(), is("Second"));
    }

    @Test
    public void isBlankTrueTest() {
        TextNode node = new TextNode("");
        Assert.assertTrue(node.isBlank());
    }

    @Test
    public void isBlankFalseTest() {
        TextNode node = new TextNode("Non-empty");
        Assert.assertFalse(node.isBlank());
    }

    @Test
    public void splitTextTest() {
        TextNode node = new TextNode("TestNode");
        TextNode splitted = node.splitText(4);
        Assert.assertThat(node.getWholeText(), is("Test"));
        Assert.assertThat(splitted.getWholeText(), is("Node"));
    }

    @Test
    public void cloneTest() {
        TextNode original = new TextNode("Test");
        TextNode cloned = original.clone();
        Assert.assertNotSame(original, cloned);
    }

    @Test
    public void createFromEncodedTest() {
        TextNode node = TextNode.createFromEncoded("&quot;", true);
        Assert.assertThat(node.getWholeText(), is("\""));
    }

    @Test
    public void normaliseWhitespaceTest() {
        String input = "  Test   String ";
        String expected = "Test String";
        Assert.assertEquals(expected, TextNode.normaliseWhitespace(input));
    }

    @Test
    public void stripLeadingWhitespaceTest() {
        String input = "    Strip this";
        String expected = "Strip this";
        Assert.assertEquals(expected, TextNode.stripLeadingWhitespace(input));
    }

    @Test
    public void lastCharIsWhitespaceTest() {
        StringBuilder builder = new StringBuilder("Test ");
        Assert.assertTrue(TextNode.lastCharIsWhitespace(builder));
    }
}
