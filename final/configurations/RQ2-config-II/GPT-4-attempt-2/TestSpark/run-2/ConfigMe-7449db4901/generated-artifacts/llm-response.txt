import ch.jalu.configme.configurationdata.CommentsConfiguration;
import org.junit.Test;
import java.util.Arrays;
import java.util.List;
import java.util.Map;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertTrue;

public class CommentsConfigurationTest{

    @Test
    public void setCommentSingleLineTest(){
        CommentsConfiguration config = new CommentsConfiguration();
        config.setComment("path1","commentLine1");
        Map<String, List<String>> comments = config.getAllComments();
        assertEquals(1,comments.size());
        assertTrue(comments.get("path1").contains("commentLine1"));
    }

    @Test
    public void setCommentMultipleLinesTest(){
        CommentsConfiguration config = new CommentsConfiguration();
        config.setComment("path1","commentLine1","commentLine2");
        Map<String, List<String>> comments = config.getAllComments();
        assertEquals(2,comments.get("path1").size());
        assertTrue(comments.get("path1").containsAll(Arrays.asList("commentLine1","commentLine2")));
    }

    @Test
    public void setCommentOverrideTest(){
        CommentsConfiguration config = new CommentsConfiguration();
        config.setComment("path1","commentLine1","commentLine2");
        config.setComment("path1","commentLine3");
        Map<String, List<String>> comments = config.getAllComments();
        assertEquals(1,comments.get("path1").size());
        assertTrue(comments.get("path1").contains("commentLine3"));
    }
    
    @Test
    public void getAllCommentsTest() {
        Map<String, List<String>> initialComments = new HashMap<>();
        initialComments.put("path4", Arrays.asList("comment4"));
        CommentsConfiguration config = new CommentsConfiguration(initialComments);
        assertEquals(initialComments, config.getAllComments());
    }
}
