        Tag tag = Tag.valueOf("div");

        Tag tag = Tag.valueOf("div");
        assertEquals("div", tag.normalName());

        assertNotNull(tag);
    }

        Tag.valueOf(null);
    }

    public void valueOfWithSettingExistingTagTest() {

        assertNotNull(tag);
    }

    public void valueOfWithSettingNullTagTest() {
        Tag.valueOf(null, ParseSettings.htmlDefault);
    }

    public void isBlockDivTagTest() {

        assertTrue(tag.isBlock());

        assertTrue(tag.formatAsBlock());
    }

    }
