import org.jsoup.Connection;
import org.jsoup.helper.UrlBuilder;
import org.junit.Assert;
import org.junit.Test;

import java.io.UnsupportedEncodingException;
import java.net.MalformedURLException;
import java.net.URL;

public class UrlBuilderTest {

    @Test
    public void UrlBuilderConstructTest() throws MalformedURLException {
        URL inputUrl = new URL("https://google.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        Assert.assertNotNull(urlBuilder);
    }

    @Test
    public void buildURLTest() throws MalformedURLException {
        URL inputUrl = new URL("https://google.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        URL outputUrl = urlBuilder.build();
        Assert.assertNotNull(outputUrl);
        Assert.assertEquals("https://google.com", outputUrl.toString());
    }

    @Test
    public void appendKeyValTest() throws MalformedURLException, UnsupportedEncodingException {
        URL inputUrl = new URL("https://google.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        Connection.KeyVal kv = Connection.KeyVal.create("key", "value");
        urlBuilder.appendKeyVal(kv);
        URL outputUrl = urlBuilder.build();
        Assert.assertNotNull(outputUrl);
        Assert.assertEquals("https://google.com?key=value", outputUrl.toString());
    }
    
    @Test(expected = MalformedURLException.class)
    public void UrlBuilderConstructTestWithInvalidURL() throws MalformedURLException {
        URL inputUrl = new URL("htp://invalid-url");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
    }

    @Test(expected = UnsupportedEncodingException.class)
    public void appendKeyValTestWithInvalidEncoding() throws MalformedURLException, UnsupportedEncodingException {
        URL inputUrl = new URL("https://google.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        //Create key value pair with invalid encoding characters
        Connection.KeyVal kv = Connection.KeyVal.create("key", "valu\uFFFFe");
        urlBuilder.appendKeyVal(kv);
    }
}
