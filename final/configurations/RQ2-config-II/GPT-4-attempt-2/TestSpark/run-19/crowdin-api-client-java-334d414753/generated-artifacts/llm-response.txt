import com.crowdin.client.projectsgroups.model.Project;
import com.crowdin.client.sourcefiles.model.ExportOptions;

import com.crowdin.client.core.http.exceptions.CrowdinApiException;
import com.crowdin.client.core.http.exceptions.HttpException;
import com.crowdin.client.core.http.exceptions.HttpBadRequestException;
import com.crowdin.client.core.http.impl.json.JacksonJsonTransformer;
import org.junit.Assert;
import org.junit.Test;

public class JacksonJsonTransformerTest {
    @Test
    public void parseValidJsonTest() {
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        String json = "{\"name\":\"test\", \"id\":1}";
        Project project = transformer.parse(json, Project.class);
        Assert.assertEquals("test", project.getName());
        Assert.assertEquals(1, project.getId());
    }

    @Test(expected = CrowdinApiException.class)
    public void parseInvalidJsonTest() {
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        String json = "{\"name:\"test\", \"id\":1}";
        Project project = transformer.parse(json, Project.class);
    }

    @Test
    public void convertObjectToJsonTest() {
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        Project project = new Project();
        project.setName("test");
        project.setId(1);
        String json = transformer.convert(project);
        Assert.assertEquals("{\"name\":\"test\", \"id\":1}", json);
    }

    @Test(expected = HttpException.class)
    public void convertInvalidObjectToJsonTest() {
        JacksonJsonTransformer transformer = new JacksonJsonTransformer();
        ExportOptions invalidObject = new ExportOptions();
        String json = transformer.convert(invalidObject);
    }
}
