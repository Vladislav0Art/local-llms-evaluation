import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertFalse;
import static org.junit.Assert.assertNotNull;
import static org.junit.Assert.assertTrue;
import org.jsoup.nodes.Attribute;
import org.jsoup.nodes.Element;
import org.junit.Test;

public class SafelistTest {

    @Test
    public void noneTest() {
        Safelist safelist = Safelist.none();
        assertNotNull(safelist);
    }

    @Test
    public void simpleTextTest() {
        Safelist safelist = Safelist.simpleText();
        assertNotNull(safelist);
    }

    @Test
    public void basicTest() {
        Safelist safelist = Safelist.basic();
        assertNotNull(safelist);
    }

    @Test
    public void basicWithImagesTest() {
        Safelist safelist = Safelist.basicWithImages();
        assertNotNull(safelist);
    }

    @Test
    public void relaxedTest() {
        Safelist safelist = Safelist.relaxed();
        assertNotNull(safelist);
    }

    @Test
    public void addTagsTest() {
        Safelist safelist = new Safelist();
        safelist.addTags("p");
        assertTrue(safelist.isSafeTag("p"));
    }

    @Test
    public void removeTagsTest() {
        Safelist safelist = new Safelist();
        safelist.addTags("p");
        safelist.removeTags("p");
        assertFalse(safelist.isSafeTag("p"));
    }

    @Test
    public void addAttributesTest() {
        Safelist safelist = new Safelist();
        Element testElement = new Element("p");
        Attribute testAttribute = new Attribute("class", "test");
        safelist.addAttributes("p", "class");
        assertTrue(safelist.isSafeAttribute("p", testElement, testAttribute));
    }

    @Test
    public void addEnforcedAttributeTest() {
        Safelist safelist = new Safelist();
        safelist.addEnforcedAttribute("p", "class", "test");
        assertEquals("test", safelist.getEnforcedAttributes("p").get("class"));
    }

    @Test
    public void removeEnforcedAttributeTest() {
        Safelist safelist = new Safelist();
        safelist.addEnforcedAttribute("p", "class", "test");
        safelist.removeEnforcedAttribute("p", "class");
        assertFalse(safelist.getEnforcedAttributes("p").hasKey("class"));
    }

    @Test
    public void preserveRelativeLinksTest() {
        Safelist safelist = new Safelist();
        safelist.preserveRelativeLinks(true);
        Element testElement = new Element("a");
        Attribute testAttribute = new Attribute("href", "/test");
        assertTrue(safelist.isSafeAttribute("a", testElement, testAttribute));
    }

    @Test
    public void addProtocolsTest() {
        Safelist safelist = new Safelist();
        safelist.addProtocols("a", "href", "http");
        Element testElement = new Element("a");
        Attribute testAttribute = new Attribute("href", "http://test");
        assertTrue(safelist.isSafeAttribute("a", testElement, testAttribute));
    }

    @Test
    public void removeProtocolsTest() {
        Safelist safelist = new Safelist();
        safelist.addProtocols("a", "href", "http");
        safelist.removeProtocols("a", "href", "http");
        Element testElement = new Element("a");
        Attribute testAttribute = new Attribute("href", "http://test");
        assertFalse(safelist.isSafeAttribute("a", testElement, testAttribute));
    }
}
