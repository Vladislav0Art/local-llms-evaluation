import io.github.vmzakharov.ecdataframe.dataframe.*;
import org.eclipse.collections.api.list.FixedSizeList;

import org.eclipse.collections.impl.list.mutable.FastList;
import org.junit.Assert;
import org.junit.Test;

import static org.mockito.Mockito.*;

public class DataFrameTest {

    @Test
    public void addRowObjectTest() {
        DataFrame df = new DataFrame("TestDF");
        df.addStringColumn("column1"); 
        df.addDoubleColumn("column2");
        df.addRow("test", 1.23);
        Assert.assertEquals(1, df.rowCount());
    }

    @Test
    public void columnCountTest() {
        DataFrame df = new DataFrame("TestDF");
        df.addStringColumn("column1"); 
        df.addDoubleColumn("column2");
        Assert.assertEquals(2, df.columnCount());
    }

    @Test
    public void getNameTest() {
        DataFrame df = new DataFrame("TestDF");
        Assert.assertEquals("TestDF", df.getName());
    }

    @Test
    public void getColumnNamedTest() {
        DataFrame df = new DataFrame("TestDF");
        df.addStringColumn("column1");
        DfColumn column = df.getColumnNamed("column1");
        Assert.assertEquals("column1", column.getName());
    }

    @Test
    public void addStringColumnTest() {
        DataFrame df = new DataFrame("TestDF");
        df.addStringColumn("column1", FastList.newListWith("value1", "value2"));
        Assert.assertEquals("value1", df.getString("column1", 0));
    }

    @Test
    public void addLongColumnTest() {
        DataFrame df = new DataFrame("TestDF");
        df.addLongColumn("column1", IntLists.immutable.of(1, 2, 3));
        Assert.assertEquals(1, df.getLong("column1", 0));
    }

    @Test
    public void addDoubleColumnTest() {
        DataFrame df = new DataFrame("TestDF");
        df.addDoubleColumn("column1", FastList.newListWith(1.23, 4.56));
        Assert.assertEquals(1.23, df.getDouble("column1", 0), 0.0001);
    }

    @Test
    public void hasColumnTest() {
        DataFrame df = new DataFrame("TestDF");
        df.addStringColumn("column1");
        Assert.assertTrue(df.hasColumn("column1"));
    }

    public void getStringColumnTest() {
        DataFrame df = new DataFrame("TestDF");

    }

    public void getObjectTestRowAndColumn() {

    }

    public void getObjectTestColumnAndRow() {

        df.addRow("hello");

    }

    public void isEmptyTrueTest() {
        DataFrame df = new DataFrame("TestDF");
        Assert.assertTrue(df.isEmpty());

    @Test
    public void isEmptyFalseTest() {
        DataFrame df = new DataFrame("TestDF");
        df.addStringColumn("column1");
        df.addRow("hello");
        Assert.assertFalse(df.isEmpty());
    }
