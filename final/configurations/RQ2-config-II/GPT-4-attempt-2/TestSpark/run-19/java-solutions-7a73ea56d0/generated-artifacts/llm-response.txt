import org.junit.Test;
import static org.junit.Assert.*;

public class OnlineStockSpanTest {

  public void nextSingleValueTest() {
    OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
    int result = onlineStockSpan.next(100);
    assertEquals(1, result);
  }

  @Test
  public void nextMultipleValuesTest() {
    OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
    onlineStockSpan.next(100);
    int result = onlineStockSpan.next(80);
    assertEquals(1, result);
  }

  @Test
  public void nextMultipleValuesIncreasingTest() {
    OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
    onlineStockSpan.next(100);
    int result = onlineStockSpan.next(120);
    assertEquals(2, result);
  }

  @Test
  public void nextZeroValuesTest() {
    OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
    int result = onlineStockSpan.next(0);
    assertEquals(1, result);
  }
  
  @Test
  public void calculateSpansEmptyArrayTest() {
    OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
    int[] result = onlineStockSpan.calculateSpans(new int[]{});
    assertArrayEquals(new int[]{}, result);
  }

  @Test
  public void calculateSpansSingleValueTest() {
    OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
    int[] result = onlineStockSpan.calculateSpans(new int[]{100});
    assertArrayEquals(new int[]{1}, result);
  }

  @Test
  public void calculateSpansMultipleValuesTest() {
    OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
    int[] result = onlineStockSpan.calculateSpans(new int[]{100, 80, 120});
    assertArrayEquals(new int[]{1, 1, 3}, result);
  }

  @Test
  public void calculateSpansZeroValuesTest() {
    OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
    int[] result = onlineStockSpan.calculateSpans(new int[]{0, 0, 0});
    assertArrayEquals(new int[]{1, 2, 3}, result);
  }  
}
