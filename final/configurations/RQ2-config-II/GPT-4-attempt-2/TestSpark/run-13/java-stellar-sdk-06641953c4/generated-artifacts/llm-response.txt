import org.stellar.sdk.xdr.PublicKey;
import org.stellar.sdk.xdr.SignerKey;
import org.junit.Assert;
import org.junit.Test;

public class KeyPairTest {

  @Test
  public void canSignTest() {
    KeyPair keyPair = KeyPair.random();
    Assert.assertTrue(keyPair.canSign());
  }

  @Test
  public void fromSecretSeedTest() {
    KeyPair keyPair1 = KeyPair.fromSecretSeed("TestKeyPairs");
    KeyPair keyPair2 = KeyPair.fromSecretSeed("TestKeyPairs".toCharArray());
    Assert.assertEquals(keyPair1.getAccountId(), keyPair2.getAccountId());
  }

  @Test
  public void randomTest() {
    KeyPair keyPair = KeyPair.random();
    Assert.assertNotNull(keyPair.getPublicKey());
  }

  @Test
  public void getAccountIdTest() {
    KeyPair keyPair = KeyPair.random();
    Assert.assertNotNull(keyPair.getAccountId());
  }

  @Test
  public void getPublicKeyTest() {
    KeyPair keyPair = KeyPair.random();
    Assert.assertNotNull(keyPair.getPublicKey());
  }

  @Test
  public void getXdrPublicKeyTest() {
    KeyPair keyPair = KeyPair.random();
    Assert.assertNotNull(keyPair.getXdrPublicKey());
  }

  @Test
  public void verifyTest() {
    KeyPair keyPair = KeyPair.random();
    byte[] data = "TestKeyPairs".getBytes();
    byte[] signature = keyPair.sign(data);
    Assert.assertTrue(keyPair.verify(data, signature));
  }

  @Test
  public void fromXdrPublicKeyTest() {
    KeyPair keyPair = KeyPair.random();
    PublicKey publicKey = keyPair.getXdrPublicKey();
    KeyPair result = KeyPair.fromXdrPublicKey(publicKey);
    Assert.assertEquals(keyPair, result);
  }

  @Test
  public void fromXdrSignerKeyTest() {
    KeyPair keyPair = KeyPair.random();
    SignerKey signerKey = keyPair.getXdrSignerKey();
    KeyPair result = KeyPair.fromXdrSignerKey(signerKey);
    Assert.assertEquals(keyPair, result);
  }

  @Test
  public void equalsTest() {
    KeyPair keyPair1 = KeyPair.random();
    KeyPair keyPair2 = KeyPair.random();
    Assert.assertNotEquals(keyPair1, keyPair2);

    keyPair2 = keyPair1;
    Assert.assertEquals(keyPair1, keyPair2);
  }
}
