import io.github.vmzakharov.ecdataframe.dataframe.DataFrame;
import io.github.vmzakharov.ecdataframe.dataframe.DfColumn;
import io.github.vmzakharov.ecdataframe.dsl.value.Value;
import io.github.vmzakharov.ecdataframe.util.ExpressionParserHelper;
import org.eclipse.collections.impl.factory.Lists;
import org.eclipse.collections.impl.tuple.Tuples;
import org.junit.Assert;
import org.junit.Test;
import org.mockito.Mockito;

public class DataFrameTest {

    @Test
    public void addStringColumnColumnNameTest() {
        DataFrame dataFrame = new DataFrame("Test");
        DataFrame result = dataFrame.addStringColumn("Column1");
        Assert.assertEquals("Test", result.getName());
    }

    @Test
    public void addStringColumnColumnNameExpressionAsStringTest() {
        DataFrame dataFrame = new DataFrame("Test");
        String expectedExpression = "x -> x.toString()";
        DataFrame result = dataFrame.addStringColumn("Column1", expectedExpression);
        Assert.assertEquals(expectedExpression, result.getColumnNamed("Column1").getName());
    }

    @Test
    public void addStringColumnColumnNameValuesTest() {
        DataFrame dataFrame = new DataFrame("Test");
        DataFrame result = dataFrame.addStringColumn("Column1", Lists.mutable.of("Value1", "Value2"));
        Assert.assertEquals("Test", result.getName());
    }

    @Test
    public void getColumnNamedTest() {
        DataFrame dataFrame = new DataFrame("Test");
        dataFrame.addStringColumn("Column1");
        DfColumn column = dataFrame.getColumnNamed("Column1");
        Assert.assertEquals("Column1", column.getName());
    }

    @Test
    public void addRowTest() {
        DataFrame dataFrame = new DataFrame("Test");
        dataFrame.addStringColumn("Column1");
        dataFrame.addRow(Lists.mutable.of(ExpressionParserHelper.toValue("Value1")));
        Assert.assertEquals(1, dataFrame.rowCount());
    }

    @Test
    public void asCsvStringTest() {
        DataFrame dataFrame = new DataFrame("Test");
        dataFrame.addStringColumn("Column1");
        dataFrame.addRow(Lists.mutable.of(ExpressionParserHelper.toValue("Value1")));
        String csv = dataFrame.asCsvString();
        Assert.assertEquals("Column1\nValue1\n", csv);
    }

    @Test
    public void addColumnColumnNameTypeTest() {
        DataFrame dataFrame = new DataFrame("Test");
        DataFrame result = dataFrame.addColumn("Column1", Value.Type.STRING);
        Assert.assertEquals("Test", result.getName());
    }

    @Test
    public void evaluateExpressionTest() {
        Value expectedValue = Mockito.mock(Value.class);
        DataFrame mockDataFrame = Mockito.mock(DataFrame.class);
        Expression mockExpression = Mockito.mock(Expression.class);
        Mockito.when(mockDataFrame.evaluateExpression(mockExpression, 1)).thenReturn(expectedValue);
        Assert.assertEquals(expectedValue, mockDataFrame.evaluateExpression(mockExpression, 1));
    }

    @Test
    public void unsortTest() {
        String expectedName = "Test";
        DataFrame mockDataFrame = Mockito.mock(DataFrame.class);
        Mockito.doCallRealMethod().when(mockDataFrame).unsort();
        mockDataFrame.unsort();
        Mockito.verify(mockDataFrame).unsort();
    }

    @Test
    public void isEmptyTest() {
        DataFrame dataFrame = new DataFrame("Test");
        Assert.assertTrue(dataFrame.isEmpty());
    }

    @Test
    public void isNotEmptyTest() {
        DataFrame dataFrame = new DataFrame("Test");
        dataFrame.addStringColumn("Column1");
        Assert.assertFalse(dataFrame.isEmpty());
    }
}
