package leetcode.medium;

import org.junit.Test;

import static org.junit.Assert.assertEquals;

public class OnlineStockSpanTest {

    @Test
    public void nextSinglePriceTest() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        int result = onlineStockSpan.next(100);
        assertEquals(1, result);
    }

    @Test
    public void nextMultiplePricesTest() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        onlineStockSpan.next(100);
        int result = onlineStockSpan.next(80);
        assertEquals(1, result);
    }

    @Test
    public void nextMultiplePricesInIncreasingOrderTest() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        onlineStockSpan.next(100);
        int result = onlineStockSpan.next(120);
        assertEquals(2, result);
    }

    @Test(expected = NullPointerException.class)
    public void nextNullPriceTest() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        onlineStockSpan.next(null);
    }

    @Test
    public void calculateSpansSinglePriceTest() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        int[] result = onlineStockSpan.calculateSpans(new int[]{100});
        assertEquals(new int[]{1}, result);
    }

    @Test
    public void calculateSpansMultiplePricesTest() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        int[] result = onlineStockSpan.calculateSpans(new int[]{100, 80, 60, 70, 60, 75, 85});
        assertEquals(new int[]{1, 1, 1, 2, 1, 4, 6}, result);
    }

    @Test(expected = NullPointerException.class)
    public void calculateSpansNullPricesTest() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        onlineStockSpan.calculateSpans(null);
    }

    @Test(expected = IllegalArgumentException.class)
    public void calculateSpansEmptyPricesTest() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        onlineStockSpan.calculateSpans(new int[]{});
    }

}
