import java.security.SecureRandom;

import static org.junit.Assert.*;

public class KeyPairTest { 

    @Test
    public void canSignTest() {
        KeyPair keyPair = KeyPair.random();
        assertTrue(keyPair.canSign());
    }

    @Test
    public void fromSecretSeedCharArrayTest() {
        char[] seed = {'a', 'b', 'c', 'd', 'e', 'f'};
        KeyPair keyPair = KeyPair.fromSecretSeed(seed);
        assertNotNull(keyPair);
    }

    @Test
    public void fromSecretSeedStringTest() {
        String seed = "abcdef";
        KeyPair keyPair = KeyPair.fromSecretSeed(seed);
        assertNotNull(keyPair);
    }

    public void fromAccountIdTest() {
        KeyPair keyPair1 = KeyPair.random();
        KeyPair keyPair2 = KeyPair.fromAccountId(keyPair1.getAccountId());
        assertEquals(keyPair1, keyPair2);
    }

    @Test
    public void fromBip39SeedTest() {
        byte[] seed = new SecureRandom().generateSeed(20);
        KeyPair keyPair = KeyPair.fromBip39Seed(seed, 1);
        assertNotNull(keyPair);
    }

    @Test
    public void getAccountIdTest() {
        KeyPair keyPair = KeyPair.random();
        assertNotNull(keyPair.getAccountId());
    }

    }

        KeyPair keyPair2 = KeyPair.fromXdrPublicKey(publicKey);
        assertEquals(keyPair1, keyPair2);
    }

    public void fromXdrSignerKeyTest() {

    public void verifyTest() {
