import org.jsoup.nodes.Attributes;
import org.jsoup.nodes.DataNode;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.TextNode;
import org.jsoup.parser.Tag;
import org.jsoup.select.Elements;
import org.jsoup.select.Evaluator;
import org.junit.Test;

import static org.junit.Assert.*;

public class ElementTest {

    @Test
    public void tagNameNormalCaseTest() {
        Element element = new Element("div");
        assertEquals("div", element.tagName());
    }

    @Test
    public void hasAttributesWithoutAttributesTest() {
        Element element = new Element("div");
        assertFalse(element.hasAttributes());
    }

    @Test
    public void hasAttributesWithAttributesTest() {
        Attributes attributes = new Attributes();
        attributes.put("class", "test");
        Element element = new Element(Tag.valueOf("div"), "", attributes);
        assertTrue(element.hasAttributes());
    }

    @Test
    public void textInitialCaseTest() {
        Element element = new Element("div");
        assertEquals("", element.text());
    }

    @Test
    public void textNormalCaseTest() {
        Element element = new Element("div");
        element.appendText("Hello");
        assertEquals("Hello", element.text());
    }

    @Test
    public void childNodeSizeWithoutChildNodesTest() {
        Element element = new Element("div");
        assertEquals(0, element.childNodeSize());
    }

    @Test
    public void childNodeSizeWithChildNodesTest() {
        Element element = new Element("div");
        element.appendChild(new Element("p"));
        assertEquals(1, element.childNodeSize());
    }

    @Test
    public void appendChildNormalCaseTest() {
        Element element = new Element("div");
        Element child = new Element("p");
        element.appendChild(child);
        assertEquals(child, element.child(0));
    }

    @Test
    public void prependChildNormalCaseTest() {
        Element element = new Element("div");
        Element firstChild = new Element("p");
        Element secondChild = new Element("a");
        element.appendChild(firstChild);
        element.prependChild(secondChild);
        assertEquals(secondChild, element.child(0));
        assertEquals(firstChild, element.child(1));
    }

    @Test
    public void selectNormalCaseTest() {
        Element element = new Element("div");
        element.appendChild(new Element("p"));
        Elements elements = element.select("p");
        assertEquals(1, elements.size());
    }

    @Test
    public void selectNoMatchesTest() {
        Element element = new Element("div");
        Elements elements = element.select("p");
        assertTrue(elements.isEmpty());
    }
    
    @Test
    public void idNormalCaseTest() {
        Element element = new Element("div");
        element.attr("id", "test");
        assertEquals("test", element.id());
    }
    
    @Test
    public void dataNodesNormalCaseTest() {
        Element element = new Element("div");
        element.appendChild(new DataNode("data"));
        List<DataNode> dataNodes = element.dataNodes();
        assertEquals(1, dataNodes.size());
        assertEquals("data", dataNodes.get(0).getWholeData());
    }
    
    @Test
    public void textNodesNormalCaseTest() {
        Element element = new Element("div");
        element.appendChild(new TextNode("text"));
        List<TextNode> textNodes = element.textNodes();
        assertEquals(1, textNodes.size());
        assertEquals("text", textNodes.get(0).text());
    }
}
