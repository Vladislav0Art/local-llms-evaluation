import io.netty.channel.Channel;
import org.junit.Test;
import org.junit.Assert;

import java.net.InetSocketAddress;
import java.net.SocketAddress;

import static org.mockito.Mockito.*;

public class GalileoProtocolDecoderTest {

    private GalileoProtocolDecoder decoder = new GalileoProtocolDecoder(new Protocol("gp"));

    @Test
    public void decodeNullMessageTest() throws Exception {
        String msg = null;
        Channel channel = mock(Channel.class);
        SocketAddress remoteAddress = new InetSocketAddress("localhost", 8080);
        Assert.assertNull(decoder.decode(channel, remoteAddress, msg));
    }

    @Test
    public void decodeEmptyMessageTest() throws Exception {
        String msg = "";
        Channel channel = mock(Channel.class);
        SocketAddress remoteAddress = new InetSocketAddress("localhost", 8080);
        Assert.assertNull(decoder.decode(channel, remoteAddress, msg));
    }

    @Test
    public void decodeInvalidFormatMessageTest() throws Exception {
        String msg = "invalid message format";
        Channel channel = mock(Channel.class);
        SocketAddress remoteAddress = new InetSocketAddress("localhost", 8080);
        Assert.assertNull(decoder.decode(channel, remoteAddress, msg));
    }

    @Test
    public void decodeValidMessageTest() throws Exception {
        String msg = "355227043578424,C,150721060058,,,,,000.000,000,0000,0,0000000000,00000000,313000*05:00,11,100";
        Channel channel = mock(Channel.class);
        SocketAddress remoteAddress = new InetSocketAddress("localhost", 8080);
        Assert.assertNotNull(decoder.decode(channel, remoteAddress, msg));
    }

    @Test
    public void decodeWithNullChannelTest() throws Exception {
        String msg = "355227043578424,C,150721060058,,,,,000.000,000,0000,0,0000000000,00000000,313000*05:00,11,100";
        Channel channel = null;
        SocketAddress remoteAddress = new InetSocketAddress("localhost", 8080);
        Assert.assertNotNull(decoder.decode(channel, remoteAddress, msg));
    }

    @Test
    public void decodeWithNullRemoteAddressTest() throws Exception {
        String msg = "355227043578424,C,150721060058,,,,,000.000,000,0000,0,0000000000,00000000,313000*05:00,11,100";
        Channel channel = mock(Channel.class);
        SocketAddress remoteAddress = null;
        Assert.assertNotNull(decoder.decode(channel, remoteAddress, msg));
    }

}
