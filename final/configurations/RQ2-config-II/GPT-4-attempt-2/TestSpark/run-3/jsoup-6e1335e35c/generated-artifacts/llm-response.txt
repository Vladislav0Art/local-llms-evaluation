package org.jsoup.nodes;

import org.jsoup.parser.Tag;
import org.jsoup.select.Elements;
import org.jsoup.select.Evaluator;
import org.jsoup.select.NodeTraversor;
import org.jsoup.select.NodeVisitor;
import org.junit.Assert;
import org.junit.Test;

import java.util.ArrayList;
import java.util.List;

public class ElementTest {

    @Test
    public void tagNameTest() {
        Element element = new Element("tag");
        Assert.assertEquals(element.tagName(), "tag");
    }

    @Test
    public void hasChildNodesTest() {
        Element element = new Element("tag");
        Assert.assertFalse(element.hasChildNodes());
    }

    @Test
    public void attrTest() {
        Element element = new Element("tag");
        element.attr("testAttr", "true");
        Assert.assertEquals(element.attr("testAttr"), "true");
    }

    @Test
    public void idTest() {
        Element element = new Element("tag");
        element.id("testId");
        Assert.assertEquals(element.id(), "testId");
    }

    @Test
    public void appendChildTest() {
        Element element = new Element("tag");
        element.appendChild(new Element("childTag"));
        Assert.assertEquals(element.child(0).nodeName(), "childTag");
    }

    @Test
    public void childNodeSizeTest() {
        Element element = new Element("tag");
        element.appendChild(new Element("childTag1"));
        element.appendChild(new Element("childTag2"));
        Assert.assertEquals(element.childNodeSize(), 2);
    }

    @Test
    public void cloneTest(){
        Element element = new Element("tag");
        element.id("testId");
        Element cloneElement = element.clone();
        Assert.assertEquals(cloneElement.toString(), element.toString());
    }

    @Test
    public void datasetTest() {
        Element element = new Element("tag");
        element.attr("data-test", "value");
        Assert.assertEquals(element.dataset().get("test"), "value");
    }

    @Test
    public void classNamesTest() {
        Element element = new Element("tag");
        element.attr("class", "class1 class2");
        Assert.assertEquals(element.classNames().size(), 2);
        Assert.assertTrue(element.classNames().contains("class1"));
        Assert.assertTrue(element.classNames().contains("class2"));
    }

    @Test
    public void selectFirstTest() {
        Element parent = new Element("parent");
        Element child = new Element("child");
        child.id("testId");
        parent.appendChild(child);
        Assert.assertEquals(parent.selectFirst("#testId"), child);
    }

    @Test
    public void cssSelectorTest() {
        Element parent = new Element("parent");
        Element child = new Element("child");
        child.id("testId");
        parent.appendChild(child);
        Assert.assertEquals(child.cssSelector(), "#testId");
    }

    @Test
    public void appendTest() {
        Element parent = new Element("parent");
        parent.append("<child id=\"testId\"></child>");
        Assert.assertEquals(parent.child(0).id(), "testId");
    }

    @Test
    public void childrenTest() {
        Element parent = new Element("parent");
        Element child1 = new Element("child");
        Element child2 = new Element("child");
        parent.appendChild(child1);
        parent.appendChild(child2);

        Elements children = parent.children();
        Assert.assertEquals(children.size(), 2);
        Assert.assertEquals(children.get(0), child1);
        Assert.assertEquals(children.get(1), child2);
    }

    @Test
    public void clearAttributesTest() {
        Element element = new Element("tag");
        element.attr("testAttr", "true");
        element.clearAttributes();
        Assert.assertTrue(element.attributes().size() == 0);
    }

}
