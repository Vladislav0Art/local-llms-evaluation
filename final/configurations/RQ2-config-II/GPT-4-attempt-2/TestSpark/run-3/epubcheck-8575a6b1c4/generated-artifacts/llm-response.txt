package com.adobe.epubcheck.tool;

import com.adobe.epubcheck.api.EPUBProfile;
import com.adobe.epubcheck.util.InvalidVersionException;
import com.adobe.epubcheck.util.Messages;
import com.adobe.epubcheck.api.LocalizableReport;
import com.adobe.epubcheck.api.Report;
import com.adobe.epubcheck.util.DefaultReportImpl;
import org.junit.Test;
import org.junit.Assert;
import org.junit.Rule;
import org.junit.rules.ExpectedException;
import org.junit.runner.RunWith;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

@RunWith(MockitoJUnitRunner.class)
public class EpubCheckerTest {

    private EpubChecker epubChecker = new EpubChecker();

    @Rule
    public ExpectedException thrown = ExpectedException.none();

    @Test
    public void getLocaleDefaultTest() {
        Locale defaultLocale = Locale.getDefault();
        Assert.assertEquals(defaultLocale, epubChecker.getLocale());
    }

    @Test
    public void runInvalidArgsTest() {
        thrown.expect(RuntimeException.class);
        thrown.expectMessage(Messages.get("missing_input_filename"));
        epubChecker.run(new String[0]);
    }

    @Test
    public void processEpubFileTest() {
        int exitCode = epubChecker.processEpubFile(new String[]{"somePath"});
        Assert.assertEquals(0, exitCode);
    }

    @Test
    public void validateFileTest() {
        String path = "SomePath";
        EPUBVersion version = EPUBVersion.VERSION_3;
        Report report = new DefaultReportImpl(path);
        EPUBProfile profile = EPUBProfile.DEFAULT;
        
        int result = epubChecker.validateFile(path, version, report, profile);

        Assert.assertEquals(0, result);
    }

    @Test
    public void validateFileInvalidVersionTest() {
        String path = "SomePath";
        EPUBVersion version = null;
        Report report = new DefaultReportImpl(path);
        EPUBProfile profile = EPUBProfile.DEFAULT;

        thrown.expect(InvalidVersionException.class);
        thrown.expectMessage("Invalid EPUB version");

        epubChecker.validateFile(path, version, report, profile);
    }

    @Test
    public void validateFileInvalidPathTest() {
        String path = null;
        EPUBVersion version = EPUBVersion.VERSION_3;
        Report report = new DefaultReportImpl(path);
        EPUBProfile profile = EPUBProfile.DEFAULT;

        thrown.expect(RuntimeException.class);
        thrown.expectMessage("Invalid file path");

        epubChecker.validateFile(path, version, report, profile);
    }

    @Test
    public void runMockTest() {
        EpubChecker epubCheckerMock = Mockito.mock(EpubChecker.class);
        Mockito.when(epubCheckerMock.run(Mockito.any(String[].class))).thenReturn(0);
        int exitCode = epubCheckerMock.run(new String[]{"somePath"});
        Mockito.verify(epubCheckerMock).run(new String[]{"somePath"});
        Assert.assertEquals(exitCode, 0);
    }

}
