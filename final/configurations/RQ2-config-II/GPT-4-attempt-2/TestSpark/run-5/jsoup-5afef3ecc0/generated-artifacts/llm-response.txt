import org.jsoup.Connection;
import org.junit.Test;
import java.io.UnsupportedEncodingException;
import java.net.MalformedURLException;
import java.net.URL;
import static org.junit.Assert.*;

public class UrlBuilderTest {
    @Test
    public void UrlBuilderValidUrlTest() throws MalformedURLException {
        URL inputUrl = new URL("http://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        assertNotNull(urlBuilder);
    }

    @Test(expected = MalformedURLException.class)
    public void UrlBuilderInvalidUrlTest() throws MalformedURLException {
        URL inputUrl = new URL("invalid url");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
    }

    @Test
    public void buildValidUrlTest() throws MalformedURLException {
        URL inputUrl = new URL("http://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        URL outputUrl = urlBuilder.build();
        assertEquals(inputUrl, outputUrl);
    }

    @Test
    public void appendKeyValValidTest() throws UnsupportedEncodingException, MalformedURLException {
        Connection.KeyVal keyVal = Connection.KeyVal.create("name", "value");
        URL inputUrl = new URL("http://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        urlBuilder.appendKeyVal(keyVal);
        URL outputUrl = urlBuilder.build();
        assertEquals("http://example.com?name=value", outputUrl.toString());
    }

    @Test(expected = UnsupportedEncodingException.class)
    public void appendKeyValInvalidEncodingTest() throws UnsupportedEncodingException {
        Connection.KeyVal keyVal = Connection.KeyVal.create("name", "välue");
        UrlBuilder urlBuilder = new UrlBuilder(new URL("http://example.com"));
        urlBuilder.appendKeyVal(keyVal);
    }

    @Test(expected = NullPointerException.class)
    public void appendKeyValNullTest() throws UnsupportedEncodingException {
        Connection.KeyVal keyVal = null;
        UrlBuilder urlBuilder = new UrlBuilder(new URL("http://example.com"));
        urlBuilder.appendKeyVal(keyVal);
    }
}
