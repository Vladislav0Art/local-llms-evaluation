import org.junit.Test;
import static org.junit.Assert.assertEquals;

public class OnlineStockSpanTest {

    public void nextThreeDifferentPricesTest() {

        assertEquals(1, onlineStockSpan.next(120));
    }

    public void nextThreeSamePricesTest() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        
        assertEquals(1, onlineStockSpan.next(100));
        assertEquals(2, onlineStockSpan.next(100));
        assertEquals(3, onlineStockSpan.next(100));
    }

    @Test
    public void nextThreePricesOrderedDescendingTest() {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        
        assertEquals(1, onlineStockSpan.next(120));
        assertEquals(2, onlineStockSpan.next(100));
        assertEquals(3, onlineStockSpan.next(80));
    }

    public void calculateSpansEmptyArrayTest() {

    }

    public void calculateSpansThreeDifferentPricesTest() {

        assertEquals(1, spans[2]);

        assertEquals(2, spans[1]);
        assertEquals(3, spans[2]);
    }
