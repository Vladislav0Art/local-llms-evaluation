import io.github.vmzakharov.ecdataframe.dsl.*;
import io.github.vmzakharov.ecdataframe.dsl.value.DecimalValue;
import io.github.vmzakharov.ecdataframe.dsl.value.StringValue;
import io.github.vmzakharov.ecdataframe.dsl.visitor.PrettyPrintVisitor;
import io.github.vmzakharov.ecdataframe.util.CollectingPrinter;
import org.junit.Test;

import static org.junit.Assert.assertEquals;

public class PrettyPrintVisitorTest {

    @Test
    public void visitAssignExprTest() {
        AssingExpr expr = new AssingExpr(new VarExpr(new StringValue("Assignment")), new DecimalExpr(new DecimalValue(15)));
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(new CollectingPrinter());
        visitor.visitAssignExpr(expr);
        assertEquals("Assignment = 15", visitor.toString());
    }

    @Test
    public void visitBinaryExprTest() {
        BinaryExpr expr = new BinaryExpr(new DecimalExpr(new DecimalValue(5)), Operator.ADD, new DecimalExpr(new DecimalValue(10)));
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(new CollectingPrinter());
        visitor.visitBinaryExpr(expr);
        assertEquals("(5 + 10)", visitor.toString());
    }

    @Test
    public void visitUnaryExprTest() {
        UnaryExpr expr = new UnaryExpr(new DecimalExpr(new DecimalValue(5)), UnaryOp.MINUS);
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(new CollectingPrinter());
        visitor.visitUnaryExpr(expr);
        assertEquals("-5", visitor.toString());
    }

    @Test
    public void visitConstExprTest() {
        StringValue expr = new StringValue("John Snow");
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(new CollectingPrinter());
        visitor.visitConstExpr(expr);
        assertEquals("'John Snow'", visitor.toString());
    }

}
