```java
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.runners.MockitoJUnitRunner;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.Node;

import static org.junit.Assert.*;

@RunWith(MockitoJUnitRunner.class)
public class ElementTest {
    
    @Test
    public void ElementTagBaseUriAttributesTest() {
        Element testElement = new Element("test", "https://test.com", new Attributes());
        assertEquals("test", testElement.TagName());
        assertEquals("https://test.com", testElement.baseUri());
    }
    
    @Test
    public void hasChildNodesTest() {
        Element testElement = new Element("test", "https://test.com", new Attributes());
        assertFalse(testElement.hasChildNodes());
    }

    @Test
    public void tagNameTest() {
        Element testElement = new Element("test");
        assertEquals("test", testElement.TagName());
        testElement.tagName("testChange");
        assertEquals("testChange", testElement.TagName());

    }

    @Test
    public void idTest() {
        Element testElement = new Element("test");
        testElement.id("test");
        assertEquals("test", testElement.id());
    }

    @Test
    public void childNodeSizeTest() {
        Element testElement =  new Element("testChild"),
                parentElement = new Element("test");

        parentElement.appendChild(testElement);

        assertEquals(1, parentElement.childNodeSize());
    }

    @Test
    public void selectTest() {
        Element selectElement = new Element("select");
        Elements resultElements = selectElement.select("div");
        assertNotNull(resultElements);
    }
      
    @Test
    public void appendChildTest() {
        Element testElement = new Element("div");
        Element childElement = new Element("p");
        testElement.appendChild(childElement);
        assertEquals(1, testElement.childrenSize());
    }

    @Test
    public void removeAttrTest() {
        Element testElement = new Element("test");
        testElement.attr("newAttr", "value");
        testElement.removeAttr("newAttr");
        assertFalse(testElement.hasAttributes());
    }
      
    @Test
    public void cloneTest() {
        Element testElement = new Element("test");
        Element cloneElement = testElement.clone();
        assertEquals(cloneElement.tagName(), testElement.tagName());
    }
}

```
The above unit tests cover the instantiation of the `Element` class with different constructors, attribute manipulation (adding and removing), modifying child nodes, and cloning an element. All these tests reinforce the expected behavior of the methods in different scenarios.
