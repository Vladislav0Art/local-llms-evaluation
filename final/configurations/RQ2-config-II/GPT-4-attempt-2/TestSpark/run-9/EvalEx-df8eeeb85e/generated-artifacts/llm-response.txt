package com.ezylang.evalex.parser;

import com.ezylang.evalex.config.ExpressionConfiguration;
import com.ezylang.evalex.config.FunctionDictionaryIfc;
import com.ezylang.evalex.config.OperatorDictionaryIfc;
import com.ezylang.evalex.functions.FunctionIfc;
import org.junit.Test;

import static org.junit.Assert.*;
import static org.mockito.Mockito.*;

import java.util.List;

public class TokenizerTest {

    @Test
    public void tokenizerValidExpressionConfigurationTest() {
        ExpressionConfiguration configuration = mock(ExpressionConfiguration.class);
        new Tokenizer("3+5", configuration);
        verify(configuration, times(1)).toString();
    }

    @Test(expected = ParseException.class)
    public void parseInvalidExpressionTest() throws ParseException {
        ExpressionConfiguration configuration = mock(ExpressionConfiguration.class);
        Tokenizer tokenizer = new Tokenizer("3++5", configuration);
        tokenizer.parse();
    }

    @Test
    public void parseValidExpressionTest() throws ParseException {
        ExpressionConfiguration configuration = mock(ExpressionConfiguration.class);
        when(configuration.getOperatorDictionary()).thenReturn(mock(OperatorDictionaryIfc.class));
        when(configuration.getFunctionDictionary()).thenReturn(mock(FunctionDictionaryIfc.class));
        new Tokenizer("3+5", configuration).parse();
        verify(configuration, times(1)).getOperatorDictionary();
        verify(configuration, times(1)).getFunctionDictionary();
    }

    @Test(expected = ParseException.class)
    public void parseStringLiteralInvalidTest() throws ParseException {
        ExpressionConfiguration configuration = mock(ExpressionConfiguration.class);
        Tokenizer tokenizer = new Tokenizer("\"Invalid", configuration);
        tokenizer.parseStringLiteral();
    }

    @Test
    public void parseStringLiteralValidTest() throws ParseException {
        ExpressionConfiguration configuration = mock(ExpressionConfiguration.class);
        Tokenizer tokenizer = new Tokenizer("\"Valid\"", configuration);
        tokenizer.parseStringLiteral();
        verify(configuration, times(0)).getOperatorDictionary();
    }
}
