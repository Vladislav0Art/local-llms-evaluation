import static org.junit.Assert.*;

public class DocumentTest {

    }

    @Test
    public void location_ReturnsCorrectLocation() {

        assertNotNull(document.connection());
    }

    public void head_ReturnsCorrectHeadElement() {
        String baseUri = "https://example.com";

        Element head = document.head();

    }

    public void body_ReturnsCorrectBodyElement() {

        Document document = new Document(baseUri);
        Element body = document.body();

    }

    public void forms_ReturnsCorrectFormsElements() {
