package org.jsoup.nodes;

import org.jsoup.helper.Validate;
import org.jsoup.internal.StringUtil;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.junit.runners.JUnit4;
import java.io.ByteArrayOutputStream;
import java.io.IOException;
import static org.junit.Assert.*;
import java.util.List;

@RunWith(JUnit4.class)
public class TextNodeTest {

    @Test
    public void nodeName_EmptyTextReturnsTextNode() {
        TextNode node = new TextNode("");
        assertEquals("text-node", node.nodeName());
    }

    @Test
    public void nodeName_TextReturnsTextNode() {
        TextNode node = new TextNode("test");
        assertEquals("text-node", node.nodeName());
    }

    @Test
    public void text_EmptyTextReturnsEmptyString() {
        TextNode node = new TextNode("");
        assertEquals("", node.text());
    }

    @Test
    public void text_TextReturnsTextValue() {
        TextNode node = new TextNode("test");
        assertEquals("test", node.text());
    }

    @Test
    public void text_MultipleSpacesReturnsMultipleSpaces() {
        TextNode node = new TextNode("   ");
        assertEquals("   ", node.text());
    }

    @Test
    public void text_NewlinesReturnNewlines() {
        TextNode node = new TextNode("\n");
        assertEquals("\n", node.text());
    }

    @Test
    public void text_SingleCharacterReturnsSingleCharacter() {
        TextNode node = new TextNode("a");
        assertEquals("a", node.text());
    }

    @Test
    public void text_MultiLineTextReturnsMultiLineText() {
        TextNode node = new TextNode("line1\nline2");
        assertEquals("line1\nline2", node.text());
    }

    @Test
    public void text_SplittingWhitespaceReturnsSplitString() {
        TextNode node = new TextNode("   ");
        assertEquals("   ", node.text());
    }

    @Test
    public void isBlank_ReturnsTrueForEmptyText() {
        TextNode node = new TextNode("");
        assertTrue(node.isBlank());
    }

    @Test
    public void isBlank_ReturnsFalseForNonEmptyText() {
        TextNode node = new TextNode("test");
        assertFalse(node.isBlank());
    }

    @Test
    public void splitText_SplittingOffsetIsZeroReturnsOriginalString() {
        TextNode node = new TextNode("test");
        TextNode splitResult = node.splitText(0);
        assertEquals(node, splitResult);
    }

    @Test
    public void splitText_SplittingOffsetIsNonZeroReturnsSplittedString() {
        TextNode node = new TextNode("test");
        TextNode splitResult = node.splitText(5);
        assertEquals("te", splitResult.text());
    }

    @Test
    public void outerHtmlHead_AppendsWholeTextToAccum() throws IOException {
        ByteArrayOutputStream accum = new ByteArrayOutputStream();
        Document.OutputSettings out = Document.OutputSettings.defaultAttribute();
        TextNode node = new TextNode("<test>");
        node.outerHtmlHead(accum, 0, out);
        assertTrue(accum.toString().contains("<test>"));
    }

    @Test
    public void outerHtmlTail_AppendsWholeTextToAccum() throws IOException {
        ByteArrayOutputStream accum = new ByteArrayOutputStream();
        Document.OutputSettings out = Document.OutputSettings.defaultAttribute();
        TextNode node = new TextNode("<test>");
        node.outerHtmlTail(accum, 0, out);
        assertTrue(accum.toString().contains("</test>"));
    }

    @Test
    public void toString_ReturnsWholeText() {
        TextNode node = new TextNode("test");
        assertEquals("test", node.toString());
    }

    @Test
    public void clone_ReturnsNewTextNode() {
        TextNode node = new TextNode("test");
        TextNode cloned = node.clone();
        assertNotNull(cloned);
        assertEquals(node, cloned);
    }

    @Test
    public void createFromEncoded_RetrievesCorrectString() {
        String encodedText = "test";
        TextNode created = TextNode.createFromEncoded(encodedText);
        assertEquals("test", created.text());
    }

    @Test
    public void normaliseWhitespace_ReturnsNormalisedString() {
        String text = "\n   ";
        String result = StringUtil.normaliseWhitespace(text);
        assertTrue(result.contains("\n"));
        assertTrue(result.contains(" "));
    }

    @Test
    public void stripLeadingWhitespace_ReturnsTrimmedString() {
        String text = "   test";
        String result = StringUtil.stripLeadingWhitespace(text);
        assertEquals("test", result);
    }

    @Test
    public void lastCharIsWhitespace_ReturnsFalseForNonSpaceCharacter() {
        StringBuilder sb = new StringBuilder();
        assertFalse(lastCharIsWhitespace(sb));
    }
}
