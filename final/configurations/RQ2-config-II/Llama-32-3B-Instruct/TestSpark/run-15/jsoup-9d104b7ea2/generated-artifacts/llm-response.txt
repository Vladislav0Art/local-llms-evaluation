public class TextNodeTest {

    public void nodeName_NoTextReturnsNull() {
        TextNode node = new TextNode("");
        assert null == node.nodeName();
    }

    public void nodeName_ReturnsNodeName() {
        TextNode node = new TextNode("test");
        assert "text".equals(node.nodeName());
    }

    @Test
    public void text_NoTextReturnsEmptyString() {
        TextNode node = new TextNode("");
        assert "".equals(node.text());
    }

    @Test
    public void text_ReturnsGivenText() {
        String text = "test";
        TextNode node = new TextNode(text);
        assert text.equals(node.text());
    }

    @Test
    public void text_MultiLineTextReturnsWholeText() {
        String text = "test\nanother test";
        TextNode node = new TextNode(text);
        assert text.equals(node.text());
    }

    public void text_SplittingCharacterChangesContent() {
        String text = "abc\ndef";
        TextNode node = new TextNode(text);
        assert "def".equals(node.text());
    }

    public void splitText_NoOffsetReturnsEmptyString() {
        TextNode node = new TextNode("abc");
        assert "".equals(node.splitText(0).text());
    }

        String text = "abcdef";
