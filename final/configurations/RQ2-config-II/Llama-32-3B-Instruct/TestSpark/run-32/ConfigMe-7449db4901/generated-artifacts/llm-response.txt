import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import java.util.Arrays;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

import static org.hamcrest.MatcherAssert.assertThat;
import static org.hamcrest.Matchers.is;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class CommentsConfigurationTest {

    @Mock
    private Map<String, List<String>> commentsMapMock;

    public CommentsConfiguration getCommentsConfiguration() {
        return new CommentsConfiguration(commentsMapMock);
    }

    @Test
    public void shouldCreateComentsConfigurationWithEmptyCommentLines() {
        String path = "/path/to/resource";
        when(commentsMapMock.get(path)).thenReturn(Arrays.asList(""));
        CommentsConfiguration commentsConfig = getCommentsConfiguration();
        Map<String, List<String>> allComments = commentsConfig.getAllComments();
        assertThat(allComments.size(), is(1));
    }

    @Test
    public void shouldCreateComentsConfigurationWithMultipleCommentLines() {
        String path = "/path/to/resource";
        when(commentsMapMock.get(path)).thenReturn(Arrays.asList("line1", "line2"));
        CommentsConfiguration commentsConfig = getCommentsConfiguration();
        Map<String, List<String>> allComments = commentsConfig.getAllComments();
        assertThat(allComments.size(), is(1));
    }

    @Test
    public void shouldCreateComentsConfigurationWithMultiplePaths() {
        String path1 = "/path/to/resource1";
        String path2 = "/path/to/resource2";
        when(commentsMapMock.get(path1)).thenReturn(Arrays.asList("line11", "line12"));
        when(commentsMapMock.get(path2)).thenReturn(Arrays.asList("line21", "line22"));
        CommentsConfiguration commentsConfig = getCommentsConfiguration();
        Map<String, List<String>> allComments = commentsConfig.getAllComments();
        assertThat(allComments.size(), is(2));
    }

    @Test
    public void shouldSetCommentForPath() {
        String path = "/path/to/resource";
        when(commentsMapMock.get(path)).thenReturn(new ArrayList<>());
        CommentsConfiguration commentsConfig = new CommentsConfiguration(commentsMapMock);
        commentsConfig.setComment(path, "line1", "line2");
        Map<String, List<String>> allComments = commentsConfig.getAllComments();
        assertThat(allComments.size(), is(1));
    }

    @Test
    public void shouldOverwriteExistingCommentForPath() {
        String path = "/path/to/resource";
        when(commentsMapMock.get(path)).thenReturn(Arrays.asList("line11", "line12"));
        CommentsConfiguration commentsConfig = new CommentsConfiguration(commentsMapMock);
        commentsConfig.setComment(path, "newLine1");
        Map<String, List<String>> allComments = commentsConfig.getAllComments();
        assertThat(allComments.size(), is(2));
    }

    @Test
    public void shouldNotAddEmptyNewLine() {
        String path = "/path/to/resource";
        when(commentsMapMock.get(path)).thenReturn(Arrays.asList());
        CommentsConfiguration commentsConfig = new CommentsConfiguration(commentsMapMock);
        commentsConfig.setComment(path, "");
        Map<String, List<String>> allComments = commentsConfig.getAllComments();
        assertThat(allComments.size(), is(1));
    }

    @Test
    public void shouldThrowExceptionWhenSettingCommentLinesWithoutPath() {
        String[] commentLines = {"line1", "line2"};
        try {
            CommentsConfiguration commentsConfig = new CommentsConfiguration(commentsMapMock);
            commentsConfig.setComment(commentLines);
            assert false;
        } catch (NullPointerException e) {
            // expected
        }
    }

    @Test
    public void shouldReturnUnmodifiableViewForAllComments() {
        String path = "/path/to/resource";
        when(commentsMapMock.get(path)).thenReturn(Arrays.asList("line1", "line2"));
        CommentsConfiguration commentsConfig = new CommentsConfiguration(commentsMapMock);
        Map<String, List<String>> allComments = commentsConfig.getAllComments();
        assert allComments instanceof UnmodifiableView;
    }
}
