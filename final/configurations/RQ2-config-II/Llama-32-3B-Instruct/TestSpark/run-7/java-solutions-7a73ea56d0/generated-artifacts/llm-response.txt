package leetcode.medium;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import java.util.Arrays;

@RunWith(MockitoJUnitRunner.class)
public class OnlineStockSpanTest {

    @InjectMocks
    private OnlineStockSpan stockSpan;

    @Mock
    private Stack<Integer> stack;

    @Test
    public void calculateSpans_EmptyArray_ReturnsEmptyArray() {
        int[] result = stockSpan.calculateSpans(new int[0]);
        assert result == new int[0];
    }

    @Test
    public void calculateSpans_SingleElementArray_ReturnsSingleElementArray() {
        int[] prices = {1};
        int[] result = stockSpan.calculateSpans(prices);
        assert Arrays.equals(result, new int[]{1});
    }

    @Test
    public void calculateSpans_TwoElementArray_ReturnsTwoElementArray() {
        int[] prices = {1, 2};
        int[] result = stockSpan.calculateSpans(prices);
        assert Arrays.equals(result, new int[]{2, 1});
    }

    @Test
    public void next_EmptyStack_Returns0() {
        Mockito.when(stack.isEmpty()).thenReturn(true);
        int result = stockSpan.next(5);
        assert result == 0;
    }

    @Test
    public void next_SingleElementStack_ReturnsPricePlusOne() {
        stack.push(1);
        Mockito.when(stack.isEmpty()).thenReturn(false);
        int price = 5;
        int result = stockSpan.next(price);
        assert result == price + 1;
    }

    @Test
    public void next_MultipleElementsStack_ReturnsPricePlusOneWhenCurrentIsSmallest() {
        stack.push(2);
        Mockito.when(stack.isEmpty()).thenReturn(false);
        int price = 5;
        int current = stack.pop();
        stockSpan.next(current);
        stack.push(price);
        Mockito.when(stack.isEmpty()).thenReturn(false);
        int result = stockSpan.next(price);
        assert result == price + 1;
    }

    @Test
    public void next_MultipleElementsStack_ReturnsPriceMinusOldWhenCurrentIsLarger() {
        stack.push(2);
        Mockito.when(stack.isEmpty()).thenReturn(false);
        int price = 5;
        int current = stack.pop();
        stockSpan.next(current);
        stack.push(price);
        Mockito.when(stack.isEmpty()).thenReturn(true);
        int result = stockSpan.next(price);
        assert result == price - current;
    }

    @Test
    public void calculateSpans_SingleElementArray_ReturnsSingleElementArray() {
        int[] prices = {1};
        int[] result = stockSpan.calculateSpans(prices);
        assert Arrays.equals(result, new int[]{2});
    }

    @Test
    public void calculateSpans_TwoElementArray_ReturnsTwoElementArray() {
        int[] prices = {1, 2};
        int[] result = stockSpan.calculateSpans(prices);
        assert Arrays.equals(result, new int[]{3, 1});
    }
}
