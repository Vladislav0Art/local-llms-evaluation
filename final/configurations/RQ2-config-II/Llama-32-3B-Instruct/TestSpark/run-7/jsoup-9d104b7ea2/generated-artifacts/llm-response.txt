package org.jsoup.nodes;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import static org.hamcrest.MatcherAssert.assertThat;
import static org.hamcrest.Matchers.is;
import static org.hamcrest.Matchers.nullValue;
import static org.junit.Assert.assertTrue;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class TextNodeTest {

    @Mock
    private String text;

    @Mock
    private StringBuilder sb;

    public void setText(String text) {
        this.text = text;
    }

    public void set Sb(StringBuilder sb) {
        this.sb = sb;
    }

    @Test
    public void testConstructor() {
        TextNode node = new TextNode(text);
        assertThat(node.text(), is(text));
    }

    @Test
    public void testGetWholeText() {
        TextNode node = new TextNode("Hello World");
        assertThat(node.getWholeText(), is("Hello World"));
    }

    @Test
    public void testIsBlank() {
        assertThat(TextNode.blank, is(true));
    }

    @Test
    public void testSplitText() {
        String text = "Hello World";
        when(text.substring(0, 5)).thenReturn("Hello ");
        TextNode node = new TextNode(text);
        TextNode expected = new TextNode(node.text().substring(0, 6));
        assertTrue(node.splitText(5).equals(expected));
    }

    @Test
    public void testOuterHtmlHead() {
        Appendable accum = mock(Appendable.class);
        Document.OutputSettings out = mock(Document.OutputSettings.class);
        when(out.getOutputMode()).thenReturn(Document.OutputMode.XHTML);
        TextNode node = new TextNode("Hello World");
        TextNode result = new TextNode("<p>Hello World</p>");
        assertEquals(result, node.outerHtmlHead(accum, 0, out));
    }

    @Test
    public void testOuterHtmlTail() {
        Appendable accum = mock(Appendable.class);
        Document.OutputSettings out = mock(Document.OutputSettings.class);
        when(out.getOutputMode()).thenReturn(Document.OutputMode.XHTML);
        TextNode node = new TextNode("Hello World");
        String result = " </p>";
        assertEquals(result, node.outerHtmlTail(accum, 0, out));
    }

    @Test
    public void testClone() {
        TextNode node = new TextNode("Hello World");
        TextNode expected = new TextNode(node.text());
        assertEquals(expected, node.clone());
    }

    @Test
    public void testFromString() {
        String text = "Hello World";
        TextNode node = TextNode.createFromEncoded(text);
        assertThat(node.text(), is(text));
    }
}
