public class KeyPairTest {

    @Test
    public void newKeyPair_fromSecretSeed(char[] seed) {
        KeyPair keyPair = KeyPair.fromSecretSeed(seed);
        assertNotNull(keyPair);
    }

    @Test
    public void newKeyPair_fromSecretSeed_string_seed() {
        char[] seed = "seed".toCharArray();
        KeyPair keyPair = KeyPair.fromSecretSeed(seed);
        assertNotNull(keyPair);
    }

    @Test
    public void newKeyPair_fromXdrPublicKey_PublicKey() {
        PublicKey publicKey = new PublicKey("publicKey");
        KeyPair keyPair = KeyPair.fromXdrPublicKey(publicKey);
        assertNotNull(keyPair);
    }

    @Test
    public void canSign_returnsFalse_whenNoPublicKey() {
        KeyPair keyPair = null;
        assertFalse(keyPair.canSign());
    }

    @Test
    public void canSign_returnsTrue_whenPublicKeypresent() {
        PublicKey publicKey = new PublicKey("publicKey");
        KeyPair keyPair = KeyPair.fromXdrPublicKey(publicKey);
        assertTrue(keyPair.canSign());
    }

    @Test
    public void sign_decodesCorrectly_fromBip39Seed_andSignatureHint() throws GeneralSecurityException, IOException {
        byte[] bip39Seed = {1, 2, 3};
        int accountNumber = 4;
        KeyPair keyPair = KeyPair.fromBip39Seed(bip39Seed, accountNumber);
        ByteArrayOutputStream bos = new ByteArrayOutputStream();
        XdrDataOutputStream xdrDataOutputStream = new XdrDataOutputStream(bos);
        byte[] data = "data".getBytes();
        DecoratedSignature signature = keyPair.sign(data);
        assertFalse(xdrDataOutputStream.write(signature.getEncoded()));
    }

    @Test
    public void sign_returnsSignatureForGivenPayload() throws GeneralSecurityException, IOException {
        KeyPair keyPair = KeyPair.random();
        ByteArrayOutputStream bos = new ByteArrayOutputStream();
        XdrDataOutputStream xdrDataOutputStream = new XdrDataOutputStream(bos);
        byte[] payload = "payload".getBytes();
        DecoratedSignature signature = keyPair.sign(payload);
        assertFalse(xdrDataOutputStream.write(signature.getEncoded()));
    }

    @Test
    public void verify_returnsTrue_whenPayloadAndSignatureAreValid() throws GeneralSecurityException, IOException {
        KeyPair keyPair = KeyPair.random();
        ByteArrayOutputStream bos = new ByteArrayOutputStream();
        XdrDataOutputStream xdrDataOutputStream = new XdrDataOutputStream(bos);
        byte[] payload = "payload".getBytes();
        DecoratedSignature signature = keyPair.sign(payload);
        assertFalse(xdrDataOutputStream.write(signature.getEncoded()));
        assertTrue(keyPair.verify(payload, signature));
    }

    @Test
    public void hashCode_returnsCorrectValue() {
        KeyPair keyPair = new KeyPair("publicKey");
        int expectedHashcode = 123;
        assertEquals(expectedHashcode, keyPair.hashCode());
    }
}
