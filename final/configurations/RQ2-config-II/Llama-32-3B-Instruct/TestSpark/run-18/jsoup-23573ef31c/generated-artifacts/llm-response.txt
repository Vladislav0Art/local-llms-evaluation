import org.junit.Test;
import org.junit.runner.RunWith;
import org.junit.runners.JUnit4;
import org.mockito.Mock;
import org.jsoup.Connection;
import org.jsoup.Document;
import org.jsoup.Element;
import org.jsoup.Selector;

import static org.junit.Assert.assertEquals;
import static org.mockito.Mockito.when;

@RunWith(JUnit4.class)
public class DocumentTest {

    @Mock
    private Connection connection;

    @Test
    public void testCreateShell() {
        Document document = Document.createShell("https://www.example.com");
        assertEquals("https://www.example.com", document.location());
    }

    @Test
    public void testLocation() {
        Document document = new Document("https://www.example.com");
        assertEquals("https://www.example.com", document.location());
    }

    @Test
    public void testConnection() {
        when(connection.getUri()).thenReturn("https://www.example.com");
        Document document = new Document("https://www.example.com");
        assertEquals("https://www.example.com", document.connection().getUri());
    }

    @Test
    public void testDocumentType() {
        Document document = new Document("https://www.example.com");
        DocumentType type = new DocumentType();
        when(document.getType()).thenReturn(type);
        assertEquals(type, document.documentType());
    }

    @Test
    public void testHead() {
        Element head = new Element("head");
        Document document = new Document("https://www.example.com");
        when(document.getHead()).thenReturn(head);
        assertEquals(head, document.head());
    }

    @Test
    public void testBody() {
        Element body = new Element("body");
        Document document = new Document("https://www.example.com");
        when(document.getBody()).thenReturn(body);
        assertEquals(body, document.body());
    }

    @Test
    public void testForms() {
        List<FormElement> forms = new ArrayList<>();
        forms.add(new FormElement());
        Document document = new Document("https://www.example.com");
        when(document.getForms()).thenReturn(forms);
        assertEquals(forms, document.forms());
    }

    @Test
    public void testExpectForm() {
        Element form = new Element("form");
        when(connection.selectFirst(Selector.form())).thenReturn(form);
        Document document = new Document("https://www.example.com");
        FormElement expectedForm = new FormElement();
        assertEquals(expectedForm, document.expectForm(".form"));
    }

    @Test
    public void testTitle() {
        Document document = new Document("https://www.example.com");
        assertEquals("", document.title());
    }

    @Test
    public void testTitleSet() {
        Document document = new Document("https://www.example.com");
        document.title("Example Title");
        assertEquals("Example Title", document.title());
    }

    @Test
    public void testCreateElement() {
        Element element = new Element();
        when(connection.createElement("div")).thenReturn(element);
        Document document = new Document("https://www.example.com");
        Element createdElement = document.createElement("div");
        assertEquals(element, createdElement);
    }

    @Test
    public void testOuterHtml() {
        Document document = new Document("https://www.example.com");
        when(document.getXml()).thenReturn("<html>Example Content</html>");
        assertEquals("<html>Example Content</html>", document.outerHtml());
    }

    @Test
    public void testText() {
        String text = "Example Text";
        Document document = new Document("https://www.example.com");
        when(document.getTextContent()).thenReturn(text);
        Element element = document.text(text);
        assertEquals(text, element.getText());
    }

    @Test
    public void testNodeName() {
        Document document = new Document("https://www.example.com");
        when(document.getNodeName()).thenReturn("html");
        assertEquals("html", document.nodeName());
    }

    @Test
    public void testCharsetSet() {
        Charset charset = Charset.forName("UTF-8");
        Document document = new Document("https://www.example.com");
        document.charset(charset);
        assertEquals(charset, document.charset());
    }

    @Test
    public void testUpdateMetaCharsetElement() {
        boolean update = true;
        when(connection.updateMetaCharsetElement()).thenReturn(update);
        Document document = new Document("https://www.example.com");
        document.updateMetaCharsetElement();
        assertEquals(true, document.updateMetaCharsetElement());
    }

    @Test
    public void testOutputSettingsSet() {
        OutputSettings outputSettings = new OutputSettings();
        Document document = new Document("https://www.example.com");
        document.outputSettings(outputSettings);
        assertEquals(outputSettings, document.outputSettings());
    }

    @Test
    public void testParserSet() {
        Parser parser = new Parser();
        when(connection.setParser(parser)).thenReturn(parser);
        Document document = new Document("https://www.example.com");
        Document parsedDocument = document.parser(parser);
        assertEquals(parsedDocument, document);
    }
}
