import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertTrue;
import static org.mockito.ArgumentMatchers.anyString;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class TagTest {

    @Mock
    private ParseSettings settings;

    @Mock
    private String tagName;

    @Test
    public void getName_ValidTag_ReturnsTagName() {
        when(settings.getName()).thenReturn(tagName);
        assertEquals(tagName, new Tag(settings).getName());
    }

    @Test
    public void normalName_ValidTagName_ReturnsNormalizedTagName() {
        when(settings.getName()).thenReturn("test");
        assertEquals("test", new Tag(settings).normalName());
    }

    @Test
    public void valueOf_ValidTagName_ReturnsTagObject() {
        when(settings.getName()).thenReturn(tagName);
        assertEquals(new Tag(settings), Tag.valueOf(tagName));
    }

    @Test
    public void isBlock_ValidTag_ReturnsTrue() {
        when(settings.isBlock(tagName)).thenReturn(true);
        assertTrue(new Tag(settings).isBlock());
    }

    @Test
    public void formatAsBlock_ValidBlockTag_ReturnsTrue() {
        when(settings.isBlock(tagName)).thenReturn(true);
        assertTrue(new Tag(settings).formatAsBlock());
    }

    @Test
    public void isInline_ValidTag_ReturnsFalse() {
        when(settings.isInline(tagName)).thenReturn(false);
        assertFalse(new Tag(settings).isInline());
    }

    @Test
    public void isEmpty ValidEmptyTagName_ReturnsTrue() {
        when(settings.getName()).thenReturn("");
        assertTrue(new Tag(settings).isEmpty());
    }

    @Test
    public void isSelfClosing_ValidTag_ReturnsFalse() {
        when(settings.isSelfClosing(tagName)).thenReturn(false);
        assertFalse(new Tag(settings).isSelfClosing());
    }

    @Test
    public void setSelfClosing_ValidTagName_ReturnsNewTag() {
        when(settings.setSelfClosing(tagName)).thenReturn(true);
        Tag newTag = new Tag(settings);
        assertTrue(newTag.isSelfClosing());
    }

    @Test
    public void equals_ValidTagsReturnEqualValue() {
        Tag tag1 = new Tag(settings);
        Tag tag2 = new Tag(settings);
        assertEquals(tag1, tag2);
    }

    @Test
    public void hashCode_ValidTagReturnsSameHashCode() {
        Tag tag1 = new Tag(settings);
        Tag tag2 = new Tag(settings);
        assertEquals(tag1.hashCode(), tag2.hashCode());
    }
}
