package org.jsoup.safety;

import org.jsoup.helper.Validate;
import org.jsoup.nodes.Attribute;
import org.jsoup.nodes.Attributes;
import org.jsoup.nodes.Element;
import java.util.HashMap;
import java.util.HashSet;
import java.util.Iterator;
import java.util.Map;
import java.util.Set;
import static org.jsoup.internal.Normalizer.lowerCase;

public class SafelistTest {

    public void testNoneSafelist() {
        Safelist safelist = Safelist.nonce();
        Validate.assertNotNull(safelist);
    }

    public void testSimpleTextSafelist() {
        Safelist safelist = Safelist.simpleText();
        Validate.assertNotNull(safelist);
        Validate.assertNotEquals("", safelist.toString());
    }

    public void testBasicSafelist() {
        Safelist safelist = Safelist.basic();
        Validate.assertNotNull(safelist);
        Validate.assertNotEquals("", safelist.toString());
    }

    public void testBasicWithImagesSafelist() {
        Safelist safelist = Safelist.basicWithImages();
        Validate.assertNotNull(safelist);
        Validate.assertNotEquals("", safelist.toString());
    }

    public void testRelaxedSafelist() {
        Safelist safelist = Safelist.relaxed();
        Validate.assertNotNull(safelist);
        Validate.assertNotEquals("", safelist.toString());
    }

    public void testNewInstanceCreation() {
        Safelist safelist1 = new Safelist();
        Safelist safelist2 = new Safelist(safelist1);
        Validate.assertSame(safelist1, safelist2);
    }

    public void testAddTagsSafelist() {
        Safelist safelist = Safelist.none();
        safelist = safelist.addTags("p", "img");
        Validate.assertNotNull(safelist);
        Validate.assertTrue(safelist.isSafeTag("p"));
        Validate.assertFalse(safelist.isSafeTag("img"));
    }

    public void testRemoveTagsSafelist() {
        Safelist safelist = Safelist.none();
        safelist = safelist.removeTags("img");
        Validate.assertNotNull(safelist);
        Validate.assertFalse(safelist.isSafeTag("img"));
        Validate.assertTrue(safelist.isSafeTag("p"));
    }

    public void testAddAttributesSafelist() {
        Safelist safelist = Safelist.none();
        safelist = safelist.addAttributes("a", "href");
        Validate.assertNotNull(safelist);
        Validate.assertFalse(safelist.isSafeAttribute("a", new Element(), new Attribute()));
        Validate.assertTrue(safelist.isSafeAttribute("a", new Element(), new Attribute("href")));
    }

    public void testRemoveAttributesSafelist() {
        Safelist safelist = Safelist.none();
        safelist = safelist.removeAttributes("a");
        Validate.assertNotNull(safelist);
        Validate.assertFalse(safelist.isSafeAttribute("a", new Element(), new Attribute()));
        Validate.assertTrue(safelist.isSafeAttribute("a", new Element(), new Attribute("href")));
    }

    public void testAddEnforcedAttributeSafelist() {
        Safelist safelist = Safelist.none();
        safelist = safelist.addEnforcedAttribute("a", "href", "http://example.com");
        Validate.assertNotNull(safelist);
        Validate.assertFalse(safelist.isSafeAttribute("a", new Element(), new Attribute()));
        Validate.assertTrue(safelist.isSafeAttribute("a", new Element(), new Attribute("href")));
    }

    public void testRemoveEnforcedAttributeSafelist() {
        Safelist safelist = Safelist.none();
        safelist = safelist.removeEnforcedAttribute("a");
        Validate.assertNotNull(safelist);
        Validate.assertFalse(safelist.isSafeAttribute("a", new Element(), new Attribute()));
        Validate.assertTrue(safelist.isSafeAttribute("a", new Element(), new Attribute("href")));
    }

    public void testPreserveRelativeLinksSafelist() {
        Safelist safelist = Safelist.none();
        safelist = safelist.preserveRelativeLinks(true);
        Validate.assertNotNull(safelist);
        Validate.assertTrue(safelist.isSafeAttribute("a", new Element(), new Attribute()));
    }

    public void testAddProtocolsSafelist() {
        Safelist safelist = Safelist.none();
        safelist = safelist.addProtocols("script", "src");

        Validate.assertFalse(safelist.isSafeAttribute("script", new Element(), new Attribute()));
        Validate.assertTrue(safelist.isSafeAttribute("script", new Element(), new Attribute("src")));

    public void testRemoveProtocolsSafelist() {

        Validate.assertNotNull(safelist);
        Validate.assertFalse(safelist.isSafeAttribute("script", new Element(), new Attribute()));
        Validate.assertTrue(safelist.isSafeAttribute("script", new Element(), new Attribute("src")));
    }

    public void testIsSafeTagSafelist() {
        Safelist safelist = Safelist.none();
        Validate.assertTrue(safelist.isSafeTag("p"));
        Validate.assertFalse(safelist.isSafeTag("img"));
    }
}
