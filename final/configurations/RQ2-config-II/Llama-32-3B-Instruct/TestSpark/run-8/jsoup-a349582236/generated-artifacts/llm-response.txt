package org.jsoup.parser;

import org.junit.Test;
import static org.junit.Assert.*;
import java.util.HashMap;
import java.util.Map;
import org.jsoup.helper.Validate;
import org.jsoup.internal.Normalizer;
import org.jsoup.parser.Tag;

public class TagTest {

    @Test
    public void getName_EmptyString_ReturnsEmptyString() {
        String tagName = "";
        assertEquals("", Tag.valueOf(tagName).getName());
    }

    @Test
    public void getName_NonEmptyString_ReturnsTaggedName() {
        String tagName = "div";
        assertEquals("div", Tag.valueOf(tagName).getName());
    }

    @Test
    public void normalName_OnlyLetterInTagName_ReturnsOriginalTagName() {
        String tagName = "a";
        assertEquals("a", Tag.valueOf(tagName).normalName());
    }

    @Test
    public void normalName_MultipleLettersInTagName_ReturnsNormalizedTagName() {
        String tagName = "hello-world";
        assertEquals("hello-world", Tag.valueOf(tagName).normalName());
    }

    @Test
    public void isBlock_AlwaysFalseForInlineTags() {
        assertTrue(Tag.valueOf("").isInline());
        assertFalse(Tag.valueOf("<br>").isBlock());
    }

    @Test
    public void formatAsBlock_AsTrue_ReturnsTagAsBlock() {
        Tag tag = new Tag("div");
        tag.setFormatAsBlock(true);
        assertTrue(tag.formatAsBlock());
    }

    @Test
    public void isInline_AlwaysFalseForBlocks() {
        assertFalse(Tag.valueOf("<div>").isInline());
    }

    @Test
    public void isEmpty_AlwaysFalseForNonEmptyTags() {
        Tag tag = new Tag("img");
        assertFalse(tag.isEmpty());
    }

    @Test
    public void isSelfClosingAlwaysFalse_ExceptForSelfClosingTags() {
        assertTrue(Tag.valueOf("<hr>").isSelfClosing());
        assertFalse(Tag.valueOf("<p>").isSelfClosing());
    }

    @Test
    public void preserveWhitespace_AlwaysTrueForNonEmptyTags() {
        Tag tag = new Tag("input");
        assertTrue(tag.preserveWhitespace());
    }
}
