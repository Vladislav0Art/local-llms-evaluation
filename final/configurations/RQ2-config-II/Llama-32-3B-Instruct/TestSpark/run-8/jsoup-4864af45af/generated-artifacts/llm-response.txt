import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import static org.jsoup.internal.Normalizer.lowerCase;
import static org.junit.Assert.assertEquals;

@RunWith(MockitoJUnitRunner.class)
public class SafelistTest {

    @Mock
    private String[] tags;

    public SafelistTest() {}

    @Test
    public void none_addTags_test() {
        Safelist safelist = Safelist.none();
        assertEquals(0, safelist.addTags(tags));
    }

    @Test
    public void simpleText_addTags_test() {
        Safelist safelist = Safelist.simpleText();
        assertEquals("simpleText", lowerCase(safelist.addTags(tags)));
    }

    @Test
    public void basic_addTags_test() {
        Safelist safelist = Safelist.basic();
        assertEquals("basic", lowerCase(safelist.addTags(tags)));
    }

    @Test
    public void basicWithImages_removeTags_test() {
        Safelist safelist = Safelist.basicWithImages().addTags("img");
        assertEquals("basicWithImages", lowerCase(safelist.removeTags(tags)));
    }

    @Test
    public void relaxed_addAttributes_test() {
        Safelist safelist = Safelist.relaxed();
        assertEquals(0, safelist.addAttributes("a", "href"));
    }

    @Test
    public void relaxed_removeAttributes_test() {
        Safelist safelist = Safelist.relaxed().addAttributes("a", "href");
        assertEquals(0, safelist.removeAttributes("a", "href"));
    }

    @Test
    public void none_addEnforcedAttribute_test() {
        Safelist safelist = Safelist.none();
        assertEquals(0, safelist.addEnforcedAttribute("div", "data-foo", "bar"));
    }

    @Test
    public void basicPreserveRelativeLinks_test() {
        Safelist safelist = Safelist.basic().preserveRelativeLinks(true);
        assertEquals(true, safelist.preserveRelativeLinks(false));
    }

    @Test
    public void none_isSafeTag_test() {
        Safelist safelist = Safelist.none();
        assertTrue(safelist.isSafeTag("div"));
    }

    @Test
    public void simpleText_isSafeAttribute_test() {
        Safelist safelist = Safelist.simpleText().addAttributes("img", "src");
        assertEquals(true, safelist.isSafeAttribute("img", Mockito.mock(Element.class), Mockito.mock(Attribute.class)));
    }

    @Test
    public void basic_getEnforcedAttributes_test() {
        Safelist safelist = Safelist.basic().addAttributes("a", "href");
        assertEquals(1, safelist.getEnforcedAttributes("a").size());
    }
}
