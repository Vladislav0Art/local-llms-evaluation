import org.junit.Test;
import static org.junit.Assert.*;
import java.io.ByteArrayOutputStream;
import java.io.IOException;

public class TextNodeTest {

    @Test
    public void createTextNode_EmptyText_ReturnsEmptyTextNode() {
        TextNode textNode = TextNode.createFromEncoded("");
        assertNull(textNode.text());
    }

    @Test
    public void textTextNode_textIsNotBlank_CreateTextNode() {
        TextNode textNode = new TextNode("Hello");
        assertNotNull(textNode.text());
        assertFalse(textNode.isBlank());
    }

    @Test
    public void textTextNode_textIsNull_CreateTextNode() {
        TextNode textNode = new TextNode(null);
        assertNull(textNode.text());
    }

    @Test
    public void splitTextByTextOffset_NoOffset_ReturnsSameTextNode() {
        TextNode textNode = new TextNode("Hello World");
        TextNode result = textNode.splitText(0);
        assertEquals(textNode, result);
    }

    @Test
    public void getWholeTextNode_EmptyText_ReturnsEmptyString() {
        TextNode textNode = new TextNode("");
        assertEquals("", textNode.getWholeText());
    }

    @Test
    public void isBlankTextNode textIsNotBlank_CreateTextNode() {
        TextNode textNode = new TextNode("Hello World");
        assertFalse(textNode.isBlank());
    }

    @Test
    public void isBlankTextNode_textIsNull_ReturnsTrue(CreateTextNode) {
        TextNode textNode = new TextNode(null);
        assertTrue(textNode.isBlank());
    }

    @Test
    public void outerHtmlHeadTextNode textIsNotBlank_CreateTextNode() throws IOException {
        TextNode textNode = new TextNode("Hello World");
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        Appendable accum = out;
        int depth = 0;
        Document.OutputSettings outSettings = null;
        textNode.outerHtmlHead(accum, depth, outSettings);
        assertNotNull(out.toString());
    }

    @Test
    public void outerHtmlTailTextNode textIsNull_CreateTextNode() throws IOException {
        TextNode textNode = new TextNode(null);
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        Appendable accum = out;
        int depth = 0;
        Document.OutputSettings outSettings = null;
        textNode.outerHtmlTail(accum, depth, outSettings);
        assertNotNull(out.toString());
    }

    @Test
    public void cloneTextNode_CreateCloneTextNode() {
        TextNode textNode = new TextNode("Hello World");
        TextNode result = textNode.clone();
        assertEquals(textNode.text(), result.text());
    }
}
