        Validate.isTrue(Safelist.isSafeTag("text"));

    public void basic_BasicTagsRemoved() {
        Safelist result = Safelist.basic();
        Validate.isNull(result.addTags(new String[] {"a", "b"}));
        Validate.isNull(result.removeTags(new String[] {"a", "b"}));
    }

    @org.junit.Test
    public void basicWithImages_ImagesTaggedAndUntagged() {
        Safelist result = Safelist.basic();
        result = result.addTags("img");
        Validate.isTrue(Safelist.isSafeTag("img"));
        result = result.removeTags(new String[] {"img"});
        Validate.isNull(result.addTags(new String[] {"img"}));
    }

    @org.junit.Test
    public void relaxed_RelaxedTagsRemoved() {
        Safelist result = Safelist.relaxed();

    }
