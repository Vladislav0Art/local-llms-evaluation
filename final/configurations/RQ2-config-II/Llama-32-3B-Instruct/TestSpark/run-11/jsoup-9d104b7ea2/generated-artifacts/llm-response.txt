import org.junit.Test;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.LeafNode;
import org.jsoup.nodes.TextNode;
import org.jsoup.nodes.Node;
import org.jsoup.nodes.Appendable;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertFalse;
import static org.junit.Assert.assertTrue;
import static org.junit.Assert.assertNull;

public class TextNodeTest {

    @Test
    public void nodeName_textReturnsTextNode() {
        TextNode textNode = new TextNode("test");
        assertEquals(TextNode.class, textNode.nodeName());
    }

    @Test
    public void nodeName_notTextNode_returnsNull() {
        LeafNode leafNode = new LeafNode();
        assertNull(leafNode.nodeName());
    }

    @Test
    public void text_textReturnsTextNodeText() {
        TextNode textNode = new TextNode("test");
        assertEquals("test", textNode.text());
    }

    @Test
    public void text_setTextReturnsNewTextNodeWithCorrectText() {
        String expectedText = "expected";
        TextNode textNode = new TextNode();
        textNode.text(expectedText);
        assertEquals(expectedText, textNode.text());
    }

    @Test
    public void getWholeText_returnsEmptyStringWhenBlank() {
        TextNode blankTextNode = new TextNode();
        assertTrue(blankTextNode.isBlank());
        assertNull(blankTextNode.getWholeText());
    }

    @Test
    public void getWholeText_returnsCorrectStringWhenNotBlank() {
        String expectedText = "expected";
        TextNode textNode = new TextNode(expectedText);
        assertEquals(expectedText, textNode.getWholeText());
    }

    @Test
    public void isBlank_textReturnsTrueWhenBlank() {
        TextNode blankTextNode = new TextNode();
        assertTrue(blankTextNode.isBlank());
    }

    @Test
    public void isBlank_textReturnsFalseWhenNotBlank() {
        String expectedText = "expected";
        TextNode textNode = new TextNode(expectedText);
        assertFalse(textNode.isBlank());
    }

    @Test
    public void splitText_offsetReturnsNewTextNodeWithCorrectSplitText() {
        TextNode originalTextNode = new TextNode("test");
        int offset = 1;
        TextNode newTextNode = originalTextNode.splitText(offset);
        assertEquals(3, newTextNode.text().length());
    }

    @Test
    public void outerHtmlHead_accumIsNotEmpty_whenContentExists() throws IOException {
        Appendable accum = new StringBuilder();
        Document.OutputSettings out = Document.OutputSettings.builder().build();
        TextNode textNode = new TextNode("test");
        textNode.outerHtmlHead(accum, 0, out);
        assertTrue(accum.length() > 0);
    }

    @Test
    public void outerHtmlTail_accumIsNotEmpty_whenContentExists() throws IOException {
        Appendable accum = new StringBuilder();
        Document.OutputSettings out = Document.OutputSettings.builder().build();
        TextNode textNode = new TextNode("test");
        textNode.outerHtmlTail(accum, 0, out);
        assertTrue(accum.length() > 0);
    }

    @Test
    public void toString_returnsCorrectString() {
        String expectedText = "expected";
        TextNode textNode = new TextNode(expectedText);
        assertEquals(expectedText, textNode.toString());
    }

    @Test
    public void clone_returnsNewTextNodeWithSameContent() {
        TextNode originalTextNode = new TextNode("test");
        TextNode clonedTextNode = originalTextNode.clone();
        assertEquals(originalTextNode.text(), clonedTextNode.text());
    }

    @Test
    public void createFromEncoded_encodedTextReturnsNewTextNodeWithCorrectEncoding() {
        String encodedText = "expected";
        TextNode textNode = TextNode.createFromEncoded(encodedText);
        assertEquals(encodedText, textNode.text());
    }
}
