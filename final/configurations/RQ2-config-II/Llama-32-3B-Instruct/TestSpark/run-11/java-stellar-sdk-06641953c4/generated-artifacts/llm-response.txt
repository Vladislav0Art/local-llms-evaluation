public class KeyPairTest {

    @org.junit.Test
    public void keyPairCreationFromPubliclyKnownEdDsaKey_works() {
        EdDSAPublicKey publicKey = EdDSAPrivateKey.generateAndExport().getPublicKey();
        KeyPair keyPair = KeyPair.fromPubliclyKnownEdDsaKey(publicKey);
        assertEquals(publicKey, keyPair.getPublicKey());
    }

    @org.junit.Test
    public void keyPairCreationFromSecretSeed_WithValidSeed_works() {
        char[] seed = "valid_seed".toCharArray();
        KeyPair keyPair = KeyPair.fromSecretSeed(seed);
        assertNotNull(keyPair.getAccountId());
    }

    @org.junit.Test
    public void keyPairCreationFromSecretSeed_WithInvalidSeed_throwsGeneralSecurityException() {
        char[] seed = new char[0];
        assertThrows(GeneralSecurityException.class, () -> KeyPair.fromSecretSeed(seed));
    }

    @org.junit.Test
    public void canSign_returnsTrueForValidKeyPair() {
        KeyPair keyPair = KeyPair.fromPubliclyKnownEdDsaKey(EdDSAPrivateKey.generateAndExport().getPublicKey());
        assertTrue(keyPair.canSign());
    }

    @org.junit.Test
    public void canSign_returnsFalseForInvalidKeyPair() {
        assertThrows(GeneralSecurityException.class, () -> KeyPair.fromSecretSeed(new char[0]));
    }
}
