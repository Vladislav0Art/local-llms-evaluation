public class W3CDomTest {

    @Test
    public void namespaceAware_trueTest() {
        assertTrue(W3CDom.namespaceAware());
    }

    @Test
    public void namespaceAware_falseTest() {
        assertFalse(W3CDom.namespaceAware(false));
    }

    @Test
    public void convert_document_test() throws Exception {
        Document doc = createDocument();
        W3CDom.convert(doc, doc);
        assertEquals(doc, W3CDom.convert(doc));
    }

    @Test
    public void asString_document_with_properties_test() throws Exception {
        Document doc = createDocument();
        String properties = "{xml:space=\"preserve\"}";
        String result = W3CDom.asString(doc, PropertiesFromMap.create(propertiesToMap(properties)));
        assertEquals("<html xmlns=\"http://www.w3.org/1999/xhtml\"><body>Test</body></html>", result);
    }

    @Test
    public void OutputHtml_test() {
        Document doc = createDocument();
        HashMap<String, String> output = W3CDom.OutputHtml();
        assertEquals("output:html", output.get("output"));
    }

    @Test
    public void OutputXml_test() {
        Document doc = createDocument();
        HashMap<String, String> output = W3CDom.OutputXml();
        assertEquals("output:xslt", output.get("output"));
    }

    @Test
    public void fromJsoup_document_test() throws Exception {
        Document jsDoc = createDocument();
        Document result = W3CDom.fromJsoup(jsDoc);
        assertNotNull(result);
        assertTrue(result.getDocument().getNamespaceURI().equals(jsDoc.getDocument().getNamespaceURI()));
    }

    @Test
    public void fromJsoup_element_test() throws Exception {
        org.jsoup.nodes.Element jsElement = createElement();
        Document result = W3CDom.fromJsoup(jsElement);
        assertNotNull(result);
        assertTrue(result.getNamespaceURI().equals(jsElement.getNamespaceURI()));
    }

    @Test
    public void selectXpath_document_test() throws Exception {
        Document doc = createDocument();
        String xpath = "/html/body";
        NodeList result = W3CDom.selectXpath(xpath, doc);
        assertNotNull(result);
        assertEquals(2, result.getLength());
    }

    @Test
    public void contextNode_document_test() {
        Document wDoc = createDocument();
        Node contextNode = W3CDom.contextNode(wDoc);
        assertTrue(contextNode instanceof Element);
    }

    @Test
    public void sourceNodes_nodeList_test() throws Exception {
        NodeList nodeList = createNodeList();
        List<org.jsoup.nodes.Node> result = W3CDom.sourceNodes(nodeList, org.jsoup.nodes.Node.class);
        assertNotNull(result);
        assertEquals(2, result.size());
    }

    private Document createDocument() throws ParserConfigurationException {
        DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
        factory.setNamespaceAware(true);
        factory.setValidating(false);
        DocumentBuilder builder = factory.newDocumentBuilder();
        return builder.parse(new DOMSource("<html><body>Test</body></html>"));
    }

    private org.jsoup.nodes.Element createElement() {
        Element element = new org.jsoup.nodes.Element("div");
        element.attr("xml:space", "preserve");
        return element;
    }

    private NodeList createNodeList() {
        NodeList nodeList = new ArrayList<>();
        for (int i = 0; i < 2; i++) {
            Node node = createDocument().createTextNode("Test" + i);
            nodeList.add(node);
        }
        return nodeList;
    }

    private String propertiesToMap(String properties) {
        Properties props = W3CDom.propertiesFromMap(Collections.singletonMap("xml:space", "preserve"));
        StringWriter writer = new StringWriter();
        Transformer transformer = TransformerFactory.newInstance().newTransformer();
        transformer.setOutputProperty(OutputKeys.METHOD, OutputKeys.XML);
        transformer.transform(new DOMSource(props), new StreamResult(writer));
        return writer.toString();
    }

    private Map<String, String> propertiesFromMap(Map<String, String> map) {
        Properties props = new Properties();
        for (Map.Entry<String, String> entry : map.entrySet()) {
            props.setProperty(entry.getKey(), entry.getValue());
        }
        return props;
    }

}
