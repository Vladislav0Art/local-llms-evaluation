import org.junit.Test;
import org.mockito.Mockito;

public class CommentTest {

    @Test
    public void newComment_isCommentNode() {
        Node comment = new Comment("data");
        assert comment instanceof Comment;
    }

    @Test
    public void newComment_nodeNameIsComment() {
        Node comment = new Comment("data");
        assertEquals("comment", comment.nodeName());
    }

    @Test
    public void newComment_dataIsEmpty() {
        Node comment = new Comment("");
        assertTrue(comment.getData().isEmpty());
    }

    @Test
    public void newComment_dataIsNotNull() {
        Node comment = new Comment("data");
        assertNotNull(comment.getData());
    }

    @Test
    public void newComment_cloneIsSame() {
        Node comment1 = new Comment("data");
        Node comment2 = comment1.clone();
        assertTrue(comment1 == comment2);
    }

    @Test
    public void outerHtmlHead_setsAccumulatorWithCommentData() throws IOException {
        Comment comment = new Comment("data");
        Appendable accum = Mockito.mock(Appendable.class);
        Document.OutputSettings out = new Document.OutputSettings();
        comment.outerHtmlHead(accum, 0, out);
        Mockito.verify(accum).append(comment.getData());
    }

    @Test
    public void outerHtmlTail_setsAccumulatorWithCommentData() throws IOException {
        Comment comment = new Comment("data");

    }

        assertAsNull(comment.asXmlDeclaration());

        Comment comment = new Comment("data");
