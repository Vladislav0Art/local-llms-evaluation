import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;
import org.junit.Test;
import static org.junit.Assert.*;
import org.mockito.Mockito;

public class DocumentTest {

    @Test
    public void createShellValidUriShouldReturnDocument() {
        Document document = Document.createShell("http://example.com");
        assertNotNull(document);
    }

    @Test
    public void createShellInvalidUriShouldThrowException() {
        assertThrows(InvalidUriException.class, () -> Document.createShell(" invalidUri "));
    }

    @Test
    public void locationShouldReturnUri() {
        String uri = "http://example.com";
        Document document = new Document(uri);
        assertEquals(uri, document.location());
    }

    @Test
    public void connectionShouldReturnConnection() {
        Connection connection = Mockito.mock(Connection.class);
        Document document = new Document("http://example.com");
        Document result = document.connection();
        assertTrue(result instanceof Document);
        assertEquals(connection, result);
    }

    @Test
    public void connectionShouldThrowExceptionOnNull() {
        assertThrows(NullPointerException.class, () -> new Document("http://example.com").connection(null));
    }

    @Test
    public void titleShouldSetTitle() {
        String title = "test";
        Document document = new Document("http://example.com");
        document.title(title);
        assertEquals(title, document.title());
    }

    @Test
    public void createElementShouldReturnElement() {
        Element element = Mockito.mock(Element.class);
        Document document = new Document("http://example.com");
        Element result = document.createElement("test");
        assertTrue(result instanceof Element);
        assertEquals(element, result);
    }

    @Test
    public void outerHtmlShouldContainAllElements() {
        Element head = Mockito.mock(Element.class);
        Element body = Mockito.mock(Element.class);
        Document document = new Document("http://example.com");
        document.head(head);
        document.body(body);

        String html = document.outerHtml();
        assertTrue(html.contains("<head>"));
        assertTrue(html.contains("</head>"));
        assertTrue(html.contains("<body>"));
        assertTrue(html.contains("</body>"));
    }

    @Test
    public void textShouldSetText() {
        String text = "test";
        Document document = new Document("http://example.com");
        Element element = Mockito.mock(Element.class);
        document.text(element, text);

        assertEquals(text, element.html());
    }
}
