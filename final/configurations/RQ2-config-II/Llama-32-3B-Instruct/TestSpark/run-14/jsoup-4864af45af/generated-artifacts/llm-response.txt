        Safelist safelist = Safelist.none();

        assertFalse(safelist.isSafeTag("script"));
    }

    @Test
    public void simpleText_SafeTagsTest() {
        Safelist safelist = Safelist.simpleText();
        assertTrue(safelist.isSafeTag("p"));
        assertTrue(safelist.isSafeTag("span"));
        assertFalse(safelist.isSafeTag("img"));
    }

    public void basic_NoAttributesTest() {

        assertTrue(safelist.isSafeAttribute("", null, null));

    }

    public void basic_WithAttributes_SafeTagsTest() {
