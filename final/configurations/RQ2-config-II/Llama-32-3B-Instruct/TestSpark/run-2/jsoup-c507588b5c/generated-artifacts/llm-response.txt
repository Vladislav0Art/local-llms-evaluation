import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.TextNode;
import org.jsoup.helper.Validate;
import org.jsoup.internal.StringUtil;
import java.io.IOException;

public class TextNodeTest {

    public static void testTextNodeCreation() {
        TextNode textNode = new TextNode("Hello World");
        assertEquals("Text", textNode.nodeName());
        assertTrue(Validate.is present(textNode.text()));
    }

    @Ignore
    public static void testNormaliseWhitespace() {
        String encodedText = "  Hello World  ";
        String result = StringUtil.normaliseWhitespace(encodedText);
        assertTrue(Validate.is present(result));
    }

    public static void testStripLeadingWhitespace() {
        String text = "\nHello\nWorld";
        String result = StringUtil.stripLeadingWhitespace(text);
        assertEquals("Hello\nWorld", result);
    }

    @Ignore
    public static void testLastCharIsWhitespace() {
        StringBuilder sb = new StringBuilder();
        assertTrue(TextNode.lastCharIsWhitespace(sb));
    }

    public static void testTextNodeGetWholeText() {
        TextNode textNode = new TextNode("Hello World");
        assertEquals("Hello World", textNode.getWholeText());
    }

    @Ignore
    public static void testTextNodeSplitTextOffsetGreaterThanLength() {
        TextNode textNode = new TextNode("Hello");
        try {
            textNode.splitText(10);
            fail();
        } catch (IOException e) {}
    }

    public static void testTextNodeIsBlank() {
        TextNode textNode = new TextNode("");
        assertTrue(textNode.isBlank());
    }

    @Ignore
    public static void testTextNodeOuterHtmlHeadAccumulation() {
        StringBuilder accum = new StringBuilder();
        int depth = 0;
        Document.OutputSettings out = new Document.OutputSettings();

        try {
            TextNode textNode = new TextNode("Hello World");
            textNode.outerHtmlHead(accum, depth, out);
            fail();
        } catch (IOException e) {}
    }

    public static void testTextNodeOuterHtmlTailAccumulation() {
        StringBuilder accum = new StringBuilder();
        int depth = 0;
        Document.OutputSettings out = new Document.OutputSettings();

        try {
            TextNode textNode = new TextNode("Hello World");
            textNode.outerHtmlTail(accum, depth, out);
            fail();
        } catch (IOException e) {}
    }

    public static void testTextNodeClone() throws IOException {
        TextNode original = new TextNode("Hello World");
        TextNode clone = original.clone();

        assertEquals(original, clone);
    }
}
