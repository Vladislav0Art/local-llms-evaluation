package org.jsoup.nodes;

import org.junit.Test;
import static org.junit.Assert.*;
import java.io.ByteArrayOutputStream;
import java.util.Arrays;
import java.util.Collections;

public class TextNodeTest {

    @Test
    public void nodeName_EmptyText_ReturnsEmptyString() {
        TextNode textNode = new TextNode("");
        assertEquals("", textNode.nodeName());
    }

    @Test
    public void nodeName_SingleCharacterText_ReturnsSingleCharacter() {
        TextNode textNode = new TextNode("a");
        assertEquals("a", textNode.nodeName());
    }

    @Test
    public void nodeName_MultiCharText_ReturnsMultiCharText() {
        TextNode textNode = new TextNode("abcdefg");
        assertEquals("abcdefg", textNode.nodeName());
    }

    @Test
    public void nodeName_TextWithQuotes_ReturnsQuotedText() {
        TextNode textNode = new TextNode("'a' 'b' 'c'");
        assertEquals("'a' 'b' 'c'", textNode.nodeName());
    }

    @Test
    public void text_EmptyText_ReturnsEmptyString() {
        TextNode textNode = new TextNode("");
        assertEquals("", textNode.text());
    }

    @Test
    public void text_SingleCharacterText_ReturnsSingleCharacter() {
        TextNode textNode = new TextNode("a");
        assertEquals("a", textNode.text());
    }

    @Test
    public void text_MultiCharText_ReturnsMultiCharText() {
        TextNode textNode = new TextNode("abcdefg");
        assertEquals("abcdefg", textNode.text());
    }

    @Test
    public void text_TextWithQuotes_ReturnsQuotedText() {
        TextNode textNode = new TextNode("'a' 'b' 'c'");
        assertEquals("'a' 'b' 'c'", textNode.text());
    }

    @Test
    public void textText_MultiCharText_ReturnsMultiCharText() {
        TextNode textNode = new TextNode("abcdefg");
        assertEquals("abcdefg", textNode.textText());
    }

    @Test
    public void splitText_WholeText_CopiesOriginalText() {
        TextNode textNode = new TextNode("abcdefg");
        TextNode result = textNode.splitText(0);
        assertEquals("abcdefg", result.text());
    }

    @Test
    public void clone_CloneCreatesNewTextNode() {
        TextNode original = new TextNode("abcdefg");
        TextNode cloned = original.clone();
        assertNotNull(cloned);
        assertNotSame(original, cloned);
    }

    @Test
    public void createFromEncoded_CreateFromEncoded_ReturnsCorrectText() {
        String encodedText = "UTF-8 encoded text";
        TextNode result = TextNode.createFromEncoded(encodedText);
        assertNotNull(result);
        assertEquals(encodedText, result.text());
    }

    @Test
    public void normaliseWhitespace_SingleCharacterText_ReturnsSingleCharacterWithNoWhitespace() {
        String original = "a";
        String expected = "a";
        String result = StringUtil.normaliseWhitespace(original);
        assertEquals(expected, result);
    }

    @Test
    public void stripLeadingWhitespace_TextWithLeadingWhitespace_ReturnsTrimmedText() {
        String original = "\t";
        String expected = "";
        String result = StringUtil.stripLeadingWhitespace(original);
        assertEquals(expected, result);
    }
}
