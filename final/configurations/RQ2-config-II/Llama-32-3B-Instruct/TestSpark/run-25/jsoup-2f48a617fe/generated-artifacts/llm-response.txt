import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.XmlDeclaration;
import org.jsoup.parser.ParseSettings;
import org.jsoup.parser.Parser;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNull;
import static org.junit.Assert.assertTrue;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class CommentTest {

    @Mock
    private Appendable accum;

    @Test
    public void newCommentWithData ShallReturnNewComment() {
        String data = "data";
        Comment comment = new Comment(data);
        assertEquals(data, comment.getData());
    }

    @Test
    public void newCommentWithoutData ShallReturnEmptyString() {
        String data = "";
        Comment comment = new Comment(data);
        assertEquals("", comment.getData());
    }

    @Test
    public void commentSetData ShallUpdateData() {
        String data = "data";
        Comment comment = new Comment(data);
        comment.setData("newData");
        assertEquals("newData", comment.getData());
    }

    @Test
    public void outerHtmlHead ShallNotThrowExceptionWithEmptyData() throws IOException {
        when(accum.append(anyString())).thenReturn(null);
        String data = "";
        Comment comment = new Comment(data);
        comment.outerHtmlHead(accum, 0, null);
        // No exception is thrown.
    }

    @Test
    public void outerHtmlHeadShallThrowIOExceptionWithEmptyData() throws IOException {
        when(accum.append(anyString())).thenThrow(IOException.class);
        String data = "";
        Comment comment = new Comment(data);
        assertThrows(IOException.class, () -> comment.outerHtmlHead(accum, 0, null));
    }

    @Test
    public void outerHtmlTailShallNotThrowException() {
        when(accum.append(anyString())).thenReturn(null);
        Comment comment = new Comment("");
        comment.outerHtmlTail(accum, 0, null);
        // No exception is thrown.
    }

    @Test
    public void clone ShallReturnNewCommentWithSameData() {
        String data = "data";
        Comment comment = new Comment(data);
        Comment clonedComment = comment.clone();
        assertEquals(data, clonedComment.getData());
    }

    @Test
    public void isXmlDeclaration ShallReturnFalse() {
        Comment comment = new Comment("");
        assertFalse(comment.isXmlDeclaration());
    }

    @Test
    public void asXmlDeclaration ShallReturnNullIfNotXmlDeclaration() {
        Comment comment = new Comment("");
        assertNull(comment.asXmlDeclaration());
    }
}
