import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.TextNode;
import org.junit.Test;
import static org.hamcrest.MatcherAssert.assertThat;
import static org.hamcrest.Matchers.is;
import static org.junit.Assert.*;

public class TextNodeTest {

    @Test
    public void text()[MethodUnderTestTextGet] {
        TextNode node = new TextNode("Hello World");
        assertThat(node.text(), is("Hello World"));
    }

    @Test
    public void textIsBlank()[MethodUnderTestTextGetBlank] {
        TextNode node = new TextNode("");
        assertTrue(node.isBlank());
    }

    @Test
    public void createFromEncoded_[ScenarioEncodedCreationTest](Document document) {
        String encodedText = "Hello%20World";
        TextNode node = TextNode.createFromEncoded(encodedText);
        assertThat(node.text(), is("Hello World"));
    }

    @Test
    public void splitText_[ScenarioSplittingTextTest](Document document) {
        TextNode node = new TextNode("Hello World");
        TextNode result1 = node.splitText(6);
        assertEquals(5, result1.text().length());
        TextNode result2 = result1.splitText(result1.text().length());
        assertTrue(result2.isBlank());
    }

    @Test
    public void normaliseWhitespace_[ScenarioNormalisedWhitespace](Document document) {
        String text = "   ";
        String expected = " ";
        assertEquals(expected, TextNode.normaliseWhitespace(text));
    }

    @Test
    public void stripLeadingWhitespace_[ScenarioStrippedWhitespace](Document document) {
        String text = "   Hello World";
        String expected = "Hello World";
        assertEquals(expected, TextNode.stripLeadingWhitespace(text));
    }

    @Test
    public void lastCharIsWhitespace_[ScenarioLastWhitespace](Document document) {
        StringBuilder sb = new StringBuilder(" ");
        assertTrue(TextNode.lastCharIsWhitespace(sb));
    }
}
