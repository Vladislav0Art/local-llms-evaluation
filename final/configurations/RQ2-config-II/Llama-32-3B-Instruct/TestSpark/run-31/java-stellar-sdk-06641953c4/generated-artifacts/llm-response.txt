public class KeyPairTest {

        EdDSAPublicKey publicKey = EdDSAPrivateKey.generate().getPublic();

    }

    @Test
    public void canSign_ReturnsTrue_WhenPubliclyAvailable() throws GeneralSecurityException {
        assertTrue(keyPair.canSign());
    }

    @Test
    public void canSign_ReturnsFalse_WhenNotPublicallyAvailable() throws GeneralSecurityException {
        EdDSAPrivateKey privateKey = EdDSAPublicKey.generate().getPrivate();
        KeyPair keyPair = new KeyPair(privateKey);
        assertFalse(keyPair.canSign());
    }

    @Test
    public void fromSecretSeed_CreatesValidKeyPair_WithCorrectPublicKey() throws GeneralSecurityException {
        char[] seed = "secretseed".toCharArray();
        KeyPair keyPair = KeyPair.fromSecretSeed(seed);
        assertNotNull(keyPair.getPublicKey());
    }

    @Test
    public void fromXdrPublicKey_ReturnsValidKeyPair_WithCorrectPublicKey() throws IOException, GeneralSecurityException {
        byte[] publicKey = new byte[]{1, 2, 3};

        assertNotNull(keyPair.getPublicKey());
    }

    public void fromAccountId_ReturnsValidKeyPair_WithCorrectAccountId() throws IOException, GeneralSecurityException {

    }

    public void getPublicKey_ReturnsNonNullPublicKey() {
