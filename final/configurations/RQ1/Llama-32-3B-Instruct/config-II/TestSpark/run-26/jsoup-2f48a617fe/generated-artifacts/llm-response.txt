import static org.junit.Assert.assertEquals;

public class CommentTest {

    private Comment comment;

    @Before
    public void setup() {
        String data = "This is a test comment";
        comment = new Comment(data);
    }

    public CommentTest() {}

    public CommentTest_commentCreatedTest() {
        // [MethodUnderTest] [Scenario]Test
        assertEquals("#comment", comment.nodeName());
    }

    public CommentTest_dataReturnedTest() {
        // [MethodUnderTest] [Scenario]Test
        assertEquals(comment.getData(), "This is a test comment");
    }

    public CommentTest_setDataTest() {
        // [MethodUnderTest] [Scenario]Test
        comment.setData("New data");
        assertEquals("New data", comment.getData());
        assertEquals(comment, comment.setData(null));
    }

    public CommentTest_outerHtmlHeadTest() {
        // [MethodUnderTest] [Scenario]Test
        Appendable accum = new StringBuilder();
        Document.OutputSettings out = Document.OutputSettings.newBuilder().setPrettyPrint(true).build();
        comment.outerHtmlHead(accum, 0, out);
        String expectedOutput = "<!--This is a test comment-->";
        assertEquals(expectedOutput, accum.toString());
    }

    public CommentTest_outerHtmlTailTest() {
        // [MethodUnderTest] [Scenario]Test
        Appendable accum = new StringBuilder();
        Document.OutputSettings out = Document.OutputSettings.newBuilder().setOutline(true).build();
        comment.outerHtmlHead(accum, 0, out);
        accum.setLength(0); // reset the output
        comment.outerHtmlTail(accum, 0, out);
        assertEquals("", accum.toString());
    }

    public CommentTest_cloneTest() {
        // [MethodUnderTest] [Scenario]Test
        Comment clonedComment = comment.clone();
        assertSame(comment, clonedComment);
    }

    public CommentTest_isXmlDeclarationTest() {
        // [MethodUnderTest] [Scenario]Test
        assertFalse(comment.isXmlDeclaration());
    }

    public CommentTest_asXmlDeclarationTest() {
        // [MethodUnderTest] [Scenario]Test
        assertNull(comment.asXmlDeclaration());
    }

    public CommentTest_isXmlDeclarationDataTest() {
        // [MethodUnderTest] [Scenario]Test
        assertEquals(false, comment.isXmlDeclarationData("This is a test"));
    }

    public CommentTest_asXmlDeclarationDataTest() {
        // [MethodUnderTest] [Scenario]Test
        String data = "<xml><tag>value</tag></xml>";
        XmlDeclaration decl = comment.asXmlDeclaration();
        assertNotNull(decl);
        assertEquals(data, decl.getContent());
    }
}

public class XmlDeclaration {
