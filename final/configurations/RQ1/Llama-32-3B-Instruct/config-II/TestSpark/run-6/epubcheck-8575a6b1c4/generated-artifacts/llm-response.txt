public class CustomMessageFileLoader implements BeanPostProcessor {
    @Override
    public Object postProcessBeforeInitialization(Object bean, String beanName) throws BeansException {
        // load custom message file from environment variable or classpath
        String customMessageFileName = System.getenv("EPUBCHECK_CUSTOM_MESSAGE_FILE");
        if (customMessageFileName != null && customMessageFileName.length() > 0) {
            File f = new File(customMessageFileName);
            if (f.exists()) {
                bean.getClass().getField("customMessageFile").set(bean, f);
                bean.getClass().getField("useCustomMessageFile").set(true);
            }
        } else {
            // no custom message file specified
        }
        return bean;
    }

    @Override
    public Object postProcessAfterInitialization(Object bean, String beanName) throws BeansException {
        return bean;
    }
}
