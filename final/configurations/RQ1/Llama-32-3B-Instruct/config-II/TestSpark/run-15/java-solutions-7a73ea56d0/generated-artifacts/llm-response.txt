    public void next_LowerPrice_ReturnsCountOfHigherPrices() {
        OnlineStockSpan obj = new OnlineStockSpan();
        obj.list.add(80);
        obj.list.add(70);
        obj.next(60);
        int result = obj.next(70);
        assertEquals(2, result);
    }

    @Test
    public void next_HigherPrice_ReturnsOne() {
        OnlineStockSpan obj = new OnlineStockSpan();
        obj.list.add(100);
        int result = obj.next(100);
        assertEquals(1, result);
    }

    public void calculateSpans_SingleElement_ReturnsCorrectSpan() {
        OnlineStockSpan obj = new OnlineStockSpan();

        assertEquals(new int[]{1}, result);
    }

    public void calculateSpans_TwoElements_ReturnsCorrectSpans() {
        OnlineStockSpan obj = new OnlineStockSpan();
        int[] prices = {80, 70};
        int[] result = obj.calculateSpans(prices);
        assertArrayEquals(new int[]{1,2}, result);

    public void calculateSpans_MultipleElements_ReturnsCorrectSpans() {
