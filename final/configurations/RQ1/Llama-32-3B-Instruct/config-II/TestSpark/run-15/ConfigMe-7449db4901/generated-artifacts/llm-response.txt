public class CommentsConfigurationTest {

    private final CommentsConfiguration commentsConfiguration;

    @InjectMocks
    public CommentsConfigurationTest() {
        this.commentsConfiguration = new CommentsConfiguration();
    }

    public void testConstructors() {
        TestUtil.assertSame(commentsConfiguration, new CommentsConfiguration());
        TestUtil.assertSame(commentsConfiguration, new CommentsConfiguration(Collections.emptyMap()));
    }

    public void testSetCommentBasic() {
        commentsConfiguration.setComment("path", "comment-line");
        TestUtil.assertMapEquals(Map.of("path", List.of("comment-line")), commentsConfiguration.getAllComments());
    }

    public void testSetCommentMultipleLines() {
        commentsConfiguration.setComment("path", "comment-line1", "comment-line2");
        TestUtil.assertMapEquals(Map.of("path", List.of("comment-line1", "comment-line2")), commentsConfiguration.getAllComments());
    }

    public void testSetCommentSingleLine() {
        commentsConfiguration.setComment("path", "\n");
        TestUtil.assertMapEquals(Map.of("path", Collections.emptyList()), commentsConfiguration.getAllComments());
    }

    public void testGetAllComments() {
        commentsConfiguration.setComment("path1", "comment-line1");
        commentsConfiguration.setComment("path2", "comment-line2");
        TestUtil.assertMapEquals(
            Map.of(
                "path1", List.of("comment-line1"),
                "path2", List.of("comment-line2")
            ),
            commentsConfiguration.getAllComments()
        );
    }

    public void testGetAllCommentsEmpty() {
        TestUtil.assertSame(Collections.emptyMap(), commentsConfiguration.getAllComments());
    }
}
