public class XmlElement {
    private String tagName;
    private Map<String, String> attributes;
    private Node child;

    public XmlElement(String tagName) {
        this.tagName = tagName;
        this.attributes = new HashMap<>();
        this.child = null;
    }

    public void addChild(Node child) {
        this.child = child;
    }

    public void setAttribute(String key, String value) {
        this.attributes.put(key, value);
    }

    public Node getChild() {
        return child;
    }
}

public class Node {
    public interface ChildConsumer {
        void accept(Node child);
    }

    private XmlElement element;

    public Node(XmlElement element) {
        this.element = element;
    }

    public void appendChild(ChildConsumer consumer) {
        if (element != null) {
            element.addChild(this);
            consumer.accept(this);
        }
    }

    public String getTagName() {
        return element.tagName();
    }

    public Map<String, String> getAttributes() {
        return element.attributes;
    }
}

public class Attributes {
    private Map<String, String> attributes;

    public Attributes() {
        this.attributes = new HashMap<>();
    }

    public void addAttribute(String key, String value) {
        attributes.put(key, value);
    }

    public String getKey() {
        return attributes.keySet().iterator().next();
    }
}

public class DOMException extends Exception {}

public class Syntax {
    public enum Element {
        XML,
        HTML
    }

    private Element syntax;

    public Syntax(Element syntax) {
        this.syntax = syntax;
    }

    public static String getValidKey(String key, Syntax syntax) throws DOMException {
        if (syntax == Syntax.XML) {
            return key.matches("^[a-zA-Z]+") ? key : null;
        } else {
            return key.matches("^[^:]+") ? key : null;
        }
    }
}
