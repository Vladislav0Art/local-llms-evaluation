import org.junit.Test;

public class TextNodeTest {

    @Test
    public void createTextNode_GivenRawText_ReturnsCorrectTextNode() {

        TextNode textNode = new TextNode(rawText);
        assertNotNull(textNode);
        assertEquals(rawText, textNode.text());
    }

    @Test
    public void createTextNode_GivenEncodedText_ReturnsCorrectTextNode() {
        String encodedText = "&lt;Hello&gt;, &amp;World!&amp;";
        TextNode textNode = TextNode.createFromEncoded(encodedText);
        assertNotNull(textNode);
        assertEquals("Hello, World!", textNode.text());
    }

    @Test
    public void text_GivenTextNode_ReturnsCorrectText() {
        String rawText = "   Hello,   World!  ";
        TextNode textNode = new TextNode(rawText);
        String expectedText = "Hello, World!";
        assertEquals(expectedText, textNode.text());
    }

    @Test
    public void text_GivenEncodedTextNode_ReturnsCorrectText() {
        String encodedText = "&lt;Hello&gt;, &amp;World!&amp;";
        TextNode textNode = TextNode.createFromEncoded(encodedText);
        String expectedText = "Hello, World!";
        assertEquals(expectedText, textNode.text());
    }

    @Test
    public void splitText_GivenTextNodeWithPositiveOffset_ReturnsCorrectTextNode() {
        String rawText = "   Hello,   World!  ";
        TextNode textNode = new TextNode(rawText);
        int offset = 10;
        TextNode expectedTextNode = new TextNode("   World!");
        TextNode actualTextNode = textNode.splitText(offset);
        assertNotNull(actualTextNode);
        assertEquals(expectedTextNode, actualTextNode);
    }

    @Test
    public void splitText_GivenTextNodeWithNegativeOffset_ReturnsBlankTextNode() {
        String rawText = "   Hello,   World!  ";
        TextNode textNode = new TextNode(rawText);
        int offset = -10;
        TextNode expectedTextNode = new TextNode();
        TextNode actualTextNode = textNode.splitText(offset);
        assertNotNull(actualTextNode);
        assertEquals(expectedTextNode, actualTextNode);
    }

    @Test
    public void isBlank_GivenTextNodeWithOnlyWhitespace_ReturnsTrue() {
        String rawText = "   ";
        TextNode textNode = new TextNode(rawText);
        assertTrue(textNode.isBlank());
    }

    @Test
    public void isBlank_GivenTextNodeWithText_ReturnsFalse() {
        String rawText = "Hello, World!";
        TextNode textNode = new TextNode(rawText);
        assertFalse(textNode.isBlank());
    }
}
