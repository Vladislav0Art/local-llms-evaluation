public class CommentsConfigurationTest {

    private CommentsConfiguration commentsConfiguration;

    @Before
    public void setup() {
        commentsConfiguration = new CommentsConfiguration();
    }

    /**
     * Test constructor
     */
    public void testConstructorEmptyComments() {
        assert commentsConfiguration.comments.isEmpty();
    }

    /**
     * Test constructor with existing comments
     */
    public void testConstructorExistingComments() {
        Map<String, List<String>> commentsMap = new HashMap<>();
        commentsMap.put("path", Arrays.asList("line1", "line2"));
        CommentsConfiguration expectedCommentsConfiguration = new CommentsConfiguration(commentsMap);
        assertEquals(expectedCommentsConfiguration.comments, commentsConfiguration.comments);
    }

    /**
     * Test setComment with no comment lines
     */
    public void testSetCommentNoCommentLines() {
        commentsConfiguration.setComment("path", "\n");
        assertTrue(isEmptyList(commentsConfiguration.getAllComments().get("path")));
    }

    /**
     * Test setComment with one or more comment lines
     */
    public void testSetCommentOneOrMoreCommentLines() {
        String[] commentLines = {"line1", "line2"};
        commentsConfiguration.setComment("path", commentLines);
        assertEquals(Arrays.asList(commentLines), commentsConfiguration.getAllComments().get("path"));
    }

    /**
     * Test setComment with multiple comment lines
     */
    public void testSetCommentMultipleCommentLines() {
        String[] commentLines = {"line1", "line2", "line3"};
        commentsConfiguration.setComment("path", commentLines);
        assertEquals(Arrays.asList(commentLines), commentsConfiguration.getAllComments().get("path"));
    }

    /**
     * Test getAllComments with empty comments
     */
    public void testGetAllCommentsEmptyComments() {
        Map<String, List<String>> expectedComments = new HashMap<>();
        CommentsConfiguration expectedCommentsConfiguration = new CommentsConfiguration(expectedComments);
        assertEquals(expectedComments, commentsConfiguration.getAllComments());
    }

    /**
     * Test getAllComments with existing comments
     */
    public void testGetAllCommentsExistingComments() {
        Map<String, List<String>> commentsMap = new HashMap<>();
        commentsMap.put("path", Arrays.asList("line1", "line2"));
        CommentsConfiguration expectedCommentsConfiguration = new CommentsConfiguration(commentsMap);
        assertEquals(expectedCommentsConfiguration.getAllComments(), commentsConfiguration.getAllComments());
    }

    /**
     * Check if list is empty
     *
     * @param list the list to check
     */
    private boolean isEmptyList(List<String> list) {
        return list == null || list.isEmpty();
    }
}
