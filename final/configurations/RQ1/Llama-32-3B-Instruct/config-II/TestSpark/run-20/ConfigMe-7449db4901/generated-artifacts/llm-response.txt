public class CommentsConfigurationTest {

    private CommentsConfiguration commentsConfiguration;
    private Map<String, List<String>> comments;

    @Mock
    private SettingsHolder settingsHolder;

    @Before
    public void setUp() {
        // Arrange
        comments = new HashMap<>();
        commentsConfiguration = new CommentsConfiguration();
    }

    /**
     * Test that a comment is added to the map.
     */
    public void [ConstructorTest]TestAddCommentToMap() {
        // Act & Assert
        comments.add("path", Arrays.asList("comment"));
        assertEquals(1, comments.size());
        assertTrue(comments.containsValue(Arrays.asList("comment")));
    }

    /**
     * Test that a comment is removed from the map.
     */
    public void [RemoveCommentTest]TestRemoveCommentFromMap() {
        // Act & Assert
        commentsConfiguration.setComment("path", "comment");
        comments.remove("path");
        assertFalse(comments.containsValue(Arrays.asList("comment")));
    }

    /**
     * Test that a comment is added to the map with a new-line.
     */
    public void [AddNewLineTest]TestAddNewLineToMap() {
        // Act & Assert
        String[] commentLines = {"comment", "\n"};
        commentsConfiguration.setComment("path", commentLines);
        assertEquals(1, comments.get("path").size());
        assertTrue(comments.get("path").contains("\n"));
    }

    /**
     * Test that a read-only view is returned.
     */
    public void [GetReadOnlyViewTest]TestGetReadOnlyView() {
        // Act & Assert
        CommentsConfiguration copy = new CommentsConfiguration();
        copy.comments.put("path", Arrays.asList("comment"));
        assertEquals(1, copy.getAllComments().size());
        assertTrue(copy.getAllComments().containsValue(Arrays.asList("comment")));
    }

    /**
     * Test that an existing comment is added.
     */
    public void [AddExistingCommentTest]TestAddExistingComment() {
        // Act & Assert
        comments.add("path", Arrays.asList("comment"));
        commentsConfiguration.setComment("path", "existing comment");
        assertEquals(1, comments.get("path").size());
        assertTrue(comments.get("path").contains("existing comment"));
    }

    /**
     * Test that an empty map is created.
     */
    public void [EmptyMapTest]TestCreateEmptyMap() {
        // Act & Assert
        commentsConfiguration = new CommentsConfiguration();
        assertEquals(0, comments.size());
    }
}
