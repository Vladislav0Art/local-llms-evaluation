import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import static org.hamcrest.MatcherAssert.assertThat;
import static org.hamcrest.Matchers.is;
import static org.junit.Assert.assertEquals;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class TextNodeTest {

    @Mock
    private Node parentNode;

    @Test
    public void givenTextNodeWhenCreatedWithEmptyStringThenBlank() {
        String text = "";
        TextNode textNode = new TextNode(text);
        assertThat(textNode.isBlank(), is(true));
    }

    @Test
    public void givenTextNodeWhenCreatedWithNonEmptyStringThenNotBlank() {
        String text = "Hello World";
        TextNode textNode = new TextNode(text);
        assertThat(textNode.isBlank(), is(false));
    }

    @Test
    public void givenTextNodeWhenCreatedWithSingleSpaceAndNextNodeIsIndentingThenSkipThisText() {
        String text = " ";
        TextNode textNode = new TextNode(text);
        when(parentNode.next()).thenReturn(new Element());
        when((Element) parentNode).shouldIndent(Mockito.any(Document.OutputSettings.class));
        assertThat(textNode.isBlank(), is(true));
    }

    @Test
    public void givenTextNodeWhenCreatedWithNonEmptyStringAndTrimLeadingAndTrailingThenTrim() {
        String text = "   Hello World  ";
        TextNode textNode = new TextNode(text);
        when(parentNode.next()).thenReturn(new Element());
        when((Element) parentNode).shouldIndent(Mockito.any(Document.OutputSettings.class));
        assertThat(StringUtil.stripLeadingWhitespace(text), is("Hello World"));
    }

    @Test
    public void givenTextNodeWhenCreatedWithEncodedStringThenUnescape() {
        String encodedText = "&lt;&gt;";
        TextNode textNode = TextNode.createFromEncoded(encodedText);
        assertThat(textNode.text(), is("<>"));
    }

    @Test
    public void givenTextNodeWhenSplitAtOffsetThenCreatesNewTextNode() {
        String text = "Hello World";
        TextNode textNode = new TextNode(text);
        when(parentNode).thenReturn(new Element());
        TextNode splitTextNode = textNode.splitText(5);
        assertEquals("World", splitTextNode.text());
    }

    @Test
    public void givenTextNodeWhenSplitsAtOffsetOfLastCharacterThenCreatesNewTextNode() {
        String text = "Hello World";
        TextNode textNode = new TextNode(text);
        when(parentNode).thenReturn(new Element());
        TextNode splitTextNode = textNode.splitText(5);
        assertEquals("", splitTextNode.text());
    }

    @Test
    public void givenTextNodeWhenSplitsAtOffsetOfLastCharacterAndParentIsElementThenCreatesNewTextNode() {
        String text = "Hello World";
        when(parentNode).thenReturn(new Element());
        TextNode textNode = new TextNode(text);
        TextNode splitTextNode = textNode.splitText(5);
        assertEquals("World", splitTextNode.text());
    }

    @Test
    public void givenTextNodeWhenSplitsAtOffsetOfLastCharacterAndParentIsDocumentThenCreatesNewTextNode() {
        String text = "Hello World";
        when(parentNode).thenReturn(new Document());
        TextNode textNode = new TextNode(text);
        TextNode splitTextNode = textNode.splitText(5);
        assertEquals("", splitTextNode.text());
    }

    @Test
    public void givenTextNodeWhenSplitsAtOffsetOfLastCharacterAndParentIsBlankThenCreatesNewTextNode() {
        String text = "Hello World";
        when(parentNode).thenReturn(new Element());
        TextNode textNode = new TextNode(text);
        TextNode splitTextNode = textNode.splitText(5);
        assertEquals("", splitTextNode.text());
    }
}
