// HtmlEmitter.java
import java.util.ArrayList;
import java.util.List;

public class HtmlEmitter {
    private List<Tag> tags;
    private StringBuilder html;

    public HtmlEmitter() {
        tags = new ArrayList<>();
        html = new StringBuilder();
    }

    public void emit(Tag tag) {
        if (tag == null) return;
        html.append("<").append(tag.tagName).append(">");
        if (tag.contents != null) html.append(tag.contents);
        if (tag.formatAsInline) html.append("</").append(tag.tagName).append(">\n");
        else html.append("</").append(tag.tagName).append(">\n");
    }

    public void emitElement(Element element) {
        Tag tag = tags.get(element.tagName);
        if (tag == null) throw new RuntimeException("Tag not found: " + element.tagName);
        emit(tag);
        if (element.contents != null) html.append(element.contents);
        html.append("</").append(element.tagName).append(">\n");
    }

    public String getHtml() {
        return html.toString();
    }
}
