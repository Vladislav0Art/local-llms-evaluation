package leetcode.medium;

import org.junit.Test;
import static org.hamcrest.CoreMatchers.is;
import static org.junit.Assert.assertThat;
import java.util.ArrayList;
import java.util.List;
import java.util.Stack;

public class OnlineStockSpanTest {

    @Test
    public void next_[PriceLessThanZeroTest] {
        OnlineStockSpan solution = new OnlineStockSpan();
        List<Integer> list = new ArrayList<>();
        solution.list = list;
        int result = solution.next(-1);
        assertThat(result, is(0));
    }

    @Test
    public void next_[PriceGreaterThanOrEqualZeroTest] {
        OnlineStockSpan solution = new OnlineStockSpan();
        List<Integer> list = new ArrayList<>();
        solution.list = list;
        int result = solution.next(10);
        assertThat(result, is(1));
    }

    @Test
    public void calculateSpans_[PriceDecreasingSequenceTest] {
        OnlineStockSpan solution = new OnlineStockSpan();
        int[] prices = {7,6,4,3,1};
        int[] result = solution.calculateSpans(prices);
        for (int i = 0; i < result.length; i++) {
            if(i == 0) assertThat(result[i], is(1));
            else assertThat(result[i], is((i - 1 - findPreviousIndex(prices, i)) + 1));
        }
    }

    @Test
    public void calculateSpans_[PriceIncreasingSequenceTest] {
        OnlineStockSpan solution = new OnlineStockSpan();
        int[] prices = {10,9,8,7};
        int[] result = solution.calculateSpans(prices);
        for (int i = 0; i < result.length; i++) {
            if(i == 0) assertThat(result[i], is(1));
            else assertThat(result[i], is((i - 1 - findPreviousIndex(prices, i)) + 1));
        }
    }

    @Test
    public void calculateSpans_[EqualPricesTest] {
        OnlineStockSpan solution = new OnlineStockSpan();
        int[] prices = {10,9,8,7};
        int[] result = solution.calculateSpans(prices);
        for (int i = 0; i < result.length; i++) {
            if(i == 0) assertThat(result[i], is(1));
            else assertThat(result[i], is((i - 1 - findPreviousIndex(prices, i)) + 1));
        }
    }

    @Test
    public void next_[MultipleConsecutiveDecreasesTest] {
        OnlineStockSpan solution = new OnlineStockSpan();
        List<Integer> list = new ArrayList<>();
        solution.list = list;
        int result = solution.next(9);
        assertThat(result, is(2));
    }

    private int findPreviousIndex(int[] prices, int i) {
        Stack<Integer> indexStack = new Stack<>();
        for (int j = 0; j < i; j++) {
            while (!indexStack.isEmpty()
                && prices[indexStack.peek()] <= prices[i])
                indexStack.pop();
            if(indexStack.isEmpty()) return -1;
            else return indexStack.peek();
        }
        return -1;
    }

}
