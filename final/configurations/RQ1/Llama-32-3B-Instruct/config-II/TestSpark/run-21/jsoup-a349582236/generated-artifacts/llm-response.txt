import java.util.HashMap;
import java.util.Map;

public class TagFactory {

    private static final Map<String, Tag> tags = new HashMap<>();

    public static void register(Tag tag) {
        tags.put(tag.getTagName(), tag);
    }

    public static Tag getTag(String tagName) {
        return tags.get(tagName);
    }
}

// TagValidator.java
import java.util.Set;

public class TagValidator {

    private final Set<String> knownTags = new HashSet<>();

    public void addKnownTag(String tagName) {
        knownTags.add(tagName);
    }

    public boolean isValidTagName(String tagName) {
        return tags.get(tagName) != null;
    }
}
