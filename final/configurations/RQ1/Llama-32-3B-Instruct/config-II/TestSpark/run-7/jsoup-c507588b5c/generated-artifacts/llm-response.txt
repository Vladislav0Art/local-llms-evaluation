import org.junit.Before;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertFalse;
import static org.junit.Assert.assertTrue;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class TextNodeTest {

    @Mock
    private Node parentNode;

    @InjectMocks
    private TextNode textNode;

    @Before
    public void setup() {
        textNode.setParentNode(parentNode);
    }

    public void testConstructorWithBlankText() {
        textNode = new TextNode("");
        assertEquals("", textNode.text());
        assertFalse(textNode.isBlank());
    }

    public void testConstructorWithNonBlankText() {
        textNode = new TextNode("Hello World");
        assertEquals("Hello World", textNode.text());
        assertFalse(textNode.isBlank());
    }

    public void testSplitTextAtOffsetZero() {
        String originalText = "Hello World";
        int offset = 0;
        when(parentNode.toString()).thenReturn(originalText);
        when(coreValue()).thenReturn(originalText);

        TextNode tailNode = textNode.splitText(offset);
        assertEquals(originalText, tailNode.text());
        assertTrue(tailNode.isBlank());
    }

    public void testSplitTextAtOffsetNonZero() {
        String originalText = "Hello World";
        int offset = 5;
        when(parentNode.toString()).thenReturn(originalText);
        when(coreValue()).thenReturn(originalText);

        TextNode tailNode = textNode.splitText(offset);
        assertEquals("World", tailNode.text());
        assertTrue(tailNode.isBlank());
    }

    public void testIsBlankWhenEmpty() {
        textNode = new TextNode("");
        assertTrue(textNode.isBlank());
    }

    public void testIsBlankWhenNonEmptyButWhitespaceOnly() {
        textNode = new TextNode("   ");
        assertTrue(textNode.isBlank());
    }

    public void testIsBlankWhenNonEmptyButContainsOtherChars() {
        textNode = new TextNode("Hello World");
        assertFalse(textNode.isBlank());
    }

    public void testGetWholeTextReturnsOriginalText() {
        when(coreValue()).thenReturn("Hello World");

        assertEquals("Hello World", textNode.getWholeText());
    }

    public void testGetWholeTextWithTrimmingEnabled() {
        when(parentNode.toString()).thenReturn("   Hello World  ");
        when(coreValue()).thenReturn("   Hello World  ");

        assertTrue(textNode.getWholeText().trim().equals("Hello World"));
    }
}
