public class NamespaceAwareDocument extends org.jsoup.nodes.Document {
    
    private final Map<String, String> namespacesStack;
    private boolean namespaceAware = true;

    public NamespaceAwareDocument(String url) {
        super(url);
        namespacesStack = new HashMap<>();
    }

    @Override
    protected void head(org.jsoup.nodes.Node source, int depth) {
        namespacesStack.push(new HashMap<>(namespacesStack.peek()));
        if (source instanceof org.jsoup.nodes.Element) {
            // ...
        }
    }

    @Override
    protected void tail(org.jsoup.nodes.Node source, int depth) {
        if (source instanceof org.jsoup.nodes.Element && getParentNode() instanceof Element) {
            namespaceAware = true;
        }
        namespacesStack.pop();
    }

    public String getXmlNS() {
        return xmlnsKey;
    }

    public String getXmlNSPrefix() {
        return xmlnsPrefix;
    }

    public void setNamespaceAware(boolean namespaceAware) {
        this.namespaceAware = namespaceAware;
    }

    public Document copyWithoutNamespaces() {
        Document doc = new NamespaceAwareDocument(getUrl());
        for (Map.Entry<String, String> entry : namespacesStack.isEmpty() ? Collections.emptyMap() : namespacesStack.entrySet()) {
            doc.getElementsByName(entry.getKey()).removeAll();
        }
        return doc;
    }
