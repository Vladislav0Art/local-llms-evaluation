package leetcode.medium;

import org.junit.Test;
import static org.junit.Assert.assertEquals;
import java.util.ArrayList;
import java.util.List;
import java.util.Stack;

public class OnlineStockSpanTest {

  @Test
  public void next[Int]WhenPriceIsHigherThanAllPreviousPricesReturnZero() {
    OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
    int result = onlineStockSpan.next(100);
    assertEquals(0, result);
  }

  @Test
  public void next[Int]WhenPriceIsEqualToFirstElementReturnOne() {
    OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
    int result = onlineStockSpan.next(10);
    assertEquals(1, result);
  }

  @Test
  public void next[Int]WhenPriceIsLowerThanAllPreviousPricesReturnPreviousCountPlusOne() {
    OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
    List<Integer> list = new ArrayList<>();
    list.add(7);
    list.add(2);
    int result = onlineStockSpan.next(1);
    assertEquals(2, result);
  }

  @Test
  public void next[Int]WhenMultiplePricesReturnPreviousCountPlusOne() {
    OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
    List<Integer> list = new ArrayList<>();
    list.add(7);
    list.add(2);
    list.add(4);
    int result = onlineStockSpan.next(6);
    assertEquals(3, result);
  }

  @Test
  public void calculateSpansForSingleElementReturnArrayWithOne() {
    OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
    int[] prices = {100};
    int[] result = onlineStockSpan.calculateSpans(prices);
    assertEquals(1, result[0]);
  }

  @Test
  public void calculateSpansForMultipleElementsReturnCorrectArray() {
    OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
    int[] prices = {10, 7, 4, 2, 5};
    int[] result = onlineStockSpan.calculateSpans(prices);
    assertEquals(1, result[0]);
    assertEquals(2, result[1]);
    assertEquals(3, result[2]);
    assertEquals(4, result[3]);
    assertEquals(2, result[4]);
  }

}
