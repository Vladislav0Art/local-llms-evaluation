public class JsoupConverterTest {
    @Test
    public void testConvertJsoupElementToDocument() {
        JsoupConverter converter = new JsoupConverter();
        Document document = converter.convert(new org.jsoup.nodes.Element("div", "class", "test"));
        Element root = document.getRootElement();

        assertEquals("div", root.getTag());
        assertTrue(root.getAttributes().containsKey("class"));
        assertEquals("test", root.getAttributes().get("class"));
    }

    @Test
    public void testConvertJsoupTextToDocument() {
        JsoupConverter converter = new JsoupConverter();
        Document document = converter.convert(new org.jsoup.nodes.TextNode("Hello World"));
        Element text = (Element) document.getRootElement().getChild(0);

        assertEquals("Hello World", text.getText());
    }

    @Test
    public void testConvertJsoupCommentToDocument() {
        JsoupConverter converter = new JsoupConverter();
        Document document = converter.convert(new org.jsoup.nodes.Comment("This is a comment"));
        Element comment = (Element) document.getRootElement().getChild(0);

        assertEquals("This is a comment", comment.getData());
    }
}
