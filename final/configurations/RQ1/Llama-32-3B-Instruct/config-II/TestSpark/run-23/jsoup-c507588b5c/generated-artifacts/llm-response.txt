package org.jsoup.nodes;

import org.junit.Before;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import java.io.IOException;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertFalse;
import static org.junit.Assert.assertNull;
import static org.junit.Assert.assertTrue;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class TextNodeTest {

    @Mock
    private Document document;

    @Mock
    private Appendable appendable;

    private TextNode textNode;

    public void setUp() {
        textNode = new TextNode("Hello World");
    }

    @Test
    public void createTextNode_GivenRawText ReturnsTextNodeWithCorrectValue() throws IOException {
        when(document.nodeType()).thenReturn(1); // 1 for TEXT_NODE

        TextNode node = new TextNode("Raw Text");

        assertEquals(node, new TextNode.createFromEncoded(node.toString()));
    }

    @Test
    public void text_GivenExistingText ReturnsStringContainingOriginalValue() {
        String expectedText = "Hello World";
        when(textNode.coreValue()).thenReturn(expectedText);

        assertEquals(expectedText, textNode.text());
    }

    @Test
    public void isBlank_GivenBlankTextNode_ReturnsTrue() {
        assertTrue(textNode.isBlank());
    }

    @Test
    public void isBlank_GivenNonBlankTextNode_ReturnsFalse() {
        assertFalse(textNode.isBlank());
    }

    @Test
    public void splitText_GivenOffset WithinBounds_ReturnsCorrectTextAfterOffset() throws IOException {
        int offset = 5;
        String expectedText = "World";
        when(document.nodeType()).thenReturn(1); // 1 for TEXT_NODE

        TextNode node = new TextNode("Hello World");
        TextNode result = new TextNode("Result");

        Mockito.when(node.coreValue()).thenReturn(node.toString());
        Mockito.when(result.coreValue()).thenReturn(expectedText);

        assertEquals(result, textNode.splitText(offset));
    }

    @Test
    public void splitText_GivenOffsetOutOfBounds_ThrowsException() {
        int offset = 10;
        when(document.nodeType()).thenReturn(1); // 1 for TEXT_NODE

        TextNode node = new TextNode("Hello World");

        Mockito.when(node.coreValue()).thenReturn(node.toString());

        IOException exception = assertThrows(IOException.class, () -> textNode.splitText(offset));
    }

    @Test
    public void outerHtml_GivenValidContext_ReturnsCorrectOuterHtml() throws IOException {
        when(document.outputSettings()).thenReturn(new Document.OutputSettings(true));

        appendable = new StringBuilder();

        TextNode node = new TextNode("Hello World");

        String result = textNode.outerHtml(appendable, 0, document);

        assertEquals("<p>Hello World</p>", result);
    }

    @Test
    public void outerHtml_GivenInvalidContext_ReturnsEmptyString() throws IOException {
        when(document.outputSettings()).thenReturn(new Document.OutputSettings(false));

        appendable = new StringBuilder();

        TextNode node = new TextNode("Hello World");

        String result = textNode.outerHtml(appendable, 0, document);

        assertEquals("", result);
    }

    @Test
    public void toString_ReturnsCorrectOuterHtml() throws IOException {
        when(document.outputSettings()).thenReturn(new Document.OutputSettings(true));

        appendable = new StringBuilder();

        TextNode node = new TextNode("Hello World");

        String result = textNode.toString();

        assertEquals("<p>Hello World</p>", result);
    }

    @Test
    public void clone_ReturnsCorrectTextNode() {
        TextNode expectedResult = new TextNode("Clone");

        when(document.nodeType()).thenReturn(1); // 1 for TEXT_NODE

        TextNode result = textNode.clone();

        assertEquals(expectedResult, result);
    }
}
