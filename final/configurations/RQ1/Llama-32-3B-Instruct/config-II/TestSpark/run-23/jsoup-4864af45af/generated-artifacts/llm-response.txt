```java
public class Element {
    private final Attributes attributes;

    public Element(Attributes attributes) {
        this.attributes = attributes;
    }

    public String absUrl(String key) {
        return attributes.get(key).toString();
    }
}

public class Attributes {
    private final Map<String, AttributeValue> attributeMap;

    public Attributes() {
        attributeMap = new HashMap<>();
    }

    public void put(String key, AttributeValue value) {
        attributeMap.put(key, value);
    }

    public AttributeValue get(String key) {
        return attributeMap.get(key);
    }
}

public class Attribute {
    private final String key;
    private final AttributeValue value;

    public Attribute(String key, AttributeValue value) {
        this.key = key;
        this.value = value;
    }

    public String getKey() {
        return key;
    }

    public AttributeValue getValue() {
        return value;
    }
}

public class AttributeValue extends TypedValue {
    public AttributeValue(String value) {
        super(value);
    }
}

public interface EnforcedAttributes {
    Attributes get();
}
```

```java
public class All implements EnforcedAttributes {
    @Override
    public Attributes get() {
        return new Attributes();
    }
}

public class DefaultEnforcer implements EnforcedAttributes {
    @Override
    public Attributes get() {
        return new Attributes();
    }
}

public class TestElement {

        this.element = element;
    }

        System.out.println(element.isSafeAttribute(tagName, new Element(new Attributes()), new Attribute(attr, new AttributeValue("")));
    }

    public static void main(String[] args) {
        EnforcedAttributes enforcedAttr = new DefaultEnforcer();
        Elements elements = new Elements(enforcedAttr);
        TestElement testElement = new TestElement(elements.createElement());
        testElement.testIsSafeAttribute(All.class.getName(), "", "href");
    }
}

public class Elements {
    private final EnforcedAttributes enforcedAttributes;

    public Elements(EnforcedAttributes enforcedAttributes) {
        this.enforcedAttributes = enforcedAttributes;
    }

    public Element createElement() {
        return new Element(new Attributes());
    }
}
```
