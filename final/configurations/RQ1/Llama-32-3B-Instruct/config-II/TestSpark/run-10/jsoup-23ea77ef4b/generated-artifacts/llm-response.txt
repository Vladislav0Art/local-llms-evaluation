public class NamespaceAwareAttributeFilter implements javax.xml.transform sourceResolver {
    private final javax.xml.transform SourceResolver parent;
    private final javax.xml.transform OutputTarget target;

    public NamespaceAwareAttributeFilter(javax.xml.transform.SourceResolver parent, javax.xml.transform.OutputTarget target) {
        this.parent = parent;
        this.target = target;
    }

    @Override
    public java.io.InputStream getInputStream(String uri, String publicId, String systemId) throws IOException, SAXException {
        return parent.getOutputTarget().getOutputStream();
    }
}

public class NamespaceAwareTransformer extends javax.xml.transform.Transformer {
    private final javax.xml.transform sourceResolver;

    public NamespaceAwareTransformer(javax.xml.transform.SourceResolver sourceResolver) {
        this.sourceResolver = sourceResolver;
    }

    @Override
    public void transform(java.io.OutputStream out, javax.xml.transform.TransformException e) throws java.io.IOException {
        if (out instanceof java.io.InputStream) {
            ((java.io.InputStream) out).close();
        }
    }

    @Override
    public javax.xml.transform.Result newResult(javax.xml.transform.Source source, javax.xml.transform.TransformerConfig config) throws java.io.IOException, java.lang.NullPointerException, java.lang.NoSuchMethodException, java.lang.ClassCastException, javax.xml.transform.TransformException {
        javax.xml.transform.OutputTarget target = new NamespaceAwareAttributeFilter(sourceResolver, new javax.xml.transform.Result(new java.io.ByteArrayOutputStream()));
        return super.newResult(source, config);
    }
}
