    }

        Config config = new Config();
        config.setPreserveRelativeLinks(false);
        
        System.out.println(config.isSafeAttribute("div", null, null));

    }
}

import org.junit.Test;
import static org.junit.Assert.*;

public class ConfigTest {

    @Test
    public void testIsSafeAttribute() {
        Config config = new Config();
        assertTrue(config.isSafeAttribute("div", null, null));

        TestConfig testConfig = new TestConfig();
        assertFalse(testConfig.isSafeAttribute(All, null, null));
    }

    @Test
    public void testPreserveRelativeLinks() {
        Config config = new Config();
        assertTrue(config.isSafeAttribute("div", null, null));
        config.setPreserveRelativeLinks(false);
        assertFalse(config.isSafeAttribute("div", null, null));

        TestConfig testConfig = new TestConfig();
        assertFalse(testConfig.isSafeAttribute(All, null, null));
    }
}
