public class Protocol {
    private final String string;

    public Protocol(String string) {
        this.string = string;
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj) return true;
        if (obj == null) return false;
        if (getClass() != obj.getClass()) return false;
        Protocol other = (Protocol) obj;
        if (!string.equals(other.string)) return false;
        return true;
    }

    @Override
    public int hashCode() {
        final int prime = 31;
        int result = 1;
        result = prime * result + ((string == null) ? 0 : string.hashCode());
        return result;
    }

    @Override
    public String toString() {
        return string;
    }
}

public class Attributes {
    private final Map<String, String> attributes;

    public Attributes() {
        this.attributes = new HashMap<>();
    }

    public void put(String key, String value) {
        attributes.put(key, value);
    }

    public boolean containsKeyIgnoreCase(String key) {
        return attributes.containsKey(key) || attributes.values().stream()
                .anyMatch(v -> v.toLowerCase().equals(key));
    }

    public Optional<String> getIgnoreCase(String key) {
        if (attributes.containsKey(key)) {
            return Optional.of(attributes.get(key));
        }
        for (String value : attributes.values()) {
            if (value.toLowerCase().equals(key)) {
                return Optional.of(value);
            }
        }
        return Optional.empty();
    }

    public Set<String> getKeys() {
        return attributes.keySet();
    }
}

public class EnforcedAttributes {
    private final Map<String, Attributes> enforced;

    public EnforcedAttributes() {
        this.enforced = new HashMap<>();
    }

    public void put(String tagName, Attributes attrs) {
        enforced.put(tagName, attrs);
    }

    public Optional<Attributes> get(String tagName) {
        return Optional.ofNullable(enforced.get(tagName));
    }
}
