public class Enforcer {
    private final EnforcedAttributesProvider provider;

    public Enforcer(EnforcedAttributesProvider provider) {
        this.provider = provider;
    }

    public boolean isAttributeSupported(String tagName, Element el, Attribute attr) {
        EnforcedAttributes enforced = provider.getEnforcedAttributes(tagName);
        return !enforced.hasAttribute(attr.getKey());
    }
}
