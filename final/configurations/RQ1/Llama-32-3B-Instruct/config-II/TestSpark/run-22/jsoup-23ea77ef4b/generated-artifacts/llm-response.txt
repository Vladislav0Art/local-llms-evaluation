public class XmlConverter {
    private Document doc;
    private Stack<HashMap<String, String>> namespacesStack = new Stack<>();
    private boolean namespaceAware = true;

    public XmlConverter(Document doc) {
        this.doc = doc;
        namespacesStack.push(new HashMap<>());
    }

    public Element convert() {
        return buildElement(null);
    }

    private void addNamespace(String prefix) {
        namespacesStack.peek().put(prefix, getUrn());
    }

    private String getUrn() {
        if (namespaceAware && namespacesStack.size() > 0) {
            return namespacesStack.peek().getFirstKey();
        }
        return null;
    }

    public Element buildElement(org.jsoup.nodes.Element element) {
        Element el = doc.createElement(element.tagName());
        copyAttributes(element, el);
        addNamespace(el.getTagName());

        for (org.jsoup.nodes.Node child : element.childNodes()) {
            if (child instanceof org.jsoup.nodes.Element) {
                el.appendChild(buildElement((org.jsoup.nodes.Element) child));
            } else if (child instanceof org.jsoup.nodes.TextNode) {
                Text text = doc.createTextNode(((org.jsoup.nodes.TextNode) child).getWholeText());
                el.appendChild(text);
            } else if (child instanceof org.jsoup.nodes.Comment) {
                Comment comment = doc.createComment(((org.jsoup.nodes.Comment) child).getData());
                el.appendChild(comment);
            } else if (child instanceof org.jsoup.nodes.DataNode) {
                Text node = doc.createTextNode(((org.jsoup.nodes.DataNode) child).getWholeData());
                el.appendChild(node);
            }
        }

        return el;
    }

    private void copyAttributes(org.jsoup.nodes.Node source, Element dest) {
        for (Attribute attribute : source.attributes()) {
            String key = Attribute.getValidKey(attribute.getKey(), getSyntax());
            if (key != null) { // null if couldn't be coerced to validity
                dest.setAttribute(key, attribute.getValue());
            }
        }
    }

    private Syntax getSyntax() {
        return doc.getSyntax();
    }
}
