public class EnforcedAttributes {
    private final Map<String, Map<AttributeKey, AttributeValue>> values = new HashMap<>();

    public void put(String tagName, Map<AttributeKey, AttributeValue> keyVals) {
        values.put(tagName, keyVals);
    }

    public boolean has(String key) {
        return values.containsKey(key) && !values.get(key).isEmpty();
    }

    public String get(String key) {
        return values.get(getTagName(key)).getOrDefault(getAttributeKey(key), "");
    }
}

public class Attributes extends EnforcedAttributes {
    private final Map<AttributeKey, AttributeValue> values = new HashMap<>();

    @Override
    protected void put(String tagName, Map<AttributeKey, AttributeValue> keyVals) {
        super.put(tagName, keyVals);
        for (Map.Entry<AttributeKey, AttributeValue> entry : keyVals.entrySet()) {
            values.put(entry.getKey(), entry.getValue());
        }
    }

    public String get(String key) {
        return super.get(key);
    }
}
