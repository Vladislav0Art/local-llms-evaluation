public class HtmlElement {
    private String tag;
    private Map<String, String> attributes = new HashMap<>();
    private Text textContent;
    private Comment commentContent;

    public HtmlElement(String tag) {
        this.tag = tag;
    }

    public void setAttribute(String key, String value) {
        attributes.put(key, value);
    }

    public void setText(String text) {
        textContent = Text.valueOf(text);
    }

    public void setComment(String comment) {
        commentContent = Comment.valueOf(comment);
    }

    public Element getElement() {
        return new Element(tag);
    }
}

public class Element extends HtmlElement {
    private List<Node> children = new ArrayList<>();

    public Element(String tag) {
        super(tag);
    }

    public void append(Node child) {
        children.add(child);
    }

    public Text getTextContent() {
        return (Text) getContent();
    }

    public Comment getCommentContent() {
        return (Comment) getContent();
    }
}

public class Text extends Node {
    private String value;

    public Text(String value) {
        this.value = value;
    }

    public static Text of(String value) {
        return new Text(value);
    }

    @Override
    public Object getContent() {
        return value;
    }
}

public class Comment extends Node {
    private String value;

    public Comment(String value) {
        this.value = value;
    }

    public static Comment of(String value) {
        return new Comment(value);
    }

    @Override
    public Object getContent() {
        return value;
    }
}

public interface Node {
    Object getContent();
}

public class DOMException extends RuntimeException {}

public class Attributes implements Iterable<String> {
    private Map<String, String> values = new HashMap<>();

    public void add(String key, String value) {
        values.put(key, value);
    }

    @Override
    public Iterator<String> iterator() {
        return values.keySet().iterator();
    }
}

public interface Syntax {
    default boolean isValid(String key) {
        return true;
    }
}

public class ElementImpl extends Element {
    private Attributes attributes = new Attributes();

    public void setAttribute(String key, String value) {
        attributes.add(key, value);
    }

    @Override
    public Iterator<String> iterator() {
        return attributes.iterator();
    }
}
