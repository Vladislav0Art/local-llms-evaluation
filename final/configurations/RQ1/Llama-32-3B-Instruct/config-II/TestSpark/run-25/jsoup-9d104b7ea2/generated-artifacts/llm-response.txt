package org.jsoup.nodes;

import org.junit.Test;
import static org.junit.Assert.*;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

@RunWith(MockitoJUnitRunner.class)
public class TextNodeTest {

    @Mock
    private LeafNode parentNode;

    public TextNode textNode(String text) {
        return new TextNode(text);
    }

    [TextNodeCreateFromEncoded][Scenario]Test {
        String encodedText = "Hello World!";
        TextNode textNode = textNode(encodedText);
        assertEquals(encodedText, TextNode.createFromEncoded(encodedText).text());
    }

    [TextNodeSplitText][PositiveOffset]Test {
        TextNode textNode = new TextNode("Hello World!");
        TextNode splitNode = textNode.splitText(5);
        assertNotNull(splitNode);
        assertTrue(splitNode.text().length() > 0);
    }

    [TextNodeSplitText][NegativeOffset]Test {
        TextNode textNode = new TextNode("Hello World!");
        assertEquals(textNode, textNode.splitText(-1));
    }

    [TextNodeSplitText][InvalidOffset]Test {
        TextNode textNode = new TextNode("Hello World!");
        assertFalse(textNode.splitText(10).equals(textNode));
    }

    [TextNodeBlank][Scenario]Test {
        TextNode textNode = textNode("");
        assertTrue(textNode.isBlank());
    }

    [TextNodeBlank][NotBlank]Test {
        TextNode textNode = textNode("Hello World!");
        assertFalse(textNode.isBlank());
    }

    [TextNodeText][Scenario]Test {
        TextNode textNode = new TextNode("Hello World!");
        assertEquals("Hello World!", textNode.text());
    }

    [TextNodeText][Blank]Test {
        TextNode textNode = textNode("");
        assertEquals("", textNode.text());
    }
}
