import org.jsoup.nodes.Comment;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.parser.ParseSettings;
import org.jsoup.parser.Parser;
import org.junit.Test;
import static org.junit.Assert.*;
import java.io.IOException;

public class CommentTest {

    @Test
    public void newCommentNode()[MethodUnderTest][Scenario]Test {
        Comment comment = new Comment("This is a test");
        assertNotNull(comment);
    }

    @Test
    public void nodeNameShouldReturnCorrectValue()[MethodUnderTest][Scenario]Test {
        Comment comment = new Comment("This is a test");
        assertEquals("#comment", comment.nodeName());
    }

    @Test
    public void getDataShouldReturnCommentContents()[MethodUnderTest][Scenario]Test {
        Comment comment = new Comment("This is a test");
        assertEquals(comment.getData(), "This is a test");
    }

    @Test
    public void setDataShouldUpdateCommentContents()[MethodUnderTest][Scenario]Test {
        Comment comment = new Comment("This is a test");
        comment.setData("New content");
        assertEquals(comment.getData(), "New content");
    }

    @Test
    public void outerHtmlHeadShouldIncludeCommentData()[MethodUnderTest][Scenario]Test {
        Comment comment = new Comment("This is a test");
        Document document = new Document();
        Appendable appendable = new StringBuilder();
        int depth = 0;
        Document.OutputSettings out = new Document.OutputSettings();
        comment.outerHtmlHead(appendable, depth, out);
        assertEquals("<!-- This is a test -->", appendable.toString());
    }

    @Test
    public void outerHtmlTailShouldNotIncludeAnyData()[MethodUnderTest][Scenario]Test {
        Comment comment = new Comment("This is a test");
        Document document = new Document();
        Appendable appendable = new StringBuilder();
        int depth = 0;
        Document.OutputSettings out = new Document.OutputSettings();
        comment.outerHtmlTail(appendable, depth, out);
        assertEquals("", appendable.toString());
    }

    @Test
    public void cloneShouldReturnNewCommentNode()[MethodUnderTest][Scenario]Test {
        Comment comment = new Comment("This is a test");
        Comment clonedComment = comment.clone();
        assertNotNull(clonedComment);
        assertEquals(comment.getNodeName(), clonedComment.getNodeName());
        assertEquals(comment.getData(), clonedComment.getData());
    }

    @Test
    public void isXmlDeclarationShouldReturnFalse()[MethodUnderTest][Scenario]Test {
        Comment comment = new Comment("This is a test");
        assertFalse(comment.isXmlDeclaration());
    }

    @Test
    public void isXmlDeclarationShouldReturnTrueForDeclaringNode()[MethodUnderTest][Scenario]Test {
        String data = "<?xml version=\"1.0\" encoding=\"UTF-8\"?>";
        Comment comment = new Comment(data);
        assertTrue(comment.isXmlDeclaration());
    }

    @Test
    public void asXmlDeclarationShouldParseCorrectly()[MethodUnderTest][Scenario]Test {
        Comment comment = new Comment("This is a test");
        XmlDeclaration decl = comment.asXmlDeclaration();
        assertNotNull(decl);
        assertEquals(comment.getData(), decl.getContents());
    }

    @Test
    public void asXmlDeclarationShouldNotIncludeEmptyNodesInResult()[MethodUnderTest][Scenario]Test {
        Comment comment = new Comment("<xml></xml>");
        XmlDeclaration decl = comment.asXmlDeclaration();
        assertNull(decl);
    }
}
