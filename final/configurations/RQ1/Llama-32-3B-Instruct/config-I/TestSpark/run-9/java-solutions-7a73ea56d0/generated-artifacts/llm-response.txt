package leetcode.medium;

import org.junit.Test;
import static org.junit.Assert.assertEquals;

public class OnlineStockSpan {

    private int[] list = new int[0];

    public void next(int price) {
        int left = 0, right = list.length - 1;
        while (left < right && list[right] >= price) {
            right--;
        }
        list = addToList(list, right + 1);
    }

    private int[] addToList(int[] array, int index) {
        int[] newArray = new int[array.length + 1];
        System.arraycopy(array, 0, newArray, 0, index);
        newArray[index] = calculateSpan(index - 1);
        System.arraycopy(array, index, newArray, index + 1, array.length - index);
        return newArray;
    }

    private int calculateSpan(int lastIndex) {
        if (lastIndex < 0) {
            return 1;
        }
        return list[lastIndex] - list[lastIndex - 1];
    }

    public int[] calculateSpans(int[] prices) {
        for (int price : prices) {
            next(price);
        }
        return list;
    }

}
