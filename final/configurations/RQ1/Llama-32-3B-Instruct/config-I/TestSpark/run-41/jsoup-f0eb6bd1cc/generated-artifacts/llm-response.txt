import org.jsoup.Connection;
import org.junit.Test;
import static org.junit.Assert.*;
import java.net.URL;
import java.util.HashMap;
import java.util.Map;

public class UrlBuilderTest {

    @Test
    public void urlBuilderConstructionNoQueryString() {
        URL u = new URL("http://example.com/path");
        UrlBuilder builder = new UrlBuilder(u);
        assertNotNull(builder.u);
        assertNull(builder.q);
    }

    @Test
    public void urlBuilderConstructionWithQueryString() {
        URL u = new URL("http://example.com/path?a=1&b=2");
        UrlBuilder builder = new UrlBuilder(u);
        assertNotNull(builder.u);
        assertSame(builder.u.getQuery(), builder.q.toString());
    }

    @Test
    public void buildUrlNormalization() throws Exception {
        String urlStr = "http://example.com/äöü?abc=def#ghi";
        URL u = new URL(urlStr);
        UrlBuilder builder = new UrlBuilder(u);
        URL normU = builder.build();
        assertNotNull(normU);
        assertSame(normU.getProtocol(), u.getProtocol());
    }

    @Test
    public void appendKeyValNormalizesStrings() {
        Connection.KeyVal kv = new Connection.KeyVal("abc", "def");
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com/path"));
        builder.appendKeyVal(kv);
        assertNotNull(builder.q.toString());
        assertSame("abc=def", builder.q.toString());
    }

    @Test
    public void appendKeyValPreservesNonAsciiCharacters() {
        String urlStr = "http://example.com/äöü?abc=def#ghi";
        URL u = new URL(urlStr);
        UrlBuilder builder = new UrlBuilder(u);
        Connection.KeyVal kv = new Connection.KeyVal("abc", String.valueOf(Character.toChars(228)));
        builder.appendKeyVal(kv);
        assertNotNull(builder.q.toString());
        assertTrue(builder.q.toString().contains("%C3%A6%C3%B6%C3%BC"));
    }

    @Test
    public void decodePartNormalizesNonAscii() throws Exception {
        String encoded = "äöü";
        String decoded = UrlBuilder.decodePart(encoded);
        assertNotNull(decoded);
        assertEquals("äöü", decoded);
    }
}

import org.jsoup.Connection;
import org.junit.Test;
import static org.junit.Assert.*;
import java.net.URL;
import java.util.HashMap;
import java.util.Map;

public class UrlBuilderTest {

    public static class ConnectionKeyVal {
        private String key;
        private String val;

        public ConnectionKeyVal(String key, String val) {
            this.key = key;
            this.val = val;
        }

        public String getKey() {
            return key;
        }

        public String getVal() {
            return val;
        }
    }

    public static class DataUtilKeyVal extends ConnectionKeyVal {
        private String decodedVal;

        public DataUtilKeyVal(String key, String val) {
            super(key, val);
            this.decodedVal = val;
        }

        public String getDecodedVal() {
            return decodedVal;
        }
    }

    public static class UrlBuilder {
        private URL u;
        private HashMap<String, String> q;

        public UrlBuilder(URL u) {
            this.u = u;
            this.q = new HashMap<>();
        }

        public void appendKeyVal(ConnectionKeyVal kv) {
            if (kv.getVal() != null) {
                q.put(kv.getKey(), kv.getVal());
            }
        }

        private String decodePart(String encodedVal) {
            return encodedVal; // TO DO implement decoding
        }

        public URL build() {
            return u;
        }
    }

    @Test
    public void urlBuilderConstructionNoQueryString() {
        URL u = new URL("http://example.com/path");
        UrlBuilder builder = new UrlBuilder(u);
        assertNotNull(builder.u);
        assertNull(builder.q);
    }

    @Test
    public void urlBuilderConstructionWithQueryString() {
        URL u = new URL("http://example.com/path?abc=def");
        UrlBuilder builder = new UrlBuilder(u);
        builder.appendKeyVal(new ConnectionKeyVal("abc", "def"));
        assertNotNull(builder.q);
        assertEquals("def", builder.q.get("abc"));
    }

    @Test
    public void appendKeyValNormalizesStrings() {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com/path"));
        builder.appendKeyVal(new ConnectionKeyVal("abc", "def"));
        assertNotNull(builder.q);
        assertEquals("abc=def", builder.q.toString());
    }

    @Test
    public void appendKeyValPreservesNonAsciiCharacters() {
        String urlStr = "http://example.com/äöü?abc=def#ghi";
        URL u = new URL(urlStr);
        UrlBuilder builder = new UrlBuilder(u);
        ConnectionKeyVal kv = new DataUtilKeyVal("abc", String.valueOf(Character.toChars(228)));
        builder.appendKeyVal(kv);
        assertNotNull(builder.q.toString());
        assertTrue(builder.q.toString().contains("%C3%A6%C3%B6%C3%BC"));
    }

    @Test
    public void decodePartNormalizesNonAscii() {
        String encoded = "äöü";
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com/path"));
        DataUtilKeyVal decoded = (DataUtilKeyVal)builder.decodePart(encoded);
        assertNotNull(decoded.getDecodedVal());
        assertEquals("äöü", decoded.getDecodedVal());
    }
}
