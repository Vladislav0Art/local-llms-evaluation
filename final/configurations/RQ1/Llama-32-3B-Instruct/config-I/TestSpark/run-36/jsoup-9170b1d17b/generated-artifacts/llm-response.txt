// Attributes.java
public class Attributes {
    private String[] keys;
    private Object[] vals;

    // getters and setters

    public void put(String key, String value) {
        int index = indexOfKey(key);
        if (index == NotFound) {
            addKeyAndValue(key, value);
        } else {
            vals[index] = value;
        }
    }

    public String get(String key) {
        int index = indexOfKey(key);
        return (index != NotFound ? (String) vals[index] : null);
    }

    private static final int NotFound = -1;

    private int indexOfKey(String key) {
        for (int i = 0; i < keys.length; i++) {
            if (keys[i].equals(key)) {
                return i;
            }
        }
        return NotFound;
    }

    public void addKeyAndValue(String key, String value) {
        String[] newKeys = new String[keys.length + 1];
        Object[] newValue = new Object[vals.length + 1];

        System.arraycopy(keys, 0, newKeys, 0, keys.length);
        System.arraycopy(vals, 0, newValue, 0, vals.length);

        newKeys[keys.length] = key;
        newValue[newValue.length - 1] = value;

        keys = newKeys;
        vals = newValue;
    }

    public int size() {
        return keys.length;
    }
}

public class TestEqualsMatcher {
    public static void main(String[] args) {
        EqualsMatcher matcher = new EqualsMatcher();
        AssertsThat asserts = new AssertsThat();

        equals("abc", "def");
        notEquals(10, 20);
        assertTrue(true);
        assertFalse(false);

        Matcher<Integer> integerMatcher = expectedMatcher().withType(Integer.class);
        Matcher<String> stringMatcher = expectedMatcher().withType(String.class);

        matches(integerMatcher, 10);
        matches(stringMatcher, "abc");

        notMatches(integerMatcher, 20);
        notMatches(stringMatcher, "def");
    }

    public static class EqualsMatcher<T> {
        private final Object expected;

        public EqualsMatcher() {}

        public EqualsMatcher(T expected) {
            this.expected = expected;
        }

        public Matcher<Object> equals(Object actual) {
            if (!actual.equals(expected)) {
                throw new AssertionError("Expected " + expected + ", but got " + actual);
            }
            return this;
        }
    }

    public static class AssertsThat {
        private final EqualsMatcher matcher;

        public AssertsThat() {
            this.matcher = new EqualsMatcher<>();
        }

        public AssertsThat withType(Class<?> type) {
            return this;
        }

        public Matcher<Object> expectedMatcher() {
            return matcher;
        }
    }

    public static class Matcher<T> {
        private final T actual;

        public Matcher(T actual) {
            this.actual = actual;
        }

        public Matcher<T> withType(Class<?> type) {
            // implementation
            return this;
        }

        public Matcher<T> not() {
            // implementation
            return new NotMatcher<>(actual);
        }
    }

    public static class NotMatcher<T> extends Matcher<T> {
        private final T actual;

        public NotMatcher(T actual) {
            super(actual);
        }

        @Override
        public boolean matches(Object obj) {
            return !super.matches(obj);
        }
    }
}
