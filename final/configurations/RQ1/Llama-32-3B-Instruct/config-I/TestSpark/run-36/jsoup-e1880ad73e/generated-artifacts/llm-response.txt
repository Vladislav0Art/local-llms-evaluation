```java
public class UrlBuilder {
    private static final String URL_SAFE = "!#$%&'*+/=?^_`{|}~-";

    public static void main(String[] args) {
        testUrlBuilder();
    }

    public static boolean decodePart(String encoded) {
        StringBuilder result = new StringBuilder();
        for (int i = 0; i < encoded.length(); i++) {
            char c = encoded.charAt(i);
            if (URL_SAFE.indexOf(c) != -1 || c == '-') {
                result.append(c);
            } else {
                try {
                    String decodedChar = java.net.URLDecoder.decode(String.valueOf(c), "UTF-8");
                    result.append(decodedChar);
                } catch (java.io.UnsupportedEncodingException e) {
                    throw new RuntimeException(e);
                }
            }
        }
        return true;
    }

    public static void testUrlBuilder() {
        assertEquals("example.co.uk", decodePart("example.co.uk"));
        assertEquals("Hello+World", UrlBuilder.appendToAscii(decodePart("Hello%2BWorld"), true, new StringBuilder()));
        assertThrows(UnsupportedEncodingException.class, () -> UrlBuilder.decodePart(""));
        assertThrows(UnsupportedEncodingException.class, () -> UrlBuilder.appendToAscii("", false, new StringBuilder()));

        assertEquals(true, decodePart("example.co.uk").equals("example.co.uk"));
        try {
            UrlBuilder.decodePart("");
            assert false;
        } catch (UnsupportedEncodingException e) {}

        assertEquals(true, UrlBuilder.appendToAscii(decodePart("Hello%2BWorld"), true, new StringBuilder()).toString().equals("Hello+World"));
        try {
            UrlBuilder.appendToAscii("", false, new StringBuilder());
            assert false;
        } catch (UnsupportedEncodingException e) {}
    }
}
```

```java
import java.net.URI;

public class UrlBuilder {
```

```java
import java.net.URI;
import java.util.HashMap;
import java.util.Map;

public class UrlBuilder {
    public static String buildUrl(String baseUrl, Map<String, String> params) {
        URI uri = new URI(baseUrl);
        StringBuilder queryString = new StringBuilder();
        for (Map.Entry<String, String> entry : params.entrySet()) {
            if (queryString.length() > 0) {
                queryString.append("&");
            }
            queryString.append(entry.getKey()).append("=").append(entry.getValue());
        }

        return uri.getScheme() + "://" + uri.getHost() + uri.getPath();
    }

    public static void main(String[] args) {
        String baseUrl = "https://example.co.uk/path/to/base";
        Map<String, String> params = new HashMap<>();
        params.put("param1", "value1");
        params.put("param2", "value2");

        // Build URL with query parameters:
        String urlWithQuery = buildUrl(baseUrl, params);
        System.out.println(urlWithQuery);

        // Decode query string:
        String decodedQueryString = decodeQueryString(urlWithQuery.split("?")[1]);
        System.out.println(decodedQueryString);
    }

    public static String decodeQueryString(String queryString) {
        StringBuilder result = new StringBuilder();
        for (int i = 0; i < queryString.length(); i++) {
            char c = queryString.charAt(i);
            if (c == '%') {
                try {
                    int value = Integer.parseInt(queryString.substring(i + 1, i + 3), 16);
                    String decodedChar = String.format("%c", (char) value);
                    result.append(decodedChar);
                    i += 2; // Skip the next two characters
                } catch (NumberFormatException e) {}
            } else {
                try {
                    result.append(new String(queryString.charAt(i)));
                } catch (Exception e) {
                    // Ignore invalid characters
                }
            }
        }
        return result.toString();
    }
}
```
