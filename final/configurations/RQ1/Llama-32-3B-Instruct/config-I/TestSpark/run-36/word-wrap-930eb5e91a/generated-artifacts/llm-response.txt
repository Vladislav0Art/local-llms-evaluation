```python
class WordWrapper:
    def __init__(self, out):
        self.out = out

    @staticmethod
    def wrap(input_text, max_length, out):
        line = ""
        words = input_text.split()
        for word in words:
            if len(line) + len(word) > max_length:
                out.writeLine(line)
                line = word
            else:
                if line:
                    line += " "
                line += word
        if line:
            out.writeLine(line)


class LineConsumer:
    def write(self, s):
        pass

    def write_new_line(self):
        pass

    def write_line(self, line):
        pass


def main():
    console_out = print

    wrap = WordWrapper(console_out)

    test1 = "This is a very long line that should be wrapped to two lines."
    test2 = "Hello World!"
    test3 = ""

    WordWrapper.wrap(test1, 20, wrap)
    WordWrapper.wrap(test2, 10, wrap)
    WordWrapper.wrap(test3, 5, wrap)


if __name__ == "__main__":
    main()
```

In this code:

-   The `WordWrapper` class takes an output consumer (`LineConsumer`) in its constructor.
-   The `wrap` method splits the input text into words and appends them to the current line. When the next word would make the line too long, it writes the current line and starts a new one with the first word of the next line.
-   The `main` function demonstrates how to use the `WordWrapper`. It creates an instance of `WordWrapper`, wraps three test strings, and prints them to the console.
