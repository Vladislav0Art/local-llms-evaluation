public class ConnectionRequest {
    private Request request;

    public static void serialiseRequestUrl(Connection.Request req, OutputStream outStream)
            throws IOException {
        URL in = req.url();
        StringBuilder url = new StringBuilder(in.getProtocol())
                .append("://")
                .append(in.getAuthority()) // includes host, port
                .append(in.getPath());
        if (in.getQuery() != null) {
            url.append('?');
            url.append(URLEncoder.encode(in.getQuery(), Charset.forName("UTF-8")));
        }
        for (Connection.KeyVal keyVal : req.data()) {
            if (!keyVal.hasInputStream()) {
                url.append('&')
                        .append(URLEncoder.encode(keyVal.key(), Charset.forName("UTF-8")))
                        .append('=')
                        .append(URLEncoder.encode(keyVal.value(), Charset.forName("UTF-8")));
            }
        }

        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(outStream));
        bw.write(url.toString());
        bw.close();
    }

    public static boolean needsMultipart(Connection.Request req) {
        for (Connection.KeyVal keyVal : req.data()) {
            if (keyVal.hasInputStream())
                return true;
        }
        return false;
    }

    public void writeTo(OutputStream outStream) throws IOException {
        BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(outStream));
        for (Connection.KeyVal keyVal : request.data()) {
            bw.write(keyVal.key() + "=" + keyVal.value());
            bw.newLine();
        }
        bw.close();
    }

    public String getContentType() {
        return null;
    }
}

public class DefaultUploadType implements ContentType {
    @Override
    public String getMimeTyp() {
        return "application/x-www-form-urlencoded";
    }
