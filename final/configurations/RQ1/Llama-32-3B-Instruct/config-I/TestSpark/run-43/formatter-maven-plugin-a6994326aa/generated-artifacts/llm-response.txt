import org.junit.Before;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.MockitoJUnitRunner;

import java.io.IOException;
import java.io.StringReader;
import java.util.Map;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertTrue;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class CssFormatterTest {

    @Mock
    private CSSFormat formatter;

    @Mock
    private ConfigurationSource cfg;

    @Mock
    private Map<String, String> options;

    @Before
    public void setUp() {
        options.put("indent", "4");
        options.put("rgbAsHex", Boolean.TRUE.toString());
        options.put("useSourceStringValues", Boolean.FALSE.toString());

        when(cfg.getOptions()).thenReturn(options);
    }

    @Test
    public void initFormatterWithDefaults() {
        new CssFormatter().init(null, cfg);
        assertTrue(CssFormatter.this.isInitialized());
    }

    @Test
    public void initFormatterWithCustomOptions() {
        options.put("indent", "8");
        when(cfg.getOptions()).thenReturn(options);

        new CssFormatter().init(null, cfg);
        assertEquals(8, ((CssFormatter) new CssFormatter()).getIndent());
        assertTrue(CssFormatter.this.isInitialized());
    }

    @Test
    public void initFormatterWithRgbAsHexDisabled() {
        options.put("rgbAsHex", Boolean.FALSE.toString());
        when(cfg.getOptions()).thenReturn(options);

        new CssFormatter().init(null, cfg);
        assertTrue(CssFormatter.this.isInitialized());
    }

    @Test
    public void initFormatterWithUseSourceStringValuesEnabled() {
        options.put("useSourceStringValues", Boolean.TRUE.toString());
        when(cfg.getOptions()).thenReturn(options);

        new CssFormatter().init(null, cfg);
        assertTrue(CssFormatter.this.isInitialized());
    }

    @Test
    public void doFormatNullCodeReturnsNull() {
        assertEquals(null, new CssFormatter().doFormat(null, LineEnding.EOL));
    }

    @Test
    public void doFormatEmptyStringReturnsEmptyString() {
        final var code = "";
        final var ending = LineEnding.EOL;
        final var result = new CssFormatter().doFormat(code, ending);
        assertEquals(code, result);
    }

    @Test
    public void doFormatWithValidCodeAndEolShouldReturnFormatedCode() throws IOException {
        final var code = "/* test */\n\tbody {\n\t  background-color: #fff;\n}\t";
        final var ending = LineEnding.EOL;

        when(formatter.getCssText(any(CSSStyleSheetImpl.class))).thenReturn(code);

        final var result = new CssFormatter().doFormat(code, ending);
        assertEquals(code, result);
    }

    @Test
    public void doFormatWithInvalidCodeAndEolShouldReturnFormatedCode() throws IOException {
        final var code = "/* test */\n\tbody {\n\t  background-color: #fff;\n}\t";
        final var ending = LineEnding.EOL;

        when(formatter.getCssText(any(CSSStyleSheetImpl.class))).thenReturn(code);

        final var result = new CssFormatter().doFormat(code, ending);
        assertEquals(code, result);
    }

    @Test
    public void doFormatWithUnparsableCodeShouldReturnFormatedCode() throws IOException {
        final var code = "/* test */\n\tbody {\n\t  background-color: #fff;\n}\t";
        final var ending = LineEnding.EOL;

        when(formatter.getCssText(any(CSSStyleSheetImpl.class))).thenThrow(new RuntimeException());

        final var result = new CssFormatter().doFormat(code, ending);
        assertTrue(result instanceof String);
    }
}

public enum LineEnding {
    EOL("\n"),
    NL("\n");

    private final String value;

    LineEnding(String value) {
        this.value = value;
    }

    public String getValue() {
        return value;
    }
}
