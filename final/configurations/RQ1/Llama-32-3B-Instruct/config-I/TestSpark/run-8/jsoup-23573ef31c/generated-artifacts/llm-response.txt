    private String type;

    public static Parser createParser(String type) {
        return new Parser(type);
    }

    public String getType() {
        return type;
    }
}

public class Connection {
    private String url;

    public static Connection createConnection(String url) {
        return new Connection(url);
    }

    public String getUrl() {
        return url;
    }
}

public class QuirksMode {
    public enum Mode {
        noQuirks,
        quirks,
        limitedQuirks
    }

    private Mode mode;

    public static void printMode(Mode mode) {}

    public QuirksMode(Mode mode) {
        this.mode = mode;
    }

    public Mode getMode() {
        return mode;
    }
}

public class Validate {

    public static void validate(int value, int minValue, int maxValue) {
        if (value < minValue || value > maxValue) {
            throw new AssertionError("Value out of range");
        }
    }

    public static void main(String[] args) {}
}

public class Test {
    @org.junit.Test
    public void testValidate() {
        Validate.validate(10, 0, 100);
    }

    @org.junit.Test
    public void testOutputSettings() {
        OutputSettings outputSettings = new OutputSettings();
        outputSettings.setPrettyPrint(true);
    }

    @org.junit.Test
    public void testParser() {
        Parser parser = Parser.createParser("test");
        System.out.println(parser.getType());
    }

    @org.junit.Test
    public void testConnection() {
        Connection connection = Connection.createConnection("https://example.com");
        System.out.println(connection.getUrl());
    }

    @org.junit.Test
    public void testQuirksMode() {
        QuirksMode quirksMode = new QuirksMode(QuirksMode.Mode.noQuirks);
        System.out.println(quirksMode.getMode());
        QuirksMode.printMode(quirksMode.getMode());
    }
}
