package leetcode.medium;

import org.junit.Test;
import static org.junit.Assert.assertEquals;
import java.util.Arrays;

public class OnlineStockSpanTest {

    @Test
    public void next_0() {
        OnlineStockSpan solution = new OnlineStockSpan();
        int result = solution.next(0);
        assertEquals(1, result);
    }

    @Test
    public void next_Negative() {
        OnlineStockSpan solution = new OnlineStockSpan();
        int result = solution.next(-5);
        assertEquals(1, result);
    }

    @Test
    public void next_SamePrice() {
        OnlineStockSpan solution = new OnlineStockSpan();
        solution.next(3);
        solution.next(3);
        solution.next(4);
        assertEquals(2, solution.next(3));
    }

    @Test
    public void calculateSpans_EmptyInput() {
        OnlineStockSpan solution = new OnlineStockSpan();
        int[] result = solution.calculateSpans(new int[0]);
        assertArrayEquals(new int[0], result);
    }

    @Test
    public void calculateSpans_OneElement() {
        OnlineStockSpan solution = new OnlineStockSpan();
        int[] result = solution.calculateSpans(new int[]{1});
        assertEquals(new int[]{1}, result);
    }

    @Test
    public void calculateSpans_MultipleElements() {
        OnlineStockSpan solution = new OnlineStockSpan();
        int[] result = solution.calculateSpans(new int[]{100, 80, 75, 60});
        assertArrayEquals(new int[]{2, 1, 4, 5}, result);
    }

    @Test
    public void calculateSpans_MultipleElementsDecreasing() {
        OnlineStockSpan solution = new OnlineStockSpan();
        int[] result = solution.calculateSpans(new int[]{100, 80, 75, 60, 65});
        assertArrayEquals(new int[]{2, 1, 4, 5, 3}, result);
    }
}
