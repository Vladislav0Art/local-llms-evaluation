        DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
        factory.setNamespaceAware(true);
        org.w3c.dom.Document document = factory.newDocumentBuilder().newDocument();

        // Parse the HTML string into a Document object
        String[] lines = input.split("\n");
        for (String line : lines) {
            if (!line.isEmpty()) {
                Node node = document.importNode(document.createElement(line), true);
                document.getDocumentElement().appendChild(node);
            }
        }

        return document;
    }
}

      element.textContent = content;
    }
    this.html += element.outerHTML;
  }

    this.addElement("p", content);
  }

class DocumentGenerator {
  constructor() {
    this.html = "";
    this.css = "";
  }

    if (content) {
      element.textContent = content;
    }
    this.html += `<${elementName}>${element.outerHTML}</${elementName}>`;
  }

  addParagraph(content) {
    this.addElement("p", content);
  }

  generateDocument(inputHtml, inputCss) {
    // Clear the HTML and CSS content
    while (this.html.length > 0) {
      this.html = "";
    }
    while (this.css.length > 0) {
      this.css = "";
    }

    // Add input HTML to the document
    const parser = new DOMParser();
    const doc = parser.parseFromString(inputHtml, "text/html");
    this.html += `<style>${inputCss}</style>`;

    if (doc.body) {
      this.html += doc.body.outerHTML;
    }
  }
}
