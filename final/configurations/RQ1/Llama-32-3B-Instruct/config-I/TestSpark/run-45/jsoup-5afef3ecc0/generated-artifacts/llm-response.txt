import org.junit.Test;
import static org.junit.Assert.*;
import java.net.URL;
import java.net.MalformedURLException;
import java.util.HashMap;
import java.util.Map;

public class UrlBuilderTest {

    @Test
    public void newUrlBuilderValidURLTest() {
        URL url = new URL("https://example.com/path?query=value#ref");
        UrlBuilder builder = new UrlBuilder(url);
        assertNotNull(builder.u);
    }

    @Test
    public void newUrlBuilderInvalidProtocolTest() {
        assertThrows(MalformedURLException.class, () -> new UrlBuilder(new URL("invalid:https://example.com/path?query=value#ref")));
    }

    @Test
    public void appendKeyValNoQueryTest() throws UnsupportedEncodingException {
        Connection.KeyVal kv = new Connection.KeyVal("key", "value");
        UrlBuilder builder = new UrlBuilder(new URL("https://example.com/path"));
        builder.appendKeyVal(kv);
        assertNotNull(builder.q);
        assertEquals("key=value", builder.q.toString());
    }

    @Test
    public void appendKeyValQueryAlreadySetTest() throws UnsupportedEncodingException {
        Connection.KeyVal kv = new Connection.KeyVal("key2", "value");
        UrlBuilder builder = new UrlBuilder(new URL("https://example.com/path?query=value#ref"));
        builder.appendKeyVal(kv);
        assertNotNull(builder.q);
        assertEquals("key=value&key2=value", builder.q.toString());
    }

    @Test
    public void buildUrlPunyCodingHostTest() throws MalformedURLException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com:8080/path?query=value#ref"));
        assertNotNull(builder.u);
        assertEquals("https%3A//example.com:8080/path?query=value#ref", ((URL)builder.u).toString());
    }

    @Test
    public void buildUrlNonAsciiPathTest() throws MalformedURLException {
        UrlBuilder builder = new UrlBuilder(new URL("http://example.com/àçéèüìòõ/paths?query=value#ref"));
        assertNotNull(builder.u);
        assertEquals("https%3A//example.com/%C3%A0%C3%A7%C3%A6%C3%A9%C3%A7%C3%A9%C3%A8%C3%A9/%C3%A9%C3%A9u%C3%A9i%C3%B1%C3%B2%20paths?query=value#ref", ((URL)builder.u).toString());
    }

    @Test
    public void buildUrlQueryAndRefAlreadySetTest() throws MalformedURLException {
        UrlBuilder builder = new UrlBuilder(new URL("https://example.com/path"));
        assertNotNull(builder.q);
        assertNotNull(builder.u);
        assertEquals("https%3A//example.com/path?query=value#ref", ((URL)builder.u).toString());
    }

    @Test
    public void buildUrlMalformedURLExceptionTest() throws MalformedURLException {
        assertThrows(MalformedURLException.class, () -> new UrlBuilder(new URL("invalid:https://example.com/path?query=value#ref")));
    }
}
