public class AppVersion {
    private String packageName;
    private String version;

    public AppVersion(String packageName, String version) {
        this.packageName = packageName;
        this.version = version;
    }

    public String getPackageName() {
        return packageName;
    }

    public String getVersion() {
        return version;
    }

    public boolean isValidName(String name) {
        try {
            parsedName = parseName(name);
            return parsedName != null && parsedName.getPackageName().equals(packageName)
                    && parsedName.getVersion().equals(version);
        } catch (InvalidFormatException e) {
            return false;
        }
    }

    private static AppVersion parsedName;

    public static AppVersion parseName(String name) throws InvalidFormatException {
        if (parsedName == null || !parsedName.getPackageName().equals(name.substring(0, name.indexOf('/'))))
            parsedName = new AppVersion(name.substring(0, name.indexOf('/')), name.substring(name.indexOf('/') + 1));
        return parsedName;
    }

public class AppVersion {
    private String packageName;
    private String version;

    public static AppVersion createAppVersion(String packageName, String version) {
        return new AppVersion(packageName, version);
    }

    public String getPackageName() {
        return packageName;
    }

    public String getVersion() {
        return version;
    }

    public boolean isValidName(String name) {
        if (name.isEmpty()) {
            return false;
        }
        try {
            parsedName = parseName(name);
            return parsedName != null && parsedName.getPackageName().equals(packageName)
                    && parsedName.getVersion().equals(version);
        } catch (InvalidFormatException e) {
            return false;
        }
    }

    private static AppVersion parsedName;

    public static AppVersion parseName(String name) throws InvalidFormatException {
        if (parsedName == null || !parsedName.getPackageName().equals(name.substring(0, name.indexOf('/'))))
            parsedName = new AppVersion(name.substring(0, name.indexOf('/')), name.substring(name.indexOf('/') + 1));
        return parsedName;
    }

    public static class Test {
        @org.junit.Test
        public void testParseValid() throws Exception {
            AppVersion appVersion = createAppVersion("com.example.package", "1.0.0-586499");
            assertEquals("com.example.package", appVersion.getPackageName());
            assertEquals("1.0.0-586499", appVersion.getVersion());
        }

        @org.junit.Test
        public void testParseInvalid() throws Exception {
            AppVersion appVersion = createAppVersion("");
            assertNull(appVersion.getPackageName());
        }

        @org.junit.Test
        public void testGetPackage() throws Exception {
            AppVersion appVersion = createAppVersion("com.example.package", "1.0.0-586499");
            assertEquals("com.example.package", appVersion.getPackageName());
        }

        @org.junit.Test
        public void testGetVersion() throws Exception {
            AppVersion appVersion = createAppVersion("com.example.package", "1.0.0-586499");
            assertEquals("1.0.0-586499", appVersion.getVersion());
        }

        @org.junit.Test
        public void testGetInvalidName() throws Exception {
            assertFalse(AppVersion.isValidName(""));
        }

        @org.junit.Test
        public void testGetValidName() throws Exception {
            AppVersion appVersion = createAppVersion("com.example.package", "1.0.0-586499");
            assertTrue(AppVersion.isValidName(appVersion.getPackageName() + "/" + appVersion.getVersion()));
        }
    }
}
