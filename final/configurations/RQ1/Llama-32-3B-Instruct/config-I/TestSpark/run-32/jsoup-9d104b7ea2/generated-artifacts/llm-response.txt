import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertTrue;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class TextNodeTest {

    @Mock
    private Node parentNode;

    public void setup() {
        when(parentNode.addNode(1, new TextNode("")));
    }

    @Test
    public void textShouldReturnUnnormalizedText() {
        TextNode textNode = new TextNode("   ");
        assertEquals("   ", textNode.text());
    }

    @Test
    public void splitTextShouldReturnNewTextNode() throws IOException {
        TextNode textNode = new TextNode("Hello World");
        when(parentNode.addNode(1, null)).thenReturn(true);
        TextNode result = textNode.splitText(7);
        assertEquals("World", result.text());
    }

    @Test
    public void splitTextShouldReturnNullIfOffsetIsGreaterThanLength() {
        TextNode textNode = new TextNode("Hello World");
        when(parentNode.addNode(1, null)).thenReturn(true);
        TextNode result = textNode.splitText(13);
        assertNull(result);
    }

    @Test
    public void isBlankShouldReturnFalseForNonEmptyText() {
        TextNode textNode = new TextNode("   ");
        assertFalse(textNode.isBlank());
    }

    @Test
    public void isBlankShouldReturnTrueForEmptyText() {
        TextNode textNode = new TextNode("");
        assertTrue(textNode.isBlank());
    }

    @Test
    public void outerHtmlHeadShouldCallIndentIfTextIsNotEmptyAndParentIsBlockTag() throws IOException, InterruptedException {
        Appendable accum = new StringBuilder();
        Document.OutputSettings out = new Document.OutputSettings(true);
        TextNode textNode = new TextNode("   ");
        when(parentNode.addNode(1, null)).thenReturn(true);
        TextNode result = textNode.splitText(7);

        outerHtmlHead(accum, 0, out);
        String html = accum.toString();
        assertTrue(html.contains("<br>"));
    }

    @Test
    public void outerHtmlShouldReturnValidHtml() throws IOException {
        Appendable accum = new StringBuilder();
        Document.OutputSettings out = new Document.OutputSettings(true);
        TextNode textNode = new TextNode("   ");
        when(parentNode.addNode(1, null)).thenReturn(true);

        outerHtmlHead(accum, 0, out);
        String html = accum.toString();
        assertTrue(html.contains("<br>"));
    }

    @Test
    public void cloneShouldReturnSameTextNode() {
        TextNode textNode = new TextNode("   ");
        TextNode cloned = textNode.clone();
        assertEquals(textNode, cloned);
    }
}

import org.junit.Test;
import static org.junit.Assert.*;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.TextNode;
import org.jsoup.nodes.Node;
import org.jsoup.nodes.Comment;

public class GeneratedTest {

    public void splitTextShouldReturnNewTextNode() throws Exception {
        Document doc = new Document();
        Node parentNode = doc.appendElement("div");
        TextNode textNode = new TextNode("");
        parentNode.appendChild(textNode);
        Node newNode = new Node();
        parentNode.addNode(0, newNode);
        assertEquals(1, parentNode.childNodes().size());
    }

    public void outerHtmlHeadShouldCallIndentIfTextIsNotEmptyAndParentIsBlockTag() throws Exception {
        Document doc = new Document();
        Node parentNode = doc.appendElement("div");
        TextNode textNode = new TextNode("");
        parentNode.appendChild(textNode);
        assertEquals(true, parentNode.addNode(0, textNode));
    }

    public void outerHtmlShouldReturnValidHtml() throws Exception {
        Document doc = new Document();
        Node parentNode = doc.appendElement("div");
        parentNode.appendChild(new Comment(""));
        assertEquals(doc.outerHtml(), parentNode.outerHtml());
    }
}

public class TestDocumentOutputSettings {
    public void testConstructor() throws Exception {
        Document.OutputSettings out = new Document.OutputSettings(true);
        assert(out.isPrettyPrint());
    }
}
