```java
import java.time.LocalDate;
import java.time.temporal.ChronoUnit;

public class TimeDeltaCalculator {

    public static long calculateTimeDifference(LocalDate start, LocalDate end) {
        return ChronoUnit.DAYS.between(start, end);
    }

    public static double estimate(LocalDate date) {
        // Estimate time difference in years
        return (double) calculateTimeDifference(date, LocalDate.now()) / 365.25;
    }

    public static double decimalYear(LocalDate date) {
        // Calculate decimal year from date
        int year = date.getYear();
        if (date.getMonthValue() > 12 || date.isAfter(LocalDate.of(year + 1, 1, 1))) {
            year++;
        }
        return (double) year / 100;
    }
}

public class GeneratedTest {

    @org.junit.Test
    public void testCalculateTimeDifference() {
        LocalDate start = LocalDate.of(2010, 1, 1);
        LocalDate end = LocalDate.now();
        org.junit.Assert.assertEquals(TimeDeltaCalculator.calculateTimeDifference(start, end), 13_444L); 
    }

    @org.junit.Test
    public void testEstimate() {
        LocalDate forDate = LocalDate.of(2022, 1, 1);
        org.junit.Assert.assertEquals(DeltaT.estimate(forDate), TimeDeltaCalculator.estimate(forDate));
    }

    @org.junit.Test
    public void testDecimalYear() {
        LocalDate forDate = LocalDate.of(500, 12, 31);
        org.junit.Assert.assertEquals(DeltaT.decimalYear(forDate), TimeDeltaCalculator.decimalYear(forDate));
    }
}
```
