import java.util.HashMap;
import java.util.Map;

public class AttributeGenerator {

    public static Class<?> generateClass(Map<String, Object> attributesMap) throws Exception {
        String className = "GeneratedAttributes";
        StringBuilder sb = new StringBuilder(className + ".java");

        sb.append("public class ").append(className).append(" {\n");
        for (Map.Entry<String, Object> entry : attributesMap.entrySet()) {
            if (entry.getValue() instanceof Map) {
                Class<?> attributeClass = generateClass((Map<String, Object>) entry.getValue());
                sb.append("    public static ").append(entry.getKey()).append(" get").append(getterName(entry.getKey())).append("() {\n");
                sb.append("        return new ").append(attributeClass.getSimpleName()).append(".class.get").append(getterName(entry.getKey())).append("();\n");
                sb.append("    }\n");
            } else {
                sb.append("    public static ").append(entry.getKey()).append(" get").append(getterName(entry.getKey())).append("() {\n");
                sb.append("        return (").append(entry.getValue().getClass().getSimpleName()).append(") attributes.get(\"").append(entry.getKey()).append("\");\n");
                sb.append("    }\n");
            }
        }

        sb.append("}\n");

        return Class.forName(className);
    }

    private static String getterName(String name) {
        return "get" + Character.toUpperCase(name.charAt(0));
    }
}

import java.util.HashMap;
import java.util.Map;

public class DataValidator {
    private Map<String, Validator> validators;

    public DataValidator() {
        this.validators = new HashMap<>();
    }

    public void addValidator(String fieldName, Validator validator) {
        validators.put(fieldName, validator);
    }

    public boolean validate(String fieldName, Object value) throws ValidationException {
        if (validators.containsKey(fieldName)) {
            return validators.get(fieldName).validate(value);
        }
        throw new ValidationException("No validator found for field: " + fieldName);
    }
}

interface Validator {
    boolean validate(Object value) throws ValidationException;
}
