public class OnlineStockSpan {

  List<Integer> list;

  public OnlineStockSpan() {
    this.list = new ArrayList<>();
  }

  public int next(int price) {
    list.add(price);
    int count = 0;
    for (int i = list.size() - 1; i >= 0; i--) {
      if (list.get(i) > price)
        break;
      count++;
    }
    return count;
  }

  public int[] calculateSpans(int[] prices) {

    int[] spans = new int[prices.length];
    spans[0] = 1; // Span of first element is always 1

    Stack<Integer> indexStack = new Stack<>();

    // Push the index of first element
    indexStack.push(0);

    for (int i = 1; i < prices.length; i++) {
      while (!indexStack.isEmpty()
          && prices[indexStack.peek()] <= prices[i])
        indexStack.pop();

      // If index stack is empty, the price at index 'i'
      // is greater than all previous values
      if (indexStack.isEmpty())
        spans[i] = i + 1;
      else
        spans[i] = i - indexStack.peek();

      indexStack.push(i);
    }

    return spans;
  }
}

  private List<Integer> list = new ArrayList<>();
  private Stack<Integer> stack = new Stack<>();

  public void add(int value) {
    list.add(value);
  }

  public int next() {
    while(!stack.isEmpty()) {
      if (list.get(stack.peek()).equals(0)) {
        stack.pop();
      } else {
        break;
      }
    }
    if (stack.isEmpty()) {
      return list.size();
    } else {
      return list.indexOf(list.get(stack.peek())) + 1;
    }
  }

  private Stack<Integer> stack = new Stack<>();

    int count = stack.isEmpty() ? 1 : list.get(stack.peek()) - value + 1;
    list.add(count);
    stack.push(list.size()-1);

  public int next(int price) {

public class OnlineStockSpan {

  private List<Integer> list = new ArrayList<>();
  private Stack<Integer> stack = new Stack<>();

  public int[] calculateSpans(int[] prices) {
    for (int price : prices) {
      while (!stack.isEmpty() && prices[stack.peek()] < price) {
        list.add(stack.pop());
      }
      list.add(list.size());
      stack.push(list.size() - 1);
    }
    int[] result = new int[list.size()];
    for (int i = 0; i < list.size(); i++) {
      result[i] = i + 1;
    }
    for (int i = 0; i < list.size(); i++) {
      if (list.get(i) != i + 1) {
        result[list.get(i)] = list.get(i) - i;
      }
    }
    return result;
  }

  public int next(int price) {
    while (!stack.isEmpty() && prices[stack.peek()] < price) {
      stack.pop();
    }
    if (stack.isEmpty()) {
      list.add(0);
    } else {
      list.set(stack.peek(), list.get(stack.peek()) + 1);
    }
    return list.size();
  }

}

  @org.junit.Test

    }
    assertEquals(2, instance.next(4));
  }

      instance.next(i);

}
