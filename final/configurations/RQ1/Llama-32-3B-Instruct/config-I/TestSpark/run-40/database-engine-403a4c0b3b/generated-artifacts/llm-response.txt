    public void testCreateTable() throws DBAppException, IOException {
        DBApp db = new DBApp();
        db.createTable("testTable", "id", new Hashtable<String, String>() {{
            put("name", "String");
            put("age", "int");
        }}, null, null);
        assertEquals(1, db.getMyTables().size());
    }

    @Test
    public void testUpdateTable() throws DBAppException {
        DBApp db = new DBApp();
        db.createTable("testTable", "id", new Hashtable<String, String>() {{
            put("name", "String");
            put("age", "int");
        }}, null, null);
        db.updateTable("testTable", "1", new Hashtable<String, Object>() {{
            put("name", "John");
            put("age", 25);
        }});
        assertEquals(1, ((Selector)db.selectFromTable(new SQLTerm[] {"id"}, new String[]{"="})).next());
    }

    @Test
    public void testDeleteTable() throws DBAppException {
        DBApp db = new DBApp();
        db.createTable("testTable", "id", new Hashtable<String, String>() {{
            put("name", "String");
            put("age", "int");
        }}, null, null);
        db.deleteFromTable("testTable", new Hashtable<String, Object>());
        assertNull(db.selectFromTable(new SQLTerm[] {"id"}, new String[]{"="}).next());
    }

    @Test
    public void testSelect() throws DBAppException {
        DBApp db = new DBApp();

            put("name", "String");
            put("age", "int");
