import java.io.InputStream;

public class FileKeyValImpl implements Connection.KeyVal {
    private final String key;
    private final String filename;
    private InputStream inputStream;

    public static Builder<FileKeyVal> create(String key, String filename) {
        return new Builder(key, filename);
    }

    private FileKeyValImpl(String key, String filename) {
        this.key = Objects.requireNonNull(key, "key cannot be null");
        Validate(notEmptyParam(filename, "filename cannot be empty"));
        this.filename = filename;
    }

    @Override
    public InputStream getInputStream() {
        if (inputStream == null) {
            try (InputStream inputStream = new FileInputStream(filename)) {
                this.inputStream = inputStream;
            } catch (IOException e) {
                throw new RuntimeException(e);
            }
        }
        return inputStream;
    }

    @Override
    public String getKey() {
        return key;
    }

    @Override
    public String getValue() {
        return filename;
    }
}

public class Builder<T> implements Iterable<Builder<T>> {

    private Builder(T value) {

    public static <T> Builder<T> create(String key, String value) {
        return new Builder<>(key, value);
    }

    @Override
    public Iterator<Builder<T>> iterator() {
        return Collections.emptyIterator();
    }
}
