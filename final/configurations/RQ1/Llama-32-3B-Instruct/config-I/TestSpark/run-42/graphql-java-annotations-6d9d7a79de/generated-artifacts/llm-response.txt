        System.out.println(publicClass.publicMethod2());
    }

public class Directive {
    public String directiveName;

    public Directive(String directiveName) {
        this.directiveName = directiveName;
    }

    public List<GraphQLDirective> getDirectiveAnnotations() {
        return Arrays.asList(new GraphQLDirective(directiveName));
    }
}

public class GraphQLDirective {
    private String name;

    public GraphQLDirective(String name) {
        this.name = name;
    }

    @Override
    public String toString() {
        return "GraphQLDirective{" +
                "name='" + name + '\'' +
                '}';
    }
}

public class TestDirective {
    @Test
    public void testGetDirectiveAnnotations() {
        Directive directive = new Directive("testDirective");
        List<GraphQLDirective> annotations = directive.getDirectiveAnnotations();
        for (GraphQLDirective annotation : annotations) {
            System.out.println(annotation);
        }
    }
}
