import static org.junit.Assert.assertEquals;

public class DeltaTTest {

    @Test
    public void [estimate][beforeFirstCentury]Test() {
        LocalDate forDate = LocalDate.of(1800, 1, 1);
        assertEquals(8.83, DeltaT.estimate(forDate), 0.00001);
    }

    @Test
    public void [estimate][afterLastCentury]Test() {
        LocalDate forDate = LocalDate.of(1900, 1, 1);
        assertEquals(-2.79, DeltaT.estimate(forDate), 0.00001);
    }

    @Test
    public void [estimate][midFirstCentury]Test() {
        LocalDate forDate = LocalDate.of(1805, 6, 15);
        assertEquals(8.83 + 0.1603 * (18 - 1) / 12 - 0.0059285 * pow((18 - 1) / 12, 2), DeltaT.estimate(forDate), 0.00001);
    }

    @Test
    public void [estimate][lastFirstCentury]Test() {
        LocalDate forDate = LocalDate.of(1799, 6, 15);
        assertEquals(8.83 + 0.1603 * (-1) / 12 - 0.0059285 * pow(-1 / 12, 2), DeltaT.estimate(forDate), 0.00001);
    }

    @Test
    public void [decimalYear][earliestDate]Test() {
        LocalDate forDate = LocalDate.of(500, 1, 1);
        assertEquals(50.0, DeltaT.decimalYear(forDate), 0.00001);
    }

    @Test
    public void [estimate][yearInEarly20thCentury]Test() {
        LocalDate forDate = LocalDate.of(1950, 6, 15);
        assertEquals(21.20 + 0.84493 * (19 - 1) / 12 - 0.076100 * pow((19 - 1) / 12, 2) + 0.0020936 * pow((19 - 1) / 12, 3), DeltaT.estimate(forDate), 0.00001);
    }

}
