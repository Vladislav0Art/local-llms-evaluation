import org.junit.jupiter.api.Test;

public class CommentsConfigurationTest {
    @Test
    void testSetComment() throws Exception {
        CommentsConfiguration commentsConfiguration = new CommentsConfiguration();
        String[] line1 = {"comment", "line", "1"};
        String[] line2 = {"comment", "line", "2"};
        String path = "/test/path";
        
        commentsConfiguration.setComments(path, line1);
        commentsConfiguration.setComments(path, line2);

        assertEquals(Collections.singletonList(line1), commentsConfiguration.getComments().get(path));
    }

    @Test
    void testGetComment() throws Exception {
        CommentsConfiguration commentsConfiguration = new CommentsConfiguration();
        String[] line1 = {"comment", "line", "1"};
        String[] line2 = {"comment", "line", "2"};
        String path = "/test/path";
        
        commentsConfiguration.setComments(path, line1);
        commentsConfiguration.setComments(path, line2);

        assertEquals(Collections.singletonList(line2), commentsConfiguration.getComments().get(path));
    }

    @Test
    void testSetCommentEmpty() throws Exception {
        CommentsConfiguration commentsConfiguration = new CommentsConfiguration();
        String[] line1 = {"comment", "line", "1"};
        
        commentsConfiguration.setComments("test/path", line1);

        assertEquals(Collections.singletonList(line1), commentsConfiguration.getComments().get("test/path"));
    }

    @Test
    void testGetCommentEmpty() throws Exception {
        CommentsConfiguration commentsConfiguration = new CommentsConfiguration();

        assertEquals(Collections.emptyList(), commentsConfiguration.getComments());
    }
}

public class CommentsConfiguration {
    public Map<String, List<String>> commentsMap;

    public CommentsConfiguration() {
        this.commentsMap = new HashMap<>();
    }

    public void setComment(String path, String comment) {
        if (commentsMap.containsKey(path)) {
            ((List<String>) commentsMap.get(path)).add(comment);
        } else {
            List<String> list = new ArrayList<>();
            list.add(comment);
            commentsMap.put(path, list);
        }
    }

    public Map<String, List<String>> getComments() {
        return commentsMap;
    }
}

import org.junit.Test;
import static org.junit.Assert.*;

public class TestCommentsConfiguration {
    @Test
    public void testAddComment() {
        CommentsConfiguration comments = new CommentsConfiguration();
        List<String> path = new ArrayList<>();
        path.add("newComment");
        comments.setComments("path", path);
        assertEquals(1, ((List<String>) comments.getComments().get("path")).size());
        ((List<String>) comments.getComments().get("path")).add("anotherNewComment");
        assertEquals(2, ((List<String>) comments.getComments().get("path")).size());
    }

    @Test
    public void testGetMap() {
        CommentsConfiguration comments = new CommentsConfiguration();
        Map<String, List<String>> map = new HashMap<>();
        comments.setComments("map", map);
        assertTrue(comments.getComments().containsKey("map"));
    }

    @Test
    public void testAddCommentMultiplePath() {
        CommentsConfiguration comments = new CommentsConfiguration();
        String path1 = "newComment";
        String path2 = "anotherNewComment";
        List<String> path1Comments = new ArrayList<>();
        path1Comments.add(path1);
        List<String> path2Comments = new ArrayList<>();
        path2Comments.add(path2);
        Map<String, List<String>> commentsMap = new HashMap<>();
        commentsMap.put("path1", path1Comments);
        commentsMap.put("path2", path2Comments);
        comments.setComments(commentsMap);
        assertEquals(2, ((List<String>) comments.getComments().get("path1")).size());
        assertEquals(1, ((List<String>) comments.getComments().get("path2")).size());
    }

    @Test
    public void testGetList() {
        CommentsConfiguration comments = new CommentsConfiguration();
        List<String> list = new ArrayList<>();
        list.add("newComment");
        comments.setComments("list", list);
        assertEquals(1, ((List<String>) comments.getComments().get("list")).size());
    }
}

class CommentsConfiguration {
    private Map<String, Object> comments;

    public void setComments(String key, Object value) {
        if (comments == null) {
            comments = new HashMap<>();
        }
        comments.put(key, value);
    }

    public Object getComments() {
        return comments;
    }
}
