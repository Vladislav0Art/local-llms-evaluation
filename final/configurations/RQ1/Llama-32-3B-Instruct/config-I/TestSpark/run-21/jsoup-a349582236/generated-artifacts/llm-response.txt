```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Tag {
    private String name;
    private Element element;

    public Tag(String name, Element element) {
        this.name = name;
        this.element = element;
    }

    public static Tag getTag(String tagName, Element element) {
        return new Tag(tagName, element);
    }
}

class TestTag {
    private Document document;
    private Element element;

    public TestTag(Document document, String tagName) {
        this.document = document;
        this.element = document.getElementById(tagName);
    }

    public static void main(String[] args) {
        // Create a new HTML document
        Document document = Jsoup.create("<html><body><div id='tag1'>Hello World!</div></body></html>");

        // Get the tag with the id 'tag1'
        TestTag testTag = new TestTag(document, "tag1");

        // Print the tag's name and element
        System.out.println("Tag Name: " + testTag.name);
        System.out.println("Element: " + testTag.element.outerHtml());
    }
}
```

In this solution:

*   We define a `Tag` class with a constructor that takes a tag name and an HTML element.
*   We create a `TestTag` class that represents a tag with a specific id in the HTML document.
*   In the `main` method, we create a new HTML document, add a div element with id 'tag1', and then create a `TestTag` instance for this element.

This solution demonstrates how to work with tags in an HTML document using Java's Jsoup library.
