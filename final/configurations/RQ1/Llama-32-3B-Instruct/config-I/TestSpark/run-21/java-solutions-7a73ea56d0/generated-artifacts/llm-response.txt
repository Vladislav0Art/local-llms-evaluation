public class OnlineStockSpan {
  private int[] prices;
  private int index = -1;

  public int[] next(int[] prices) {
    this.prices = prices;
    return getSpans();
  }

  private int[] getSpans() {
    int[] result = new int[prices.length];
    for (int i = index + 1; i < prices.length; i++) {
      int count = 1;
      while (i - count >= 0 && prices[i] > prices[i - count]) {
        count++;
      }
      result[i] = count;
      index = i;
    }
    return result;
  }

  public static void main(String[] args) {
    OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
    int[] test1 = {0};
    System.out.println(java.util.Arrays.toString(onlineStockSpan.next(test1)));

    int[] test2 = {-10, -5, 0, 3, 20};
    System.out.println(java.util.Arrays.toString(onlineStockSpan.next(test2)));

    int[] test3 = {10, 7, 4, 9, 2};
    System.out.println(java.util.Arrays.toString(onlineStockSpan.next(test3)));

    int[] test4 = {};

      java.util.Arrays.toString(onlineStockSpan.next(test4));

      System.out.println(e.getMessage());

      onlineStockSpan.next(test5);

      System.out.println(e.getMessage());
    }
  }
}

public class OnlineStockSpan {
  public int[] next(int[] prices) {
    int[] result = new int[prices.length];
    int index = -1;
    for (int i = 0; i < prices.length; i++) {
      while (index >= 0 && prices[index] < prices[i]) {
        index--;
      }
      if (index == -1) {
        result[i] = 1;
      } else {
        result[i] = i - index;
      }
      index++;
    }
    return result;
  }

  public static void main(String[] args) {
    OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
