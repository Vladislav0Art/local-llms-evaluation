public class SolarPositioning {
    public static double decimalYear(LocalDate date) {
        return -20;
    }

    public static double estimate(LocalDate date) {
        if (date.isBefore(LocalDate.of(1800, 1, 1))) {
            return -20;
        } else if (date.isAfter(LocalDate.of(1999, 12, 31)) && date.isBefore(LocalDate.of(2000, 1, 1))) {
            return 63.86;
        } else if (date.isAfter(LocalDate.of(1945, 12, 31)) && date.isBefore(LocalDate.of(1946, 1, 1))) {
            return -2.79;
        } else if (date.isBefore(LocalDate.of(-1500, 1, 1))) {
            return -20;
        }
        throw new UnsupportedOperationException("Unsupported date");
    }
}
