/**
 * A registry for storing and retrieving Tag objects.
 */
public class TagRegistry {
    private static final Map<String, Tag> tags = new HashMap<>();

    /**
     * Registers a new Tag object with the given tag name.
     *
     * @param tag the Tag object to register
     */
    public static void register(Tag tag) {
        if (!tags.containsKey(tag.getTagName())) {
            tags.put(tag.getTagName(), tag);
        } else {
            throw new RuntimeException("Tag with name '" + tag.getTagName() + "' already exists.");
        }
    }

    /**
     * Retrieves a Tag object by its name.
     *
     * @param tagName the name of the Tag object to retrieve
     * @return the Tag object associated with the given name, or null if no such Tag exists
     */
    public static Tag getTag(String tagName) {
        return tags.get(tagName);
    }
}

/**
 * Represents a generic HTML tag.
 */
public class Tag {
    private String tagName;
    private boolean isBlock;

    /**
     * Constructs a new Tag object with the given tag name and block status.
     *
     * @param tagName  the name of the HTML tag
     * @param isBlock   whether the tag is a block element
     */
    public Tag(String tagName, boolean isBlock) {
        this.tagName = tagName;
        this.isBlock = isBlock;
    }

    // Getters and setters

    public String getTagName() {
        return tagName;
    }

    public void setTagName(String tagName) {
        this.tagName = tagName;
    }

    public boolean isBlock() {
        return isBlock;
    }

    public void setBlock(boolean block) {
        isBlock = block;
    }
}

/**
 * Represents a HTML img tag.
 */
public class ImgTag extends Tag {
    private String src;

    /**
     * Constructs a new ImgTag object with the given tag name, src attribute, and block status.
     *
     * @param tagName  the name of the HTML tag
     * @param src        the src attribute value
     * @param isBlock    whether the tag is a block element
     */
    public ImgTag(String src, boolean isBlock) {
        super(src, isBlock);
        this.src = src;
    }

    // Getters and setters

    public String getSrc() {
        return src;
    }

    public void setSrc(String src) {
        this.src = src;
    }
}

 * Represents a HTML link tag.

    private String href;

     * Constructs a new LinkTag object with the given tag name, href attribute, and block status.
