    public LineWrapper(String input) {
        this.input = trim(input);
    }

    public static String lineToString(String s) {
        return s.trim();
    }

    public static String trim(String s) {
        int start = 0;
        while (start < s.length() && Character.isWhitespace(s.charAt(start))) {
            start++;
        }
        if (start == s.length()) {
            return "";
        }
        int end = s.length() - 1;
        while (end >= 0 && Character.isWhitespace(s.charAt(end))) {
            end--;
        }
        if (end < 0) {
            return "";
        }
        return s.substring(start, end + 1);
    }

    @Override

        return lineToString(input);
