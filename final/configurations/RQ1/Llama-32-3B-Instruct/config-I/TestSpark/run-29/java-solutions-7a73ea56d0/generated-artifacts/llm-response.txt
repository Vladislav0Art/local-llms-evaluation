package leetcode.medium;

import org.junit.Test;
import static org.junit.Assert.*;
import java.util.Arrays;
import java.util.List;
import java.util.Stack;
public class OnlineStockSpanTest {

  private List<Integer> list = new ArrayList<>();
  private OnlineStockSpan stockSpan = new OnlineStockSpan();

  @Test
  public void [Next][ZeroPrice]Test() {
    assertEquals(1, stockSpan.next(0));
    assertTrue(stockSpan.list.contains(0));
  }

  @Test
  public void [Next][PositivePrice]Test() {
    assertEquals(1, stockSpan.next(100));
    assertTrue(stockSpan.list.contains(100));
  }

  @Test
  public void [Next][DecreasingSequence]Test() {
    list.clear();
    stockSpan.next(10);
    stockSpan.next(20);
    stockSpan.next(15);
    assertEquals(2, stockSpan.next(30));
    assertTrue(stockSpan.list.contains(30));
  }

  @Test
  public void [CalculateSpans][SingleElement]Test() {
    int[] prices = {100};
    int[] result = stockSpan.calculateSpans(prices);
    assertTrue(Arrays.equals(result, new int[]{1}));
  }

  @Test
  public void [CalculateSpans][MultipleElements]Test() {
    int[] prices = {10,20,15,30};
    int[] result = stockSpan.calculateSpans(prices);
    assertTrue(Arrays.equals(result, new int[]{2,3,4,5}));
  }

  @Test
  public void [CalculateSpans][IncreasingSequence]Test() {
    list.clear();
    stockSpan.next(10);
    stockSpan.next(20);
    stockSpan.next(30);
    stockSpan.next(40);
    assertEquals(new int[]{2,3,4,5}, stockSpan.calculateSpans(new int[]{10,20,30,40}));
  }

  @Test
  public void [CalculateSpans][DecreasingSequence]Test() {
    list.clear();
    stockSpan.next(10);
    stockSpan.next(20);
    stockSpan.next(15);
    assertEquals(new int[]{2,1}, stockSpan.calculateSpans(new int[]{30,20}));
  }

}
