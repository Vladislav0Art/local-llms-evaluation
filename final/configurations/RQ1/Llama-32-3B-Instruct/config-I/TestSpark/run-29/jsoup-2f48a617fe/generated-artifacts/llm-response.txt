package org.jsoup.nodes;

import org.junit.Test;
import org.mockito.Mockito;

public class CommentTest {

    @Test
    public void constructor_hasValidData() {
        Comment comment = new Comment("Hello World!");
        assertEquals("Hello World!", comment.getData());
    }

    @Test
    public void constructor_hasEmptyData() {
        Comment comment = new Comment("");
        assertEquals("", comment.getData());
    }

    @Test(expected = NullPointerException.class)
    public void isXmlDeclaration_throwsNullPointerException_onNullData() {
        Comment comment = new Comment(null);
        comment.isXmlDeclaration();
    }

    @Test
    public void isXmlDeclaration_returnsTrue_forXmlDeclarationData() {
        Comment comment = new Comment("!xml declaration");
        assertTrue(comment.isXmlDeclaration());
    }

    @Test
    public void isXmlDeclaration_returnsFalse_forNonXmlDeclarationData() {
        Comment comment = new Comment("Hello World!");
        assertFalse(comment.isXmlDeclaration());
    }

    @Test
    public void asXmlDeclaration_returnsNull_ifNotAnXmlDeclaration() {
        Comment comment = new Comment("!xml declaration");
        assertNull(comment.asXmlDeclaration());
    }

    @Test
    public void asXmlDeclaration_returnsNullIfDataIsInvalidXmlDeclarationData() {
        Comment comment = new Comment("Hello World!");
        assertNull(comment.asXmlDeclaration());
    }
}

class MockParser {
    private String parsedData;

    public void setParsedData(String data) {
        this.parsedData = data;
    }

    public String getParsedData() {
        return parsedData;
    }
}

public class XmlDeclarationTest {

    @Test
    public void clone_returnsSameInstance() {
        Comment comment = new Comment("Hello World!");
        Comment clonedComment = new Comment(comment.clone());
        assertEquals(comment, clonedComment);
    }

    @Test
    public void clone_returnsDifferentInstance_ifModified() {
        Comment comment = new Comment("Hello World!");
        Comment modifiedComment = new Comment(comment.getData() + " Modified");
        Comment clonedComment = new Comment(modifiedComment.clone());
        assertNotEquals(comment, clonedComment);
    }
}
