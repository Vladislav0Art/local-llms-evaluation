public class TextNodeTests {
    @Test
    public void testTextNode() {
        TextNode textNode = new TextNode("Hello, World!");
        assertEquals("Hello, World!", textNode.getWholeText());
    }

    @Test
    public void testTextNodeSplit() {
        TextNode textNode = new TextNode("Hello, World!");
        int splitOffset = 5;
        String tail = textNode.splitText(splitOffset);
        assertEquals("World!", tail);
        assertEquals(6, tail.length());
    }

    @Test
    public void testTextNodeGetWholeTextLength() {
        TextNode textNode1 = new TextNode("Hello, World!");
        int splitOffset1 = 5;
        String tail1 = textNode1.splitText(splitOffset1);
        assertEquals("World!", tail1);
        assertEquals(6, tail1.length());
        
        TextNode textNode2 = new TextNode("This is a test.");
        int splitOffset2 = 15;
        String tail2 = textNode2.splitText(splitOffset2);
        assertEquals("test.", tail2);
        assertEquals(4, tail2.length());
    }

    public static class TextNode {
        private String value;

        public TextNode(String value) {
            this.value = value;
        }

        public String getWholeText() {
            return value;
        }

        public String splitText(int offset) {
            int length = Math.min(offset, value.length());
            return value.substring(length);
        }

        public String getWholeTextLength() {
            return String.valueOf(value.length());
        }
    }
}
