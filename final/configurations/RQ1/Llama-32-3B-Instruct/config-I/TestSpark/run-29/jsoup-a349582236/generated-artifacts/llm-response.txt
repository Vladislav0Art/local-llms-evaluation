public class Tag {
    public String tagName;
    public boolean empty = false;
    public boolean isBlock = true;
    public boolean formatAsBlock = true;
    public boolean preserveWhitespace = false;
    public boolean formList = false;
    public boolean formSubmit = false;

    public Tag(String tagName) {
        this.tagName = tagName;
    }
}

public class TagValidator {

    private static Map<String, Tag> tags = new HashMap<>();

    public static void register(Tag tag) {
        tags.put(tag.tagName, tag);
    }

    public static void validate(String tagName) {
        Tag tag = tags.get(tagName);
        Validate.notNull(tag);

        if (tag.empty) {
            System.out.println("Tag '" + tagName + "' should not be empty.");
        }
    }

    public static boolean isBlock(String tagName) {
        return tags.get(tagName).isBlock;
    }

    public static void setInline(String tagName) {
        Tag tag = tags.get(tagName);
        if (tag != null && !tag.isBlock) {
            System.out.println("Tag '" + tagName + "' should be block.");
        }
    }

    public static boolean isFormatAsInline(String tagName) {
        return tags.get(tagName).formatAsBlock;
    }

    public static void setPreserveWhitespace(String tagName) {
        Tag tag = tags.get(tagName);
        if (tag != null && !tag.preserveWhitespace) {
            System.out.println("Tag '" + tagName + "' should preserve whitespace.");
        }
    }

    public static boolean isFormListed(String tagName) {
        return tags.get(tagName).formList;
    }

    public static void setFormSubmit(String tagName) {
        Tag tag = tags.get(tagName);
        if (tag != null && !tag.formSubmit) {
            System.out.println("Tag '" + tagName + "' should be form submit.");
        }
    }

    public static class Validate {
        public static void notNull(Object obj) {
            if (obj == null) {
                throw new NullPointerException();
            }
        }
    }

    public static void main(String[] args) {
        // Test cases
        register(new Tag("div"));
        validate("div");
        System.out.println(isBlock("div"));  // prints: true

        Tag tag = new Tag("p");
        tag.formatAsBlock = false;
        register(tag);
        setInline("p");

        tag.preserveWhitespace = true;
        setPreserveWhitespace("pre");

        tag.formList = true;
        setFormListed("button");

        System.out.println(isFormatAsInline("p"));  // prints: false
        System.out.println(setPreserveWhitespace("pre"));  // prints: Tag 'pre' should preserve whitespace.
    }
}

### Example Code (Python)

The following Python example demonstrates how to create and validate HTML tags.

```python
class Tag:
    def __init__(self, tag_name, empty=False, is_block=True, format_as_block=True, preserve_whitespace=False, form_list=False, form_submit=False):
        self.tag_name = tag_name
        self.empty = empty
        self.is_block = is_block
        self.format_as_block = format_as_block
        self.preserve_whitespace = preserve_whitespace
        self.form_list = form_list
        self.form_submit = form_submit

def validate_tag(tag_name, tags):
    """
    Validates a given HTML tag.

    Args:
        tag_name (str): The name of the HTML tag to be validated.
        tags (list): A list of Tag objects representing the HTML tags.

    Returns:
        bool: True if the tag is valid; False otherwise.
    """
    # Check if the tag exists in the given list
    for tag in tags:
        if tag.tag_name == tag_name:
            # Validate each property of the tag
            if tag.empty:
                print(f"Tag '{tag_name}' should not be empty.")
                return False

            if not tag.is_block and (not tag.format_as_block or not tag.preserve_whitespace):
                print(f"Tag '{tag_name}' should be a block element or format as a block with whitespace preservation.")
                return False

            if tag.form_list:
                print(f"Tag '{tag_name}' should not form a list.")
                return False

            if tag.form_submit:
                print(f"Tag '{tag_name}' should not submit the form.")
                return False

    # If no issues are found, the tag is valid
    return True

# Example usage
tags = [
    Tag("div"),
    Tag("p", empty=True),
    Tag("span")
]

validate_tag("div", tags)
print(validate_tag("p", tags))
```

### Explanation

The provided Python code defines a `Tag` class to represent HTML tags. Each tag has several properties, such as name, emptiness, block status, format-as-block status, whitespace preservation, list formation, and form submission.

The `validate_tag` function checks if the given tag exists in the specified list of tags and then validates each property of the tag against its expected values. If any validation fails, it prints an error message indicating what is wrong with the tag.

In the example usage section, we create a list of `Tag` objects representing different HTML elements (div, p, span). We then use the `validate_tag` function to validate these tags one by one and print any error messages that may occur during validation.
