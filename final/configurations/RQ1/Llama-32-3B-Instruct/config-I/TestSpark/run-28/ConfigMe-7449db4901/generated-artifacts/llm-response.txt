    public void setCommentTest_AddingLinesWithEmptyLine() {
        CommentsConfiguration config = new CommentsConfiguration();
        String path = "path1";
        String line1 = "\n";
        String line2 = "line2";
        List<String> lines = new ArrayList<>(Arrays.asList(line1, line2));
        config.setComment(path, line1, line2);
        assertEquals(Arrays.asList(line2), config.comments.get(path));
    }

    @Test
    public void setCommentTest_OverridingExistingLines() {
        CommentsConfiguration config = new CommentsConfiguration();
        String path = "path1";
        String line1 = "line1\n";

        CommentsConfiguration config = new CommentsConfiguration();

        assertTrue(map instanceof org.jetbrains.annotations.UnmodifiableView);
    }
}
