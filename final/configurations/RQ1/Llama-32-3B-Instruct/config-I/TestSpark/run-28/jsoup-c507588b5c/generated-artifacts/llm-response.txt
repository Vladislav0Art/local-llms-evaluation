public class TextNode {
    private String text;
    private Node parentNode;

    public TextNode(String rawText) {
        this.text = preprocessRawText(rawText);
        this.parentNode = null;
    }

    private String preprocessRawText(String rawText) {
        return rawText.replace("\n", " ").replace("\t", " ");
    }

    public String text() {
        return text;
    }

    public Node parentNode() {
        return parentNode;
    }

    public void setParentNode(Node parentNode) {
        this.parentNode = parentNode;
    }

    public Node clone() {
        return new TextNode(this.text);
    }

    public static TextNode createFromEncoded(String encodedText) {
        StringBuilder sb = new StringBuilder();
        for (int i=0; i<encodedText.length(); i++) {
            char c = encodedText.charAt(i);
            if (c == '<') {
                if (sb.length() > 0 && sb.charAt(sb.length()-1) != '>') {
                    sb.append('&');
                }
                sb.append(c);
            } else if (c == ' ') {
                sb.append(' ');
            } else if (c == '>') {
                sb.append('>');
            } else {
                sb.append(c);
            }
        }
        return new TextNode(sb.toString());
    }

    public void outerHtmlHead(Appendable accum, int index, Document.OutputSettings out) throws IOException {
        accum.append("<span style=\"display: inline;");
        accum.append(index > 0 ? "text-indent: -" + index + "px;" : "");
        accum.append("background-color: #f2f2f2; padding: 5px; border-radius: 3px;">");
        accum.append(text);
        accum.append("</span>");
    }

    public void outerHtmlTail(Appendable accum, int index, Document.OutputSettings out) throws IOException {
        if (index > 0) {
            accum.append("&#160;");
        }
    }

    public boolean isBlank() {
        return text.trim().isEmpty();
    }

    public static TextNode splitText(String rawText, int offset) {
        if (offset >= rawText.length()) {
            return null;
        }
        String before = rawText.substring(0, offset);
        String after = rawText.substring(offset);
        return new TextNode(before);
    }
}

public class TextNodeTest {

    public void testPreprocessRawText() {
        String rawText = "Hello World!";
        assertEquals("Hello World!", TextNode.preprocessRawText(rawText));
        assertEquals("\n   \t", TextNode.preprocessRawText("\n   \t"));
    }

    public void testIsBlank() {
        TextNode textNode1 = new TextNode("");
        TextNode textNode2 = new TextNode("\n   \t");
        assertTrue(TextNode.isBlank(textNode1));
        assertFalse(TextNode.isBlank(textNode2));
    }

    public void testSplitText() {
        String rawText = "Hello World!";
        int offset = 5;
        TextNode expectedBefore = new TextNode("Hello ");
        TextNode expectedAfter = new TextNode("World!");
        assertEquals(expectedBefore, TextNode.splitText(rawText, offset).clone());
        assertEquals(expectedAfter, TextNode.splitText(rawText, rawText.length()).clone());
    }

    public void testOuterHtmlHead() {
        String expectedOutput = "<span style=\"display: inline; text-indent: -0px; background-color: #f2f2f2; padding: 5px; border-radius: 3px;\">Hello World!</span>";
        TextNode textNode = new TextNode("Hello World!");
        Appendable accum = new StringBuilder();
        Document.OutputSettings out = new Document.OutputSettings(true, true);
        try {
            textNode.outerHtmlHead(accum, 0, out);
            assertEquals(expectedOutput, accum.toString());
        } catch (IOException e) {
            fail(e.getMessage());
        }
    }

    public void testOuterHtmlTail() {
        TextNode expectedOutput = "&#160;";
        String inputText = "Hello World!";
        Appendable accum = new StringBuilder();
        Document.OutputSettings out = new Document.OutputSettings(true, true);
        try {
            TextNode.text.outerHtmlTail(accum, inputText.length(), out);
        } catch (IOException e) {
            fail(e.getMessage());
        }
    }

    public void testClone() {
        String expectedOutput1 = "Hello World!";
        String expectedOutput2 = "";
        TextNode textNode1 = new TextNode("Hello World!");
        TextNode textNode2 = new TextNode("");
        assertEquals(expectedOutput1, textNode1.clone().text);
        assertEquals(expectedOutput2, textNode2.clone().text);
    }

    public void testCreateFromEncoded() {
        String encodedText = "<span>Hello World!</span>";
        String expectedOutput = "Hello World!";
        TextNode textNode = TextNode.createFromEncoded(encodedText);
        assertEquals(expectedOutput, textNode.text);
    }
}
