        stack = new int[0];
        top = 0;
    }

        }
        return result;
    }

    private int calculateSpan(int price) {
        int count = 1;
        while (!result.isEmpty() && result.peek().value <= price) {
            count += result.peek().count;
            result.remove();
        }
        return count;
    }

    public void push(int value, int count) {
        while (result.size() > 0 && result.peek().value >= value) {
            result.remove();
        }
        result.add(new Result(value, count));
    }

    private class Result {
        int value;
        int count;

        public Result(int value, int count) {
            this.value = value;
            this.count = count;
        }
    }

    public static void main(String[] args) {
        StockSpan stockSpan = new StockSpan();
        List<Integer> priceList = Arrays.asList(100, 80, 75, 70, 60);

        for (int price : priceList) {
            stockSpan.push(price, 1);
            while (!stockSpan.result.isEmpty() && stockSpan.next() == price) {
                int count = stockSpan.result.remove().count;
                stockSpan.push(price, count + 1);
            }
            System.out.println(stockSpan.next());
        }

        Result[] result = stockSpan.calculateSpans();
        for (Result r : result) {
            System.out.print("(" + r.value + ", " + r.count + ") ");
        }
    }

Problem Statement: 
Given an array of daily temperatures, calculate the number of consecutive days you would've had to look at to see each temperature, from left to right.

Example:
Input: [73, 74, 75, 71, 69, 72, 76, 73]
Output: [1, 1, 4, 2, 1, 1, 0, 0]

Code:

```java
import java.util.Stack;

public class StockSpan {
    private int[] result;
    private Stack<Integer> stack;
    private int index;

    public StockSpan() {
        stack = new Stack<>();
        result = new int[0];
        index = 0;
    }

    public void add(int price) {
        while (!stack.isEmpty() && price > stack.peek()) {
            pop();
        }
        addToResult(price);
    }

    private void pop() {
        int count = index - stack.pop();
        result = Arrays.copyOf(result, ++index);
        result[index - 1] += count;
    }

    private void addToResult(int price) {
        stack.push(price);
        result = Arrays.copyOf(result, ++index);
        result[index - 1] = 1;
    }

    public int next() {
        return result[result.length - 1];
    }

    public static void main(String[] args) {
        StockSpan stockSpan = new StockSpan();
        int[] priceArray = {73, 74, 75, 71, 69, 72, 76, 73};
        for (int i = 0; i < priceArray.length; i++) {
            stockSpan.add(priceArray[i]);
            System.out.println(stockSpan.next());
        }
    }
}
```

This code uses a stack to keep track of the temperatures. When a new temperature is added, it pops elements from the stack until the top element is less than or equal to the current temperature. The count of consecutive days for each temperature is calculated by popping all elements and adding them back with their respective counts.

Problem Statement: 
Given an array of daily temperatures, calculate the number of consecutive days you would've had to look at to see each temperature, from left to right.

Example:
Input: [73, 74, 75, 71, 69, 72, 76, 73]

```java
import java.util.Stack;

public class StockSpan {
    public int[] getDailyTemperatures(int[] temperatures) {
        Stack<Integer> stack = new Stack<>();
        int[] result = new int[temperatures.length];
        for (int i = 0; i < temperatures.length; i++) {
            while (!stack.isEmpty() && temperatures[i] > temperatures[stack.peek()]) {
                int top = stack.pop();
                int daysToNextGreaterTemp = i - stack.peek();
                result[top] = daysToNextGreaterTemp;
            }
            stack.push(i);
        }
        return result;
    }

    public static void main(String[] args) {
        StockSpan stockSpan = new StockSpan();
        int[] temperatures = {73, 74, 75, 71, 69, 72, 76, 73};
        int[] result = stockSpan.getDailyTemperatures(temperatures);
        for (int i = 0; i < result.length; i++) {
            System.out.print(result[i] + " ");
        }
    }
}
```

This code uses a stack to keep track of the indices of the temperatures. When a new temperature is encountered, it pops elements from the stack until the top element is less than or equal to the current temperature. The number of consecutive days for each temperature is calculated by subtracting the index of the previous greater temperature from the current index.
