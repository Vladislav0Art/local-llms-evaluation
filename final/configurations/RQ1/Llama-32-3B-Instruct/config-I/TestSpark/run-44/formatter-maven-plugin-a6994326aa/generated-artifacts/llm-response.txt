package net.revelc.code.formatter.css;

import org.hamcrest.MatcherAssert;
import org.hamcrest.Matchers;
import java.io.IOException;
import java.util.Map;

public class CssFormatter {

    public boolean init(Map<String, String> options, ConfigurationSource cfg) {
        // Check if the indent value is valid
        int indent = Integer.parseInt(options.get("indent"));
        MatcherAssert.assertThat(indent, Matchers.positive());

        // Check if the rgbAsHex option is set to true
        MatcherAssert.assertThat(options.get("rgbAsHex"), Matchers.is("true"));

        // Initialize the formatter with the given options and configuration source
        return true;
    }

    public String doFormat(String code, LineEnding ending) {
        // Split the code into lines
        String[] lines = code.split("\n");

        // Format each line according to the indent value
        StringBuilder formattedCode = new StringBuilder();
        for (String line : lines) {
            if (line.contains("{")) {
                // Check if the line contains a closing bracket
                MatcherAssert.assertThat(ending, Matchers.is(LineEnding.LF));
                formattedCode.append(line);
            } else {
                // Format the line according to the indent value
                formattedCode.append(getIndent(options.get("indent")));
                formattedCode.append(line).append("\n");
            }
        }

        return formattedCode.toString();
    }

    public boolean isInitialized() {
        // Check if the formatter has been initialized with the given options
        MatcherAssert.assertThat(init(Map.of(), Map.of()), Matchers.is(true));
        return true;
    }

    private String getIndent(String indent) {
        // Convert the indent value from a string to an integer and use it as the number of spaces
        int numSpaces = Integer.parseInt(indent);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < numSpaces; i++) {
            sb.append(" ");
        }
        return sb.toString();
    }
}

public class ConfigurationSource {

    public String get(String key) {
        // Return a default value if the key is not found
        MatcherAssert.assertThat(key, Matchers.is("indent"));
        return "4";
    }

    public String get(String key, Map<String, String> options) {
        // Return the value from the options map if it exists
        MatcherAssert.assertThat(options, Matchers.containsKey(key));
        return options.get(key);
    }
}

public class LineEnding {
    public static enum Type {
        LF,
        CRLF
    }

    private final Type type;

    public LineEnding(Type type) {
        this.type = type;
    }
}

    }
