```java
        System.out.println(kv.value);

        key = "key";

        System.out.println(kv1.value);
```

```java
public class UrlBuilderTest {

    @Test
    public void testUrlBuilder() {
        String url = "http://example.com/path?query=hello#fragment";
        URI uri = UrlBuilder.decodePart(URI.create(url).getPath())
                .replace('?', '!')
                .split('&')
                .stream()
                .map(part -> part.split('=')[0])
                .collect(Collectors.joining("&"));
        uri = new URL(uri + "?query=" + url.replace("?", " "))
            .toURI();
        uri = new URI(uri.getScheme(), uri.getHost(), uri.getPath() + "#fragment", uri.getQuery(),
                null, uri.getFragment());
        System.out.println(uri);
    }
}

public class UrlBuilder {

    public static String decodePart(String encoded) {
        return encoded;
    }

    public static String encodePart(String value) {
        return value;
    }

    public static URI createUrl(String url) {
        String[] parts = url.split("://");
        String scheme = parts[0];
        String hostPort = parts[1].split("/");
        String pathFragment = "";
        if (hostPort.length > 1) {
            String host = hostPort[0];
            String port = hostPort[1];
            pathFragment = host + ":" + port;
            hostPort = host;
        }
        String query = url.split("?")[1] != null ? "?" + url.split("\\?")[1] : "";
        return new URI(scheme, host, encodePart(pathFragment), encodePart(hostPort), query);
    }

    public static void main(String[] args) {
        System.out.println(UrlBuilder.decodePart("http://example.com/path"));
        System.out.println(UrlBuilder.encodePart("http://localhost:8080/"));
    }
}
```
