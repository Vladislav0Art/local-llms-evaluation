        Pattern pattern = Pattern.compile("^([a-zA-Z]+)-?\\d+(?:\\.\\d+)*-(\\d{3,4})-?(h[1-9]|h[A-Ha-hJ-j])-(.*)$");

        } else {
            return buildJob;
        }
