package ch.jalu.configme.configurationdata;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import java.util.Arrays;
import java.util.List;
import java.util.Map;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertFalse;
import static org.junit.Assert.assertTrue;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class CommentsConfigurationTest {

    @Mock
    private Map<String, List<String>> commentsMap;

    @Test
    public void newConstructor_createsNewEmptyCommentsMap() {
        CommentsConfiguration comments = new CommentsConfiguration();
        assertEquals(0, comments.getComments().size());
    }

    @Test
    public void newConstructor_passesCommentsMap() {
        Map<String, List<String>> comments = Collections.singletonMap("test", Arrays.asList("line1"));
        CommentsConfiguration commentsConfig = new CommentsConfiguration(comments);
        assertEquals(comments, commentsConfig.getComments());
    }

    @Test
    public void setComment_setsNewLinesForExistingPath() {
        CommentsConfiguration comments = new CommentsConfiguration();
        Map<String, List<String>> commentsMapCopy = new HashMap<>(comments.getComments());
        when(commentsMap).thenReturn(commentsMapCopy);
        comments.setComment("test", "line1", "line2");
        assertEquals(Arrays.asList("line1", "line2"), comments.getComments().get("test"));
    }

    @Test
    public void setComment_setsEmptyLineWithoutCommentMarker() {
        CommentsConfiguration comments = new CommentsConfiguration();
        Map<String, List<String>> commentsMapCopy = new HashMap<>(comments.getComments());
        when(commentsMap).thenReturn(commentsMapCopy);
        comments.setComment("test", "\n");
        assertEquals(Collections.singletonList("\n"), comments.getComments().get("test"));
    }

    @Test
    public void getAllComments_returnsUnmodifiableView() {
        CommentsConfiguration comments = new CommentsConfiguration();
        Map<String, List<String>> allComments = comments.getAllComments();
        assertTrue(allComments instanceof java.util.Map);
        assertFalse(allComments instanceof java.util.List);
    }
}
