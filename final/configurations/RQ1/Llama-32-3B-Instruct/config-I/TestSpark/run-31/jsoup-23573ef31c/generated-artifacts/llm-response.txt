```java
// OutputSettings.java
public enum OutputSettings {
    DEFAULT_OUTPUT_SETTINGS(4, 10, false),

    private final int maxPaddingWidth;

// Website.java
import java.util.Scanner;

public class Website {
    private final String name;
    private QuirksMode quirksMode;
    private OutputSettings outputSettings;
    private Scanner scanner;

    public Website(String name) {
        this.name = name;
        this.quirksMode = QuirksMode.NO_QUIRKS;
        this.outputSettings = OutputSettings.DEFAULT_OUTPUT_SETTINGS;
        this.scanner = new Scanner(System.in);
    }

    public void setQuirksMode(QuirksMode mode) {
        this.quirksMode = mode;
    }

    public void setOutputSettings(OutputSettings settings) {
        this.outputSettings = settings;
    }

    public void run() {
        System.out.println("Welcome to " + name + "!");

        while (true) {
            System.out.println("\nCurrent Quirks Mode: " + getQuirksMode());
            System.out.println("Current Output Settings: " + outputSettings);
            System.out.print("\nEnter 'q' to quit or enter a number to change quirks mode: ");
            String input = scanner.nextLine();

            if (input.equalsIgnoreCase("q")) {
                break;
            } else if (input.matches("\\d+")) {
                try {
                    int choice = Integer.parseInt(input);
                    switch (choice) {
                        case 1:
                            setQuirksMode(QuirksMode.QUIRKS);
                            break;
                        case 2:
                            setQuirksMode(QuirksMode.LIMITED_QUIRKS);
                            break;
                        default:
                            System.out.println("Invalid choice. Please try again.");
                    }
                } catch (NumberFormatException e) {
                    System.out.println("Invalid input. Please enter a number or 'q' to quit.");
                }
            } else {
                System.out.println("Invalid input. Please enter a number or 'q' to quit.");
            }

            System.out.print("\nEnter 'o' to change output settings or enter a letter to change quirks mode: ");
            String outputInput = scanner.nextLine();

            if (outputInput.equalsIgnoreCase("o")) {
                setOutputSettings(getNextOutputSetting());
            } else if (outputInput.matches("[a-z]")) {
                char choice = outputInput.charAt(0);
                switch (choice) {
                    case 'q':
                        setQuirksMode(QuirksMode.QUIRKS);
                        break;
                    case 'p':
                        setQuirksMode(QuirksMode.PRETTY_PRINT_QUIRKS);
                        break;
                    default:
                        System.out.println("Invalid choice. Please try again.");
                }
            } else {
                System.out.println("Invalid input. Please enter 'o' to change output settings or a letter to change quirks mode.");
            }
        }

        scanner.close();
    }

    private OutputSettings getNextOutputSetting() {
        System.out.print("\nEnter 1 for default settings, 2 for pretty print settings: ");
        String input = scanner.nextLine();

        try {
            int choice = Integer.parseInt(input);
            switch (choice) {
                case 1:
                    return OutputSettings.DEFAULT_OUTPUT_SETTINGS;
                case 2:
                    return OutputSettings.PRETTY_PRINT_OUTPUT_SETTINGS;
                default:
                    System.out.println("Invalid choice. Defaulting to default settings.");
                    return OutputSettings.DEFAULT_OUTPUT_SETTINGS;
            }
        } catch (NumberFormatException e) {
            System.out.println("Invalid input. Defaulting to default settings.");
            return OutputSettings.DEFAULT_OUTPUT_SETTINGS;
        }
    }

    public QuirksMode getQuirksMode() {
        return quirksMode;
    }

    @Override
    public String toString() {
        return "Website{" +
                "name='" + name + '\'' +
                ", quirksMode=" + quirksMode +
                ", outputSettings=" + outputSettings +
                '}';
    }
}

// Main.java
public class Main {

    public static void main(String[] args) {
        Website website = new Website("Example Website");
        website.run();
    }
}
```

This code defines a `Website` class that handles user input to change the quirks mode and output settings. The `run()` method contains a loop that continuously prompts the user for input until they choose to quit.
