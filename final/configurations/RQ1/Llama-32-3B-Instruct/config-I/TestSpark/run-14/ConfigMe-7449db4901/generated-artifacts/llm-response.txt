public class ConfigurationDataTestBase extends TestCase {

    public CommentsConfiguration createClassUnderTest() {
        return new CommentsConfiguration();
    }

    public Map<String, List<String>> getCommentsForTestClassUnderTest() {
        Map<String, List<String>> comments = new HashMap<>();
        comments.put("path1", Arrays.asList("line1"));
        return comments;
    }
}

public class CommentsConfiguration extends Configuration {

    private Map<String, List<String>> comments;

    public CommentsConfiguration(Map<String, List<String>> comments) {
        this.comments = comments;
    }

    public void setComment(String path, String value) {
        if (path == null || value == null) {
            throw new NullPointerException();
        }
        if (!comments.containsKey(path)) {
            comments.put(path, new ArrayList<>());
        }
        comments.get(path).add(value);
    }

    public Map<String, List<String>> getComments() {
        return comments;
    }

    @Override
    public String toString() {
        return super.toString() + " - " + comments;
    }
}
