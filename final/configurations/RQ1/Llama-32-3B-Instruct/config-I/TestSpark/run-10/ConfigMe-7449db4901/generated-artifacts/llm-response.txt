```java
public class CommentsConfigurationTest {

    private CommentsConfiguration commentsConfiguration;
    private Map<String, List<String>> comments;

    @Before
    public void setup() {
        comments = new HashMap<>();
        commentsConfiguration = new CommentsConfiguration(comments);
    }

    }

    }
```

```java
public class CommentConfiguration {
    private Map<String, List<String>> comments;

    public CommentConfiguration() {
        this.comments = new HashMap<>();
    }

    public void addComment(String key, String value) {
        if (!comments.containsKey(key)) {
            comments.put(key, new ArrayList<>());
        }
        comments.get(key).add(value);
    }

    @Override
    public Map<String, List<String>> getAllComments() {
        return Collections.unmodifiableMap(comments);
    }
}

public class CommentTest {

    @Before
    public void setup() {
        this.commentsConfiguration = new CommentConfiguration();
    }

    @Test(expected = NullPointerException.class)
    public void testGetAllCommentsReturnsNullWhenNoCommentsExist() {
        this.commentsConfiguration.getAllComments().size() > 0;
    }

    @Test
    public void testGetAllCommentsReturnsEmptyMap() {
        Map<String, List<String>> map = new HashMap<>();
        commentsConfiguration.getAllComments().equals(map);
    }

    @Test
    public void testAddCommentAndGetAllComments() {
        String key = "key";
        String value = "value";
        this.commentsConfiguration.addComment(key, value);
        assertNotEquals(Collections.emptyMap(), this.commentsConfiguration.getAllComments());
        assertEquals(1, ((List<String>) this.commentsConfiguration.getAllComments().get(key)).size());
    }
}
```
