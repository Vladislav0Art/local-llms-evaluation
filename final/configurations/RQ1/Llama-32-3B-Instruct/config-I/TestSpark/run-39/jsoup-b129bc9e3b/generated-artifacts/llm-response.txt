import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.net.HttpURLConnection;
import java.net.URL;

public class ConnectionRequestBuilder {

    private URL url;

    public ConnectionRequestBuilder() {}

    public void buildRequest(String method, String path) throws IOException {
        // Set the URL
        this.url = new URL("http://" + "localhost" + ":" + 8080 + path);

        // Create a connection to the URL
        HttpURLConnection connection = (HttpURLConnection) url.openConnection();
        connection.setRequestMethod(method);
        connection.setRequestProperty("Content-Type", "application/json");
    }

    public void buildRequest(String method, String path, String param) throws IOException {
        // Set the URL
        this.url = new URL("http://" + "localhost" + ":" + 8080 + path);

        // Create a connection to the URL
        HttpURLConnection connection = (HttpURLConnection) url.openConnection();
        connection.setRequestMethod(method);
        connection.setRequestProperty("Content-Type", "application/json");
        connection.setDoOutput(true);

        // Set the parameter in the request body
        String requestBody = "{\"" + param + "\": true}";
        connection.getOutputStream().write(requestBody.getBytes());
    }

    public void buildRequest(String method, String path, String[] params) throws IOException {
        // Set the URL
        this.url = new URL("http://" + "localhost" + ":" + 8080 + path);

        // Create a connection to the URL
        HttpURLConnection connection = (HttpURLConnection) url.openConnection();
        connection.setRequestMethod(method);
        connection.setRequestProperty("Content-Type", "application/json");
        connection.setDoOutput(true);

        // Set the parameter in the request body
        StringBuilder requestBodyBuilder = new StringBuilder("{\"");
        for (String param : params) {
            requestBodyBuilder.append("\"").append(param).append("\": true,\"");
        }
        requestBodyBuilder.deleteCharAt(requestBodyBuilder.length() - 1);
        String requestBody = requestBodyBuilder.toString();
        connection.getOutputStream().write(requestBody.getBytes());
    }

    public void buildRequest(String method, String path, int param) throws IOException {
        // Set the URL
        this.url = new URL("http://" + "localhost" + ":" + 8080 + path);

        // Create a connection to the URL
        HttpURLConnection connection = (HttpURLConnection) url.openConnection();
        connection.setRequestMethod(method);
        connection.setRequestProperty("Content-Type", "application/json");
        connection.setDoOutput(true);

        // Set the parameter in the request body
        String requestBody = "{\"" + param + "\": true}";
        connection.getOutputStream().write(requestBody.getBytes());
    }

}
