public class CommentNodeTest {

    private static final String XML_DECLARATION = "!DOCTYPE html PUBLIC \"-//W3C//DTD HTML 4.01 Transitional//EN\" \"http://www.w3.org/TR/html4/loose.dtd\">";
    private static final String XML_DECLARATION_WITH_SPACES = " !DOCTYPE html PUBLIC \"-//W3C//DTD HTML 4.01 Transitional//EN\" \"http://www.w3.org/TR/html4/loose.dtd\" >";

    public void testConstructor() {
        Comment comment = new Comment(XML_DECLARATION);
        assert comment.getData().equals(XML_DECLARATION);
    }

    public void testGetNodeName() {
        Comment comment = new Comment("");
        assertEquals("#comment", comment.nodeName());
    }

    public void testGetData() {
        Comment comment = new Comment(XML_DECLARATION);
        assertEquals(XML_DECLARATION, comment.getData());
    }

    public void testGetDataEmptyString() {
        Comment comment = new Comment("");
        assertNull(comment.getData());
    }

    public void testSetData() {
        Comment comment = new Comment("");
        comment.setData(XML_DECLARATION);
        assertEquals(XML_DECLARATION, comment.getData());
    }

    public void testOuterHtmlHead() throws IOException {
        Comment comment = new Comment(XML_DECLARATION);
        Appendable accum = mock(Appendable.class);
        Document.OutputSettings out = mock(Document.OutputSettings.class);
        when(out.prettyPrint()).thenReturn(true);
        when(out.outline()).thenReturn(false);
        when(comment.parentNode()).thenReturn(new Element("#root"));
        int depth = 0;
        comment.outerHtmlHead(accum, depth, out);
        verify(accum).append(anyString());
    }

    public void testOuterHtmlTail() throws IOException {
        Comment comment = new Comment("");
        Appendable accum = mock(Appendable.class);
        Document.OutputSettings out = mock(Document.OutputSettings.class);
        int depth = 0;
        comment.outerHtmlTail(accum, depth, out);
        verify(accum).append(anyString());
    }

    public void testToString() throws IOException {
        Comment comment = new Comment(XML_DECLARATION);
        assertEquals(comment.outerHtml(), comment.toString());
    }

    public void testClone() {
        Comment comment = new Comment(XML_DECLARATION);
        Comment clonedComment = comment.clone();
        assertNotNull(clonedComment);
        assertEquals(comment, clonedComment);
    }

    public void testIsXmlDeclaration() {
        Comment comment = new Comment(XML_DECLARATION);
        assertTrue(comment.isXmlDeclaration());
    }

    public void testIsXmlDeclarationEmptyString() {
        Comment comment = new Comment("");
        assertFalse(comment.isXmlDeclaration());
    }

    public void testAsXmlDeclaration() throws IOException {
        Comment comment = new Comment(XML_DECLARATION);
        XmlDeclaration xmlDeclaration = comment.asXmlDeclaration();
        assertNull(xmlDeclaration);
    }

    public void testAsXmlDeclarationXmlDeclaration() throws IOException {
        Element element = mock(Element.class);
        when(element.tagName()).thenReturn("html");
        String data = XML_DECLARATION;
        XmlDeclaration xmlDeclaration = new XmlDeclaration(data, true);
        Comment comment = new Comment(data);
        XmlDeclaration parsedDeclaration = comment.asXmlDeclaration();
        assertEquals(xmlDeclaration, parsedDeclaration);
    }

    public void testAsXmlDeclarationEmptyString() throws IOException {
        Element element = mock(Element.class);
        when(element.tagName()).thenReturn("html");
        String data = "";
        XmlDeclaration xmlDeclaration = new XmlDeclaration(data, false);
        Comment comment = new Comment(data);
        XmlDeclaration parsedDeclaration = comment.asXmlDeclaration();
        assertNull(parsedDeclaration);
    }
}
