        this.month = Integer.toString(monthNumber);
        return (year % 4 + year / 4 - 2 * (year % 100) + year / 400) * 29.53058868;
    }

    public int getMonthAsInt(String monthStr) {
        if (monthStr.equals("12")) {
            return 11; // December
        } else {
            return Integer.parseInt(monthStr);
        }
    }

    @Override
    public String toString() {
        return "SolarPosition{" +
                "year=" + year +
                ", month=" + month +
