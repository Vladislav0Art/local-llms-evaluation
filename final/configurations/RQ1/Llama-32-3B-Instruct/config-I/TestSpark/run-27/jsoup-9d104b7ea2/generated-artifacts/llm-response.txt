### Text Preprocessing Example
```java
import java.util.ArrayList;
import java.util.List;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

public class TextPreprocessing {

    public static List<String> preprocessText(String input) {
        // Convert to lowercase
        String lowerCaseInput = input.toLowerCase();

        // Remove non-alphanumeric characters and punctuation
        String cleanedInput = cleanInput(lowerCaseInput);

        // Tokenize the text into words
        String[] tokens = tokenize(cleanedInput);

        // Filter out empty tokens (e.g., from multiple consecutive whitespace)
        List<String> filteredTokens = new ArrayList<>();
        for (String token : tokens) {
            if (!token.isEmpty()) {
                filteredTokens.add(token);
            }
        }

        return filteredTokens;
    }

    private static String cleanInput(String input) {
        Pattern pattern = Pattern.compile("[^a-zA-Z0-9\\s]");
        Matcher matcher = pattern.matcher(input);
        return matcher.replaceAll("");
    }

    private static String[] tokenize(String input) {
        String[] words = input.split("\\s+");
        return words;
    }

    public static void main(String[] args) {
        String originalText = "Hello, World! This is a test sentence.";
        List<String> preprocessedTokens = preprocessText(originalText);
        System.out.println("Preprocessed Tokens:");
        for (String token : preprocessedTokens) {
            System.out.println(token);
        }
    }
}
```
### Explanation

1.  **Convert to Lowercase**: We start by converting the input text to lowercase using the `toLowerCase()` method.
2.  **Remove Non-Alphanumeric Characters and Punctuation**: Next, we use a regular expression pattern (`[^a-zA-Z0-9\\s]`) to match and replace any non-alphanumeric characters (excluding whitespace) with an empty string, effectively removing them from the text.
3.  **Tokenize the Text into Words**: We then split the cleaned input text into individual words using the `split()` method with a regular expression pattern (`\\s+`).
4.  **Filter Out Empty Tokens**: To remove any duplicate whitespace that may have resulted in empty tokens, we create an `ArrayList` to store unique non-empty tokens.

import java.util.ArrayList;
import java.util.List;

public class TextPreprocessing {

    public static List<String> preprocessText(String input) {
        // Convert to lowercase
        String lowerCaseInput = input.toLowerCase();

        // Remove non-alphanumeric characters and punctuation
        String cleanedInput = cleanInput(lowerCaseInput);

        // Tokenize the text into words
        String[] tokens = tokenize(cleanedInput);

        // Stem each token using Porter Stemmer
        List<String> stemmedTokens = stemTokens(tokens);

        return stemmedTokens;
    }

    private static String cleanInput(String input) {
        StringBuilder cleanedInput = new StringBuilder();
        for (char c : input.toCharArray()) {
            if (Character.isLetterOrDigit(c)) {
                cleanedInput.append(c);
            }
        }
        return cleanedInput.toString();
    }

    private static String[] tokenize(String input) {
        String[] words = input.split("\\s+");
        return words;
    }

    private static List<String> stemTokens(String[] tokens) {
        List<String> stemmedTokens = new ArrayList<>();
        for (String token : tokens) {
            Stemmer porterStemmer = new PorterStemmer();
            stemmedTokens.add(porterStemmer.stem(token));
        }
        return stemmedTokens;
    }

    private static class PorterStemmer {

        public String stem(String word) {
            if (word.length() < 3) {
                return word;
            }

            if (word.endsWith("ing")) {
                return stemWord(word.substring(0, word.length() - 3));
            } else if (word.endsWith("ed")) {
                return stemWord(word.substring(0, word.length() - 2));
            }

            return word.substring(0, Math.max(word.lastIndexOf("s"), word.lastIndexOf("x")));
        }

        private String stemWord(String word) {
            if (word.length() < 2) {
                return word;
            }

            if (word.endsWith("le")) {
                return stemWord(word.substring(0, word.length() - 2));
            } else if (word.endsWith("e") && !Character.isLetter(word.charAt(word.length() - 1))) {
                return stemWord(word.substring(0, word.length() - 1));
            }

            return word;
        }
    }

    public static void main(String[] args) {
        String originalText = "Helloing World! This is a test sentence.";
        List<String> preprocessedTokens = preprocessText(originalText);
        System.out.println("Preprocessed Tokens:");
        for (String token : preprocessedTokens) {
            System.out.println(token);
        }
    }
}
