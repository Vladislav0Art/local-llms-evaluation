        boolean result = Config.class.isAssignableFrom(el.getClass()) && config.preserveRelativeLinks;
        org.junit.Assert.assertTrue(result);
    }

    @org.junit.Test
    public void testIsSafeAttributeEnforced() {
        Element el = new Element();
        Attribute attr = new Attribute("href", "#anchor");
        boolean result = "enforced".equals(Config.class.getName()) && config.preserveRelativeLinks;
        org.junit.Assert.assertTrue(result);
    }
}

public class Config {

    private boolean preserveRelativeLinks;

    public void setPreserveRelativeLinks(boolean preserveRelativeLinks) {
        this.preserveRelativeLinks = preserveRelativeLinks;

        if (tagName.equals("enforced")) {
            return config.preserveRelativeLinks;

        return Config.class.isAssignableFrom(el.getClass());
