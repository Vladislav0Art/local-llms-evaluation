import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.util.Map;

public class Connection {
    private HttpClient client = HttpClient.newHttpClient();

    public static Connection newRequest(String uri, String method, Map<String, String> body) throws IOException, InterruptedException {
        // Create a new HttpRequest
        HttpRequest.Builder requestBuilder = HttpRequest.newBuilder();
        requestBuilder.uri(URI.create(uri));
        requestBuilder.method(method);
        
        if (body != null) {
            // Convert the map to a JSON string
            String json = "{";
            for (Map.Entry<String, String> entry : body.entrySet()) {
                json += "\"" + entry.getKey() + "\":\"" + entry.getValue() + "\",";
            }
            json += "}";
            
            // Set the content type and body of the request
            requestBuilder.header("Content-Type", "application/json");
            requestBuilder.body(json.getBytes());
        }

        HttpRequest request = requestBuilder.build();

        // Execute the request and get a HttpResponse object
        return new Connection().executeRequest(request);
    }

    private HttpResponse<String> executeRequest(HttpRequest request) throws IOException, InterruptedException {
        HttpResponse<String> response = client.send(request, HttpClient.BodyHandlers.ofString());

        // Handle the response as needed (e.g., print its status code)
        System.out.println("Status Code: " + response.statusCode());
        return response;
    }
}
