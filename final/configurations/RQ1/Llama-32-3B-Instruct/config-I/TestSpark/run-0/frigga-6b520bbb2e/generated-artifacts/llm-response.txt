public class AppVersionTest {

    @org.junit.Test
    public void testParseName() {
        String amiName = "myApp/1.0.0-123456";
        AppVersion appVersion = AppVersion.parseName(amiName);
        assertEquals("myApp", appVersion.getPackageName());
        assertEquals("1.0.0-123456", appVersion.getVersion());
    }

    @org.junit.Test
    public void testSetAndGetMethods() {
        String packageName = "myApp";
        String version = "1.0.0-123456";
        AppVersion appVersion = new AppVersion();
        appVersion.setPackageName(packageName);
        appVersion.setVersion(version);
        assertEquals(packageName, appVersion.getPackageName());
        assertEquals(version, appVersion.getVersion());
    }

    @org.junit.Test
    public void testAllMethods() {
        String packageName = "myApp";
        String version = "1.0.0-123456";
        AppVersion appVersion = new AppVersion();
        appVersion.setPackageName(packageName);
        appVersion.setVersion(version);
        appVersion.setBuildJobName("myBuild");
        appVersion.setBuildNumber("12345");
        appVersion.setCommit("abc123");
        appVersion.setChangelist("def456");

        assertEquals(packageName, appVersion.getPackageName());
        assertEquals(version, appVersion.getVersion());
        assertEquals("myBuild", appVersion.getBuildJobName());
        assertEquals("12345", appVersion.getBuildNumber());
        assertEquals("abc123", appVersion.getCommit());
        assertEquals("def456", appVersion.getChangelist());
