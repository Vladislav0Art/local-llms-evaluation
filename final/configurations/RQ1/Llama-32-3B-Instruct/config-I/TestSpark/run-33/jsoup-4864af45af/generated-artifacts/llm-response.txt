```java
// Item.java
public class Item {
    private final String title;
    private final String link;
    private final String description;

    public Item(String title, String link, String description) {
        this.title = title;
        this.link = link;
        this.description = description;
    }

    public String getTitle() {
        return title;
    }

    public String getLink() {
        return link;
    }

    public String getDescription() {
        return description;
    }
}

// Feed.java
public class Feed implements FeedInterface {
    private final String name;
    private final List<Item> items;

    public Feed(String name, List<Item> items) {
        this.name = name;
        this.items = items;
    }

    @Override
    public String getName() {
        return name;
    }

    @Override
    public List<Item> getItems() {
        return items;
    }
}

// RSSReader.java
public class RSSReader implements ReaderInterface {
    private final Feed feed;

    public RSSReader(Feed feed) {
        this.feed = feed;
    }

    @Override
    public void readFeed() {
        for (Item item : feed.getItems()) {
            System.out.println("Title: " + item.getTitle());
            System.out.println("Link: " + item.getLink());
            System.out.println("Description: " + item.getDescription());
            System.out.println();
        }
    }
}

// ReaderInterface.java
public interface ReaderInterface {
    void readFeed();
}

// FeedInterface.java
public interface FeedInterface {
    String getName();
    List<Item> getItems();
}
```
