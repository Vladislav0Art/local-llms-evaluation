    public static List<String> getTopComments(Map<String, @UnmodifiableView List<String>> allComments, int num) {

        List<String> topComments = new ArrayList<>();

    public static Map<String, Integer> getCommentCount(Map<String, @UnmodifiableView List<String>> allComments) {
        Map<String, Integer> commentCounts = new HashMap<>();
        for (Map.Entry<String, @UnmodifiableView List<String>> entry : allComments.entrySet()) {

        return allComments.containsKey(commentId);
    }

    public static void printAllComments(Map<String, @UnmodifiableView List<String>> allComments) {
        for (Map.Entry<String, @UnmodifiableView List<String>> entry : allComments.entrySet()) {
            CommentUtil.printCommentInfo(entry.getKey(), entry.getValue().get(0), entry.getValue().size());
        }
    }

    public static int getAverageCommentRating(Map<String, @UnmodifiableView List<String>> allComments) {
        double totalRating = 0;
        for (Map.Entry<String, @UnmodifiableView List<String>> entry : allComments.entrySet()) {
            if (!entry.getValue().isEmpty()) {
                double averageRating = 0;
                for (String comment : entry.getValue()) {
                    averageRating += calculateCommentRating(comment);
                }
                totalRating += averageRating / entry.getValue().size();
            }
        }
        return (int) totalRating;
    }

    private static double calculateCommentRating(String comment) {
        // Simple rating calculation: count of words * 0.1
        int words = 0;
        for (String word : comment.split(" ")) {
            if (!word.isEmpty()) {
                words++;
            }
        }
        return words * 0.1;
    }

    public static List<String> getMostActiveUsers(Map<String, @UnmodifiableView List<String>> allComments) {
        Map<String, Integer> userActivity = new HashMap<>();
        for (Map.Entry<String, @UnmodifiableView List<String>> entry : allComments.entrySet()) {
            for (String username : entry.getValue()) {
                userActivity.put(username, userActivity.getOrDefault(username, 0) + 1);
            }
        }
        List<Map.Entry<String, Integer>> sortedUsers = new ArrayList<>(userActivity.entrySet());
        sortedUsers.sort((e1, e2) -> e2.getValue() - e1.getValue());

        List<String> mostActiveUsers = new ArrayList<>();
        for (int i = 0; i < Math.min(10, sortedUsers.size()); i++) {
            mostActiveUsers.add(sortedUsers.get(i).getKey());
        }
        return mostActiveUsers;
    }
