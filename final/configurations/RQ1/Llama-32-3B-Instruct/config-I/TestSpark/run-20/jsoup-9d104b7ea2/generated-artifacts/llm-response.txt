public class TextNode {
    private String value;

    public TextNode(String value) {
        this.value = value;
    }

    public String getWholeText() {
        return value;
    }

    public boolean isBlank() {
        return StringUtil.stripLeadingWhitespace(value).isEmpty();
    }

    public void splitText(int offset) {
        int index = value.indexOf('\n', offset);
        if (index == -1) {
            index = value.length();
        }
        String textBefore = value.substring(0, index);
        String textAfter = value.substring(index + 1);
        this.value = textBefore;
    }

    public void outerHtmlHead(Appendable accum, int offset, Document.OutputSettings out) throws IOException {
        StringBuilder sb = new StringBuilder();
        sb.append("<div>");
        sb.append(value);
        sb.append("</div>");
        out.prettyPrint(sb.toString());
    }

    public void outerHtmlTail(Appendable accum, int offset, Document.OutputSettings out) {
    }

    @Override
    public String toString() {
        return "<div>" + value + "</div>";
    }
}

public class StringUtil {

    public static String stripLeadingWhitespace(String text) {
        while (text.startsWith(" ")) {
            text = text.substring(1);
        }
        return text;
    }

    public static String normaliseWhitespace(String text) {
        if (!text.contains("\n")) {
            return text;
        } else {
            StringBuilder sb = new StringBuilder();
            int index = 0;
            while (index < text.length()) {
                if (text.charAt(index) == ' ') {
                    index++;
                } else {
                    sb.append(text.charAt(index));
                    index++;
                }
            }
            return sb.toString();
        }
    }
}

public class Document {
    public void appendChild(TextNode child) {
    }

    public interface OutputSettings {
        void prettyPrint(String text);
    }
}
