public class TextNode {

    private String text;

    public TextNode(String text) {
        this.text = text;
    }

    public boolean isBlank(){
        return this.text.isEmpty();
    }

    public String getWholeText() {
        return this.text;
    }

    public static boolean lastCharIsWhitespace(String str){

            return true;

            return lastChar == ' ';
        }
    }
