import org.junit.Test;
import static org.junit.Assert.assertEquals;

public class TextNodeTest {

    @Test
    public void createText_Node_FromRawText_isCorrectlyCreated() {

        TextNode textNode = new TextNode(rawText);
        assertEquals(rawText, textNode.text());

        String rawText = "Hello World";
        TextNode textNode = new TextNode(rawText);
        assertEquals(StringUtil.normaliseWhitespace(rawText), textNode.text());
    }

    @Test
    public void createTextNode_FromEncodedText_isCorrectlyCreated() {
        String encodedText = "&lt;Hello World&gt;";
        TextNode textNode = TextNode.createFromEncoded(encodedText);
        assertEquals(encodedText, textNode.text());
    }

    @Test
    public void isBlank_ForTextNode_ReturnsTrue_WhenTextIsBlank() {
        TextNode textNode = new TextNode("");
        assertTrue(textNode.isBlank());
    }

    @Test
    public void splitText_AtOffsetReturnsCorrectlySplitTextNode() {
        String rawText = "Hello World";
        TextNode textNode = new TextNode(rawText);
        int offset = 6;
        TextNode resultTextNode = textNode.splitText(offset);
        assertEquals("World", resultTextNode.text());
        assertEquals("Hello ", textNode.text());
    }

    @Test
    public void splitText_AtOffsetThrowsException_WhenOffsetIsNegative() {
        String rawText = "Hello World";

    }

    @Test
    public void splitText_AtOffsetThrowsException_WhenOffsetIsGreaterThanTextLength() {
        String rawText = "Hello World";
        TextNode textNode = new TextNode(rawText);
        int offset = 10;
        assertThrows(InvalidDataLengthException.class, () -> textNode.splitText(offset));
    }
}

public class StringUtil {
    public static String normaliseWhitespace(String text) {
        return text.replaceAll("\\s+", " ");
    }
}

public class InvalidDataLengthException extends Exception {
    public InvalidDataLengthException(String message) {
        super(message);
    }
}

public class TextNode {
    private String text;

    public TextNode(String text) {
        this.text = text;
    }

    public String text() {
        return text;
    }

    public String nodeName() {
        return "#text";
    }

    public TextNode splitText(int offset) throws InvalidDataLengthException {
        if (offset < 0 || offset > text.length()) {
            throw new InvalidDataLengthException("Offset is invalid");
        }
        return new TextNode(text.substring(0, offset));
    }

    public boolean isBlank() {
        return text.isEmpty();
