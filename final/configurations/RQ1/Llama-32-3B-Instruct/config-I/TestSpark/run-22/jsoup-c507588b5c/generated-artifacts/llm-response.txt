        this.tag = tag;
    }

    public int siblingIndex() {

        return new Element();

    public void indent(OutputStreamWriter writer, int depth, Document.OutputSettings out) {}

public class TextNode extends Node {
    private String text;

    public TextNode(String text) {
        this.text = text;
    }

    public String text() {
        return text;
    }

    public boolean isBlank() {
        return StringUtil.isEmpty(text);
    }

    public void setText(String text) {
        this.text = text;
    }

    public Element getWholeText() {
        return new Element(text);
    }

    public TextNode splitText(int offset) {
        return new TextNode("");
    }

    public void outerHtmlHead(OutputStreamWriter writer, int depth, Document.OutputSettings outSettings) throws IOException {
        writer.append(outSettings.prettyPrint ? "\n" : "");
        writer.append("<").append(tag()).append(">");
        indent(writer, 0, outSettings);
        if (siblingIndex() > 0 || !isBlank()) {
            writer.write("\n");
        }
    }

    public void outerHtmlTail(OutputStreamWriter writer, int depth, Node parent) throws IOException {
        writer.append("</").append(tag()).append(">");
    }
}

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.select.Elements;

public class Main {

    public static String getHtml(String html) {
        return Jsoup.parse(html).html();
    }

    public static boolean hasClass(Elements elements, String className) {
        for (Element element : elements) {
            if (element.hasAttr("class") && element.attr("class").contains(className)) {
                return true;
            }
        }
        return false;
    }

    public static void main(String[] args) {
        // Test 1
        Document doc = Jsoup.parse("<html><body class='test-class'><p>Hello World!</p></body></html>");
        System.out.println(getHtml(doc.html()));
        System.out.println(hasClass(doc.select("body"), "test-class"));  // true

        // Test 2
        doc = Jsoup.parse("<html><body><p>Hello World!</p></body></html>");
        System.out.println(getHtml(doc.html()));
        System.out.println(hasClass(doc.select("body"), "test-class"));  // false

        // Test 3
        doc = Jsoup.parse("<html><body class='test-class'><div>Hello World!</div></body></html>");
        System.out.println(getHtml(doc.html()));
        System.out.println(hasClass(doc.select("body"), "test-class"));  // true
    }
}
