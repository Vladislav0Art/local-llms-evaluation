```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Attribute;
import org.jsoup.nodes.Comment;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.Text;

public class Element {
    private String tag;
    private Map<String, String> attributes = new HashMap<>();
    private Text content = null;

    public String getTag() {
        return tag;
    }

    public void setTag(String tag) {
        this.tag = tag;
    }

    public Map<String, String> getAttributes() {
        return attributes;
    }

    public void addAttribute(String key, String value) {
        attributes.put(key, value);
    }

    public Text getContent() {
        return content;
    }

    public void setContent(String text) {
        content = new Text(text);
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder("<" + tag);
        for (Map.Entry<String, String> entry : attributes.entrySet()) {
            sb.append(" " + entry.getKey() + "=\"" + entry.getValue() + "\"");
        }
        if (content != null) {
            sb.append(">").append(content.toString()).append("</").append(tag).append(">");
        } else {
            sb.append("/>").append(tag).append(">");
        }
        return sb.toString();
    }

    public static Element createElement(Element parent, String tag, Map<String, String> attributes, Text content) {
        Element element = new Element();
        element.setTag(tag);
        for (Map.Entry<String, String> entry : attributes.entrySet()) {
            element.addAttribute(entry.getKey(), entry.getValue());
        }
        if (content != null) {
            element.setContent(content.toString());
        } else {
            element.setContent("");
        }
        return element;
    }

    public static Element createElement(Element parent, org.jsoup.nodes.Element jsoupElement, int depth) {
        Element element = new Element();
        element.setTag(jsoupElement.tagName());
        for (Attribute attribute : jsoupElement.attributes()) {
            element.addAttribute(attribute.getKey(), attribute.getValue());
        }
        if (jsoupElement.hasText()) {
            Text textContent = new Text(jsoupElement.text());
            element.setContent(textContent.toString());
        } else if (jsoupElement.childNodes().size() > 0) {
            for (org.jsoup.nodes.Node childNode : jsoupElement.childNodes()) {
                if (childNode instanceof org.jsoup.nodes.Element) {
                    Element childEl = createElement(element, (org.jsoup.nodes.Element) childNode, depth + 1);
                    element.getChildren().add(childEl);
                } else if (childNode instanceof org.jsoup.nodes.TextNode) {
                    Text textContent = new Text(((org.jsoup.nodes.TextNode) childNode).text());
                    element.getContent().append(textContent.toString());
                }
            }
        }
        return element;
    }

    public static void main(String[] args) {
        Document document = Jsoup.connect("https://www.example.com").get();
        Element root = createElement(null, (Element) document.body(), 1);
        System.out.println(root.toString());
    }
}
```
