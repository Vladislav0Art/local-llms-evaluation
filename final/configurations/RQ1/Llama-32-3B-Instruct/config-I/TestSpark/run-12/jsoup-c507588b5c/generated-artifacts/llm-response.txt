public class TextNode {
    private String value;

    public TextNode(String value) {
        this.value = value;
    }

    public String text() {
        return value;
    }

    public void setText(String value) {
        this.value = value;
    }

    public String getWholeText() {
        return value.trim();
    }

    public TextNode splitText(int offset) {
        if (offset < 0 || offset > value.length()) {
            throw new IndexOutOfBoundsException("Offset is out of range");
        }
        String tail = value.substring(offset);
        return new TextNode(tail);
    }

    public void outerHtmlHead(ByteArrayOutputStream outStream, int offset, Document.OutputSettings settings) {
        if (!settings.isPrettyPrint) {
            outStream.write("<p>");
        }
        if (offset > 0 && offset < value.length()) {
            outStream.write(value.substring(0, offset));
        }
    }

    public void outerHtmlTail(ByteArrayOutputStream outStream, int offset, Document.OutputSettings settings) {
        if (!settings.isPrettyPrint) {
            outStream.write("</p>");
        }
        outStream.write(value.substring(offset).trim());
    }

    public static TextNode createFromEncoded(String encodedText) {
        return new TextNode(encodedText);
    }

    @Override
    public String toString() {
        return value;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;

        TextNode that = (TextNode) o;
        return Objects.equals(value, that.value);
    }

    public int hashCode() {
        return value.hashCode();
    }
}
