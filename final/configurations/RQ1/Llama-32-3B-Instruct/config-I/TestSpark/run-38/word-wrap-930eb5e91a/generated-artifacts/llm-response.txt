====================== Iteration #2 ======================
Here's a Java implementation of word wrapping for a text output stream:

```java
import java.io.*;
import java.util.*;

public class WordWrap implements LineConsumer {

    private final double MAX_WIDTH;
    private final PrintWriter writer;

    public WordWrap(PrintWriter writer, double maxWidth) {
        this.writer = writer;
        this.MAX_WIDTH = maxWidth;
    }

    @Override
    public void write(String input) throws IOException {
        wrap(input);
    }

    private void wrap(String text) throws IOException {
        StringBuilder line = new StringBuilder();
        String[] words = text.split("\\s+");

        for (String word : words) {
            if (line.length() + word.length() > MAX_WIDTH) {
                writer.println(line.toString());
                line.setLength(0);
            }
            appendWordToLine(line, word);
        }

        if (!line.toString().isEmpty()) {
            writer.println(line.toString());
        }
    }

    private void appendWordToLine(StringBuilder line, String word) {
        if (line.length() > 0 && line.charAt(line.length() - 1) == ' ') {
            line.setLength(line.length() - 1);
        }
        line.append(word);
    }
}
```

====================== Iteration #5 ======================
Here's a Java implementation of word wrapping for a text output stream:

```java
import java.io.*;
import java.util.*;

public class WordWrap {
    private final LineConsumer writer;
    private double maxWidth;

    public WordWrap(LineConsumer writer, double maxWidth) {
        this.writer = writer;
        this.maxWidth = maxWidth;
    }

    public void wrap(String input) throws IOException {
        String[] words = input.split("\\s+");
        StringBuilder currentLine = new StringBuilder();

        for (String word : words) {
            if (currentLine.length() + word.length() + 1 > maxWidth) {
                writer.write(currentLine.toString());
                currentLine.setLength(0);
            }

            if (!currentLine.toString().isEmpty()) {
                currentLine.append(' ');
            }
            currentLine.append(word);

            if (currentLine.length() == 0) {
                currentLine.append(word);
            }
        }

        if (currentLine.length() > 0) {
            writer.write(currentLine.toString());
        }
    }

    public void writeLine(LineConsumer writer, StringBuilder line) throws IOException {
        writer.write(line.toString());
        line.setLength(0);
    }
}

public interface LineConsumer {
    void write(String input) throws IOException;
}

public class ConsoleLineConsumer implements LineConsumer {
    @Override
    public void write(String input) throws IOException {
        System.out.println(input);
    }
}
```
