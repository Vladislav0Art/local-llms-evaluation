public class TagAttributeEnforcer {

    private final Map<String, Attributes> attributes = new HashMap<>();
    private final Map<String, AttributeValue> enforcedAttributes = new HashMap<>();
    private final Set<Protocol> protocols;

    public TagAttributeEnforcer() {
        this.protocols = new HashSet<>();
    }

    public void addAttribute(String tagName, String attributeName, String attributeValue) {
        Attributes attributes = getAttributes(tagName);
        if (attributes == null) {
            attributes = new Attributes();
            this.attributes.put(tagName, attributes);
        }
        
        AttributeKey key = AttributeKey.valueOf(attributeName);
        if (!protocols.contains(Protocol.valueOf(attributeValue))) {
            attributes.add(key, Protocol.valueOf(attributeValue));
        }
    }

    public void addEnforcedAttribute(String tagName, String attributeName, String attributeValue) {
        Map<AttributeKey, AttributeValue> keyVals = getEnforcedAttributes(tagName);
        if (keyVals == null) {
            keyVals = new HashMap<>();
            this.enforcedAttributes.put(tagName, keyVals);
        }
        
        AttributeKey key = AttributeKey.valueOf(attributeName);
        keyVals.put(key, AttributeValue.valueOf(attributeValue));
    }

    public void addProtocol(String protocol) {
        protocols.add(Protocol.valueOf(protocol));
    }

    private Attributes getAttributes(String tagName) {
        return attributes.get(tagName);
    }

    private Map<AttributeKey, AttributeValue> getEnforcedAttributes(String tagName) {
        return enforcedAttributes.get(tagName);
    }
}

public class TagAttributeEnforcerTest {

    @Test
    public void testAddAttribute() {
        TagAttributeEnforcer enforcer = new TagAttributeEnforcer();
        
        String tagName = "div";
        String attributeName = "href";
        String attributeValue = "#anchor";
        
        enforcer.addAttribute(tagName, attributeName, attributeValue);
        
        Assert.assertTrue(enforcer.getAttributes(tagName).containsKey(AttributeKey.valueOf(attributeName)));
