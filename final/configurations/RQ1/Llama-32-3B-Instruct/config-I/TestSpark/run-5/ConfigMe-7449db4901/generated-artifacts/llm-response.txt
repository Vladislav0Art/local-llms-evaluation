    }

    public void newCommentsConfiguration_customMap() {

        assertEquals(customMap, comments.getComments());
    }

    }

    public void setComment_overwritesExistingEntry() {
        Map<String, List<String>> map = new HashMap<>();

        assertEquals(Collections.singletonList("newComment"), comments.getComments().get("path1"));

        Map<String, List<String>> map = new HashMap<>();
        map.put("path1", Arrays.asList("comment1", "comment2"));

        Map<String, @UnmodifiableView List<String>> unmodifiableMap = comments.getAllComments();
        assertTrue(unmodifiableMap instanceof java.util.Map);
    }

    public void getAllComments_returnsCorrectPaths() {
        Map<String, List<String>> map = new HashMap<>();
        map.put("path1", Arrays.asList("comment1", "comment2"));

        assertTrue(comments.getAllComments().containsKey("path1"));
    }
