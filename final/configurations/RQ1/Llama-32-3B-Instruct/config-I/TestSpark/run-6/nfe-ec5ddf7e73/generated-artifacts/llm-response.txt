public class DFStringValidador {
    public static boolean placaDeVeiculo(String placa, String mensagemErro) {
        if (placa.length() == 7 && isAlfabetico(placa)) {
            return true;
        } else {
            System.out.println(mensagemErro);
            return false;
        }
    }

    private static boolean isAlfabetico(String palavra) {
        for (char c : palavra.toCharArray()) {
            if (!Character.isLetter(c)) {
                return false;
            }
        }
        return true;
    }

    public static boolean validaIntervalo(String renavam, int inicio, int fim, String mensagemErro) {
        if (renavam.length() == 8 && renavam.startsWith("RR") && renavam.endsWith("R")) {
            if (Integer.parseInt(renavam.substring(1, 3)) >= inicio && Integer.parseInt(renavam.substring(1, 3)) <= fim) {
                return true;
            } else {
                System.out.println(mensagemErro);
                return false;
            }
        } else {
            System.out.println(mensagemErro);
            return false;
        }
    }

    public static boolean capacidadeNDigitos(String valor, String mensagemErro, int quantidadeDeDigitos) {
        if (valor.length() == quantidadeDeDigitos && valor.matches("\\d+")) {
            return true;
        } else {
            System.out.println(mensagemErro);
            return false;
        }
    }
