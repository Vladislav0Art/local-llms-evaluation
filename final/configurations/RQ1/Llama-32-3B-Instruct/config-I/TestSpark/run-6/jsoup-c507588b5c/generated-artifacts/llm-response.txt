public class stripConsecutiveWhitespace_RemoveAll {
    public String process(String s) {
        if (s == null || s.isEmpty()) {
            return s;
        } else {
            StringBuilder builder = new StringBuilder();
            for (int i = 0; i < s.length(); i++) {
                char c = s.charAt(i);
                if (!Character.isWhitespace(c)) {
                    builder.append(c);
                }
            }
            return builder.toString();
        }
    }
}
