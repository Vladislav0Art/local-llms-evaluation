    public void isBlank_GivenTextNode_ReturnsFalse() {

    public void text(OutputStream out) throws IOException {} // Changed method name to avoid conflict with outerHtmlHead
    public void outerHtmlHead(OutputStream out, int offset, OutputSettings settings) throws IOException {} 
    private int bufferIndex = 0; // Made it private and added initialization
}

public class OutputStream extends java.io.OutputStream {
    byte[] buffer;
    int bufferIndex;

    public OutputStream(int bufferSize) { // Added constructor
        buffer = new byte[bufferSize];
        bufferIndex = 0;
    }

    @Override
    public void write(int b) throws IOException { 
        buffer[bufferIndex++] = (byte) b; // Replaced Arrays.copyOf with manual indexing
    }
    public void writeBytes(byte[] bytes) { // Added method to avoid duplicate method name
        System.arraycopy(bytes, 0, this.buffer, bufferIndex, bytes.length);
        bufferIndex += bytes.length;
    }

    @Override
    public byte[] getBuffer() {
        return Arrays.copyOf(buffer, bufferIndex); 
    }
}

public class ByteArrayOutputStream extends OutputStream { // Made it extend OutputStream instead of implementing methods from it
    public ByteArrayOutputStream(int bufferSize) {
        super(bufferSize);
    }
}
