```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Comment;
import org.jsoup.nodes.Document;
import org.jsoup.select.Elements;

public class GeneratedTest {

    private static final String HTML_STRING = "<html><body>Comment <!-- This is a comment --> <p>Hello World!</p></body></html>";

    @org.junit.Test
    public void testRemoveComments() {
        Document doc = Jsoup.parse(HTML_STRING);
        Elements comments = doc.getAllElements().select("comment");

        for (Element comment : comments) {
            comment.remove();
        }

        String modifiedHtml = doc.toString();

        // Check if all comments were removed from the original HTML string
        org.junit.Assert.assertFalse(modifiedHtml.contains("\n" + "    <!-- This is a comment -->\n"));
        org.junit.Assert.assertTrue(modifiedHtml.contains("<p>Hello World!</p>"));
    }
}
```

This code tests the removal of comments from an HTML string using Jsoup. It selects all comments, removes them, and then checks if they were successfully removed by verifying that the modified HTML no longer contains the original comment.
