```java
public class TextNode {
    private String value;

    public TextNode(String value) {
        this.value = value;
    }

    public Object splitText(int offset) throws IOException {
        if (offset < 0 || offset > value.length()) {
            throw new IndexOutOfBoundsException();
        }
        return new TextNode(value.substring(offset));
    }

    public String outerHtml() {
        StringBuilder sb = new StringBuilder();
        sb.append("<p>");
        for (int i = 0; i < value.length(); i++) {
            if (value.charAt(i) == '<') {
                break;
            } else {
                sb.append(value.charAt(i));
            }
        }
        sb.append("</p>");
        return sb.toString();
    }

    public String outerHtmlHead(StringBuilder sb, int offset, Document.OutputSettings settings) throws IOException {
        sb.append("<div>");
        for (int i = 0; i < value.length(); i++) {
            if (value.charAt(i) == '<') {
                break;
            } else {
                sb.append(value.charAt(i));
            }
        }
        sb.append("</div>");
        return "";
    }
}

class StringUtil {
    public static String normaliseWhitespace(String text) {
        return text.replaceAll("\\s+", " ");
    }
}

enum Document.OutputSettings {
    STATIC;
}
```
