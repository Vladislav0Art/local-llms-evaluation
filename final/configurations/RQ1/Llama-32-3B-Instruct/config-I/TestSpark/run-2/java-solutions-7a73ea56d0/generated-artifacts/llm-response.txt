public class StockSpanTest {
    @Test
    public void testBasicExample() {
        StockSpan onlineStockSpan = new StockSpan();
        int[] result1 = onlineStockSpan.next(new int[]{3, 10, 15});
        assertEquals(new int[]{1, 1, 2}, result1);
    }

    @Test
    public void testNextWithNullValue() throws Exception {
        StockSpan onlineStockSpan = new StockSpan();
        assertThrows(NullPointerException.class, () -> onlineStockSpan.next(null));
    }

    @Test
    public void testAdd() {
        StockSpan onlineStockSpan = new StockSpan();
        onlineStockSpan.add(1);
        onlineStockSpan.add(2);
        onlineStockSpan.add(3);
        assertEquals(new int[]{2, 4, 6}, onlineStockSpan.next());
    }

    @Test
    public void testAddMultiple() {
        StockSpan onlineStockSpan = new StockSpan();
        for (int i = 1; i <= 10000; i++) {
            onlineStockSpan.add(i);
        }
        int[] result = onlineStockSpan.next();
        assertEquals(new int[]{1, 2, 3}, Arrays.copyOf(result, result.length));
    }
}

public class StockSpan {
    private int[] stock;
    private int index;

    public StockSpan() {
        this.stock = new int[101];
        this.index = -1;
    }

    public void add(int x) {
        index++;
        stock[index] = x;
    }

    public int next() {
        while (index > 0 && stock[index] <= stock[index - 1]) {
            --index;
        }
        return index == 0 ? 1 : index + 1;
    }
}
