import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.Node;
import org.jsoup.parser.*;
import org.junit.Test;
import org.mockito.Mockito;

import java.io.StringReader;
import java.util.List;

import static org.junit.Assert.*;

public class XmlTreeBuilderTests {

    @Test
    public void initialiseParseTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        xmlTreeBuilder.initialiseParse(new StringReader("<a></a>"), "http://test.com", new Parser(xmlTreeBuilder));
        assertEquals("http://test.com", xmlTreeBuilder.getBaseUri());
        assertEquals(1, xmlTreeBuilder.stack.size());
    }

    @Test
    public void parseReaderBaseUriTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        Document doc = xmlTreeBuilder.parse(new StringReader("<a></a>"), "http://test.com");
        assertEquals("http://test.com", doc.baseUri());
    }

    @Test
    public void parseStringBaseUriTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        Document doc = xmlTreeBuilder.parse("<a></a>", "http://test.com");
        assertEquals("http://test.com", doc.baseUri());
    }

    @Test
    public void processTokenTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        Token token = new Token.StartTag();
        token.tagName = "p";
        assertTrue(xmlTreeBuilder.process(token));
    }

    @Test
    public void processTokenEOFTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        Token token = new Token.EOF();
        assertTrue(xmlTreeBuilder.process(token));
    }

    @Test(expected = IllegalArgumentException.class)
    public void processTokenInvalidTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        Token token = Mockito.mock(Token.class);
        xmlTreeBuilder.process(token);
    }

    @Test
    public void popStackToCloseTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        Token.EndTag endTag = new Token.EndTag("p");
        xmlTreeBuilder.stack.add(new Element("p", "http://test.com"));
        xmlTreeBuilder.popStackToClose(endTag);
        assertTrue(xmlTreeBuilder.stack.isEmpty());
    }

    @Test
    public void parseFragmentStringTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        List<Node> nodes = xmlTreeBuilder.parseFragment("<a></a>", "http://test.com", new Parser(xmlTreeBuilder));
        assertTrue(nodes.get(0) instanceof Element);
    }

    @Test
    public void parseFragmentElementTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        Element element = new Element("p", "http://test.com");
        List<Node> nodes = xmlTreeBuilder.parseFragment("<a></a>", element, "http://test.com", new Parser(xmlTreeBuilder));
        assertTrue(nodes.get(0) instanceof Element);
    }
}
