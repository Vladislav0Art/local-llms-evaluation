import org.jsoup.Connection;
import org.jsoup.helper.UrlBuilder;
import org.junit.Test;
import java.net.URL;
import java.util.ArrayList;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertThrows;
import static org.mockito.Mockito.mock;
import static org.mockito.Mockito.when;

public class UrlBuilderTest {

    @Test
    public void buildBasicUrlTest() throws Exception {
        URL url = new URL("http://www.test.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        URL newUrl = urlBuilder.build();
        assertEquals(newUrl, url);
    }

    @Test
    public void buildUrlWithQueryTest() throws Exception {
        URL url = new URL("http://www.test.com?param=value");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        URL newUrl = urlBuilder.build();
        assertEquals(newUrl, url);
    }

    @Test
    public void buildUrlWithMalformedUrlTest() throws Exception {
        URL url = new URL("ht://www.test.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        assertThrows(Exception.class, urlBuilder::build);
    }

    @Test
    public void appendKeyValTest() throws Exception {
        URL url = new URL("http://www.test.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        Connection.KeyVal kv = mock(Connection.KeyVal.class);
        when(kv.key()).thenReturn("param");
        when(kv.value()).thenReturn("value");
        urlBuilder.appendKeyVal(kv);
        URL newUrl = urlBuilder.build();
        assertEquals(newUrl, new URL("http://www.test.com?param=value"));
    }

    @Test
    public void appendKeyValUnsupportedEncodingTest() throws Exception {
        URL url = new URL("http://www.test.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        Connection.KeyVal kv = mock(Connection.KeyVal.class);
        when(kv.key()).thenReturn("param");
        when(kv.value()).thenReturn(new String(new byte[]{(byte) 129}, "ISO-8859-1"));
        assertThrows(Exception.class, () -> urlBuilder.appendKeyVal(kv));
    }

    @Test
    public void buildUrlWithNonAsciiHostTest() throws Exception {
        URL url = new URL("http://www.测试.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        URL newUrl = urlBuilder.build();
        assertEquals(newUrl.getHost(), "www.xn--0zwm56d.com");
    }

    @Test
    public void buildUrlWithNonAsciiPathTest() throws Exception {
        URL url = new URL("http://www.test.com/测试");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        URL newUrl = urlBuilder.build();
        assertEquals(newUrl.getPath(), "/%E6%B5%8B%E8%AF%95");
    }
}
