import io.netty.buffer.ByteBuf;
import io.netty.buffer.Unpooled;
import io.netty.channel.Channel;
import org.junit.Test;
import org.mockito.Mockito;
import org.traccar.model.Position;
import org.traccar.protocol.WatchProtocolDecoder;
import java.net.SocketAddress;
import java.nio.charset.StandardCharsets;
import static org.junit.Assert.*;
import static org.mockito.Mockito.mock;

public class WatchProtocolDecoderTest {

    private String id = "idTest";
    private String content = "contentTest";

    private ByteBuf getByteBuf(String data) {
        return Unpooled.wrappedBuffer(data.getBytes(StandardCharsets.US_ASCII));
    }

    @Test
    public void decodePositionScenarioOneTest() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(null);
        assertNull(decoder.decodePosition(null, "invalidDataTest"));
    }

    @Test
    public void getHasIndexTrueTest() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(null);
        decoder.getHasIndex();
    }

    @Test
    public void getManufacturerTest() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(null);
        decoder.getManufacturer();
    }

    @Test
    public void sendResponseWithIndexTest() {
        Channel channel = Mockito.mock(Channel.class);
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(null);
        decoder.sendResponse(channel, id, "indexTest", content);
        Mockito.verify(channel, Mockito.times(1)).writeAndFlush(Mockito.any());
    }

    @Test
    public void sendResponseWithoutIndexTest() {
        Channel channel = Mockito.mock(Channel.class);
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(null);
        decoder.sendResponse(channel, id, null, content);
        Mockito.verify(channel, Mockito.times(1)).writeAndFlush(Mockito.any());
    }

    @Test
    public void decodeAlarmStatusZeroTest() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(null);
        assertEquals("lowBattery", decoder.decodeAlarm(0));
    }

    @Test
    public void decodeAlarmStatusOneTest() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(null);
        assertEquals("geofenceExit", decoder.decodeAlarm(1));
    }

    @Test
    public void decodeAlarmStatusOtherTest() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(null);
        assertNull(decoder.decodeAlarm(333));
    }

    @Test
    public void decodeTest() throws Exception {
        Channel channel = Mockito.mock(Channel.class);
        ByteBuf byteBuf = getByteBuf("[CS*1*0016*LK]");
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(null);
        assertNotNull(decoder.decode(channel, mock(SocketAddress.class), byteBuf));
    }

    @Test
    public void decodeAlarmNullTest() {
        WatchProtocolDecoder decoder = new WatchProtocolDecoder(null);
        assertNull(decoder.decodeAlarm(23));
    }

}
