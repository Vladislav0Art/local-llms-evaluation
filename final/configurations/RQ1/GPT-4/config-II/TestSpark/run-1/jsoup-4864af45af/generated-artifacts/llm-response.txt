import org.jsoup.nodes.Attribute;
import org.jsoup.nodes.Element;
import org.jsoup.safety.Safelist;
import org.junit.Test;

import static org.junit.Assert.assertFalse;
import static org.junit.Assert.assertTrue;

public class SafelistTest {

    @Test
    public void noneIsSafeTagTest() {
        Safelist safelist = Safelist.none();
        String tag = "p";
        assertFalse(safelist.isSafeTag(tag));
    }

    @Test
    public void simpleTextIsSafeTagTest() {
        Safelist safelist = Safelist.simpleText();
        String tag = "em";
        assertTrue(safelist.isSafeTag(tag));
    }

    @Test
    public void basicIsSafeTagTest() {
        Safelist safelist = Safelist.basic();
        String tag = "a";
        assertTrue(safelist.isSafeTag(tag));
    }

    @Test
    public void basicWithImagesIsSafeTagTest() {
        Safelist safelist = Safelist.basicWithImages();
        String tag = "img";
        assertTrue(safelist.isSafeTag(tag));
    }

    @Test
    public void relaxedIsSafeTagTest() {
        Safelist safelist = Safelist.relaxed();
        String tag = "div";
        assertTrue(safelist.isSafeTag(tag));
    }

    @Test
    public void addTagsTest() {
        Safelist safelist = Safelist.none();
        String tag = "p";
        safelist.addTags(tag);
        assertTrue(safelist.isSafeTag(tag));
    }

    @Test
    public void removeTagsTest() {
        Safelist safelist = Safelist.simpleText();
        String tag = "em";
        safelist.removeTags(tag);
        assertFalse(safelist.isSafeTag(tag));
    }

    @Test
    public void isSafeAttributeTest() {
        Safelist safelist = Safelist.basic();
        Element element = new Element("a");
        Attribute attribute = new Attribute("href", "http://url.com");

        assertTrue(safelist.isSafeAttribute("a", element, attribute));
    }

    @Test
    public void addEnforcedAttributeTest() {
        Safelist safelist = Safelist.none();
        Element element = new Element("a");
        Attribute attribute = new Attribute("rel", "nofollow");
        safelist.addEnforcedAttribute("a", "rel", "nofollow");

        assertTrue(safelist.isSafeAttribute("a", element, attribute));
    }

    @Test
    public void removeEnforcedAttributeTest() {
        Safelist safelist = Safelist.none();
        Element element = new Element("a");
        Attribute attribute = new Attribute("rel", "nofollow");
        safelist.addEnforcedAttribute("a", "rel", "nofollow");
        safelist.removeEnforcedAttribute("a", "rel");

        assertFalse(safelist.isSafeAttribute("a", element, attribute));
    }

    @Test
    public void preserveRelativeLinksTest() {
        Safelist safelist = Safelist.none();
        safelist.preserveRelativeLinks(true);

        Element element = new Element("a");
        Attribute attribute1 = new Attribute("href", "/relative/url");
        Attribute attribute2 = new Attribute("href", "http://url.com");

        assertTrue(safelist.isSafeAttribute("a", element, attribute1));
        assertTrue(safelist.isSafeAttribute("a", element, attribute2));
    }

    @Test
    public void addProtocolsTest() {
        Safelist safelist = Safelist.none();
        Element element = new Element("a");
        Attribute attribute = new Attribute("href", "ftp://url.com");
        safelist.addProtocols("a", "href", "ftp");

        assertTrue(safelist.isSafeAttribute("a", element, attribute));
    }

    @Test
    public void removeProtocolsTest() {
        Safelist safelist = Safelist.basic();
        Element element = new Element("a");
        Attribute attribute = new Attribute("href", "ftp://url.com");
        safelist.removeProtocols("a", "href", "ftp");

        assertFalse(safelist.isSafeAttribute("a", element, attribute));
    }
}
