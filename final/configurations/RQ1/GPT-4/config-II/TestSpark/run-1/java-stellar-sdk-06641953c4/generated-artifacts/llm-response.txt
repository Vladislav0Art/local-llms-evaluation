import org.junit.Test;
import org.junit.Assert;
import org.mockito.Mockito;
import net.i2p.crypto.eddsa.EdDSAEngine;
import net.i2p.crypto.eddsa.EdDSAPrivateKey;
import net.i2p.crypto.eddsa.EdDSAPublicKey;
import org.stellar.sdk.xdr.SignatureHint;

public class KeyPairTest {

    @Test
    public void KeyPairPublicConstructorTest(){
        EdDSAPublicKey publicKey = Mockito.mock(EdDSAPublicKey.class);
        KeyPair keyPair = new KeyPair(publicKey);
        Assert.assertNotNull(keyPair);
        Assert.assertEquals(publicKey, keyPair.getPublicKey());
    }

    @Test
    public void KeyPairPrivateConstructorTest(){
        EdDSAPublicKey publicKey = Mockito.mock(EdDSAPublicKey.class);
        EdDSAPrivateKey privateKey = Mockito.mock(EdDSAPrivateKey.class);
        KeyPair keyPair = new KeyPair(publicKey, privateKey);
        Assert.assertNotNull(keyPair);
        Assert.assertEquals(publicKey, keyPair.getPublicKey());
        Assert.assertTrue(keyPair.canSign());
    }

    @Test(expected = NullPointerException.class)
    public void KeyPairNullPublicConstructorTest(){
        KeyPair keyPair = new KeyPair(null);
    }

    @Test
    public void canSignTest() {
        EdDSAPublicKey publicKey = Mockito.mock(EdDSAPublicKey.class);
        EdDSAPrivateKey privateKey = Mockito.mock(EdDSAPrivateKey.class);
        KeyPair keyPair = new KeyPair(publicKey, privateKey);
        Assert.assertTrue(keyPair.canSign());

        KeyPair keyPairWithoutPrivateKey = new KeyPair(publicKey);
        Assert.assertFalse(keyPairWithoutPrivateKey.canSign());
    }

    @Test
    public void fromAccountIdTest() {
        String accountId = "AccountId";
        KeyPair keyPair = KeyPair.fromAccountId(accountId);
        Assert.assertNotNull(keyPair);
        Assert.assertEquals(accountId, keyPair.getAccountId());
    }

    @Test
    public void fromPublicKeyTest() {
        byte[] publicKeyBytes = new byte[32];
        KeyPair keyPair = KeyPair.fromPublicKey(publicKeyBytes);
        Assert.assertNotNull(keyPair);
        Assert.assertArrayEquals(publicKeyBytes, keyPair.getPublicKey());
    }

    @Test
    public void signTest() throws Exception {
        byte[] data = "some data".getBytes();
        EdDSAPublicKey publicKey = Mockito.mock(EdDSAPublicKey.class);
        EdDSAPrivateKey privateKey = Mockito.mock(EdDSAPrivateKey.class);
        KeyPair keyPair = new KeyPair(publicKey, privateKey);
        EdDSAEngine mockedEngine = Mockito.mock(EdDSAEngine.class);
        Mockito.when(mockedEngine.sign(Mockito.any())).thenReturn(data);
        keyPair.sign(data);
        Mockito.verify(mockedEngine).sign(data);
    }

    @Test(expected = RuntimeException.class)
    public void signWithoutPrivateKeyTest() throws Exception {
        byte[] data = "some data".getBytes();
        EdDSAPublicKey publicKey = Mockito.mock(EdDSAPublicKey.class);
        KeyPair keyPair = new KeyPair(publicKey);
        keyPair.sign(data);
    }

    @Test
    public void verifyTest() throws Exception {
        byte[] data = "some data".getBytes();
        byte[] signature = "signature".getBytes();
        EdDSAPublicKey publicKey = Mockito.mock(EdDSAPublicKey.class);
        KeyPair keyPair = new KeyPair(publicKey);
        EdDSAEngine mockedEngine = Mockito.mock(EdDSAEngine.class);
        Mockito.when(mockedEngine.verify(Mockito.any())).thenReturn(true);
        boolean result = keyPair.verify(data, signature);
        Assert.assertTrue(result);
    }

    @Test  
    public void getSignatureHintTest() {  
        EdDSAPublicKey publicKey = Mockito.mock(EdDSAPublicKey.class);  
        KeyPair keyPair = new KeyPair(publicKey);  
        SignatureHint hint = keyPair.getSignatureHint();  
        Assert.assertNotNull(hint);  
    }  
}
