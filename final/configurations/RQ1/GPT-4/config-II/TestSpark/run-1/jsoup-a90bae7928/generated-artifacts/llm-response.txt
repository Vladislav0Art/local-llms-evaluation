package org.jsoup.parser;

import org.jsoup.nodes.*;
import org.junit.Assert;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import java.io.Reader;
import java.util.List;

import static org.mockito.ArgumentMatchers.any;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class XmlTreeBuilderTest {

    @Mock
    private Reader reader;

    @Mock
    private Parser parser;

    @Test
    public void defaultSettingsTest() {
        XmlTreeBuilder treeBuilder = new XmlTreeBuilder();
        Assert.assertEquals(ParseSettings.preserveCase, treeBuilder.defaultSettings());
    }

    @Test
    public void initialiseParseTest() {
        XmlTreeBuilder treeBuilder = new XmlTreeBuilder();
        treeBuilder.initialiseParse(reader, "", parser);
        Assert.assertFalse(treeBuilder.stack.isEmpty());
    }

    @Test
    public void newInstanceTest() {
        XmlTreeBuilder treeBuilder = new XmlTreeBuilder();
        Assert.assertNotNull(treeBuilder.newInstance());
    }

    @Test
    public void processTest() {
        XmlTreeBuilder treeBuilder = new XmlTreeBuilder();
        Token token = new Token.EndTag();
        Assert.assertTrue(treeBuilder.process(token));
    }

    @Test
    public void insertNodeTest() {
        XmlTreeBuilder treeBuilder = new XmlTreeBuilder();
        treeBuilder.initialiseParse(reader, "", parser);
        Element element = new Element(Tag.valueOf("test"), "test");
        treeBuilder.insertNode(element);
        Assert.assertEquals(element, treeBuilder.currentElement().childNode(0));
    }

    @Test
    public void insertTest() {
        XmlTreeBuilder treeBuilder = new XmlTreeBuilder();
        treeBuilder.initialiseParse(reader, "", parser);
        Token.StartTag startTag = new Token.StartTag();
        startTag.name("test").selfClosing();
        Element element = treeBuilder.insert(startTag);
        Assert.assertEquals("test", element.tagName());
    }

    @Test
    public void popStackToCloseTest() {
        XmlTreeBuilder treeBuilder = new XmlTreeBuilder();
        treeBuilder.initialiseParse(reader,"", parser);
        Token.StartTag startTag = new Token.StartTag().name("test");
        treeBuilder.insert(startTag);
        Token.EndTag endTag = new Token.EndTag().name("test");
        treeBuilder.popStackToClose(endTag);
        Assert.assertEquals(1, treeBuilder.stack.size());
    }

    @Test
    public void parseFragmentTest() throws Exception {
        XmlTreeBuilder treeBuilder = new XmlTreeBuilder();
        treeBuilder.initialiseParse(reader,"", parser);
        List<Node> nodes = treeBuilder.parseFragment("<test>content</test>", "", parser);
        Assert.assertEquals(1, nodes.size());
    }

    @Test
    public void parseFragmentWithContextTest() throws Exception {
        Element context = new Element(Tag.valueOf("test"), "test");
        when(parser.parseFragment(any(), any(), any())).thenReturn(null);
        XmlTreeBuilder treeBuilder = new XmlTreeBuilder();
        treeBuilder.initialiseParse(reader,"", parser);
        Assert.assertNull(treeBuilder.parseFragment("<test>content</test>", context, "", parser));
    }
}
