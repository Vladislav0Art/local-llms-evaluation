import org.jsoup.parser.ParseSettings;
import org.jsoup.parser.Tag;
import org.junit.Test;

import static org.junit.Assert.*;

public class TagTest {

    @Test
    public void getNameTest() {
        Tag t = new Tag("div");
        assertEquals("div", t.getName());
    }

    @Test
    public void normalNameTest() {
        Tag t = new Tag("Div");
        assertEquals("div", t.normalName());
    }

    @Test
    public void valueOfTest() {
        Tag t1 = Tag.valueOf("div");
        Tag t2 = Tag.valueOf("Div", ParseSettings.preserveCase);
        assertEquals("div", t1.getName());
        assertEquals("Div", t2.getName());
    }

    @Test
    public void isBlockTest() {
        Tag t = new Tag("div");
        assertTrue(t.isBlock());
    }

    @Test
    public void isInlineTest() {
        Tag t = new Tag("span");
        assertFalse(t.isBlock());
    }

    @Test
    public void isEmptyTest() {
        Tag t = new Tag("img");
        assertTrue(t.isEmpty());
    }

    @Test
    public void isSelfClosingTest() {
        Tag t = new Tag("img");
        assertTrue(t.isSelfClosing());
    }

    @Test
    public void isKnownTagTest() {
        assertTrue(Tag.isKnownTag("img"));
        assertFalse(Tag.isKnownTag("unknownTag"));
    }

    @Test
    public void preserveWhitespaceTest() {
        Tag t = new Tag("pre");
        assertTrue(t.preserveWhitespace());
    }

    @Test
    public void isFormListedTest() {
        Tag t = new Tag("input");
        assertTrue(t.isFormListed());
    }

    @Test
    public void isFormSubmittableTest() {
        Tag t = new Tag("input");
        assertTrue(t.isFormSubmittable());
    }

    @Test
    public void equalsTest() {
        Tag t1 = new Tag("div");
        Tag t2 = new Tag("div");
        assertEquals(t1, t2);
    }

    @Test
    public void hashCodeTest() {
        Tag t = new Tag("div");
        assertNotNull(t.hashCode());
    }

    @Test
    public void toStringTest() {
        Tag t = new Tag("div");
        assertEquals("div", t.toString());
    }

    @Test
    public void cloneTest() {
        Tag t1 = new Tag("div");
        Tag t2 = t1.clone();
        assertEquals(t1, t2);
    }
}
