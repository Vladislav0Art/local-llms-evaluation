import ch.jalu.configme.configurationdata.CommentsConfiguration;
import org.junit.Test;
import org.mockito.Mockito;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import static org.hamcrest.CoreMatchers.is;
import static org.hamcrest.MatcherAssert.assertThat;

public class CommentsConfigurationTest {

    @Test
    public void setCommentTest() {
        CommentsConfiguration cc = new CommentsConfiguration();
        cc.setComment("path1", "Test Comment1", "Test Comment2");
        Map<String, List<String>> comments = cc.getAllComments();

        assertThat(comments.size(), is(1));
        assertThat(comments.get("path1").size(), is(2));
        assertThat(comments.get("path1").get(0), is("Test Comment1"));
        assertThat(comments.get("path1").get(1), is("Test Comment2"));
    }

    @Test
    public void setCommentOverridePreviousValueTest() {
        CommentsConfiguration cc = new CommentsConfiguration();
        cc.setComment("path1", "Old Comment");
        cc.setComment("path1", "New Comment");
        Map<String, List<String>> comments = cc.getAllComments();

        assertThat(comments.get("path1").size(), is(1));
        assertThat(comments.get("path1").get(0), is("New Comment"));
    }

    @Test
    public void getAllCommentsTest() {
        Map<String, List<String>> mockComments = new HashMap<>();
        mockComments.put("path1", Arrays.asList("Test Comment1", "Test Comment2"));
        CommentsConfiguration cc = new CommentsConfiguration(mockComments);
        Map<String, List<String>> comments = cc.getAllComments();

        assertThat(comments, is(mockComments));
    }
}
