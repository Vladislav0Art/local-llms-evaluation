import org.jsoup.Connection;
import org.junit.Test;

import java.net.MalformedURLException;
import java.net.URL;
import java.net.URISyntaxException;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNotNull;
import static org.junit.Assert.assertNull;

public class UrlBuilderTest {

    @Test
    public void urlBuilderConstructorTest() throws MalformedURLException {
        URL url = new URL("http://example.com?param='value'");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        assertNotNull(urlBuilder);
    }

    @Test
    public void buildNonAsciiProtocolTest() throws MalformedURLException, URISyntaxException {
        URL url = new URL("ht%20tp://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        URL normalizedURL = urlBuilder.build();
        assertEquals("http://example.com", normalizedURL.toString());
    }

    @Test
    public void appendKeyValTest() throws MalformedURLException, UnsupportedEncodingException {
        URL url = new URL("http://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        Connection.KeyVal keyValue = Connection.KeyVal.create("param", "value");
        urlBuilder.appendKeyVal(keyValue);
        assertEquals("param=value", urlBuilder.q.toString());
    }

    @Test
    public void buildWithQueryTest() throws MalformedURLException, URISyntaxException {
        URL url = new URL("http://example.com?param=value");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        URL normalizedURL = urlBuilder.build();
        assertEquals("http://example.com?param=value", normalizedURL.toString());
    }

    @Test
    public void buildWithNonAsciiPathTest() throws MalformedURLException, URISyntaxException {
        URL url = new URL("http://example.com/%20path");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        URL normalizedURL = urlBuilder.build();
        assertEquals("http://example.com/ path", normalizedURL.toString());
    }

    @Test
    public void buildWithRefTest() throws MalformedURLException, URISyntaxException {
        URL url = new URL("http://example.com#ref");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        URL normalizedURL = urlBuilder.build();
        assertEquals("http://example.com#ref", normalizedURL.toString());
    }

    @Test
    public void buildWithNonAsciiHostTest() throws MalformedURLException, URISyntaxException {
        URL url = new URL("http://\u2211.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        URL normalizedURL = urlBuilder.build();
        assertEquals("http://xn--74h.com", normalizedURL.toString());
    }

    @Test(expected = RuntimeException.class)
    public void decodePartUnsupportedEncodingExceptionTest() throws MalformedURLException {
        URL url = new URL("http://example.com");
        UrlBuilder urlBuilder = new UrlBuilder(url);
        urlBuilder.u = new UrlBuilder(url).build();
        urlBuilder.decodePart("%ZZ");
    }

    @Test
    public void normalizeQueryTest() {
        String result = UrlBuilder.normalizeQuery("param=value");
        assertEquals("param=value", result);
    }

    @Test
    public void normalizeRefTest() {
        String result = UrlBuilder.normalizeRef("ref");
        assertEquals("ref", result);
    }
}
