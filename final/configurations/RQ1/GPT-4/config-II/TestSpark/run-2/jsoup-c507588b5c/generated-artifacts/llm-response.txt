import org.jsoup.nodes.TextNode;
import org.junit.Test;

import static junit.framework.TestCase.assertEquals;
import static junit.framework.TestCase.assertTrue;
import static org.junit.Assert.assertFalse;

public class TextNodeTest {

    @Test
    public void TextNodeConstructorTest() {
        String value = "Test value";
        TextNode node = new TextNode(value);

        assertEquals(value, node.getWholeText());
    }

    @Test
    public void nodeNameTest() {
        TextNode node = new TextNode("Test value");

        assertEquals("#text", node.nodeName());
    }

    @Test
    public void textTest() {
        String value = "   Test value   ";
        TextNode node = new TextNode(value);

        assertEquals(value.trim(), node.text()); //Asserting with whitespace normalised
    }

    @Test
    public void getWholeTextTest() {
        String value = "Test value";
        TextNode node = new TextNode(value);

        assertEquals(value, node.getWholeText());
    }

    @Test
    public void isBlankTrueTest() {
        TextNode node = new TextNode(" ");

        assertTrue(node.isBlank());
    }

    @Test
    public void isBlankFalseTest() {
        TextNode node = new TextNode("Test value");

        assertFalse(node.isBlank());
    }

    @Test
    public void splitTextTest() {
        String value = "Test value";
        TextNode node = new TextNode(value);

        TextNode newNode = node.splitText(4);

        assertEquals("Test", node.getWholeText());
        assertEquals(" value", newNode.getWholeText());
    }

    @Test
    public void cloneTest() {
        String value = "Test value";
        TextNode node = new TextNode(value);

        TextNode cloneNode = node.clone();

        assertEquals(cloneNode.getWholeText(), node.getWholeText());
    }

    @Test
    public void createFromEncodedTest() {
        String value = "&amp;Test value";
        TextNode node = TextNode.createFromEncoded(value);

        assertEquals("&Test value", node.getWholeText());
    }

    @Test
    public void normaliseWhitespaceTest() {
        String value = "    Normalise whitespace    ";
        String normalised = TextNode.normaliseWhitespace(value);

        assertEquals("Normalise whitespace", normalised);
    }

    @Test
    public void stripLeadingWhitespaceTest() {
        String value = "    Strip leading whitespace";
        String stripped = TextNode.stripLeadingWhitespace(value);

        assertEquals("Strip leading whitespace", stripped);
    }

    @Test
    public void lastCharIsWhitespaceTrueTest() {
        StringBuilder sb = new StringBuilder(" ");
        assertTrue(TextNode.lastCharIsWhitespace(sb));
    }

    @Test
    public void lastCharIsWhitespaceFalseTest() {
        StringBuilder sb = new StringBuilder("A");
        assertFalse(TextNode.lastCharIsWhitespace(sb));
    }
}
