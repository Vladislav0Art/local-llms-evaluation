package org.jsoup.helper;

import org.jsoup.Connection;
import org.junit.Assert;
import org.junit.Test;
import java.net.URL;

public class UrlBuilderTest {

    @Test
    public void buildUrlWithNoQueryTest() throws Exception {
        UrlBuilder urlBuilder = new UrlBuilder(new URL("https://google.com"));
        URL url = urlBuilder.build();
        Assert.assertEquals("https://google.com", url.toString());
    }

    @Test
    public void buildUrlWithQueryTest() throws Exception {
        UrlBuilder urlBuilder = new UrlBuilder(new URL("https://google.com/search?q=jsoup"));
        URL url = urlBuilder.build();
        Assert.assertEquals("https://google.com/search?q=jsoup", url.toString());
    }

    @Test
    public void buildUrlWithSpecialCharacterTest() throws Exception {
        UrlBuilder urlBuilder = new UrlBuilder(new URL("https://google.com/search?q=jsoup%20library"));
        URL url = urlBuilder.build();
        Assert.assertEquals("https://google.com/search?q=jsoup+library", url.toString());
    }

    @Test
    public void appendKeyValTestAndBuild() throws Exception {
        UrlBuilder urlBuilder = new UrlBuilder(new URL("https://google.com/search?q=jsoup"));
        urlBuilder.appendKeyVal(new Connection.KeyVal("language", "java"));
        URL url = urlBuilder.build();
        Assert.assertEquals("https://google.com/search?q=jsoup&language=java", url.toString());
    }

    @Test
    public void buildURLWithNonAsciiCharacterTest() throws Exception {
        UrlBuilder urlBuilder = new UrlBuilder(new URL("https://google.com/search?q=jsoup✈️"));
        URL url = urlBuilder.build();
        Assert.assertEquals("https://google.com/search?q=jsoup%E2%9C%88%EF%B8%8F", url.toString());
    }

    @Test
    public void buildUrlWithFragmentTest() throws Exception {
        UrlBuilder urlBuilder = new UrlBuilder(new URL("https://google.com#fragment"));
        URL url = urlBuilder.build();
        Assert.assertEquals("https://google.com#fragment", url.toString());
    }

    @Test(expected = RuntimeException.class)
    public void decodePartWithInvalidEncodingTest() throws Exception {
        String part = new String(new byte[] {(byte)0xC3, (byte)0x28}, "UTF-8");
        UrlBuilder.decodePart(part);
    }
}
