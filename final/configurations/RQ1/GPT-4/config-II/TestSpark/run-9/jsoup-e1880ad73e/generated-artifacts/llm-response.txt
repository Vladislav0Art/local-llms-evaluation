import org.jsoup.Connection;
import org.junit.Test;
import java.io.UnsupportedEncodingException;
import java.net.URL;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNotEquals;

public class UrlBuilderTest {
    
    @Test
    public void UrlBuilderTest() throws Exception {
        URL inputUrl = new URL("http://test.com?query=value");
        UrlBuilder builder = new UrlBuilder(inputUrl);
        assertEquals("http://test.com?query=value", builder.u.toString());
        assertEquals("query=value", builder.q.toString());
    }

    @Test
    public void buildTest() throws Exception {
        URL inputUrl = new URL("http://test.com?query=value");
        UrlBuilder builder = new UrlBuilder(inputUrl);
        builder.build();
        assertEquals("http://test.com?query=value", builder.u.toString());
    }

    @Test
    public void appendKeyValTest() throws Exception {
        URL inputUrl = new URL("http://test.com");
        UrlBuilder builder = new UrlBuilder(inputUrl);
        builder.appendKeyVal(new Connection.KeyVal("key", "value"));
        assertEquals("key=value", builder.q.toString());
    }

    @Test(expected = UnsupportedEncodingException.class)
    public void appendKeyValTestWithStringEncoding() throws Exception {
        URL inputUrl = new URL("http://test.com");
        UrlBuilder builder = new UrlBuilder(inputUrl);
        builder.appendKeyVal(new Connection.KeyVal("key", "\uD83D\uDE00"));
    }

    @Test
    public void appendToAsciiTest() throws Exception {
        URL inputUrl = new URL("http://test.com?query=value");
        UrlBuilder builder = new UrlBuilder(inputUrl);
        StringBuilder sb = new StringBuilder();
        UrlBuilder.appendToAscii("test + char", true, sb);
        assertEquals("test+%20char", sb.toString());
    }

    @Test
    public void appendToAsciiTestWithoutSpaceAsPlus() throws Exception {
        URL inputUrl = new URL("http://test.com?query=value");
        UrlBuilder builder = new UrlBuilder(inputUrl);
        StringBuilder sb = new StringBuilder();
        UrlBuilder.appendToAscii("test + char", false, sb);
        assertEquals("test%20+%20char", sb.toString());
    }
}
