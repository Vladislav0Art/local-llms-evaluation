import org.davidmoten.text.utils.WordWrap;
import org.davidmoten.text.utils.WordWrap.Builder;
import org.junit.Assert;
import org.junit.Test;

import java.io.StringReader;
import java.nio.charset.StandardCharsets;
import java.util.HashSet;
import java.util.Set;

public class WordWrapTest {

    @Test
    public void fromClasspathUtf8Test() {
        Builder builder = WordWrap.fromClasspathUtf8("src/test/resources/test.txt");
        String wrapResult = builder.wrap();

        Assert.assertNotNull(wrapResult);
    }

    @Test
    public void fromFileCharsetTest() {
        Builder builder = WordWrap.from(new File("src/test/resources/test.txt"), StandardCharsets.UTF_8);
        String wrapResult = builder.wrap();

        Assert.assertNotNull(wrapResult);
    }

    @Test(expected = IORuntimeException.class)
    public void fromFileCharsetTestWhenFileNotFound() {
        WordWrap.from(new File("NotExisting/Nonexisting.txt"), StandardCharsets.UTF_8);
    }

    @Test
    public void builderMaxWidthTest() {
        Builder builder = WordWrap.from(new StringReader("test width"));
        builder.maxWidth(100);
        String wrapResult = builder.wrap();

        Assert.assertNotNull(wrapResult);
    }

    @Test(expected = IllegalArgumentException.class)
    public void builderMaxWidthTestWhenMaxWidthIsZero() {
        Builder builder = getDefaultBuilder();
        builder.maxWidth(0);
    }

    @Test
    public void builderStringWidthTest() {
        Builder builder = getDefaultBuilder();
        builder.stringWidth(s -> s.length() * 2);

        Assert.assertEquals("test width", builder.wrap());
    }

    @Test
    public void builderNewLineTest() {
        Builder builder = getDefaultBuilder();
        builder.newLine("test new line");

        Assert.assertEquals("test width", builder.wrap());
    }

    @Test
    public void builderExtraWordCharsSetTest() {
        Builder builder = getDefaultBuilder();
        Set<Character> extraWordChars = new HashSet<>();
        extraWordChars.add('_');
        builder.extraWordChars(extraWordChars);

        Assert.assertEquals("test width", builder.wrap());
    }

    @Test
    public void builderExtraWordCharsStringTest() {
        Builder builder = getDefaultBuilder();
        builder.extraWordChars("testString");

        Assert.assertEquals("test width", builder.wrap());
    }

    @Test
    public void builderIncludeExtraWordCharsTest() {
        Builder builder = getDefaultBuilder();
        builder.includeExtraWordChars("include");

        Assert.assertEquals("test width", builder.wrap());
    }

    @Test
    public void builderExcludeExtraWordCharsTest() {
        Builder builder = getDefaultBuilder();
        builder.excludeExtraWordChars("exclude");

        Assert.assertEquals("test width", builder.wrap());
    }

    @Test
    public void builderInsertHyphensTest() {
        Builder builder = getDefaultBuilder();
        builder.insertHyphens(true);

        Assert.assertEquals("test width", builder.wrap());
    }

    @Test
    public void builderBreakWordsTest() {
        Builder builder = getDefaultBuilder();
        builder.breakWords(true);

        Assert.assertEquals("test width", builder.wrap());
    }

    private Builder getDefaultBuilder() {
        return WordWrap.from(new StringReader("test width"));
    }
}
