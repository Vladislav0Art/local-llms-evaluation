import io.github.vmzakharov.ecdataframe.dsl.*;
import io.github.vmzakharov.ecdataframe.dsl.value.DecimalValue;
import io.github.vmzakharov.ecdataframe.dsl.value.LongValue;
import io.github.vmzakharov.ecdataframe.dsl.visitor.PrettyPrintVisitor;
import io.github.vmzakharov.ecdataframe.dsl.visitor.UnaryOpString;
import io.github.vmzakharov.ecdataframe.util.CollectingPrinter;
import org.eclipse.collections.impl.factory.Lists;
import org.junit.Assert;
import org.junit.Test;

public class PrettyPrintVisitorTest {
  
    @Test
    public void exprToStringTest() {
        AliasExpr aliasExpr = new AliasExpr(new VarExpr("varExpr1"), "aliasExpr1");
        String prettyPrint = PrettyPrintVisitor.exprToString(aliasExpr);
        Assert.assertEquals("aliasExpr1 : varExpr1", prettyPrint);
    }

    @Test
    public void visitAssignExprTest() {
        AssignExpr assignExpr = new AssignExpr("var1", new LongValue(123L), false);
        PrettyPrintVisitor prettyPrintVisitor = new PrettyPrintVisitor(new CollectingPrinter());
        assignExpr.accept(prettyPrintVisitor);
        Assert.assertEquals("var1 = 123", prettyPrintVisitor.toString());
    }

    @Test
    public void visitBinaryExprTest() {
        BinaryExpr binaryExpr = new BinaryExpr(new LongValue(123L), new DecimalValue(456.789D), BinaryOp.MUL);
        String prettyPrint = PrettyPrintVisitor.exprToString(binaryExpr);
        Assert.assertEquals("(123 * 456.789)", prettyPrint);
    }

    @Test
    public void visitUnaryExprTest() {
        UnaryExpr unaryExpr = new UnaryExpr(UnaryOpString.ADD, new LongValue(13L));
        String prettyPrint = PrettyPrintVisitor.exprToString(unaryExpr);
        Assert.assertEquals("+ ( 13 )", prettyPrint);

        unaryExpr = new UnaryExpr(UnaryOpString.SUB, new LongValue(15L));
        prettyPrint = PrettyPrintVisitor.exprToString(unaryExpr);
        Assert.assertEquals("- ( 15 )", prettyPrint);
    }

    @Test
    public void visitFunctionCallExprTest() {
        FunctionCallExpr functionCallExpr = new FunctionCallExpr("functionName1", Lists.mutable.of(new LongValue(246L), new DecimalValue(741.852D)));
        String prettyPrint = PrettyPrintVisitor.exprToString(functionCallExpr);
        Assert.assertEquals("functionName1(246, 741.852)", prettyPrint);
    }

    @Test
    public void visitPropertyPathExprTest() {
        PropertyPathExpr propertyPathExpr = new PropertyPathExpr("path1.path2.path3");
        String prettyPrint = PrettyPrintVisitor.exprToString(propertyPathExpr);
        Assert.assertEquals("path1.path2.path3", prettyPrint);
    }

    @Test
    public void visitProjectionExprTest() {
        ProjectionExpr projectionExpr = new ProjectionExpr(Lists.mutable.of(new LongValue(369L)));
        String prettyPrint = PrettyPrintVisitor.exprToString(projectionExpr);
        Assert.assertEquals("project {369}", prettyPrint);
    }

    @Test
    public void visitAliasExprTest() {
        AliasExpr aliasExpr = new AliasExpr(new LongValue(123L), "alias1");
        String prettyPrint = PrettyPrintVisitor.exprToString(aliasExpr);
        Assert.assertEquals("alias1 : 123", prettyPrint);
    }

    @Test
    public void visitVectorExprTest() {
        VectorExpr vectorExpr = new VectorExpr(Lists.mutable.of(new LongValue(123L), new LongValue(456L)));
        String prettyPrint = PrettyPrintVisitor.exprToString(vectorExpr);
        Assert.assertEquals("(123, 456)", prettyPrint);
    }

    @Test
    public void visitIndexExprTest() {
        IndexExpr indexExpr = new IndexExpr(new VarExpr("vectorVar1"), new LongValue(5L));
        String prettyPrint = PrettyPrintVisitor.exprToString(indexExpr);
        Assert.assertEquals("vectorVar1[5]", prettyPrint);
    }

    @Test
    public void visitDecimalExprTest() {
        DecimalExpr decimalExpr = new DecimalExpr(new DecimalValue(741852963.12345D), new LongValue(2L));
        String prettyPrint = PrettyPrintVisitor.exprToString(decimalExpr);
        Assert.assertEquals("[741852963.12345, 2]", prettyPrint);
    }

    @Test
    public void visitIfElseExprTest() {
        IfElseExpr ifElseExpr = new IfElseExpr(new LongValue(11L), new VarExpr("ifVar1"), new VarExpr("elseVar1"), true);
        String prettyPrint = PrettyPrintVisitor.exprToString(ifElseExpr);
        Assert.assertEquals("11 ? ifVar1 : elseVar1", prettyPrint);
    }
}
