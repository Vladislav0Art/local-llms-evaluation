import io.github.vmzakharov.ecdataframe.util.CollectingPrinter;
import org.junit.Test;

import static org.junit.Assert.assertEquals;

public class PrettyPrintVisitorTest {

    @Test
    public void exprToStringScenarioTest() {
        AnonymousScript anonymousScript = new AnonymousScript();
        anonymousScript.addExpression(new VarExpr("var1"));

        String result = PrettyPrintVisitor.exprToString(anonymousScript);

        assertEquals("var1", result);
    }

    @Test
    public void visitAssignExprScenarioTest() {
        AssingExpr assingExpr = new AssingExpr("var1", new VarExpr("var2"));
        CollectingPrinter printer = new CollectingPrinter();
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(printer);

        visitor.visitAssignExpr(assingExpr);

        assertEquals("var1 = var2", printer.toString());
    }

    @Test
    public void visitBinaryExprScenarioTest() {
        BinaryExpr binaryExpr = new BinaryExpr(new VarExpr("var1"), "+", new VarExpr("var2"));
        CollectingPrinter printer = new CollectingPrinter();
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(printer);

        visitor.visitBinaryExpr(binaryExpr);

        assertEquals("(var1 + var2)", printer.toString());
    }

    @Test
    public void visitUnaryExprScenarioTest() {
        UnaryExpr unaryExpr = new UnaryExpr("-", new VarExpr("var1"));
        CollectingPrinter printer = new CollectingPrinter();
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(printer);

        visitor.visitUnaryExpr(unaryExpr);

        assertEquals("-(var1)", printer.toString());
    }

    @Test
    public void visitConstExprScenarioTest() {
        Value value = new Value(10);
        CollectingPrinter printer = new CollectingPrinter();
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(printer);

        visitor.visitConstExpr(value);

        assertEquals("10", printer.toString());
    }

    @Test
    public void visitFunctionCallExprScenarioTest() {
        FunctionCallExpr functionCallExpr = new FunctionCallExpr("func");
        functionCallExpr.addParameter(new VarExpr("var1"));
        CollectingPrinter printer = new CollectingPrinter();
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(printer);

        visitor.visitFunctionCallExpr(functionCallExpr);

        assertEquals("func(var1)", printer.toString());
    }

}
