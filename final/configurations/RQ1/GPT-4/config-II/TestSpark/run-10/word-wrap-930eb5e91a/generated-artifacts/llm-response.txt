import java.io.*;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

import org.davidmoten.text.utils.WordWrap;
import org.davidmoten.text.utils.WordWrap.Builder;
import org.junit.Test;

import static org.junit.Assert.*;

public class WordWrapTest {

    @Test
    public void fromReaderTest() {
        Reader reader = new StringReader("sample text");
        Builder builder = WordWrap.from(reader);
        assertNotNull(builder);
    }

    @Test
    public void fromCharSequenceTest() {
        CharSequence text = "sample text";
        Builder builder = WordWrap.from(text);
        assertNotNull(builder);
    }

    @Test
    public void fromUtf8StreamTest() {
        InputStream inputStream = new ByteArrayInputStream("sample text".getBytes());
        Builder builder = WordWrap.fromUtf8(inputStream);
        assertNotNull(builder);
    }

    @Test
    public void fromFileTest() throws FileNotFoundException {
        File file = new File("./src/test/resources/file.txt");
        Builder builder = WordWrap.from(file, StandardCharsets.UTF_8);
        assertNotNull(builder);
    }

    @Test(expected = IORuntimeException.class)
    public void fromFileNotFoundTest() {
        File file = new File("fileNotFound.txt");
        Builder builder = WordWrap.from(file, StandardCharsets.UTF_8);
    }

    @Test
    public void BuilderExtraWordCharsSetTest() {
        Set<Character> set = new HashSet<Character>() {
            {
                add('s');
            }
        };
        WordWrap.Builder builder = new WordWrap.Builder(new StringReader(""), false)
                .extraWordChars(set);
        assertNotNull(builder);
    }

    @Test
    public void BuilderExtraWordCharsStringTest() {
        WordWrap.Builder builder = new WordWrap.Builder(new StringReader(""), false)
                .extraWordChars("s");
        assertNotNull(builder);
    }

    @Test
    public void BuilderIncludeExtraWordCharsTest() {
        WordWrap.Builder builder = new WordWrap.Builder(new StringReader(""), false)
                .includeExtraWordChars("s");
        assertNotNull(builder);
    }

    @Test
    public void BuilderExcludeExtraWordCharsTest() {
        WordWrap.Builder builder = new WordWrap.Builder(new StringReader(""), false)
                .excludeExtraWordChars("s");
        assertNotNull(builder);
    }

    @Test
    public void BuilderInsertHyphensTest() {
        WordWrap.Builder builder = new WordWrap.Builder(new StringReader(""), false)
                .insertHyphens(true);
        assertNotNull(builder);
    }
    
    @Test
    public void BuilderBreakWordsTest() {
        WordWrap.Builder builder = new WordWrap.Builder(new StringReader(""), false)
                .breakWords(true);
        assertNotNull(builder);
    }

    @Test
    public void BuilderWrapTest() {
        String result = new WordWrap.Builder(new StringReader("sample text"), false)
                .wrap();
        assertNotNull(result);
    }

    @Test
    public void BuilderWrapToListTest() {
        List<String> result = new WordWrap.Builder(new StringReader("sample text"), false)
                .wrapToList();
        assertNotNull(result);
    }
}
