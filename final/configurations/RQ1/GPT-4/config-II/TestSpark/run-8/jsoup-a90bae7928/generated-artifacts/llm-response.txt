import org.jsoup.helper.Validate;
import org.jsoup.nodes.*;
import org.jsoup.parser.*;
import org.junit.Assert;
import org.junit.Test;

import java.io.StringReader;

public class XmlTreeBuilderTest {

    @Test
    public void parseGivenInputAndBaseUriTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        Document result = xmlTreeBuilder.parse("<tag>", "http://base.uri");
        Assert.assertEquals("http://base.uri", result.baseUri());
    }

    @Test
    public void parseGivenReaderInputAndBaseUriTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        Document result = xmlTreeBuilder.parse(new StringReader("<tag>"), "http://base.uri");
        Assert.assertEquals("http://base.uri", result.baseUri());
    }

    @Test
    public void newInstanceTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        Assert.assertNotNull(xmlTreeBuilder.newInstance());
    }

    @Test(expected = IllegalArgumentException.class)
    public void processGivenInvalidTokenTypeTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        xmlTreeBuilder.process(new Token.Reset());
    }

    @Test
    public void insertNodeTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        xmlTreeBuilder.initialiseParse(new StringReader("<tag>"), "http://base.uri", new Parser(xmlTreeBuilder));
        xmlTreeBuilder.insertNode(new TextNode("textnode"));
        Assert.assertEquals("textnode", xmlTreeBuilder.doc.text());
    }

    @Test
    public void insertStartTagTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        xmlTreeBuilder.initialiseParse(new StringReader("<tag>"), "http://base.uri", new Parser(xmlTreeBuilder));
        xmlTreeBuilder.insert(new Token.StartTag("tag"));
        Assert.assertEquals("tag", xmlTreeBuilder.doc.text());
    }

    @Test
    public void insertCommentTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        xmlTreeBuilder.initialiseParse(new StringReader("<tag>"), "http://base.uri", new Parser(xmlTreeBuilder));
        xmlTreeBuilder.insert(new Token.Comment("comment", true));
        Assert.assertEquals("", xmlTreeBuilder.doc.text());
    }

    @Test
    public void insertCharacterTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        xmlTreeBuilder.initialiseParse(new StringReader("<tag c=\"character\">"), "http://base.uri",
                new Parser(xmlTreeBuilder));
        xmlTreeBuilder.insert(new Token.Character("character", true));
        Assert.assertEquals("", xmlTreeBuilder.doc.text());
    }

    @Test
    public void insertDoctypeTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        xmlTreeBuilder.initialiseParse(new StringReader("<!DOCTYPE html>"), "http://base.uri",
                new Parser(xmlTreeBuilder));
        xmlTreeBuilder.insert(new Token.Doctype("html"));
        Assert.assertEquals("", xmlTreeBuilder.doc.text());
    }

    @Test
    public void popStackToCloseTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        xmlTreeBuilder.initialiseParse(new StringReader("<tag></tag>"), "http://base.uri",
                new Parser(xmlTreeBuilder));
        xmlTreeBuilder.insert(new Token.StartTag("tag"));
        xmlTreeBuilder.popStackToClose(new Token.EndTag("tag"));
        Assert.assertEquals("", xmlTreeBuilder.doc.text());
    }

    @Test
    public void parseFragmentTest() {
        XmlTreeBuilder xmlTreeBuilder = new XmlTreeBuilder();
        Element element = new Element(new Tag("tag"), "http://base.uri");
        Assert.assertNotNull(xmlTreeBuilder.parseFragment("<tag>", element, "http://base.uri", new Parser(xmlTreeBuilder)));
    }
}
