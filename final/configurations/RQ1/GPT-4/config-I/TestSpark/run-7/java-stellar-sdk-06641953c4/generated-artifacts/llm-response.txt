```java
import net.i2p.crypto.eddsa.EdDSAPrivateKey;
import net.i2p.crypto.eddsa.EdDSAPublicKey;
import net.i2p.crypto.eddsa.KeyPairGenerator;
import org.junit.Test;
import org.mockito.Mockito;
import org.stellar.sdk.*;

import java.security.KeyPair;
import java.security.Security;
import java.util.Arrays;

import static org.junit.Assert.*;

public class KeyPairTest {

    @Test
    public void KeyPairConstructorWithTwoParamsTest() {
        EdDSAPublicKey publicKey = Mockito.mock(EdDSAPublicKey.class);
        EdDSAPrivateKey privateKey = Mockito.mock(EdDSAPrivateKey.class);
        org.stellar.sdk.KeyPair keyPair = new org.stellar.sdk.KeyPair(publicKey, privateKey);
        assertNotNull(keyPair);
    }

    @Test
    public void canSignTest() {
        java.security.KeyPair keypair = new KeyPairGenerator().generateKeyPair();
        org.stellar.sdk.KeyPair stellarKeyPair = org.stellar.sdk.KeyPair.fromSecretSeed(((EdDSAPrivateKey) keypair.getPrivate()).getSeed());
        assertTrue(stellarKeyPair.canSign());
    }

    @Test
    public void fromSecretSeedWithBytesTest() {
        byte[] bytes = {0, 1, 0, 1, 0, 1, 0, 1};
        org.stellar.sdk.KeyPair keyPair = org.stellar.sdk.KeyPair.fromSecretSeed(bytes);
        assertNotNull(keyPair);
    }

    @Test
    public void fromPublicKeyTest() {
        byte[] bytes = {0, 1, 0, 1, 0, 1, 0, 1};
        org.stellar.sdk.KeyPair keyPair = org.stellar.sdk.KeyPair.fromPublicKey(bytes);
        assertNotNull(keyPair);
    }

    @Test
    public void getXdrPublicKeyTest() {
        byte[] bytes = {0, 1, 0, 1, 0, 1, 0, 1};
        org.stellar.sdk.KeyPair keyPair = org.stellar.sdk.KeyPair.fromPublicKey(bytes);
        assertTrue(keyPair.getXdrPublicKey() instanceof org.stellar.sdk.xdr.PublicKey);
    }

    @Test(expected = RuntimeException.class)
    public void signTest() {
        byte[] bytes = {0, 1, 0, 1, 0, 1, 0, 1};
        org.stellar.sdk.KeyPair keyPair = org.stellar.sdk.KeyPair.fromPublicKey(bytes);
        keyPair.sign(bytes);
    }

    @Test
    public void verifyTest() {
        java.security.KeyPair keypair = new KeyPairGenerator().generateKeyPair();
        org.stellar.sdk.KeyPair stellarKeyPair = org.stellar.sdk.KeyPair.fromSecretSeed(((EdDSAPrivateKey) keypair.getPrivate()).getSeed());
        byte[] data = "TestData".getBytes();
        byte[] signature = stellarKeyPair.sign(data);
        assertTrue(stellarKeyPair.verify(data, signature));
    }

    @Test
    public void equalsTest() {
        java.security.KeyPair keypair1 = new KeyPairGenerator().generateKeyPair();
        org.stellar.sdk.KeyPair stellarKeyPair1 = org.stellar.sdk.KeyPair.fromSecretSeed(((EdDSAPrivateKey) keypair1.getPrivate()).getSeed());

        java.security.KeyPair keypair2 = new KeyPairGenerator().generateKeyPair();
        org.stellar.sdk.KeyPair stellarKeyPair2 = org.stellar.sdk.KeyPair.fromSecretSeed(((EdDSAPrivateKey) keypair2.getPrivate()).getSeed());

        assertFalse(stellarKeyPair1.equals(stellarKeyPair2));
        assertTrue(stellarKeyPair1.equals(stellarKeyPair1));
    }
}
```

```java
import net.i2p.crypto.eddsa.EdDSAPrivateKey;
import net.i2p.crypto.eddsa.EdDSAPublicKey;
import net.i2p.crypto.eddsa.KeyPairGenerator;
import org.junit.Test;

import java.security.GeneralSecurityException;

import static org.junit.Assert.*;

public class KeyPairTest {

    @Test
    public void KeyPairConstructorWithPublicKeyTest() {
        EdDSAPublicKey publicKey = new EdDSAPublicKey(new byte[32]);
        org.stellar.sdk.KeyPair keyPair = new org.stellar.sdk.KeyPair(publicKey);

        assertNotNull(keyPair);
        assertFalse(keyPair.canSign());
    }

    @Test
    public void KeyPairConstructorWithBothKeysTest() {
        EdDSAPublicKey publicKey = new EdDSAPublicKey(new byte[32]);
        EdDSAPrivateKey privateKey = new EdDSAPrivateKey(new byte[64]);
        org.stellar.sdk.KeyPair keyPair = new org.stellar.sdk.KeyPair(publicKey, privateKey);

        assertNotNull(keyPair);
        assertTrue(keyPair.canSign());
    }

    @Test
    public void fromSecretSeedCharTest() {
        org.stellar.sdk.KeyPair keyPair = org.stellar.sdk.KeyPair.fromSecretSeed(new char[]{});

        assertNotNull(keyPair);
    }

    @Test
    public void fromSecretSeedStringTest() {
        org.stellar.sdk.KeyPair keyPair = org.stellar.sdk.KeyPair.fromSecretSeed("");

        assertNotNull(keyPair);
    }

    @Test
    public void fromSecretSeedBytesTest() {
        org.stellar.sdk.KeyPair keyPair = org.stellar.sdk.KeyPair.fromSecretSeed(new byte[32]);

        assertNotNull(keyPair);
    }

    public void fromPublicKeyWithInvalidKeyTest() {
        org.stellar.sdk.KeyPair.fromPublicKey(new byte[31]);
    }

    @Test

        assertNotNull(keyPair);
    }

    public void randomTest() {
        org.stellar.sdk.KeyPair keyPair = org.stellar.sdk.KeyPair.random();

        assertNotNull(keyPair);
    }

        assertEquals(32, keyPair.getPublicKey().length);

        byte[] data = new byte[0];
        byte[] signature = keyPair.sign(data);

        assertNotNull(signature);
    }

    public void signWithoutPrivateKeyTest() {

        keyPair.sign(data);
    }

        org.stellar.sdk.KeyPair keyPair = org.stellar.sdk.KeyPair.random();

        byte[] data = new byte[0];

        assertFalse(keyPair.verify(data, new byte[64]));
    }
```

```java
import org.junit.Test;
import org.stellar.sdk.xdr.PublicKey;

import static org.junit.Assert.assertTrue;


public class KeyPairTest {

    @Test(expected = RuntimeException.class)
    public void fromPublicKeyWithInvalidKeyTest() {
        org.stellar.sdk.KeyPair.fromPublicKey(new byte[31]);
    }

    @Test
    public void fromPublicKeyTest() {
        org.stellar.sdk.KeyPair keyPair = org.stellar.sdk.KeyPair.fromPublicKey(new byte[32]);
        assertTrue(keyPair instanceof org.stellar.sdk.KeyPair);
    }

    @Test
    public void fromSecretSeedStringTest() {
        String seedString = "SB6MJOG3ERDBV34AZI6WFS6Y2JI4W3LDMX3GCLVZ526536JZ3SC54ZPA";
        org.stellar.sdk.KeyPair keyPair = org.stellar.sdk.KeyPair.fromSecretSeed(seedString);
        assertTrue(keyPair instanceof org.stellar.sdk.KeyPair);
    }

    @Test
    public void fromSecretSeedTest() {
        char[] seed = "SB6MJOG3ERDBV34AZI6WFS6Y2JI4W3LDMX3GCLVZ526536JZ3SC54ZPA".toCharArray();
        org.stellar.sdk.KeyPair keyPair = org.stellar.sdk.KeyPair.fromSecretSeed(seed);
        assertTrue(keyPair instanceof org.stellar.sdk.KeyPair);
    }

    @Test
    public void fromXdrPublicKeyTest() {
        PublicKey publicKey = new PublicKey();
        publicKey.setDiscriminant(org.stellar.sdk.xdr.PublicKeyType.PUBLIC_KEY_TYPE_ED25519);
        org.stellar.sdk.xdr.Uint256 uint256 = new org.stellar.sdk.xdr.Uint256();
        uint256.setUint256(new byte[32]);
        publicKey.setEd25519(uint256);

        org.stellar.sdk.KeyPair keyPair = org.stellar.sdk.KeyPair.fromXdrPublicKey(publicKey);
        assertTrue(keyPair instanceof org.stellar.sdk.KeyPair);
    }

    @Test
    public void randomTest() {
        org.stellar.sdk.KeyPair keyPair = org.stellar.sdk.KeyPair.random();
        assertTrue(keyPair instanceof org.stellar.sdk.KeyPair);
    }

}
```
