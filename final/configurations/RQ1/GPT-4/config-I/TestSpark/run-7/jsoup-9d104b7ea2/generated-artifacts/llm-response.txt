package org.jsoup.nodes;

import org.jsoup.nodes.Document.OutputSettings;
import org.junit.Test;
import java.io.IOException;

import static org.junit.Assert.*;

public class TextNodeTest {

    @Test
    public void textNodeBlankTest() {
        TextNode node = new TextNode(" ");
        assertEquals(node.text(), " ");
        assertTrue(node.isBlank());
    }

    @Test
    public void textNodeWellFormedTest() {
        TextNode node = new TextNode("Hello World");
        assertEquals(node.text(), "Hello World");
        assertFalse(node.isBlank());
    }

    @Test
    public void nodeNameTest() {
        TextNode node = new TextNode("Node");
        assertEquals(node.nodeName(), "#text");
    }

    @Test
    public void nodeSetTextTest() {
        TextNode node = new TextNode("Old Text");
        assertEquals(node.text(), "Old Text");
        node.text("New Text");
        assertEquals(node.text(), "New Text");
    }

    @Test
    public void getWholeTextTest() {
        TextNode node = new TextNode("Whole Text");
        assertEquals(node.getWholeText(), "Whole Text");
    }

    @Test
    public void splitTextValidOffset() {
        TextNode node = new TextNode("Node");
        TextNode splitNode = node.splitText(2);
        assertEquals(node.text(), "No");
        assertEquals(splitNode.text(), "de");
    }

    @Test(expected = IllegalArgumentException.class)
    public void splitTextInvalidNegativeOffset() {
        TextNode node = new TextNode("Node");
        node.splitText(-1);
    }

    @Test(expected = IllegalArgumentException.class)
    public void splitTextInvalidOffsetGreaterThanLength() {
        TextNode node = new TextNode("Node");
        node.splitText(5);
    }

    @Test
    public void outerHtmlHeadTest() throws IOException {
        TextNode node = new TextNode("Hello");
        StringBuilder accum = new StringBuilder();
        node.outerHtmlHead(accum, 0, new OutputSettings());
        assertEquals(accum.toString(), "Hello");
    }

    @Test
    public void outerHtmlTailTest() throws IOException {
        TextNode node = new TextNode("Hello");
        StringBuilder accum = new StringBuilder();
        node.outerHtmlTail(accum, 0, new OutputSettings());
        assertEquals(accum.toString(), "");
    }

    @Test
    public void cloneTest() {
        TextNode node = new TextNode("Hello");
        TextNode clonedNode = node.clone();
        assertNotNull(clonedNode);
        assertNotSame(node, clonedNode);
        assertEquals(node.text(), clonedNode.text());
    }

    @Test
    public void createFromEncodedTest() {
        TextNode node = TextNode.createFromEncoded("Hello &amp; World");
        assertEquals(node.getWholeText(), "Hello & World");
    }

    @Test
    public void normaliseWhitespaceTest() {
        String normalisedText = TextNode.normaliseWhitespace(" Hello    World ");
        assertEquals(normalisedText, "Hello World");
    }

    @Test
    public void stripLeadingWhitespaceTest() {
        String strippedText = TextNode.stripLeadingWhitespace("   Hello World ");
        assertEquals(strippedText, "Hello World ");
    }

    @Test
    public void lastCharIsWhitespaceTest() {
        StringBuilder sb = new StringBuilder("Hello World ");
        assertTrue(TextNode.lastCharIsWhitespace(sb));
        sb = new StringBuilder("Hello World");
        assertFalse(TextNode.lastCharIsWhitespace(sb));
    }
}
