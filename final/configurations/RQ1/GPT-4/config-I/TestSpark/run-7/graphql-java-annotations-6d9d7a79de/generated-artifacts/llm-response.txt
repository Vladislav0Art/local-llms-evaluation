import static graphql.schema.GraphQLArgument.newArgument;
import static graphql.schema.GraphQLScalarType.newScalar;
import static graphql.schema.GraphQLDirective.newDirective;

import graphql.annotations.annotationTypes.GraphQLField;
import graphql.annotations.annotationTypes.directives.GraphQLDirectiveLocations;
import graphql.annotations.annotationTypes.directives.activation.GraphQLDirectives;
import graphql.annotations.annotationTypes.directives.definition.GraphQLDirectiveDefinition;
import graphql.annotations.processor.ProcessingElementsContainer;
import graphql.annotations.processor.exceptions.GraphQLAnnotationsException;
import graphql.annotations.processor.util.DirectivesUtil;
import graphql.annotations.processor.util.NamingKit;
import graphql.schema.GraphQLArgument;
import graphql.schema.GraphQLDirective;
import graphql.schema.GraphQLScalarType;
import graphql.schema.GraphQLType;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.junit.runners.JUnit4;

import java.lang.reflect.AnnotatedElement;
import java.util.Arrays;
import java.util.HashMap;

import static org.hamcrest.CoreMatchers.notNullValue;
import static org.junit.Assert.assertThat;
import static org.junit.Assert.fail;
import static org.mockito.Mockito.mock;
import static org.mockito.Mockito.when;

@RunWith(JUnit4.class)
public class DirectivesBuilderTests {

    @GraphQLDirectiveDefinition(name = "simpleDirective")
    @GraphQLDirectiveLocations("FIELD_DEFINITION")
    @interface SimpleDirective {

        @GraphQLField
        int intValue();
    }

    @GraphQLDirectives(value = {
            @GraphQLDirectives.Directive(name = "simpleDirective", argumentsValues = {"5"})
    })
    public static class SimpleClass {}

    @Test
    public void buildObjectInDirectiveRegistryTest(){
        AnnotatedElement annotatedElementMock = mock(AnnotatedElement.class);
        ProcessingElementsContainer containerMock = mock(ProcessingElementsContainer.class);
        GraphQLDirective directive = newDirective().name("simpleDirective").argument(newArgument().name("intValue").type(newScalar().build()).build()).build();
        containerMock.getDirectiveRegistry().put("simpleDirective", new DirectivesUtil.DirectiveInfo(directive));
        when(containerMock.getDirectiveRegistry()).thenReturn(containerMock.getDirectiveRegistry());
        when(containerMock.getNamingKit()).thenReturn(new NamingKit());
        when(annotatedElementMock.getAnnotations()).thenReturn(SimpleClass.class.getAnnotations());
        when(annotatedElementMock.getAnnotation(GraphQLDirectives.class)).thenReturn(Arrays.stream(SimpleClass.class.getAnnotations()).filter(annotation -> annotation instanceof GraphQLDirectives).map(annotation -> ((GraphQLDirectives) annotation)).findFirst().orElse(null));

        DirectivesBuilder directivesBuilder = new DirectivesBuilder(annotatedElementMock, containerMock);

        boolean exceptionOccured = false;
        try {
            GraphQLDirective[] result = directivesBuilder.build();
            assertThat(result, notNullValue());
            assertThat(result[0], notNullValue());
        } catch (GraphQLAnnotationsException ex){
            exceptionOccured = true;
        }
        assert !exceptionOccured;
    }

}
