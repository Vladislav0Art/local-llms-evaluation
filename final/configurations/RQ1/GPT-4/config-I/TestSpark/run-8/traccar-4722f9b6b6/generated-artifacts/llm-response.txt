        position = new Position();
        position.setProtocol("watch");
    }

    @Test
    public void decodeAlarmTest() {
        assertNull(decoder.decodeAlarm(23));
        assertEquals(Position.ALARM_LOW_BATTERY, decoder.decodeAlarm(1));

        assertEquals(Position.ALARM_SOS, decoder.decodeAlarm(65_536));
