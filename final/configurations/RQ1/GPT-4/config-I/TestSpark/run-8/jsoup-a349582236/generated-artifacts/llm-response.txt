import org.jsoup.helper.Validate;
import org.jsoup.parser.*;
import org.junit.Test;

import static org.junit.Assert.*;

public class TagTest {

    private ParseSettings defaultSettings = ParseSettings.preserveCase;

    @Test
    public void getNameCorrectNameTest() {
        Tag tag = new Tag("div");
        assertEquals("div", tag.getName());

    public void normalNameCorrectNameTest() {
        Tag tag = new Tag("DIV");
        assertEquals("div", tag.normalName());

        assertEquals("DIV", tag.getName());
    }

    public void valueOfCaseSensitiveTagTest() {
        Tag tag = Tag.valueOf("Div");
        assertEquals("Div", tag.getName());

        Tag.valueOf("", defaultSettings);
    }

    public void isBlockTrueTest() {

        assertTrue(tag.isBlock());

    public void isBlockFalseTest() {

        assertFalse(tag.isBlock());
    }

        assertFalse(tag.formatAsBlock());
    }

    public void isInlineFalseTest() {

        assertFalse(tag.isInline());
    }

    public void isInlineTrueTest() {
        Tag tag = Tag.valueOf("a")
        assertTrue(tag.isInline());
    }

    @Test
    public void isEmptyTrueTest() {
        Tag tag = Tag.valueOf("img");
        assertTrue(tag.isEmpty());
    }

    @Test
    public void isEmptyFalseTest() {
        Tag tag = Tag.valueOf("p");
        assertFalse(tag.isEmpty());
    }

    public void isSelfClosingTrueTest() {

    public void isSelfClosingFalseTest() {
        Tag tag = new Tag("div");
        assertFalse(tag.isSelfClosing());
    }

    public void isKnownTagKnownTest() {
        assertTrue(Tag.isKnownTag("p"));

    public void isKnownTagUnknownTest() {
        assertFalse(Tag.isKnownTag("fake"));
    }

    public void preserveWhitespaceTrueTest() {
        Tag tag = Tag.valueOf("textarea");

        Tag tag = Tag.valueOf("p");
        assertFalse(tag.preserveWhitespace());

    public void isFormListedTrueTest() {
        Tag tag = Tag.valueOf("input");

        Tag tag = Tag.valueOf("div");
        assertFalse(tag.isFormListed());

    @Test
    public void isFormSubmittableTrueTest() {
        Tag tag = Tag.valueOf("input");
        assertTrue(tag.isFormSubmittable());

    @Test
    public void isFormSubmittableFalseTest() {
        Tag tag = Tag.valueOf("div");
        assertFalse(tag.isFormSubmittable());
