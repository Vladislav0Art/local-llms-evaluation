package net.revelc.code.formatter.css;

import static org.junit.Assert.*;

import org.junit.Test;
import java.util.HashMap;
import java.util.Map;

import net.revelc.code.formatter.ConfigurationSource;
import net.revelc.code.formatter.LineEnding;

public class CssFormatterTest {

    @Test
    public void initDefaultTest() {
        CssFormatter formatter = new CssFormatter();
        assertFalse(formatter.isInitialized());
        ConfigurationSource configSource = new ConfigurationSource.Std();
        Map<String, String> options = new HashMap<>();
        formatter.init(options, configSource);
        assertTrue(formatter.isInitialized());
    }

    @Test
    public void initNonDefaultTest() {
        CssFormatter formatter = new CssFormatter();
        assertFalse(formatter.isInitialized());
        ConfigurationSource configSource = new ConfigurationSource.Std();
        Map<String, String> options = new HashMap<>();
        options.put("indent", "6");
        options.put("rgbAsHex", "false");
        options.put("useSourceStringValues", "true");
        formatter.init(options, configSource);
        assertTrue(formatter.isInitialized());
    }

    @Test
    public void doFormatNullTest() throws IOException {
        CssFormatter formatter = new CssFormatter();
        ConfigurationSource configSource = new ConfigurationSource.Std();
        Map<String, String> options = new HashMap<>();
        formatter.init(options, configSource);
        String code = "\t;";
        String formattedCode = formatter.doFormat(code, LineEnding.AUTO);
        assertNull(formattedCode);
    }

    @Test
    public void doFormatNotNullTest() throws IOException {
        CssFormatter formatter = new CssFormatter();
        ConfigurationSource configSource = new ConfigurationSource.Std();
        Map<String, String> options = new HashMap<>();
        formatter.init(options, configSource);
        String code = ".example { color: red }";
        String formattedCode = formatter.doFormat(code, LineEnding.AUTO);
        assertNotNull(formattedCode);
    }

    @Test
    public void isInitializedFalseTest() {
        CssFormatter formatter = new CssFormatter();
        assertFalse(formatter.isInitialized());
    }
}
