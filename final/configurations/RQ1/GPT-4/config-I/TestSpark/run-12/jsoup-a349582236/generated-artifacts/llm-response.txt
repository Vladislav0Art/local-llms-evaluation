import static org.junit.Assert.*;

import org.jsoup.parser.Tag;
import org.jsoup.parser.ParseSettings;
import org.junit.Test;

public class TagTest {

    @Test
    public void getNameBlockTagTest() {
        Tag t = Tag.valueOf("html");
        assertEquals("html", t.getName());
    }

    @Test
    public void getNameInlineTagTest() {
        Tag t = Tag.valueOf("mark");
        assertEquals("mark", t.getName());
    }

    @Test
    public void normalNameTest() {
        Tag t = Tag.valueOf("HTML");
        assertEquals("html", t.normalName());
    }

    @Test
    public void valueOfUnknownTagTest() {
        Tag t = Tag.valueOf("unknown");
        assertEquals("unknown", t.getName());
    }
    
    @Test
    public void isBlockTest() {
        Tag t = Tag.valueOf("p");
        assertTrue(t.isBlock());
    }
    
    @Test
    public void isInlineTest() {
        Tag t = Tag.valueOf("img");
        assertTrue(t.isInline());
    }
    
    @Test
    public void isEmptyTest() {
        Tag t = Tag.valueOf("img");
        assertTrue(t.isEmpty());
    }   

    @Test
    public void isSelfClosingTest() {
        Tag t = Tag.valueOf("img");
        assertTrue(t.isSelfClosing());
    }
    
    @Test
    public void isFormListedTest() {
        Tag t = Tag.valueOf("input");
        assertTrue(t.isFormListed());
    }

    @Test
    public void isFormSubmittableTest() {
        Tag t = Tag.valueOf("input");
        assertTrue(t.isFormSubmittable());
    }
    
    @Test
    public void isKnownTagTest() {
        assertTrue(Tag.isKnownTag("p"));
    }

    @Test
    public void formatAsBlockTest() {
        Tag t = Tag.valueOf("title");
        assertFalse(t.formatAsBlock());
    }
    
    @Test
    public void preserveWhitespaceTest() {
        Tag t = Tag.valueOf("pre");
        assertTrue(t.preserveWhitespace());
    }
    
    @Test
    public void cloneTagTest() {
        Tag t1 = Tag.valueOf("p");
        Tag t2 = t1.clone();
        assertNotSame(t1, t2);
        assertEquals(t1, t2);
    }
    
    @Test
    public void valueOfCaseSensitiveTest() {
        Tag t = Tag.valueOf("HTML", ParseSettings.preserveCase);
        assertEquals("HTML", t.getName());
    }

    @Test
    public void equalsTest() {
        Tag t = Tag.valueOf("p");
        Tag tag = t.clone().setSelfClosing();
        assertFalse(t.equals(tag));
    }

    @Test(expected = RuntimeException.class)
    public void testCloneThrowsException() {
        Tag t = new Tag("p") {
            @Override
            public Tag clone() {
                try {
                    throw new CloneNotSupportedException();
                } catch (CloneNotSupportedException e) {
                    throw new RuntimeException(e);
                }
            }
        };
        t.clone();
    }

}
