import graphql.annotations.processor.retrievers.fieldBuilders.DirectivesBuilder;
import graphql.annotations.processor.ProcessingElementsContainer;
import graphql.annotations.processor.exceptions.GraphQLAnnotationsException;
import graphql.schema.GraphQLArgument;
import graphql.schema.GraphQLDirective;
import graphql.schema.GraphQLScalarType;
import graphql.schema.GraphQLTypeReference;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.junit.MockitoJUnitRunner;
import org.mockito.Mock;
import org.mockito.InjectMocks;

import java.util.Collections;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertThrows;
import static org.mockito.Mockito.*;

@RunWith(MockitoJUnitRunner.class)
public class DirectivesBuilderTest {

    @Mock
    private ProcessingElementsContainer container;

    @Mock
    private GraphQLScalarType scalarType;

    @Test(expected = GraphQLAnnotationsException.class)
    public void buildDirectiveNotFoundExceptionTest() {
        DirectivesBuilder builder = new DirectivesBuilder(MyClass1.class, container);
        when(container.getDirectiveRegistry()).thenReturn(Collections.emptyMap());
        builder.build();
    }

    @Test
    public void buildDirectiveSuccessTest() {
        GraphQLDirective directive = GraphQLDirective.newDirective().name("name").argument(GraphQLArgument.newArgument().name("arg1").type(scalarType).value("value1").build()).build();
        DirectivesBuilder builder = new DirectivesBuilder(MyClass2.class, container);
        when(container.getDirectiveRegistry()).thenReturn(Collections.singletonMap("name", new ProcessingElementsContainer.GraphQLDirectiveContainer(directive, "name")));
        GraphQLDirective[] directives = builder.build();
        assertEquals(1, directives.length);
        assertEquals("name", directives[0].getName());
        assertEquals("arg1", directives[0].getArguments().get(0).getName());
        assertEquals("value1", directives[0].getArguments().get(0).getValue());
    }

    @Test
    public void buildDirectiveTooMuchArgumentsTest() {
        GraphQLDirective directive = GraphQLDirective.newDirective().name("dirName").build();
        DirectivesBuilder builder = new DirectivesBuilder(MyClass3.class, container);
        when(container.getDirectiveRegistry()).thenReturn(Collections.singletonMap("dirName", new ProcessingElementsContainer.GraphQLDirectiveContainer(directive, "dirName")));
        Throwable e = assertThrows(GraphQLAnnotationsException.class, () -> builder.build());
        assertEquals("Directive 'dirName' is supplied with more argument values than it supports", e.getMessage());
    }
}

class MyClass2 { 

    String getMyValue() {  
        return "myvalue"; 
    } 
}

import org.junit.Test;

import org.mockito.Mock;
import org.mockito.InjectMocks;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertThrows;
import static org.mockito.Mockito.*;

@RunWith(MockitoJUnitRunner.class)
public class DirectivesBuilderTest {

    @Mock
    private ProcessingElementsContainer container;

    @Test
    public void buildDirectiveNotFoundExceptionTest() {
        DirectivesBuilder builder = new DirectivesBuilder(MyClass1.class, container);
        when(container.getDirectiveRegistry()).thenReturn(Collections.emptyMap());
        Exception exception = assertThrows(GraphQLAnnotationsException.class, builder::build);
        assertEquals("No directive named value is found in the directive registry", exception.getMessage());
    }

    @Test
    public void buildDirectiveSuccessTest() {
        Map<String, GraphQLDirective> directiveRegistry = new HashMap<>();
        GraphQLDirective directive = GraphQLDirective.newDirective()
                .name("MyDirective")
                .build();
        directiveRegistry.put("MyDirective", directive);

        when(container.getDirectiveRegistry()).thenReturn(directiveRegistry);

        DirectivesBuilder builder = new DirectivesBuilder(MyClass2.class, container);
        GraphQLDirective[] directives = builder.build();
        assertEquals(1, directives.length);
        assertEquals("MyDirective", directives[0].getName());
    }

    @Test
    public void buildDirectiveTooMuchArgumentsTest() {
        Map<String, GraphQLDirective> directiveRegistry = new HashMap<>();
        GraphQLDirective directive = GraphQLDirective.newDirective()
                .name("MyDirective")
                .argument(
                        GraphQLArgument.newArgument()
                                .name("arg1")
                                .type(GraphQLTypeReference.typeRef("String"))
                                .defaultValue("default")
                                .build())
                .build();
        directiveRegistry.put("MyDirective", directive);

        when(container.getDirectiveRegistry()).thenReturn(directiveRegistry);

        DirectivesBuilder builder = new DirectivesBuilder(MyClass3.class, container);
        GraphQLAnnotationsException exception = assertThrows(GraphQLAnnotationsException.class, builder::build);
        assertEquals("Directive 'MyDirective' is supplied with more argument values than it supports", exception.getMessage());
    }

}

}

}
