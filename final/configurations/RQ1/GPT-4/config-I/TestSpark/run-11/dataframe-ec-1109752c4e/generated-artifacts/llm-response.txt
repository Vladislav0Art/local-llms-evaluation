import io.github.vmzakharov.ecdataframe.dsl.*;
import io.github.vmzakharov.ecdataframe.dsl.value.StringValue;
import io.github.vmzakharov.ecdataframe.dsl.visitor.PrettyPrintVisitor;
import org.eclipse.collections.impl.factory.Lists;
import org.junit.Test;

import static org.junit.Assert.assertEquals;

public class PrettyPrintVisitorTest {

    @Test
    public void visitAssignExprEscapedTest() {
        AssingExpr expr = new AssingExpr("varName", new StringExpression("testValue"));
        expr.setEscaped(true);
        assertEquals("${varName} = 'testValue'", PrettyPrintVisitor.exprToString(expr));
    }

    @Test
    public void visitAssignExprNotEscapedTest() {
        AssingExpr expr = new AssingExpr("varName", new StringExpression("testValue"));
        expr.setEscaped(false);
        assertEquals("varName = 'testValue'", PrettyPrintVisitor.exprToString(expr));
    }

    @Test
    public void visitBinaryExprTest() {
        BinaryExpr expr = new BinaryExpr(new StringExpression("left"), BinaryOp.ADD, new StringExpression("right"));
        assertEquals("('left' + 'right')", PrettyPrintVisitor.exprToString(expr));
    }

    @Test
    public void visitUnaryExprPrefixTest() {
        UnaryExpr expr = new UnaryExpr(UnaryOp.NOT, new StringExpression("testValue"));
        assertEquals("!('testValue')", PrettyPrintVisitor.exprToString(expr));
    }

    @Test
    public void visitConstExprTest() {
        ConstExpr expr = new StringExpression("testValue");
        assertEquals("'testValue'", PrettyPrintVisitor.exprToString(expr));
    }

    @Test
    public void visitFunctionCallExprTest() {
        FunctionCallExpr expr = new FunctionCallExpr("functionName", Lists.mutable.of(new StringExpression("parameter1"), new StringExpression("parameter2")));
        assertEquals("functionName('parameter1', 'parameter2')", PrettyPrintVisitor.exprToString(expr));
    }

    @Test
    public void visitPropertyPathExprTest() {
        PropertyPathExpr expr = new PropertyPathExpr(Lists.mutable.of("property1", "property2"));
        assertEquals("property1.property2", PrettyPrintVisitor.exprToString(expr));
    }

    @Test
    public void visitAnonymousScriptExprTest() {
        AnonymousScript expr = new AnonymousScript(Lists.mutable.of(new StringExpression("testValue")));
        assertEquals("'testValue'", PrettyPrintVisitor.exprToString(expr));
    }
    
    @Test
    public void visitVarExprEscapedTest() {  
        VarExpr expr = new VarExpr("testName");  
        expr.setEscaped(true);  
        assertEquals("${testName}", PrettyPrintVisitor.exprToString(expr));  
    }  
    
    @Test  
    public void visitVarExprNotEscapedTest() {  
        VarExpr expr = new VarExpr("testName");  
        expr.setEscaped(false);  
        assertEquals("testName", PrettyPrintVisitor.exprToString(expr));  
    }  

    // Add similar test methods for other methods in PrettyPrintVisitor class to achieve full class coverage
}
