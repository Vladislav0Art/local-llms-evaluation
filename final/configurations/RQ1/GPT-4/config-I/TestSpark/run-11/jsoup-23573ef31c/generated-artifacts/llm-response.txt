import org.junit.Assert;
import org.junit.Test;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.TextNode;

import java.nio.charset.StandardCharsets;
import java.util.List;

public class DocumentTest {

    @Test
    public void DocumentConstructorTest() {
        String baseUri = "https://www.github.com";
        Document doc = new Document(baseUri);
        Assert.assertEquals(doc.location(), baseUri);
    }

    @Test
    public void DocumentCreateShellTest() {
        String baseUri = "https://www.github.com";
        Document doc = Document.createShell(baseUri);

        Assert.assertEquals(doc.location(), baseUri);
        Assert.assertNotNull(doc.select("html"));
        Assert.assertNotNull(doc.select("head"));
        Assert.assertNotNull(doc.select("body"));
    }

    @Test
    public void DocumentLocationTest() {
        String baseUri = "https://stackoverflow.com";
        Document doc = new Document(baseUri);
        Assert.assertEquals(doc.location(), baseUri);
    }

    @Test
    public void DocumentHeadTest() {
        String baseUri = "http://www.google.com";
        Document doc = Document.createShell(baseUri);
        Element head = doc.head();
        Assert.assertNotNull(head);
    }

    @Test
    public void DocumentBodyTest(){
        String baseUri = "http://www.google.com";
        Document doc = Document.createShell(baseUri);
        Element body = doc.body();
        Assert.assertNotNull(body);
    }

    @Test
    public void DocumentTitleSettingTest(){
        String baseUri = "http://www.google.com";
        Document doc = Document.createShell(baseUri);
        String titleContent = "Google";
        doc.title(titleContent);
        Assert.assertEquals(doc.title(),titleContent);
    }

    @Test
    public void DocumentTitleFetchingTest(){
        String baseUri = "http://www.google.com";
        Document doc = Document.createShell(baseUri);
        Assert.assertEquals(doc.title(),"");
    }

    @Test
    public void DocumentOuterHtmlTest(){
        String baseUri = "http://www.google.com";
        Document doc = Document.createShell(baseUri);
        doc.head().appendChild(new TextNode("Test"));

        String expectedHtml = "<html>\n <head>\n  Test\n </head>\n <body></body>\n</html>\n";
        Assert.assertEquals(doc.outerHtml(), expectedHtml);
    }

    @Test
    public void DocumentCharSetTest(){
        String baseUri = "http://www.google.com";
        Document doc = Document.createShell(baseUri);

        doc.charset(StandardCharsets.UTF_8);
        Assert.assertEquals(doc.charset(), StandardCharsets.UTF_8);
    }

    @Test
    public void DocumentUpdateMetaCharsetElementTest() {
        String baseUri = "http://www.google.com";
        Document doc = Document.createShell(baseUri);
        doc.updateMetaCharsetElement(true);
        Assert.assertTrue(doc.updateMetaCharsetElement());
    }
    
    @Test
    public void DocumentCloneTest() {
        String baseUri = "http://www.google.com";
        Document doc = Document.createShell(baseUri);
        Document docCloned = doc.clone();
        Assert.assertEquals(doc.outerHtml(), docCloned.outerHtml());
    }
}
