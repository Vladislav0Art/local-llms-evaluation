        assertFalse(decoder.getHasIndex());
    }

    public void getManufacturerTest() {
        assertNull(decoder.getManufacturer());
    }

    public void decodeAlarmTest() {

        assertEquals(Position.ALARM_FALL_DOWN, decoder.decodeAlarm(22));
    }

    @Test
    public void decodePositionTest() {
        assertNull(decoder.decodePosition(null, "INVALID_DATA"));
    }
