import org.jsoup.Connection;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.powermock.core.classloader.annotations.PrepareForTest;
import org.powermock.modules.junit4.PowerMockRunner;

import java.io.UnsupportedEncodingException;
import java.net.URL;
import java.nio.charset.StandardCharsets;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertTrue;
import static org.junit.Assert.fail;
import static org.powermock.api.mockito.PowerMockito.mockStatic;
import static org.powermock.api.mockito.PowerMockito.when;

@RunWith(PowerMockRunner.class)
@PrepareForTest({Connection.KeyVal.class, UrlBuilder.class})
public class UrlBuilderTest {

    @Mock
    Connection.KeyVal kv;

    @Test
    public void constructorAndBuildURLTest() {
        try {
            URL url = new URL("https://test.com");
            UrlBuilder urlBuilder = new UrlBuilder(url);
            assertEquals(url, urlBuilder.build());
        } catch (Exception ex) {
            fail("No exception should be thrown.");
        }
    }

    @Test
    public void appendKeyValWithNonNullQueryTest() {
        try {
            URL url = new URL("https://test.com?key=value");
            UrlBuilder urlBuilder = new UrlBuilder(url);
            when(kv.key()).thenReturn("newKey");
            when(kv.value()).thenReturn("newValue");
            urlBuilder.appendKeyVal(kv);
            assertEquals(url.getQuery() + "&" +
                    java.net.URLEncoder.encode("newKey", StandardCharsets.UTF_8.name()) + "=" +
                    java.net.URLEncoder.encode("newValue", StandardCharsets.UTF_8.name()), urlBuilder.q.toString());
        } catch (Exception ex) {
            fail("No exception should be thrown.");
        }
    }

    @Test
    public void appendKeyValWithNullQueryTest() {
        try {
            URL url = new URL("https://test.com");
            UrlBuilder urlBuilder = new UrlBuilder(url);
            when(kv.key()).thenReturn("newKey");
            when(kv.value()).thenReturn("newValue");
            urlBuilder.appendKeyVal(kv);
            assertEquals(java.net.URLEncoder.encode("newKey", StandardCharsets.UTF_8.name()) + "=" +
                    java.net.URLEncoder.encode("newValue", StandardCharsets.UTF_8.name()), urlBuilder.q.toString());
        } catch (Exception ex) {
            fail("No exception should be thrown.");
        }
    }

    @Test
    public void buildWithNonNullQueryAndNonNullRefTest() {
        try {
            URL url = new URL("https://test.com?key=value#ref");
            UrlBuilder urlBuilder = new UrlBuilder(url);
            assertTrue(urlBuilder.build().toString().contains("key=value"));
            assertTrue(urlBuilder.build().toString().contains("#ref"));
        } catch (Exception ex) {
            fail("No exception should be thrown.");
        }
    }

    @Test
    public void buildWithNullQueryAndNonNullRefTest() {
        try {
            URL url = new URL("https://test.com#ref");
            UrlBuilder urlBuilder = new UrlBuilder(url);
            assertTrue(urlBuilder.build().toString().contains("#ref"));
        } catch (Exception ex) {
            fail("No exception should be thrown.");
        }
    }

    @Test
    public void buildWithNonNullQueryAndNullRefTest() {
        try {
            URL url = new URL("https://test.com?key=value");
            UrlBuilder urlBuilder = new UrlBuilder(url);
            assertTrue(urlBuilder.build().toString().contains("key=value"));
        } catch (Exception ex) {
            fail("No exception should be thrown.");
        }
    }

    @Test
    public void buildWithInvalidURLTest() {
        try {
            mockStatic(Validate.class);
            URL url = new URL("https://test.com#ref");
            UrlBuilder urlBuilder = new UrlBuilder(url);
            when(Validate.assertFail("")).thenReturn(false);
            assertTrue(url.toString().equals(urlBuilder.build().toString()));
        } catch (Exception ex) {
            fail("No exception should be thrown.");
        }
    }

    @Test
    public void decodePartWithValidPartTest() {
        try {
            String decodedPart = "test123";
            String encodedPart = java.net.URLEncoder.encode(decodedPart, StandardCharsets.UTF_8.name());
            assertEquals(decodedPart, UrlBuilder.decodePart(encodedPart));
        } catch (Exception ex) {
            fail("No exception should be thrown.");
        }
    }

    @Test(expected = RuntimeException.class)
    public void decodePartWithInvalidPartTest() {
        UrlBuilder.decodePart("\uD844\uDF0F");
    }

    @Test
    public void appendToAsciiTest() {
        try {
            String original = "sample";
            StringBuilder sb = new StringBuilder();
            UrlBuilder.appendToAscii(original, false, sb);
            assertEquals(original, sb.toString());
        } catch (UnsupportedEncodingException ex) {
            fail("No exception should be thrown.");
        }
    }
}
