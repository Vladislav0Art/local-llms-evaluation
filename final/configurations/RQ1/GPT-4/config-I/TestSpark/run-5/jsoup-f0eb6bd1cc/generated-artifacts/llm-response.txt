import org.junit.Test;
import org.jsoup.Connection;
import org.jsoup.helper.UrlBuilder;
import java.net.URL;

import static org.junit.Assert.assertEquals;

public class UrlBuilderTest {

    @Test
    public void buildTest() throws Exception {
        URL inputURL = new URL("http://www.example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputURL);
        URL resultURL = urlBuilder.build();

        assertEquals(inputURL, resultURL);
    }

    @Test
    public void appendKeyValTest() throws Exception {
        URL inputURL = new URL("http://www.example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputURL);
        Connection.KeyVal kv = Connection.KeyVal.create("key", "value");

        urlBuilder.appendKeyVal(kv);

        URL resultURL = urlBuilder.build();

        assertEquals("http://www.example.com?key=value", resultURL.toString());
    }

    @Test(expected = RuntimeException.class)
    public void decodePartTest() throws Exception {
        final String encoded = "\uC881";

        UrlBuilder.decodePart(encoded);
    }
}

            @Override

            }

                return "value";
            }

            @Override
            public Connection.KeyVal value(String value) {
                return null;
            }
        };

        urlBuilder.appendKeyVal(kv);

        URL resultURL = urlBuilder.build();

        assertEquals("http://www.example.com?key=value", resultURL.toString());
    }
}

import org.junit.Test;
import org.jsoup.Connection;
import java.net.URL;

import static org.junit.Assert.assertEquals;
import static org.mockito.Mockito.mock;
import static org.mockito.Mockito.when;

public class UrlBuilderTest {

    @Test
    public void buildTest() throws Exception {
        URL inputURL = new URL("http://www.example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputURL);
        URL resultURL = urlBuilder.build();

        assertEquals(inputURL, resultURL);
    }

    @Test
    public void appendKeyValTest() throws Exception {
        URL inputURL = new URL("http://www.example.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputURL);
        
        Connection.KeyVal kv = mock(Connection.KeyVal.class);
        when(kv.key()).thenReturn("key");
        when(kv.value()).thenReturn("value");

        urlBuilder.appendKeyVal(kv);

        URL resultURL = urlBuilder.build();

        assertEquals("http://www.example.com?key=value", resultURL.toString());
    }
}
