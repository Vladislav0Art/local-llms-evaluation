import static org.junit.Assert.*;

        UrlBuilder builder = new UrlBuilder(url);
        assertEquals(new URI("http://www.test.com:80"), builder.build().toURI());
    }

    @Test(expected = MalformedURLException.class)
    public void buildInvalidProtocolTest() throws Exception {
        URL url = new URL("invalid://www.test.com:80");
        UrBuilder builder = new UrlBuilder(url);
        builder.build();
    }

    @Test
    public void buildWithHostUnicodeCharTest() throws Exception {
        URL url = new URL("http://www.tëst.com:80");
        UrlBuilder builder = new UrlBuilder(url);
        assertEquals(new URI("http://www.xn--tst-qla.com:80"), builder.build().toURI());
    }

    @Test
    public void appendKeyValTest() throws Exception {
        URL url = new URL("http://www.test.com");
        UrlBuilder builder = new UrlBuilder(url);
        Connection.KeyVal keyVal = new Connection.KeyVal() {
            @Override
            public String key() {
                return "testKey";
            }
            @Override
            public String value() {
                return "testValue";
            }
            @Override
            public KeyVal key(String key) {
                return this;
            }
            @Override
            public KeyVal value(String value) {
                return this;
            }
        };
        builder.appendKeyVal(keyVal);
        assertEquals(new URI("http://www.test.com?testKey=testValue"), builder.build().toURI());
    }
}
