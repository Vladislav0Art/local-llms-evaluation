import ch.jalu.configme.exception.ConfigMeException;
import ch.jalu.configme.properties.IntegerProperty;
import ch.jalu.configme.properties.Property;
import org.junit.Rule;
import org.junit.Test;
import org.junit.rules.ExpectedException;

import java.util.List;

import static org.hamcrest.Matchers.contains;
import static org.junit.Assert.assertThat;

public class PropertyListBuilderTest {

    @Rule
    public ExpectedException thrown = ExpectedException.none();

    private PropertyListBuilder underTest = new PropertyListBuilder();

    // Test for add method
    @Test
    public void addUniquePropertiesTest() {
        // Given
        Property<?> property1 = new IntegerProperty("DataSource.mysql.Port", 3306);
        Property<?> property2 = new IntegerProperty("DataSource.mysql.MaxConnections", 10);
        Property<?> property3 = new IntegerProperty("Security.LoginAttempts", 5);
        
        // When
        underTest.add(property1);
        underTest.add(property2);
        underTest.add(property3);

        // Then
        List<Property<?>> properties = underTest.create();
        assertThat(properties, contains(property1, property2, property3));
    }

    @Test
    public void addDuplicatePropertyTest() {        
        // Given
        Property<?> property = new IntegerProperty("DataSource.mysql.Port", 3306);
        underTest.add(property);

        // Then
        thrown.expect(ConfigMeException.class);
        thrown.expectMessage("Path at '" + property.getPath() + "' already exists");
        
        // When
        underTest.add(property);
    }

    // Test for create method
    @Test
    public void createWithEmptyPropertiesTest() {
        // When
        List<Property<?>> properties = underTest.create();

        // Then
        assertThat(properties.isEmpty(), is(true));
    }

    // Test for getRootEntries method
    @Test
    public void getRootEntriesWithPropertiesTest() {
        // Given
        Property<?> property = new IntegerProperty("DataSource.mysql.Port", 3306);
        underTest.add(property);

        // When
        Map<String, Object> entries = underTest.getRootEntries();

        // Then
        assertThat(entries.containsKey("DataSource"), is(true));
    }
}
