import static org.junit.Assert.assertEquals;
import static org.mockito.ArgumentMatchers.any;
import static org.mockito.Mockito.doReturn;
import static org.mockito.Mockito.spy;

import graphql.annotations.processor.ProcessingElementsContainer;
import graphql.annotations.processor.exceptions.GraphQLAnnotationsException;
import graphql.annotations.processor.retrievers.fieldBuilders.DirectivesBuilder;
import graphql.schema.GraphQLArgument;
import graphql.schema.GraphQLDirective;
import graphql.schema.Coercing;
import graphql.schema.GraphQLScalarType;

import java.lang.annotation.Annotation;
import java.lang.reflect.Method;
import java.util.ArrayList;
import java.util.List;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

@RunWith(MockitoJUnitRunner.class)
public class DirectivesBuilderTest {

    @Mock
    private ProcessingElementsContainer container;

    @Mock
    private GraphQLDirective.Builder directiveBuilder;

    @Mock
    private GraphQLDirective directive;

    @Mock
    private GraphQLArgument argument;

    @Mock
    private Coercing coercing;

    @Test(expected = GraphQLAnnotationsException.class)
    public void buildNoDirectiveInRegistryTest() {
        Annotation annotation = new DummyDirectiveAnnotation();
        DirectivesBuilder builder = new DirectivesBuilder(annotation, container);

        builder.build();
    }

    @Test
    public void buildOneDirectiveTest() throws Exception {
        Annotation annotation = new DummyDirectiveAnnotation();
        DirectivesBuilder builder = new DirectivesBuilder(annotation, container);

        List<GraphQLArgument> argumentList = new ArrayList<>();
        argumentList.add(argument);

        Object argumentValue = new Object();

        doReturn(directive).when(container.getDirectiveRegistry()).get("dummy");
        doReturn(directiveBuilder).when(directive).newDirective();
        doReturn(argumentList).when(directive).getArguments();
        doReturn(GraphQLScalarType.newScalar().coercing(coercing).name("dummy").build()).when(argument).getType();
        doReturn(argumentValue).when(coercing).parseValue(argumentValue);

        GraphQLDirective[] result = builder.build();

        assertEquals(1, result.length);
        assertEquals(directive, result[0]);
    }

    @Test(expected = GraphQLAnnotationsException.class)
    public void transformArgsAnnotationToManyArgumentsTest() throws Exception {
        Annotation annotation = new DummyDirectiveAnnotation();
        DirectivesBuilder builder = new DirectivesBuilder(annotation, container);

        List<GraphQLArgument> argumentList = new ArrayList<>();

        doReturn(directiveBuilder).when(directive).newDirective();
        doReturn(argumentList).when(directive).getArguments();

        builder.transformArgs(directive, annotation);
    }

    @Test
    public void transformArgumentTest() throws Exception {
        Annotation annotation = new DummyDirectiveAnnotation();
        DirectivesBuilder builder = spy(new DirectivesBuilder(annotation, container));
        
        List<GraphQLArgument> argumentList = new ArrayList<>();
        argumentList.add(argument);

        Object argumentValue = new Object();
        Method[] methods = annotation.annotationType().getDeclaredMethods();

        doReturn(GraphQLScalarType.newScalar().coercing(coercing).name("dummy").build()).when(argument).getType();
        doReturn(argumentValue).when(coercing).parseValue(any());
        doReturn(argumentValue).when(methods[0]).invoke(annotation);

        builder.transformArgument(annotation, directiveBuilder, argumentList, 0);
    }

    @Test(expected = GraphQLAnnotationsException.class)
    public void transformArgumentNotAScalarTest() throws Exception {
        Annotation annotation = new DummyDirectiveAnnotation();
        DirectivesBuilder builder = new DirectivesBuilder(annotation, container);
        
        List<GraphQLArgument> argumentList = new ArrayList<>();
        argumentList.add(argument);

        builder.transformArgument(annotation, directiveBuilder, argumentList, 0);
    }

    private @interface DummyDirectiveAnnotation {
        String name() default "dummy";
    }
}

    public void buildTest_NoDirectiveInRegistry() {
        Annotation object = mock(Annotation.class);
        ProcessingElementsContainer container = mock(ProcessingElementsContainer.class);

        when(container.getDirectiveRegistry()).thenReturn(Collections.emptyMap());

        DirectivesBuilder builder = new DirectivesBuilder(object, container);
        builder.build();
    }

    @Test
    public void buildTest_DirectiveInRegistry() {
        Annotation object = mock(Annotation.class);
        ProcessingElementsContainer container = mock(ProcessingElementsContainer.class);
        Map<String, Collection<GraphQLArgument>> directivesRegistry = new HashMap<>();

        GraphQLArgument arg = GraphQLArgument.newArgument().name("A1").type(GraphQLScalarType.newScalar().name("T1").build()).build();
        directivesRegistry.put("D1", Collections.singleton(arg));

        when(container.getDirectiveRegistry()).thenReturn(directivesRegistry);
        Method[] methods = { mock(Method.class) };
        when(object.annotationType()).thenReturn(Directive.class);
        when(object.annotationType().getDeclaredMethods()).thenReturn(methods);

        DirectivesBuilder builder = new DirectivesBuilder(object, container);
        GraphQLDirective[] directives = builder.build();
        assertEquals(1, directives.length);
        assertEquals(1, directives[0].getArguments().size());
    }
