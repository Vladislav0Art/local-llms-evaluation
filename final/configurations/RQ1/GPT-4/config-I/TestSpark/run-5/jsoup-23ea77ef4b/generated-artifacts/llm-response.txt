import org.jsoup.nodes.*;
import org.junit.Test;
import org.w3c.dom.Document;
import org.w3c.dom.Node;

import javax.xml.parsers.ParserConfigurationException;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

import static org.junit.Assert.*;

public class W3CDomTest {
    @Test
    public void namespaceAwareDefaultCheckTest() throws ParserConfigurationException {
        W3CDom w3CDom = new W3CDom();
        assertTrue(w3CDom.namespaceAware());
    }

    @Test
    public void namespaceAwareSetFalseTest() throws ParserConfigurationException {
        W3CDom w3CDom = new W3CDom();
        w3CDom.namespaceAware(false);
        assertFalse(w3CDom.namespaceAware());
    }

    @Test
    public void convertTest() throws ParserConfigurationException {
        Document in = org.jsoup.Jsoup.parse("<html><head><title>Title</title></head><body>Body</body></html>").ownerDocument();
        W3CDom w3CDom = new W3CDom();
        Document out = w3CDom.fromJsoup(in);
        assertEquals("html", out.getDocumentElement().getTagName());
    }

    @Test
    public void asStringWithPropertiesTest() throws ParserConfigurationException {
        Document in = org.jsoup.Jsoup.parse("<html><head><title>Title</title></head><body>Body</body></html>").ownerDocument();
        W3CDom w3CDom = new W3CDom();
        Document out = w3CDom.fromJsoup(in);

        Map<String, String> properties = new HashMap<>();
        properties.put("method", "html");
        assertEquals("<html><head><meta></meta><title>Title</title></head><body>Body</body></html>",
                W3CDom.asString(out, properties).trim());
    }

    @Test
    public void propertiesFromMapTest() throws ParserConfigurationException {
        Map<String, String> map = new HashMap<>();
        map.put("method", "html");
        assertEquals("html", W3CDom.propertiesFromMap(map).get("method"));
    }

    @Test
    public void outputHtmlTest() throws ParserConfigurationException {
        HashMap<String, String> outputHtml = W3CDom.OutputHtml();
        assertEquals("html", outputHtml.get("method"));
    }

    @Test
    public void outputXmlTest() throws ParserConfigurationException {
        HashMap<String, String> outputXml = W3CDom.OutputXml();
        assertEquals("xml", outputXml.get("method"));
    }

    @Test
    public void contextNodeTest() throws ParserConfigurationException {
        Document in = org.jsoup.Jsoup.parse("<html><head><title>Title</title></head><body>Body</body></html>").ownerDocument();
        W3CDom w3CDom = new W3CDom();
        Document out = w3CDom.fromJsoup(in);
        Node contextNode = w3CDom.contextNode(out);
        assertEquals("html", contextNode.getNodeName());
    }

    @Test
    public void selectXpathTest() throws ParserConfigurationException {
        Document in = org.jsoup.Jsoup.parse("<html><head><title>Title</title></head><body>Body</body></html>").ownerDocument();
        W3CDom w3CDom = new W3CDom();
        Document out = w3CDom.fromJsoup(in);
        assertEquals("html", w3CDom.selectXpath("/", out).item(0).getNodeName());
    }


    @Test
    public void sourceNodesTest() throws ParserConfigurationException {
        Document in = org.jsoup.Jsoup.parse("<html><head><title>Title</title></head><body>Body</body></html>").ownerDocument();
        W3CDom w3CDom = new W3CDom();
        Document out = w3CDom.fromJsoup(in);
        List<Element> elements = w3CDom.sourceNodes(out.getChildNodes(), Element.class);
        assertEquals(1, elements.size());
        assertTrue(elements.get(0) instanceof Element);
    }

    @Test
    public void asStringTest() throws ParserConfigurationException {
        Document in = org.jsoup.Jsoup.parse("<html><head><title>Title</title></head><body>Body</body></html>").ownerDocument();
        W3CDom w3CDom = new W3CDom();
        Document out = w3CDom.fromJsoup(in);
        assertEquals("<html><head><meta></meta><title>Title</title></head><body>Body</body></html>",
                w3CDom.asString(out).trim());
    }
}
