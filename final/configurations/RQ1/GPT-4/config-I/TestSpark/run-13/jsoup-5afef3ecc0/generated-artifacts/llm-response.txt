```java
import org.jsoup.Connection;
import org.junit.Test;
import static org.junit.Assert.*;

import java.net.MalformedURLException;
import java.net.URL;

public class UrlBuilderTest {

    @Test
    public void buildTestNoQueryNoRef() throws MalformedURLException {
        URL inputUrl = new URL("http://www.google.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);

        URL result = urlBuilder.build();
        assertEquals("http://www.google.com", result.toString());
    }

    @Test
    public void buildTestQueryNoRef() throws MalformedURLException {
        URL inputUrl = new URL("http://www.google.com?test=query");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);

        URL result = urlBuilder.build();
        assertEquals("http://www.google.com?test=query", result.toString());
    }

    @Test
    public void buildTestRefNoQuery() throws MalformedURLException {
        URL inputUrl = new URL("http://www.google.com#test");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);

        URL result = urlBuilder.build();
        assertEquals("http://www.google.com#test", result.toString());
    }

    @Test
    public void buildTestRefAndQuery() throws MalformedURLException {
        URL inputUrl = new URL("http://www.google.com?test=query#test");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);

        URL result = urlBuilder.build();
        assertEquals("http://www.google.com?test=query#test", result.toString());
    }

    @Test(expected = RuntimeException.class)
    public void buildTestInvalidURL() throws MalformedURLException {
        URL inputUrl = new URL("http::invalid_url");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);

        URL result = urlBuilder.build();
    }

    @Test
    public void appendKeyValTest() throws Exception {
        URL inputUrl = new URL("http://www.google.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        Connection.KeyVal kv = Connection.KeyVal.create("key1", "value1");
```

```java
        urlBuilder.appendKeyVal(kv);
        URL result = urlBuilder.build();

        assertEquals("http://www.google.com?key1=value1", result.toString());
    }
}
```

```java
import org.jsoup.Connection;
import org.junit.Test;

import java.net.MalformedURLException;
import java.net.URL;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.fail;

public class UrlBuilderTest {

    @Test
    public void buildNoQueryNoRefTest() throws MalformedURLException {
        URL inputUrl = new URL("http://www.google.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        URL result = urlBuilder.build();
        assertEquals("http://www.google.com", result.toString());
    }

    @Test
    public void buildQueryNoRefTest() throws MalformedURLException {
        URL inputUrl = new URL("http://www.google.com?param=value");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        URL result = urlBuilder.build();
        assertEquals("http://www.google.com?param=value", result.toString());
    }

    @Test
    public void buildNoQueryWithRefTest() throws MalformedURLException {
        URL inputUrl = new URL("http://www.google.com#ref");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        URL result = urlBuilder.build();
        assertEquals("http://www.google.com#ref", result.toString());
    }
    
    @Test
    public void appendKeyValTest() throws Exception {
        URL inputUrl = new URL("http://www.google.com");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        Connection.KeyVal kv = new Connection.KeyVal() {
            @Override
            public String key() {
                return "key";
            }

            @Override
            public KeyVal key(String s) {
                return this;
            }

            @Override
            public String value() {
                return "value";
            }

            @Override
            public KeyVal value(String s) {
                return this;
            }
            
            @Override
            public String contentType() {
                return null;
            }

            @Override
            public KeyVal contentType(String s) {
                return this;
            }
        };
        urlBuilder.appendKeyVal(kv);
        URL result = urlBuilder.build();
        assertEquals("http://www.google.com?key=value", result.toString());
    }
    
    @Test(expected = RuntimeException.class)
    public void buildInvalidURLTest() throws MalformedURLException {
        URL inputUrl = new URL("http:invalid_url");
        UrlBuilder urlBuilder = new UrlBuilder(inputUrl);
        urlBuilder.build();
    }
}
```
