    }

    public void runHelpArgumentTest() {
        EpubChecker epubChecker = mock(EpubChecker.class);
        String[] args = {"-help"};

        epubChecker.run(args);
        verify(epubChecker, times(1)).run(args);
    }

    public void getLocaleDefaultTest(){

        verify(epubChecker, times(1)).dumpMessageDictionary(report);
    }

    @Test
    public void printEpubCheckCompletedTest() throws IOException {
        Report report = mock(Report.class);

        verify(outWriter, times(1)).setQuiet(false);
    }

    public void displayHelpTest() {

        verify(outWriter, times(1)).println(String.format(Messages.getInstance().get("help_text"), EpubCheck.version()));
    }
