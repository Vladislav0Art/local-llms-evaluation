package io.github.vmzakharov.ecdataframe.dsl.visitor;

import io.github.vmzakharov.ecdataframe.dsl.*;
import io.github.vmzakharov.ecdataframe.util.CollectingPrinter;
import org.junit.Test;
import static org.junit.Assert.assertEquals;

public class PrettyPrintVisitorTest {
    @Test
    public void exprToStringTest() {
        Expression expr = new VarExpr(new ExprContext(), "variable");
        String outStr = PrettyPrintVisitor.exprToString(expr);
        assertEquals("variable", outStr);
    }

    @Test
    public void visitAssignExprTest() {
        AssingExpr assignExpr = new AssingExpr(new ExprContext(), "var", new BinaryExpr(new ExprContext(), new VarExpr("v1"), new VarExpr("v2"), BinaryOp.PLUS));
        assignExpr.accept(new PrettyPrintVisitor(new CollectingPrinter()));
        assertEquals("var = (v1 + v2)", PrettyPrintVisitor.exprToString(assignExpr));
    }

    @Test
    public void visitBinaryExprTest() {
        BinaryExpr binaryExpr = new BinaryExpr(new ExprContext(), new VarExpr("v1"), new VarExpr("v2"), BinaryOp.PLUS);
        assertEquals("(v1 + v2)", PrettyPrintVisitor.exprToString(binaryExpr));
    }

    @Test
    public void visitUnaryExprTest(){
        UnaryExpr unaryExpr = new UnaryExpr(new ExprContext(), new VarExpr("var"), UnaryOp.MINUS);
        assertEquals("-var", PrettyPrintVisitor.exprToString(unaryExpr));
    }

    @Test
    public void visitConstExprTest(){
        Value value = Value.newValue(5.0);
        PrettyPrintVisitor.visitor.visitConstExpr(value);
        assertEquals("5", visitor.printer.toString());
    }

    @Test
    public void visitFunctionCallExprTest(){
        FunctionCallExpr functionCallExpr = new FunctionCallExpr(new ExprContext(), "testFun", new VarExpr("param"));
        assertEquals("testFun(param)", PrettyPrintVisitor.exprToString(functionCallExpr));
    }

    @Test
    public void visitPropertyPathExprTest(){
        PropertyPathExpr propertyPathExpr = new PropertyPathExpr(new ExprContext(), "property1.property2");
        assertEquals("property1.property2", PrettyPrintVisitor.exprToString(propertyPathExpr));
    }

    @Test
    public void visitVectorExprTest(){
        VectorExpr vectorExpr = new VectorExpr(new ExprContext(), new DecimalExpr(new ExprContext(), Value.newValue(5), Value.newValue(2)));
        assertEquals("(5, 2)", PrettyPrintVisitor.exprToString(vectorExpr));
    }

    @Test
    public void visitAliasExprTest(){
        AliasExpr aliasExpr = new AliasExpr(new ExprContext(), "alias", new VarExpr("value"));
        assertEquals("alias : value", PrettyPrintVisitor.exprToString(aliasExpr));
    }

    @Test
    public void visitIndexExprTest(){
        IndexExpr indexExpr = new IndexExpr(new ExprContext(), new VarExpr("var"), new DecimalExpr(new ExprContext(), Value.newValue(5), Value.newValue(2)));
        assertEquals("var[5, 2]", PrettyPrintVisitor.exprToString(indexExpr));
    }

    @Test
    public void visitDecimalExprTest(){
        DecimalExpr decimalExpr = new DecimalExpr(new ExprContext(), Value.newValue(5), Value.newValue(2));
        assertEquals("[5, 2]", PrettyPrintVisitor.exprToString(decimalExpr));
    }
}
