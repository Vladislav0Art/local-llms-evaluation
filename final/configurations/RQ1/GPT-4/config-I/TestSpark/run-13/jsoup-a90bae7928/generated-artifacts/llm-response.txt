import junit.framework.TestCase;
import org.jsoup.parser.*;
import org.jsoup.nodes.*;

import java.io.StringReader;
import java.util.List;

public class XmlTreeBuilderTest extends TestCase {

    public void testInitialiseParseTest() {
        Reader input = new StringReader("<test></test>");
        String baseUri = "http://test.com";
        XmlTreeBuilder builder = new XmlTreeBuilder();
        Parser parser = new Parser(builder);

        builder.initialiseParse(input, baseUri, parser);

        assertEquals(baseUri, builder.getBaseUri());
        assertTrue(builder.getStack().contains(builder.getDocument()));
        assertFalse(builder.getDocument().outputSettings().prettyPrint());
    }

    public void testParseReaderBaseUriTest() {
        Reader input = new StringReader("<test></test>");
        String baseUri = "http://test.com";
        XmlTreeBuilder builder = new XmlTreeBuilder();

        Document document = builder.parse(input, baseUri);
        
        assertEquals("<test></test>", document.outerHtml());
        assertEquals(baseUri, document.baseUri());
    }

    public void testParseStringBaseUriTest() {
        String input = "<test></test>";
        String baseUri = "http://test.com";
        XmlTreeBuilder builder = new XmlTreeBuilder();

        Document document = builder.parse(input, baseUri);

        assertEquals("<test></test>", document.outerHtml());
        assertEquals(baseUri, document.baseUri());
    }
    
    public void testProcessStartTagTest() {
        XmlTreeBuilder builder = new XmlTreeBuilder();
        Token.StartTag startTag = new Token.StartTag();
        startTag.tagName("test");
        Token token = startTag;

        builder.process(token);

        assertTrue(builder.getStack().get(1) instanceof Element);
        assertEquals("test", builder.getStack().get(1).nodeName());
    }
    
    public void testProcessEndTagTest() {
        XmlTreeBuilder builder = new XmlTreeBuilder();
        Token.StartTag startTag = new Token.StartTag();
        startTag.tagName("test");
        Token tokenStart = startTag;
        builder.process(tokenStart);

        Token.EndTag endTag = new Token.EndTag();
        endTag.tagName("test");
        Token tokenEnd = endTag;
        builder.process(tokenEnd);

        assertFalse(builder.getStack().contains(tokenEnd));
    }
    
    public void testInsertNodeTest() {
        XmlTreeBuilder builder = new XmlTreeBuilder();
        Node node = new Element("test");

        builder.insertNode(node);
        
        assertEquals(2, builder.getStack().size());
        assertEquals(node, builder.getStack().get(1));
    }
    
    // More unit tests can be added here for methods like insert(), popStackToClose(), parseFragment(), etc.
}

import org.jsoup.nodes.Comment;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.parser.*;
import org.junit.Test;
import static org.junit.Assert.*;

import java.io.StringReader;
import java.util.ArrayList;
import java.util.List;

public class XmlTreeBuilderTest {
    @Test
    public void initialiseParseTest() {
        final Reader input = new StringReader("<test></test>");
        final String baseUri = "http://test.com";
        final XmlTreeBuilder builder = new XmlTreeBuilder();
        final Parser parser = new Parser(builder);

        builder.initialiseParse(input, baseUri, parser);

        assertEquals(Document.OutputSettings.Syntax.xml, builder.doc.outputSettings().syntax());
        assertEquals(Entities.EscapeMode.xhtml, builder.doc.outputSettings().escapeMode());
    }

    @Test
    public void parseTest() {
        final Reader input = new StringReader("<test></test>");
        final String baseUri = "http://test.com";
        final XmlTreeBuilder builder = new XmlTreeBuilder();
        Document document = builder.parse(input, baseUri );

        assertNotNull(document);
        assertEquals("<test></test>", document.toString());
    }

    @Test
    public void processTest() {
        final Token token = new Token.StartTag();
        token.asStartTag().name = "test";

        final XmlTreeBuilder builder = new XmlTreeBuilder();

        boolean ret = builder.process(token);

        assertTrue(ret);

        final List<Element> result = new ArrayList<>();

        for (Element element : builder.stack) {
            result.add(element);
        }

        assertNotNull(result);
        assertTrue(result.get(0) instanceof Document);
    }

    // Tests for insert, insertNode, popStackToClose, parseFragment etc. Can be added..

    @Test(expected = IllegalArgumentException.class)
    public void processExceptionTest() {
        final Token token = new Token();
        token.type = Token.TokenType.Comment;
        final XmlTreeBuilder builder = new XmlTreeBuilder();
        builder.process(token);
    }

}
