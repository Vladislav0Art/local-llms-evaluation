import io.github.vmzakharov.ecdataframe.dsl.*;
import io.github.vmzakharov.ecdataframe.dsl.operator.BinaryOperator;
import io.github.vmzakharov.ecdataframe.dsl.operator.UnaryOperator;
import io.github.vmzakharov.ecdataframe.dsl.value.Value;
import io.github.vmzakharov.ecdataframe.dsl.visitor.PrettyPrintVisitor;
import io.github.vmzakharov.ecdataframe.util.CollectingPrinter;
import static org.junit.Assert.assertEquals;

import org.eclipse.collections.impl.factory.Lists;
import org.junit.Test;

public class PrettyPrintVisitorTest {
    
    @Test
    public void exprToStringTest() {
        VarExpr varExpr = new VarExpr("TestVariableName");
        assertEquals("TestVariableName", PrettyPrintVisitor.exprToString(varExpr));
    }

    @Test
    public void visitAssignExprTest() {
        AssingExpr assignExpr = new AssingExpr("TestVariableName", new DecimalExpr("10"));
        PrettyPrintVisitor prettyPrintVisitor = new PrettyPrintVisitor(new CollectingPrinter());
        assignExpr.accept(prettyPrintVisitor);
        assertEquals("TestVariableName = 10", prettyPrintVisitor.toString());
    }

    @Test
    public void visitBinaryExprTest() {
        BinaryExpr binaryExpr = new BinaryExpr(new StringExpr("hello "), BinaryOperator.PLUS, new StringExpr("world"));
        PrettyPrintVisitor prettyPrintVisitor = new PrettyPrintVisitor(new CollectingPrinter());
        binaryExpr.accept(prettyPrintVisitor);
        assertEquals("(hello  + world)", prettyPrintVisitor.toString());
    }

    @Test
    public void visitUnaryExprTest() {
        UnaryExpr unaryExpr = new UnaryExpr(new VarExpr("TestVariableName"), UnaryOperator.MINUS);
        PrettyPrintVisitor prettyPrintVisitor = new PrettyPrintVisitor(new CollectingPrinter());
        unaryExpr.accept(prettyPrintVisitor);
        assertEquals("-TestVariableName", prettyPrintVisitor.toString());
    }

    @Test
    public void visitConstExprTest() {
        Value constExpr = new StringExpr("TestStringValue");
        PrettyPrintVisitor prettyPrintVisitor = new PrettyPrintVisitor(new CollectingPrinter());
        prettyPrintVisitor.visitConstExpr(constExpr);
        assertEquals("'TestStringValue'", prettyPrintVisitor.toString());
    }

    @Test
    public void visitFunctionCallExprTest() {
        FunctionCallExpr functionCallExpr = new FunctionCallExpr("TestFunction", Lists.fixedSize.of(new VarExpr("TestVarParam")));
        PrettyPrintVisitor prettyPrintVisitor = new PrettyPrintVisitor(new CollectingPrinter());
        functionCallExpr.accept(prettyPrintVisitor);
        assertEquals("TestFunction(TestVarParam)", prettyPrintVisitor.toString());
    }

    @Test
    public void visitPropertyPathExprTest() {
        PropertyPathExpr propertyPathExpr = new PropertyPathExpr("some.property.path");
        PrettyPrintVisitor prettyPrintVisitor = new PrettyPrintVisitor(new CollectingPrinter());
        propertyPathExpr.accept(prettyPrintVisitor);
        assertEquals("some.property.path", prettyPrintVisitor.toString());
    }

    @Test
    public void visitAliasExprTest() {
        AliasExpr aliasExpr = new AliasExpr("TestAlias", new VarExpr("TestVariableName"));
        PrettyPrintVisitor prettyPrintVisitor = new PrettyPrintVisitor(new CollectingPrinter());
        aliasExpr.accept(prettyPrintVisitor);
        assertEquals("TestAlias : TestVariableName", prettyPrintVisitor.toString());
    }  
    
    @Test
    public void visitVectorExprTest() {
        VectorExpr vectorExpr = new VectorExpr(Lists.fixedSize.of(new VarExpr("TestVariableName")));
        PrettyPrintVisitor prettyPrintVisitor = new PrettyPrintVisitor(new CollectingPrinter());
        vectorExpr.accept(prettyPrintVisitor);
        assertEquals("(TestVariableName)", prettyPrintVisitor.toString());
    }

    @Test
    public void visitIndexExprTest() { 
        IndexExpr indexExpr = new IndexExpr(new VarExpr("TestVectorVariableName"), new VarExpr("TestIndexVariableName"));
        PrettyPrintVisitor prettyPrintVisitor = new PrettyPrintVisitor(new CollectingPrinter());
        indexExpr.accept(prettyPrintVisitor);
        assertEquals("TestVectorVariableName[TestIndexVariableName]", prettyPrintVisitor.toString());
    }

    @Test
    public void visitDecimalExprTest() {
        DecimalExpr decimalExpr = new DecimalExpr(new VarExpr("TestVariableName1"), new VarExpr("TestVariableName2"));
        PrettyPrintVisitor prettyPrintVisitor = new PrettyPrintVisitor(new CollectingPrinter());
        decimalExpr.accept(prettyPrintVisitor);
        assertEquals("[TestVariableName1,TestVariableName2]", prettyPrintVisitor.toString());
    }

    @Test
    public void visitIfElseExprTest() {
        IfElseExpr ifElseExpr = new IfElseExpr(new VarExpr("TestConditionVariableName"), new VarExpr("TestIfScriptVarName"), new VarExpr("TestElseScriptVarName"));
        PrettyPrintVisitor prettyPrintVisitor = new PrettyPrintVisitor(new CollectingPrinter());
        ifElseExpr.accept(prettyPrintVisitor);
        assertEquals("TestConditionVariableName ? TestIfScriptVarName : TestElseScriptVarName", prettyPrintVisitor.toString());
    }
