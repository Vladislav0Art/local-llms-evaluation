import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNotNull;
import static org.junit.Assert.assertTrue;
import static org.mockito.Mockito.mock;

import java.util.HashMap;
import java.util.Map;

import net.revelc.code.formatter.ConfigurationSource;
import net.revelc.code.formatter.LineEnding;

import org.junit.Test;
 
public class CssFormatterTest {
    // Mock the ConfigurationSource object
    ConfigurationSource cfg = mock(ConfigurationSource.class);

    @Test
    public void initTest() {
        Map<String, String> options = new HashMap<String, String>();
        options.put("indent", "4");
        options.put("rgbAsHex", "true");
        options.put("useSourceStringValues", "false");
        CssFormatter formatter = new CssFormatter();
        formatter.init(options,cfg);
        assertTrue(formatter.isInitialized());
    }

    @Test
    public void initDefaultTest() {
        Map<String, String> options = new HashMap<String, String>();
        CssFormatter formatter = new CssFormatter();
        formatter.init(options,cfg);
        assertTrue(formatter.isInitialized());
    }

    @Test
    public void doFormatTest() throws IOException {
        Map<String, String> options = new HashMap<String, String>();
        CssFormatter formatter = new CssFormatter();
        formatter.init(options,cfg);
        String formattedCode = formatter.doFormat("h1 \n{color: tab;}", LineEnding.CRLF);
        assertNotNull(formattedCode);
        assertEquals("h1 {\n  color: \\9;\n}", formattedCode);
    }

    @Test
    public void doFormatUnchangedCodeTest() throws IOException {
        Map<String, String> options = new HashMap<String, String>();
        CssFormatter formatter = new CssFormatter();
        formatter.init(options,cfg);
        String code = "h1 {\n  color: \\9;\n}";
        String formattedCode = formatter.doFormat(code, LineEnding.CRLF);
        assertEquals(null, formattedCode);
    }

    @Test
    public void isInitializedTest() {
        Map<String, String> options = new HashMap<String, String>();
        CssFormatter formatter = new CssFormatter();
        formatter.init(options,cfg);
        assertTrue(formatter.isInitialized());
    }

    @Test
    public void isInitializedFalseTest() {
        CssFormatter formatter = new CssFormatter();
        assertFalse(formatter.isInitialized());
    }
}
