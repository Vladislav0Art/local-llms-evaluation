        DirectivesBuilder directivesBuilder = new DirectivesBuilder(method, container);

        Mockito.when(container.getDirectiveRegistry()).thenReturn(Collections.emptyMap());

        directivesBuilder.build();
    }

    @Test(expected = GraphQLAnnotationsException.class)
    public void buildTestWithTooMuchArgumentsError() {
        Method method = Mockito.mock(Method.class);
        ProcessingElementsContainer container = Mockito.mock(ProcessingElementsContainer.class);
        DirectivesBuilder directivesBuilder = new DirectivesBuilder(method, container);

        Directive directive = Mockito.mock(Directive.class);

        directivesBuilder.build();

    public void buildTestSuccessfully() {

        ProcessingElementsContainer container = Mockito.mock(ProcessingElementsContainer.class);
