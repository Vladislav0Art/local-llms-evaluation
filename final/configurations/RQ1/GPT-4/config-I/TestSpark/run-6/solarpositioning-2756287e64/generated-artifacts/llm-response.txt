package net.e175.klaus.solarpositioning;

import org.junit.Test;

import java.time.ZoneOffset;
import java.time.ZonedDateTime;
import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNotNull;

public class Grena3Test {

    private static final double DELTA = 0.01;
    
    @Test
    public void calculateSolarPositionLatitudeZeroLongitudeZeroTest() {
        ZonedDateTime date = ZonedDateTime.now(ZoneOffset.UTC);
        AzimuthZenithAngle solarPosition = Grena3.calculateSolarPosition(date, 0, 0, 0);
        assertNotNull(solarPosition);
    }

    @Test
    public void calculateSolarPositionTypicalValuesTest() {
        ZonedDateTime date = ZonedDateTime.of(2020, 8, 1, 12, 0, 0, 0, ZoneOffset.UTC);
        AzimuthZenithAngle solarPosition = Grena3.calculateSolarPosition(date, 50, 20, DeltaT.estimate(date), 1010, 20);
        assertEquals(180, solarPosition.getAzimuth(), DELTA);
        assertEquals(30, solarPosition.getZenithAngle(), DELTA);
    }

    @Test(expected = IllegalArgumentException.class)
    public void calculateSolarPositionInvalidPressureTest() {
        ZonedDateTime date = ZonedDateTime.now(ZoneOffset.UTC);
        Grena3.calculateSolarPosition(date, 50, 20, DeltaT.estimate(date), -1, 20);
    }

    @Test(expected = IllegalArgumentException.class)
    public void calculateSolarPositionInvalidTemperatureTest() {
        ZonedDateTime date = ZonedDateTime.now(ZoneOffset.UTC);
        Grena3.calculateSolarPosition(date, 50, 20, DeltaT.estimate(date), 1010, -300);
    }

    @Test
    public void calculateSolarPositionEdgeDatesTest() {
        double deltaT = DeltaT.estimate(ZonedDateTime.now(ZoneOffset.UTC));
        ZonedDateTime startDate = ZonedDateTime.of(2010, 1, 1, 12, 0, 0, 0, ZoneOffset.UTC);
        ZonedDateTime endDate = ZonedDateTime.of(2110, 12, 31, 12, 0, 0, 0, ZoneOffset.UTC);

        AzimuthZenithAngle startSolarPosition = Grena3.calculateSolarPosition(startDate, 50, 20, deltaT, 1010, 20);
        AzimuthZenithAngle endSolarPosition = Grena3.calculateSolarPosition(endDate, 50, 20, deltaT, 1010, 20);

        assertNotNull(startSolarPosition);
        assertNotNull(endSolarPosition);
    }
}
