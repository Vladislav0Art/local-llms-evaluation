import org.junit.Assert;
import org.junit.Test;
import org.mockito.Mockito;
import net.i2p.crypto.eddsa.EdDSAEngine;
import net.i2p.crypto.eddsa.EdDSAPrivateKey;
import net.i2p.crypto.eddsa.EdDSAPublicKey;

import java.security.GeneralSecurityException;
import java.security.MessageDigest;

public class KeyPairTest {

    @Test
    public void KeyPairCreationTest() {
        EdDSAPublicKey publicKey = Mockito.mock(EdDSAPublicKey.class);
        EdDSAPrivateKey privateKey = Mockito.mock(EdDSAPrivateKey.class);
        KeyPair keyPair = new KeyPair(publicKey, privateKey);
        Assert.assertTrue(keyPair.canSign());
    }

    @Test(expected = RuntimeException.class)
    public void signMethodPrivateKeyNullTest() {
        EdDSAPublicKey publicKey = Mockito.mock(EdDSAPublicKey.class);
        KeyPair keyPair = new KeyPair(publicKey);
        byte[] data = new byte[10];
        keyPair.sign(data);
    }

    @Test
    public void signMethodTest() throws GeneralSecurityException {
        EdDSAPublicKey publicKey = Mockito.mock(EdDSAPublicKey.class);
        EdDSAPrivateKey privateKey = Mockito.mock(EdDSAPrivateKey.class);
        KeyPair keyPair = new KeyPair(publicKey, privateKey);

        MessageDigest digest = Mockito.mock(MessageDigest.class);
        Mockito.when(digest.getInstance(Mockito.any())).thenReturn("SHA-512");

        Signature sg = Mockito.mock(Signature.class);
        Mockito.when(sg.initSign(privateKey)).thenReturn("");

        byte[] data = new byte[10];
        keyPair.sign(data);
        Mockito.verify(sg).initSign(privateKey);
        Mockito.verify(sg).update(data);
    }

    @Test(expected = RuntimeException.class)
    public void verifyMethodGeneralSecurityExceptionTest() throws GeneralSecurityException {
        EdDSAPublicKey publicKey = Mockito.mock(EdDSAPublicKey.class);
        EdDSAPrivateKey privateKey = Mockito.mock(EdDSAPrivateKey.class);
        KeyPair keyPair = new KeyPair(publicKey, privateKey);
        Mockito.when(publicKey.getParams()).thenThrow(GeneralSecurityException.class);
        keyPair.verify(new byte[10], new byte[10]);
    }

    @Test
    public void verifyMethodTest() throws GeneralSecurityException {
        EdDSAPublicKey publicKey = Mockito.mock(EdDSAPublicKey.class);
        EdDSAPrivateKey privateKey = Mockito.mock(EdDSAPrivateKey.class);
        KeyPair keyPair = new KeyPair(publicKey, privateKey);

        EdDSAEngine sgr = Mockito.mock(EdDSAEngine.class);
        Mockito.when(sgr.initVerify(publicKey)).thenReturn(null);

        byte[] data = new byte[10];
        byte[] signature = new byte[10];
        keyPair.verify(data, signature);
        Mockito.verify(sgr).initVerify(publicKey);
        Mockito.verify(sgr).update(data);
    }
}
