public class WatchProtocolDecoderTest {

    public static void setUp() {

        decoder = new WatchProtocolDecoder(protocol);
    }

    @AfterClass

        decoder = null;
    }

    @Test

        assertEquals(expectedAlarm, decoder.decodeAlarm(status));
    }

    @Test
    public void decodeAlarmGeoFenceExitTest() {
        int status = 1;
        String expectedAlarm = Position.ALARM_GEOFENCE_EXIT;
        assertEquals(expectedAlarm, decoder.decodeAlarm(status));
    }

    @Test
    public void decodeAlarmGeoFenceEnterTest() {
        int status = 2;
        String expectedAlarm = Position.ALARM_GEOFENCE_ENTER;
        assertEquals(expectedAlarm, decoder.decodeAlarm(status));
    }

    @Test
    public void decodeAlarmPowerCutTest() {
        int status = 14;
        String expectedAlarm = Position.ALARM_POWER_CUT;
        assertEquals(expectedAlarm, decoder.decodeAlarm(status));
    }

    @Test
    public void decodeAlarmSOSTest() {
        int status = 16;
        String expectedAlarm = Position.ALARM_SOS;
        assertEquals(expectedAlarm, decoder.decodeAlarm(status));
    }

    @Test
    public void decodeAlarmFallDownTest() {
        int status = 21;
        String expectedAlarm = Position.ALARM_FALL_DOWN;
        assertEquals(expectedAlarm, decoder.decodeAlarm(status));
    }

    @Test
    public void decodeAlarmNoAlarmTest() {
        int status = 3;
        assertNull(decoder.decodeAlarm(status));
    }

    @Test
    public void getHasIndexTest() {
        assertFalse(decoder.getHasIndex());
    }

    @Test
    public void getManufacturerBeforeDecodeTest() {
        assertNull(decoder.getManufacturer());
    }

    @Test
    public void getManufacturerAfterDecodeTest() throws Exception {
        String expectedManufacturer = "TC";
        ByteBuf buf = Unpooled.copiedBuffer("[TC012345678910*LK]", StandardCharsets.US_ASCII);
        decoder.decode(null, null, buf);
        assertEquals(expectedManufacturer, decoder.getManufacturer());
    }

    @Test
    public void decodeUDTest() throws Exception {
        ByteBuf buf = Unpooled.copiedBuffer("[TC012345678910*UD,201119,A,4608.2035,N,00608.2035,E,000.8,0.0,10.0,4,2,10,20,1,0,abcd,a,b,c,d,e,]", StandardCharsets.US_ASCII);
        Position position = (Position) decoder.decode(null, null, buf);
        assertEquals(46.136725, position.getLatitude(), 0.000001d);
        assertEquals(6.136725, position.getLongitude(), 0.000001d);
    }

    @Test
    public void decodeALKTest() throws Exception {
        ByteBuf buf = Unpooled.copiedBuffer("[TC012345678910*AL,2]", StandardCharsets.US_ASCII);
        Position position = (Position) decoder.decode(null, null, buf);
        assertEquals(Position.ALARM_GEOFENCE_EXIT, position.getAttributes().get(Position.KEY_ALARM));
    }

    @Test
    public void decodeInvalidTest() throws Exception {
        ByteBuf buf = Unpooled.copiedBuffer("[TC012345678910*XX]", StandardCharsets.US_ASCII);
        assertNull(decoder.decode(null, null, buf));
    }

    @Test
    public void decodePositionInvalidDataTest() {
        DeviceSession deviceSession = Mockito.mock(DeviceSession.class);
        when(deviceSession.getDeviceId()).thenReturn(12345L);
        assertNull(decoder.decodePosition(deviceSession, "invalid"));
    }

    @Test
    public void decodePositionValidDataTest() {
        DeviceSession deviceSession = Mockito.mock(DeviceSession.class);
        when(deviceSession.getDeviceId()).thenReturn(12345L);
        Position position = decoder.decodePosition(deviceSession, "201119,092550,A,4608.2035,N,00608.2035,E,000.8,0.0,10.0,4,2,10,20,1");
        assertEquals(new Date(120, 10, 20, 9, 25, 50), position.getDeviceTime());
        assertEquals(12345L, position.getDeviceId().longValue());
    }
}

    }

    public static void tearDown() {

    }


    @Test
    public void decodeTk(){

        SocketAddress remoteAddress = new InetSocketAddress(InetAddress.getByName("localhost"), 55000);

        SessionManager sessionManager = Mockito.mock(SessionManager.class);
        DeviceSession deviceSession = Mockito.mock(DeviceSession.class);

        when(sessionManager.getDeviceSession(any(), any(String.class))

        )

        when(deviceSession.getDeviceId()).thenReturn(7800000000L);

        assertEquals(7800000000L, position.getDeviceId().longValue());

    }
