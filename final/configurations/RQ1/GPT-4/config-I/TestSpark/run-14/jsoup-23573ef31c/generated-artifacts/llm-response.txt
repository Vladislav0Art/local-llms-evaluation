package org.jsoup.nodes;

import org.jsoup.Connection;
import org.jsoup.Jsoup;
import org.jsoup.parser.Parser;
import org.jsoup.select.Elements;
import org.junit.Assert;
import org.junit.Test;

import java.nio.charset.StandardCharsets;
import java.util.List;

import static org.mockito.Mockito.mock;
import static org.mockito.Mockito.when;

public class DocumentTest {

    @Test
    public void DocumentTest() {
        Document document = new Document("http://test.com/");
        Assert.assertEquals("#document", document.nodeName());
    }

    @Test
    public void createShellTest() {
        Document document = Document.createShell("http://test.com/");
        Assert.assertNotNull(document);
    }

    @Test
    public void locationTest() {
        Document document = new Document("http://test.com/");
        String location = document.location();
        Assert.assertEquals("http://test.com/", location);
    }

    @Test
    public void documentTypeTest() {
        Document document = new Document("http://test.com/");
        Assert.assertNull(document.documentType());
    }

    @Test
    public void htmlElTest() {
        Document document = new Document("http://test.com/");
        Assert.assertNotNull(document.htmlEl());
    }

    @Test
    public void headTest() {
        Document document = new Document("http://test.com/");
        Assert.assertNotNull(document.head());
    }

    @Test
    public void bodyTest() {
        Document document = new Document("http://test.com/");
        Assert.assertNotNull(document.body());
    }

    @Test
    public void formsTest() {
        Document document = new Document("http://test.com/");
        List<FormElement> forms = document.forms();
        Assert.assertTrue(forms.isEmpty());
    }

    @Test(expected = IllegalArgumentException.class)
    public void expectFormTest() {
        Document document = new Document("http://test.com/");
        document.expectForm("test");
    }

    @Test
    public void titleTestDefault() {
        Document document = new Document("http://test.com/");
        Assert.assertEquals("", document.title());
    }

    @Test
    public void titleTestSet() {
        Document document = new Document("http://test.com/");
        document.title("TestTitle");
        Assert.assertEquals("TestTitle", document.title());
    }

    @Test
    public void outerHtmlTest() {
        Document document = new Document("http://test.com/");
        Assert.assertTrue(document.outerHtml().contains("<html>"));
    }

    @Test
    public void textTest() {
        Document document = new Document("http://test.com/");
        document.text("test text");
        Assert.assertEquals("test text", document.text());
    }

    @Test
    public void charsetTest() {
        Document document = new Document("http://test.com/");
        document.charset(StandardCharsets.UTF_8);
        Assert.assertEquals(StandardCharsets.UTF_8, document.charset());
    }

    @Test
    public void updateMetaCharsetElementTest() {
        Document document = new Document("http://test.com/");
        document.updateMetaCharsetElement(true);
        Assert.assertTrue(document.updateMetaCharsetElement());
    }

    @Test
    public void cloneTest() {
        Document document = new Document("http://test.com/");
        Assert.assertNotNull(document.clone());
    }

    @Test
    public void shallowCloneTest() {
        Document document = new Document("http://test.com/");
        Assert.assertNotNull(document.shallowClone());
    }

    @Test
    public void outputSettingsTest() {
        Document document = new Document("http://test.com/");
        Document.OutputSettings outputSettings = mock(Document.OutputSettings.class);
        when(outputSettings.prettyPrint()).thenReturn(true);
        document.outputSettings(outputSettings);
        Assert.assertTrue(document.outputSettings().prettyPrint());
    }
}
