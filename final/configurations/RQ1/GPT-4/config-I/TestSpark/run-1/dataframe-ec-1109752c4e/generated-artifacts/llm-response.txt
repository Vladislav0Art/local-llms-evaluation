import io.github.vmzakharov.ecdataframe.dsl.AliasExpr;
import io.github.vmzakharov.ecdataframe.dsl.AssingExpr;
import io.github.vmzakharov.ecdataframe.dsl.BinaryExpr;
import io.github.vmzakharov.ecdataframe.dsl.DecimalExpr;
import io.github.vmzakharov.ecdataframe.dsl.FunctionScript;
import io.github.vmzakharov.ecdataframe.dsl.FunctionCallExpr;
import io.github.vmzakharov.ecdataframe.dsl.VarExpr;
import io.github.vmzakharov.ecdataframe.dsl.visitor.PrettyPrintVisitor;
import io.github.vmzakharov.ecdataframe.dsl.value.DoubleValue;
import io.github.vmzakharov.ecdataframe.dsl.value.StringValue;
import io.github.vmzakharov.ecdataframe.util.CollectingPrinter;
import org.junit.Test;
import static org.junit.Assert.assertEquals;

public class PrettyPrintVisitorTest {

    @Test
    public void visitAssignExprTest() {
        AssingExpr expr = new AssingExpr("x", new DoubleValue(10.0));
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(new CollectingPrinter());
        expr.accept(visitor);
        assertEquals("x = 10", visitor.exprToString(expr));
    }

    @Test
    public void visitBinaryExprTest() {
        BinaryExpr expr = new BinaryExpr(new VarExpr("x"), "+", new DoubleValue(10.0));
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(new CollectingPrinter());
        expr.accept(visitor);
        assertEquals("(x + 10)", visitor.exprToString(expr));
    }

    @Test
    public void visitFunctionCallExprTest() {
        FunctionCallExpr expr = new FunctionCallExpr("sum", new StringValue("x"));
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(new CollectingPrinter());
        expr.accept(visitor);
        assertEquals("sum(\"x\")", visitor.exprToString(expr));
    }

    @Test
    public void visitFunctionScriptExprTest() {
        var vb = new VarExpr("b");
        var vc = new VarExpr("c");
        FunctionScript functionScript = new FunctionScript("func", "a")
            .addExpression(new AssingExpr("b", new VarExpr("a")))
            .addExpression(new AssingExpr("c", new VarExpr("a")));

        PrettyPrintVisitor visitor = new PrettyPrintVisitor();
        visitor.visitFunctionScriptExpr(functionScript);

        assertEquals("function func(a)\n{\n  b = a\n  c = a\n}", visitor.exprToString(functionScript));
    }

    @Test
    public void visitDecimalExprTest() {
        DecimalExpr expr = new DecimalExpr(new DoubleValue(10), new DoubleValue(12.0));
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(new CollectingPrinter());
        expr.accept(visitor);
        assertEquals("[10, 12]", visitor.exprToString(expr));
    }

    @Test
    public void escapeVarNameIfNeededTest() {
        AssingExpr expr = new AssingExpr("var", new DoubleValue(10.0), true);
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(new CollectingPrinter());
        if (visitor.escapeVarNameIfNeeded(expr.getVarName(), expr.isEscaped()).equals("${" + expr.getVarName() + "}")) {
            assertEquals("${var}", visitor.escapeVarNameIfNeeded(expr.getVarName(), expr.isEscaped()));
        }
    }
}

import io.github.vmzakharov.ecdataframe.dsl.*;
import io.github.vmzakharov.ecdataframe.dsl.visitor.PrettyPrintVisitor;
import io.github.vmzakharov.ecdataframe.util.CollectingPrinter;
import org.junit.Test;

import static org.junit.Assert.assertEquals;
import static org.mockito.Mockito.mock;

public class PrettyPrintVisitorTest {

    @Test
    public void visitAssignExprTest() {
        AssingExpr assingExpr = new AssingExpr("variable", mock(Expression.class));
        AssingExpr expr = new AssingExpr(assingExpr.getVarName(), assingExpr.getExpression());
        expr.setEscaped(true);
        CollectingPrinter printer = new CollectingPrinter();
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(printer);
        visitor.visitAssignExpr(expr);
        assertEquals("${variable} =", printer.toString());
    }

    @Test
    public void visitBinaryExprTest() {
        BinaryExpr binaryExpr = new BinaryExpr(mock(Expression.class), "*", mock(Expression.class));
        BinaryExpr expr = new BinaryExpr(binaryExpr.getOperand1(), binaryExpr.getOperation(), binaryExpr.getOperand2());
        CollectingPrinter printer = new CollectingPrinter();
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(printer);
        visitor.visitBinaryExpr(expr);
        assertEquals("*", printer.toString());
    }

    @Test
    public void visitFunctionCallExprTest() {
        FunctionCallExpr functionCallExpr = new FunctionCallExpr("sum", mock(Expression.class));
        FunctionCallExpr expr = new FunctionCallExpr(functionCallExpr.getFunctionName(), functionCallExpr.getParameters().get(0));
        CollectingPrinter printer = new CollectingPrinter();
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(printer);
        visitor.visitFunctionCallExpr(expr);
        assertEquals("sum", printer.toString());
    }

    @Test
    public void visitFunctionScriptExprTest() {
        FunctionScript functionScript = new FunctionScript("func");
        FunctionScript expr = new FunctionScript(functionScript.getName());
        CollectingPrinter printer = new CollectingPrinter();
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(printer);
        visitor.visitFunctionScriptExpr(expr);
        assertEquals("function func", printer.toString());
    }

    @Test
    public void visitDecimalExprTest() {
        DecimalExpr decimalExpr = new DecimalExpr(mock(Expression.class), mock(Expression.class));
        DecimalExpr expr = new DecimalExpr(decimalExpr.unscaledValueExpr(), decimalExpr.scaleExpr());
        CollectingPrinter printer = new CollectingPrinter();
        PrettyPrintVisitor visitor = new PrettyPrintVisitor(printer);
        visitor.visitDecimalExpr(decimalExpr);
        assertEquals("[]", printer.toString());
    }
}
