import static org.junit.Assert.*;

import leetcode.medium.OnlineStockSpan;
import org.junit.Test;

public class OnlineStockSpanTest {

  @Test
  public void nextNewPriceTest() {
    OnlineStockSpan stockSpan = new OnlineStockSpan();
    int result = stockSpan.next(100);
    assertEquals(1, result);
  }

  @Test
  public void nextPriceIncrementTest() {
    OnlineStockSpan stockSpan = new OnlineStockSpan();
    stockSpan.next(100);
    int result = stockSpan.next(101);
    assertEquals(1, result);
  }

  @Test
  public void nextPriceDecrementTest() {
    OnlineStockSpan stockSpan = new OnlineStockSpan();
    stockSpan.next(100);
    int result = stockSpan.next(99);
    assertEquals(2, result);
  }

  @Test
  public void nextEqualPriceTest() {
    OnlineStockSpan stockSpan = new OnlineStockSpan();
    stockSpan.next(100);
    int result = stockSpan.next(100);
    assertEquals(2, result);
  }
  
  @Test
  public void calculateSpansSingleElementTest() {
    OnlineStockSpan stockSpan = new OnlineStockSpan();
    int[] prices = new int[] {100};
    int[] spans = stockSpan.calculateSpans(prices);
    assertArrayEquals(new int[] {1}, spans);
  }

  @Test
  public void calculateSpansMultipleElementTest() {
    OnlineStockSpan stockSpan = new OnlineStockSpan();
    int[] prices = new int[] {100, 103, 99, 98, 105};
    int[] spans = stockSpan.calculateSpans(prices);
    assertArrayEquals(new int[] {1, 2, 1, 1, 5}, spans);
  }

  @Test(expected = ArrayIndexOutOfBoundsException.class)
  public void calculateSpansEmptyArrayTest() {
    OnlineStockSpan stockSpan = new OnlineStockSpan();
    int[] prices = new int[] {};
    stockSpan.calculateSpans(prices);
  }
}
