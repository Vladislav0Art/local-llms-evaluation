import org.jsoup.Connection;
import org.jsoup.parser.Parser;
import org.jsoup.select.Elements;
import org.junit.Assert;
import org.junit.Test;
import org.mockito.Mockito;

import java.nio.charset.Charset;
import java.nio.charset.StandardCharsets;
import java.util.List;

import static org.mockito.Mockito.*;

public class DocumentTest {

    @Test
    public void DocumentConstructorTest() {
        Document document = new Document("http://test.com");
        Assert.assertEquals("http://test.com", document.location());
    }

    @Test
    public void createShellTest() {
        Document document = Document.createShell("http://test.com");
        Assert.assertEquals("http://test.com", document.location());
        Assert.assertEquals("<html>\n <head></head>\n <body></body>\n</html>", document.html());
    }

    @Test
    public void locationTest() {
        Document document = new Document("http://test.com");
        Assert.assertEquals("http://test.com", document.location());
    }

    @Test
    public void connectionTest() {
        Document document = new Document("http://test.com");
        Connection connection = Mockito.mock(Connection.class);
        document.connection(connection);
        Assert.assertEquals(connection, document.connection());
    }

    @Test
    public void titleTest() {
        Document document = Document.createShell("http://test.com");
        document.title("Test Title");
        Assert.assertEquals("Test Title", document.title());
    }

    @Test
    public void charsetTest() {
        Document document = new Document("http://test.com");
        document.charset(StandardCharsets.UTF_16);
        Assert.assertEquals(StandardCharsets.UTF_16, document.charset());
    }

    @Test
    public void updateMetaCharsetElementTest() {
        Document document = new Document("http://test.com");
        document.updateMetaCharsetElement(true);
        Assert.assertTrue(document.updateMetaCharsetElement());
    }

    @Test
    public void parserTest() {
        Document document = new Document("http://test.com");
        Parser parser = Mockito.mock(Parser.class);
        document.parser(parser);
        Assert.assertEquals(parser, document.parser());
    }

    @Test
    public void cloneTest() {
        Document document = new Document("http://test.com");
        Document cloned = document.clone();
        Assert.assertNotSame(document, cloned);
        Assert.assertEquals(document.location(), cloned.location());
    }

    @Test
    public void shallowCloneTest() {
        Document document = new Document("http://test.com");
        Document shallowCloned = document.shallowClone();
        Assert.assertNotSame(document, shallowCloned);
        Assert.assertEquals(document.location(), shallowCloned.location());
    }
    
    @Test
    public void formsTest() {
        Document document = new Document("http://test.com");
        List<FormElement> forms = document.forms();
        Assert.assertNotNull(forms);
    }

    @Test(expected = IllegalArgumentException.class)
    public void expectFormTest() {
        Document document = new Document("http://test.com");
        document.expectForm("form");
    }

    @Test
    public void createElementTest() {
        Document document = new Document("http://test.com");
        Element element = document.createElement("div");
        Assert.assertEquals("div", element.tagName());
    }
}
