import org.jsoup.Connection;
import org.junit.Assert;
import org.junit.Test;
import java.net.MalformedURLException;
import java.net.URL;

import static org.mockito.Mockito.mock;
import static org.mockito.Mockito.when;

public class UrlBuilderTest {

    @Test
    public void buildTest_withQuery() throws MalformedURLException {
        UrlBuilder testInstance = new UrlBuilder(new URL("https://www.example.com/test?param=value"));
        URL resultUrl = testInstance.build();
        Assert.assertEquals("https://www.example.com/test?param=value", resultUrl.toString());
    }

    @Test
    public void buildTest_WithoutQuery() throws MalformedURLException {
        UrlBuilder testInstance = new UrlBuilder(new URL("https://www.example.com/test"));
        URL resultUrl = testInstance.build();
        Assert.assertEquals("https://www.example.com/test", resultUrl.toString());
    }

    @Test
    public void buildTest_WithSpecialChars() throws MalformedURLException {
        UrlBuilder testInstance = new UrlBuilder(new URL("https://www.example.com/test/äöü?param=value"));
        URL resultUrl = testInstance.build();
        Assert.assertEquals("https://www.example.com/test/%C3%A4%C3%B6%C3%BC?param=value", resultUrl.toString());
    }

    @Test
    public void appendKeyValTest_NewKeyVal() throws MalformedURLException {
        UrlBuilder testInstance = new UrlBuilder(new URL("https://www.example.com/test"));
        Connection.KeyVal kv = mock(Connection.KeyVal.class);
        when(kv.key()).thenReturn("param");
        when(kv.value()).thenReturn("value");
        testInstance.appendKeyVal(kv);
        URL resultUrl = testInstance.build();
        Assert.assertEquals("https://www.example.com/test?param=value", resultUrl.toString());
    }

    @Test
    public void appendKeyValTest_AddToExisting() throws MalformedURLException {
        UrlBuilder testInstance = new UrlBuilder(new URL("https://www.example.com/test?param=value"));
        Connection.KeyVal kv = mock(Connection.KeyVal.class);
        when(kv.key()).thenReturn("param2");
        when(kv.value()).thenReturn("value2");
        testInstance.appendKeyVal(kv);
        URL resultUrl = testInstance.build();
        Assert.assertEquals("https://www.example.com/test?param=value&param2=value2", resultUrl.toString());
    }
}
