```java
import org.davidmoten.text.utils.WordWrap;
import org.davidmoten.text.utils.WordWrap.Builder;
import org.junit.Test;
import java.io.StringReader;
import java.util.Set;
import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertTrue;

public class WordWrapTest {

	@Test
	public void fromReaderTest() {
		Builder builder = WordWrap.from(new StringReader("This is a sample text."));
		assertEquals(80, builder.maxWidth.intValue());
	}

	@Test
	public void fromClasspathUtf8Test() {
		Builder builder = WordWrap.fromClasspathUtf8("sample.txt");
		assertEquals(80, builder.maxWidth.intValue());
	}

	@Test
	public void fromTextTest() {
		Builder builder = WordWrap.from("This is a sample text.");
		assertEquals(80, builder.maxWidth.intValue());
	}

	@Test
	public void maxWidthTest() {
		Builder builder = WordWrap.from("This is a sample text.").maxWidth(100);
		assertEquals(100, builder.maxWidth.intValue());
	}

	@Test
	public void newLineTest() {
		Builder builder = WordWrap.from("This is a sample text.").newLine("--");
		assertEquals("--", builder.newLine);
	}

	@Test
	public void breakWordsTest() {
		Builder builder = WordWrap.from("This is a sample text.").breakWords(false);
		assertFalse(builder.breakWords);
	}

	@Test
	public void extraWordCharsTest() {
		Set<Character> chars = new HashSet<>();
		chars.add('a');
		Builder builder = WordWrap.from("This is a sample text.").extraWordChars(chars);
		assertEquals(chars, builder.extraWordChars);
	}

	@Test
	public void wrapTest() {
		String result = WordWrap.from("This is a sample text.").wrap();
		assertTrue(result instanceof String);
	}

	@Test
	public void wrapWithHyphenTest() {
		String result = WordWrap.from("This-really-long-word-with-hyphen test text").insertHyphens(true).wrap();
		assertEquals("This-really-long-word-with-hyphen test text", result);
	}

	@Test
	public void wrapWithoutHyphenTest() {
		String result = WordWrap.from("This really long word without hyphen test text").insertHyphens(false).wrap();
		assertEquals("This really long word without hyphen test text", result);
	}
    
    @Test(expected = IllegalArgumentException.class)
    public void maxWidthLessThanEqualToZeroTest() {
        WordWrap.from("Test").maxWidth(0);
    }
}
```

```java
import org.davidmoten.text.utils.WordWrap;
import org.junit.Test;
import java.io.StringReader;
import java.util.Arrays;
import static org.junit.Assert.*;

public class WordWrapTest {

	@Test
	public void fromReaderTest() {
		String result = WordWrap.from(new StringReader("This is a sample text.")).wrap();
		assertEquals("This is a sample text.", result);
	}

	@Test
	public void fromTextTest() {
		String result = WordWrap.from("This is a sample text.").wrap();
		assertEquals("This is a sample text.", result);
	}

	@Test
	public void maxWidthTest() {
		String result = WordWrap.from("This is a sample text with more than thirty characters.").maxWidth(30).wrap();
		assertTrue(Arrays.stream(result.split("\n")).allMatch(s -> s.length() <= 30));
	}

	@Test
	public void newLineTest() {
		String result = WordWrap.from("This is a sample text.\nThis is a second line.").newLine("--").wrap();
		assertTrue(result.contains("--"));
	}

	@Test
	public void breakWordsTest() {
		String result = WordWrap.from("ThisIsAReallyLongWord ThatCannotFitInALine").breakWords(true).maxWidth(10).wrap();
		assertTrue(result.contains("ThisIsARea-\n" + "llyLongWor-\n" + "d"));
	}

	@Test
	public void insertHyphensTest() {
		String result = WordWrap.from("ThisIsAReallyLongWord ThatCannotFitInALine").breakWords(true).insertHyphens(false).maxWidth(10).wrap();
		assertTrue(result.contains("ThisIsARea\n" + "llyLongWor\n" + "d"));
	}

	@Test
	public void wrapTest() {
		String result = WordWrap.from("This is a sample text.").wrap();
		assertTrue(result instanceof String);
	}

    @Test(expected = IllegalArgumentException.class)
    public void maxWidthLessThanEqualToZeroTest() {
        WordWrap.from("Test").maxWidth(0).wrap();
    }
}
```
