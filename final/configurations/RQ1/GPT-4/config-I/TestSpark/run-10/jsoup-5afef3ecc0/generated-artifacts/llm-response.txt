import org.junit.Test;
import static org.junit.Assert.*;
import org.jsoup.Connection;
import java.net.MalformedURLException;
import java.net.URL;
import org.jsoup.helper.UrlBuilder;

public class UrlBuilderTest {
    
    @Test
    public void buildTest_NormalizationNeeded() throws MalformedURLException {
        URL url = new URL("http://www.domain.com/path?a=b");
        UrlBuilder builder = new UrlBuilder(url);
        URL result = builder.build();

        assertEquals("http", result.getProtocol());
        assertEquals("www.domain.com", result.getHost());
        assertEquals("/path", result.getPath());
        assertEquals("a=b", result.getQuery());
    }

    @Test
    public void buildTest_NoNormalizationNeeded() throws MalformedURLException {
        URL url = new URL("http://www.domain.com/path");
        UrlBuilder builder = new UrlBuilder(url);
        URL result = builder.build();

        assertEquals("http", result.getProtocol());
        assertEquals("www.domain.com", result.getHost());
        assertEquals("/path", result.getPath());
        assertNull(result.getQuery());
    }

    @Test(expected = RuntimeException.class)
    public void buildTest_MalformedUrl() throws MalformedURLException {
        URL url = new URL("htp:/www.domain.com");
        UrlBuilder builder = new UrlBuilder(url);
        builder.build();
    }

    @Test
    public void appendKeyValTest_NewQueryParameter() throws MalformedURLException {
        URL url = new URL("http://www.domain.com/path");
        UrlBuilder builder = new UrlBuilder(url);

        builder.appendKeyVal(new Connection.KeyVal("key", "value"));
        URL result = builder.build();

        assertEquals("http", result.getProtocol());
        assertEquals("www.domain.com", result.getHost());
        assertEquals("/path", result.getPath());
        assertEquals("key=value", result.getQuery());
    }

    @Test
    public void appendKeyValTest_AppendToExistingQueryParameter() throws MalformedURLException {
        URL url = new URL("http://www.domain.com/path?a=b");
        UrlBuilder builder = new UrlBuilder(url);

        builder.appendKeyVal(new Connection.KeyVal("key", "value"));
        URL result = builder.build();

        assertEquals("http", result.getProtocol());
        assertEquals("www.domain.com", result.getHost());
        assertEquals("/path", result.getPath());
        assertFalse(result.getQuery().contains("&"));
        assertTrue(result.getQuery().contains("a=b"));
        assertTrue(result.getQuery().contains("key=value"));
    }
}
