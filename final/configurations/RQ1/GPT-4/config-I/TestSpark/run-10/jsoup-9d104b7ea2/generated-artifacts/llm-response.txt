package org.jsoup.nodes;

import org.junit.Assert;
import org.junit.Test;

public class TextNodeTests {
    
    @Test
    public void nodeNameTest() {
        TextNode textNode = new TextNode("Test Node");
        Assert.assertEquals("#text", textNode.nodeName());
    }

    @Test
    public void textTest() {
        TextNode textNode = new TextNode(" Test Node  ");
        Assert.assertEquals("Test Node", textNode.text());
    }
    
    @Test
    public void getWholeTextTest() {
        TextNode textNode = new TextNode(" Test Node  ");
        Assert.assertEquals(" Test Node  ", textNode.getWholeText());
    }

    @Test
    public void textAfterSetTest() {
        TextNode textNode = new TextNode("Test Node");
        Assert.assertEquals("New Node Text", textNode.text("New Node Text").text());
    }
    
    @Test
    public void isBlankTrueTest() {
        TextNode textNode = new TextNode(" ");
        Assert.assertEquals(true, textNode.isBlank());
    }
    
    @Test
    public void isBlankFalseTest() {
        TextNode textNode = new TextNode("Test Node");
        Assert.assertEquals(false, textNode.isBlank());
    }
    
    @Test
    public void splitTextNodeTest() {
        TextNode textNode = new TextNode("TestNodeSplit");
        TextNode splitNode = textNode.splitText(8);
        Assert.assertEquals("TestNode", textNode.text());
        Assert.assertEquals("Split", splitNode.text());
    }
    
    @Test(expected = IllegalArgumentException.class)
    public void splitTextNodeNegativeOffsetTest() {
        TextNode textNode = new TextNode("Test Node");
        textNode.splitText(-2);
    }
    
    @Test(expected = IllegalArgumentException.class)
    public void splitTextNodeBigOffsetTest() {
        TextNode textNode = new TextNode("Test Node");
        textNode.splitText(100);
    }
    
    @Test
    public void cloneNodeTest() {
        TextNode textNode = new TextNode("Test Node");
        Assert.assertNotSame(textNode, textNode.clone());
        Assert.assertEquals(textNode.text(), textNode.clone().text());
    }

    @Test
    public void createFromEncodedTest() {
        String htmlEncoded = "&lt;p&gt;Hello World&lt;/p&gt;";
        TextNode textNode = TextNode.createFromEncoded(htmlEncoded);
        Assert.assertEquals("<p>Hello World</p>", textNode.getWholeText());
    }

    @Test
    public void normaliseWhiteSpaceTest() {
        String text = " Test\nNode  ";
        Assert.assertEquals("Test Node", TextNode.normaliseWhitespace(text));
    }

    @Test
    public void stripLeadingWhiteSpace() {
        String text = "  Test Node";
        Assert.assertEquals("Test Node", TextNode.stripLeadingWhitespace(text));
    }

    @Test
    public void lastCharIsWhiteSpace() {
        StringBuilder sb = new StringBuilder("Test Node ");
        Assert.assertTrue(TextNode.lastCharIsWhitespace(sb));
    }
}
