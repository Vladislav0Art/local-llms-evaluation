import org.jsoup.parser.ParseSettings;
import org.jsoup.parser.Tag;
import org.junit.Assert;
import org.junit.Test;

import java.lang.reflect.Field;

public class TagTests {

    @Test
    public void getNameAndNormalNameTest() throws Exception {
        Tag tag = new Tag("SomeTag");
        Assert.assertEquals("SomeTag", tag.getName());
        Assert.assertEquals("sometag", tag.normalName());
    }

    @Test
    public void valueOfWithSettingsTest() throws Exception {
        Tag tag = Tag.valueOf("SomeTagWithSettings", ParseSettings.htmlDefault);
        Assert.assertEquals("sometagwithsettings", tag.normalName());
    }

    @Test(expected = IllegalArgumentException.class)
    public void valueOfWithSettingsEmptyTagNameTest() throws Exception {
        Tag.valueOf("", ParseSettings.htmlDefault);
    }

    @Test
    public void valueOfTest() throws Exception {
        Tag tag = Tag.valueOf("SomeTagWithoutSettings");
        Assert.assertEquals("sometagwithoutsettings", tag.normalName());
    }

    @Test
    public void isBlockAndFormatAsBlockTest() throws Exception {
        Tag tag = new Tag("randomTag");
        Assert.assertFalse(tag.isBlock());
        Assert.assertTrue(tag.formatAsBlock());
        
        tag = Tag.valueOf("pre", ParseSettings.htmlDefault);
        Assert.assertTrue(tag.isBlock());
        Assert.assertTrue(tag.formatAsBlock());
    }

    @Test
    public void isInlineTest() {
        Tag tag = new Tag("randomTag");
        Assert.assertTrue(tag.isInline());

        tag = Tag.valueOf("pre", ParseSettings.htmlDefault);
        Assert.assertFalse(tag.isInline());
    }

    @Test
    public void isEmptyTest() {
        Tag tag = new Tag("randomTag");
        Assert.assertFalse(tag.isEmpty());

        tag = Tag.valueOf("img", ParseSettings.htmlDefault);
        Assert.assertTrue(tag.isEmpty());
    }

    @Test
    public void isSelfClosingTest() {
        Tag tag = new Tag("randomTag");
        Assert.assertFalse(tag.isSelfClosing());

        tag = Tag.valueOf("img", ParseSettings.htmlDefault);
        Assert.assertTrue(tag.isSelfClosing());
    }

    @Test
    public void isKnownTagTest() throws NoSuchFieldException, IllegalAccessException {
        Tag tag = new Tag("unknownTag");
        Assert.assertFalse(tag.isKnownTag());

        Field tagsField = Tag.class.getDeclaredField("tags");
        tagsField.setAccessible(true);
        ((Map<String, Tag>)tagsField.get(null)).put("unknownTag", tag);

        Assert.assertTrue(Tag.isKnownTag("unknownTag"));
    }

    @Test
    public void preserveWhitespaceTest() {
        Tag tag = Tag.valueOf("unknownTag");
        Assert.assertFalse(tag.preserveWhitespace());

        tag = Tag.valueOf("pre", ParseSettings.htmlDefault);
        Assert.assertTrue(tag.preserveWhitespace());
    }

    @Test
    public void isFormListedTest() {
        Tag tag = Tag.valueOf("unknownTag");
        Assert.assertFalse(tag.isFormListed());

        tag = Tag.valueOf("button", ParseSettings.htmlDefault);
        Assert.assertTrue(tag.isFormListed());
    }

    @Test
    public void isFormSubmittableTest() {
        Tag tag = Tag.valueOf("unknownTag");
        Assert.assertFalse(tag.isFormSubmittable());

        tag = Tag.valueOf("button", ParseSettings.htmlDefault);
        Assert.assertTrue(tag.isFormSubmittable());
    }
}
