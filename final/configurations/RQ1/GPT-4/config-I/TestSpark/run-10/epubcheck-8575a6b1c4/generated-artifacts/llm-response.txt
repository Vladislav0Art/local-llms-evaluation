package com.adobe.epubcheck.tool;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertNotNull;
import static org.mockito.Mockito.*;

import java.io.File;
import java.util.Locale;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;
import org.w3c.epubcheck.core.Checker;
import org.w3c.epubcheck.util.Archive;

import com.adobe.epubcheck.api.EpubCheck;

@RunWith(MockitoJUnitRunner.class)
public class EpubCheckerTest {

    @Mock
    private EpubCheck epubCheckMock;

    @Test
    public void getLocaleDefaultLocaleTest() {
        EpubChecker epubChecker = new EpubChecker();
        Locale locale = epubChecker.getLocale();
        assertNotNull(locale);
        assertEquals(Locale.getDefault(), locale);
    }

    @Test(expected = RuntimeException.class)
    public void runRunWithNoArgsTest() {
        String[] args = {};
        EpubChecker epubChecker = new EpubChecker();
        epubChecker.run(args);
    }

    @Test
    public void processEpubFileRunWithSomeArgsTest() {
        String[] args = { "-v", "3.0", "someFilePath" };
        EpubChecker epubChecker = new EpubChecker();
        epubChecker.processEpubFile(args);
    }

    @Test(expected = RuntimeException.class)
    public void runRunWithWrongArgsTest() {
        String[] args = { "-v", "5.0", "someFilePath" };
        EpubChecker epubChecker = new EpubChecker();
        epubChecker.run(args);
    }

    @Test
    public void validateFileFileExistsHTTPSTest() throws Exception {
        EpubChecker epubChecker = new EpubChecker();
        int result = epubChecker.validateFile("https://google.com", EPUBVersion.VERSION_2, mock(Report.class), EPUBProfile.DEFAULT);
        assertEquals(1, result);
    }
    
    @Test
    public void validateFileFileDoesNotExistTest() {
        EpubChecker epubChecker = new EpubChecker();
        int result = epubChecker.validateFile("nonExistentfilepath", EPUBVersion.VERSION_2, mock(Report.class), EPUBProfile.DEFAULT);
        assertEquals(1, result);
    }
        
    @Test
    public void processFileExpandedTest() throws Exception {
        EpubChecker epubChecker = spy(EpubChecker.class);
        epubChecker.expanded = true;
        epubChecker.path = System.getProperty("java.io.tmpdir");
        epubChecker.validateFile = mock(Archive.class);
        epubChecker.processFile(mock(Report.class));
        verify(epubChecker, times(1)).validateFile(anyString(), any(EPUBVersion.class), any(Report.class), any(EPUBProfile.class));
    }

    @Test
    public void processFileNonExpandedTest() throws Exception {
        EpubChecker epubChecker = spy(EpubChecker.class);
        epubChecker.expanded = false;
        epubChecker.path = "foo.epub";
        epubChecker.processFile(mock(Report.class));
        verify(epubChecker, times(1)).validateFile(anyString(), any(EPUBVersion.class), any(Report.class), any(EPUBProfile.class));
    }
    
    @Test
    public void validateFileCheckerInstanceEpubCheckTest() throws Exception {
        EpubChecker checker = new EpubChecker();
        int result = checker.validateFile(new File("src/test/resources/epub/valid.epub").getAbsolutePath(), 
            EPUBVersion.VERSION_3, mock(Report.class), EPUBProfile.DEFAULT);
        assertEquals(0, result);
    }

    @Test
    public void createXmlReportTest() throws Exception {
        EpubChecker epubChecker = new EpubChecker();
        epubChecker.xmlOutput = true;
        epubChecker.path = "unfound.epub";
        Report report = epubChecker.createReport();
        assertNotNull(report);
    }
    
    @Test
    public void createJsonReportTest() throws Exception {
        EpubChecker epubChecker = new EpubChecker();
        epubChecker.jsonOutput = true;
        epubChecker.path = "unfound.epub";
        Report report = epubChecker.createReport();
        assertNotNull(report);
    }
}
