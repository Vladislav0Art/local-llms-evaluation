Here is an example of unit tests for `org.jsoup.helper.UrlBuilder` class. It covers the following requirements:

* No `@Before` and `@After` test methods.
* Test cases are atomic.
* All tests are written using JUnit 4.
* Mocking is used only for `decodePart(String)` method.
* Method names follow the template: `[MethodUnderTest][Scenario]Test` using only English letters.

```java
import org.jsoup.helper.UrlBuilder;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;
import java.net.URL;
import static org.mockito.Mockito.when;
import static org.junit.Assert.assertEquals;

@RunWith(MockitoJUnitRunner.class)
public class UrlBuilderTest {

    @Mock
    private UrlBuilder urlBuilder;

    @Test
    public void buildWithNullQueryTest() throws Exception {
        URL inputUrl = new URL("http://example.com");
        URL expectedUrl = new URL("http://example.com");

        when(urlBuilder.decodePart("example.com")).thenReturn("example.com");

        URL actualUrl = urlBuilder.build(inputUrl);
        assertEquals(expectedUrl, actualUrl);
    }

    @Test
    public void buildWithQueryTest() throws Exception {
        URL inputUrl = new URL("http://example.com?q=test");

        when(urlBuilder.decodePart("example.com")).thenReturn("example.com");

        URL actualUrl = urlBuilder.build(inputUrl);
        assertEquals(expectedUrl, actualUrl);
    }
