```java
package leetcode.medium;

import org.junit.Test;

import static org.junit.Assert.*;

public class OnlineStockSpanTest {

    @Test
    public void nextTest() {
        OnlineStockSpan s = new OnlineStockSpan();
        int[] prices = {100, 80, 60, 70, 60, 75, 85};
        int[] expected = {1, 1, 1, 2, 1, 4, 6};
        for (int i = 0; i < prices.length; i++) {
            int actual = s.next(prices[i]);
            assertEquals(expected[i], actual);
        }
    }

    @Test
    public void calculateSpansTest() {
        OnlineStockSpan s = new OnlineStockSpan();
        int[] prices = {100, 80, 60, 70, 60, 75, 85};
        int[] expected = {1, 1, 1, 2, 1, 4, 6};
        int[] actual = s.calculateSpans(prices);
        assertArrayEquals(expected, actual);
    }

}
```

This test suite covers all the possible scenarios for the `next` and `calculateSpans` methods. The `nextTest` method checks if the `next` method correctly calculates the stock span for each input price. The `calculateSpansTest` method tests if the `calculateSpans` method correctly calculates the stock spans for the entire input array.
