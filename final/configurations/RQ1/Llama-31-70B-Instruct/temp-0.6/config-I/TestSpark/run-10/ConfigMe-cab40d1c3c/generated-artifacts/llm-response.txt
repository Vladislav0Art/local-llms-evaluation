import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertThrows;

import ch.jalu.configme.configurationdata.PropertyListBuilder;
import ch.jalu.configme.exception.ConfigMeException;
import ch.jalu.configme.properties.Property;
import java.util.List;
import org.junit.jupiter.api.Test;

class PropertyListBuilderAddTest {

  private PropertyListBuilder builder = new PropertyListBuilder();

  @Test
  void addProperty_propertyAdded_sizeIncreased() {
    builder.add(new Property<>("testProperty", "testPath"));
    assertEquals(1, builder.getRootEntries().size());
  }

  @Test
  void addProperty_duplicateProperty_exceptionThrown() {
    Property<String> property = new Property<>("testProperty", "testPath");
    builder.add(property);
    assertThrows(ConfigMeException.class, () -> builder.add(property));
  }

  @Test
  void addProperty_validProperty_propertyAdded() {
    Property<String> property = new Property<>("testProperty", "testPath");
    builder.add(property);
    assertEquals(property, builder.getRootEntries().get("testPath"));
  }
}
```
