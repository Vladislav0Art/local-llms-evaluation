import java.net.URL;
import static org.junit.jupiter.api.Assertions.*;

class UrlBuilderTest {
    private UrlBuilder urlBuilder;

    @BeforeEach
    void setUp() throws MalformedURLException {
        urlBuilder = new UrlBuilder(new URL("https://www.example.com"));
    }

    @Test
    void constructor() {
        assertNotNull(urlBuilder);
    }

    @Test
    void build() {
        assertNotNull(urlBuilder.build());
    }

    @Test
    void buildWithQuery() throws MalformedURLException {
        urlBuilder = new UrlBuilder(new URL("https://www.example.com?param=value"));
        assertNotNull(urlBuilder.build());
    }

    @Test
    void buildWithRef() throws MalformedURLException {
        urlBuilder = new UrlBuilder(new URL("https://www.example.com#ref"));
        assertNotNull(urlBuilder.build());
    }

    @Test
    void buildWithQueryAndRef() throws MalformedURLException {
        urlBuilder = new UrlBuilder(new URL("https://www.example.com?param=value#ref"));
        assertNotNull(urlBuilder.build());
    }
}
