package ch.jalu.configme.configurationdata;

import ch.jalu.configme.exception.ConfigMeException;
import ch.jalu.configme.properties.Property;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.junit.MockitoJUnitRunner;

import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

import static org.junit.Assert.assertEquals;
import static org.junit.Assert.assertThrows;
import static org.mockito.Mockito.when;

@RunWith(MockitoJUnitRunner.class)
public class PropertyListBuilderTest {

    @InjectMocks
    private PropertyListBuilder builder;

    @Test
    public void addProperty_NoExistingPath_AddsProperty() {
        Property<String> property = new Property<>("test", "value");
        builder.add(property);
        assertEquals(1, builder.create().size());
    }

    @Test
    public void addProperty_ExistingPath_ThrowsException() {
        Property<String> property = new Property<>("test", "value");
        builder.add(property);
        assertThrows(ConfigMeException.class, () -> builder.add(property));
    }

    @Test
    public void addProperty_NullProperty_ThrowsException() {
        assertThrows(NullPointerException.class, () -> builder.add(null));
    }

    @Test
    public void create_EmptyBuilder_ReturnsEmptyList() {
        assertEquals(0, builder.create().size());
    }

    @Test
    public void create_AddedProperties_ReturnsAddedProperties() {
        Property<String> property1 = new Property<>("test1", "value1");
        Property<String> property2 = new Property<>("test2", "value2");
        builder.add(property1);
        builder.add(property2);
        assertEquals(2, builder.create().size());
    }

    @Test
    public void create_AddedProperties_GroupedByPath() {
        Property<String> property1 = new Property<>("test1", "value1");
        Property<String> property2 = new Property<>("test1.sub", "value2");
        Property<String> property3 = new Property<>("test2", "value3");
        builder.add(property1);
        builder.add(property2);
        builder.add(property3);
        List<Property<?>> result = builder.create();
        assertEquals(3, result.size());
        assertEquals("test1", result.get(0).getPath());
        assertEquals("test1.sub", result.get(1).getPath());
        assertEquals("test2", result.get(2).getPath());
