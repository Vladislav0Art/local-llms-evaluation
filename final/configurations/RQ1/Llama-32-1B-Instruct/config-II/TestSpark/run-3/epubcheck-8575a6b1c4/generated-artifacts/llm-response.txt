public static class EpubCheck {
    private String version;
    private String locale;

    public String getVersion() {
        return version;
    }

    public void setVersion(String version) {
        this.version = version;
    }

    public String getLocale() {
        return locale;
    }

    public void setLocale(String locale) {
        this.locale = locale;
    }
}

public class EpubCheckCommand {
    private EpubCheck epubCheck;

    public static void main(String[] args) throws Exception {
        new EpubCheck().run(args);
    }

    public void run(String[] args) throws Exception {
        if (args.length < 1 || args[0].equals("--version")) {
            displayVersion();
            return;
        }
        String path = null;

        for (String arg : args) {
            if (arg.startsWith("-") && !arg.startsWith("--")) {
                System.err.println(String.format(messages.get("unrecognized_argument"), arg));
                displayHelp();
                return;
            } else if (arg.equals("--help") || arg.equals("?")) {
                displayHelp();
                return;
            }

            String[] parts = arg.split("\\+");
            if (parts.length == 1) {
                path = null;
            } else if (parts[0].equals("path")) {
                path = parts[1];
            } else {
                epubCheck = new EpubCheck();
                epubCheck.setVersion(parts[0]);
                epubCheck.setLocale(parts[1]);

                switch (parts[2]) {
                    case "mode":
                        if (args.length < 4) {
                            System.err.println(String.format(messages.get("unrecognized_argument"), parts[3]));
                            displayHelp();
                            return;
                        }
                        String mode = args[3];
                        epubCheck.setMode(mode);
                        break;
                    case "version":
                        epubCheck.setVersion(parts[3]);
                        break;
                }

                switch (parts[2]) {
                    case "list_checks":
                        if (args.length < 4) {
                            System.err.println(String.format(messages.get("unrecognized_argument"), parts[3]));
                            displayHelp();
                            return;
                        }
                        String listChecksOut = args[3];
                        epubCheck.setListChecks(listChecksOut);
                        break;
                    case "custom_messages":
                        if (args.length < 4) {
                            System.err.println(String.format(messages.get("unrecognized_argument"), parts[3]));
                            displayHelp();
                            return;
                        }
                        String customMessageFile = args[3];
                        epubCheck.setCustomMessageFile(customMessageFile);
                        break;
                    case "locale":
                        if (args.length < 2) {
                            System.err.println(String.format(messages.get("unrecognized_argument"), parts[1]));
                            displayHelp();
                            return;
                        }
                        String localeTag = args[1];
                        epubCheck.setLocale(localeTag);
                        break;

                    default:
                        if (path == null || path.matches("\\.[Ee][Pp][Uu][Bb]")) {
                            System.err.println(String.format(messages.get("mode_version_ignored"), parts[2]));
                            continue;
                        }
                }

                testMatch(args, path);

                epubCheck.setCustomMessageFileFromEnvironment();
            }
        }
    }

    private void testMatch(String[] args, String path) {
        for (String arg : args) {
            if (arg.startsWith("-") && !arg.startsWith("--")) {
                System.err.println(String.format(messages.get("unrecognized_argument"), arg));
                displayHelp();
                return;
            } else if (arg.equals("--help") || arg.equals("?")) {
                displayHelp();
                return;
            }

            String[] parts = arg.split("\\+");
            if (!parts.length) {
                continue;
            }
        }
    }
}
