/**
 * A utility class for serializing HTTP requests into URL strings.
 */
public class HttpRequestSerializer {

    /**
     * Serializes an object into a URL string.
     *
     * @param obj the object to serialize
     * @return the serialized URL string
     */
    public static String serialize(Object obj) {
        // Use the String.format method to construct the URL string
        return String.format(
            "GET %s HTTP/1.1\r\n",
            HttpRequestSerializer.buildUrl(obj)
        );
    }

    /**
     * Builds a URL string for an object of type KeyVal.
     *
     * @param obj the object to serialize
     * @return the serialized URL string
     */
    private static String buildUrl(KeyVal obj) {
        // Get the query parameters from the key-value pairs
        Map<String, List<Connection.KeyVal>> queryParams = getQueryParams(obj);

        // Create a query string with the params
        StringBuilder queryStringBuilder = new StringBuilder();
        for (Map.Entry<String, List<Connection.KeyVal>> entry : queryParams.entrySet()) {
            queryStringBuilder.append(entry.getKey()).append("=").append(String.join("&", entry.getValue().map(KeyVal::key)).append("\r\n");
        }

        // Add the query string to the URL
        String url = "GET " + obj.key() + "?" + queryStringBuilder.toString();

        return url;
    }

    /**
     * Gets the query parameters from a KeyVal object.
     *
     * @param obj the object to get params from
     * @return the query parameters as a map of (key, value) pairs
     */
    private static Map<String, List<Connection.KeyVal>> getQueryParams(KeyVal obj) {
        Map<String, List<Connection.KeyVal>> queryParams = new HashMap<>();
        for (Connection.KeyVal keyVal : obj.data()) {
            queryParams.computeIfAbsent(keyVal.key(), k -> new ArrayList<>()).add(keyVal);
        }
        return queryParams;
    }

    /**
     * Serializes a Collection of KeyVal objects into a URL string.
     *
     * @param objects the collection to serialize
     * @return the serialized URL string
     */
    public static String serialize(Collection<KeyVal> objects) {
        StringBuilder queryStringBuilder = new StringBuilder();
        for (KeyVal keyVal : objects) {
            queryStringBuilder.append(keyVal.key()).append("=").append(String.join("&", keyVal.value()))
                .append("\r\n");
        }

        return "GET / HTTP/1.1\r\n" + queryStringBuilder.toString() + "\r\n";
    }
}
