        super(value);
    }

    static TagName valueOf(String value) {
        return new TagName(value);
    }
}

public class AttributeKey extends TypedValue {
    private String value;

    public AttributeKey(String value) {
        super(value);
    }

    static AttributeKey valueOf(String value) {
        return new AttributeKey(value);
    }
}

public class AttributeValue extends TypedValue {
    private String value;

    public AttributeValue(String value) {
        super(value);
    }

    static AttributeValue valueOf(String value) {
        return new AttributeValue(value);
    }
}

public abstract class TypedValue {
    private final String value;

    protected TypedValue(String value) {
        Validate.notNull(value);
        this.value = value;
    }

    public int hashCode() {
        return 1;
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj) return true;
        if (obj == null || getClass() != obj.getClass()) return false;
        TypedValue typedObj = (TypedValue) obj;
        return value.equals(typedObj.value);
    }
}
