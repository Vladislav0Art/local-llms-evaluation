public class TestOnlineStockSpan {
    public void testNextTest() throws Exception {
        OnlineStockSpan test = new OnlineStockSpan();
        assert test.next(10) == 1;
        assert test.next(7) == 2;
        assert test.next(5) == 3;
        assert test.next(6) == 4;
        assert test.next(9) == 5;
    }

    public void testCalculateSpansTest() throws Exception {
        OnlineStockSpan test = new OnlineStockSpan();
        int[] prices1 = {7, 6, 5, 4, 3, 2, 1};
        int[] expectedSpans1 = {1, 2, 3, 4, 5, 6, 7};

        int[] prices2 = {10, 20, 30, 40, 50, 60};
        int[] expectedSpans2 = {1, 2, 3, 4, 5, 6};

        assert test.calculateSpans(prices1).equals(expectedSpans1);
        assert test.calculateSpans(prices2).equals(expectedSpans2);

        prices1 = new int[]{10, 20};
        expectedSpans1 = new int[] {1, 2};
        test.next(10);
        test.next(20);
        assert test.calculateSpans(prices1).equals(expectedSpans1);
    }

    public void testNextWithEmptyPriceTest() throws Exception {
        OnlineStockSpan test = new OnlineStockSpan();
        assert test.next(0) == 1;
    }
}
