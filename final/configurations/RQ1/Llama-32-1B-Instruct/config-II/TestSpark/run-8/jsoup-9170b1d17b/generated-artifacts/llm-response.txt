class AbstractMap<K, V> extends Map<K, V> {
    private final String dataPrefix;

    public AbstractMap(String dataPrefix) {
        this.dataPrefix = dataPrefix;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        AbstractMap<?, ?> map = (AbstractMap<?, ?>) o;
        return Objects.equals(dataPrefix, map.dataPrefix);
    }

    @Override
    public int hashCode() {
        return Objects.hash(dataPrefix);
    }
}

class Attrs {
    private final String[] keys;
    private final String[] vals;

    public Attrs(String[] keys, String[] vals) {
        this.keys = keys;
        this.vals = vals;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        Attrs attributes = (Attrs) o;
        int keyDiff = 0;
        int valDiff = 0;
        for (int i = 0; i < keys.length && i < attributes.keys.length; i++) {
            if (!keys[i].equals(attributes.keys[i])) keyDiff++;
        }
        for (int i = 0; i < values.length && i < attributes.vals.length; i++) {
            if (!vals[i].equals(attributes.vals[i])) valDiff++;
        }
        return keyDiff == 0 && valDiff == 0;
    }

    @Override
    public int hashCode() {
        int result = 1;
        for (int i = 0; i < keys.length && i < values.length; i++) {
            result = 31 * result + Integer.valueOf(keys[i]).hashCode();
            result = 31 * result + Integer.valueOf(values[i]).hashCode();
        }
        return result;
    }

        }
    }
