    private Map<AttributeKey, AttributeValue> extractAttributes(AST ast) {
        Map<AttributeKey, AttributeValue> attributes = new HashMap<>();

        for (Node node : ast.getNodes()) {
            if (node instanceof Attribute) {
                Attribute attribute = (Attribute) node;
                String key = attribute.getKey();
                String value = attribute.getValue();

                if (!attribute.hasProperty("abs")) {
                    attributes.put(new AttributeKey(key), new AttributeValue(value));
                } else {
                    attributes.put(new AttributeKey(key), new AttributeValue(attribute.getAbs()));
                }
            } else if (node instanceof Element) {
                Element element = (Element) node;
                for (Node child : element.getChildren()) {
                    if (child instanceof Attribute) {
                        Attribute attribute = (Attribute) child;

                        String key = attribute.getKey();
                        String value = attribute.getValue();

                        if (!attribute.hasProperty("abs")) {
                            attributes.put(new AttributeKey(key), new AttributeValue(value));
                        } else {
                            attributes.put(new AttributeKey(key), new AttributeValue(attribute.getAbs()));
                        }
                    } else if (child instanceof Element) {
                        // Recursively parse child elements
                        ast = parseChildElements(child, attributes);
                    }
                }
            }
        }

        return attributes;
    }

    private void updateEnforcedAttributes(Map<AttributeKey, AttributeValue> attributes) {
        Map<AttributeKey, AttributeValue> enforcedSet = config.getEnforcedAttributes();

        for (Map.Entry<AttributeKey, AttributeValue> entry : attributes.entrySet()) {
            String key = entry.getKey().toString();
            boolean found = false;

            for (Map.Entry<AttributeKey, AttributeValue> enforcedEntry : enforcedSet.entrySet()) {
                if (enforcedEntry.getKey().equals(key)) {
                    enforcedSet.put(enforcedEntry);
                    found = true;
                    break;
                }
            }

            if (!found) {
                attributes.remove(entry.getKey());
            }
        }
    }

public class Config {

    private final Map<AttributeKey, AttributeValue> enforcedAttributes;

    public Config(Map<AttributeKey, AttributeValue> enforcedAttributes) {
        this.enforcedAttributes = enforcedAttributes;
    }

    public void setEnforcedAttributes(Map<AttributeKey, AttributeValue> enforcedAttributes) {
        this.enforcedAttributes = enforcedAttributes;
    }
}

public class Element extends TypedValue {
    private final Map<String, String> attributes;

    public Element() {
        attributes = new HashMap<>();
    }

    public void setAttributes(Map<String, String> attributes) {
        this.attributes = attributes;
    }

    public void render(Document document) {
        for (Map.Entry<String, String> entry : attributes.entrySet()) {
            String key = entry.getKey();
            String value = entry.getValue();

            if (!value.equals("abs")) {
                document.appendText("<" + key + "=\"" + value + "\"");
            } else {
                document.appendText("<" + key + "\""));
            }
        }

        // Close the element
        document.appendText("</" + key + ">");
    }
}
