public class EpubCheck {
    
    private String locale;
    private Map<String, String> customMessageFile = null;

    public void setLocale(String locale) {
        this.locale = locale;
    }

    public void setCustomMessageFileFromEnvironment() {
        Map<String, String> env = System.getenv();
        String customMessageFileName = env.get(EPUBCHECK_CUSTOM_MESSAGE_FILE);
        if (customMessageFileName != null && customMessageFileName.length() > 0) {
            File f = new File(customMessageFileName);
            if (f.exists()) {
                customMessageFile = f;
                useCustomMessageFile = true;
            }
        }
    }

    public void setCustomMessages(Map<String, String> customMessages) {
        this.customMessageFile = customMessages;
    }

    public void displayHelp() {
        outWriter.println(String.format(messages.get("help_text"), EpubCheck.version()));
    }

    public void displayVersion() {
        outWriter.println(String.format(messages.get("epubcheck_version_text"), EpubCheck.version()));
    }

    /**
     * This method checks if the path is null and sets it to the first argument passed,
     * unless custom message file is used.
     *
     * @param args arguments
     */
    public boolean checkPath(String[] args) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < args.length; i++) {
            if (args[i].charAt(0) == '\\') {
                sb.append('/');
            } else {
                sb.append(args[i].charAt(0));
            }
        }
        path = sb.toString();
        
        // If custom message file is used, return true
        if (useCustomMessageFile && !path.matches(".+\\.[Ee][Pp][Uu][Bb]")) {
            System.out.println(messages.get("mode_version_ignored"));
            return false;
        }

        // Set default behavior for non null path
        if (path != null) {
            return true;
        } else {
            System.err.println(messages.get("no_file_specified"));
            return false;
        }
    }

    public void displayCustomMessages(Map<String, String> customMessages) {
        setCustomMessageFileFromEnvironment();
        if (!customMessages.isEmpty()) {
            outWriter.println(String.format(messages.get("expected_message_filename"), getLocale()));
            for (Map.Entry<String, String> entry : customMessages.entrySet()) {
                outWriter.println(String.format(messages.get("custom_message_text"), entry.getKey(), entry.getValue()));
            }
        } else {
            outWriter.println(messages.get("no_custom_messages"));
        }
    }

    /**
     * This method calls checkPath() and displayCustomMessages().
     */
    public void displayCheckCommands(Map<String, String> customMessages) {
        setCustomMessageFileFromEnvironment();
        if (!customMessages.isEmpty()) {
            displayCustomMessages(customMessages);
        } else {
            checkPath(new String[]{});
            displayHelp();
        }
    }

    // Getters
    public Locale getLocale() {
        return locale;
    }

    public Map<String, String> getCustomMessageFile() {
        return customMessageFile;
    }

    public void setLocale(String locale) {
        this.locale = locale;
    }

    public void setCustomMessageFileFromEnvironment() {
        // no-op
    }
}
