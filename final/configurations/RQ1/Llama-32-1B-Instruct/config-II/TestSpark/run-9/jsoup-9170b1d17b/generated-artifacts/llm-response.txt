public class Main {

        return keys.length;

        return Arrays.asList(keys);

    public String getValue(int index) {
        if (index >= 1 && index <= getKeysSize()) {
            return vals[index - 1];
        } else {
            return null;
        }
    }

    // Lowercase method
    private static String lowerCase(String s) {
        for (int i = 0; i < s.length(); i++) {
            s = s.toLowerCase();
        }
        return s;
    }

    public void remove(int index) {
        if (index >= 1 && index <= getKeysSize()) {
            delete(index);
        } else {
            throw new IndexOutOfBoundsException("Index out of range");
        }
    }

    // Helper method for removing elements from the map
    private static void delete(int index) {
        keys[index] = null;
        vals[index] = null;
    }

    public boolean isEmpty() {
        return getKeysSize() == 0;
    }

    public String dataPrefix() {
        return dataPrefix;
    }

    // Helper method to count the number of duplicates
    private static int deduplicate(ParseSettings settings) {
        if (settings.preserveAttributeCase()) {
            return 0;
        }
        Set<String> duplicateKeys = new HashSet<>();
        for (int i = 1; i < getKeysSize(); i++) {
            String key = keys[i];
            if (!duplicateKeys.add(lowerCase(key))) {
                // If a duplicate is found, remove the first occurrence
                remove(0);
            }
        }
        return duplicateKeys.size();
    }

    public static class ParseSettings {
        private boolean preserveAttributeCase;

        public ParseSettings(boolean preserveAttributeCase) {
            this.preserveAttributeCase = preserveAttributeCase;
        }

        // Getter and Setter methods for preserveAttributeCase
    }

    public static class Attribute {
        private String key;
        private String value;

        public Attribute(String key, String value) {
            this.key = lowerCase(key);
            this.value = value;
        }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (o == null || getClass() != o.getClass()) return false;
            Attribute that = (Attribute) o;
            return key.equals(that.key) && Objects.equals(value, that.value);
        }

        @Override
        public int hashCode() {
            return Objects.hash(key, value);
        }
    }

            s = s.toLowerCase();
        }
        return s.startsWith(dataPrefix);
    }
