import java.util.*;

class TestUtils {
    public static Map<String, String> generateData(Map<String, Object> data) {
        return new HashMap<>();
    }

    public static List<Map.Entry<String, Integer>> generateData(List<Object> data) {
        List<Map.Entry<String, Integer>> entries = new ArrayList<>();
        for (Object obj : data) {
            if (obj instanceof Map && ((Map<?, ?>) obj).size() > 0) {
                Map<String, Object> map = (Map<String, Object>) obj;
                for (String key : map.keySet()) {
                    entries.add(new AbstractMap.SimpleEntry<>(key, map.get(key)));
                }
            } else if (obj instanceof List && ((List<?, ?>) obj).size() > 0) {
                List<Object> list = (List<?>) obj;
                for (Object o : list) {
                    if (o instanceof Map) {
                        Map<String, Object> map = (Map<String, Object>) o;
                        for (String key : map.keySet()) {
                            entries.add(new AbstractMap.SimpleEntry<>(key, map.get(key)));
                        }
                    } else if (o instanceof Integer) {
                        entries.add(new AbstractMap.SimpleEntry<>((Integer) o, 0));
                    } else {
                        throw new RuntimeException("Unsupported type: " + o.getClass());
                    }
                }
            } else {
                throw new RuntimeException("Invalid data format");
            }
        }

        return entries;
    }
}

public class Main {

    public static void main(String[] args) {
        Map<String, Object> data = TestUtils.generateData(Arrays.asList(
                Collections.singletonMap("a", 1),
                Collections.singletonMap("b", "x"),
                Arrays.asList(2, 3)
        ));
        List<Map.Entry<String, Integer>> entries = TestUtils.generateData(List.of(new HashMap<>(), new HashMap<>(), new int[]{0, 1})); // map, list of integers
        System.out.println(entries);
    }

    public static String lowerCase(String s) {
        return s.toLowerCase();
    }
}
