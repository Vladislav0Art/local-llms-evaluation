import org.w3c.dom.Document;
import org.w3c.dom.Element;
import org.w3c.dom.NodeList;

import javax.xml.parsers.DocumentBuilder;
import javax.xml.parsers.DocumentBuilderFactory;
import java.io.ByteArrayInputStream;
import java.util.HashMap;
import java.util.Map;

public class DocumentGenerator {
    public static void main(String[] args) throws Exception {
        // Define the root element
        Document doc = DocumentBuilderFactory.newInstance().newDocumentBuilder()
                .newDocument();
        Element root = doc.createElement("html");
        root.setAttribute("xmlns", "http://www.w3.org/1999/xhtml");
        doc.appendChild(root);

        // Add head and body to the document
        Element head = doc.createElement("head");
        root.appendChild(head);
        head.appendChild(doc.importNode(doc.getDocumentElement(), true));

        Element body = doc.createElement("body");
        root.appendChild(body);
        body.appendChild(doc.importNode(doc.getDocumentElement(), true));

        // Define elements
        Element paragraph1 = doc.createElement("p");
        doc.appendChild(paragraph1);

        Element h2Title = doc.createElement("h2");
        h2Title.setAttribute("id", "title");
        h2Title.setTextContent("Hello World!");
        body.appendChild(h2Title);

        Element titleElement = doc.createElement("title");
        paragraph1.appendChild(titleElement);
        titleElement.setTextContent("Hello World!");

        // Write to file
        ByteArrayInputStream bis = new ByteArrayInputStream(doc.toxml().getBytes());
        org.w3c.dom.Document document = DocumentBuilderFactory.newInstance().newDocumentBuilder().parse(bis);
        System.out.println(document.getDocumentElement().getNodeName() + "\n");
    }
}
