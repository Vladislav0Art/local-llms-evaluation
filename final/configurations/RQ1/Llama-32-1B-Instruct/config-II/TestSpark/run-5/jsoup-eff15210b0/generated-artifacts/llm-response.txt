/**
 * Response class.
 *
 * This class is responsible for encoding the request URL into a format that can be sent in an HTTP response.
 */
public class Response {

    /**
     * The response body.
     */
    private String responseBody;

    public Response() {
        Validate.notNullParam(responseBody, "responseBody");
    }

    /**
     * Sets the response body.
     *
     * @param responseBody The response body to be set.
     */
    public void setResponseBody(String responseBody) {
        this.responseBody = responseBody;
    }

    /**
     * Returns the response body as a string.
     *
     * @return The response body as a string.
     */
    public String getResponseBody() {
        return responseBody;
    }
}

/**
 * Response data class.
 *
 * This class is used to hold the data that will be sent in an HTTP response.
 */
public class responseData {

    /**
     * The key-value pair.
     */
    private KeyVal keyVal;

    public responseData() {
        Validate.notNullParam(keyVal, "keyVal");
    }

    /**
     * Sets the key-value pair.
     *
     * @param keyVal The key-value pair to be set.
     */
    public void setKeyVal(KeyVal keyVal) {
        this.keyVal = keyVal;
    }

    /**
     * Returns the key-value pair as a string.
     *
     * @return The key-value pair as a string.
     */
    public String getKeyVal() {
        return keyVal.toString();
    }
}

/**
 * URL builder class.
 *
 * This class is used to build the request URL by appending the query parameters and key-value pairs to it.
 */
public class UrlBuilder {

    /**
     * The base URL.
     */
    private String baseUrl;

    public UrlBuilder(String baseUrl) {
        Validate.notNullParam(baseUrl, "baseUrl");
        this.baseUrl = baseUrl;
    }

    /**
     * Appends a query parameter.
     *
     * @param key The key for the query parameter.
     * @param value The value for the query parameter.
     */
    public void appendQueryParameter(String key, String value) {
        Validate.notNullParam(key);
        Validate.notNullParam(value);
        this.baseUrl += "?" + "key=" + URLEncoder.encode(key, "UTF-8") + "&value=" + URLEncoder.encode(value, "UTF-8");
    }

    /**
     * Appends a key-value pair.
     *
     * @param key The key for the key-value pair.
     * @param value The value for the key-value pair.
     */
    public void appendKeyVal(String key, String value) {
        Validate.notNullParam(key);
        this.baseUrl += "/" + encodeMimeName(key) + "=" + URLEncoder.encode(value, "UTF-8");
    }

    /**
     * Returns the URL with query parameters and key-value pairs appended to it.
     *
     * @return The URL with query parameters and key-value pairs appended to it.
     */
    public String build() {
        return baseUrl;
    }
}

/**
 * Mime name utility class.
 *
 * This class is used to encode a string into a mime name for use in the request URL.
 */
public class MimeNameUtil {

    /**
     * Encodes a string into a mime name.
     *
     * @param str The string to be encoded.
     * @return The mime name of the string.
     */
    public static String encodeMimeName(String str) {
        return "text/" + str;
    }

    /**
     * Checks if a string contains an input stream.
     *
     * @param str The string to be checked.
     * @return True if the string contains an input stream, false otherwise.
     */
    public static Validate.isFalse(str.contains("inputstream"), "InputStream data not supported in URL query string.");

    /**
     * Returns a key-value pair as a string.
     *
     * @param key The key for the key-value pair.
     * @return The key-value pair as a string.
     */
    public static String getKeyVal(String key) {
        return key + "=";
    }
}

/**
 * Key value interface class.
 *
 * This interface is used to represent a key-value pair.
 */
public interface KeyValue {

    /**
     * Sets the key for the key-value pair.
     *
     * @param key The new key for the key-value pair.
     * @return this object
     */
    void setKey(String key);

    /**
     * Returns the key as a string.
     *
     * @return The key as a string.
     */
    String getKey();

    /**
     * Sets the value for the key-value pair.
     *
     * @param value The new value for the key-value pair.
     * @return this object
     */
    void setValue(String value);

    /**
     * Returns the value as a string.
     *
     * @return The value as a string.
     */
    String getValue();
}

public class Main {

    public static void main(String[] args) {
        Response response = new Response();

        // Set the response body
        response.setResponseBody("Hello World!");

        // Append query parameters to the URL
        UrlBuilder urlBuilder = new UrlBuilder("/users");
        urlBuilder.appendQueryParameter("name", "John Doe");
        String url = urlBuilder.build();
        System.out.println(url);  // Output: /users?name=John+Doe

        // Set the key-value pair as a string
        KeyVal keyVal = new KeyVal();
        keyVal.setValue("username", "john_doe");
        response.setKeyVal(keyVal);

        // Get the query parameters as a string
        String queryString = MimeNameUtil.getKeyVal("name");
        System.out.println(queryString);  // Output: username

    }
}
