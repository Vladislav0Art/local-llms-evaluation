import org.junit.Test;
import static org.junit.Assert.*;

public class TextNodeTest {

    @Test
    public void testBlankNodes() {
        TextNode node = new TextNode("   ");
        assertTrue(node.isBlank());
    }

    @Test
    public void testSplitText_NoOffset() {
        TextNode node = new TextNode("Hello, World!");
        TextNode result = node.splitText(0);
        assertEquals(node, result);
    }

    @Test
    public void testSplitText_EmptyString() {
        TextNode node = new TextNode("");
        TextNode result = node.splitText(0);
        assertTrue(result.isBlank());
    }

    @Test
    public void testSplitText_OkayOffset() {
        TextNode node = new TextNode("Hello, World!");
        TextNode result = node.splitText(10);
        assertEquals(node, result);
    }

    @Test
    public void testSplitText_OkayOffset_Narrower() {
        TextNode node = new TextNode("Hello, World!");
        TextNode result = node.splitText(5);
        assertTrue(result.isBlank());
    }

    @Test
    public void testIsBlank_TextOnly() {
        TextNode node = new TextNode("   ");
        assertEquals(false, node.isBlank());
    }

    @Test
    public void testGetWholeText_EqualToOriginal() {
        TextNode node = new TextNode("Hello, World!");
        String expected = "Hello, World!";
        String actual = node.getText();
        assertEquals(expected, actual);
    }

    @Test
    public void testGetWholeText_WrapInQuotes() {
        TextNode node = new TextNode("\"Hello, World!\"");
        String expected = "\"Hello, World!\"
";
        String actual = node.getText();
        assertEquals(expected, actual);
    }

    @Test
    public void testText_SplitAtOffset_WithWhitespace() {
        TextNode node = new TextNode("Hello   ,  World!");
        TextNode result = node.splitText(6);
        assertEquals(node, result);
    }

    @Test
    public void testText_SplitAtOffset_Okay() {
        TextNode node = new TextNode("Hello  ,   World!");
        TextNode result = node.splitText(4);
        assertEquals(node, result);
    }
}
