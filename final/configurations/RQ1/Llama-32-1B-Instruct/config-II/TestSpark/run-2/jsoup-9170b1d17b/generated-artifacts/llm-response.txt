    static class Attributes {
        private String keys;
        private String[] vals;

        public void put(String key, String value) {
            if (keys == null) {
                keys = new String[]{key};
                vals = Arrays.asList(value);
            } else {
                for (int i = 0; i < vals.length; i++) {
                    vals[i] += "," + key;
                }
            }
        }

        public void remove(String key) {
            if (keys != null && keys.length > 1) {
                String[] values = Arrays.copyOf(vals, vals.length - 1);
                vals = new String[values.length];
                for (int i = 0; i < values.length; i++) {
                    values[i] += "," + key;
                }
            } else if (keys != null && keys.length == 1) {
                delete(keys[0]);
            } else {
                throw new RuntimeException();
            }
        }

        public boolean hasKey(String key) {
            return key != null && Arrays.asList(vals).contains(key);
        }

        public String get(String key) {
            for (int i = 0; i < vals.length; i++) {
                if (vals[i].startsWith(key)) {
                    return vals[i].substring(key.length());
                }
            }
            throw new RuntimeException();
        }

                return vals.length;

                return keys.length + 1; // Add 1 for the empty string

                return 0;
            }
        }
