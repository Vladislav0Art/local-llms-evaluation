public class EpubCheck {

    private static final String HELP_TEXT = "EpubCheck command-line tool.\n";
    private static final String VERSION_TEXT = "Version: ";
    private static final String EPUB_CHECK_VERSION_TEXT = "EpubCheck version: ";

    public static void main(String[] args) {
        boolean result;
        for (String arg : args) {
            if (!arg.startsWith("-")) {
                System.err.println(arg);
                return false;
            }
            switch (arg) {
                case "--help":
                    displayHelp();
                    break;
                case "--version":
                    System.out.println(VERSION_TEXT + EpubCheck.EPUB_CHECK_VERSION_TEXT + EpubCheck.EpubCheck.version());
                    break;
                default:
                    result = runEpubCheck(arg);
            }
        }
    }

    private static boolean runEpubCheck(String arg) {
        if (arg.equals("--help")) {
            return true;
        } else if (arg.startsWith("--version")) {
            System.out.println(VERSION_TEXT + EpubCheck.EPUB_CHECK_VERSION_TEXT + EpubCheck.EpubCheck.version());
            return false;
        } else if (arg.contains("path=")) {
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < arg.length(); i++) {
                if (arg.charAt(i) == '\\') {
                    sb.append('/');
                } else {
                    sb.append(arg.charAt(i));
                }
            }
            return true;
        } else {
            return false;
        }
    }

    private void displayHelp() {
        System.out.println(HELP_TEXT + EpubCheck.EpubCheck.version());
    }
}
