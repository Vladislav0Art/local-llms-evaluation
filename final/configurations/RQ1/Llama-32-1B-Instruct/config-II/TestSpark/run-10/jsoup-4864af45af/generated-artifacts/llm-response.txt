import java.util.HashMap;
import java.util.Map;

public class ProtocolTest {

    public static void main(String[] args) {
        // Test cases
        testValidProtocol("rel:stylesheet", new Element(), " stylesheet");

        testValidProtocol("#anchor", new Element(), "#anchor");
        testValidProtocol("http://example.com/path", new Element(), "path");

        testInvalidProtocol("invalid:protocol", new Element(), "");
    }

    public static void testValidProtocol(String value, Object el, String attr) {
        TestProtocol test = new TestProtocol();
        if (!test.testValidProtocol(el, attr, value)) {
            fail("Failed to test valid protocol");
        }
    }

    public static void testInvalidProtocol(String value, Object el, String attr) {
        TestProtocol test = new TestProtocol();
        if (test.testValidProtocol(el, attr, value)) {
            fail("Expected invalid protocol, got valid one");
        }
    }

    private static class Element extends TypedValue {
        @Override
        public int hashCode() {
            final int prime = 31;
            int result = 1;
            result = prime * result + ((value == null) ? 0 : value.hashCode());
            return result;
        }

        @Override
        public boolean equals(Object obj) {
            if (this == obj) return true;
            if (obj == null) return false;
            if (getClass() != obj.getClass()) return false;
            Element other = (Element) obj;
            if (value == null) {
                return other.value == null;
            } else {
                return value.equals(other.value);
            }
        }

        @Override
        public String toString() {
            return value;
        }
    }

    private static class TestProtocol extends TypedValue {

        public static boolean testValidProtocol(Element el, Attribute attr, String value) {
            if (preserveRelativeLinks) {
                // Check relative URL in the attribute's value
                try {
                    new Element().absUrl(attr.getKey()).equals(value);
                } catch (Exception e) {
                    fail("Expected abs URL for relative link");
                }
            }

            return !attr.getValue().isEmpty() && value.startsWith("#") && !value.matches(".*\\s.*");
        }

    }

}
