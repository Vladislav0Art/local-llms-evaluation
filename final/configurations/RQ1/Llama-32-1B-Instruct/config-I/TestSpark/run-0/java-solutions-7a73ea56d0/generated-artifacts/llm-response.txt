    public int calculateSpans(int[] prices) {
        int result = 0;
        for (int i = 1; i < prices.length; i++) {
            if (prices[i] > prices[i-1]) {
                result += (i - 1);
            }
            prices[i] = Math.max(prices[i], result + prices[i]);
