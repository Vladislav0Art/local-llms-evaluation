    /**
     * Test if this text node is blank -- that is, empty or only whitespace (including newlines).
     *
     * @param document Document
     * @return true if this document is empty or only whitespace, false if it contains any text content.
     */
    public static boolean isBlank(Document document) {
        return document != null && !document.getText().isEmpty();
    }

    /**
     * Test if the given node is a blank node -- that is, an element with no children and no parent.
     *
     * @param document Document
     * @return true if the given node is a blank node, false otherwise.
     */
    public static boolean isBlank(Node document) {
        return document == null || document.getChildren().isEmpty() && !document.getParent() instanceof Node;
    }

    /**
     * Test if this text node has more than one sibling.
     *
     * @param document Document
     * @return true if the given node has at least two siblings, false otherwise.
     */
    public static boolean hasMoreSiblings(Document document) {
        return document != null && !document.getChildren().isEmpty() && (document.getChildren().size() > 1 || document.getParent() instanceof Node);
    }

    /**
     * Test if the given text node is blank.
     *
     * @param node The text node to check.
     * @return true if the node is empty or only whitespace, false otherwise.
     */
    public static boolean isEmptyNode(Node node) {
        return node.getTextContent().isEmpty();
    }

    /**
     * Test if a document has any blank nodes.
     *
     * @param document The document to check for blank nodes.
     * @return true if the document contains any blank nodes, false otherwise.
     */
    public static boolean hasBlankNodes(Document document) {
        NodeList nodeList = document.getChildNodes();
        for (int i = 0; i < nodeList.getLength(); i++) {
            Node node = (Node) nodeList.item(i);
            if (node.getNodeType() == Node.ELEMENT_NODE && !isEmptyNode(node)) {
                return true;
            }
        }
        return false;
    }

    /**

     * @param document The document to check for blank nodes.

        NodeList nodeList = document.getChildNodes();
        for (int i = 0; i < nodeList.getLength(); i++) {
