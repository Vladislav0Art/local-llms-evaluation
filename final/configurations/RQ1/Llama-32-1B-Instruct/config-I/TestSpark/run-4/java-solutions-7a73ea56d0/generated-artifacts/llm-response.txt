public class Solution {
    public int[] movingStacks(int[][] piles, int k) {
        int n = piles.length;
        int[] ans = new int[n];

        for (int i = 0; i < n; i++) {
            while (piles[i].length > 1 && k > 0) {
                ans[i] += pileLength(piles[i]) * piles[i][k];
                pile(k, -piles[i][k]);
                k -= 1;
            }
        }

        return ans;
    }

    private int pile(int[] arr, int val) {
        while (arr.length > 0 && val == arr[arr.length - 1]) {
            arr.pop();
        }
        if (val != -1) {
            arr.push(val);
        } else {
            arr.clear();
        }
        return arr.length;
    }

    private int pileLength(int[] arr) {
        for (int i = 0; i < arr.length; i++) {
            if (arr[i] > 0) {
                return i + 1;
            }
        }
        return -1;
    }
}

        int[] ans = new int[n];

            arr[i] = piles[i].length;

                ans[index]++;
                pile(piles, -1);
                k -= 1;

                    arr[i] = 0;

import java.util.*;

public class Solution {
    public int[] movingStacks(int[][] piles, int k) {
        Deque<Integer> ans = new ArrayDeque<>();
        
        // Initialize array to store final stacks
        int[] arr = new int[piles.length];
        for (int i = 0; i < piles.length; i++) {
            arr[i] = piles[i].length;
        }

        while (k > 0) {
            Deque<Integer> curStacks = new ArrayDeque<>();
            Deque<Integer> nextStacks = new ArrayDeque<>();
            
            // Push all stacks with one position to go into the queue
            for (int i = 0; i < arr.length; i++) {
                if (arr[i] > 1) {
                    curStacks.addLast(arr[i]);
                }
            }

            while (!curStacks.isEmpty()) {
                int len = curStacks.poll();
                
                // Move the stack to the next position
                for (int i = 0; i < arr.length; i++) {
                    if (i + len <= arr.length) {
                        nextStacks.add(i, arr[i] - 1);
                    } else {
                        break;
                    }
                }

                int index = 0;

                // Move the stack to the correct position
                while (index < piles.length && piles[index].length > 1) {
                    ans.set(index, ans.get(index) + pile(piles[index], -1));
                    nextStacks.addLast(-piles[index][k]);
                    k -= 1;
                }

                for (int i = index; i < piles.length; i++) {
                    if (i >= arr.length || piles[i].length != 1) {
                        break;
                    }
                    
                    int val = -piles[i][k];
                    nextStacks.add(i, val);
                    k -= 1;

                }

                // Update the array of stacks
                for (int i : nextStacks) {
                    if (!curStacks.contains(arr[i])) {
                        arr[i] = 0;
                    } else {
                        arr[i] += piles[i].length * -1;
                    }
                }
            }

            k -= curStacks.size();
        }

        return ans.toArray(new int[0]);
    }

    private int pile(int[] arr, int val) {
        while (arr.length > 0 && val == arr[arr.length - 1]) {
            arr.pop();
        }
        if (val != -1) {
            arr.push(val);
        } else {
            arr.clear();

        return arr.length;

            if (arr[i] > 0) {
                return i + 1;
            }
        }
        return -1;
    }
}

import java.util.*;

public class Solution {
    public int[] movingStacks(int[][] piles, int k) {
        Deque<Integer> ans = new ArrayDeque<>();
        
        // Initialize array to store final stacks
        Deque<Integer>[] queue = new LinkedList[piles.length];
        for (int i = 0; i < piles.length; i++) {
            queue[i] = new LinkedList<>();
            queue[i].add(piles[i].length);
        }

        while (k > 0) {
            Deque<Integer> curStacks = new ArrayDeque<>();
            Deque<Integer> nextStacks = new ArrayDeque<>();
            
            // Push all stacks with one position to go into the queue
            for (int i : queue) {
                if (i >= piles[i].length && k > 0) {
                    int len = i - piles[i].length;
                    curStacks.addLast(piles[i][k]);
                    nextStacks.addLast(-piles[i][k]);
                    k -= len;
                }
            }

            while (!curStacks.isEmpty()) {
                int len = curStacks.poll();
                
                // Move the stack to the next position
                for (int i : queue) {
                    if (i > piles[i].length && k > 0) {
                        ans.add(i - len);
                        queue[i - len].add(piles[i][k]);
                        break;
                    }
                }

                int index = 0;

                // Move the stack to the correct position
                while (index < queue.length && queue[index].size() > piles[index].length) {
                    nextStacks.addLast(-queue[index].pollFirst());
                    k -= 1;
                }

                for (int i : queue) {
                    if (i < piles[i].length) {
                        ans.add(i + 1);
                    }
                }
            }

            k -= curStacks.size();
        }

        return ans.toArray(new int[0]);
    }

    private int pile(int[] arr, int val) {
        while (arr.length > 0 && val == arr[arr.length - 1]) {
            arr.pop();
        }
        if (val != -1) {
            arr.push(val);
        } else {
            arr.clear();
        }
        return arr.length;
    }

    private int pileLength(int[] arr) {
        for (int i = 0; i < arr.length; i++) {
            if (arr[i] > 0) {
                return i + 1;
            }
        }
        return -1;
    }
}
