import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.assertTrue;

public class AttributeUtilTest {

    @Test
    void testIsValidAnchor() {
        String[] validStrings = {"#anchor", "test#anchor"};
        for (String s : validStrings) {
            assertTrue(AttributeUtil.isValidAnchor(s));
        }
    }

    @Test
    void testInvalidAnchor() {
        String[] invalidStrings = {"#anchor", "", "#anchor#", "#anchor"},
                                "test#anchor", "test#anchor2", "test/anchor1", "test/anchor2"};
        for (String s : invalidStrings) {
            assertFalse(AttributeUtil.isValidAnchor(s));
        }
    }

    @Test
    void testIsValidUrl() {
        String[] validUrls = {"http://example.com", "https://www.example.com/path"};
        for (String url : validUrls) {
            assertTrue AttributeUtil.isValidUrl(url));
        }
    }

    @Test
    void testIsValidEmail() {
        String[] validEmails = {"test@example.com", "valid_email@invalid_domain.com"};
        for (String email : validEmails) {
            assertTrue(AttributeUtil.isValidEmail(email));
        }
    }

    public static class AttributeUtil {
        public boolean isValidUrl(String url) {
            try {
                java.net.URI uri = new java.net.URI(url);
                return uri.getScheme() != null && uri.getHost() != null;
            } catch (Exception e) {
                throw new AssertionError(e);
            }
        }

        public boolean isValidEmail(String email) {
            String[] emailParts = email.split("@");
            if (emailParts.length < 2 || emailParts[1].isEmpty()) {
                return false;
            }
            try {
                java.net.URI uri = new java.net.URI(emailParts[0] + "." + emailParts[1]);
                return true;
            } catch (Exception e) {
                throw new AssertionError(e);
            }
        }

        public boolean isNumber(String str) {
            for (char c : str.toCharArray()) {
                if (!Character.isDigit(c)) {
                    return false;
                }
            }
            return true;
        }

        public boolean isValidUrlPart(String part) {
            String[] validParts = {"http", "https"};
            for (String s : validParts) {
                if (!part.equals(s)) {
                    return false;
                }
            }
            return true;
        }

    }

}

import org.junit.jupiter.api.Test;

public class AttributeUtilTest {

    @Test
    void testIsValidAnchor() {
        String[] validStrings = {"#anchor", "test#anchor"};
        for (String s : validStrings) {
            assertTrue(AttributeUtil.isValidAnchor(s));
        }
    }

    @Test
    void testInvalidAnchor() {
        String[] invalidStrings = {"#", "anchor", "#", "#"};
        for (String s : invalidStrings) {
            assertFalse(AttributeUtil.isValidAnchor(s));
        }
    }

    @Test
    public void testIsValidUrl() {
        AttributeUtil attributeUtil = new AttributeUtil();
        assertTrue(attributeUtil.isValidUrl("http://example.com"));
        assertTrue(attributeUtil.isValidUrl("https://www.example.com/path"));
    }

    @Test
    public void testIsValidEmail() {
        AttributeUtil attributeUtil = new AttributeUtil();
        assertTrue(attributeUtil.isValidEmail("test@example.com"));
        assertTrue(attributeUtil.isValidEmail("valid_email@invalid_domain.com"));
    }

    @Test
    public void testInvalidUrl() {
        AttributeUtil attributeUtil = new AttributeUtil();
        assertFalse(AttributeUtil.isValidUrl("#anchor"));
        assertFalse(AttributeUtil.isValidUrl("http://example.com/path"));
        assertFalse(AttributeUtil.isValidUrl("https://www.example.com/"));
    }

    @Test
    public void testIsValidEmailPart() {
        AttributeUtil attributeUtil = new AttributeUtil();
        assertTrue(attributeUtil.isNumber("#anchor"));
        assertTrue(attributeUtil.isNumber("test#anchor"));
        assertTrue(attributeUtil.isNumber("#anchor#"));
        assertTrue(attributeUtil.isNumber("#anchor"));
        assertFalse(attributeUtil.isNumber(""));
    }

    @Test
    public void testIsValidEmailPartNotNumber() {
        AttributeUtil attributeUtil = new AttributeUtil();
        assertFalse(attributeUtil.isValidEmailPart("#anchor"));
        assertFalse(attributeUtil.isValidEmailPart("test#anchor"));
        assertFalse(attributeUtil.isValidEmailPart("#anchor#"));
        assertFalse(attributeUtil.isValidEmailPart("#anchor"));
    }

    @Test
    public void testIsValidEmail() {
        AttributeUtil attributeUtil = new AttributeUtil();
