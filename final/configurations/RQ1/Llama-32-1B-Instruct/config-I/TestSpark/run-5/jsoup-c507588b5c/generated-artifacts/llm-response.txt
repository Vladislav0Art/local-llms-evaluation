public class TextNodeTests {

    @Test
    public void testBlankText() {
        TextNode node = new TextNode("   \n\t\r\n");
        assertTrue(node.isBlank());
    }

    @Test
    public void testSplitText() {
        String text = "Hello World!";
        TextNode node1 = new TextNode(text.substring(0, 4));
        TextNode node2 = new TextNode(text.substring(5));
        TextNode node3 = new TextNode(" ");
        assert node1.toString().equals(node3.toString());
    }

    @Test
    public void testSplitTextOffSetZero() {
        String text = "Hello World!";
        TextNode node = new TextNode(text);
        TextNode tail = node.splitText(0);
        assertNotNull(tail);
    }

    @Test
    public void testSplitTextNonZeroOffset() {
        String text = "Hello World!";
        TextNode node = new TextNode(text);
        TextNode tail = node.splitText(5);
        assertNotNull(tail);
    }

    @Test
    public void testBlankAfterText() {
        String text = "Hello World!";
        TextNode node = new TextNode("   \n\t\r\n");
        assertTrue(node.isBlank());
    }

    @Test
    public void testRemoveNewlines() {
        String text = "Hello World!\n";
        TextNode node = new TextNode(text);
        assertEquals("", node.text(""));
    }

    @Test
    public void testRemoveSpacesBetweenTags() {
        String text = "<p>Hello <span>World!</span></p>";
        TextNode node = new TextNode(text);
        assertEquals("<p>Hello World!</p>", node.toString());
    }

    @Test
    public void testStripLeadingAndTrailingWhitespace() {
        String text = "   \n\t\r\n";
        TextNode node = new TextNode(text);
        assertEquals("Hello World!", node.text(""));
    }

    @Test
    public void testEntitiesEscape() throws Exception {
        String text = "&amp;lt;p&amp;gt;Hello &amp;lt;span&gt;World!&amp;lt;/span&gt;";
        TextNode node = new TextNode(text);
        assertEquals("&amp;lt;p&amp;gt;Hello &amp;lt;span&gt;World!&amp;lt;/span&gt;", node.toString());
    }

}
