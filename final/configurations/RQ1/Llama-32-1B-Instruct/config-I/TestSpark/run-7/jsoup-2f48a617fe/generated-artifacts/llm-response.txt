import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.parser.ParseSettings;

public class CommentTest {

    public static void main(String[] args) {
        Document doc = new Document();
        ParseSettings parseSettings = new ParseSettings(doc, new Parser());
        parseSettings.setMode(ParserMode.TEXT);

        String originalComment = "!xml declaration content";
        Element comment = doc.createElement("comment");
        comment.appendChild(doc.createTextNode(originalComment));
        doc.body().appendChild(comment);

        Comment parsedComment = new Comment(originalComment);
        Assert.assertTrue(parsedComment.isXmlDeclaration());

        expectedContent = "<!xml declaration content>";
        Comment expectedContentComment = new Comment(expectedContent);
        Element expectedContentElement = doc.createElement("comment");
        expectedContentElement.appendChild(doc.createTextNode(expectedContent));
        doc.body().appendChild(expectedContentElement);

        Assert.assertEquals(expectedContent, parsedContent.getData());

        parsedComment = (Comment) originalComment.clone();
        assertNotSame(originalComment, parsedComment);

        comment = new Comment("!xml declaration content");
        parsedComment = (Comment) comment.asXmlDeclaration();

        expectedContent = "<!xml declaration content>";
        Element expectedContentElement = doc.createElement("comment");
        expectedContentElement.appendChild(doc.createTextNode(expectedContent));
        doc.body().appendChild(expectedContentElement);

        Assert.assertEquals(expectedContent, parsedContent.getData());

        comment = new Comment("!original comment content");
        parsedComment = (Comment) comment.clone();

        expectedContent = "<!xml declaration content>";
        Element expectedContentElement = doc.createElement("comment");
        expectedContentElement.appendChild(doc.createTextNode(expectedContent));
        doc.body().appendChild(expectedContentElement);

        Assert.assertEquals(originalComment, parsedContent.getData());
    }

    public static void testIsXmlDeclaration() {
        String originalContent = "!xml declaration content";
        Comment comment = new Comment(originalContent);
        Assert.assertTrue(comment.isXmlDeclaration());
    }

    public static void testAsXmlDeclaration() {
        String expectedContent = "<!xml declaration content>";
        Comment originalComment = new Comment("#original comment content");
        Comment parsedComment = (Comment) originalComment.asXmlDeclaration();
        Assert.assertEquals(expectedContent, parsedComment.getData());
    }

    public static void testClone() {
        Comment originalComment = new Comment("!original comment content");
        Comment clonedComment = originalComment.clone();

        String expectedContent = "!xml declaration content";
        Element expectedCommentElement = doc.createElement("comment");
        expectedCommentElement.appendChild(doc.createTextNode(expectedContent));
        doc.body().appendChild(expectedCommentElement);

        AssertNotSame(originalComment, clonedComment);
    }
}
