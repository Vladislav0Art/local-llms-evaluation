    public void setup() {
        onlineStockSpan = new OnlineStockSpan();
        list = new ArrayList<>();
    }

    public void testNext_1() {
        int price = 10;
        int[] result = onlineStockSpan.next(price);
        assertEquals(1, result); // Span of first element is always 1
    }

    @Test
    public void testNext_Sorted_1() {
        int price = 0;
        int[] result = onlineStockSpan.next(price);
        assertEquals(1, result); // Span of first element is always 1
    }

    @Test
    public void testNext_NaN() {
        double price = Double.NaN;
        onlineStockSpan.next(price);
        assertOnlineStockSpanIsNaN(list, price);
    }

    @Test
    public void testNext_MultipleValues_10() {
        int[] prices = { 9, 8, 7, 6 };
        int[] result = onlineStockSpan.calculateSpans(prices);
        assertOnlineStockSpanIsCorrect(result, prices);
    }

    private boolean assertOnlineStockSpanIsNaN(List<Integer> list, double price) {
        if (price != Double.NaN && !list.contains(price)) {
            return true;
        }
        for (int i = 0; i < list.size(); i++) {
            if (!Math.abs(list.get(i) - price) <= 1e-6 && Math.abs(list.get(i) - price) > 0.00001) {
                return false;
            }
        }
        return true;
    }

    private boolean assertOnlineStockSpanIsCorrect(int[] result, int[] prices) {
        for (int i = 0; i < prices.length; i++) {
            if (prices[i] != result[i]) {
                // Do nothing
            } else if (result[i] == 1) {
                continue;
            } else {
                assertOnlineStockSpanIsNaN(list, prices[i]);
            }
        }
        return true;
    }
}

public static class List {
    public static <T> T[] newArrayList() {
        return (T[]) new Object[0];
    }

    public static void add(T[] array, T element) {
        if (array.length == 0) {
            throw new RuntimeException("ArrayList is empty");
        }
        int index = 0;
        while (index < array.length && array[index] != null) {
            index++;
        }
        if (index >= array.length || array[index] != null) {
            array = java.util.Arrays.copyOf(array, index + 1);
        }
        array[index] = element;
    }

    public static void remove(T[] array, T element) {
        int index = -1;
        for (int i = 0; i < array.length; i++) {
            if (array[i] == null) continue;
            if (array[i].equals(element)) {
                index = i;
                break;
            }
        }

        if (index != -1) {
            System.arraycopy(array, 0, array, 0, index);
            System.arraycopy(array, index + 1, array, index, array.length - index - 1);
        }
    }

    public static <T> T[] newArrayList(T element) {
        return new ArrayList<T>().toArray(new T[arraySize(element)]);
    }

    public static int arraySize(T element) {
        if (element == null) {
            throw new NullPointerException("List contains a null element");
        } else {
            return 1;
        }
    }
}

public class OnlineStockSpan {

    private List stockPrices;

    public OnlineStockSpan() {
        this.stockPrices = new ArrayList();
    }

    public void addPrice(int price) {
        stockPrices.add(stockPrices.arraySize(price));
    }

    public int getLatestPrice(int[] prices, int index) {
        return stockPrices.get(index);
    }
}

public class Main {

    public static void main(String[] args) {
        OnlineStockSpan onlineStockSpan = new OnlineStockSpan();
        for (int i = 0; i < 10; i++) {
            onlineStockSpan.addPrice(i);
        }

        System.out.println("Latest price at index 0: " + onlineStockSpan.getLatestPrice(new int[]{1, 2, 3}, 0));
    }
}
