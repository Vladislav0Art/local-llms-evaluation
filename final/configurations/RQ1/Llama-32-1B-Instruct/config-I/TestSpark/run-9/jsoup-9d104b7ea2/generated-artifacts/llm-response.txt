import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.nodes.TextNode;

public class TextNodeTest {

    public static void testIsBlank() {
        TextNode textNode = new TextNode("test");
        assertEquals(false, textNode.isBlank());
    }

    public static String splitText(TextNode textNode, int start) {
        Document doc = new Document();
        appendAll(textNode, doc);
        Element head = doc.getElementsByTagName("head").item(0);
        if (head == null || !head.hasText()) {
            return "";
        }
        head.appendText("\n");
        String[] lines = splitString(head.text());
        for (int i = 1; i < lines.length; i++) {
            lines[i] += "\n";
        }
        TextNode result = new TextNode(lines[0]);
        appendAll(textNode, result);
        return result.toString();
    }

    public static String outerHtmlHead(Appendable accum, int start, OutputSettings settings) {
        Document doc = new Document();
        appendAll(doc, accum);
        Element head = doc.getElementsByTagName("head").item(0);
        if (head == null || !head.hasText()) {
            return "";
        }
        head.appendText("\n");
        String[] lines = splitString(head.text());
        for (int i = 1; i < lines.length; i++) {
            lines[i] += "\n";
        }
        TextNode result = new TextNode(lines[0]);
        appendAll(doc, result);
        return result.toString();
    }

    public static void clone(TextNode textNode) {
        TextNode clonedTextNode = new TextNode(textNode.text());
        if (textNode.isBlank()) {
            clonedTextNode.isBlank = true;
        } else {
            clonedTextNode.strip();
        }
        appendAll(clonedTextNode, textNode);
    }

    private static String[] splitString(String str) {
        return str.split("\n");
    }

    private static void appendAll(Appendable appendable, TextNode node) {
        if (node instanceof Appendable) {
            ((Appendable) node).append(appendable);
        } else {
            appendable.append(node.toString());
        }
    }
}

public class TextNodeTest {

        Document doc = new Document();
        Elements elements = doc.getAllElements();
        for (Element element : elements) {
            if (!element.text().isEmpty()) {
                break;
            }
        }
        assertTrue(textNodeIsBlank(doc, "Hello World!"));
    }

    public static String splitText(TextNode textNode) {
        String html = textNode.strip().outerHtmlHead("", 0);
        return Jsoup.parse(html).text();
    }

    public static String outerHtmlHead(Appendable accum, int start, OutputSettings settings) {
        Document doc = new Document();
        appendAll(doc, accum);
        Elements elements = doc.getAllElements();
        for (int i = 1; i < elements.length(); i++) {
            if (elements.get(i).tagName().equals("head")) {
                break;
            }
        }
        String html = Jsoup.parse(elements.toString()).outerHtmlHead("", start, settings);
        return html;
    }

    public static void clone(TextNode textNode) {
        TextNode clonedTextNode = new TextNode(textNode.text());
        if (textNode.isBlank()) {
            clonedTextNode.isBlank = true;
        } else {
            clonedTextNode.strip();
        }
        appendAll(clonedTextNode, textNode);
    }

    private static boolean textNodeIsBlank(Document doc, String text) {
        Elements elements = doc.getAllElements();
        for (Element element : elements) {
            if (!element.text().isEmpty()) {
                return false;
            }
        }
        return true;
    }
