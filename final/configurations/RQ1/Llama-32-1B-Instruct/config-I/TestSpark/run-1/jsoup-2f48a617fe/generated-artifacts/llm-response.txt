import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class CommentTest {

    @Test
    public void testCommentNode() {
        String html = "<html><body><div id=\"test-div\"><span id=\"test-comment1\"></span></div></body></html>";
        Document doc = Jsoup.parse(html);
        
        Element element = doc.getElementById("test-div");
        Element span1 = doc.getElementById("test-comment1");
        
        System.out.println(span1.text());
    }

    @Test
    public void testCommentText() {
        String html = "<html><body><div id=\"test-div\"><span id=\"test-comment1\"></span></div></body></html>";
        Document doc = Jsoup.parse(html);
        
        Element element = doc.getElementById("test-div");
        System.out.println(element.text());
    }

    @Test
    public void testCommentAppend() {
        String html = "<html><body><div id=\"test-div\"><span id=\"test-comment1\"></span></div></body></html>";
        Document doc = Jsoup.parse(html);
        
        Element element = doc.getElementById("test-div");
        Element span1 = doc.getElementById("test-comment1");
        
        // Append new comment
        span1.appendChild(doc.selectFirst("[id=\"test-comment2\"]").first());
        
        System.out.println(span1.text());
    }

    @Test
    public void testCommentRemove() {
        String html = "<html><body><div id=\"test-div\"><span id=\"test-comment1\"></span></div></body></html>";
        Document doc = Jsoup.parse(html);
        
        Element element = doc.getElementById("test-div");
        Element span1 = doc.getElementById("test-comment1");
        
        // Remove original comment
        span1.remove();
        
        System.out.println(span1.text());
    }

    @Test
    public void testCommentTextRemove() {
