import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Main {
    private Map<String, String> enforcedAttributes = new HashMap<>();

    public void addAttribute(String attributeKey, String value) {
        enforcedAttributes.put(attributeKey, value);
    }

    public boolean testValidProtocol(Element element, String tag, Protocol protocol) {
        if (!enforcedAttributes.containsKey("tag")) return false;
        Document doc = Jsoup.parse(element.outerHtml());
        Element myTag = doc.select(tag).first();
        if (myTag == null || !testValidProtocol(myTag, "example", protocol)) return false;
        for (Attribute attribute : enforcedAttributes.values()) {
            String value = attribute.getValue();
            Element element = doc.select(String.valueOf(attribute.getKey())).first();
            if (!element.hasAttr(attribute.getKey()) || !element.attr(attribute.getKey()).equals(value)) return false;
        }
        return true;
    }

        addAttribute("tag", "example");

        Protocol protocol = new Protocol("https://example.com");
        boolean isValidProtocol = testValidProtocol(Element.el, "myTag", protocol);
        System.out.println(isValidProtocol);  // Output: true
    }

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Main {
    private Map<String, String> enforcedAttributes = new HashMap<>();

    public void addAttribute(String attributeKey, String value) {
        enforcedAttributes.put(attributeKey, value);
    }

    public boolean testValidProtocol(Element element, String tag, Protocol protocol) {
        if (!enforcedAttributes.containsKey("tag")) return false;
        Document doc = Jsoup.parse(element.outerHtml());
        Element elementTag = doc.select(String.valueOf(tag)).first();
        if (elementTag == null || !elementTag.tagName().equals(tag)) return false;

        Attribute attribute = new Attribute(tag, tag);
        for (Attribute attr : enforcedAttributes.values()) {
            if (!attr.getKey().equals(attribute.key) || !attr.getValue().equals(attribute.value)) return false;
        }

        elementTag.select("input[type=hidden][name=" + protocol.url + "]").remove();
        for (String hiddenInputValue : elementTag.attr("hidden").split(";")) {
            String[] parts = hiddenInputValue.split("=");
            if (parts.length != 2) continue;
            attribute.key = parts[0];
            attribute.value = parts[1].trim();
        }

        return true;
    }

    public static class Attribute {
        private String key;
        private String value;

        public Attribute(String key, String value) {
            this.key = key;
            this.value = value;
        }

        public String getKey() {
            return key;
        }

        public String getValue() {
            return value;
        }
    }

    public static class ElementTag {
        public Document document;

        public ElementTag(Document document) {
            this.document = document;
        }
    }

    public static class Protocol {
        private String url;
    }
}

class Test {
    public void testValidProtocol() {
        Main main = new Main();
        // create some test elements
        ElementTag element1 = new ElementTag(main.document);
        main.addAttribute("test1", "value1");
        main.addAttribute("test2", "value2");

        ElementTag element2 = new ElementTag(main.document);
        main.addAttribute("test3", "value3");

        Protocol protocol = new Main.Protocol();
        boolean isValidProtocol = main.testValidProtocol(element1, "input[type=hidden][name=test1]", protocol);
        System.out.println(isValidProtocol);

        // add some hidden input values
        element2.attr("input[type=hidden][name=test1]").value = "new_value";
    }
}

class Main {
    public static void main(String[] args) {
        new Test().testValidProtocol();
    }
}
