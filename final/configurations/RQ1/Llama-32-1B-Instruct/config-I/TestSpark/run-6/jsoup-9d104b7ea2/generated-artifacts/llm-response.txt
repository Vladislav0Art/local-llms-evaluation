    }
}

class TextNode extends org.jsoup.nodes.Text {

    public TextNode(String text) {
        super(text);
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;

        TextNode that = (TextNode) o;

        return !java.util.Objects.equals(getText(), that.getText());
    }
}

class NodeFilter extends org.jsoup.nodes.NodeFilter {

    public NodeFilter(Node node) {
        super(node);
    }

    @Override
    public boolean accept(Node n) {
        if (!(n instanceof TextNode)) {
            return false;
        }
        return true;
    }

    @Override
    public void accept(Node node, Document ownerDocument) {
        if (!node.text().equals("Hello World!")) {
            System.out.println(node.text());
        }
    }
}

public class Main {

    public static String filterByText(Node node, Document doc) {
        NodeFilter filter = new NodeFilter(node);
        if (!filter.accept(node)) return "";
        String text = node.text();
        for (Node child : node.children()) {
            if (!filter.accept(child, doc)) {
                System.out.println(text + " - " + child.toString());
                break;
            }
        }
        return text;
    }

    public static void main(String[] args) {
        String input1 = "Hello World!";
        Document doc1 = new Document();
        Elements tags1 = doc1.getElementsByTag("p");
        for (Element tag : tags1) {
            System.out.println(tag.text());
        }

        Element parent1 = new Element("body").appendChild(new TextNode(doc1.createTextNode(input1)));
        System.out.println(filterByText(parent1, doc1));

        String input2 = "No text here!";
        Document doc2 = new Document();
        Elements tags2 = doc2.getElementsByTag("p");
        for (Element tag : tags2) {
            System.out.println(tag.text());
        }

        Element parent2 = new Element("body").appendChild(new TextNode(doc2.createTextNode(input2)));
        System.out.println(filterByText(parent2, doc2));
    }
}

class TagFilter extends org.jsoup.nodes.NodeFilter {

    public TagFilter(Node node) {
        super(node);
    }

    @Override
    public boolean accept(Node n) {
        if (!(n instanceof TextNode)) return false;
        return true;
    }

    @Override
    public void accept(Node node, Document ownerDocument) {
        String text = node.text();
        for (Node child : node.children()) {
            System.out.println(text + " - " + child.toString());
        }
    }
}
