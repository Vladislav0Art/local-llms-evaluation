            this.content = "";
        }

                content = str;
            } else {
                content += " <" + tag + ">" + str + "</" + tag + ">";
            }
        }

        @Override
        public String toString() {
            return "<" + tag + ">" + content + "</" + tag + ">";
        }
    }

    public static void main(String[] args) {

        Element rootElement = new Element("root");
        xml.addXmlElement("root", rootElement);
        rootElement.appendContent("<tag>");

        Element childElement = new Element("child");
        xml.addXmlElement("child", childElement);
        childElement.appendContent("<content>Text</content>");

import java.io.File;
import java.io.FileNotFoundException;
import java.util.Scanner;

public class XmlFileParser {
    public static void main(String[] args) throws FileNotFoundException {
        File file = new File("example.xml");
        String xmlString = readXmlFromFile(file);
        parseXml(xmlString);
    }

    private static String readXmlFromFile(File file) throws FileNotFoundException {
        Scanner scanner = new Scanner(file);
        StringBuilder xmlString = new StringBuilder();
        while (scanner.hasNextLine()) {
            xmlString.append(scanner.nextLine());
        }
        scanner.close();
        return xmlString.toString();
    }

    private static void parseXml(String xmlString) {
        System.out.println("Parsing XML...");
        XmlFileParser xmlParser = new XmlFileParser();
        xmlParser.parse(xmlString);
        System.out.println("XML parsing complete.");
    }

    private static class XmlFileParser {
        private Map<String, String> elements = new HashMap<>();

        public void parse(String xmlString) throws Exception {
            // Remove XML declaration
            int pos = xmlString.indexOf("<!");
            if (pos == -1) {
                throw new Exception("XML file does not contain an XML declaration");
            }
            xmlString = xmlString.substring(0, pos);

            // Parse elements
            String[] lines = xmlString.split("\n");
            for (String line : lines) {
                if (line.trim().startsWith("<")) {
                    String tagName = line.trim();
                    String content = "";
                    while (!line.isEmpty() && !line.startsWith("/")) {
                        if (tagName.equals("tag")) {
                            break;
                        }
                        content += line + " ";
                        line = xmlString.substring(xmlString.indexOf(line) + 1, xmlString.indexOf(">", line));
                    }

                    // Create element
                    String[] keyValue = content.split("=");
                    XmlElement element = new XmlElement(keyValue[0].trim());
                    if (keyValue.length > 1) {
                        element.appendContent(keyValue[1].trim());
                    }

                    elements.put(tagName, element.toString());
                }
            }
        }
    }

    private static class XmlElement {
        private String tag;

        public XmlElement(String tag) {
            this.tag = tag;
        }

        public void appendContent(String content) {
            if (!content.isEmpty()) {
                System.out.println(content);
            } else {
                System.out.print("<" + tag + ">");
            }
        }

        @Override
        public String toString() {
            return "<" + tag + ">" + content + "</" + tag + ">";
        }
    }
}

In this solution, the `XmlFileParser` class reads an XML file from a file and parses its elements. It removes any XML declaration that may be present at the beginning of the file.
