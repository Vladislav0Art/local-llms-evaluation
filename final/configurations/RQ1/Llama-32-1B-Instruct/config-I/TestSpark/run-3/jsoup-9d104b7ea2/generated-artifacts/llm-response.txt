        assertEquals("<a><b>text</b></a>", node.text());
    }

    @Test
    public void splitTextWithOffsetAndTrimTrailing() {

        TextNode node = new TextNode(splitText(6));
        assertEquals("<a><b>text</b></a>", node.text());
    }

    @Test
    public void LastCharIsWhitespace() {
        assertTrue(lastCharIsWhitespace());
    }

    private boolean isBlank() {
        return inputText.replaceAll("\\s+", "").isEmpty();
    }

    private boolean lastCharIsWhitespace() {
        int count = 0;
        for (int i = 0; i < inputText.length(); i++) {
            if (Character.isWhitespace(inputText.charAt(i))) {
                count++;
            }
        }
        return count == 0;
    }
}

public class TextNode {

    private String text;

    public TextNode() {
        this.text = "";
    }

    public void append(String element) {
        if (text.isEmpty()) {
            this.text = element;
        } else {
            this.text += element;
        }
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof TextNode)) return false;
        TextNode other = (TextNode) o;
        return text.equals(other.text);
    }

    @Override
    public int hashCode() {
        return text.hashCode();
    }

    public String getText() {
        return this.text;
    }
}

public class Main {

    public static void main(String[] args) {
        TextNode test1 = new TextNode();
        test1.append("Hello World");
        System.out.println(test1.getText());

        TestSparkConfig config = new TestSparkConfig();

        }

        TextNode test2 = new TextNode();
