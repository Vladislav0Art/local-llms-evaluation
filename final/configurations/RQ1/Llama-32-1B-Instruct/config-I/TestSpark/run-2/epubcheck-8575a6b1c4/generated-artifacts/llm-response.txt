            File file = new File(path);
            if (!file.exists()) {
                return false; // non-existent file
            }
            if (!file.isFile()) {
                return false; // not a file
            }

            try (BufferedReader br = new BufferedReader(new FileReader(file))) {
                String line;
                while ((line = br.readLine()) != null) {
                    line = line.trim();
                    if (!line.startsWith("#") && !line.isEmpty() && !line.contains("u2:")) {
                        return false; // no u2: markers
                    }
                }
            } catch (Exception e) {
                System.out.println(e.getMessage());
                return false;
            }

            return true;
        }

        public boolean useCustomMessageFile() {
            String customMessageFileName = EpubCheck.getCustomMessageFileName();
            File file = new File(customMessageFileName);
            if (file.exists()) {
                return true; // u2: markers in custom message file
            }
            return false; // no custom message file found
        }

        public boolean getCustomMessageFileName() {
            String customMessageFileName = EpubCheck.getCustomMessageFileName();
