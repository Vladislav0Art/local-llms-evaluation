```java
import org.enforce.attributes.Attribute;
import org.enforce.attributes.Name;
import org.enforce.attributes.ClassName;

public class HtmlUtils {

    public static void enforceAttributes(String html) {
        parse(html);
    }

    private static void parse(String html) {
        String[] lines = html.split("\n");
        for (String line : lines) {
            if (line.startsWith("<")) {
                parseTag(line);
            } else if (line.startsWith(">")) {
                parseEndTag();
            }
        }
    }

    private static void parseTag(String tag) {
        int start = tag.indexOf(">");
        String tagName = tag.substring(1, start).trim();

        if (!isValidTagName(tagName)) {
            System.out.println("Invalid tag name: " + tagName);
            return;
        }

        for (Attribute attr : getAttributes()) {
            if (attr.getName().equals(tagName)) {
                addEnforceableAttr(attr.getValue());
            }
        }
    }

    private static void parseEndTag() {
        int start = findNext("<", html);
        String tagName = html.substring(start + 1, start + 5).trim();

        if (!isValidTagName(tagName)) {
            System.out.println("Invalid tag name: " + tagName);
            return;
        }

        for (Attribute attr : getAttributes()) {
            if (attr.getName().equals(tagName)) {
                removeEnforceableAttr(attr.getValue());
            }
        }
    }

    private static String findNext(String tag, String html) {
        int index = html.indexOf(tag);
        while (index != -1 && !tag.equals(html.substring(index + tag.length()))) {
            index = html.indexOf(tag, index + 1);
        }
        return index;
    }

    private static void addEnforceableAttr(String value) {
        if (!ENFORCEABLE_ATTRS.contains(value)) {
            ENFORCEABLE_ATTRS.add(value);
            System.out.println("Added enforceable attribute: " + value);
        }
    }

    private static void removeEnforceableAttr(String value) {
        if (ENFORCEABLE_ATTRS.contains(value)) {
            ENFORCEABLE_ATTRS.remove(value);
            System.out.println("Removed enforceable attribute: " + value);
        }
    }

    private static List<Attribute> getAttributes() {
        return new ArrayList<>(Arrays.asList(
                new Attribute(Name.of("class"), new ClassName()),
                new Attribute(Name.of("style"), new ClassName())
        ));
    }

    private static boolean isValidTagName(String tagName) {
        return !tagName.isEmpty();
    }
}
```

In this implementation, the `enforceAttributes` method uses a recursive approach to parse HTML. It splits the input string into lines and iterates over each line.

If a line starts with `<`, it's a tag opening. The `parseTag` method is called for each tag, checking if it's valid and adding any enforceable attributes found in the tag name.

The same logic applies to closing tags, where the `parseEndTag` method finds the next `<` tag, checks its name for enforceability, and removes any enforceable attributes.

Finally, the `getAttributes` method returns a list of all enforceable attributes in the HTML.
