public class Main {

    public static void main(String[] args) {
        String[] command = {"-help"};
        boolean result = Main.main(command);
        System.out.println(result);

        command = new String[]{"-version", "1.0"};
        result = Main.main(command);
        System.out.println(result);

        command = new String[]{"-version", "-listChecks"};
        result = Main.main(command);
        System.out.println(result);

        command = new String[]{"-help", "--verbose"};
        result = Main.main(command);
        System.out.println(result);
    }

    public static boolean main(String[] args) {
        EpubCheck epubCheck = new EpubCheck();
        return epubCheck.main(args);
    }
}

class EpubCheck {

    public boolean main(String[] command) {
        try {
            Process process = Runtime.getRuntime().exec(command);
            int exitValue = process.waitFor();

            if (exitValue == 0) {
                System.out.println("epubcheck: " + command[0] + " -help");
                return true;
            } else {
                throw new RuntimeException();
            }
        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());
            return false;
        }
    }

    public static class CustomMessage {
        public String message = "This is a custom message.";
    }
}

            Process process = Runtime.getRuntime().exec(command);
            IOUtils.write(process.getInputStream().readAllBytes(), System.out, 1 << 26);
