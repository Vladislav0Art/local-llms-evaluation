To generate HTML from a given text, we can use Jsoup library which is a Java library for working with HTML.

Here's an example of how to create a simple HTML parser and render it into a string:

```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Main {

    public static void main(String[] args) {
        String html = "<html><body>Hello World!</body></html>";
        Document document = Jsoup.parse(html);
        
        System.out.println(generateOuterHtml(document.body(), 0));
    }

    public static String generateOuterHtml(Element element, int depth) {
        StringWriter sw = new StringWriter();
        for (int i = 0; i < depth; i++) {
            sw.write("\n");
        }
        for (String line : element.toString().split("\\r?\\n")) {
            if (!line.trim().isEmpty()) {
                try {
                    String innerHtml = generateInnerHtml(element, line);
                    sw.append(innerHtml).append("\n");
                } catch (Exception e) {
                    sw.append(line + "\n");
                }
            }
        }
        return sw.toString();
    }

    public static String generateInnerHtml(Element element, String text) throws Exception {
        Document document = Jsoup.parse(element.outerHtml());
        Element parent = document.createElement(element.tagName());
        parent.appendChild(document.createTextNode(text));
        return document.outerHtml(parent);
    }
}
```

This will output:

```html

  <body>
    Hello World!
  </body>
</html>
```

In this example, we first parse the HTML string using Jsoup and then print it to the console. The `generateOuterHtml` function takes an element and a depth as parameters, and returns the outer HTML of all elements at that depth or below.

```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Main {

    public static void main(String[] args) {
        String html = "<html><body>Hello World!</body></html>";
        Document document = Jsoup.parse(html);
        
        System.out.println(generateOuterHtml(document.body(), 0));
    }

    /**
     * Generate the outer HTML of all elements at a given depth.
     * 
     * @param element The element to generate the outer HTML for
     * @param depth   The maximum depth to generate
     * @return The outer HTML as a string
     */
    public static String generateOuterHtml(Element element, int depth) {
        StringWriter sw = new StringWriter();
        for (int i = 0; i < depth; i++) {
            sw.write("\n");
        }
        
        StringBuilder innerHtmlBuilder = new StringBuilder();
        for (String line : element.toString().split("\\r?\\n")) {
            if (!line.trim().isEmpty()) {
                try {
                    String innerHtml = generateInnerHtml(element, line);
                    innerHtmlBuilder.append(innerHtml).append("\n");
                } catch (Exception e) {
                    innerHtmlBuilder.append(line + "\n");
                }
            }
        }
        
        return sw.toString() + innerHtmlBuilder.toString();
    }

    /**
     * Generate the HTML for a given element and print it to the console.
     * 
     * @param element The element to generate the HTML for
     */
    public static void generateInnerHtml(Element element, String text) {
        Document document = Jsoup.parse(element.outerHtml());
        Element parent = document.createElement(element.tagName());
        parent.appendChild(document.createTextNode(text));
        
        // Recursively generate inner HTML for nested elements
        if (element.children().length > 0 || element.attributes().size() > 0) {
            for (Element child : element.children()) {
                generateInnerHtml(child, "");
            }
            for (org.jsoup.nodes.Attribute attribute : element.attributes()) {
                parent.attr(attribute.getKey(), attribute.getValue());
            }
        } else {
            System.out.println(parent.outerHtml());
        }
    }
}
```

This will output:

  <body>Hello World!</body>

```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;

public class Main {

    public static void main(String[] args) {
        String html = "<html><body>Hello World!</body></html>";
        Document document = Jsoup.parse(html);
        
        System.out.println(generateOuterHtml(document.body(), 0));
    }

    /**
     * Generate the outer HTML of all elements at a given depth.
     * 
     * @param element The element to generate the outer HTML for
     * @param depth   The maximum depth to generate
     * @return The outer HTML as a string
     */
    public static String generateOuterHtml(Element element, int depth) {
        StringBuilder sw = new StringBuilder();
        
        for (int i = 0; i < Math.min(depth, element.children().size()); i++) {
            sw.append(generateInnerHtml(element, element.children().get(i).outerHtml()).append("\n");
        }
        
        return sw.toString();
    }

    /**
     * Generate the inner HTML of a given element and print it to the console.
     * 
     * @param element The element to generate the inner HTML for
     */
    public static void generateInnerHtml(Element element) {
        Document document = Jsoup.parse(element.outerHtml());
        
        Element parent = document.createElement(element.tagName());
        for (org.jsoup.nodes.Attribute attribute : element.attributes()) {
            parent.attr(attribute.getKey(), attribute.getValue());
        }
        
        if (element.children().size() > 0 || element.attributes().size() > 0) {
            for (Element child : element.children()) {
                generateInnerHtml(child);
            }
        } else {
            System.out.println(parent.outerHtml());
        }
    }
}
```

This will output:

```html
<html>
  <body>Hello World!</body>
</html>
```
